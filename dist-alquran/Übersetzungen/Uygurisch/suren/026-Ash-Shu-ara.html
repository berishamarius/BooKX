بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
طسٓمٓ ﴿١﴾
طا، سىم، مىم
تِلْكَ ءَايَـٰتُ ٱلْكِتَـٰبِ ٱلْمُبِينِ ﴿٢﴾
بۇ، روشەن مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) ئايەتلىرىدۇر
لَعَلَّكَ بَـٰخِعٌ نَّفْسَكَ أَلَّا يَكُونُوا۟ مُؤْمِنِينَ ﴿٣﴾
(ئى مۇھەممەد!) ئۇلارنىڭ ئىمان ئېيتمىغانلىقىدىن ئۆزۈڭنى ھالاك قىلىۋېتىشىڭ مۇمكىن
إِن نَّشَأْ نُنَزِّلْ عَلَيْهِم مِّنَ ٱلسَّمَآءِ ءَايَةً فَظَلَّتْ أَعْنَـٰقُهُمْ لَهَا خَـٰضِعِينَ ﴿٤﴾
ئەگەر بىز ئۇلارنىڭ (ئىمان ئېيتىشىنى) خالايدىغان بولساق، ئۇلارغا ئاسماندىن (ئۇلارنى ئىمانغا مەجبۇرلايدىغان) بىر ئايەتنى (ئالامەتنى) نازىل قىلاتتۇقتە، ئۇنىڭغا ئۇلار باش ئەگكەن بولاتتى
وَمَا يَأْتِيهِم مِّن ذِكْرٍ مِّنَ ٱلرَّحْمَـٰنِ مُحْدَثٍ إِلَّا كَانُوا۟ عَنْهُ مُعْرِضِينَ ﴿٥﴾
ئۇلارغا مەرھەمەتلىك اﷲ تەرىپىدىن قۇرئاندىن يېڭىدىن بىرنەرسە نازىل بولسىلا، ئۇلار ئۇنىڭدىن يۈز ئۆرۈيدۇ
فَقَدْ كَذَّبُوا۟ فَسَيَأْتِيهِمْ أَنۢبَـٰٓؤُا۟ مَا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿٦﴾
ئۇلار راستلا (قۇرئاننى) ئىنكار قىلدى، مەسخىرە قىلغان نەرسىنىڭ خەۋەرلىرى (يەنى ئاقىۋىتى) ئۇلارغا كېلىدۇ
أَوَلَمْ يَرَوْا۟ إِلَى ٱلْأَرْضِ كَمْ أَنۢبَتْنَا فِيهَا مِن كُلِّ زَوْجٍ كَرِيمٍ ﴿٧﴾
ئۇلار زېمىننى (يەنى زېمىننىڭ ئاجايىباتلىرىنى) كۈزەتمىدىمۇ؟ زېمىندا تۈرلۈك پايدىلىق ئۆسۈملۈكلەرنى ئۆستۈردۇق
إِنَّ فِى ذَٰلِكَ لَـَٔايَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿٨﴾
بۇنىڭدا ئەلۋەتتە (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) ئالامەت بار، ئۇلارنىڭ تولىسى ئىمان ئېيتقۇچى بولمىدى
وَإِنَّ رَبَّكَ لَهُوَ ٱلْعَزِيزُ ٱلرَّحِيمُ ﴿٩﴾
سېنىڭ پەرۋەردىگارىڭ ئىشەنچلىك ھالدا غالىبتۇر، ناھايىتى مېھرىباندۇر
وَإِذْ نَادَىٰ رَبُّكَ مُوسَىٰٓ أَنِ ٱئْتِ ٱلْقَوْمَ ٱلظَّـٰلِمِينَ ﴿١٠﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭ مۇساغا نىدا قىلدى: «سەن زالىم قەۋمگە بارغىن
قَوْمَ فِرْعَوْنَ ۚ أَلَا يَتَّقُونَ ﴿١١﴾
(ئۇلار) پىرئەۋننىڭ قەۋمىدۇر، ئۇلار (اﷲ نىڭ جازالىشىدىن) قورقمامدۇ؟»
قَالَ رَبِّ إِنِّىٓ أَخَافُ أَن يُكَذِّبُونِ ﴿١٢﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! مەن ئىشەنچلىك ھالدا ئۇلارنىڭ مېنى ئىنكار قىلىشىدىن قورقىمەن
وَيَضِيقُ صَدْرِى وَلَا يَنطَلِقُ لِسَانِى فَأَرْسِلْ إِلَىٰ هَـٰرُونَ ﴿١٣﴾
(ئۇلارنىڭ مېنى ئىنكار قىلىشىدىن) يۈرىكىم سىقىلىدۇ، تىلىم كېكەچ، شۇڭا (ماڭا ياردەمدە بولۇش ئۈچۈن) ھارۇننى (خۇدانىڭ ئەلچىسى) قىلىپ ئەۋەتكىن
وَلَهُمْ عَلَىَّ ذَنۢبٌ فَأَخَافُ أَن يَقْتُلُونِ ﴿١٤﴾
(ئۇلارنىڭ دەۋاسىچە) ئۇلارنىڭ ئالدىدا مېنىڭ گۇناھىم بار، ئۇلارنىڭ مېنى ئۆلتۈرۈشىدىن قورقىمەن»
قَالَ كَلَّا ۖ فَٱذْهَبَا بِـَٔايَـٰتِنَآ ۖ إِنَّا مَعَكُم مُّسْتَمِعُونَ ﴿١٥﴾
اﷲ ئېيتتى: «ھەرگىز ئۇنداق ئەمەس (يەنى سېنى ئۇلار ھەرگىز ئۆلتۈرەلمەيدۇ)، سىلەر مېنىڭ مۆجىزىلىرىمنى ئېلىپ بېرىڭلار، بىز ئىشەنچلىك ھالدا سىلەر بىلەن بىللە (مۇنازىرەڭلارنى) ئاڭلاپ تۇرىمىز
فَأْتِيَا فِرْعَوْنَ فَقُولَآ إِنَّا رَسُولُ رَبِّ ٱلْعَـٰلَمِينَ ﴿١٦﴾
سىلەر پىرئەۋنگە بېرىپ: بىز ئىشەنچلىك ھالدا ئالەملەرنىڭ پەرۋەردىگارىنىڭ ئەلچىسىمىز
أَنْ أَرْسِلْ مَعَنَا بَنِىٓ إِسْرَٰٓءِيلَ ﴿١٧﴾
ئىسرائىل ئەۋلادىنى (قۇللۇقۇڭدىن) بوشاتقىن، ئۇلار بىز بىلەن (شامغا) كەتسۇن دەڭلار»
قَالَ أَلَمْ نُرَبِّكَ فِينَا وَلِيدًا وَلَبِثْتَ فِينَا مِنْ عُمُرِكَ سِنِينَ ﴿١٨﴾
(ئۇلار پىرئەۋنگە كېلىپ ئەلچىلىكنى يەتكۈزدى) پىرئەۋن (مۇساغا) ئېيتتى: «سېنى بىز ئۆيىمىزدە كىچىكىڭدىن تەربىيلەپ چوڭ قىلمىدۇقمۇ؟ ئارىمىزدا كۆپ يىللار تۇرمىدىڭمۇ؟
وَفَعَلْتَ فَعْلَتَكَ ٱلَّتِى فَعَلْتَ وَأَنتَ مِنَ ٱلْكَـٰفِرِينَ ﴿١٩﴾
سەن ھېلىقى قىلغان ئىشىڭنى قىلمىدىڭمۇ؟ (يەنى قىبتىنى ئۆلتۈرمىدىڭمۇ؟) سەن تۇزكورلاردىنسەن»
قَالَ فَعَلْتُهَآ إِذًا وَأَنَا۠ مِنَ ٱلضَّآلِّينَ ﴿٢٠﴾
مۇسا ئېيتتى: «مەن ئۇ ئىشنى قىلغان چېغىمدا نادانلاردىن ئىدىم
فَفَرَرْتُ مِنكُمْ لَمَّا خِفْتُكُمْ فَوَهَبَ لِى رَبِّى حُكْمًا وَجَعَلَنِى مِنَ ٱلْمُرْسَلِينَ ﴿٢١﴾
سىلەردىن قورقۇپ، سىلەردىن قاچتىم، پەرۋەردىگارىم ماڭا ھېكمەت ئاتا قىلدى، مېنى خۇدانىڭ ئەلچىسى قىلدى
وَتِلْكَ نِعْمَةٌ تَمُنُّهَا عَلَىَّ أَنْ عَبَّدتَّ بَنِىٓ إِسْرَٰٓءِيلَ ﴿٢٢﴾
سەن ئىسرائىل ئەۋلادىنى قۇل قىلدىڭ، سېنىڭ ماڭا مىننەت قىلغان نېمىتىڭ ئەنە شۇدۇر»
قَالَ فِرْعَوْنُ وَمَا رَبُّ ٱلْعَـٰلَمِينَ ﴿٢٣﴾
پىرئەۋن ئېيتتى: «ئالەملەرنىڭ پەرۋەردىگارى دېگەن نېمە؟»
قَالَ رَبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَآ ۖ إِن كُنتُم مُّوقِنِينَ ﴿٢٤﴾
مۇسا ئېيتتى: «ئۇ ئاسمانلارنىڭ، زېمىننىڭ ۋە ئۇلارنىڭ ئارىسىدىكى پۈتۈن مەخلۇقاتلارنىڭ پەرۋەردىگارىدۇر. ئەگەر سىلەر ھەقىقىي ئىشىنىدىغان بولساڭلار»
قَالَ لِمَنْ حَوْلَهُۥٓ أَلَا تَسْتَمِعُونَ ﴿٢٥﴾
پىرئەۋن چۆرىسىدىكىلەرگە: «(ئۇنىڭ جاۋابىنى) ئاڭلاۋاتامسىلەر؟» دېدى
قَالَ رَبُّكُمْ وَرَبُّ ءَابَآئِكُمُ ٱلْأَوَّلِينَ ﴿٢٦﴾
مۇسا ئيتتى: «(ئۇ) سىلەرنىڭ پەرۋەردىگارىڭلاردۇر ۋە سىلەرنىڭ ئاتا - بوۋاڭلارنىڭ پەرۋەردىگارىدۇر»
قَالَ إِنَّ رَسُولَكُمُ ٱلَّذِىٓ أُرْسِلَ إِلَيْكُمْ لَمَجْنُونٌ ﴿٢٧﴾
پىرئەۋن ئېيتتى: «سىلەرگە ئەۋەتىلگەن (بۇ) ئەلچى ئەلۋەتتە مەجنۇندۇر»
قَالَ رَبُّ ٱلْمَشْرِقِ وَٱلْمَغْرِبِ وَمَا بَيْنَهُمَآ ۖ إِن كُنتُمْ تَعْقِلُونَ ﴿٢٨﴾
مۇسا ئېيتتى: «(ئۇ) مەشرىقنىڭ، مەغرىبنىڭ ۋە ئۇلارنىڭ ئارىسىدىكى مەخلۇقاتنىڭ پەرۋەردىگارىدۇر. ئەگەر سىلەر چۈشىنىدىغان بولساڭلار»
قَالَ لَئِنِ ٱتَّخَذْتَ إِلَـٰهًا غَيْرِى لَأَجْعَلَنَّكَ مِنَ ٱلْمَسْجُونِينَ ﴿٢٩﴾
پىرئەۋن ئېيتتى: «ئەگەر مەندىن غەيرىنى ئىلاھ قىلىۋالىدىغان بولساڭ، سېنى چوقۇم زىندانغا تاشلايمەن»
قَالَ أَوَلَوْ جِئْتُكَ بِشَىْءٍ مُّبِينٍ ﴿٣٠﴾
مۇسا ئېيتتى: «(مېنىڭ راست خۇدانىڭ ئەلچىسى ئىكەنلىكىمنى ئىسپاتلايدىغان) روشەن دەلىل كەلتۈرسەممۇ (زىندانغا تاشلامسەن؟)»
قَالَ فَأْتِ بِهِۦٓ إِن كُنتَ مِنَ ٱلصَّـٰدِقِينَ ﴿٣١﴾
پىرئەۋن ئېيتتى: «ئەگەر سۆزۈڭ راست بولسا روشەن دەلىلىڭنى كەلتۈرگىن»
فَأَلْقَىٰ عَصَاهُ فَإِذَا هِىَ ثُعْبَانٌ مُّبِينٌ ﴿٣٢﴾
مۇسا ھاسىسىنى تاشلىۋىدى، ناگاھان ئوپئوچۇق ئەجدىھاغا ئايلاندى
وَنَزَعَ يَدَهُۥ فَإِذَا هِىَ بَيْضَآءُ لِلنَّـٰظِرِينَ ﴿٣٣﴾
مۇسا (قوينىدىن) قولىنى چىقىرىۋىدى، ناگاھان ئۇ قارىغۇچىلارغا (نۇر چاقناپ تۇرىدىغان) ئاپئاق بولۇپ كۆرۈندى
قَالَ لِلْمَلَإِ حَوْلَهُۥٓ إِنَّ هَـٰذَا لَسَـٰحِرٌ عَلِيمٌ ﴿٣٤﴾
پىرئەۋن چۆرىسىدىكى (قەۋمنىڭ) چوڭلىرىغا ئېيتتى: «بۇ ئىشەنچلىك ھالدا ئۇستا سېھىرگەر ئىكەن، ئۇ سېھىر ئارقىلىق سىلەرنى زېمىنىڭلاردىن ھەيدەپ چىقارماقچى، (بۇ ھەقتە) نېمە مەسلىھەت بېرىسىلەر؟»
يُرِيدُ أَن يُخْرِجَكُم مِّنْ أَرْضِكُم بِسِحْرِهِۦ فَمَاذَا تَأْمُرُونَ ﴿٣٥﴾
پىرئەۋن چۆرىسىدىكى (قەۋمنىڭ) چوڭلىرىغا ئېيتتى: «بۇ ئىشەنچلىك ھالدا ئۇستا سېھىرگەر ئىكەن، ئۇ سېھىر ئارقىلىق سىلەرنى زېمىنىڭلاردىن ھەيدەپ چىقارماقچى، (بۇ ھەقتە) نېمە مەسلىھەت بېرىسىلەر؟»
قَالُوٓا۟ أَرْجِهْ وَأَخَاهُ وَٱبْعَثْ فِى ٱلْمَدَآئِنِ حَـٰشِرِينَ ﴿٣٦﴾
ئۇلار ئېيتتى: «(ئۇلارنىڭ ئىشىنى) تەخىر قىلغىن، شەھەرلەرگە (سېھىرگەرلەرنى) يىققۇچى كىشىلەرنى ئەۋەتكىن
يَأْتُوكَ بِكُلِّ سَحَّارٍ عَلِيمٍ ﴿٣٧﴾
ئۇلار سېنىڭ ھوزۇرۇڭغا ناھايىتى ئۇستا سېھىرگەرلەرنىڭ ھەممىسىنى ئېلىپ كەلسۇن»
فَجُمِعَ ٱلسَّحَرَةُ لِمِيقَـٰتِ يَوْمٍ مَّعْلُومٍ ﴿٣٨﴾
مۇئەييەن كۈندىكى بەلگىلەنگەن ۋاقىتتا سېھىرگەرلەر توپلاندى
وَقِيلَ لِلنَّاسِ هَلْ أَنتُم مُّجْتَمِعُونَ ﴿٣٩﴾
كىشىلەرگە: «سىلەر يىغىلىپ بولدۇڭلارمۇ؟ ئەگەر سېھىرگەرلەر غالىب چىقسا ئۇلارغا بويسۇنۇشىمىز مۇمكىن» دېيىلدى
لَعَلَّنَا نَتَّبِعُ ٱلسَّحَرَةَ إِن كَانُوا۟ هُمُ ٱلْغَـٰلِبِينَ ﴿٤٠﴾
كىشىلەرگە: «سىلەر يىغىلىپ بولدۇڭلارمۇ؟ ئەگەر سېھىرگەرلەر غالىب چىقسا ئۇلارغا بويسۇنۇشىمىز مۇمكىن» دېيىلدى
فَلَمَّا جَآءَ ٱلسَّحَرَةُ قَالُوا۟ لِفِرْعَوْنَ أَئِنَّ لَنَا لَأَجْرًا إِن كُنَّا نَحْنُ ٱلْغَـٰلِبِينَ ﴿٤١﴾
سېھىرگەرلەر پىرئەۋننىڭ قېشىغا يېتىپ كەلگەندە، ئۇلار پىرئەۋنگە: «ئەگەر غەلىبە قىلساق بىزگە چوقۇم مۇكاپات بېرىلەمدۇ؟» دېدى
قَالَ نَعَمْ وَإِنَّكُمْ إِذًا لَّمِنَ ٱلْمُقَرَّبِينَ ﴿٤٢﴾
پىرئەۋن: «ھەئە، ئۇ چاغدا (سىلەرگە مۇكاپات بېرىلىدۇ، ئۇنىڭ ئۈستىگە) سىلەر چوقۇم مېنىڭ يېقىن ئادەملىرىمدىن بولۇپ قالىسىلەر» دېدى
قَالَ لَهُم مُّوسَىٰٓ أَلْقُوا۟ مَآ أَنتُم مُّلْقُونَ ﴿٤٣﴾
مۇسا ئۇلارغا: «تاشلايدىغان نەرسەڭلارنى تاشلاڭلار!» دېدى
فَأَلْقَوْا۟ حِبَالَهُمْ وَعِصِيَّهُمْ وَقَالُوا۟ بِعِزَّةِ فِرْعَوْنَ إِنَّا لَنَحْنُ ٱلْغَـٰلِبُونَ ﴿٤٤﴾
ئۇلار ئارغامچىلىرىنى، ھاسسىلىرىنى تاشلىدى. ئۇلار: «پىرئەۋننىڭ كاتتىلىقى بىلەن قەسەمكى، بىز شەك شۈبھىسىز غەلىبە قىلىمىز» دېدى
فَأَلْقَىٰ مُوسَىٰ عَصَاهُ فَإِذَا هِىَ تَلْقَفُ مَا يَأْفِكُونَ ﴿٤٥﴾
مۇسا ھاسىسىنى تاشلىدى، (ئۇ ئەجدىھاغا ئايلىىنىپ) ئۇلارنىڭ ئويدۇرما نەرسىلىرىنى يۈتۈۋەتتى
فَأُلْقِىَ ٱلسَّحَرَةُ سَـٰجِدِينَ ﴿٤٦﴾
سىھىرگەرلەر سەجدىگە باردى
قَالُوٓا۟ ءَامَنَّا بِرَبِّ ٱلْعَـٰلَمِينَ ﴿٤٧﴾
ئۇلار: «ئالەملەرنىڭ پەرۋەردىگارىغا - مۇسا ۋە ھارۇننىڭ پەرۋەردىگارىغا - ئىمان ئېيتتۇق» دېدى
رَبِّ مُوسَىٰ وَهَـٰرُونَ ﴿٤٨﴾
ئۇلار: «ئالەملەرنىڭ پەرۋەردىگارىغا - مۇسا ۋە ھارۇننىڭ پەرۋەردىگارىغا - ئىمان ئېيتتۇق» دېدى
قَالَ ءَامَنتُمْ لَهُۥ قَبْلَ أَنْ ءَاذَنَ لَكُمْ ۖ إِنَّهُۥ لَكَبِيرُكُمُ ٱلَّذِى عَلَّمَكُمُ ٱلسِّحْرَ فَلَسَوْفَ تَعْلَمُونَ ۚ لَأُقَطِّعَنَّ أَيْدِيَكُمْ وَأَرْجُلَكُم مِّنْ خِلَـٰفٍ وَلَأُصَلِّبَنَّكُمْ أَجْمَعِينَ ﴿٤٩﴾
پىرئەۋن: «مەن رۇخسەت قىلماي تۇرۇپ سىلەر مۇساغا ئىمان ئېيتتىڭلار، ئۇ چوقۇم سىلەرگە سېھىرنى ئۈگەتكەن باشلىقىڭلار ئىكەن، (سىلەرنى قانداق جازالايدىغانلىقىمنى) ئۇزۇنغا قالماي بىلىسىلەر، قولۇڭلارنى، پۇتۇڭلارنى چوقۇم ئوڭ - چەپ قىلىپ (يەنى ئوڭ قولۇڭلار بىلەن سول پۇتۇڭلارنى ياكى سول قولۇڭلار بىلەن ئوڭ پۇتۇڭلارنى) كېسىمەن، ھەممىڭلارنى چوقۇم دارغا ئاسىمەن» دېدى
قَالُوا۟ لَا ضَيْرَ ۖ إِنَّآ إِلَىٰ رَبِّنَا مُنقَلِبُونَ ﴿٥٠﴾
سېھىرگەرلەر: «(بۇنىڭدىمۇ بىزگە) ھېچ زىيان يوق، بىز ئەلۋەتتە پەرۋەردىگارىمىزنىڭ دەرگاھىغا قايتقۇچىلارمىز، بىز مۇساغا ھەممىدىن بۇرۇن ئىمان ئېيتقۇچىلار بولغانلىقىمىز ئۈچۈن، پەرۋەردىگارىمىزنىڭ خاتالىقلىرىمىزنى مەغپىرەت قىلىشىنى ئەلۋەتتە ئۈمىد قىلىمىز» دېدى
إِنَّا نَطْمَعُ أَن يَغْفِرَ لَنَا رَبُّنَا خَطَـٰيَـٰنَآ أَن كُنَّآ أَوَّلَ ٱلْمُؤْمِنِينَ ﴿٥١﴾
سېھىرگەرلەر: «(بۇنىڭدىمۇ بىزگە) ھېچ زىيان يوق، بىز ئەلۋەتتە پەرۋەردىگارىمىزنىڭ دەرگاھىغا قايتقۇچىلارمىز، بىز مۇساغا ھەممىدىن بۇرۇن ئىمان ئېيتقۇچىلار بولغانلىقىمىز ئۈچۈن، پەرۋەردىگارىمىزنىڭ خاتالىقلىرىمىزنى مەغپىرەت قىلىشىنى ئەلۋەتتە ئۈمىد قىلىمىز» دېدى
۞ وَأَوْحَيْنَآ إِلَىٰ مُوسَىٰٓ أَنْ أَسْرِ بِعِبَادِىٓ إِنَّكُم مُّتَّبَعُونَ ﴿٥٢﴾
بىز مۇساغا: «كېچىدە بەندىلىرىمنى (يەنى بەنى ئىسرائىلنى) ئېلىپ ئاتلانغىن. سىلەر ئەلۋەتتە قوغلىنىسىلەر، (يەنى پىرئەۋن ئۆز قەۋمى بىلەن سىلەرنى قوغلاپ چىقىدۇ)» دەپ ۋەھيى قىلدۇق
فَأَرْسَلَ فِرْعَوْنُ فِى ٱلْمَدَآئِنِ حَـٰشِرِينَ ﴿٥٣﴾
پىرئەۋن شەھەرلەرگە (ئەسكەر) توپلىغۇچىلارنى ئەۋەتتى
إِنَّ هَـٰٓؤُلَآءِ لَشِرْذِمَةٌ قَلِيلُونَ ﴿٥٤﴾
(پىرئەۋن ئېيتتىكى) «بۇ كىشىلەر ئىشەنچلىك ھالدا بىر ئوچۇم ئادەملەردۇر
وَإِنَّهُمْ لَنَا لَغَآئِظُونَ ﴿٥٥﴾
ئۇلار ئىشەنچلىك ھالدا بىزنىڭ ئاچچىقىمىزنى كەلتۈرۈپ قويدى
وَإِنَّا لَجَمِيعٌ حَـٰذِرُونَ ﴿٥٦﴾
بىز ئىشەنچلىك ھالدا ئېھتىياتچان جامائەمىز»
فَأَخْرَجْنَـٰهُم مِّن جَنَّـٰتٍ وَعُيُونٍ ﴿٥٧﴾
بىز ئولارنى (يەنى پىرئەۋن بىلەن ئۇنىڭ قەۋمىنى) باغلاردىن، بۇلاقلاردىن، خەزىنىلەردىن ۋە ئېسىل تۇرالغۇدىن ئايرىۋەتتۇق
وَكُنُوزٍ وَمَقَامٍ كَرِيمٍ ﴿٥٨﴾
بىز ئولارنى (يەنى پىرئەۋن بىلەن ئۇنىڭ قەۋمىنى) باغلاردىن، بۇلاقلاردىن، خەزىنىلەردىن ۋە ئېسىل تۇرالغۇدىن ئايرىۋەتتۇق
كَذَٰلِكَ وَأَوْرَثْنَـٰهَا بَنِىٓ إِسْرَٰٓءِيلَ ﴿٥٩﴾
شۇنداق قىلىپ ئۇلارنى بەنى ئىسرائىلغا مىراس قىلىپ بەردۇق
فَأَتْبَعُوهُم مُّشْرِقِينَ ﴿٦٠﴾
ئۇلار (يەنى پىرئەۋن بىلەن ئۇنىڭ قوشۇنى) ئۇلارنى كۈن چىققان چاغدا قوغلاپ چىقتى
فَلَمَّا تَرَٰٓءَا ٱلْجَمْعَانِ قَالَ أَصْحَـٰبُ مُوسَىٰٓ إِنَّا لَمُدْرَكُونَ ﴿٦١﴾
ئىككى توپ (يەنى پىرئەۋن توپى بىلەن مۇسا ئەلەيھىسسالامنىڭ توپى) بىر - بىرىنى كۆرۈپ تۇرۇشقان چاغدا، مۇسانىڭ ئادەملىرى: «ئۇلار: (يەنى پىرئەۋن بىلەن قوشۇنى) بىزگە چوقۇم يېتىشىۋالىدىغان بولدى» دېدى
قَالَ كَلَّآ ۖ إِنَّ مَعِىَ رَبِّى سَيَهْدِينِ ﴿٦٢﴾
مۇسا ئېيتتى: «ئۇنداق بولمايدۇ (يەنى ھەرگىز يېتىشەلمەيدۇ) پەرۋەردىگارىم ئىشەنچلىك ھالدا مەن بىلەن بىللە، مېنى (قۇتۇلۇش يولىغا) باشلايدۇ»
فَأَوْحَيْنَآ إِلَىٰ مُوسَىٰٓ أَنِ ٱضْرِب بِّعَصَاكَ ٱلْبَحْرَ ۖ فَٱنفَلَقَ فَكَانَ كُلُّ فِرْقٍ كَٱلطَّوْدِ ٱلْعَظِيمِ ﴿٦٣﴾
بىز مۇساغا: «ھاساڭ بىلەن دەرياغا ئۇرغىن» دەپ ۋەھيى قىلدۇق، (مۇسا ھاسىسى بىلەن ئۇرۇۋېدى) دەريا يېرىلدى، (ھەر بىر يېرىلغان) قىسمى چوڭ تاغدەك بولۇپ قالدى
وَأَزْلَفْنَا ثَمَّ ٱلْـَٔاخَرِينَ ﴿٦٤﴾
ئىككىنچى بىر گۇرۇھنى بۇ يەرگە يېقىنلاشتۇردۇق (يەنى پىرئەۋن بىلەن قوشۇنىنى بەنى ئىسرائىلنىڭ ئارقىسىدىن دەرياغا كىرگۈزدۇق)
وَأَنجَيْنَا مُوسَىٰ وَمَن مَّعَهُۥٓ أَجْمَعِينَ ﴿٦٥﴾
مۇسا بىلەن ئۇنىڭ ھەمراھلىرىنى پۈتۈنلەي قۇتقۇزدۇق
ثُمَّ أَغْرَقْنَا ٱلْـَٔاخَرِينَ ﴿٦٦﴾
ئاندىن ئىككىنچى گۇرۇھنى (يەنى پىرئەۋن بىلەن قەۋمىنى) غەرق قىلدۇق
إِنَّ فِى ذَٰلِكَ لَـَٔايَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿٦٧﴾
بۇنىڭدا (يەنى پىرئەۋن بىلەن ئۇنىڭ قەۋمىنىڭ غەرق بولۇشىدا) ئەلۋەتتە (چوڭ) ئىبرەت بار، ئۇلارنىڭ تولىسى ئىمان ئېيتقۇچى بولمىدى
وَإِنَّ رَبَّكَ لَهُوَ ٱلْعَزِيزُ ٱلرَّحِيمُ ﴿٦٨﴾
سېنىڭ پەرۋەردىگارىڭ ئىشەنچلىك ھالدا غالىبتۇر، ناھايىتى مېھرىباندۇر
وَٱتْلُ عَلَيْهِمْ نَبَأَ إِبْرَٰهِيمَ ﴿٦٩﴾
ئۇلارغا ئىبراھىمنىڭ قىسسىسىنى ئوقۇپ بەرگىن
إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا تَعْبُدُونَ ﴿٧٠﴾
ئۆز ۋاقتىدا ئۇ ئاتىسىغا ۋە قەۋمىگە: «نېمىگە ئىبادەت قىلىسىلەر؟» دېدى
قَالُوا۟ نَعْبُدُ أَصْنَامًا فَنَظَلُّ لَهَا عَـٰكِفِينَ ﴿٧١﴾
ئۇلار: «بۇتلارغا ئىبادەت قىلىمىز، بۇ ئىبادەتنى داۋالاشتۇرىمىز» دېيىشتى
قَالَ هَلْ يَسْمَعُونَكُمْ إِذْ تَدْعُونَ ﴿٧٢﴾
ئىبراھىم ئېيتتى: «دۇئا قىلغان چېغىڭلاردا ئۇلار (دۇئايىڭلارنى ئاڭلامدۇ؟)
أَوْ يَنفَعُونَكُمْ أَوْ يَضُرُّونَ ﴿٧٣﴾
يا سىلەرگە پايدا يەتكۈزەلەمدۇ؟ يا زىيان يەتكۈزەلەمدۇ؟»
قَالُوا۟ بَلْ وَجَدْنَآ ءَابَآءَنَا كَذَٰلِكَ يَفْعَلُونَ ﴿٧٤﴾
ئۇلار ئېيتتى: «ئۇنداق ئەمەس، ئاتا - بوۋىلىرىمىزنىڭ شۇنداق قىلغانلىقىنى بايقىدۇق»
قَالَ أَفَرَءَيْتُم مَّا كُنتُمْ تَعْبُدُونَ ﴿٧٥﴾
ئىبراھىم ئېيتتى: «ئېيتىپ بېقىڭلارچۇ! سىلەرنىڭ ئىبادەت قىلغىنىڭلار نېمە؟
أَنتُمْ وَءَابَآؤُكُمُ ٱلْأَقْدَمُونَ ﴿٧٦﴾
سىلەرنىڭ ئەڭ قەدىمكى ئەجدادلىرىڭلارنىڭ ئىبادەت قىلغىنى نېمە؟
فَإِنَّهُمْ عَدُوٌّ لِّىٓ إِلَّا رَبَّ ٱلْعَـٰلَمِينَ ﴿٧٧﴾
ئۇلار (يەنى بۇتلار) مېنىڭ دۈشمىنىمدۇر، پەقەت ئالەملەرنىڭ پەرۋەردىگارى (مېنىڭ پەرۋەردىگارىمدۇر)
ٱلَّذِى خَلَقَنِى فَهُوَ يَهْدِينِ ﴿٧٨﴾
ئۇ مېنى ياراتقان، ئۇ مېنى ھىدايەت قىلىدۇ
وَٱلَّذِى هُوَ يُطْعِمُنِى وَيَسْقِينِ ﴿٧٩﴾
ئۇ مېنى تاماق بىلەن تەمىنلەيدۇ، ئۇسسۇلۇق بىلەن تەمىنلەيدۇ
وَإِذَا مَرِضْتُ فَهُوَ يَشْفِينِ ﴿٨٠﴾
ئاغرىپ قالسام ئۇ مېنى ساقايتىدۇ
وَٱلَّذِى يُمِيتُنِى ثُمَّ يُحْيِينِ ﴿٨١﴾
ئۇ مېنى قەبزى روھ قىلىدۇ، كېيىن يەنە تىرىلدۈرىدۇ
وَٱلَّذِىٓ أَطْمَعُ أَن يَغْفِرَ لِى خَطِيٓـَٔتِى يَوْمَ ٱلدِّينِ ﴿٨٢﴾
قىيامەت كۈنى ئۇنىڭ مېنىڭ خاتالىقلىرىمنى مەغپىرەت قىلىشىنى ئۈمىد قىلىمەن
رَبِّ هَبْ لِى حُكْمًا وَأَلْحِقْنِى بِٱلصَّـٰلِحِينَ ﴿٨٣﴾
پەرۋەردىگارىم! ماڭا ئىلىم ھېكمەت ئاتا قىلغىن، مېنى ياخشىلارغا قوشقىن
وَٱجْعَل لِّى لِسَانَ صِدْقٍ فِى ٱلْـَٔاخِرِينَ ﴿٨٤﴾
كېيىنكىلەر ئارىسىدا ياخشى نامىمنى قالدۇرغىن
وَٱجْعَلْنِى مِن وَرَثَةِ جَنَّةِ ٱلنَّعِيمِ ﴿٨٥﴾
مېنى نازۇنېمەتلىك مەڭگۈلۈك باغلارنىڭ ۋارىسلىرىدىن قىلغىن
وَٱغْفِرْ لِأَبِىٓ إِنَّهُۥ كَانَ مِنَ ٱلضَّآلِّينَ ﴿٨٦﴾
ئاتامغا مەغپىرەت قىلغىن، ئۇ ئىشەنچلىك ھالدا گۇمراھلاردىن بولدى
وَلَا تُخْزِنِى يَوْمَ يُبْعَثُونَ ﴿٨٧﴾
ئۇلار (يەنى خالايىقلار) (ھېساب بېرىش ئۈچۈن) تىرىلدۈرىلىدىغان كۈندە مېنى رەسۋا قىلمىغىن
يَوْمَ لَا يَنفَعُ مَالٌ وَلَا بَنُونَ ﴿٨٨﴾
ئۇ كۈنى (ھېچ كىشىگە) مال ۋە ئوغۇللار پايدا يەتكۈزەلمەيدۇ
إِلَّا مَنْ أَتَى ٱللَّهَ بِقَلْبٍ سَلِيمٍ ﴿٨٩﴾
پەقەت (اﷲ نىڭ دەرگاھى) غا پاك قەلب بىلەن كەلگەن ئادەمگىلا پايدا يەتكۈزۈلىدۇ»
وَأُزْلِفَتِ ٱلْجَنَّةُ لِلْمُتَّقِينَ ﴿٩٠﴾
مەڭگۈلۈك باغلار تەقۋادارلارغا يېقىنلاشتۇرىلىدۇ
وَبُرِّزَتِ ٱلْجَحِيمُ لِلْغَاوِينَ ﴿٩١﴾
دوزاخ گۇمراھلارغا ئوچۇق كۆرسىتىلىدۇ
وَقِيلَ لَهُمْ أَيْنَ مَا كُنتُمْ تَعْبُدُونَ ﴿٩٢﴾

فَيَقُولُوا۟ هَلْ نَحْنُ مُنظَرُونَ ﴿٢٠٣﴾

أَفَبِعَذَابِنَا يَسْتَعْجِلُونَ ﴿٢٠٤﴾

مَآ أَغْنَىٰ عَنْهُم مَّا كَانُوا۟ يُمَتَّعُونَ ﴿٢٠٧﴾

بۇ ئۇلارغا لايىق ئەمەس. ئۇلارمۇ (ئۇنىڭغا) قادىر بولالمايدۇ
إِنَّهُمْ عَنِ ٱلسَّمْعِ لَمَعْزُولُونَ ﴿٢١٢﴾

فَلَا تَدْعُ مَعَ ٱللَّهِ إِلَـٰهًا ءَاخَرَ فَتَكُونَ مِنَ ٱلْمُعَذَّبِينَ ﴿٢١٣﴾

وَأَنذِرْ عَشِيرَتَكَ ٱلْأَقْرَبِينَ ﴿٢١٤﴾
