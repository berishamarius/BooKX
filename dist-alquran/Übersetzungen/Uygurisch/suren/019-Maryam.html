بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
كٓهيعٓصٓ ﴿١﴾
كاف، ھا، يا، عين، صاد
ذِكْرُ رَحْمَتِ رَبِّكَ عَبْدَهُۥ زَكَرِيَّآ ﴿٢﴾
بۇ، پەرۋەردىگارىڭنىڭ بەندىسى زەكەرىياغا قىلغان مەرھىمىتىنىڭ بايانىدۇر
إِذْ نَادَىٰ رَبَّهُۥ نِدَآءً خَفِيًّا ﴿٣﴾
ئۇ ئۆز ۋاقتىدا پەرۋەردىگارىغا پەس ئاۋازدا مۇناجات قىلدى
قَالَ رَبِّ إِنِّى وَهَنَ ٱلْعَظْمُ مِنِّى وَٱشْتَعَلَ ٱلرَّأْسُ شَيْبًا وَلَمْ أَكُنۢ بِدُعَآئِكَ رَبِّ شَقِيًّا ﴿٤﴾
ئۇ ئېيتتى: «پەرۋەردىگارىم! مەن ئىشەنچلىك ھالدا كۈچ - قۇۋۋىتىمدىن كەتتىم، چېچىم ئاقاردى. پەرۋەردىگارىم! ساڭا دۇئا قىلىپ ئۈمىدسىز بولۇپ قالغىنىم يوق
وَإِنِّى خِفْتُ ٱلْمَوَٰلِىَ مِن وَرَآءِى وَكَانَتِ ٱمْرَأَتِى عَاقِرًا فَهَبْ لِى مِن لَّدُنكَ وَلِيًّا ﴿٥﴾
مەن تاغىلىرىمنىڭ بالىلىرىنىڭ ئورنۇمنى باسالماسلىقلىرىدىن (يەنى دىننى قولدىن بېرىپ قويۇشلىرىدىن) ئىشەنچلىك ھالدا ئەنسىرەيمەن، مېنىڭ ئايالىم تۇغماستۇر، (پەزلىڭدىن) ماڭا بىر (ياخشى) پەرزەنت ئاتا قىلغىن
يَرِثُنِى وَيَرِثُ مِنْ ءَالِ يَعْقُوبَ ۖ وَٱجْعَلْهُ رَبِّ رَضِيًّا ﴿٦﴾
ئۇ ماڭا ۋارىس بولغاي ۋە يەئقۇب ئەۋلادىغا ۋارىس بولغاي، پەرۋەردىگارىم! ئۇنى سېنىڭ رازىلىقىڭغا ئېرىشىدىغان قىلغىن»
يَـٰزَكَرِيَّآ إِنَّا نُبَشِّرُكَ بِغُلَـٰمٍ ٱسْمُهُۥ يَحْيَىٰ لَمْ نَجْعَل لَّهُۥ مِن قَبْلُ سَمِيًّا ﴿٧﴾
(اﷲ ئېيتتىكى) «ئى زەكەرىيا! بىز ئىشەنچلىك ھالدا ساڭا (خۇدانىڭ ئەلچى پەرىشتىلىرى ئارقىلىق) يەھيا ئىسىملىك بىر ئوغۇل بىلەن خۇش خەۋەر بېرىمىز، ئىلگىرى ھېچ ئادەمنى ئۇنىڭغا ئىسىمداش قىلمىدۇق»
قَالَ رَبِّ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَكَانَتِ ٱمْرَأَتِى عَاقِرًا وَقَدْ بَلَغْتُ مِنَ ٱلْكِبَرِ عِتِيًّا ﴿٨﴾
زەكەرىيا ئېيتتى: «ئايالىم تۇغماس تۇرسا، مەن قېرىپ مۈكچىيىپ كەتكەن تۇرسام، مېنىڭ قانداقمۇ ئوغلۇم بولسۇن؟»
قَالَ كَذَٰلِكَ قَالَ رَبُّكَ هُوَ عَلَىَّ هَيِّنٌ وَقَدْ خَلَقْتُكَ مِن قَبْلُ وَلَمْ تَكُ شَيْـًٔا ﴿٩﴾
(نىدا قىلغان پەرىشتە) ئېيتتى: «ئىش ئەنە شۇنداقتۇر، پەرۋەردىگارىڭ ئېيتتى: بۇ مەن ئۈچۈن ئاساندۇر، سېنى مەن ياراتتىم، ھالبۇكى سەن بۇرۇن يوق ئىدىڭ (يەنى سېنى يوقتىن بار قىلغاندەك، يەھيانى ئىككىڭلاردىن تۆرەلدۈرۈشكە قادىرمەن)»
قَالَ رَبِّ ٱجْعَل لِّىٓ ءَايَةً ۚ قَالَ ءَايَتُكَ أَلَّا تُكَلِّمَ ٱلنَّاسَ ثَلَـٰثَ لَيَالٍ سَوِيًّا ﴿١٠﴾
زەكەرىيا ئېيتتى: «پەرۋەردىگارىم! (ئايالىمنىڭ ھامىلدار بولغانلىقىغا) ماڭا بىر ئالامەت قىلىپ بەرسەڭ». اﷲ ئېيتتى: «سېنىڭ ئالامىتىڭ (شۇكى) ساق تۇرۇپ ئۈچ كېچە - كۈندۈز كىشىلەرگە سۆز قىلالمايسەن»
فَخَرَجَ عَلَىٰ قَوْمِهِۦ مِنَ ٱلْمِحْرَابِ فَأَوْحَىٰٓ إِلَيْهِمْ أَن سَبِّحُوا۟ بُكْرَةً وَعَشِيًّا ﴿١١﴾
زەكەرىيا نامازگاھىدىن قەۋمىنىڭ ئالدىغا چىقىپ، ئۇلارنى ئەتىگەن - ئاخشامدا تەسبىھ ئېيتىشقا ئىشارەت قىلدى
يَـٰيَحْيَىٰ خُذِ ٱلْكِتَـٰبَ بِقُوَّةٍ ۖ وَءَاتَيْنَـٰهُ ٱلْحُكْمَ صَبِيًّا ﴿١٢﴾
(يەھيا تۇغۇلۇپ چوڭ بولغاندا ئۇنىڭغا) «ئى يەھيا! مۇقەددەس كىتاب-ۋەھىينى (يەنى تەۋراتنى) تىرىشىپ ئوقۇغىن» (دېدۇق). ئۇنىڭغا بالىلىق چېغىدىلا ئەقىل - پاراسەت ئاتا قىلدۇق
وَحَنَانًا مِّن لَّدُنَّا وَزَكَوٰةً ۖ وَكَانَ تَقِيًّا ﴿١٣﴾
بىز ئۇنىڭغا مەرھەمەت قىلىش ۋە ئۇنى (ناچار خىسلەتلەردىن) پاك قىلىش يۈزىسىدىن (شۇنداق قىلدۇق)، يەھيا تەقۋادار ئىدى
وَبَرًّۢا بِوَٰلِدَيْهِ وَلَمْ يَكُن جَبَّارًا عَصِيًّا ﴿١٤﴾
ئۇنى ئاتا - ئانىسىغا كۆيۈمچان (قىلدۇق)، ئۇ مۇتەكەببىر (يەنى ئاتا - ئانىسىنى قاخشاتقۇچى)، (پەرۋەردىگارىغا) ئاسىيلىق قىلغۇچى بولمىدى
وَسَلَـٰمٌ عَلَيْهِ يَوْمَ وُلِدَ وَيَوْمَ يَمُوتُ وَيَوْمَ يُبْعَثُ حَيًّا ﴿١٥﴾
ئۇ تۇغۇلغان كۈنىدە، ۋاپات بولغان كۈنىدە، تىرىلىپ (قەبرىدىن) تۇرغۇزۇلغان كۈنىدە (اﷲ تائالا تەرىپىدىن بولغان) ئامانلىققا ئېرىشىدۇ
وَٱذْكُرْ فِى ٱلْكِتَـٰبِ مَرْيَمَ إِذِ ٱنتَبَذَتْ مِنْ أَهْلِهَا مَكَانًا شَرْقِيًّا ﴿١٦﴾
(ئۇلارغا) قۇرئاندا مەريەم (قىسسىسى) نى بايان قىلغىن، ئەينى زاماندا ئۇ ئائىلىسىدىن ئايرىلىپ شەرق تەرەپتىكى بىر جايغا باردى
فَٱتَّخَذَتْ مِن دُونِهِمْ حِجَابًا فَأَرْسَلْنَآ إِلَيْهَا رُوحَنَا فَتَمَثَّلَ لَهَا بَشَرًا سَوِيًّا ﴿١٧﴾
ئۇ پەردە تارتىپ كىشىلەردىن يۇشۇرۇندى، ئۇنىڭغا بىزنىڭ روھىمىزنى (يەنى جىبرىئىل ئەلەيھىسسالامنى) ئەۋەتتۇق، ئۇ مەريەمگە بېجىرىم ئادەم سۈرىتىدە كۆرۈندى
قَالَتْ إِنِّىٓ أَعُوذُ بِٱلرَّحْمَـٰنِ مِنكَ إِن كُنتَ تَقِيًّا ﴿١٨﴾
مەريەم ئېيتتى: «مەن ئىشەنچلىك ھالدا مەرھەمەتلىك اﷲ قا سېغىنىپ سېنىڭ چېقىلىشىڭدىن پاناھ تىلەيمەن، ئەگەر سەن تەقۋادار بولساڭ (ماڭا چېقىلمىغىن)»
قَالَ إِنَّمَآ أَنَا۠ رَسُولُ رَبِّكِ لِأَهَبَ لَكِ غُلَـٰمًا زَكِيًّا ﴿١٩﴾
جىبرىئىل ئېيتتى: «مەن ساڭا بىر پاك ئوغۇل بېرىش ئۈچۈن ئەۋەتىلگەن پەرۋەردىگارىڭنىڭ ئەلچىسىمەن»
قَالَتْ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَلَمْ يَمْسَسْنِى بَشَرٌ وَلَمْ أَكُ بَغِيًّا ﴿٢٠﴾
مەريەم ئېيتتى: «ماڭا كىشى يېقىنلاشمىغان تۇرسا، مەن پاھىشە قىلمىغان تۇرسام، قانداقمۇ مېنىڭ ئوغۇل بالام بولسۇن؟»
قَالَ كَذَٰلِكِ قَالَ رَبُّكِ هُوَ عَلَىَّ هَيِّنٌ ۖ وَلِنَجْعَلَهُۥٓ ءَايَةً لِّلنَّاسِ وَرَحْمَةً مِّنَّا ۚ وَكَانَ أَمْرًا مَّقْضِيًّا ﴿٢١﴾
جىبرىئىل ئېيتتى: «ئەمەلىيەت سەن دېگەندەكتۇر، (لېكىن پەرۋەردىگارىڭ ئېيتتى: ئۇ ماڭا ئاساندۇر. ئۇنى كىشىلەرگە قۇدرىتىمىزنى كۆرسىتىدىغان) دەلىل ۋە بىز تەرەپتىن بولغان مەرھەمەت قىلدۇق، بۇ تەقدىر قىلىنىپ بولغان ئىشتۇر»
۞ فَحَمَلَتْهُ فَٱنتَبَذَتْ بِهِۦ مَكَانًا قَصِيًّا ﴿٢٢﴾
مەريەم ھامىلدار بولدى، (ئائىلىسىدىن) يىراق بىر جايغا كەتتى
فَأَجَآءَهَا ٱلْمَخَاضُ إِلَىٰ جِذْعِ ٱلنَّخْلَةِ قَالَتْ يَـٰلَيْتَنِى مِتُّ قَبْلَ هَـٰذَا وَكُنتُ نَسْيًا مَّنسِيًّا ﴿٢٣﴾
تولغاقنىڭ قاتتىقلىقى ئۇنى (يۆلىنىۋېلىش ئۈچۈن) بىر خورما دەرىخىنىڭ يېنىغا كېلىۋېلىشقا مەجبۇر قىلدى. مەريەم ئېيتتى: «كاشكى مەن بۇنىڭدىن ئىلگىرى ئۆلۈپ كەتكەن بولسامچۇ! كىشىلەر تەرىپىدىن ئۇنتۇلۇپ كەتكەن (ئەرزىمەس) نەرسىگە ئايلىنىپ كەتسەمچۇ!»
فَنَادَىٰهَا مِن تَحْتِهَآ أَلَّا تَحْزَنِى قَدْ جَعَلَ رَبُّكِ تَحْتَكِ سَرِيًّا ﴿٢٤﴾
جىبرىئىل ئۇنىڭغا خورما دەرىخىنىڭ ئاستىدىن نىدا قىلدى: «غەم قىلمىغىن، پەرۋەردىگارىڭ سېنىڭ ئاستىڭدىن (ئېقىپ تۇرىدىغان) بىر ئېرىقنى پەيدا قىلدى
وَهُزِّىٓ إِلَيْكِ بِجِذْعِ ٱلنَّخْلَةِ تُسَـٰقِطْ عَلَيْكِ رُطَبًا جَنِيًّا ﴿٢٥﴾
خورما دەرىخىنى ئۆز تەرىپىڭگە ئىرغىتقىن، ئالدىڭغا پىشقان يېڭى خورمىلار تۆكۈلىدۇ
فَكُلِى وَٱشْرَبِى وَقَرِّى عَيْنًا ۖ فَإِمَّا تَرَيِنَّ مِنَ ٱلْبَشَرِ أَحَدًا فَقُولِىٓ إِنِّى نَذَرْتُ لِلرَّحْمَـٰنِ صَوْمًا فَلَنْ أُكَلِّمَ ٱلْيَوْمَ إِنسِيًّا ﴿٢٦﴾
(بۇ شېرىن خورمىدىن) يېگىن، (بۇ تاتلىق سۇدىن ئىچكىن ۋە خۇرسەن بولغىن، بىرەر ئادەمنى كۆرسەڭ (ئۇ بالا توغرۇلۇق سورىسا): مەن ئىشەنچلىك ھالدا مەرھەمەتلىك اﷲ قا ۋەدە بەردىم، بۈگۈن ھېچ ئادەمگە سۆز قىلمايمەن دېگىن»
فَأَتَتْ بِهِۦ قَوْمَهَا تَحْمِلُهُۥ ۖ قَالُوا۟ يَـٰمَرْيَمُ لَقَدْ جِئْتِ شَيْـًٔا فَرِيًّا ﴿٢٧﴾
مەريەم بوۋاقنى (يەنى ئىسا ئەلەيھىسسالامنى) كۆتۈرۈپ قەۋمىنىڭ ئالدىغا كەلدى، ئۇلار ئېيتتى: «ئى مەريەم! سەن ئىشەنچلىك ھالدا غەلىتە ئىش قىلدىڭ
يَـٰٓأُخْتَ هَـٰرُونَ مَا كَانَ أَبُوكِ ٱمْرَأَ سَوْءٍ وَمَا كَانَتْ أُمُّكِ بَغِيًّا ﴿٢٨﴾
ئى ھارۇننىڭ ھەمشىرىسى! سېنىڭ ئاتاڭ يامان ئادەم ئەمەس ئىدى، ئاناڭمۇ يولدىن چىققان خوتۇن ئەمەس ئىدى»
فَأَشَارَتْ إِلَيْهِ ۖ قَالُوا۟ كَيْفَ نُكَلِّمُ مَن كَانَ فِى ٱلْمَهْدِ صَبِيًّا ﴿٢٩﴾
مەريەم بوۋاقنى كۆرسەتتى، ئۇلار: «بۆشۈكتىكى بوۋاققا قانداق سۆزلەيمىز» دېدى
قَالَ إِنِّى عَبْدُ ٱللَّهِ ءَاتَىٰنِىَ ٱلْكِتَـٰبَ وَجَعَلَنِى نَبِيًّا ﴿٣٠﴾
بوۋاق ئېيتتى: «مەن ئىشەنچلىك ھالدا اﷲ نىڭ بەندىسىمەن، ماڭا مۇقەددەس كىتاب-ۋەھىينى بەردى (يەنى ئىنجىلنى بېرىدۇ) ۋە مېنى خۇدانىڭ ئەلچىسى قىلدى (يەنى قىلىدۇ)
وَجَعَلَنِى مُبَارَكًا أَيْنَ مَا كُنتُ وَأَوْصَـٰنِى بِٱلصَّلَوٰةِ وَٱلزَّكَوٰةِ مَا دُمْتُ حَيًّا ﴿٣١﴾
قەيەردە بولاي مېنى بەرىكەتلىك قىلدى، ھاياتلا بولىدىكەنمەن، ماڭا نامازنى، زاكات (پاكلاش ئۇلۇشى)نى ئادا قىلىشنى تەۋسىيە قىلدى
وَبَرًّۢا بِوَٰلِدَتِى وَلَمْ يَجْعَلْنِى جَبَّارًا شَقِيًّا ﴿٣٢﴾
مېنى ئانامغا كۆيۈمچان قىلدى، مېنى مۇتەكەببىر، شەقى قىلمىدى
وَٱلسَّلَـٰمُ عَلَىَّ يَوْمَ وُلِدتُّ وَيَوْمَ أَمُوتُ وَيَوْمَ أُبْعَثُ حَيًّا ﴿٣٣﴾
مەن تۇغۇلغان كۈنۈمدە، ۋاپات بولغان كۈنۈمدە، تىرىلىپ قەبرەمدىن تۇرغۇزۇلغان كۈنۈمدە (اﷲ تائالا تەرىپىدىن بولغان) ئامانلىققا ئېرىشىمەن»
ذَٰلِكَ عِيسَى ٱبْنُ مَرْيَمَ ۚ قَوْلَ ٱلْحَقِّ ٱلَّذِى فِيهِ يَمْتَرُونَ ﴿٣٤﴾
ئەنە شۇ ئىسا ئىبىن مەريەمنىڭ (قىسسىسىدۇر). ئىسا ئۇلار دەتالاش قىلىۋاتقان ئۆزىنڭ ئىشىنىڭ (ھەقىقىتى) ئۈستىدە ھەق گەپنى قىلدى
مَا كَانَ لِلَّهِ أَن يَتَّخِذَ مِن وَلَدٍ ۖ سُبْحَـٰنَهُۥٓ ۚ إِذَا قَضَىٰٓ أَمْرًا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ﴿٣٥﴾
«اﷲ نىڭ شەنىگە بالىسى بولۇش لايىق ئەمەس، اﷲ (بالىدىن، شېرىكتىن) پاكتۇر، اﷲ بىرەر ئىش قىلىشنى ئىرادە قىلسا، ئۇنىڭغا ۋۇجۇتقا كەل دەيدۇ - دە ئۇ ۋۇجۇتقا كېلىدۇ

أَسْمِعْ بِهِمْ وَأَبْصِرْ يَوْمَ يَأْتُونَنَا ۖ لَـٰكِنِ ٱلظَّـٰلِمُونَ ٱلْيَوْمَ فِى ضَلَـٰلٍ مُّبِينٍ ﴿٣٨﴾

إِذْ قَالَ لِأَبِيهِ يَـٰٓأَبَتِ لِمَ تَعْبُدُ مَا لَا يَسْمَعُ وَلَا يُبْصِرُ وَلَا يُغْنِى عَنكَ شَيْـًٔا ﴿٤٢﴾
ئۆز ۋاقتىدا ئىبراھىم ئاتىسىغا ئېيتتى: «ئى ئاتا نېمىشقا ئاڭلىمايدىغان، كۆرمەيدىغان ۋە ساڭا ھېچ نەرسىگە ئەسقاتمايدىغان نەرسىلەرگە چوقۇنىسەن

يَـٰٓأَبَتِ إِنِّىٓ أَخَافُ أَن يَمَسَّكَ عَذَابٌ مِّنَ ٱلرَّحْمَـٰنِ فَتَكُونَ لِلشَّيْطَـٰنِ وَلِيًّا ﴿٤٥﴾
ئى ئاتا! سېنىڭ مەرھەمەتلىك اﷲ نىڭ ئازابىغا قېلىشىڭدىن، شەيتان (ئازدۇرغۇچى)نىڭ ھەمراھىغا ئايلىنىپ قېلىشىڭدىن قورقىمەن»
قَالَ أَرَاغِبٌ أَنتَ عَنْ ءَالِهَتِى يَـٰٓإِبْرَٰهِيمُ ۖ لَئِن لَّمْ تَنتَهِ لَأَرْجُمَنَّكَ ۖ وَٱهْجُرْنِى مَلِيًّا ﴿٤٦﴾
ئۇ (يەنى ئازەر ئېيتتى: «ئى ئىبراھىم! سەن مېنىڭ ئىلاھلىرىمدىن يۈز ئۆرۈمەكچىمۇسەن، ئەگەر سەن (مۇنداق قىلىشتىن) يانمىساڭ، چوقۇم سېنى تاش - كېسەك قىلىپ ئۆلتۈرىمەن، ئۇزاق زامان مەندىن يىراق تۇرغىن»
قَالَ سَلَـٰمٌ عَلَيْكَ ۖ سَأَسْتَغْفِرُ لَكَ رَبِّىٓ ۖ إِنَّهُۥ كَانَ بِى حَفِيًّا ﴿٤٧﴾
ئىبراھىم ئېيتتى: «ساڭا ئامانلىق بولسۇن، پەرۋەردىگارىمدىن ساڭا مەغپىرەت تىلەيمەن، اﷲ ماڭا ئىشەنچلىك ھالدا مېھرىباندۇر
وَأَعْتَزِلُكُمْ وَمَا تَدْعُونَ مِن دُونِ ٱللَّهِ وَأَدْعُوا۟ رَبِّى عَسَىٰٓ أَلَّآ أَكُونَ بِدُعَآءِ رَبِّى شَقِيًّا ﴿٤٨﴾
سىلەردىن ۋە سىلەر اﷲ نى قويۇپ چوقۇنۇۋاتقان نەرسىلەردىن يىراق بولىمەن، پەرۋەردىگارىمغا ئىبادەت قىلىمەن، پەرۋەردىگارىمنىڭ ئىبادىتىدىن مەھرۇم بولۇپ قالماسلىقنى ئۈمىد قىلىمەن»
فَلَمَّا ٱعْتَزَلَهُمْ وَمَا يَعْبُدُونَ مِن دُونِ ٱللَّهِ وَهَبْنَا لَهُۥٓ إِسْحَـٰقَ وَيَعْقُوبَ ۖ وَكُلًّا جَعَلْنَا نَبِيًّا ﴿٤٩﴾

وَوَهَبْنَا لَهُم مِّن رَّحْمَتِنَا وَجَعَلْنَا لَهُمْ لِسَانَ صِدْقٍ عَلِيًّا ﴿٥٠﴾
ئۇلارغا بىزنىڭ رەھمىتىمىزنى ئاتا قىلدۇق، ئۇلارنىڭ نامىنى بەلەن ۋە يۇقىرى قىلدۇق
وَٱذْكُرْ فِى ٱلْكِتَـٰبِ مُوسَىٰٓ ۚ إِنَّهُۥ كَانَ مُخْلَصًا وَكَانَ رَسُولًا نَّبِيًّا ﴿٥١﴾
مۇقەددەس كىتاب-ۋەھىيتا (يەنى قۇرئاندا) مۇسا (قىسىسى) نى بايان قىلغىن، شۈبھىسىزكى، ئۇ اﷲ ئىبادىتىگە خاس قىلغان ھەم رەسۇل (يەنى ئەلچى)، ھەم خۇدانىڭ ئەلچىسى ئىدى
وَنَـٰدَيْنَـٰهُ مِن جَانِبِ ٱلطُّورِ ٱلْأَيْمَنِ وَقَرَّبْنَـٰهُ نَجِيًّا ﴿٥٢﴾
ئۇنىڭغا بىز تۇر تېغىنىڭ ئوڭ تەرىپىدىن نىدا قىلدۇق، ئۇنى بىز مۇناجات ئۈچۈن (دەرگاھىمىزغا) يېقىنلاشتۇردۇق
وَوَهَبْنَا لَهُۥ مِن رَّحْمَتِنَآ أَخَاهُ هَـٰرُونَ نَبِيًّا ﴿٥٣﴾
ئۇنىڭغا بىز مەرھەمەت قىلىپ قېرىندىشى ھارۇننى خۇدانىڭ ئەلچىسى قىلىپ بەردۇق
وَٱذْكُرْ فِى ٱلْكِتَـٰبِ إِسْمَـٰعِيلَ ۚ إِنَّهُۥ كَانَ صَادِقَ ٱلْوَعْدِ وَكَانَ رَسُولًا نَّبِيًّا ﴿٥٤﴾

وَكَانَ يَأْمُرُ أَهْلَهُۥ بِٱلصَّلَوٰةِ وَٱلزَّكَوٰةِ وَكَانَ عِندَ رَبِّهِۦ مَرْضِيًّا ﴿٥٥﴾
ئۇ ئائىلىسىدىكىلەرنى ناماز ئوقۇشقا بۇيرۇيتتى، پەرۋەردىگارىنىڭ دەرگاھىدا ئۇنىڭ رازىلىقىغا ئېرىشكەن ئىدى
وَٱذْكُرْ فِى ٱلْكِتَـٰبِ إِدْرِيسَ ۚ إِنَّهُۥ كَانَ صِدِّيقًا نَّبِيًّا ﴿٥٦﴾

تِلْكَ ٱلْجَنَّةُ ٱلَّتِى نُورِثُ مِنْ عِبَادِنَا مَن كَانَ تَقِيًّا ﴿٦٣﴾
بەندىلىرىمىز ئىچىدىكى تەقۋادارلارغا بىز مىراس قىلىپ بېرىدىغان مەڭگۈلۈك باغلار ئەنە شۇ
وَمَا نَتَنَزَّلُ إِلَّا بِأَمْرِ رَبِّكَ ۖ لَهُۥ مَا بَيْنَ أَيْدِينَا وَمَا خَلْفَنَا وَمَا بَيْنَ ذَٰلِكَ ۚ وَمَا كَانَ رَبُّكَ نَسِيًّا ﴿٦٤﴾
بىز (دۇنياغا) پەرۋەردىگارىڭنىڭ ئەمرى بىلەن چۈشىمىز، بىزنىڭ ئالدىمىزدىكى، ئارقىمىزدىكى ۋە ئالدى - كەينىمىز ئارىسىدىكى ئىشلارنىڭ ھەممىسى اﷲ نىڭ ئىلكىدىدۇر، پەرۋەردىگارىڭ (بەندىلىرىنىڭ ئەمەللىرىدىن ھېچ نەرسىنى) ئۇنتۇمايدۇ
رَّبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا فَٱعْبُدْهُ وَٱصْطَبِرْ لِعِبَـٰدَتِهِۦ ۚ هَلْ تَعْلَمُ لَهُۥ سَمِيًّا ﴿٦٥﴾
ئۇ ئاسمانلارنىڭ، زېمىن ۋە ئۇلارنىڭ ئارىسىدىكى شەيئىلەرنىڭ پەرۋەردىگارىدۇر، ئۇنىڭغا ئىبادەت قىلغىن، اﷲ نىڭ ئىبادىتىگە چىداملىق بولغىن، سەن ئۇنىڭ شېرىكى بارلىقىنى بىلەمسەن؟
وَيَقُولُ ٱلْإِنسَـٰنُ أَءِذَا مَا مِتُّ لَسَوْفَ أُخْرَجُ حَيًّا ﴿٦٦﴾
ئىنسان (يەنى ئۆلگەندىن كېيىن تىرىلىشىگە ئېتىقاد قىلمايدىغان ھەقىقەتنى رەت قىلغۇچى): «مەن چوقۇم تىرىلەمدىم؟» دەيدۇ
أَوَلَا يَذْكُرُ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ مِن قَبْلُ وَلَمْ يَكُ شَيْـًٔا ﴿٦٧﴾
ئىنسان بىز ئۇنى ئىلگىرى يوقتىن بار قىلغانلىقىمىزنى ئەسلىمەمدۇ؟
فَوَرَبِّكَ لَنَحْشُرَنَّهُمْ وَٱلشَّيَـٰطِينَ ثُمَّ لَنُحْضِرَنَّهُمْ حَوْلَ جَهَنَّمَ جِثِيًّا ﴿٦٨﴾
پەرۋەردىگارىڭ بىلەن قەسەمكى، ئۇلارنى ئەلۋەتتە (ئۇلارنى ئازدۇرغان) شەيتان (ئازدۇرغۇچى)لار بىلەن قوشۇپ يىغىمىز، ئاندىن ئۇلارنى مەڭگۈلۈك ئوتنىڭ چۆرىسىدە تىزلاندۇرۇپ ئولتۇرغۇزغان ھالدا ئەلۋەتتە ھازىر قىلىمىز
ثُمَّ لَنَنزِعَنَّ مِن كُلِّ شِيعَةٍ أَيُّهُمْ أَشَدُّ عَلَى ٱلرَّحْمَـٰنِ عِتِيًّا ﴿٦٩﴾
ئاندىن ھەربىر گۇرۇھنىڭ ئىچىدىن مەرھەمەتلىك اﷲ قا ئەڭ ئاسىي بولغانلىرىنى تارتىپ چىقىرىمىز
ثُمَّ لَنَحْنُ أَعْلَمُ بِٱلَّذِينَ هُمْ أَوْلَىٰ بِهَا صِلِيًّا ﴿٧٠﴾
ئاندىن بىز دوزاخ ئازابىغا ئەڭ لايىق بولغانلارنى ئەلۋەتتە ئوبدان بىلىمىز
وَإِن مِّنكُمْ إِلَّا وَارِدُهَا ۚ كَانَ عَلَىٰ رَبِّكَ حَتْمًا مَّقْضِيًّا ﴿٧١﴾
سىلەرنىڭ ئىچىڭلاردىن دوزاخقا بارمايدىغان بىرەر كىشىمۇ قالمايدۇ، بۇ پەرۋەردىگارىڭنىڭ ئۆزگەرمەس ھۆكمىدۇر
ثُمَّ نُنَجِّى ٱلَّذِينَ ٱتَّقَوا۟ وَّنَذَرُ ٱلظَّـٰلِمِينَ فِيهَا جِثِيًّا ﴿٧٢﴾
ئاندىن ئەڭ تەقۋادارلارنى (مەڭگۈلۈك ئوتدىن) قۇتقۇزىمىز، زالىملارنى مەڭگۈلۈك ئوتدە تىزلىنىپ ئولتۇرغان ھالدا قويىمىز
وَإِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُنَا بَيِّنَـٰتٍ قَالَ ٱلَّذِينَ كَفَرُوا۟ لِلَّذِينَ ءَامَنُوٓا۟ أَىُّ ٱلْفَرِيقَيْنِ خَيْرٌ مَّقَامًا وَأَحْسَنُ نَدِيًّا ﴿٧٣﴾
ئۇلارغا بىزنىڭ روشەن ئايەتلىرىمىز تىلاۋەت قىلىنغان چاغدا، ھەقىقەتنى رەت قىلغۇچىلار مۆمىنلەرگە: «ئىككى گۇرۇھتىن قايسىنىڭ ئورنى ياخشى، سورۇنى گۈزەل؟» دېدى
وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هُمْ أَحْسَنُ أَثَـٰثًا وَرِءْيًا ﴿٧٤﴾
ئۇلاردىن بۇرۇن (بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغان) نۇرغۇن ئۈممەتلەرنى ھالاك قىلدۇق، بۇلار مەيلى مال - مۈلۈك جەھەتتىن بولسۇن، مەيلى تاشقى كۆرۈنۈش جەھەتتىن بولسۇن، ئۇلاردىن ئارتۇق ئىدى
قُلْ مَن كَانَ فِى ٱلضَّلَـٰلَةِ فَلْيَمْدُدْ لَهُ ٱلرَّحْمَـٰنُ مَدًّا ۚ حَتَّىٰٓ إِذَا رَأَوْا۟ مَا يُوعَدُونَ إِمَّا ٱلْعَذَابَ وَإِمَّا ٱلسَّاعَةَ فَسَيَعْلَمُونَ مَنْ هُوَ شَرٌّ مَّكَانًا وَأَضْعَفُ جُندًا ﴿٧٥﴾
(ئى مۇھەممەد!) ئېيتقىنكى، «كىمكى گۇمراھلىقتا بولىدىكەن، مەرھەمەتلىك اﷲ ئۇنىڭغا مۆھلەت بىرىدۇ، تاكى ئۇلار ئازاب ۋە قىيامەتتىن ئىبارەت اﷲ ئۇلارغا ۋەدە قىلغان نەرسىلەرنى كۆرگەندە ئاندىن كىمنىڭ ئورنى يامان ۋە كىمنىڭ لەشكىرى ئاجىز ئىكەنلىكىنى بىلىدۇ»
وَيَزِيدُ ٱللَّهُ ٱلَّذِينَ ٱهْتَدَوْا۟ هُدًى ۗ وَٱلْبَـٰقِيَـٰتُ ٱلصَّـٰلِحَـٰتُ خَيْرٌ عِندَ رَبِّكَ ثَوَابًا وَخَيْرٌ مَّرَدًّا ﴿٧٦﴾
اﷲ ھىدايەت تاپقۇچىلارنىڭ ھىدايىتىنى زىيادە قىلىدۇ، باقى قالىدىغان ياخشى ئەمەللەر ساۋاب جەھەتتىن ۋە ئاقىۋەت جەھەتتىن پەرۋەردىگارىڭنىڭ دەرگاھىدا ئارتۇقتۇر
أَفَرَءَيْتَ ٱلَّذِى كَفَرَ بِـَٔايَـٰتِنَا وَقَالَ لَأُوتَيَنَّ مَالًا وَوَلَدًا ﴿٧٧﴾
بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغان ۋە ماڭا (ئاخىرەتتە) ئەلۋەتتە مال - مۈلۈك، ئەۋلاد بېرىلىدۇ دېگەن ئادەمنى كۆردۈڭمۇ؟
أَطَّلَعَ ٱلْغَيْبَ أَمِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ﴿٧٨﴾
ئۇ غەيبنى بىلگەنمۇ؟ ياكى مەرھەمەتلىك اﷲ نىڭ ئەھدىگە ئېرىشكەنمۇ؟
كَلَّا ۚ سَنَكْتُبُ مَا يَقُولُ وَنَمُدُّ لَهُۥ مِنَ ٱلْعَذَابِ مَدًّا ﴿٧٩﴾
ھەرگىز ئۇنداق بولغان ئەمەس، ئۇنىڭ دېگەنلىرىنى خاتىرىلەپ قويىمىز، ئۇنىڭغا بېرىدىغان ئازابنى كۈچەيتىمىز
وَنَرِثُهُۥ مَا يَقُولُ وَيَأْتِينَا فَرْدًا ﴿٨٠﴾
ئۇنىڭ دېگەنلىرىنى (يەنى مېلىنى ۋە ئەۋلادىنى) باشقىلارغا مىراس قىلىپ قالدۇرىمىز، ئۇ ئالدىمىزغا تەنھا كېلىدۇ
وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لِّيَكُونُوا۟ لَهُمْ عِزًّا ﴿٨١﴾
ئۇلار (يەنى مۇشرىكلار) ئىززەتكە ئېرىشىش ئۈچۈن (يەنى بۇتلىرىنىڭ شاپائىتى بىلەن اﷲ نىڭ ئالدىدا ئىززەتلىك بولۇش ئۈچۈن)، اﷲ نى قويۇپ ئىلاھلار (يەنى بۇتلار) غا چوقۇندى
كَلَّا ۚ سَيَكْفُرُونَ بِعِبَادَتِهِمْ وَيَكُونُونَ عَلَيْهِمْ ضِدًّا ﴿٨٢﴾
ئىش ئۇلارنىڭ گۇمان قىلغىنىدەك ئەمەس (يەنى بۇتلىرى ئۇلارنىڭ اﷲ ئالدىدا ئىززەت تېپىشلىرىنىڭ سەۋەبى بولالمايدۇ)، ئۇلارنىڭ (يەنى مۇشرىكلارنىڭ) چوقۇنۇشىنى ئىنكار قىلىدۇ، ئۇلار (يەنى مۇشرىكلار) نىڭ دۈشمىنىگە ئايلىنىدۇ
أَلَمْ تَرَ أَنَّآ أَرْسَلْنَا ٱلشَّيَـٰطِينَ عَلَى ٱلْكَـٰفِرِينَ تَؤُزُّهُمْ أَزًّا ﴿٨٣﴾
بىزنىڭ شەيتان (ئازدۇرغۇچى)لارنى ھەقىقەتنى رەت قىلغۇچىلارغا ئەۋەتكەنلىكىمىزنى (يەنى مۇسەللەت قىلغانلىقىمىزنى) بىلمەمسەن؟ شەيتان (ئازدۇرغۇچى)لار ئۇلارنى (گۇناھلارغا) قىزىقتۇرىدۇ
فَلَا تَعْجَلْ عَلَيْهِمْ ۖ إِنَّمَا نَعُدُّ لَهُمْ عَدًّا ﴿٨٤﴾
ئۇلارنىڭ ھالاك بولۇشىغا ئالدىراپ كەتمىگىن، (ئۇلارنىڭ كۈنى ئاز قالغان بولۇپ) ئۇنى بىز ساناپ تۇرىمىز
يَوْمَ نَحْشُرُ ٱلْمُتَّقِينَ إِلَى ٱلرَّحْمَـٰنِ وَفْدًا ﴿٨٥﴾
شۇ كۈندە (يەنى قىيامەت كۈنىدە) تەقۋادارلارنى مەرھەمەتلىك پەرۋەردىگارنىڭ دەرگاھىغا ئىززەت - ئىكرام بىلەن يىغىمىز
وَنَسُوقُ ٱلْمُجْرِمِينَ إِلَىٰ جَهَنَّمَ وِرْدًا ﴿٨٦﴾
گۇناھكارلارنى ئۇلار چاڭقىغان ھالدا مەڭگۈلۈك ئوتگە ھەيدەيمىز
لَّا يَمْلِكُونَ ٱلشَّفَـٰعَةَ إِلَّا مَنِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ﴿٨٧﴾
مەرھەمەتلىك اﷲ نىڭ ئەھدىگە ئېرىشكەنلەردىن باشقىلار شاپائەت قىلىشقا قادىر بولالمايدۇ
وَقَالُوا۟ ٱتَّخَذَ ٱلرَّحْمَـٰنُ وَلَدًا ﴿٨٨﴾
ئۇلار: «اﷲ نىڭ بالىسى بار» دېيىشتى
لَّقَدْ جِئْتُمْ شَيْـًٔا إِدًّا ﴿٨٩﴾
سىلەر شەك - شۈبھىسىز قەبىھ بىر سۆزنى قىلدىڭلار
تَكَادُ ٱلسَّمَـٰوَٰتُ يَتَفَطَّرْنَ مِنْهُ وَتَنشَقُّ ٱلْأَرْضُ وَتَخِرُّ ٱلْجِبَالُ هَدًّا ﴿٩٠﴾
بۇ سۆزنىڭ (يامانلىقىدىن) ئاسمانلار پارچىلىنىپ كەتكىلى، يەر يېرىلىپ كەتكىلى، تاغلار گۇمران بولۇپ كەتكىلى تاس قالدى
أَن دَعَوْا۟ لِلرَّحْمَـٰنِ وَلَدًا ﴿٩١﴾
بۇ شۇنىڭ ئۈچۈنكى، ئۇلار اﷲ نىڭ بالىسى بار دەپ دەۋا قىلدى
وَمَا يَنۢبَغِى لِلرَّحْمَـٰنِ أَن يَتَّخِذَ وَلَدًا ﴿٩٢﴾
بالىسى بولۇش اﷲ قا لايىق ئەمەستۇر
إِن كُلُّ مَن فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ إِلَّآ ءَاتِى ٱلرَّحْمَـٰنِ عَبْدًا ﴿٩٣﴾
ئاسمانلاردا ۋە زېمىندا اﷲ قا بەندە بولۇپ بويسۇنمايدىغان ھېچ مەخلۇق يوقتۇر
لَّقَدْ أَحْصَىٰهُمْ وَعَدَّهُمْ عَدًّا ﴿٩٤﴾
شەك - شۈبھىسىزكى، اﷲ ئۇلارنى تولۇق ئىگەللىدى ۋە سانىنى مۇكەممەل بىلدى
وَكُلُّهُمْ ءَاتِيهِ يَوْمَ ٱلْقِيَـٰمَةِ فَرْدًا ﴿٩٥﴾
قىيامەت كۈنى ئۇلارنىڭ ھەممىسى اﷲ نىڭ دەرگاھىغا تەنھا كېلىدۇ
إِنَّ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ سَيَجْعَلُ لَهُمُ ٱلرَّحْمَـٰنُ وُدًّا ﴿٩٦﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلار ئۈچۈن اﷲ ئىشەنچلىك ھالدا (ئۇلارنىڭ دىللىرىدا) مۇھەببەت پەيدا قىلىدۇ
فَإِنَّمَا يَسَّرْنَـٰهُ بِلِسَانِكَ لِتُبَشِّرَ بِهِ ٱلْمُتَّقِينَ وَتُنذِرَ بِهِۦ قَوْمًا لُّدًّا ﴿٩٧﴾
(ئى مۇھەممەد!) تەقۋادارلارغا خۇش - خەۋەر بېرىشىڭ ۋە سەركەش قەۋمنى ئاگاھلاندۇرۇشۇڭ ئۈچۈن، بىز قۇرئاننى سېنىڭ تىلىڭ بىلەن چۈشىنىشلىك قىلىپ نازىل قىلدۇق
وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هَلْ تُحِسُّ مِنْهُم مِّنْ أَحَدٍ أَوْ تَسْمَعُ لَهُمْ رِكْزًۢا ﴿٩٨﴾
ئۇلاردىن ئىلگىرى نۇرغۇن ئۈممەتلەرنى ھالاك قىلدۇق، ئۇلاردىن بىرەر ئەھەدىنى كۆرەلەمسەن؟ ياكى ئۇلاردىن بىرەر پەس ئاۋاز ئاڭلىيالامسەن؟ (يەنى ئۇلار شۇنداق تەلتۆكۈس ھالاك قىلىندىكى، ئۇلاردىن كۆرگىلى ۋە ئاۋازىنى ئاڭلىغىلى ھېچ ئەھەدى قالمىدى)
