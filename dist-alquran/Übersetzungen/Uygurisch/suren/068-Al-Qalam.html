بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
نٓ ۚ وَٱلْقَلَمِ وَمَا يَسْطُرُونَ ﴿١﴾
نۇن. قەلەم بىلەن ۋە ئۇلار يازغان نەرسىلەر بىلەن قەسەمكى،
مَآ أَنتَ بِنِعْمَةِ رَبِّكَ بِمَجْنُونٍ ﴿٢﴾
(ئى مۇھەممەد!) سەن پەرۋەردىگارىڭنىڭ نېمىتى بىلەن، (مۇشرىكلار ئېيتقاندەك) مەجنۇن ئەمەسسەن
وَإِنَّ لَكَ لَأَجْرًا غَيْرَ مَمْنُونٍ ﴿٣﴾
مۇقەررەركى، سەن ئەلۋەتتە ئۆكسىمەس ساۋابقا ئېرىشىسەن
وَإِنَّكَ لَعَلَىٰ خُلُقٍ عَظِيمٍ ﴿٤﴾
(ئى مۇھەممەد!) سەن ئىشەنچلىك ھالدا بۈيۈك ئەخلاققا ئىگىسەن
فَسَتُبْصِرُ وَيُبْصِرُونَ ﴿٥﴾
كەلگۈسىدە قايسىڭلارنىڭ مەجنۇن ئىكەنلىكىنى سەن كۆرۈسەن، ئۇلارمۇ كۆرىدۇ
بِأَييِّكُمُ ٱلْمَفْتُونُ ﴿٦﴾
كەلگۈسىدە قايسىڭلارنىڭ مەجنۇن ئىكەنلىكىنى سەن كۆرۈسەن، ئۇلارمۇ كۆرىدۇ
إِنَّ رَبَّكَ هُوَ أَعْلَمُ بِمَن ضَلَّ عَن سَبِيلِهِۦ وَهُوَ أَعْلَمُ بِٱلْمُهْتَدِينَ ﴿٧﴾
شۈبھىسىزكى، سېنىڭ پەرۋەردىگارىڭ ئۆزىنىڭ يولىدىن ئازغانلارنى ئوبدان بىلىدۇ، ھىدايەت تاپقۇچىلارنىمۇ ئوبدان بىلىدۇ
فَلَا تُطِعِ ٱلْمُكَذِّبِينَ ﴿٨﴾
ئىنكار قىلغۇچىلارغا ئىتائەت قىلمىغىن
وَدُّوا۟ لَوْ تُدْهِنُ فَيُدْهِنُونَ ﴿٩﴾
ئۇلار سېنىڭ ئۆزلىرى بىلەن مادارا قىلىشىڭنى، شۇنىڭ بىلەن ئۆزلىرىنىڭ (سەن بىلەن) مادارا قىلىشىنى ئارزۇ قىلىدۇ
وَلَا تُطِعْ كُلَّ حَلَّافٍ مَّهِينٍ ﴿١٠﴾
قەسەمخور، پەس، غەيۋەتخور، سۇخەنچى، بېخىل، ھەددىدىن ئاشقۇچى، گۇناھكار، قوپال، ئۇنىڭ ئۈستىگە ھارامدىن بولغان ئادەم (يەنى ۋەلىد ئىبن مۇغىرە) گە ئىتائەت قىلمىغىن
هَمَّازٍ مَّشَّآءٍۭ بِنَمِيمٍ ﴿١١﴾
قەسەمخور، پەس، غەيۋەتخور، سۇخەنچى، بېخىل، ھەددىدىن ئاشقۇچى، گۇناھكار، قوپال، ئۇنىڭ ئۈستىگە ھارامدىن بولغان ئادەم (يەنى ۋەلىد ئىبن مۇغىرە) گە ئىتائەت قىلمىغىن
مَّنَّاعٍ لِّلْخَيْرِ مُعْتَدٍ أَثِيمٍ ﴿١٢﴾
قەسەمخور، پەس، غەيۋەتخور، سۇخەنچى، بېخىل، ھەددىدىن ئاشقۇچى، گۇناھكار، قوپال، ئۇنىڭ ئۈستىگە ھارامدىن بولغان ئادەم (يەنى ۋەلىد ئىبن مۇغىرە) گە ئىتائەت قىلمىغىن
عُتُلٍّۭ بَعْدَ ذَٰلِكَ زَنِيمٍ ﴿١٣﴾
قەسەمخور، پەس، غەيۋەتخور، سۇخەنچى، بېخىل، ھەددىدىن ئاشقۇچى، گۇناھكار، قوپال، ئۇنىڭ ئۈستىگە ھارامدىن بولغان ئادەم (يەنى ۋەلىد ئىبن مۇغىرە) گە ئىتائەت قىلمىغىن
أَن كَانَ ذَا مَالٍ وَبَنِينَ ﴿١٤﴾
بىزنىڭ ئايەتلىرىمىز ئۇنىڭغا ئوقۇپ بېرىلسە، ئۇنىڭ مال - مۈلكى ۋە ئوغۇللىرى بولغانلىقى ئۈچۈن، «(بۇ) بۇرۇنقىلارنىڭ ھېكايىلىرىدۇر»، دەيدۇ
إِذَا تُتْلَىٰ عَلَيْهِ ءَايَـٰتُنَا قَالَ أَسَـٰطِيرُ ٱلْأَوَّلِينَ ﴿١٥﴾
بىزنىڭ ئايەتلىرىمىز ئۇنىڭغا ئوقۇپ بېرىلسە، ئۇنىڭ مال - مۈلكى ۋە ئوغۇللىرى بولغانلىقى ئۈچۈن، «(بۇ) بۇرۇنقىلارنىڭ ھېكايىلىرىدۇر»، دەيدۇ
سَنَسِمُهُۥ عَلَى ٱلْخُرْطُومِ ﴿١٦﴾
ئۇنىڭ بۇرنىنى داغلاپ بەلگە سالىمىز
إِنَّا بَلَوْنَـٰهُمْ كَمَا بَلَوْنَآ أَصْحَـٰبَ ٱلْجَنَّةِ إِذْ أَقْسَمُوا۟ لَيَصْرِمُنَّهَا مُصْبِحِينَ ﴿١٧﴾
شۈبھىسىزكى، بىز ئۇلارنى (يەنى مەككە ئاھالىسىنى) مەلۇم باغنىڭ ئىگىلىرىنى سىنىغاندەك (قەھەتچىلىك، ئاچارچىلىققا دۇچار قىلىش بىلەن) سىنىدۇق. ئۇلار (يەنى باغنىڭ ئىگىلىرى) ئۇنىڭ مېۋىلىرىنى (كەمبەغەللەر ئۇقمىسۇن، ئۇلارغا ھېچ نەرسە بەرمەيلى دېيىشىپ) سەھەردىلا ئۈزۈۋېلىشقا قەسەم ئىچىشكەن ئىدى
وَلَا يَسْتَثْنُونَ ﴿١٨﴾
ئۇلار خۇدا خالىسا دېمىگەن ئىدى
فَطَافَ عَلَيْهَا طَآئِفٌ مِّن رَّبِّكَ وَهُمْ نَآئِمُونَ ﴿١٩﴾
ئۇلار ئۇخلاۋاتقاندا، باغقا پەرۋەردىگارىڭ تەرىپىدىن ئازاب (يەنى يانغىن) نازىل بولۇپ، باغ (كۆيۈپ) قاپقارا كۈلدەك بولۇپ قالدى
فَأَصْبَحَتْ كَٱلصَّرِيمِ ﴿٢٠﴾
ئۇلار ئۇخلاۋاتقاندا، باغقا پەرۋەردىگارىڭ تەرىپىدىن ئازاب (يەنى يانغىن) نازىل بولۇپ، باغ (كۆيۈپ) قاپقارا كۈلدەك بولۇپ قالدى
فَتَنَادَوْا۟ مُصْبِحِينَ ﴿٢١﴾
ئۇلار: «ئەگەر (مېۋىلەرنى) ئۈزمەكچى بولساڭلار باغقا بېرىڭلار» دەپ، ئەتىگەندە بىر - بىرىنى توۋلاشتى
أَنِ ٱغْدُوا۟ عَلَىٰ حَرْثِكُمْ إِن كُنتُمْ صَـٰرِمِينَ ﴿٢٢﴾
ئۇلار: «ئەگەر (مېۋىلەرنى) ئۈزمەكچى بولساڭلار باغقا بېرىڭلار» دەپ، ئەتىگەندە بىر - بىرىنى توۋلاشتى
فَٱنطَلَقُوا۟ وَهُمْ يَتَخَـٰفَتُونَ ﴿٢٣﴾
ئۇلار ماڭدى، يولدا كېتىۋېتىپ: «بۈگۈن باغقا ھەرگىز بىر مىسكىنمۇ كىرمىسۇن» دېيىشىپ پىچىرلاشتى
أَن لَّا يَدْخُلَنَّهَا ٱلْيَوْمَ عَلَيْكُم مِّسْكِينٌ ﴿٢٤﴾
ئۇلار ماڭدى، يولدا كېتىۋېتىپ: «بۈگۈن باغقا ھەرگىز بىر مىسكىنمۇ كىرمىسۇن» دېيىشىپ پىچىرلاشتى
وَغَدَوْا۟ عَلَىٰ حَرْدٍ قَـٰدِرِينَ ﴿٢٥﴾
ئۇلار: «(مىسكىنلەرنى باغقا كىرىشتىن) توسىيالايمىز دېگەن ئوي بىلەن، ئەتىگەندە تۇرۇشتى
فَلَمَّا رَأَوْهَا قَالُوٓا۟ إِنَّا لَضَآلُّونَ ﴿٢٦﴾
ئۇلار باغنىڭ (كۆيۈپ قاپقارا بولۇپ كەتكەنلىكىنى) كۆرۈپ: «بىز ئىشەنچلىك ھالدا ئېزىپ قېلىپتۇق، ياق بىز (باغنىڭ مېۋىسىدىن) قۇرۇق قېلىپتۇق» دېدى
بَلْ نَحْنُ مَحْرُومُونَ ﴿٢٧﴾
ئۇلار باغنىڭ (كۆيۈپ قاپقارا بولۇپ كەتكەنلىكىنى) كۆرۈپ: «بىز ئىشەنچلىك ھالدا ئېزىپ قېلىپتۇق، ياق بىز (باغنىڭ مېۋىسىدىن) قۇرۇق قېلىپتۇق» دېدى
قَالَ أَوْسَطُهُمْ أَلَمْ أَقُل لَّكُمْ لَوْلَا تُسَبِّحُونَ ﴿٢٨﴾
ئۇلارنىڭ ياخشىسى (يەنى ئەڭ ئەقىللىقى): «سىلەر نېمىشقا اﷲ قا تەۋبە قىلىپ تەسبىھ ئېيتمايسىلەر دېمىگەنمىدىم؟ دېدى
قَالُوا۟ سُبْحَـٰنَ رَبِّنَآ إِنَّا كُنَّا ظَـٰلِمِينَ ﴿٢٩﴾
ئۇلار: «پەرۋەردىگارىمىز! سېنى پاك ئېتىقاد قىلىمىزكى، بىز (مىسكىنلەرنىڭ ھەققىنى مەنئى قىلىش بىلەن) ئىشەنچلىك ھالدا زۇلۇم قىلغۇچىلاردىن بولدۇق» دېدى
فَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَلَـٰوَمُونَ ﴿٣٠﴾
شۇنىڭ بىلەن ئۇلار بىر - بىرگە قارىشىپ ئۆز ئارا مالامەت قىلىشتى
قَالُوا۟ يَـٰوَيْلَنَآ إِنَّا كُنَّا طَـٰغِينَ ﴿٣١﴾
ئۇلار دېيىشتى: «ۋاي بىزگە! بىز ئىشەنچلىك ھالدا ھەددىمىزدىن ئېشىپتۇق
عَسَىٰ رَبُّنَآ أَن يُبْدِلَنَا خَيْرًا مِّنْهَآ إِنَّآ إِلَىٰ رَبِّنَا رَٰغِبُونَ ﴿٣٢﴾
ئۇمىدكى، پەرۋەردىگارىمىز بىزگە بۇنىڭدىن ياخشىراق باغنى بەرگەي، بىز ئەلۋەتتە پەرۋەردىگارىمىزدىن ئۈمىد قىلغۇچىلارمىز»
كَذَٰلِكَ ٱلْعَذَابُ ۖ وَلَعَذَابُ ٱلْـَٔاخِرَةِ أَكْبَرُ ۚ لَوْ كَانُوا۟ يَعْلَمُونَ ﴿٣٣﴾
ئازاب مانا مۇشۇنداقتۇر (يەنى قۇرەيشكىمۇ باغ ئىگىلىرىگە چۈشكەن ئازاب چۈشىدۇ). ئەگەر ئۇلار بىلسە، ئاخىرەتنىڭ ئازابى تېخىمۇ قاتتىقتۇر
إِنَّ لِلْمُتَّقِينَ عِندَ رَبِّهِمْ جَنَّـٰتِ ٱلنَّعِيمِ ﴿٣٤﴾
تەقۋادارلار پەرۋەردىگارىڭنىڭ دەرگاھىدا ئىشەنچلىك ھالدا نازۇنېمەتلىك مەڭگۈلۈك باغلارلەردىن بەھرىمەن بولىدۇ
أَفَنَجْعَلُ ٱلْمُسْلِمِينَ كَٱلْمُجْرِمِينَ ﴿٣٥﴾
بىز مۇسۇلمانلارنى ھەقىقەتنى رەت قىلغۇچىلار بىلەن باراۋەر قىلامدۇق؟
مَا لَكُمْ كَيْفَ تَحْكُمُونَ ﴿٣٦﴾
سىلەرگە نېمە بولدى؟ قانداقچە مۇنداق ھۆكۈم چىقىرىسىلەر!
أَمْ لَكُمْ كِتَـٰبٌ فِيهِ تَدْرُسُونَ ﴿٣٧﴾
ياكى سىلەردە (ئاسماندىن نازىل بولغان) بىر مۇقەددەس كىتاب-ۋەھىي بولۇپ، ئۇنىڭدىن سىلەرگە (قىيامەت كۈنى) خالىغىنىڭلار بار، دېگەننى ئوقۇدۇڭلارمۇ؟
إِنَّ لَكُمْ فِيهِ لَمَا تَخَيَّرُونَ ﴿٣٨﴾
ياكى سىلەردە (ئاسماندىن نازىل بولغان) بىر مۇقەددەس كىتاب-ۋەھىي بولۇپ، ئۇنىڭدىن سىلەرگە (قىيامەت كۈنى) خالىغىنىڭلار بار، دېگەننى ئوقۇدۇڭلارمۇ؟
أَمْ لَكُمْ أَيْمَـٰنٌ عَلَيْنَا بَـٰلِغَةٌ إِلَىٰ يَوْمِ ٱلْقِيَـٰمَةِ ۙ إِنَّ لَكُمْ لَمَا تَحْكُمُونَ ﴿٣٩﴾
ياكى سىلەر ھۆكۈم قىلغان نەرسەڭلارنىڭ بولۇشىغا بىز سىلەر بىلەن قىيامەت كۈنى ئاندىن ۋاقتى توشىدىغان مەھكەم بىر ئەھدى تۈزۈشكەنمۇ؟
سَلْهُمْ أَيُّهُم بِذَٰلِكَ زَعِيمٌ ﴿٤٠﴾
ئۇلاردىن سوراپ باققىنكى، ئۇلارنىڭ ئارىسىدا قايسىسى يۇقىرىقى ئىشلارغا كېپىللىك قىلالايدۇ؟
أَمْ لَهُمْ شُرَكَآءُ فَلْيَأْتُوا۟ بِشُرَكَآئِهِمْ إِن كَانُوا۟ صَـٰدِقِينَ ﴿٤١﴾
ياكى ئۇلارنىڭ (يۇقىرىقى ئىشلارغا كېپىللىك قىلىدىغان شېرىكلىرى) بارمۇ؟ ئەگەر ئۇلار (دەۋاسىدا) راستچىل بولسا، شېرىكلىرىنى كەلتۈرۈپ باقسۇن
يَوْمَ يُكْشَفُ عَن سَاقٍ وَيُدْعَوْنَ إِلَى ٱلسُّجُودِ فَلَا يَسْتَطِيعُونَ ﴿٤٢﴾
ئۇ كۈندە (يەنى قىيامەت كۈنىدە) ئىش قىيىنلىشىدۇ. ئۇلار (ئىمانىنى سىناش ئۈچۈن) سەجدە قىلىشقا چاقىرىلىدۇ، ئۇلار سەجدە قىلالمايدۇ
خَـٰشِعَةً أَبْصَـٰرُهُمْ تَرْهَقُهُمْ ذِلَّةٌ ۖ وَقَدْ كَانُوا۟ يُدْعَوْنَ إِلَى ٱلسُّجُودِ وَهُمْ سَـٰلِمُونَ ﴿٤٣﴾
ئۇلار (قورقۇنچتىن) تىكىلىپ قارىيالمايدۇ، ئۇلارغا خارلىق يۈزلىنىدۇ، ئۇلار (دۇنيادا) ساغلام چاغلىرىدا سەجدىگە دەۋەت قىلىنغان ئىدى (ئۇلار سەجدىدىن باش تارتاتتى)
فَذَرْنِى وَمَن يُكَذِّبُ بِهَـٰذَا ٱلْحَدِيثِ ۖ سَنَسْتَدْرِجُهُم مِّنْ حَيْثُ لَا يَعْلَمُونَ ﴿٤٤﴾
بۇ قۇرئاننى ئىنكار قىلغانلارنى ماڭا قويۇۋەتكىن، ئۇلارنى تۇيدۇرماستىن، ئاستا - ئاستا ھالاكەتكە يۈزلەندۈرىمىز
وَأُمْلِى لَهُمْ ۚ إِنَّ كَيْدِى مَتِينٌ ﴿٤٥﴾
ئۇلارغا مۆھلەت بېرىمەن، شۈبھىسىزكى، مېنىڭ تەدبىرىم پۇختىدۇر
أَمْ تَسْـَٔلُهُمْ أَجْرًا فَهُم مِّن مَّغْرَمٍ مُّثْقَلُونَ ﴿٤٦﴾
ياكى ئۇلاردىن (دىنغا دەۋەت قىلغانلىقىڭغا) ھەق سوراپ، ئۇلارنىڭ (ئىقتىسادىي) يۈكىنى ئېغىرلاشتۇرۇۋەتتىڭمۇ؟
أَمْ عِندَهُمُ ٱلْغَيْبُ فَهُمْ يَكْتُبُونَ ﴿٤٧﴾
ياكى ئۇلارنىڭ يېنىدا غەيب (يەنى غەيبنى ئىچىگە ئالغان لەۋھۇلمەھپۇز) بار بولۇپ، ئۇلار (ئېيتىدىغانلىرىنى شۇنىڭغا) يازامدۇ؟
فَٱصْبِرْ لِحُكْمِ رَبِّكَ وَلَا تَكُن كَصَاحِبِ ٱلْحُوتِ إِذْ نَادَىٰ وَهُوَ مَكْظُومٌ ﴿٤٨﴾
پەرۋەردىگارىڭنىڭ ھۆكمىگە سەۋر قىلغىن، (ئالدىراڭغۇلۇقتا، زېرىككەكلىكتە) بېلىقنىڭ ساھىبىگە (يەنى يۇنۇسقا) ئوخشاش بولمىغىن، ئۆز ۋاقتىدا ئۇ (بېلىقنىڭ قارنىدا تۇرۇپ) غەمكىن ھالدا پەرۋەردىگارىغا ئىلتىجا قىلدى
لَّوْلَآ أَن تَدَٰرَكَهُۥ نِعْمَةٌ مِّن رَّبِّهِۦ لَنُبِذَ بِٱلْعَرَآءِ وَهُوَ مَذْمُومٌ ﴿٤٩﴾
ئەگەر ئۇ پەرۋەردىگارىنىڭ نېمىتىگە ئېرىشمىسە ئىدى، ئەلۋەتتە، ئۇ (قىلمىشىغا) مالامەت قىلىنغان ھالدا (دەرەخمۇ يوق، تاغمۇ يوق) بىر دالىغا تاشلانغان بولاتتى
فَٱجْتَبَـٰهُ رَبُّهُۥ فَجَعَلَهُۥ مِنَ ٱلصَّـٰلِحِينَ ﴿٥٠﴾
پەرۋەردىگارى ئۇنى (خۇدانىڭ ئەلچىسىلىككە) تاللىدى، ئۇنى ياخشىلاردىن قىلدى
وَإِن يَكَادُ ٱلَّذِينَ كَفَرُوا۟ لَيُزْلِقُونَكَ بِأَبْصَـٰرِهِمْ لَمَّا سَمِعُوا۟ ٱلذِّكْرَ وَيَقُولُونَ إِنَّهُۥ لَمَجْنُونٌ ﴿٥١﴾
ھەقىقەتنى رەت قىلغۇچىلار قۇرئاننى ئاڭلىغان چاغلىرىدا (ساڭا بولغان دۈشمەنلىكىنىڭ قاتتىقلىقىدىن يامان) كۆزلىرى بىلەن سېنى يىقىتىۋېتىشكە تاس قالىدۇ. سېنى: «شۈبھىسىز بىر مەجنۇن» دېيىشىدۇ
وَمَا هُوَ إِلَّا ذِكْرٌ لِّلْعَـٰلَمِينَ ﴿٥٢﴾
قۇرئان پەقەت تامام ئەھلى جاھان ئۈچۈن ۋەز - نەسىھەتتۇر
