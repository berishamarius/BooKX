Al-Anbya · The Prophets
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
ٱقْتَرَبَ لِلنَّاسِ حِسَابُهُمْ وَهُمْ فِى غَفْلَةٍ مُّعْرِضُونَ ﴿١﴾
كىشىلەرگە ئۇلارنىڭ (ئەمەللىرىدىن) ھېساب ئېلىنىدىغان ۋاقىت (يەنى قىيامەت) توغرۇلۇق ئويلىنىشتىن) يۈز ئۆرۈمەكتىدۇر
مَا يَأْتِيهِم مِّن ذِكْرٍ مِّن رَّبِّهِم مُّحْدَثٍ إِلَّا ٱسْتَمَعُوهُ وَهُمْ يَلْعَبُونَ ﴿٢﴾
ئۇلارغا پەرۋەردىگارى تەرىپىدىن (قۇرئاندا) يېڭى ۋەز - نەسىھەت نازىل بولسىلا، ئۇلار ئۇنىڭغا مەسخىرە قىلغان ھالدا قۇلاق سېلىشىدۇ
لَاهِيَةً قُلُوبُهُمْ ۗ وَأَسَرُّوا۟ ٱلنَّجْوَى ٱلَّذِينَ ظَلَمُوا۟ هَلْ هَـٰذَآ إِلَّا بَشَرٌ مِّثْلُكُمْ ۖ أَفَتَأْتُونَ ٱلسِّحْرَ وَأَنتُمْ تُبْصِرُونَ ﴿٣﴾
ئۇلارنىڭ دىللىرى (اﷲ نىڭ كالامىدىن) غەپلەتتىدۇر، ھەقىقەتنى رەت قىلغۇچىلار ئۆز ئارا: «بۇ (يەنى مۇھەممەد ئەلەيھىسسالام) پەقەت سىلەرگە ئوخشاش (ئاددىي) بىر ئىنسان، سىلەر سېھىرنى كۆرۈپ تۇرۇپ ئۇنى قوبۇل قىلامسىلەر؟» دەپ يوشۇرۇن سۆزلىشىدۇ
قَالَ رَبِّى يَعْلَمُ ٱلْقَوْلَ فِى ٱلسَّمَآءِ وَٱلْأَرْضِ ۖ وَهُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٤﴾
(خۇدانىڭ ئەلچىسى) ئېيتتى: «پەرۋەردىگارىم ئاسمان، زېمىندا ئېيتىلغان سۆزلەرنى بىلىپ تۇرىدۇر، اﷲ (سۆزلىرىڭلارنى) ئاڭلاپ تۇرغۇچىدۇر، (ئەھۋالىڭلارنى) بىلىپ تۇرغۇچىدۇر»
بَلْ قَالُوٓا۟ أَضْغَـٰثُ أَحْلَـٰمٍۭ بَلِ ٱفْتَرَىٰهُ بَلْ هُوَ شَاعِرٌ فَلْيَأْتِنَا بِـَٔايَةٍ كَمَآ أُرْسِلَ ٱلْأَوَّلُونَ ﴿٥﴾
(ھەقىقەتنى رەت قىلغۇچىلار قۇرئان سېھرىدۇر دەپلا قالماستىن) بەلكى ئۇ شائىردۇر، ئىلگىرىكى خۇدانىڭ ئەلچى-رەسۇللىرىگە ئوخشاش بىزگە بىرەر مۆجىزە كەلتۈرسۇن» دېيىشتى
مَآ ءَامَنَتْ قَبْلَهُم مِّن قَرْيَةٍ أَهْلَكْنَـٰهَآ ۖ أَفَهُمْ يُؤْمِنُونَ ﴿٦﴾
ئۇلاردىن ئىلگىرى، بىز ھالاك قىلغان شەھەرلەردىن (يەنى ئاھالىسىدىن) ھېچقايسىسى ئىمان ئېيتمىغان تۇرسا، (مۆجىزىلەرنى كۆرسەتسەڭ) ئۇلار ئىمان ئېيتامدۇ؟
وَمَآ أَرْسَلْنَا قَبْلَكَ إِلَّا رِجَالًا نُّوحِىٓ إِلَيْهِمْ ۖ فَسْـَٔلُوٓا۟ أَهْلَ ٱلذِّكْرِ إِن كُنتُمْ لَا تَعْلَمُونَ ﴿٧﴾
(ئى مۇھەممەد!) سەندىن ئىلگىرى - (خۇدانىڭ ئەلچى پەرىشتىلىرىنى ئەمەس) پەقەت ئىنسانلارنىلا خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق، ئەگەر (بۇنى) بىلمىسەڭلار، ئەھلى ئىلىمدىن (يەنى تەۋرات، ئىنجىللارنى بىلىدىغانلاردىن) سوراڭلار
وَمَا جَعَلْنَـٰهُمْ جَسَدًا لَّا يَأْكُلُونَ ٱلطَّعَامَ وَمَا كَانُوا۟ خَـٰلِدِينَ ﴿٨﴾
خۇدانىڭ ئەلچى-رەسۇللىرىنى تاماق يېمەيدىغان جەسەت قىلغىنىمىز يوق (يەنى ئۇلار بارلىق كىشىلەرگە ئوخشاش يەيدۇ، ئىچىدۇ) ھەمدە ئۇلار (دۇنيادا ئۆلمەي) مەڭگۈ قالىدىغان بولغىنى يوق (يەنى ئۇلارمۇ بارلىق كىشىلەرگە ئوخشاش ۋاپات بولىدۇ)
ثُمَّ صَدَقْنَـٰهُمُ ٱلْوَعْدَ فَأَنجَيْنَـٰهُمْ وَمَن نَّشَآءُ وَأَهْلَكْنَا ٱلْمُسْرِفِينَ ﴿٩﴾
ئاندىن ئۇلارغا (ئۇلارغا ياردەم بېرىش، ئۇلارنى ئىنكار قىلغانلارنى ھالاك قىلىشتىن ئىبارەت ۋەدىمىزنى راست قىلىپ بەردۇق، ئۇلارنى ۋە بىز خالىغانلارنى قۇتقۇزدۇق، (كۇفرىدا ۋە گۇمراھلىقتا) ھەددىدىن ئاشقۇچىلارنى ھالاك قىلدۇق
لَقَدْ أَنزَلْنَآ إِلَيْكُمْ كِتَـٰبًا فِيهِ ذِكْرُكُمْ ۖ أَفَلَا تَعْقِلُونَ ﴿١٠﴾
سىلەرنىڭ شان - شەرىپىڭلارنى ئۆز ئىچىگە ئالغان مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) سىلەرگە ئىشەنچلىك ھالدا نازىل قىلىپ بەردۇق. (بۇنى) چۈشەنمەمسىلەر؟
وَكَمْ قَصَمْنَا مِن قَرْيَةٍ كَانَتْ ظَالِمَةً وَأَنشَأْنَا بَعْدَهَا قَوْمًا ءَاخَرِينَ ﴿١١﴾
بىز (ئاھالىسى) كۇففار بولغان نۇرغۇن شەھەرلەرنى ھالاك قىلدۇق. ئۇلاردىن كېيىن (ئورنىغا) باشقا قەۋمنى پەيدا قىلدۇق
فَلَمَّآ أَحَسُّوا۟ بَأْسَنَآ إِذَا هُم مِّنْهَا يَرْكُضُونَ ﴿١٢﴾
ئۇلار بىزنىڭ ئازابىمىزنىڭ (كەلگەنلىكىنى) ھېس قىلغان چاغدا ئۇشتۇمتۇت قاچىدۇ
لَا تَرْكُضُوا۟ وَٱرْجِعُوٓا۟ إِلَىٰ مَآ أُتْرِفْتُمْ فِيهِ وَمَسَـٰكِنِكُمْ لَعَلَّكُمْ تُسْـَٔلُونَ ﴿١٣﴾
(خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارغا مەسخىرە قىلىپ ئېيتتى) قاچماڭلار، باياشات تۇرمۇشۇڭلارغا، تۇرالغۇ جايلىرىڭلارغا قايتىڭلار، سىلەردىن (مۇھىم ئىشلاردا مەسلىھەت) سورىلىشى مۇمكىن»
قَالُوا۟ يَـٰوَيْلَنَآ إِنَّا كُنَّا ظَـٰلِمِينَ ﴿١٤﴾
ئۇلار: «ۋاي ئىسىت! ئىشەنچلىك ھالدا بىز زالىم بولۇپتۇق» دېيىشتى
فَمَا زَالَت تِّلْكَ دَعْوَىٰهُمْ حَتَّىٰ جَعَلْنَـٰهُمْ حَصِيدًا خَـٰمِدِينَ ﴿١٥﴾
ئۇلارنى ئورۇلغان زىرائەتتەك ھالاك قىلىپ، جانلىرىنى ئالغىنىمىزغا قەدەر، ئۇلار يۇقىرىقى پەريادىنى تەكرارلاپ تۇردى
وَمَا خَلَقْنَا ٱلسَّمَآءَ وَٱلْأَرْضَ وَمَا بَيْنَهُمَا لَـٰعِبِينَ ﴿١٦﴾
بىز ئاسماننى، زېمىننى ۋە ئۇلارنىڭ ئارىسىدىكى نەرسىلەرنى ئويناپ ياراتقىنىمىز يوق (يەنى ئۇلارنى بىكار ياراتماستىن، قۇدرىتىمىزنىڭ ئالامەتلىرى سۈپىتىدە ياراتتۇق)
لَوْ أَرَدْنَآ أَن نَّتَّخِذَ لَهْوًا لَّٱتَّخَذْنَـٰهُ مِن لَّدُنَّآ إِن كُنَّا فَـٰعِلِينَ ﴿١٧﴾
ئەگەر بىز ئىچ پۇشۇقى قىلماقچى بولساق، قېشىمىزدىكى نەرسىلەر بىلەن قىلاتتۇق، ئەگەر (شۇنى) ئىرادە قىلغان بولساق (لېكىن بۇ ئۇلۇغ زاتىمىزغا مۇھال بولغانلىقتىن بۇنداق قىلىشنى ئىرادە قىلمىدۇق)
بَلْ نَقْذِفُ بِٱلْحَقِّ عَلَى ٱلْبَـٰطِلِ فَيَدْمَغُهُۥ فَإِذَا هُوَ زَاهِقٌ ۚ وَلَكُمُ ٱلْوَيْلُ مِمَّا تَصِفُونَ ﴿١٨﴾
بىز ھەق ئارقىلىق باتىلغا ھۇجۇم قىلىمىز، ھەق باتىلنى يوقىتىدۇ، باتىل ناگاھان يوقىلىدۇ، (ئى كۇففارلار جامائەسى! اﷲ نى بالىسى بار، خوتۇنى بار دەپ) سۈپەتلىگەن سۆزۈڭلاردىن سىلەرگە ۋاي!
وَلَهُۥ مَن فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۚ وَمَنْ عِندَهُۥ لَا يَسْتَكْبِرُونَ عَنْ عِبَادَتِهِۦ وَلَا يَسْتَحْسِرُونَ ﴿١٩﴾
ئاسمانلاردىكى ۋە زېمىندىكىلەرنىڭ ھەممىسى اﷲ نىڭ ئىلكىدىدۇر، اﷲ نىڭ دەرگاھىدىكىلەر (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى) اﷲ نىڭ ئىبادىتىدىن تەكەببۇرلۇق قىلىپ باش تارتمايدۇ ۋە ھېرىپمۇ قالمايدۇ
يُسَبِّحُونَ ٱلَّيْلَ وَٱلنَّهَارَ لَا يَفْتُرُونَ ﴿٢٠﴾
ئۇلار كېچە - كۈندۈز تەسبىھ ئېيتىدۇ، بوشاشمايدۇ
أَمِ ٱتَّخَذُوٓا۟ ءَالِهَةً مِّنَ ٱلْأَرْضِ هُمْ يُنشِرُونَ ﴿٢١﴾
ئۇلار (يەنى مۇشرىكلار) زېمىندىكى نەرسىلەرنى ئىلاھ قىلىۋالدى، ئۇلار (يەنى بۇ نەرسىلەر) ئۆلۈكلەرنى تىرىلدۈرەلەمدۇ؟
لَوْ كَانَ فِيهِمَآ ءَالِهَةٌ إِلَّا ٱللَّهُ لَفَسَدَتَا ۚ فَسُبْحَـٰنَ ٱللَّهِ رَبِّ ٱلْعَرْشِ عَمَّا يَصِفُونَ ﴿٢٢﴾
ئەگەر ئاسمان - زېمىندا اﷲ تىن باشقا ئىلاھلار بولسا ئىدى، (كائىناتنىڭ تەرتىپى) ئەلۋەتتە بۇزۇلاتتى، ئەرشنىڭ پەرۋەردىگارى اﷲ ئۇلارنىڭ سۈپەتلىگەن نەرسىلىرىدىن پاكتۇر
لَا يُسْـَٔلُ عَمَّا يَفْعَلُ وَهُمْ يُسْـَٔلُونَ ﴿٢٣﴾
اﷲ نىڭ قىلغانلىرىدىن سوئال - سوراق قىلىنمايدۇ، ئۇلاردىن سوئال - سوراق قىلىنىدۇ
أَمِ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ ءَالِهَةً ۖ قُلْ هَاتُوا۟ بُرْهَـٰنَكُمْ ۖ هَـٰذَا ذِكْرُ مَن مَّعِىَ وَذِكْرُ مَن قَبْلِى ۗ بَلْ أَكْثَرُهُمْ لَا يَعْلَمُونَ ٱلْحَقَّ ۖ فَهُم مُّعْرِضُونَ ﴿٢٤﴾
ئۇلار اﷲ نى قويۇپ، باشقا مەبۇدلارنى ئىلاھ قىلىۋالدىمۇ؟ ئېيتقىنكى: «(بۇنىڭغا) دەلىلىڭلارنى كۆرسىتىڭلار، مانا بۇ مەن بىلەن بىللە بولغانلارنىڭ مۇقەددەس كىتاب-ۋەھىيى (يەنى قۇرئان) دۇر ۋە مەندىن ئىلگىرىكىلەرنىڭ (تەۋرات، ئىنجىل قاتارلىق) مۇقەددەس كىتاب-ۋەھىيىدۇر، ئۇلارنىڭ تولىسى ھەقنى ئۇقمايدۇ، ئۇلار (تەۋھىد) تىن يۈز ئۆرۈگۈچىلەردۇر»
وَمَآ أَرْسَلْنَا مِن قَبْلِكَ مِن رَّسُولٍ إِلَّا نُوحِىٓ إِلَيْهِ أَنَّهُۥ لَآ إِلَـٰهَ إِلَّآ أَنَا۠ فَٱعْبُدُونِ ﴿٢٥﴾
(ئى مۇھەممەد!) سەندىن ئىلگىرى ئەۋەتىلگەن خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ ھەممىسىگە: «مەندىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، ماڭىلا ئىبادەت قىلىڭلار» دەپ ۋەھيى قىلدۇق
وَقَالُوا۟ ٱتَّخَذَ ٱلرَّحْمَـٰنُ وَلَدًا ۗ سُبْحَـٰنَهُۥ ۚ بَلْ عِبَادٌ مُّكْرَمُونَ ﴿٢٦﴾
ئۇلار (يەنى مۇشرىكلار): «اﷲ نىڭ (خۇدانىڭ ئەلچى پەرىشتىلىرىدىن) بالىسى بار» دېيىشتى، ئۇنداق ئەمەس. اﷲ (بۇنىڭدىن) پاكتۇر، (خۇدانىڭ ئەلچى پەرىشتىلىرى اﷲ نىڭ) ھۆرمەتلىك بەندىلىرىدۇر
لَا يَسْبِقُونَهُۥ بِٱلْقَوْلِ وَهُم بِأَمْرِهِۦ يَعْمَلُونَ ﴿٢٧﴾
ئۇلار اﷲ قا ئالدى بىلەن سۆز قىلىشقا پېتىنىلمايدۇ، ئۇلار اﷲ نىڭ ئەمرى بويىچە ئىش قىلىدۇ
يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلَا يَشْفَعُونَ إِلَّا لِمَنِ ٱرْتَضَىٰ وَهُم مِّنْ خَشْيَتِهِۦ مُشْفِقُونَ ﴿٢٨﴾
اﷲ ئۇلارنىڭ قىلغانلىرىنى ۋە قىلماقچى بولغانلىرىنى بىلىپ تۇرىدۇ، ئۇلار اﷲ رازى بولغانلارغىلا شاپائەت قىلىدۇ، اﷲ نىڭ ھەيۋىسىدىن تىترەپ تۇرىدۇ
۞ وَمَن يَقُلْ مِنْهُمْ إِنِّىٓ إِلَـٰهٌ مِّن دُونِهِۦ فَذَٰلِكَ نَجْزِيهِ جَهَنَّمَ ۚ كَذَٰلِكَ نَجْزِى ٱلظَّـٰلِمِينَ ﴿٢٩﴾
ئۇلاردىن كىمكى، مەن اﷲ تىن باشقا ئىلاھمەن، دەيدىكەن، ئۇنى دوزاخ بىلەن جازالايمىز، زالىملارنى شۇنداق جازالايمىز
أَوَلَمْ يَرَ ٱلَّذِينَ كَفَرُوٓا۟ أَنَّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ كَانَتَا رَتْقًا فَفَتَقْنَـٰهُمَا ۖ وَجَعَلْنَا مِنَ ٱلْمَآءِ كُلَّ شَىْءٍ حَىٍّ ۖ أَفَلَا يُؤْمِنُونَ ﴿٣٠﴾
ھەقىقەتنى رەت قىلغۇچىلار بىلەمدۇكى، ئاسمانلار بىلەن زېمىن بىر - بېرىگە تۇتاش ئىدى، ئۇلارنى ئايرىۋەتتۇق، ھەممە جانلىق مەۋجۇداتنى سۇدىن ياراتتۇق، ئۇلار (اﷲ نىڭ قۇدرىتىگە) ئىشىنەمدۇ؟
وَجَعَلْنَا فِى ٱلْأَرْضِ رَوَٰسِىَ أَن تَمِيدَ بِهِمْ وَجَعَلْنَا فِيهَا فِجَاجًا سُبُلًا لَّعَلَّهُمْ يَهْتَدُونَ ﴿٣١﴾
يەر تەۋرەپ ئۇلارنىڭ خاتىرجەمسىز بولماسلىقى ئۈچۈن، بىز زېمىندا تاغلارنى ياراتتۇق، ئۇلارنى (سەپەردە) كۆزلىگەن مەقسەتلەرگە يېتىۋالسۇن دەپ، تاغلاردا كەڭ يوللارنى ياراتتۇق
وَجَعَلْنَا ٱلسَّمَآءَ سَقْفًا مَّحْفُوظًا ۖ وَهُمْ عَنْ ءَايَـٰتِهَا مُعْرِضُونَ ﴿٣٢﴾
ئاسماننى بىز بىخەتەر ئۆگزە قىلدۇق، ھالبۇكى، ئۇلار (يەنى مۇشرىكلار) ئاسماندىكى (كۈن، ئاي، يۇلتۇزلارغا ئوخشاش اﷲ نىڭ بارلىقىنى ۋە ئۇنىڭ قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلەردىن غەپلەتتىدۇر
وَهُوَ ٱلَّذِى خَلَقَ ٱلَّيْلَ وَٱلنَّهَارَ وَٱلشَّمْسَ وَٱلْقَمَرَ ۖ كُلٌّ فِى فَلَكٍ يَسْبَحُونَ ﴿٣٣﴾
اﷲ كېچىنى ۋە كۈندۈزنى، كۈننى ۋە ئاينى ياراتتى. ئۇلارنىڭ ھەر بىرى (ئۆزىگە خاس) پەلەكتە سەير قىلىدۇ
وَمَا جَعَلْنَا لِبَشَرٍ مِّن قَبْلِكَ ٱلْخُلْدَ ۖ أَفَإِي۟ن مِّتَّ فَهُمُ ٱلْخَـٰلِدُونَ ﴿٣٤﴾
(ئى مۇھەممەد!) سەندىن ئىلگىرى (ئىنسانلاردىن) ھېچ كىشىنى مەڭگۈ ياشاتقىنىمىز يوق، سەن ئۆلسەڭ، ئۇلار مەڭگۈ ياشامدۇ؟ (بۇنداق بولۇش ھەرگىزمۇ مەڭگۈ ياشاتقىنىمىز يوق، سەن ئۆلسەك، ئۇلار مەڭگۈ ياشامدۇ؟ (بۇنداق بولۇش ھەرگىزمۇ ئۇلارغا مۇيەسسەر بولمايدۇ، ئۇلارنىڭ ھەممىسى ئۆلىدۇ)
كُلُّ نَفْسٍ ذَآئِقَةُ ٱلْمَوْتِ ۗ وَنَبْلُوكُم بِٱلشَّرِّ وَٱلْخَيْرِ فِتْنَةً ۖ وَإِلَيْنَا تُرْجَعُونَ ﴿٣٥﴾
ھەر بىر جان ئۆلۈمنىڭ تەمىنى تېتىغۇچىدۇر. سىلەرنى بىز يامانلىققا مۇپتىلا قىلىش، ياخشىلىق بېرىش ئارقىلىق سىنايمىز سىلەر بىزنىڭ دەرگاھىمىزغا قايتىسىلەر (ئەمەلىڭلارغا يارىشا جازالايمىز)
وَإِذَا رَءَاكَ ٱلَّذِينَ كَفَرُوٓا۟ إِن يَتَّخِذُونَكَ إِلَّا هُزُوًا أَهَـٰذَا ٱلَّذِى يَذْكُرُ ءَالِهَتَكُمْ وَهُم بِذِكْرِ ٱلرَّحْمَـٰنِ هُمْ كَـٰفِرُونَ ﴿٣٦﴾
ھەقىقەتنى رەت قىلغۇچىلار سىنى كۆرگەن چاغدا: «سىلەرنىڭ مەبۇدلىرىڭلارنى ئەيىبلەيدىغان ئادەم مۇشۇمۇ؟» (دېيىشىپ) سېنى مەسخىرە قىلىشىدۇ، ھالبۇكى، ئۇلار مەرھەمەتلىك اﷲ تەرىپىدىن نازىل بولغان قۇرئاننى ئىنكار قىلىشىدۇ
خُلِقَ ٱلْإِنسَـٰنُ مِنْ عَجَلٍ ۚ سَأُو۟رِيكُمْ ءَايَـٰتِى فَلَا تَسْتَعْجِلُونِ ﴿٣٧﴾
ئىنسان ئالدىراڭغۇ يارىتىلغان، سىلەرگە ئالامەتلىرىمنى (يەنى ماڭا ئاسىيلىق قىلغانلارنى جازالايدىغانلىقىمنى) كۆرسىتىمەن. (ئۇنىڭ ۋاقتى سائىتىدىن بۇرۇن) مېنى ئالدىرىتىپ كەتمەڭلار
وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٣٨﴾
ئۇلار: «(سۆزۈڭلاردا) راستچىل بولساڭلار، بۇ ۋەدە (يەنى بىزگە قىلىنغان ئازابنىڭ يۈز بېرىدىغان ۋاقتى) قاچان؟» دېيىشىدۇ
لَوْ يَعْلَمُ ٱلَّذِينَ كَفَرُوا۟ حِينَ لَا يَكُفُّونَ عَن وُجُوهِهِمُ ٱلنَّارَ وَلَا عَن ظُهُورِهِمْ وَلَا هُمْ يُنصَرُونَ ﴿٣٩﴾
ئەگەر ھەقىقەتنى رەت قىلغۇچىلار بۇ چاغدا يۈزلىرىدىن، دۈمبىلىرىدىن ئازابنى دەپئى قىلالمايدىغانلىقلىرىنى ۋە ھېچقانداق ياردەمگە ئېرىشەلمەيدىغانلىقلىرىنى بىلسە ئىدى (ئازابنى بالدۇر يۈز بېرىشىنى تەلەپ قىلمايتتى)
بَلْ تَأْتِيهِم بَغْتَةً فَتَبْهَتُهُمْ فَلَا يَسْتَطِيعُونَ رَدَّهَا وَلَا هُمْ يُنظَرُونَ ﴿٤٠﴾
لېكىن قىيامەت ئۇلارغا تۇيۇقسىز كېلىدۇ - دە، ئۇلارنى ھەيرەتتە قالدۇرىدۇ، ئۇلار قىيامەتكە قارشىلىق كۆرسىتەلمەيدۇ، ئۇلارغا (تەۋبە قىلىۋېلىش ۋە ئۆزرە بايان قىلىش ئۈچۈن) مۆھلەتمۇ بېرىلمەيدۇ
وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍ مِّن قَبْلِكَ فَحَاقَ بِٱلَّذِينَ سَخِرُوا۟ مِنْهُم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿٤١﴾
سەندىن ئىلگىرى ئىشەنچلىك ھالدا نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرى مەسخىرە قىلىندى، ئۇلارنى مەسخىرە قىلغۇچىلارغا مەسخىرە قىلغاننىڭ جازاسى بولغان ئازاب نازىل بولدى
قُلْ مَن يَكْلَؤُكُم بِٱلَّيْلِ وَٱلنَّهَارِ مِنَ ٱلرَّحْمَـٰنِ ۗ بَلْ هُمْ عَن ذِكْرِ رَبِّهِم مُّعْرِضُونَ ﴿٤٢﴾
(ئى مۇھەممەد! مەسخىرە قىلغۇچىلارغا) «سىلەرنى كېچىدە ۋە كۈندۈزدە مەرھەمەتلىك اﷲ نىڭ ئازابىدىن كىم ساقلايدۇ؟» دېگىن بەلكى ئۇلار پەرۋەردىگارىنىڭ كالامىدىن يۈز ئۆرۈگۈچىلەردۇر
أَمْ لَهُمْ ءَالِهَةٌ تَمْنَعُهُم مِّن دُونِنَا ۚ لَا يَسْتَطِيعُونَ نَصْرَ أَنفُسِهِمْ وَلَا هُم مِّنَّا يُصْحَبُونَ ﴿٤٣﴾
ئۇلاردىن بىزدىن باشقا ئۆزلىرىنى (قوغدايدىغان) مەبۇدلىرى بارمۇ ئۇلارنىڭ (چوقۇنۇۋاتقان) مەبۇدلىرى ئۆزلىرىنى قوغداشقىمۇ قادىر بولالمايدۇ، بىزنىڭ ياردىمىمىزگىمۇ ئېرىشەلمەيدۇ
بَلْ مَتَّعْنَا هَـٰٓؤُلَآءِ وَءَابَآءَهُمْ حَتَّىٰ طَالَ عَلَيْهِمُ ٱلْعُمُرُ ۗ أَفَلَا يَرَوْنَ أَنَّا نَأْتِى ٱلْأَرْضَ نَنقُصُهَا مِنْ أَطْرَافِهَآ ۚ أَفَهُمُ ٱلْغَـٰلِبُونَ ﴿٤٤﴾
بۇلارنى (يەنى مۇشرىكلارنى) ۋە ئۇلارنىڭ ئەجداتلىرىنى (دۇنيانىڭ مال - مۈلكىدىن) بەھرىمەن قىلدۇق، ھەتتا (شۇ نېمەت ئىچىدە) ئۇلارنىڭ ئۆمرى ئۇزۇن بولدى. ئۇلارنىڭ زېمىننىڭ ئەتراپىنى (مۇسۇلمانلارغا ئىگەللىتىش بىلەن) قەدەممۇقەدەم تارايتىپ بېرىۋاتقانلىقىمىزنى ئۇلارمۇ بىلەمدۇ؟ (ئەھۋال مۇشۇنداق ئىكەن) ئۇلار غالىب ھېسابلىنامدۇ؟ (مەغلۇپ ھېسابلىنامدۇ؟)
قُلْ إِنَّمَآ أُنذِرُكُم بِٱلْوَحْىِ ۚ وَلَا يَسْمَعُ ٱلصُّمُّ ٱلدُّعَآءَ إِذَا مَا يُنذَرُونَ ﴿٤٥﴾
(ئى مۇھەممەد!) «مەن سىلەرنى پەقەت ۋەھيى ئارقىلىق ئاگاھلاندۇرىمەن» دېگىن. گاسلار ئاگاھلاندۇرۇلغان چاغدا دەۋەتنى ئاڭلىمايدۇ (ئى مۇشرىكلار! سىلەر سەركەشلىكىڭلارنىڭ قاتتىقلىقىدىن خۇددى گاسقا ئوخشاش دەۋەتنى ئاڭلىمايسىلەر، ۋەز - نەسىھەتنى قوبۇل قىلمايسىلەر)
وَلَئِن مَّسَّتْهُمْ نَفْحَةٌ مِّنْ عَذَابِ رَبِّكَ لَيَقُولُنَّ يَـٰوَيْلَنَآ إِنَّا كُنَّا ظَـٰلِمِينَ ﴿٤٦﴾
ئەگەر ئۇلار پەرۋەردىگارىڭدىن ئازغىنا ئازاب يەتسىلا، ئۇلار چوقۇم: «ۋاي ئىسىت! (اﷲ نىڭ خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلىپ) ئىشەنچلىك ھالدا (ئۆزىمىزگە) زۇلۇم قىلىپتۇق» دەيدۇ
وَنَضَعُ ٱلْمَوَٰزِينَ ٱلْقِسْطَ لِيَوْمِ ٱلْقِيَـٰمَةِ فَلَا تُظْلَمُ نَفْسٌ شَيْـًٔا ۖ وَإِن كَانَ مِثْقَالَ حَبَّةٍ مِّنْ خَرْدَلٍ أَتَيْنَا بِهَا ۗ وَكَفَىٰ بِنَا حَـٰسِبِينَ ﴿٤٧﴾
قىيامەت كۈنى بىز (ئەمەللەر تارتىلىدىغان) ئادالەت تارازىسىنى ئورنىتىمىز، ھېچ ئادەمگە قىلچە ئۇۋال قىلىنمايدۇ (يەنى ياخشى ئادەمنىڭ ياخشىلىقى كېمەيتىۋېتىلمەيدۇ، يامان ئادەمنىڭ يامانلىقى ئاشۇرۇۋېتىلمەيدۇ) ئەگەر ئۇنىڭ قىچا چاغلىق ئەمەلى بولسىمۇ، ئۇنى ھازىر قىلىمىز، (بەندىلەرنىڭ ئەمەللىرىدىن) ھېساب ئېلىشقا بىز يېتەرلىكمىز
وَلَقَدْ ءَاتَيْنَا مُوسَىٰ وَهَـٰرُونَ ٱلْفُرْقَانَ وَضِيَآءً وَذِكْرًا لِّلْمُتَّقِينَ ﴿٤٨﴾

تەقۋادارلار پەرۋەردىگارىنى كۆرمەي تۇرۇپ، ئۇنىڭدىن قورقىدۇ ھەمدە ئۇلار قىيامەتنىڭ (دەھشىتىدىن) تىترىگۈچىلەردۇر

قَالَ بَل رَّبُّكُمْ رَبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ٱلَّذِى فَطَرَهُنَّ وَأَنَا۠ عَلَىٰ ذَٰلِكُم مِّنَ ٱلشَّـٰهِدِينَ ﴿٥٦﴾
ئىبراھىم ئېيتتى: «ئۇنداق ئەمەس، سىلەرنىڭ (ئىبادەت قىلىشىڭلارغا لايىق) پەرۋەردىگارىڭلار ئاسمانلارنىڭ ۋە زېمىننىڭ پەرۋەردىگارىدۇر، ئۇلارنى ئۇ ياراتقان، مەنمۇ ئۇنىڭغا (قەتئىي دەلىللەر بىلەن) گۇۋاھلىق قىلغۇچىلاردىنمەن
وَتَٱللَّهِ لَأَكِيدَنَّ أَصْنَـٰمَكُم بَعْدَ أَن تُوَلُّوا۟ مُدْبِرِينَ ﴿٥٧﴾
اﷲ بىلەن قەسەمكى، سىلەر قايتىپ كەتكەندىن كېيىن چوقۇم بۇتلىرىڭلارنى (چېقىۋېتىشنىڭ) ئامالىنى قىلىمەن»
فَجَعَلَهُمْ جُذَٰذًا إِلَّا كَبِيرًا لَّهُمْ لَعَلَّهُمْ إِلَيْهِ يَرْجِعُونَ ﴿٥٨﴾
ئىبراھىم بۇتلارنى چېقىپ پارچە - پارچە قىلىۋەتتى، ئۇلارنىڭ قايتىپ كېلىپ (بۇتلارنى كىمنىڭ چاققانلىقىنى سورىشى ئۈچۈن) پەقەت چوڭ بۇتنى (چاقماي) قالدۇردى
قَالُوا۟ مَن فَعَلَ هَـٰذَا بِـَٔالِهَتِنَآ إِنَّهُۥ لَمِنَ ٱلظَّـٰلِمِينَ ﴿٥٩﴾
(ئۇلار بايرىمىدىن قايتىپ كېلىپ بۇتلىرىنىڭ چېقىۋېتىلگەنلىكىنى كۆرۈپ) «بۇتلىرىمىزنى مۇنداق قىلغان كىم؟ ئۇ ئىشەنچلىك ھالدا زالىم ئىكەن» دېدى
قَالُوا۟ سَمِعْنَا فَتًى يَذْكُرُهُمْ يُقَالُ لَهُۥٓ إِبْرَٰهِيمُ ﴿٦٠﴾
(ئۇلارنىڭ بەزىسى) ئىبراھىم ئاتلىق بىر يىگىتنىڭ ئۇلارنى ئەيىبلىگەنلىكىنى ئاڭلىغان ئىدۇق (يەنى شۇ قىلغان بولۇشى مۇمكىن)» دېدى
قَالُوا۟ فَأْتُوا۟ بِهِۦ عَلَىٰٓ أَعْيُنِ ٱلنَّاسِ لَعَلَّهُمْ يَشْهَدُونَ ﴿٦١﴾
ئۇلار (يەنى نەمرۇد ۋە ئۇنىڭ قەۋمىنىڭ چوڭلىرى): «ئىبراھىمنى كىشىلەرنىڭ ئالدىغا ئېلىپ كېلىڭلار، (ئىبراھىمنىڭ بۇتلارنى ئەيىبلىگەنلىكىگە) ئۇلار گۇۋاھلىق بەرسۇن» دېدى)
قَالُوٓا۟ ءَأَنتَ فَعَلْتَ هَـٰذَا بِـَٔالِهَتِنَا يَـٰٓإِبْرَٰهِيمُ ﴿٦٢﴾
ئۇلار: «ئى ئىبراھىم! بۇتلىرىمىزنى مۇشۇنداق قىلغان سەنمۇ؟» دېدى.
قَالَ بَلْ فَعَلَهُۥ كَبِيرُهُمْ هَـٰذَا فَسْـَٔلُوهُمْ إِن كَانُوا۟ يَنطِقُونَ ﴿٦٣﴾
(ئىبراھىم: «(ياق) بەلكى ئۇلارنىڭ مۇنۇ چوڭى شۇنداق قىلدى، ئەگەر ئۇلار سۆزلىيەلىسە، (كىمنىڭ چاققانلىقىنى) ئۇلارنىڭ ئۆزلىرىدىن سوراپ بېقىڭلار!» دېدى
فَرَجَعُوٓا۟ إِلَىٰٓ أَنفُسِهِمْ فَقَالُوٓا۟ إِنَّكُمْ أَنتُمُ ٱلظَّـٰلِمُونَ ﴿٦٤﴾
ئۇلار بىر - بىرىنى ئەيىبلەپ: «(سىلەر بۇتلارغا چوقۇنۇشتا) ئىشەنچلىك ھالدا زالىمسىلەر» دېدى
ثُمَّ نُكِسُوا۟ عَلَىٰ رُءُوسِهِمْ لَقَدْ عَلِمْتَ مَا هَـٰٓؤُلَآءِ يَنطِقُونَ ﴿٦٥﴾
ئاندىن ئۇلار (خىجالەتچىلىكتىن) باشلىرىنى تۆۋەن سېلىپ: «ئۇلارنىڭ گەپ قىلالمايدىغانلىقىنى سەن ئوبدان بىلىسەنغۇ؟» دېدى
قَالَ أَفَتَعْبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَنفَعُكُمْ شَيْـًٔا وَلَا يَضُرُّكُمْ ﴿٦٦﴾
ئىبراھىم ئېيتتى: «سىلەر اﷲ نى قويۇپ سىلەرگە ھېچ پايدا يەتكۈزەلمەيدىغان ۋە زىيان يەتكۈزەلمەيدىغان نەرسىلەرگە چوقۇنامسىلەر»
أُفٍّ لَّكُمْ وَلِمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ ۖ أَفَلَا تَعْقِلُونَ ﴿٦٧﴾
سىلەرگە ۋە اﷲ نى قويۇپ سىلەر چوقۇنۇۋاتقان بۇتلارغا تۈفى! سىلەر چۈشەنمەمسىلەر؟»
قَالُوا۟ حَرِّقُوهُ وَٱنصُرُوٓا۟ ءَالِهَتَكُمْ إِن كُنتُمْ فَـٰعِلِينَ ﴿٦٨﴾
ئۇلار ئېيتتى: «ئىبراھىمنى كۆيدۈرۈڭلار، ئىلاھلىرىڭلارغا ياردەم بېرىڭلار، ئەگەر ئۇلارغا ھەقىقىي ياردەم بەرمەكچى بولساڭلار»
قُلْنَا يَـٰنَارُ كُونِى بَرْدًا وَسَلَـٰمًا عَلَىٰٓ إِبْرَٰهِيمَ ﴿٦٩﴾
بىز ئوتقا: «ئى ئوت! ئىبراھىمغا سالقىن ۋە ئامانلىق (بەخش ئېتىدىغان) بولۇپ بەرگىن» دېدۇق
وَأَرَادُوا۟ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَخْسَرِينَ ﴿٧٠﴾
ئۇلار ئىبراھىمغا زىيانكەشلىك قىلماقچى بولدى، بىز ئۇلارنى ئەڭ زىيان تارتقۇچى قىلدۇق
وَنَجَّيْنَـٰهُ وَلُوطًا إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا لِلْعَـٰلَمِينَ ﴿٧١﴾
بىز ئىبراھىم بىلەن لۇتنى قۇتقۇزدۇق، ئۇلارنى بىز ئەھلى جاھان ئۈچۈن بەرىكەتلىك زېمىنغا كۆچۈردۇق
وَوَهَبْنَا لَهُۥٓ إِسْحَـٰقَ وَيَعْقُوبَ نَافِلَةً ۖ وَكُلًّا جَعَلْنَا صَـٰلِحِينَ ﴿٧٢﴾
بىز ئۇنىڭغا (يەنى ئىبراھىمغا) ئىسھاقنى بەردۇق، (ئۇ پەرۋەردىگارىدىن سورىمىسىمۇ) يەئقۇبنى ئوشۇق بەردۇق (يەنى يەئقۇب ئەلەيھىسسالامنى نەۋرە قىلىپ ياكى ئىبراھى ئەلەيھىسسالامنىڭ تەلىۋىدىن زىيادە قىلىپ بەردۇق)، ئۇلارنىڭ ھەممىسىنىڭ ياخشى ئادەملەر قىلدۇق
وَجَعَلْنَـٰهُمْ أَئِمَّةً يَهْدُونَ بِأَمْرِنَا وَأَوْحَيْنَآ إِلَيْهِمْ فِعْلَ ٱلْخَيْرَٰتِ وَإِقَامَ ٱلصَّلَوٰةِ وَإِيتَآءَ ٱلزَّكَوٰةِ ۖ وَكَانُوا۟ لَنَا عَـٰبِدِينَ ﴿٧٣﴾
ئۇلارنى بىزنىڭ ئەمرىمىز بويىچە (كىشىلەرنى بىزنىڭ دىنىمىزغا) يېتەكلەيدىغان پېشۋالار (يەنى خۇدانىڭ ئەلچى-رەسۇللىرى) قىلدۇق، ئۇلارغا بىز ياخشى ئىشلارنى قىلىشنى، ناماز ئوقۇشنى، زاكات (پاكلاش ئۇلۇشى) بېرىشنى ۋەھيى قىلىدۇ، ئۇلار بىزگە خالىس ئىبادەت قىلاتتى
وَلُوطًا ءَاتَيْنَـٰهُ حُكْمًا وَعِلْمًا وَنَجَّيْنَـٰهُ مِنَ ٱلْقَرْيَةِ ٱلَّتِى كَانَت تَّعْمَلُ ٱلْخَبَـٰٓئِثَ ۗ إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ فَـٰسِقِينَ ﴿٧٤﴾
لۇتقا خۇدانىڭ ئەلچىسىلىكنى ۋە ئىلىمنى ئاتا قىلدۇق (لۇت ئىبراھىم ئەلەيھىسسالامغا ئىمان ئېيتقان، ئەگەشكەن ۋە ئۇنىڭ بىلەن ھىجرەت قىلغان ئىدى)، ئۇنىڭ (ئائىلىسى بەچچىۋازلىق، قاراقچىلىق قاتارلىق) يامان قىلىقلارنى قىلىدىغان شەھەردىن قۇتقۇزدۇق، ئىشەنچلىك ھالدا ئۇلار يامان پاسىق قەۋم ئىدى
وَأَدْخَلْنَـٰهُ فِى رَحْمَتِنَآ ۖ إِنَّهُۥ مِنَ ٱلصَّـٰلِحِينَ ﴿٧٥﴾
بىز ئۇنى رەھمىتىمىز دائىرىسىگە كىرگۈزدۇق، ئۇ ئىشەنچلىك ھالدا ياخشىلاردىن ئىدى
وَنُوحًا إِذْ نَادَىٰ مِن قَبْلُ فَٱسْتَجَبْنَا لَهُۥ فَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ﴿٧٦﴾
نۇھنىڭ (قىسسىسىنى بايان قىلغىن). ئۆز ۋاقتىدا ئۇ (مەزكۇر خۇدانىڭ ئەلچى-رەسۇللىرىدىن) بۇرۇن (قەۋمىگە، پەرۋەردىگارىم يەر يۈزىدە كۇففارلاردىن بىرنىمۇ قويمىغىن، دەپ بەتدۇئا قىلغان ئىدى)، بىز ئۇنىڭ دۇئاسىنى ئىجابەت قىلدۇق، ئۇنى ۋە ئۇنىڭ ئائىلىسىدىكىلەرنى (ۋە ئۇنىڭ بىلەن بىللە بولغان مۆمىنلەرنى) بۈيۈك بالادىن (يەنى توپان بالاسىدا غەرق بولۇشتىن) قۇتقۇزدۇق (يەنى نۇھنى قەۋمىنىڭ شەررىدىن قۇتقۇزۇپ، ئۇلارنى ھالاك قىلدۇق)
وَنَصَرْنَـٰهُ مِنَ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَـٰتِنَآ ۚ إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ فَأَغْرَقْنَـٰهُمْ أَجْمَعِينَ ﴿٧٧﴾
بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغان قەۋمگە قارشى ئۇنىڭغا ياردەم بەردۇق، ئۇلار ئىشەنچلىك ھالدا يامان قەۋم ئىدى. ئۇلارنىڭ (بىرنىمۇ قويماي) ھەممىسىنى غەرق قىلدۇق
وَدَاوُۥدَ وَسُلَيْمَـٰنَ إِذْ يَحْكُمَانِ فِى ٱلْحَرْثِ إِذْ نَفَشَتْ فِيهِ غَنَمُ ٱلْقَوْمِ وَكُنَّا لِحُكْمِهِمْ شَـٰهِدِينَ ﴿٧٨﴾
داۋۇت بىلەن سۇلايماننىڭ (قىسسىسىنى بايان قىلغىن). ئۆز ۋاقتىدا بىر قەۋمنىڭ قويى كېچىسى زىرائەتنى يەپ (بۇزۇۋەتكەندە)، ئۇ ئىككىسى زىرائەت توغرىسىدا ھۆكۈم چىقارغان ئىدى. ئۇلارنىڭ ھۆكمىگە بىز شاھىت ئىدۇق
فَفَهَّمْنَـٰهَا سُلَيْمَـٰنَ ۚ وَكُلًّا ءَاتَيْنَا حُكْمًا وَعِلْمًا ۚ وَسَخَّرْنَا مَعَ دَاوُۥدَ ٱلْجِبَالَ يُسَبِّحْنَ وَٱلطَّيْرَ ۚ وَكُنَّا فَـٰعِلِينَ ﴿٧٩﴾
قانداق ھۆكۈم قىلىشنى بىز سۇلايمانغا بىلدۈردۇق، تەسبىھ ئېيتىشقا مۇسەخخەر قىلدۇق، (شۇنداق قىلىشقا) بىز قادىر ئىدۇق
وَعَلَّمْنَـٰهُ صَنْعَةَ لَبُوسٍ لَّكُمْ لِتُحْصِنَكُم مِّنۢ بَأْسِكُمْ ۖ فَهَلْ أَنتُمْ شَـٰكِرُونَ ﴿٨٠﴾
جەڭلەردە سىلەرنى (يارىدار بولۇشتىن) ساقلاش مەقسىتىدە، داۋۇدقا سىلەر ئۈچۈن ساۋۇت ياساشنى ئۆگەتتۇق، سىلەر (بۇنىڭغا) شۈكۈر قىلامسىلەر؟ (يەنى شۈكۈر قىلىڭلار ۋە نېمەتنىڭ قەدرىنى بىلىڭلار)
وَلِسُلَيْمَـٰنَ ٱلرِّيحَ عَاصِفَةً تَجْرِى بِأَمْرِهِۦٓ إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا ۚ وَكُنَّا بِكُلِّ شَىْءٍ عَـٰلِمِينَ ﴿٨١﴾
سۇلايمانغا قاتتىق چىقىدىغان شامالنى مۇسەخخەر قىلىپ بەردۇق، شامال سۇلايماننىڭ ئەمرى بويىچە، بىز بەرىكەتلىك قىلغان زېمىنغا (يەنى شامغا) باراتتى، بىز ھەممىنى بىلگۈچىمىز
وَمِنَ ٱلشَّيَـٰطِينِ مَن يَغُوصُونَ لَهُۥ وَيَعْمَلُونَ عَمَلًا دُونَ ذَٰلِكَ ۖ وَكُنَّا لَهُمْ حَـٰفِظِينَ ﴿٨٢﴾
سۇلايمانغا يەنە بىر قىسىم شەيتان (ئازدۇرغۇچى)لارنى مۇسەخخەر قىلىپ بەردۇق، ئۇلار دېڭىزلاردا چۆكۈپ ئۇنىڭغا ئۈنچە - مارجان (گۆھەرلەرنى سۈزۈپ چىقاتتى)، ئۇلار يەنە (شەھەرلەرنى، ئېگىز سارايلارنى سېلىشتەك) باشقا ئىشلارنىمۇ قىلاتتى، بىز ئۇلارنى (سۇلايماننىڭ ئىتائىتىدىن چىقىشتىن) ساقلايتتۇق
۞ وَأَيُّوبَ إِذْ نَادَىٰ رَبَّهُۥٓ أَنِّى مَسَّنِىَ ٱلضُّرُّ وَأَنتَ أَرْحَمُ ٱلرَّٰحِمِينَ ﴿٨٣﴾

بىلەمسەنكى، ئۇ (يەنى سىلەرگە بېرىلىدىغان جازانىڭ كېچىكىشى) سىلەر ئۈچۈن سىناق ۋە ۋاقىتلىق پايدىلىنىش بولۇشى مۇمكىن»
قَـٰلَ رَبِّ ٱحْكُم بِٱلْحَقِّ ۗ وَرَبُّنَا ٱلرَّحْمَـٰنُ ٱلْمُسْتَعَانُ عَلَىٰ مَا تَصِفُونَ ﴿١١٢﴾
(خۇدانىڭ ئەلچىسى ئېيتتى) «پەرۋەردىگارىم! (مېنىڭ بىلەن مەككە كۇففارلىرىنىڭ ئارىسىدا) ئادىللىق بىلەن ھۆكۈم چىقارغىن، بىزنىڭ پەرۋەردىگارىمىز مەرھەمەتلىك اﷲ دۇر، سىلەرنىڭ بوھتان سۆزلىرىڭلارغا قارشى اﷲ تىن ياردەم تىلەيمەن
