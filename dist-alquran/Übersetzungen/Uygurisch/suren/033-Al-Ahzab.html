بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
يَـٰٓأَيُّهَا ٱلنَّبِىُّ ٱتَّقِ ٱللَّهَ وَلَا تُطِعِ ٱلْكَـٰفِرِينَ وَٱلْمُنَـٰفِقِينَ ۗ إِنَّ ٱللَّهَ كَانَ عَلِيمًا حَكِيمًا ﴿١﴾
ئى خۇدانىڭ ئەلچىسى! اﷲ قا تەقۋادارلىق قىلىشنى (داۋاملاشتۇرغىن)، ھەقىقەتنى رەت قىلغۇچىلارغا ۋە مۇناپىقلارغا ئىتائەت قىلمىغىن، اﷲ ئىشەنچلىك ھالدا (بەندىلەرنىڭ ئەمەللىرىنى) بىلىپ تۇرغۇچىدۇر، (ئۇلارنىڭ ئىشلىرىنى) ھېكمەت بىلەن باشقۇرغۇچىدۇر
وَٱتَّبِعْ مَا يُوحَىٰٓ إِلَيْكَ مِن رَّبِّكَ ۚ إِنَّ ٱللَّهَ كَانَ بِمَا تَعْمَلُونَ خَبِيرًا ﴿٢﴾
ساڭا پەرۋەردىگارىڭ تەرىپىدىن ۋەھيى قىلىنغان نەرسىگە (يەنى قۇرئانغا) ئەگەشكىن. اﷲ ئىشەنچلىك ھالدا قىلغان ئەمەلىڭلاردىن تولۇق خەۋەرداردۇر
وَتَوَكَّلْ عَلَى ٱللَّهِ ۚ وَكَفَىٰ بِٱللَّهِ وَكِيلًا ﴿٣﴾
(ھەممە ئىشىڭدا) اﷲ قا يۆلەنگىن، اﷲ (ساڭا) ھامىي بولۇشقا يېتەرلىكتۇر
مَّا جَعَلَ ٱللَّهُ لِرَجُلٍ مِّن قَلْبَيْنِ فِى جَوْفِهِۦ ۚ وَمَا جَعَلَ أَزْوَٰجَكُمُ ٱلَّـٰٓـِٔى تُظَـٰهِرُونَ مِنْهُنَّ أُمَّهَـٰتِكُمْ ۚ وَمَا جَعَلَ أَدْعِيَآءَكُمْ أَبْنَآءَكُمْ ۚ ذَٰلِكُمْ قَوْلُكُم بِأَفْوَٰهِكُمْ ۖ وَٱللَّهُ يَقُولُ ٱلْحَقَّ وَهُوَ يَهْدِى ٱلسَّبِيلَ ﴿٤﴾
اﷲ ھېچ ئادەمنىڭ ئىچىدە ئىككى يۈرەك ياراتقىنى يوق، سىلەر زىھار قىلغان ئاياللىرىڭلارنى اﷲ سىلەرگە ئانا قىلغىنى يوق، سىلەر بالا قىلىۋالغانلارنىمۇ اﷲ (ھەممە ھۆكۈمدە ئۆز پۇشتۇڭلاردىن بولغان) بالىلىرىڭلارنىڭ ھۆكمىدە قىلمىدى، بۇ (يەنى ئۆز پۇشتۇڭلاردىن بولمىغانلارنى بالا دېيىش) پەقەت ئاغزىڭلاردا ئېيتىلغان سۆزدۇر، اﷲ ھەق (سۆز) نى ئېيتىدۇ، توغرا يولغا باشلايدۇ
ٱدْعُوهُمْ لِـَٔابَآئِهِمْ هُوَ أَقْسَطُ عِندَ ٱللَّهِ ۚ فَإِن لَّمْ تَعْلَمُوٓا۟ ءَابَآءَهُمْ فَإِخْوَٰنُكُمْ فِى ٱلدِّينِ وَمَوَٰلِيكُمْ ۚ وَلَيْسَ عَلَيْكُمْ جُنَاحٌ فِيمَآ أَخْطَأْتُم بِهِۦ وَلَـٰكِن مَّا تَعَمَّدَتْ قُلُوبُكُمْ ۚ وَكَانَ ٱللَّهُ غَفُورًا رَّحِيمًا ﴿٥﴾
ئۇلارنى ئاتىلىرىنىڭ ئىسىملىرى بىلەن چاقىرىڭلار، بۇ اﷲ نىڭ دەرگاھىدا توغرىدۇر، ئەگەر ئۇلارنىڭ ئاتىلىرىنى بىلمىسەڭلار، ئۇ چاغدا ئۇلار سىلەرنىڭ دىنىي قېرىنداشلىرىڭلاردۇر، دوستلىرىڭلاردۇر، سىلەر سەۋەنلىكتىن قىلىپ سالغان ئىشلاردا سىلەرگە ھېچ گۇناھ بولمايدۇ، لېكىن قەستەن قىلغان ئىشىڭلاردا (سىلەرگە گۇناھ بولىدۇ)، اﷲ ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
ٱلنَّبِىُّ أَوْلَىٰ بِٱلْمُؤْمِنِينَ مِنْ أَنفُسِهِمْ ۖ وَأَزْوَٰجُهُۥٓ أُمَّهَـٰتُهُمْ ۗ وَأُو۟لُوا۟ ٱلْأَرْحَامِ بَعْضُهُمْ أَوْلَىٰ بِبَعْضٍ فِى كِتَـٰبِ ٱللَّهِ مِنَ ٱلْمُؤْمِنِينَ وَٱلْمُهَـٰجِرِينَ إِلَّآ أَن تَفْعَلُوٓا۟ إِلَىٰٓ أَوْلِيَآئِكُم مَّعْرُوفًا ۚ كَانَ ذَٰلِكَ فِى ٱلْكِتَـٰبِ مَسْطُورًا ﴿٦﴾
خۇدانىڭ ئەلچىسى مۆمىنلەرگە ئۇلارنىڭ ئۆز نەپسىلىرىدىنمۇ چارىدۇر، خۇدانىڭ ئەلچىسىنىڭ ئاياللىرى ئۇلارنىڭ ئانىلىرىدۇر، مۇقەددەس كىتاب-ۋەھىيۇللاھتا ئۇرۇق - تۇغقانلارنىڭ بەزىسى بەزىسىگە (مىراس ئېلىشتا) مۆمىنلەردىن ۋە مۇھاجىرلاردىن يېقىندۇر، لېكىن (مۆمىنلەردىن ۋە مۇھاجىرلاردىن بولغان) دوستلىرىڭلارغا ۋەسىيەت قىلساڭلار دۇرۇس بولىدۇ، بۇ (يەنى تۇغقانلارنىڭ ئۆز ئارا مىراسقا ۋارىسلىق قىلىشى) مۇقەددەس كىتاب-ۋەھىيتا (يەنى اﷲ نىڭ مۇقەددەس كىتاب-ۋەھىيىدا) يېزىلغاندۇر
وَإِذْ أَخَذْنَا مِنَ ٱلنَّبِيِّـۧنَ مِيثَـٰقَهُمْ وَمِنكَ وَمِن نُّوحٍ وَإِبْرَٰهِيمَ وَمُوسَىٰ وَعِيسَى ٱبْنِ مَرْيَمَ ۖ وَأَخَذْنَا مِنْهُم مِّيثَـٰقًا غَلِيظًا ﴿٧﴾
ئۆز ۋاقتىدا بىز خۇدانىڭ ئەلچى-رەسۇللىرىدىن ئەھدە ئالدۇق، سەندىن، نۇھتىن، ئىبراھىمدىن، مۇسادىن، مەريەم ئوغلى ئىسادىن مەھكەم ئەھدە ئالدۇق
لِّيَسْـَٔلَ ٱلصَّـٰدِقِينَ عَن صِدْقِهِمْ ۚ وَأَعَدَّ لِلْكَـٰفِرِينَ عَذَابًا أَلِيمًا ﴿٨﴾
اﷲ (قىيامەت كۈنى) راستچىللاردىن راستچىللىقى توغرۇلۇق سورايدۇ (يەنى خۇدانىڭ ئەلچى-رەسۇللىرىدىن خۇدانىڭ ئەلچىسىلىكىنى يەتكۈزگەنلىكىنى سورايدۇ)، اﷲ ھەقىقەتنى رەت قىلغۇچىلارغا قاتتىق ئازاب تەييارلىدى
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَآءَتْكُمْ جُنُودٌ فَأَرْسَلْنَا عَلَيْهِمْ رِيحًا وَجُنُودًا لَّمْ تَرَوْهَا ۚ وَكَانَ ٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرًا ﴿٩﴾
ئى مۆمىنلەر! اﷲ نىڭ سىلەرگە بەرگەن نېمىتىنى ئەسلەڭلار، ئۆز ۋاقتىدا ئۈستۈڭلارغا (كۇففارلاردىن ئۇيۇشقان ئۇرۇشقۇچى) قوشۇن كەلگەن ئىدى. بىز ئۇلارغا قارشى بوران ۋە سىلەرگە كۆرۈنمەيدىغان قوشۇن (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرىنى) ئەۋەتتۇق، اﷲ سىلەرنىڭ قىلغان ئىشىڭلارنى كۆرۈپ تۇرغۇچىدۇر
إِذْ جَآءُوكُم مِّن فَوْقِكُمْ وَمِنْ أَسْفَلَ مِنكُمْ وَإِذْ زَاغَتِ ٱلْأَبْصَـٰرُ وَبَلَغَتِ ٱلْقُلُوبُ ٱلْحَنَاجِرَ وَتَظُنُّونَ بِٱللَّهِ ٱلظُّنُونَا۠ ﴿١٠﴾
ئەينى ۋاقىتتا ئۇلار يۇقىرى تەرىپىڭلاردىنمۇ، تۆۋەن تەرىپىڭلاردىنمۇ (ھۇجۇم قىلىپ) كەلگەن ئىدى. بۇ چاغدا كۆزۈڭلار چەكچىيىپ قالغان، يۈرىكىڭلار ئاغزىڭلارغا قاپلىشىپ قالغان، اﷲ ھەققىدە تۈرلۈك گۇمانلاردا بولغان ئىدىڭلار
هُنَالِكَ ٱبْتُلِىَ ٱلْمُؤْمِنُونَ وَزُلْزِلُوا۟ زِلْزَالًا شَدِيدًا ﴿١١﴾
بۇ يەردە مۆمىنلەر سىنالغان ۋە (بېشىغا كەلگەن كۈننىڭ قاتتىقلىقىدىن گويا ئۇلارنى يەر سىلكىگەندەك) قاتتىق تەۋرىنىشكە ئۇچرىغان ئىدى
وَإِذْ يَقُولُ ٱلْمُنَـٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ مَّا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥٓ إِلَّا غُرُورًا ﴿١٢﴾
ئەينى ۋاقىتتا مۇناپىقلار ۋە دىللىرىدا (مۇناپىقلىق) كېسىلى بارلار: «اﷲ ۋە ئۇنىڭ پەيغەمبىرى بىزلەرگە قۇرۇق ۋەدىلەرنى قىلىپتىكەن» دېيىشەتتى
وَإِذْ قَالَت طَّآئِفَةٌ مِّنْهُمْ يَـٰٓأَهْلَ يَثْرِبَ لَا مُقَامَ لَكُمْ فَٱرْجِعُوا۟ ۚ وَيَسْتَـْٔذِنُ فَرِيقٌ مِّنْهُمُ ٱلنَّبِىَّ يَقُولُونَ إِنَّ بُيُوتَنَا عَوْرَةٌ وَمَا هِىَ بِعَوْرَةٍ ۖ إِن يُرِيدُونَ إِلَّا فِرَارًا ﴿١٣﴾
ئەينى ۋاقىتتا ئۇلاردىن بىر گۇرۇھ ئادەم: «ئى يەسرىبلىكلەر! بۇ يەردە تۇرماي قايتىڭلار!» دېيىشەتتى، ئۇلاردىن يەنە بىر گۇرۇھ ئادەم: «بىزنىڭ ئۆيلىرىمىز ئىشەنچلىك ھالدا ئوچۇقچىلىقتا» دېيىشىپ، خۇدانىڭ ئەلچىسىدىن رۇخسەت سورىشاتتى، بۇلارنىڭ ئۆيلىرى ئوچۇقچىلىقتا ئەمەس، ئۇلار پەقەت قېچىشنىلا ئويلايدۇ
وَلَوْ دُخِلَتْ عَلَيْهِم مِّنْ أَقْطَارِهَا ثُمَّ سُئِلُوا۟ ٱلْفِتْنَةَ لَـَٔاتَوْهَا وَمَا تَلَبَّثُوا۟ بِهَآ إِلَّا يَسِيرًا ﴿١٤﴾
ئەگەر شەھەرگە تۆت تەرەپتىن ھۇجۇم قىلىپ كىرىلسە، ئاندىن ئۇلاردىن ئىمانىدىن قايتىش، مۇسۇلمانلارغا قارشى ئۇرۇش قىلىش تەلەپ قىلىنسا، ئۇلار بۇ تەلەپنى ئۈزلۈكسىز رەۋىشتە ئاز ۋاقىت ئىچىدىلا قوبۇل قىلاتتى
وَلَقَدْ كَانُوا۟ عَـٰهَدُوا۟ ٱللَّهَ مِن قَبْلُ لَا يُوَلُّونَ ٱلْأَدْبَـٰرَ ۚ وَكَانَ عَهْدُ ٱللَّهِ مَسْـُٔولًا ﴿١٥﴾
ئۇلار (يەنى مۇناپىقلار) ئىلگىرى اﷲ قا (جەڭدە) چېكىنمەسلىككە ۋەدە بەرگەن ئىدى. اﷲ قا بەرگەن ۋەدىسى سورىلىدۇ
قُل لَّن يَنفَعَكُمُ ٱلْفِرَارُ إِن فَرَرْتُم مِّنَ ٱلْمَوْتِ أَوِ ٱلْقَتْلِ وَإِذًا لَّا تُمَتَّعُونَ إِلَّا قَلِيلًا ﴿١٦﴾
ئېيتقىنكى، «ئەگەر سىلەر ئۆلۈمدىن يا ئۆلتۈرۈلۈشتىن قاچساڭلار، قېچىشنىڭ سىلەرگە ھېچقانداق پايدىسى يوق، قاچقاندىمۇ سىلەر (دۇنيادىن) ئازغىنا (ۋاقىت) بەھرىمەن بولالايسىلەر»
قُلْ مَن ذَا ٱلَّذِى يَعْصِمُكُم مِّنَ ٱللَّهِ إِنْ أَرَادَ بِكُمْ سُوٓءًا أَوْ أَرَادَ بِكُمْ رَحْمَةً ۚ وَلَا يَجِدُونَ لَهُم مِّن دُونِ ٱللَّهِ وَلِيًّا وَلَا نَصِيرًا ﴿١٧﴾
ئېيتقىنكى: «ئەگەر اﷲ سىلەرگە بىرەر يامانلىقنى ئىرادە قىلسا، كىم اﷲ قا قارشى تۇرۇپ سىلەرنى قوغدىيالايدۇ، ئەگەر اﷲ سىلەرگە رەھمەت قىلىشنى ئىرادە قىلسا (كىم ئۇنىڭغا ئارىلىشالايدۇ؟)» ئۇلار ئۆزلىرى ئۈچۈن اﷲ تىن باشقا ھېچ ئىگە ۋە ھېچ ياردەمچى تاپالمايدۇ
۞ قَدْ يَعْلَمُ ٱللَّهُ ٱلْمُعَوِّقِينَ مِنكُمْ وَٱلْقَآئِلِينَ لِإِخْوَٰنِهِمْ هَلُمَّ إِلَيْنَا ۖ وَلَا يَأْتُونَ ٱلْبَأْسَ إِلَّا قَلِيلًا ﴿١٨﴾
اﷲ ئىچىڭلاردىكى باشقىلارنى (جىھادتىن) توسقۇچىلارنى ۋە قېرىنداشلىرىغا (مۇھەممەدنى ۋە ساھابىلىرىنى تەرك ئېتىپ) بىزنىڭ قېشىمىزغا كېلىڭلار دېگۈچىلەرنى ئوبدان بىلىدۇ، ئۇلار جەڭگە (رىيا قىلىپ) ئاندا - ساندا قاتنىشىدۇ
أَشِحَّةً عَلَيْكُمْ ۖ فَإِذَا جَآءَ ٱلْخَوْفُ رَأَيْتَهُمْ يَنظُرُونَ إِلَيْكَ تَدُورُ أَعْيُنُهُمْ كَٱلَّذِى يُغْشَىٰ عَلَيْهِ مِنَ ٱلْمَوْتِ ۖ فَإِذَا ذَهَبَ ٱلْخَوْفُ سَلَقُوكُم بِأَلْسِنَةٍ حِدَادٍ أَشِحَّةً عَلَى ٱلْخَيْرِ ۚ أُو۟لَـٰٓئِكَ لَمْ يُؤْمِنُوا۟ فَأَحْبَطَ ٱللَّهُ أَعْمَـٰلَهُمْ ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًا ﴿١٩﴾
ئۇلار سىلەرگە (ياردەمدە بولۇشقا) بېخىلدۇر، ئەگەر (ئۇلارغا دۈشمەن تەرەپتىن) خەۋپ كېلىدىغان بولسا، ئۇلارنىڭ ساڭا سەكراتقا چۈشۈپ قالغان ئادەمدەك (قورقۇنچتىن) كۆزلىرىنى پىلدىر لىتىپ قاراۋاتقانلىقىنى كۆرىسەن. قورقۇنچ كەتكەندە، ئۇلار پۇل - مالغا ئاچكۆز بولغان ھالدا سىلەرنى كەسكىن تىللار بىلەن رەنجىتىدۇ. ئۇلار (چىن دىللىرى بىلەن) ئىمان ئېيتقان ئەمەس. (ئۇلارنىڭ كۇفرى سەۋەبلىك) ئەمەللىرىنى اﷲ بىكار قىلدى، بۇ اﷲ ئۈچۈن ئاساندۇر
يَحْسَبُونَ ٱلْأَحْزَابَ لَمْ يَذْهَبُوا۟ ۖ وَإِن يَأْتِ ٱلْأَحْزَابُ يَوَدُّوا۟ لَوْ أَنَّهُم بَادُونَ فِى ٱلْأَعْرَابِ يَسْـَٔلُونَ عَنْ أَنۢبَآئِكُمْ ۖ وَلَوْ كَانُوا۟ فِيكُم مَّا قَـٰتَلُوٓا۟ إِلَّا قَلِيلًا ﴿٢٠﴾
ئۇلار (يەنى مۇناپىقلار) (قاتتىق قورققانلىقتىن) ئىتتىپاقداش قوشۇننى چېكىنمىدى دەپ ئويلايدۇ، ئەگەر ئىتتىپاقداش قوشۇن قايتا كەلسە، ئۇلار (قىردا) ئەئرابىلارنىڭ (يەنى قىر ئەرەبلىرىنىڭ) يېنىدا تۇرۇپ سىلەرنىڭ خەۋەرلىرىڭلارنى ئۇققاچ (تىنچ يېتىشىنى) ئارزۇ قىلىدۇ، ئۇلار (ئۇرۇش ۋاقتىدا) ئاراڭلاردا بولغان تەقدىردىمۇ، (قورققانلىقتىن) جەڭگە (رىيا قىلىپ) ئاندا - ساندا قاتناشقان بولاتتى
لَّقَدْ كَانَ لَكُمْ فِى رَسُولِ ٱللَّهِ أُسْوَةٌ حَسَنَةٌ لِّمَن كَانَ يَرْجُوا۟ ٱللَّهَ وَٱلْيَوْمَ ٱلْـَٔاخِرَ وَذَكَرَ ٱللَّهَ كَثِيرًا ﴿٢١﴾
سىلەرگە - اﷲ نى، ئاخىرەت كۈنىنى ئۈمىد قىلغان ۋە اﷲ نى كۆپ ياد ئەتكەنلەرگە - رەسۇلۇللاھ ئەلۋەتتە ياخشى ئۈلگىدۇر
وَلَمَّا رَءَا ٱلْمُؤْمِنُونَ ٱلْأَحْزَابَ قَالُوا۟ هَـٰذَا مَا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥ وَصَدَقَ ٱللَّهُ وَرَسُولُهُۥ ۚ وَمَا زَادَهُمْ إِلَّآ إِيمَـٰنًا وَتَسْلِيمًا ﴿٢٢﴾
مۆمىنلەر ئىتتىپاقداش قوشۇننى كۆرگەن چاغدا: «بۇ (يەنى قېيىن ئەھۋالدا قالغاندا دۈشمەن ئۈستىدىن غەلىبە قىلىش) اﷲ ۋە ئۇنىڭ پەيغەمبىرى بىزگە ۋەدە قىلغان ئىشتۇر. اﷲ ۋە ئۇنىڭ پەيغەمبىرى راست ئېيتتى» دېيىشتى. (بۇ ئىش) ئۇلارنىڭ (اﷲ قا بولغان) ئىمانىنى ۋە (اﷲ نىڭ بۇيرۇقلىرىغا بولغان) بويسۇنۇشىنى تېخىمۇ كۈچەيتتى
مِّنَ ٱلْمُؤْمِنِينَ رِجَالٌ صَدَقُوا۟ مَا عَـٰهَدُوا۟ ٱللَّهَ عَلَيْهِ ۖ فَمِنْهُم مَّن قَضَىٰ نَحْبَهُۥ وَمِنْهُم مَّن يَنتَظِرُ ۖ وَمَا بَدَّلُوا۟ تَبْدِيلًا ﴿٢٣﴾
مۆمىنلەرنىڭ ئىچىدە (رەسۇلۇللاھ بىلەن غازاتقا چىققاندا ساباتلىق بولۇپ، شېھىت بولغانغا قەدەر دادىللىق بىلەن جەڭ قىلىش توغرۇلۇق) اﷲ قا بەرگەن ئەھدىنى ئىشقا ئاشۇرغان نۇرغۇن كىشىلەر بار. ئۇلارنىڭ بەزىسى (ئەھدىگە ۋاپا قىلىپ) شېھىت بولدى، بەزىسى (شېھىت بولۇشنى) كۈتكمەكتە، ئۇلار (پەرۋەردىگارىغا بەرگەن ئەھدىسىنى) ھەرگىز ئۆزگەرتكىنى يوق
لِّيَجْزِىَ ٱللَّهُ ٱلصَّـٰدِقِينَ بِصِدْقِهِمْ وَيُعَذِّبَ ٱلْمُنَـٰفِقِينَ إِن شَآءَ أَوْ يَتُوبَ عَلَيْهِمْ ۚ إِنَّ ٱللَّهَ كَانَ غَفُورًا رَّحِيمًا ﴿٢٤﴾
اﷲ راستچىللارغا راستچىللىقى ئۈچۈن (ئاخىرەتتە ئەڭ ياخشى) مۇكاپات بېرىدۇ، مۇناپىقلارنى (ئەھدىنى بۇزغۇچى مۇناپىقلارنى مۇناپىق پېتى بۇ دۇنيادىن كەتكۈزۈپ) ئەگەر خالىسا ئازابلايدۇ، يا ئۇلارنى تەۋبىگە مۇۋەپپەق قىلىدۇ، اﷲ ئىشەنچلىك ھالدا (بەندىلىرىگە) ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
وَرَدَّ ٱللَّهُ ٱلَّذِينَ كَفَرُوا۟ بِغَيْظِهِمْ لَمْ يَنَالُوا۟ خَيْرًا ۚ وَكَفَى ٱللَّهُ ٱلْمُؤْمِنِينَ ٱلْقِتَالَ ۚ وَكَانَ ٱللَّهُ قَوِيًّا عَزِيزًا ﴿٢٥﴾
اﷲ ھەقىقەتنى رەت قىلغۇچىلار (يەنى مەدىنىگە ھۇجۇم قىلغان ئىتتىپاقداش قوشۇن) نى خاپا قايتۇردى، ئۇلار مەقسىتىگە يېتەلمىدى، اﷲ (دۈشمەنگە بوران ۋە خۇدانىڭ ئەلچى پەرىشتىلىرىنى ئاپىرىدە قىلىپ) مۆمىنلەرنى ئۇرۇشسىز غەلىبىگە ئېرىشتۈردى، اﷲ كۈچلۈكتۇر، غالىبتۇر
وَأَنزَلَ ٱلَّذِينَ ظَـٰهَرُوهُم مِّنْ أَهْلِ ٱلْكِتَـٰبِ مِن صَيَاصِيهِمْ وَقَذَفَ فِى قُلُوبِهِمُ ٱلرُّعْبَ فَرِيقًا تَقْتُلُونَ وَتَأْسِرُونَ فَرِيقًا ﴿٢٦﴾
اﷲ ئەھلى مۇقەددەس كىتاب-ۋەھىيتىن مۇشرىكلارغا ياردەم بەرگەنلەرنى (يەنى رەسۇلۇللاھ بىلەن قىلغان ئەھدىنى بۇزغان بەنى قۇرەيزە يەھۇدىيلىرىنى) قۇرغانلىرىدىن چۈشۈردى، ئۇلارنىڭ دىللىرىغا قورقۇنچ سالدى. (ئۇلارنىڭ) بىر بۆلۈكىنى ئۆلتۈردۇڭلار ۋە بىر بۆلۈكىنى ئەسىر ئالدىڭلار
وَأَوْرَثَكُمْ أَرْضَهُمْ وَدِيَـٰرَهُمْ وَأَمْوَٰلَهُمْ وَأَرْضًا لَّمْ تَطَـُٔوهَا ۚ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ قَدِيرًا ﴿٢٧﴾
اﷲ سىلەرگە ئۇلارنىڭ زېمىنىنى، قورو - جايلىرىنى، مال - مۈلۈكلىرىنى ۋە (تېخى) سىلەرنىڭ قەدىمىڭلار دەسسەپ باقمىغان زېمىننى مىراس قىلىپ بەردى، اﷲ ھەر نەرسىگە قادىردۇر
يَـٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ إِن كُنتُنَّ تُرِدْنَ ٱلْحَيَوٰةَ ٱلدُّنْيَا وَزِينَتَهَا فَتَعَالَيْنَ أُمَتِّعْكُنَّ وَأُسَرِّحْكُنَّ سَرَاحًا جَمِيلًا ﴿٢٨﴾
ئى خۇدانىڭ ئەلچىسى! (ئارتۇق خىراجەت سوراپ سېنى رەنجىتكەن) ئاياللىرىڭغا ئېيتقىنكى، «ئەگەر سىلەر دۇنيا تېرىكچىلىكىنى (يەنى پاراۋان تۇرمۇشنى) ۋە دۇنيانىڭ زىبۇ زىننىتىنى كۆزلىسەڭلار، كېلىڭلار، سىلەرگە بىر ئاز نەرسە بېرەي، سىلەرنى چىرايلىقچە قويۇپ بېرەي
وَإِن كُنتُنَّ تُرِدْنَ ٱللَّهَ وَرَسُولَهُۥ وَٱلدَّارَ ٱلْـَٔاخِرَةَ فَإِنَّ ٱللَّهَ أَعَدَّ لِلْمُحْسِنَـٰتِ مِنكُنَّ أَجْرًا عَظِيمًا ﴿٢٩﴾
ئەگەر سىلەر اﷲ نى ۋە اﷲ نىڭ پەيغەمبىرىنى، ئاخىرەت يۇرتىنى ئىختىيار قىلساڭلار، اﷲ ئىشەنچلىك ھالدا ئىچىڭلاردىكى ياخشى ئىش قىلغۇچىلارغا چوڭ ساۋابنى (يەنى كۆز كۆرمىگەن، قۇلاق ئاڭلىمىغان ۋە ئىنساننىڭ كۆڭلىگە كەلمىگەن نازۇنېمەتلەر بار مەڭگۈلۈك باغلارنى) تەييار قىلدى»
يَـٰنِسَآءَ ٱلنَّبِىِّ مَن يَأْتِ مِنكُنَّ بِفَـٰحِشَةٍ مُّبَيِّنَةٍ يُضَـٰعَفْ لَهَا ٱلْعَذَابُ ضِعْفَيْنِ ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًا ﴿٣٠﴾
ئى خۇدانىڭ ئەلچىسىنىڭ ئاياللىرى! ئىچىڭلاردىن كىمكى ئاشكارا قەبىھ ئىشنى قىلىدىكەن، ئۇنىڭغا ھەسسىلەپ ئازاب قىلىنىدۇ، بۇ اﷲ قا ئاساندۇر
۞ وَمَن يَقْنُتْ مِنكُنَّ لِلَّهِ وَرَسُولِهِۦ وَتَعْمَلْ صَـٰلِحًا نُّؤْتِهَآ أَجْرَهَا مَرَّتَيْنِ وَأَعْتَدْنَا لَهَا رِزْقًا كَرِيمًا ﴿٣١﴾
سىلەردىن كىمكى اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە ئىتائەت قىلسا ۋە ياخشى ئەمەل قىلسا، ئۇنىڭغا ساۋابنى ئىككى قاتلاپ بېرىمىز، ئۇنىڭغا (مەڭگۈلۈك باغلارتە) ئېسىل رىزىق تەييارلىدۇق
يَـٰنِسَآءَ ٱلنَّبِىِّ لَسْتُنَّ كَأَحَدٍ مِّنَ ٱلنِّسَآءِ ۚ إِنِ ٱتَّقَيْتُنَّ فَلَا تَخْضَعْنَ بِٱلْقَوْلِ فَيَطْمَعَ ٱلَّذِى فِى قَلْبِهِۦ مَرَضٌ وَقُلْنَ قَوْلًا مَّعْرُوفًا ﴿٣٢﴾
ئى خۇدانىڭ ئەلچىسىنىڭ ئاياللىرى! سىلەر باشقا ئاياللارنىڭ ھېچبىرىگە ئوخشىمايسىلەر، سىلەر (يات ئەرلەرگە سۆز قىلغاندا) نازاكەت بىلەن سۆز قىلماڭلار، (نازاكەت بىلەن سۆز قىلساڭلار) دىلىدا نىفاق بار ئادەم (سىلەرگە قارىتا) تەمەدە بولۇپ قالىدۇ، (گۇماندىن خالى) ياخشى سۆزنى سۆزلەڭلار
وَقَرْنَ فِى بُيُوتِكُنَّ وَلَا تَبَرَّجْنَ تَبَرُّجَ ٱلْجَـٰهِلِيَّةِ ٱلْأُولَىٰ ۖ وَأَقِمْنَ ٱلصَّلَوٰةَ وَءَاتِينَ ٱلزَّكَوٰةَ وَأَطِعْنَ ٱللَّهَ وَرَسُولَهُۥٓ ۚ إِنَّمَا يُرِيدُ ٱللَّهُ لِيُذْهِبَ عَنكُمُ ٱلرِّجْسَ أَهْلَ ٱلْبَيْتِ وَيُطَهِّرَكُمْ تَطْهِيرًا ﴿٣٣﴾
ئۆيلىرىڭلاردا ئولتۇرۇڭلار، ئىلگىرىكى جاھىلىيەت دەۋرىدىكى ئاياللارنىڭ ياسىنىپ چىققىنىدەك ياسىنىپ چىقماڭلار، ناماز ئوقۇڭلار، زاكات (پاكلاش ئۇلۇشى) بېرىڭلار، اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە ئىتائەت قىلىڭلار، ئى خۇدانىڭ ئەلچىسىنىڭ ئائىلىسىدىكىلەر! اﷲ سىلەردىن گۇناھنى ساقىت قىلىشنى ۋە سىلەرنى تامامەن پاك قىلىشنى خالايدۇ
وَٱذْكُرْنَ مَا يُتْلَىٰ فِى بُيُوتِكُنَّ مِنْ ءَايَـٰتِ ٱللَّهِ وَٱلْحِكْمَةِ ۚ إِنَّ ٱللَّهَ كَانَ لَطِيفًا خَبِيرًا ﴿٣٤﴾
ئۆيلىرىڭلاردا اﷲ نىڭ ئايەتلىرىدىن ۋە ھېكمەت (يەنى خۇدانىڭ ئەلچىسى ئەلەيھىسسالامنىڭ ھەدىسلىرى) تىن ئوقۇلۇۋاتقان نەرسىلەرنى ياد ئېتىڭلار، اﷲ ئىشەنچلىك ھالدا سىرلەرنى بىلگۈچىدۇر، ھەممىدىن خەۋەرداردۇر
إِنَّ ٱلْمُسْلِمِينَ وَٱلْمُسْلِمَـٰتِ وَٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَـٰتِ وَٱلْقَـٰنِتِينَ وَٱلْقَـٰنِتَـٰتِ وَٱلصَّـٰدِقِينَ وَٱلصَّـٰدِقَـٰتِ وَٱلصَّـٰبِرِينَ وَٱلصَّـٰبِرَٰتِ وَٱلْخَـٰشِعِينَ وَٱلْخَـٰشِعَـٰتِ وَٱلْمُتَصَدِّقِينَ وَٱلْمُتَصَدِّقَـٰتِ وَٱلصَّـٰٓئِمِينَ وَٱلصَّـٰٓئِمَـٰتِ وَٱلْحَـٰفِظِينَ فُرُوجَهُمْ وَٱلْحَـٰفِظَـٰتِ وَٱلذَّٰكِرِينَ ٱللَّهَ كَثِيرًا وَٱلذَّٰكِرَٰتِ أَعَدَّ ٱللَّهُ لَهُم مَّغْفِرَةً وَأَجْرًا عَظِيمًا ﴿٣٥﴾
مۇسۇلمان ئەرلەر ۋە مۇسۇلمان ئاياللارغا، مۆمىن ئەرلەر ۋە مۆمىن ئاياللارغا، تائەت - ئىبادەت قىلغۇچى ئەرلەر ۋە تائەت - ئىبادەت قىلغۇچى ئاياللارغا، راستچىل ئەرلەر ۋە راستچىل ئاياللارغا، سەۋر قىلغۇچى ئەرلەر ۋە سەۋر قىلغۇچى ئاياللارغا، خۇدادىن قورققۇچى ئەرلەر ۋە خۇدادىن قورققۇچى ئاياللارغا، سەدىقە بەرگۈچى ئەرلەر ۋە سەدىقە بەرگۈچى ئاياللارغا، روزا تۇتقۇچى ئەرلەر ۋە روزا تۇتقۇچى ئاياللارغا، نەپسىلىرىنى ھارامدىن ساقلىغۇچى ئەرلەر ۋە نەپسىلىرىنى ھارامدىن ساقلىغۇچى ئاياللارغا، اﷲ نى كۆپ زىكرى قىلغۇچى ئەرلەر ۋە اﷲ نى كۆپ زىكرى قىلغۇچى ئاياللارغا اﷲ مەغپىرەت ۋە كاتتا ساۋاب تەييارلىدى
وَمَا كَانَ لِمُؤْمِنٍ وَلَا مُؤْمِنَةٍ إِذَا قَضَى ٱللَّهُ وَرَسُولُهُۥٓ أَمْرًا أَن يَكُونَ لَهُمُ ٱلْخِيَرَةُ مِنْ أَمْرِهِمْ ۗ وَمَن يَعْصِ ٱللَّهَ وَرَسُولَهُۥ فَقَدْ ضَلَّ ضَلَـٰلًا مُّبِينًا ﴿٣٦﴾
اﷲ ۋە ئۇنىڭ پەيغەمبىرى بىرەر ئىشتا ھۆكۈم چىقارغان چاغدا، ئەر - ئايال مۆمىنلەرنىڭ ئۆز ئىشىدا ئىختىيارلىقى بولمايدۇ (يەنى اﷲ ۋە اﷲ نىڭ پەيغەمبىرى بىرەر ئىشتا ھۆكۈم چىقارغان ئىكەن، ھېچ ئادەمنىڭ ئۇنىڭغا مۇخالىپەتچىلىك قىلىشىغا بولمايدۇ)، كىمكى اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە ئاسىيلىق قىلسا، ئىشەنچلىك ھالدا ئۇ ئوپئوچۇق ئازغان بولىدۇ
وَإِذْ تَقُولُ لِلَّذِىٓ أَنْعَمَ ٱللَّهُ عَلَيْهِ وَأَنْعَمْتَ عَلَيْهِ أَمْسِكْ عَلَيْكَ زَوْجَكَ وَٱتَّقِ ٱللَّهَ وَتُخْفِى فِى نَفْسِكَ مَا ٱللَّهُ مُبْدِيهِ وَتَخْشَى ٱلنَّاسَ وَٱللَّهُ أَحَقُّ أَن تَخْشَىٰهُ ۖ فَلَمَّا قَضَىٰ زَيْدٌ مِّنْهَا وَطَرًا زَوَّجْنَـٰكَهَا لِكَىْ لَا يَكُونَ عَلَى ٱلْمُؤْمِنِينَ حَرَجٌ فِىٓ أَزْوَٰجِ أَدْعِيَآئِهِمْ إِذَا قَضَوْا۟ مِنْهُنَّ وَطَرًا ۚ وَكَانَ أَمْرُ ٱللَّهِ مَفْعُولًا ﴿٣٧﴾
ئۆز ۋاقتىدا سەن اﷲ نېمەت بەرگەن، سەنمۇ ئىنئام قىلغان كىشىگە: «خوتۇنۇڭنى نىكاھىڭدا تۇتقىن، اﷲ تىن قورققىن!» دېدىڭ، اﷲ ئاشكارىلىماقچى بولغان نەرسىنى كۆڭلۈڭدە يوشۇردۇڭ، كىشىلەرنىڭ تەنە قىلىشىدىن قورقتۇڭ، اﷲ تىن قۇرقۇشۇڭ ئەڭ ھەقلىق ئىدى. مۆمىنلەرگە ئۇلارنىڭ بالا قىلىۋالغان ئوغۇللىرىنىڭ قويۇپ بەرگەن خوتۇنلىرىنى نىكاھلاپ ئالسا گۇناھ بولماسلىقى ئۈچۈن، زەينەبنى زەيد قويۇۋەتكەندىن كېيىن ساڭا نىكاھلاپ بەردۇق، اﷲ نىڭ (سېنىڭ زەينەبنى ئېلىشىڭ توغرىسىدىكى) ئەمرى چوقۇم ئورۇنلىنىدۇ
مَّا كَانَ عَلَى ٱلنَّبِىِّ مِنْ حَرَجٍ فِيمَا فَرَضَ ٱللَّهُ لَهُۥ ۖ سُنَّةَ ٱللَّهِ فِى ٱلَّذِينَ خَلَوْا۟ مِن قَبْلُ ۚ وَكَانَ أَمْرُ ٱللَّهِ قَدَرًا مَّقْدُورًا ﴿٣٨﴾
اﷲ خۇدانىڭ ئەلچىسىگە ھالال قىلغان ئىشقا ھېچ گۇناھ بولمايدۇ. (بۇ) اﷲ نىڭ ئۆتكەنكى (خۇدانىڭ ئەلچى-رەسۇللىرىگە) تۇتقان يولىدۇر. اﷲ نىڭ ئەمرى ئەزەلدىنلا بېكىپ كەتكەن (ئۆزگەرمەس) ھۆكۈمدۇر
ٱلَّذِينَ يُبَلِّغُونَ رِسَـٰلَـٰتِ ٱللَّهِ وَيَخْشَوْنَهُۥ وَلَا يَخْشَوْنَ أَحَدًا إِلَّا ٱللَّهَ ۗ وَكَفَىٰ بِٱللَّهِ حَسِيبًا ﴿٣٩﴾
ئۇلار (يەنى خۇدانىڭ ئەلچى-رەسۇللىرى) اﷲ نىڭ ئەمر - پەرمانلىرىنى يەتكۈزىدۇ، اﷲ تىن قورقىدۇ، اﷲ تىن باشقا ھېچ كىشىدىن قورقمايدۇ (ئى مۇھەممەد! سەنمۇ اﷲ تىن غەيرىيدىن قورقماسلىقتا شۇ خۇدانىڭ ئەلچى-رەسۇللىرىگە ئەگەشكىن)، اﷲ (ھەممە ئىشلاردىن) ھېساب ئېلىشقا يېتەرلىكتۇر
مَّا كَانَ مُحَمَّدٌ أَبَآ أَحَدٍ مِّن رِّجَالِكُمْ وَلَـٰكِن رَّسُولَ ٱللَّهِ وَخَاتَمَ ٱلنَّبِيِّـۧنَ ۗ وَكَانَ ٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمًا ﴿٤٠﴾
مۇھەممەد ئاراڭلاردىكى ئەرلەردىن ھېچبىرىنىڭ ئاتىسى ئەمەس، لېكىن (ئۇ) اﷲ نىڭ پەيغەمبىر ى ۋە خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ ئاخىرقىسىدۇر (اﷲ ئۇنىڭ بىلەن خۇدانىڭ ئەلچىسىلىكنى ئاخىرلاشتۇرغان، ئۇنىڭدىن كېيىن ھېچقانداق خۇدانىڭ ئەلچىسى كەلمەيدۇ)، اﷲ ھەممە نەرسىنى بىلگۈچىدۇر (سىلەرنىڭ سۆز - ھەرىكىتىڭلاردىن ھېچ نەرسە اﷲ قا مەخپىي ئەمەس)
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱذْكُرُوا۟ ٱللَّهَ ذِكْرًا كَثِيرًا ﴿٤١﴾
ئى مۆمىنلەر! اﷲ نى كۆپ ياد قىلىڭلار
وَسَبِّحُوهُ بُكْرَةً وَأَصِيلًا ﴿٤٢﴾
ئۇنىڭغا ئەتىگەن - ئاخشامدا تەسبىھ ئېيتىڭلار (چۈنكى بۇ ئىككى ۋاقىتتا خۇدانىڭ ئەلچى پەرىشتىلىرى چۈشۈپ تۇرغانلىقتىن، ئۇ ۋاقىتلارنىڭ ئەۋزىلى ھېسابلىنىدۇ)
هُوَ ٱلَّذِى يُصَلِّى عَلَيْكُمْ وَمَلَـٰٓئِكَتُهُۥ لِيُخْرِجَكُم مِّنَ ٱلظُّلُمَـٰتِ إِلَى ٱلنُّورِ ۚ وَكَانَ بِٱلْمُؤْمِنِينَ رَحِيمًا ﴿٤٣﴾
اﷲ سىلەرنى قاراڭغۇلۇقتىن نۇرغا (يەنى گۇمراھلىقتىن ھىدايەتكە) چىقىرىش ئۈچۈن، سىلەرگە رەھمەت قىلىپ تۇرىدۇ، ئۇنىڭ پەرىشتىلىرى سىلەرگە مەغپىرەت تەلەپ قىلىپ تۇرىدۇ، اﷲ مۆمىنلەرگە ناھايىتى مېھرىباندۇر
تَحِيَّتُهُمْ يَوْمَ يَلْقَوْنَهُۥ سَلَـٰمٌ ۚ وَأَعَدَّ لَهُمْ أَجْرًا كَرِيمًا ﴿٤٤﴾
ئۇلار اﷲ قا مۇلاقات بولغان كۈندە ئۇلارغا اﷲ تەرىپىدىن يوللىنىدىغان سالام (سىلەرگە) ئامانلىق بولسۇن (دېگەن سۆزدىن ئىبارەت)، اﷲ ئۇلارغا ئېسىل مۇكاپات تەييارلىدى
يَـٰٓأَيُّهَا ٱلنَّبِىُّ إِنَّآ أَرْسَلْنَـٰكَ شَـٰهِدًا وَمُبَشِّرًا وَنَذِيرًا ﴿٤٥﴾
ئى خۇدانىڭ ئەلچىسى! سېنى بىز ئىشەنچلىك ھالدا گۇۋاھچى، بېشارەتچى، ئاگاھلاندۇرغۇچى، اﷲ نىڭ ئىزنى بىلەن ئۇنىڭ (بىر لىكىگە، تائەت - ئىبادىتىگە) دەۋەت قىلغۇچى ۋە نۇرلۇق چىراق قىلىپ ئەۋەتتۇق
وَدَاعِيًا إِلَى ٱللَّهِ بِإِذْنِهِۦ وَسِرَاجًا مُّنِيرًا ﴿٤٦﴾
ئى خۇدانىڭ ئەلچىسى! سېنى بىز ئىشەنچلىك ھالدا گۇۋاھچى، بېشارەتچى، ئاگاھلاندۇرغۇچى، اﷲ نىڭ ئىزنى بىلەن ئۇنىڭ (بىر لىكىگە، تائەت - ئىبادىتىگە) دەۋەت قىلغۇچى ۋە نۇرلۇق چىراق قىلىپ ئەۋەتتۇق
وَبَشِّرِ ٱلْمُؤْمِنِينَ بِأَنَّ لَهُم مِّنَ ٱللَّهِ فَضْلًا كَبِيرًا ﴿٤٧﴾

وَلَا تُطِعِ ٱلْكَـٰفِرِينَ وَٱلْمُنَـٰفِقِينَ وَدَعْ أَذَىٰهُمْ وَتَوَكَّلْ عَلَى ٱللَّهِ ۚ وَكَفَىٰ بِٱللَّهِ وَكِيلًا ﴿٤٨﴾
ھەقىقەتنى رەت قىلغۇچىلارغا ۋە مۇناپىقلارغا ئىتائەت قىلمىغىن، ئۇلارنىڭ (ساڭا قىلغان) ئازار - كۈلپەتلىرىگە پەرۋا قىلمىغىن (اﷲ قا تەۋەككۈل قىلغىن)، اﷲ (ئۇنىڭغا) تەۋەككۈل قىلغانلارغا يېتەرلىكتۇر

ئى مۆمىنلەر! سىلەر مۆمىن ئاياللارنى ئالساڭلار، ئاندىن ئۇلارغا يېقىنچىلىق قىلىشتىن بۇرۇن ئۇلارنى قويۇۋەتسەڭلار، بۇ چاغدا ئۇلار سىلەر ئۈچۈن ئىددەت تۈتماسلىقى كېرەك، ئۇلارغا ئاز - تولا نەرسە بېرىڭلار، ئۇلارنى چىرايلىقچە قويۇپ بېرىڭلار
يَـٰٓأَيُّهَا ٱلنَّبِىُّ إِنَّآ أَحْلَلْنَا لَكَ أَزْوَٰجَكَ ٱلَّـٰتِىٓ ءَاتَيْتَ أُجُورَهُنَّ وَمَا مَلَكَتْ يَمِينُكَ مِمَّآ أَفَآءَ ٱللَّهُ عَلَيْكَ وَبَنَاتِ عَمِّكَ وَبَنَاتِ عَمَّـٰتِكَ وَبَنَاتِ خَالِكَ وَبَنَاتِ خَـٰلَـٰتِكَ ٱلَّـٰتِى هَاجَرْنَ مَعَكَ وَٱمْرَأَةً مُّؤْمِنَةً إِن وَهَبَتْ نَفْسَهَا لِلنَّبِىِّ إِنْ أَرَادَ ٱلنَّبِىُّ أَن يَسْتَنكِحَهَا خَالِصَةً لَّكَ مِن دُونِ ٱلْمُؤْمِنِينَ ۗ قَدْ عَلِمْنَا مَا فَرَضْنَا عَلَيْهِمْ فِىٓ أَزْوَٰجِهِمْ وَمَا مَلَكَتْ أَيْمَـٰنُهُمْ لِكَيْلَا يَكُونَ عَلَيْكَ حَرَجٌ ۗ وَكَانَ ٱللَّهُ غَفُورًا رَّحِيمًا ﴿٥٠﴾
ئى خۇدانىڭ ئەلچىسى! ساڭا بىز سەن مەھرىلىرىنى بەرگەن ئاياللىرىڭنى، اﷲ ساڭا غەنىمەت قىلىپ بەرگەن، ئەسىر چۈشكەن ئاياللاردىن ئۆزۈڭ مالىك بولغانلارنى ھالال قىلدۇق، ئاتاڭنىڭ ئەر قېرىندىشى ۋە ھەمشىرە قېرىندىشىنىڭ قىزلىرىنى، ئاناڭنىڭ ئەر قېرىندىشى ۋە ھەمشىرە قېرىندىشىنىڭ قىزلىرىنى (ساڭا ھالال قىلدۇق). بۇلار سەن بىلەن بىرلىكتە ھىجرەت قىلدى. ئەگەر بىر مۆمىن خوتۇن ئۆزىنى خۇدانىڭ ئەلچىسىگە بەخش قىلسا، خۇدانىڭ ئەلچىسى ئۇنى (مەھرىسىز) ئېلىشىنى خالىسا، بۇنى مۆمىنلەرگە ئەمەس، يالغۇز ساڭىلا (ھالال قىلدۇق)، ئۇلارنىڭ ئاياللىرى ۋە چۆرىلىرى توغرۇلۇق بىزنىڭ چىقارغان بەلگىلىمىلىرىمىز بىزگە مەلۇمدۇر، ساڭا ھەرەج بولۇپ قالماسلىقى ئۈچۈن (سېنى ئالاھىدىلىكلەرگە ئىگە قىلدۇق)، اﷲ ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
۞ تُرْجِى مَن تَشَآءُ مِنْهُنَّ وَتُـْٔوِىٓ إِلَيْكَ مَن تَشَآءُ ۖ وَمَنِ ٱبْتَغَيْتَ مِمَّنْ عَزَلْتَ فَلَا جُنَاحَ عَلَيْكَ ۚ ذَٰلِكَ أَدْنَىٰٓ أَن تَقَرَّ أَعْيُنُهُنَّ وَلَا يَحْزَنَّ وَيَرْضَيْنَ بِمَآ ءَاتَيْتَهُنَّ كُلُّهُنَّ ۚ وَٱللَّهُ يَعْلَمُ مَا فِى قُلُوبِكُمْ ۚ وَكَانَ ٱللَّهُ عَلِيمًا حَلِيمًا ﴿٥١﴾
ئاياللىرىڭدىن خالىغىنىڭنىڭ نۆۋىتىنى ئارقىغا سۈرسەڭ، خالىغىنىڭ بىلەن (نۆۋىتىدىن باشقا ۋاقىتتا) بىر يەردە بولساڭ بولىۋېرىدۇ، (ۋاقىتلىق) ئايرىلىپ تۇرغان ئاياللىرىڭدىن بىرى بىلەن ھەمتۆشەك بولماقچى بولساڭ، ساڭا ھېچ گۇناھ يوقتۇر، (مانا) بۇ ئۇلارنىڭ خۇشال بولۇشىغا، قايغۇرماسلىقىغا، بەرگەنلىرىڭگە ئۇلارنىڭ ھەممىسىنىڭ رازى بولۇشىغا ئەڭ يېقىندۇر، اﷲ دىلىڭلاردىكىنى بىلىپ تۇرىدۇ، اﷲ ھەممىنى بىلگۈچىدۇر، ھەلىمدۇر (يەنى جازالاشقا ئالدىراپ كەتمەيدۇ)
لَّا يَحِلُّ لَكَ ٱلنِّسَآءُ مِنۢ بَعْدُ وَلَآ أَن تَبَدَّلَ بِهِنَّ مِنْ أَزْوَٰجٍ وَلَوْ أَعْجَبَكَ حُسْنُهُنَّ إِلَّا مَا مَلَكَتْ يَمِينُكَ ۗ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ رَّقِيبًا ﴿٥٢﴾
(ئى خۇدانىڭ ئەلچىسى! ئىلكىڭدىكى ئاياللىرىڭدىن باشقا) يەنە خوتۇن ئالساڭ، ساڭا ھالال بولمايدۇ، باشقا ئاياللارنىڭ جامالى سېنى ئەجەبلەندۈرگەن تەقدىردىمۇ، ئۇلارنى يەڭگۈشلەش (ساڭا ھالال بولمايدۇ). پەقەت سېنىڭ ئىلكىڭدىكى چۆرىلەر بۇنىڭدىن مۇستەسنا، اﷲ ھەممە نەرسىنى كۆزىتىپ تۇرغۇچىدۇر
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَدْخُلُوا۟ بُيُوتَ ٱلنَّبِىِّ إِلَّآ أَن يُؤْذَنَ لَكُمْ إِلَىٰ طَعَامٍ غَيْرَ نَـٰظِرِينَ إِنَىٰهُ وَلَـٰكِنْ إِذَا دُعِيتُمْ فَٱدْخُلُوا۟ فَإِذَا طَعِمْتُمْ فَٱنتَشِرُوا۟ وَلَا مُسْتَـْٔنِسِينَ لِحَدِيثٍ ۚ إِنَّ ذَٰلِكُمْ كَانَ يُؤْذِى ٱلنَّبِىَّ فَيَسْتَحْىِۦ مِنكُمْ ۖ وَٱللَّهُ لَا يَسْتَحْىِۦ مِنَ ٱلْحَقِّ ۚ وَإِذَا سَأَلْتُمُوهُنَّ مَتَـٰعًا فَسْـَٔلُوهُنَّ مِن وَرَآءِ حِجَابٍ ۚ ذَٰلِكُمْ أَطْهَرُ لِقُلُوبِكُمْ وَقُلُوبِهِنَّ ۚ وَمَا كَانَ لَكُمْ أَن تُؤْذُوا۟ رَسُولَ ٱللَّهِ وَلَآ أَن تَنكِحُوٓا۟ أَزْوَٰجَهُۥ مِنۢ بَعْدِهِۦٓ أَبَدًا ۚ إِنَّ ذَٰلِكُمْ كَانَ عِندَ ٱللَّهِ عَظِيمًا ﴿٥٣﴾
ئى مۆمىنلەر! خۇدانىڭ ئەلچىسىنىڭ ئۆيلىرىگە كىرمەڭلار، پەقەت تاماققا چاقىرىلغان ۋاقتىڭلاردىلا كىرىڭلار، (باشقا ۋاقىتتا رۇخسەت بىلەن كىرگەندە) تاماقنىڭ پىشىشىنى كۈتۈپ ئولتۇرماڭلار، لېكىن چاقىرىلغاندا كىرىڭلار يۇ، تاماق يەپ بولغاندىن كېيىن تارقاپ كېتىڭلار، تاماقتىن كېيىن) پاراڭ سېلىشىپ ئولتۇرماڭلار، بۇ (يەنى تاماقتىن كېيىن پاراڭ سېلىشىپ ئولتۇرۇش) خۇدانىڭ ئەلچىسىنى رەنجىتىدۇ، خۇدانىڭ ئەلچىسى (بۇنى ئېيتىشقا) سىلەردىن خىجىل بولىدۇ، اﷲ ھەق (نى ئېيتىش) تىن خىجىل بولمايدۇ، سىلەر خۇدانىڭ ئەلچىسىنىڭ ئاياللىرىدىن بىر نەرسە سوراپ (ئالماقچى) بولساڭلار، پەردە ئارقىسىدىن سوراپ ئېلىڭلار، مۇنداق قىلىش سىلەرنىڭ دىللىرىڭلارنىمۇ، ئۇلارنىڭ دىللىرىنىمۇ ئەڭ پاك تۇتىدۇ، سىلەرنىڭ رەسۇلۇللاھنى رەنجىتىپ قويۇشۇڭلار دۇرۇس ئەمەس، رەسۇلۇللاھتىن كېيىن ئۇنىڭ ئاياللىرىنى ئېلىشىڭلارمۇ مەڭگۈ دۇرۇس ئەمەس، بۇنداق قىلىش (يەنى رەسۇلۇللاھنى رەنجىتىش ۋە ئۇنىڭ ئاياللىرىنى ئېلىش) ئىشەنچلىك ھالدا اﷲ نىڭ دەرگاھىدا چوڭ گۇناھتۇر
إِن تُبْدُوا۟ شَيْـًٔا أَوْ تُخْفُوهُ فَإِنَّ ٱللَّهَ كَانَ بِكُلِّ شَىْءٍ عَلِيمًا ﴿٥٤﴾
بىرەر نەرسىنى ئاشكارىلىساڭلارمۇ، يا (دىلىڭلاردا) يوشۇرساڭلارمۇ (اﷲ ھامان بىلىدۇ)، چۈنكى اﷲ ئىشەنچلىك ھالدا ھەممە نەرسىنى بىلگۈچىدۇر
لَّا جُنَاحَ عَلَيْهِنَّ فِىٓ ءَابَآئِهِنَّ وَلَآ أَبْنَآئِهِنَّ وَلَآ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ أَخَوَٰتِهِنَّ وَلَا نِسَآئِهِنَّ وَلَا مَا مَلَكَتْ أَيْمَـٰنُهُنَّ ۗ وَٱتَّقِينَ ٱللَّهَ ۚ إِنَّ ٱللَّهَ كَانَ عَلَىٰ كُلِّ شَىْءٍ شَهِيدًا ﴿٥٥﴾
ئۇلارغا (يەنى خۇدانىڭ ئەلچىسى ئەلەيھىسسالامنىڭ ئاياللىرىغا) ئاتىلىرى بىلەن، ئوغۇللىرى بىلەن، قېرىنداشلىرى بىلەن، قېرىنداشلىرىنىڭ ئوغۇللىرى بىلەن، ھەمشىرىلىرىنىڭ ئوغۇللىرى بىلەن، مۆمىن ئاياللار بىلەن، قۇل، چۆرىلىرى بىلەن كۆرۈشۈش ھېچ گۇناھ ئەمەس، (ئى خۇدانىڭ ئەلچىسىنىڭ ئاياللىرى!) اﷲ تىن قورقۇڭلار، اﷲ ئىشەنچلىك ھالدا ھەممە نەرسىنى كۆزىتىپ تۇرغۇچىدۇر
إِنَّ ٱللَّهَ وَمَلَـٰٓئِكَتَهُۥ يُصَلُّونَ عَلَى ٱلنَّبِىِّ ۚ يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ صَلُّوا۟ عَلَيْهِ وَسَلِّمُوا۟ تَسْلِيمًا ﴿٥٦﴾
اﷲ ئىشەنچلىك ھالدا خۇدانىڭ ئەلچىسىگە رەھمەت يوللايدۇ (يەنى خۇدانىڭ ئەلچىسى ئەلەيھىسسالامغا رەھمەت قىلىدۇ، شەنىنى ئۇلۇغ قىلىدۇ، دەرىجىسىنى ئۈستۈن قىلىدۇ)، خۇدانىڭ ئەلچى پەرىشتىلىرىمۇ ئىشەنچلىك ھالدا (خۇدانىڭ ئەلچىسىگە) مەغپىرەت تەلەپ قىلىدۇ، ئى مۆمىنلەر! سىلەر خۇدانىڭ ئەلچىسىگە دۇرۇد ئېيتىڭلار ۋە ئامانلىق تىلەڭلار (چۈنكى رەسۇلۇللاھنىڭ سىلەرنىڭ ئۈستۈڭلاردىكى ھەققى چوڭدۇر، ئۇ سىلەرنى گۇمراھلىقتىن ھىدايەتكە، زۇلمەتتىن يورۇقلۇققا چىقارغۇچىدۇر)
إِنَّ ٱلَّذِينَ يُؤْذُونَ ٱللَّهَ وَرَسُولَهُۥ لَعَنَهُمُ ٱللَّهُ فِى ٱلدُّنْيَا وَٱلْـَٔاخِرَةِ وَأَعَدَّ لَهُمْ عَذَابًا مُّهِينًا ﴿٥٧﴾
اﷲ نى ۋە ئۇنىڭ پەيغەمبىرىنى رەنجىتىدىغانلار (يەنى اﷲ ۋە رەسۇلۇللاھ يامان كۆرىدىغان ئىشلارنى قىلىدىغانلار) نى اﷲ دۇنيا ۋە ئاخىرەتتە رەھمىتىدىن يىراق قىلىدۇ ۋە ئۇلارغا خار قىلغۇچى ئازابنى تەييار لايدۇ
وَٱلَّذِينَ يُؤْذُونَ ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَـٰتِ بِغَيْرِ مَا ٱكْتَسَبُوا۟ فَقَدِ ٱحْتَمَلُوا۟ بُهْتَـٰنًا وَإِثْمًا مُّبِينًا ﴿٥٨﴾
مۆمىنلەر ۋە مۆمىنەلەرگە قىلمىغان ئىشلارنى (چاپلاپ) ئۇلارنى رەنجىتىدىغانلار (شۇ) بۇھتاننى ۋە روشەن گۇناھنى ئۈستىگە ئارتىۋالغان بولىدۇ
يَـٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ وَبَنَاتِكَ وَنِسَآءِ ٱلْمُؤْمِنِينَ يُدْنِينَ عَلَيْهِنَّ مِن جَلَـٰبِيبِهِنَّ ۚ ذَٰلِكَ أَدْنَىٰٓ أَن يُعْرَفْنَ فَلَا يُؤْذَيْنَ ۗ وَكَانَ ٱللَّهُ غَفُورًا رَّحِيمًا ﴿٥٩﴾
ئى خۇدانىڭ ئەلچىسى! ئاياللىرىڭغا، قىزلىرىڭغا ۋە مۆمىنلەرنىڭ ئاياللىرىغا ئېيتقىنكى، پۈركەنجى بىلەن بەدىنىنى ئورىۋالسۇن، بۇنداق قىلغاندا ئۇلارنىڭ (ھۈر ئاياللار ئىكەنلىكى) ئەڭ ئوڭاي تونۇلىدۇ - دە، باشقىلار ئۇلارغا چېقىلمايدۇ. اﷲ (بەندىلىرىگە) ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
۞ لَّئِن لَّمْ يَنتَهِ ٱلْمُنَـٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ وَٱلْمُرْجِفُونَ فِى ٱلْمَدِينَةِ لَنُغْرِيَنَّكَ بِهِمْ ثُمَّ لَا يُجَاوِرُونَكَ فِيهَآ إِلَّا قَلِيلًا ﴿٦٠﴾
ئەگەر مۇناپىقلار، (نىفاقتىن) دىللىرىدا كېسەل بارلار، مەدىنىدە يالغان خەۋەر تارقاتقۇچىلار (قىلمىشلىرىدىن) يانمىسا، (ئى مۇھەممەد) ئەلۋەتتە بىز سېنى ئۇلارغا مۇسەللەت قىلىمىز، ئاندىن ئۇلار مەدىنىدە لەنەتكە ئۇچرىغان ھالدا سەن بىلەن ئازغىنا ۋاقىت تۇرالايدۇ (يەنى ئۇلار مەدىنىدىن سۈرگۈن قىلىنىپ ھەيدەپ چىقىرىلىدۇ، چىقىپ كېتىشكە تەييارلىق قىلىۋېلىش ئۈچۈن مەدىنىدە پەقەت ئازغىنا ۋاقىت تۇرالايدۇ). ئۇلار قەيەردە بايقالسا، شۇ يەردە تۇتۇلىدۇ ۋە ئۆلتۈرۈلىدۇ
مَّلْعُونِينَ ۖ أَيْنَمَا ثُقِفُوٓا۟ أُخِذُوا۟ وَقُتِّلُوا۟ تَقْتِيلًا ﴿٦١﴾
ئەگەر مۇناپىقلار، (نىفاقتىن) دىللىرىدا كېسەل بارلار، مەدىنىدە يالغان خەۋەر تارقاتقۇچىلار (قىلمىشلىرىدىن) يانمىسا، (ئى مۇھەممەد) ئەلۋەتتە بىز سېنى ئۇلارغا مۇسەللەت قىلىمىز، ئاندىن ئۇلار مەدىنىدە لەنەتكە ئۇچرىغان ھالدا سەن بىلەن ئازغىنا ۋاقىت تۇرالايدۇ (يەنى ئۇلار مەدىنىدىن سۈرگۈن قىلىنىپ ھەيدەپ چىقىرىلىدۇ، چىقىپ كېتىشكە تەييارلىق قىلىۋېلىش ئۈچۈن مەدىنىدە پەقەت ئازغىنا ۋاقىت تۇرالايدۇ). ئۇلار قەيەردە بايقالسا، شۇ يەردە تۇتۇلىدۇ ۋە ئۆلتۈرۈلىدۇ
سُنَّةَ ٱللَّهِ فِى ٱلَّذِينَ خَلَوْا۟ مِن قَبْلُ ۖ وَلَن تَجِدَ لِسُنَّةِ ٱللَّهِ تَبْدِيلًا ﴿٦٢﴾
بۇ (يەنى مۇناپىقلارغا اﷲ نىڭ تۇتقان يولى) اﷲ نىڭ ئۆتكەنكى ئۈممەتلەرگە تۇتقان يولىدۇر، اﷲ نىڭ تۇتقان يولىدا ھېچقانداق ئۆزگىرىش تاپالمايسەن
يَسْـَٔلُكَ ٱلنَّاسُ عَنِ ٱلسَّاعَةِ ۖ قُلْ إِنَّمَا عِلْمُهَا عِندَ ٱللَّهِ ۚ وَمَا يُدْرِيكَ لَعَلَّ ٱلسَّاعَةَ تَكُونُ قَرِيبًا ﴿٦٣﴾
كىشىلەر سەندىن قىيامەتنىڭ ۋاقتىنى سورايدۇ. «ئۇنى (مەن بىلمەيمەن) پەقەت اﷲ بىلىدۇ» دېگىن. سەن نېمە بىلىسەن، قىيامەت يېقىن بولۇشى مۇمكىن
إِنَّ ٱللَّهَ لَعَنَ ٱلْكَـٰفِرِينَ وَأَعَدَّ لَهُمْ سَعِيرًا ﴿٦٤﴾
اﷲ ئىشەنچلىك ھالدا ھەقىقەتنى رەت قىلغۇچىلارنى رەھمىتىدىن يىراق قىلدى ۋە ئۇلارغا دوزاخنى تەييارلىدى
خَـٰلِدِينَ فِيهَآ أَبَدًا ۖ لَّا يَجِدُونَ وَلِيًّا وَلَا نَصِيرًا ﴿٦٥﴾
ئۇلار دوزاختا مەڭگۈ قالىدۇ، ئۇلار ھېچقانداق (ئۇلارنى قوغدايدىغان) دوست ۋە (ئۇلاردىن ئازابنى دەپئى قىلىدىغان) ياردەمچى تاپالمايدۇ
يَوْمَ تُقَلَّبُ وُجُوهُهُمْ فِى ٱلنَّارِ يَقُولُونَ يَـٰلَيْتَنَآ أَطَعْنَا ٱللَّهَ وَأَطَعْنَا ٱلرَّسُولَا۠ ﴿٦٦﴾
دوزاختا ئۇلارنىڭ يۈزلىرى تۈرۈلۈپ كېتىدىغان كۈندە، ئۇلار: «كاشكى بىز اﷲ قا ئىتائەت قىلغان بولساقچۇ! خۇدانىڭ ئەلچىسىگە ئىتائەت قىلغان بولساقچۇ!» دەيدۇ
وَقَالُوا۟ رَبَّنَآ إِنَّآ أَطَعْنَا سَادَتَنَا وَكُبَرَآءَنَا فَأَضَلُّونَا ٱلسَّبِيلَا۠ ﴿٦٧﴾
ئۇلار: «پەرۋەردىگارىمىز! بىز ئىشەنچلىك ھالدا باشلىقلىرىمىزغا، كاتتىلىرىمىزغا ئىتائەت قىلدۇق، ئۇلار بىزنى توغرا يولدىن ئازدۇردى، پەرۋەردىگارىمىز! ئۇلارغا ئازابنى ئىككى ھەسسە بەرگىن ۋە ئۇلارغا قاتتىق لەنەت قىلغىن» دەيدۇ
رَبَّنَآ ءَاتِهِمْ ضِعْفَيْنِ مِنَ ٱلْعَذَابِ وَٱلْعَنْهُمْ لَعْنًا كَبِيرًا ﴿٦٨﴾
ئۇلار: «پەرۋەردىگارىمىز! بىز ئىشەنچلىك ھالدا باشلىقلىرىمىزغا، كاتتىلىرىمىزغا ئىتائەت قىلدۇق، ئۇلار بىزنى توغرا يولدىن ئازدۇردى، پەرۋەردىگارىمىز! ئۇلارغا ئازابنى ئىككى ھەسسە بەرگىن ۋە ئۇلارغا قاتتىق لەنەت قىلغىن» دەيدۇ
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَكُونُوا۟ كَٱلَّذِينَ ءَاذَوْا۟ مُوسَىٰ فَبَرَّأَهُ ٱللَّهُ مِمَّا قَالُوا۟ ۚ وَكَانَ عِندَ ٱللَّهِ وَجِيهًا ﴿٦٩﴾
ئى مۆمىنلەر! سىلەر مۇساغا ئازار قىلغان كىشىلەر (يەنى بەنى ئىسرائىل) دەك بولماڭلار، اﷲ مۇسانى ئۇلارنىڭ ئېيتقانلىرىدىن (يەنى تۆھمەتلىرىدىن) ئاقلىدى، مۇسا اﷲ نىڭ دەرگاھىدا ئابرۇيلۇق ئىدى
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَقُولُوا۟ قَوْلًا سَدِيدًا ﴿٧٠﴾
ئى مۆمىنلەر! اﷲ تىن قورقۇڭلار، توغرا سۆزنى قىلىڭلار
يُصْلِحْ لَكُمْ أَعْمَـٰلَكُمْ وَيَغْفِرْ لَكُمْ ذُنُوبَكُمْ ۗ وَمَن يُطِعِ ٱللَّهَ وَرَسُولَهُۥ فَقَدْ فَازَ فَوْزًا عَظِيمًا ﴿٧١﴾
اﷲ سىلەرنىڭ ئەمەللىرىڭلارنى تۈزەيدۇ (يەنى سىلەرنى ياخشى ئەمەللەرگە مۇۋەپپەق قىلىدۇ)، گۇناھلىرىڭلارنى مەغپىرەت قىلىدۇ، كىمكى اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە ئىتائەت قىلسا زور مۇۋەپپەقىيەت قازانغان بولىدۇ
إِنَّا عَرَضْنَا ٱلْأَمَانَةَ عَلَى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَٱلْجِبَالِ فَأَبَيْنَ أَن يَحْمِلْنَهَا وَأَشْفَقْنَ مِنْهَا وَحَمَلَهَا ٱلْإِنسَـٰنُ ۖ إِنَّهُۥ كَانَ ظَلُومًا جَهُولًا ﴿٧٢﴾
شۈبھىسىزكى، بىز ئامانەتنى (يەنى پەرزلەرنى ۋە شەرىئەت تەكلىپلىرىنى) ئاسمانلارغا، زېمىنغا ۋە تاغلارغا تەڭلىدۇق، ئۇلار ئۇنى ئۈستىگە ئالمىدى، ئۇنىڭ (ئېغىرلىقى) دىن قورقتى، ئۇنى ئىنسان ئۈستىگە ئالدى، ئىنسان ئىشەنچلىك ھالدا (ئۆزىگە) زۇلۇم قىلغۇچىدۇر، ئىشەنچلىك ھالدا ناداندۇر
لِّيُعَذِّبَ ٱللَّهُ ٱلْمُنَـٰفِقِينَ وَٱلْمُنَـٰفِقَـٰتِ وَٱلْمُشْرِكِينَ وَٱلْمُشْرِكَـٰتِ وَيَتُوبَ ٱللَّهُ عَلَى ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَـٰتِ ۗ وَكَانَ ٱللَّهُ غَفُورًا رَّحِيمًۢا ﴿٧٣﴾
(مەزكۇر ئامانەتكە خىيانەت قىلغانلىقلىرى ئۈچۈن) مۇناپىق ئەرلەرنى، مۇناپىق ئاياللارنى، مۇشرىك ئەرلەرنى، مۇشرىك ئاياللارنى اﷲ ئازابلايدۇ، (مەزكۇر ئامانەتكە رىئايە قىلغانلىقلىرى ئۈچۈن) مۆمىن ئەرلەر ۋە مۆمىن ئاياللارنى اﷲ ئەپۇ قىلىدۇ، (اﷲ مۆمىنلەرگە) ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
