الإسراء
Al-Isra · The Night Journey
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
سُبْحَـٰنَ ٱلَّذِىٓ أَسْرَىٰ بِعَبْدِهِۦ لَيْلًا مِّنَ ٱلْمَسْجِدِ ٱلْحَرَامِ إِلَى ٱلْمَسْجِدِ ٱلْأَقْصَا ٱلَّذِى بَـٰرَكْنَا حَوْلَهُۥ لِنُرِيَهُۥ مِنْ ءَايَـٰتِنَآ ۚ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ﴿١﴾
اﷲ (بارچە نۇقساندىن) پاكتۇر، ئۇ (مۇھەممەد ئەلەيھىسسالامغا) قۇدرىتىمىزنىڭ دەلىللىرىنى كۆرسىتىش ئۈچۈن، بەندىسىنى (يەنى مۇھەممەد ئەلەيھىسسالامنى) بىر كېچىدە مەسجىدى ھەرەمدىن ئەتراپىنى بەرىكەتلىك قىلغان مەسجىدى ئەقساغا ئېلىپ كەلدى. ئىشەنچلىك ھالدا اﷲ ھەممىنى ئاڭلاپ تۇرغۇچىدۇر، ھەممىنى كۆرۈپ تۇرغۇچىدۇر
وَءَاتَيْنَا مُوسَى ٱلْكِتَـٰبَ وَجَعَلْنَـٰهُ هُدًى لِّبَنِىٓ إِسْرَٰٓءِيلَ أَلَّا تَتَّخِذُوا۟ مِن دُونِى وَكِيلًا ﴿٢﴾
مۇساغا مۇقەددەس كىتاب-ۋەھىينى (يەنى تەۋراتنى) ئاتا قىلدۇق، مۇقەددەس كىتاب-ۋەھىينى ئىسرائىل ئەۋلادىغا يېتەكچى قىلدۇق، (ئۇلارغا) مەندىن غەيرىنى ھامىي (يەنى پەرۋەردىگار) قىلىۋالماڭلار!» (دېدۇق)
ذُرِّيَّةَ مَنْ حَمَلْنَا مَعَ نُوحٍ ۚ إِنَّهُۥ كَانَ عَبْدًا شَكُورًا ﴿٣﴾
ئى بىز نۇھ بىلەن بىللە كېمىگە چۈشۈرگەنلەرنىڭ ئەۋلادلىرى! (ئەجدادىڭلارنى غەرق بولۇپ كېتىشتىن قۇتقۇزدۇق، اﷲ نىڭ نېمىتىگە شۈكۈر قىلىڭلار) نۇھ ئىشەنچلىك ھالدا كۆپ شۈكۈر قىلغۇچى بەندە ئىدى
وَقَضَيْنَآ إِلَىٰ بَنِىٓ إِسْرَٰٓءِيلَ فِى ٱلْكِتَـٰبِ لَتُفْسِدُنَّ فِى ٱلْأَرْضِ مَرَّتَيْنِ وَلَتَعْلُنَّ عُلُوًّا كَبِيرًا ﴿٤﴾
ئىسرائىل ئەۋلادىغا مۇقەددەس كىتاب-ۋەھىيتا (يەنى تەۋراتتا): «پەلەستىن زېمىنىدا چوقۇم ئىككى قېتىم بۇزغۇنچىلىق قىلىسىلەر ۋە زۇلۇم قىلىش بىلەن، اﷲ چەكلىگەن ئىشلارنى قىلىش بىلەن تولىمۇ ھەددىڭلاردىن ئېشىپ كېتىسىلەر» دەپ ۋەھيى قىلدۇق
فَإِذَا جَآءَ وَعْدُ أُولَىٰهُمَا بَعَثْنَا عَلَيْكُمْ عِبَادًا لَّنَآ أُو۟لِى بَأْسٍ شَدِيدٍ فَجَاسُوا۟ خِلَـٰلَ ٱلدِّيَارِ ۚ وَكَانَ وَعْدًا مَّفْعُولًا ﴿٥﴾
بىرىنچى قېتىملىق بۇزغۇنچىلىقنىڭ (جازاسى) ۋەدە قىلىنغان ۋاقىت كەلگەندە، (سىلەرنى جازالاش ئۈچۈن) سىلەرگە كۈچلۈك بەندىلىرىمىزنى ئەۋەتتۇق، (ئۇلار سىلەرنى ئۆلتۈرۈش، بۇلاڭ - تالاڭ قىلىش ئۈچۈن) ئۆيلەرنى ئاختۇردى. (سىلەرگە دۈشمەننى مۇسەللەت قىلىش) چوقۇم ئىشقا ئاشىدىغان ۋەدە ئىدى
ثُمَّ رَدَدْنَا لَكُمُ ٱلْكَرَّةَ عَلَيْهِمْ وَأَمْدَدْنَـٰكُم بِأَمْوَٰلٍ وَبَنِينَ وَجَعَلْنَـٰكُمْ أَكْثَرَ نَفِيرًا ﴿٦﴾
ئاندىن (يەنى تەۋبە قىلغىنىڭلاردىن) كېيىن سىلەرنىڭ ئۇلاردىن ئۈستۈنلۈكۈڭلارنى ئەسلىگە كەلتۈردۇق (يەنى دۈشمىنىڭلارنى ھالاك قىلدۇق، سىلەرنى ئۇنىڭ ئۈستىدىن غالىب قىلدۇق)، سىلەرگە نۇرغۇن مال ۋە ئەۋلادلار ئاتا قىلدۇق، سانىڭلارنى كۆپ قىلدۇق
إِنْ أَحْسَنتُمْ أَحْسَنتُمْ لِأَنفُسِكُمْ ۖ وَإِنْ أَسَأْتُمْ فَلَهَا ۚ فَإِذَا جَآءَ وَعْدُ ٱلْـَٔاخِرَةِ لِيَسُـۥٓـُٔوا۟ وُجُوهَكُمْ وَلِيَدْخُلُوا۟ ٱلْمَسْجِدَ كَمَا دَخَلُوهُ أَوَّلَ مَرَّةٍ وَلِيُتَبِّرُوا۟ مَا عَلَوْا۟ تَتْبِيرًا ﴿٧﴾
(ئى ئسرائىل ئەۋلادى!) ئەگەر ياخشىلىق قىلساڭلار، ياخشىلىقىڭلار ئۆزۈڭلار ئۈچۈندۇر، يامانلىق قىلساڭلارمۇ ئۆزۈڭلار ئۈچۈندۇر، ئىككىنچى قېتىملىق بۇزغۇنچىلىق (جازاسىنىڭ) ۋەدە قىلىنغان ۋاقتى كەلگەندە، (دۈشمەنلەرنىڭ خار قىلىشى بىلەن) يۈزۈڭلاردا قايغۇنىڭ ئالامەتلىرى كۆرۈلۈشى ئۈچۈن، ئۇلارنىڭ بەيتۇلمۇقەددەسكە دەسلەپكى قېتىمدا كىرگەندەك كىرىپ (ئۇنى خاراب قىلىشى) ئۈچۈن، ئىشغال قىلغانلىكى يەرنى ۋەيران قىلىشى ئۈچۈن (ئۇلارنى بىز ئەۋەتتۇق)
عَسَىٰ رَبُّكُمْ أَن يَرْحَمَكُمْ ۚ وَإِنْ عُدتُّمْ عُدْنَا ۘ وَجَعَلْنَا جَهَنَّمَ لِلْكَـٰفِرِينَ حَصِيرًا ﴿٨﴾
(ئەگەر تەۋبە قىلساڭلار) پەرۋەردىگارىڭلارنىڭ سىلەرگە رەھىم قىلىشى مۇھەققەق يېقىندۇر، ئەگەر سىلەر قايتساڭلار (يەنى قايتا بۇزغۇنچىلىق قىلساڭلار)، بىز قايتىمىز (يەنى بىز سىلەرنى قايتا جازالايمىز). مەڭگۈلۈك ئوتنى ھەقىقەتنى رەت قىلغۇچىلار ئۈچۈن زىندان قىلدۇق
إِنَّ هَـٰذَا ٱلْقُرْءَانَ يَهْدِى لِلَّتِى هِىَ أَقْوَمُ وَيُبَشِّرُ ٱلْمُؤْمِنِينَ ٱلَّذِينَ يَعْمَلُونَ ٱلصَّـٰلِحَـٰتِ أَنَّ لَهُمْ أَجْرًا كَبِيرًا ﴿٩﴾
بۇ قۇرئان ئىشەنچلىك ھالدا ئەڭ توغرا يولغا باشلايدۇ، ياخشى ئىشلارنى قىلىدىغان مۆمىنلەرگە ئۇلارنىڭ چوڭ مۇكاپاتقا ئېرىشىدىغانلىقى بىلەن خۇش خەۋەر بېرىدۇ
وَأَنَّ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْـَٔاخِرَةِ أَعْتَدْنَا لَهُمْ عَذَابًا أَلِيمًا ﴿١٠﴾
(يەنە مۆمىنلەرگە خۇش خەۋەر بېرىدۇكى) ئاخىرەتكە ئىشەنمەيدىغانلارغا قاتتىق ئازاب تەييارلىدۇق
وَيَدْعُ ٱلْإِنسَـٰنُ بِٱلشَّرِّ دُعَآءَهُۥ بِٱلْخَيْرِ ۖ وَكَانَ ٱلْإِنسَـٰنُ عَجُولًا ﴿١١﴾
ئىنسان (ئۆزىگە ۋە بالا - چاقىلىرىغا) خەيرىلىك دۇئا قىلغاندەك، (ئاچچىقى كەلگەندە) بەتدۇئا قىلىدۇ، ئىنسان ئالدىراڭغۇدۇر
وَجَعَلْنَا ٱلَّيْلَ وَٱلنَّهَارَ ءَايَتَيْنِ ۖ فَمَحَوْنَآ ءَايَةَ ٱلَّيْلِ وَجَعَلْنَآ ءَايَةَ ٱلنَّهَارِ مُبْصِرَةً لِّتَبْتَغُوا۟ فَضْلًا مِّن رَّبِّكُمْ وَلِتَعْلَمُوا۟ عَدَدَ ٱلسِّنِينَ وَٱلْحِسَابَ ۚ وَكُلَّ شَىْءٍ فَصَّلْنَـٰهُ تَفْصِيلًا ﴿١٢﴾
كېچە بىلەن كۈندۈزنى (كامالى قۇدرىتىمىزنى كۆرسىتىدىغان) ئىككى ئالامەت قىلدۇق، كېچىنىڭ ئالامىتىنى كۆتۈرۈۋەتتۇق، پەرۋەردىگارىڭلارنىڭ پەزلىنى تەلەپ قىلىشىڭلار (يەنى تۇرمۇش لازىمەتلىكلىرىنى تېپىش يولىدا سەئيى ھەرىكەتتە بولۇشۇڭلار) ئۈچۈن، (كېچە بىلەن كۈندۈزنىڭ ئالمىشىشى ئارقىلىق كۈنلەرنىڭ، ئايلارنىڭ) يىللارنىڭ سانىنى ۋە ھېسابىنى بىلىشىڭلار ئۈچۈن، كۈندۈزنىڭ ئالامىتىنى يورۇقلۇق قىلدۇق، (دىنىي ۋە دۇنياۋى) ئىشلارنىڭ ھەممىسىنى تەپسىلىي بايان قىلدۇق
وَكُلَّ إِنسَـٰنٍ أَلْزَمْنَـٰهُ طَـٰٓئِرَهُۥ فِى عُنُقِهِۦ ۖ وَنُخْرِجُ لَهُۥ يَوْمَ ٱلْقِيَـٰمَةِ كِتَـٰبًا يَلْقَىٰهُ مَنشُورًا ﴿١٣﴾
ھەر بىر ئىنساننىڭ ئەمەلىنى ئۇنىڭ بوينىغا ئېسىپ قويىمىز (يەنى ئىنساننىڭ قىلغان ھەرقانداق ئەمەلى خۇددى بويۇنچاق بويۇندىن ئايرىلمىغىنىدەك ئۇنىڭدىن ھەرگىز ئايرىلمايدۇ، شۇنىڭغا يارىشا جازا بېرىلىدۇ)، قىيامەت كۈنى ئۇنىڭ نامە - ئەمالىنى كۆرسىتىمىز، ئۇ ئۇنى ئوچۇق. كۆرىدۇ
ٱقْرَأْ كِتَـٰبَكَ كَفَىٰ بِنَفْسِكَ ٱلْيَوْمَ عَلَيْكَ حَسِيبًا ﴿١٤﴾
(ئۇنىڭغا) «نامە - ئەمالىڭنى ئوقۇغىن! بۈگۈن ئۆزۈڭگە (يەنى بۇ قىلمىشلىرىڭغا) ئۆزۈڭ گۇۋاھ بولۇشۇڭ. كۆرىدۇ» دېيىلىدۇ
مَّنِ ٱهْتَدَىٰ فَإِنَّمَا يَهْتَدِى لِنَفْسِهِۦ ۖ وَمَن ضَلَّ فَإِنَّمَا يَضِلُّ عَلَيْهَا ۚ وَلَا تَزِرُ وَازِرَةٌ وِزْرَ أُخْرَىٰ ۗ وَمَا كُنَّا مُعَذِّبِينَ حَتَّىٰ نَبْعَثَ رَسُولًا ﴿١٥﴾
كىمكى ھىدايەت تاپىدىكەن، ھىدايەت تاپقانلىقىنىڭ پايدىسى ئۇنىڭ ئۆزى ئۈچۈندۇر. كىمكى ئازىدىكەن، ئازغانلىقنىڭ زىيىنى ئۇنىڭ ئۆزى ئۈچۈندۇر، بىر ئادەم يەنە بىر ئادەمنىڭ گۇناھىنى ئۈستىگە ئالمايدۇ. خۇدانىڭ ئەلچىسى ئەۋەتمەي تۇرۇپ (ھېچ ئادەمنى جازالىغىنىمىز يوق
وَإِذَآ أَرَدْنَآ أَن نُّهْلِكَ قَرْيَةً أَمَرْنَا مُتْرَفِيهَا فَفَسَقُوا۟ فِيهَا فَحَقَّ عَلَيْهَا ٱلْقَوْلُ فَدَمَّرْنَـٰهَا تَدْمِيرًا ﴿١٦﴾
بىرەر شەھەر (ئاھالىسى) نى ھالاك قىلماقچى بولساق، ئۇنىڭدىكى دۆلەتمەن ئادەملەرنى (بىزگە ئىتائەت قىلىشقا بۇيرۇيمىز، ئۇلار ئىتائەت قىلماي، پىسقى - پاسات قىلىدۇ - دە، ئۇلارغا ئازابىمىز تېگىشلىك بولىدۇ، شۇنىڭ بىلەن ئۇلارنى قورقۇنچلۇق رەۋىشتە ھالاك قىلىمىز
وَكَمْ أَهْلَكْنَا مِنَ ٱلْقُرُونِ مِنۢ بَعْدِ نُوحٍ ۗ وَكَفَىٰ بِرَبِّكَ بِذُنُوبِ عِبَادِهِۦ خَبِيرًۢا بَصِيرًا ﴿١٧﴾
نۇھتىن كېيىن (خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلغان) نۇرغۇن ئۈممەتلەرنى ھالاك قىلدۇق، پەرۋەردىگارىڭ بەندىلىرىنىڭ گۇناھلىرىنى تولۇق بىلىپ تۇرۇشتا، كۆرۈپ تۇرۇشتا يېتەرلىكتۇر
مَّن كَانَ يُرِيدُ ٱلْعَاجِلَةَ عَجَّلْنَا لَهُۥ فِيهَا مَا نَشَآءُ لِمَن نُّرِيدُ ثُمَّ جَعَلْنَا لَهُۥ جَهَنَّمَ يَصْلَىٰهَا مَذْمُومًا مَّدْحُورًا ﴿١٨﴾
بىز دۇنيانى كۆزلىگەنلەر ئىچىدىن خالىغان ئادەمگە (ئۇنىڭ خالىغىنىنى ئەمەس) بىزنىڭ خالىغىنىمىزنى بېرىمىز، ئاندىن ئۇنى (ئاخىرەتتە) مەڭگۈلۈك ئوت بىلەن جازالايمىز، مەڭگۈلۈك ئوتگە ئۇ خارلانغان، (اﷲ نىڭ رەھمىتىدىن) قوغلانغان ھالدا كىرىدۇ
وَمَنْ أَرَادَ ٱلْـَٔاخِرَةَ وَسَعَىٰ لَهَا سَعْيَهَا وَهُوَ مُؤْمِنٌ فَأُو۟لَـٰٓئِكَ كَانَ سَعْيُهُم مَّشْكُورًا ﴿١٩﴾
كىمكى ئاخىرەتنى كۆزلەيدىكەن ۋە مۆمىن بولۇپ ئاخىرەتكە لايىق ئەمەللەرنى قىلىدىكەن، مۇنداق ئادەملەرنىڭ قىلغان ئەمەلى (اﷲ نىڭ دەرگاھىدا) قوبۇل بولىدۇ
كُلًّا نُّمِدُّ هَـٰٓؤُلَآءِ وَهَـٰٓؤُلَآءِ مِنْ عَطَآءِ رَبِّكَ ۚ وَمَا كَانَ عَطَآءُ رَبِّكَ مَحْظُورًا ﴿٢٠﴾
بۇلارنىڭ ۋە ئۇلارنىڭ ھەر ئىككىسىگە پەرۋەردىگارىڭنىڭ ئاتا قىلغىنىدىن بېرىمىز. پەرۋەردىگارىڭنىڭ ئاتا قىلغىنى (ھېچ ئادەمدىن) مەنئى قىلىنمايدۇ
ٱنظُرْ كَيْفَ فَضَّلْنَا بَعْضَهُمْ عَلَىٰ بَعْضٍ ۚ وَلَلْـَٔاخِرَةُ أَكْبَرُ دَرَجَـٰتٍ وَأَكْبَرُ تَفْضِيلًا ﴿٢١﴾
ئۇلارنىڭ بەزىسىنى بەزىسىدىن قانداق ئۈستۈن قىلغانلىقىمىزغا قارىغىن، ئاخىرەت دەرىجىلىرىدىكى (پەرق) ئەلۋەتتە تېخىمۇ چوڭدۇر ۋە ئۈستۈنلۈك (جەھەتتىكى پەرق) زىيادە چوڭدۇر

سىلەر تاش ياكى تۆمۈر ياكى ھاياتلىقنى تەسەۋۋۇر قىلىش تېخىمۇ قىيىن بولغان بىر نەرسە بولۇپ كەتسەڭلارمۇ (اﷲ سىلەرنى تىرىلدۈرىدۇ) ئۇلار: ‹‹بىزنى كىم تىرىلدۈرىدۇ؟» دېگىن، ئۇلار ساڭا باشلىرىنى لىڭشىتىپ: ‹‹ئۇ قاچان بولىدۇ؟ ›› دېيىشىدۇ، (سەن) ‹‹ئۈمىدكى، ئۇ يېقىندا بولىدۇ›› دېگىن

سىلەر تاش ياكى تۆمۈر ياكى ھاياتلىقنى تەسەۋۋۇر قىلىش تېخىمۇ قىيىن بولغان بىر نەرسە بولۇپ كەتسەڭلارمۇ (اﷲ سىلەرنى تىرىلدۈرىدۇ) ئۇلار: ‹‹بىزنى كىم تىرىلدۈرىدۇ؟» دېگىن، ئۇلار ساڭا باشلىرىنى لىڭشىتىپ: ‹‹ئۇ قاچان بولىدۇ؟ ›› دېيىشىدۇ، (سەن) ‹‹ئۈمىدكى، ئۇ يېقىندا بولىدۇ›› دېگىن

رَّبُّكُمُ ٱلَّذِى يُزْجِى لَكُمُ ٱلْفُلْكَ فِى ٱلْبَحْرِ لِتَبْتَغُوا۟ مِن فَضْلِهِۦٓ ۚ إِنَّهُۥ كَانَ بِكُمْ رَحِيمًا ﴿٦٦﴾
پەرۋەردىگارىڭلار، سىلەرنى پەزلىنى (يەنى پەزلىدىن بولغان رىزقىڭلارنى) تەلەپ قىلسۇن دەپ، كېمىلەرنى سىلەر ئۈچۈن دېڭىزدا ماڭدۇرۇپ بېرىدۇ، ئۇلار ئىشەنچلىك ھالدا سىلەرگە ناھايىتى مېھرىباندۇر
وَإِذَا مَسَّكُمُ ٱلضُّرُّ فِى ٱلْبَحْرِ ضَلَّ مَن تَدْعُونَ إِلَّآ إِيَّاهُ ۖ فَلَمَّا نَجَّىٰكُمْ إِلَى ٱلْبَرِّ أَعْرَضْتُمْ ۚ وَكَانَ ٱلْإِنسَـٰنُ كَفُورًا ﴿٦٧﴾
سىلەر دېڭىزدا بىرەر ئاپەتكە يولۇققان چېغىڭلاردا، ئېسىڭلارغا سىلەر چوقۇنۇۋاتقان مەبۇدلار كەلمەي، پەقەت اﷲ لا كېلىدۇ (يەنى ئىنسان مۇنداق چاغدا اﷲ تىن باشقىغا ئىلتىجا قىلمايدۇ)، اﷲ سىلەرنى (غەرق بولۇشتىن) قۇتقۇزۇپ، (ئامان - ئېسەن) قۇرۇقلۇققا چىقارغاندا يۈز ئۆرۈيسىلەر (يەنى ئىخلاسىڭلار تۈگەپ ئاۋۋالقى ھالىتىڭلارغا قايتىۋالىسىلەر). ئىنسان كۇفرىلىق قىلغۇچىدۇر (يەنى اﷲ نىڭ نېمەتلىرىنى ئىنكار قىلىش ئىنساننىڭ تەبىئىتىدۇر)
أَفَأَمِنتُمْ أَن يَخْسِفَ بِكُمْ جَانِبَ ٱلْبَرِّ أَوْ يُرْسِلَ عَلَيْكُمْ حَاصِبًا ثُمَّ لَا تَجِدُوا۟ لَكُمْ وَكِيلًا ﴿٦٨﴾
(دېڭىزدا غەرق بولۇپ كېتىشتىن قۇتۇلغىنىڭلار بىلەن) اﷲ نىڭ سىلەرنى يەرگە يۇتقۇزۇۋېتىشىدىن، ياكى سىلەرگە ئاسماندىن تاش ياغدۇرۇپ ھالاك قىلىشىدىن، ئاندىن سىلەرنى (اﷲ نىڭ ئازابىدىن ساقلايدىغان) ھېچ ھامىي تاپالماسلىقىڭلاردىن قورقمامسىلەر؟
أَمْ أَمِنتُمْ أَن يُعِيدَكُمْ فِيهِ تَارَةً أُخْرَىٰ فَيُرْسِلَ عَلَيْكُمْ قَاصِفًا مِّنَ ٱلرِّيحِ فَيُغْرِقَكُم بِمَا كَفَرْتُمْ ۙ ثُمَّ لَا تَجِدُوا۟ لَكُمْ عَلَيْنَا بِهِۦ تَبِيعًا ﴿٦٩﴾
ياكى اﷲ نىڭ سىلەرنى دېڭىزدا قايتا قاتناتقۇزۇپ (دېڭىزدىكى چېغىڭلاردا) قارا بوران چىقىرىپ، كۈفرىلىق قىلغانلىقىڭلار تۈپەيلىدىن سىلەرنى غەرق قىلىۋېتىشىدىن، ئاندىن سىلەرگە بىزدىن ئۆچ ئېلىپ بېرىدىغان ھېچ ئەھدى تاپالماسلىقىڭلاردىن قورقمامسىلەر؟
۞ وَلَقَدْ كَرَّمْنَا بَنِىٓ ءَادَمَ وَحَمَلْنَـٰهُمْ فِى ٱلْبَرِّ وَٱلْبَحْرِ وَرَزَقْنَـٰهُم مِّنَ ٱلطَّيِّبَـٰتِ وَفَضَّلْنَـٰهُمْ عَلَىٰ كَثِيرٍ مِّمَّنْ خَلَقْنَا تَفْضِيلًا ﴿٧٠﴾
شەك - شۈبھىسىزكى، بىز ئادەم بالىلىرىنى ھۆرمەتلىك قىلدۇق، ئۇلارنى قۇرۇقلۇقتا (ئۇلۇغلارغا) مىندۇردۇق، دېڭىزدا (كېمىلەرگە) چىقاردۇق، ئۇلارنى شېرىن يېمەكلىكلەر بىلەن رىزىقلاندۇردۇق، ئۇلارنى مەخلۇقاتلىرىمىزنىڭ نۇرغۇنىدىن ئۈستۈن قىلدۇق
يَوْمَ نَدْعُوا۟ كُلَّ أُنَاسٍۭ بِإِمَـٰمِهِمْ ۖ فَمَنْ أُوتِىَ كِتَـٰبَهُۥ بِيَمِينِهِۦ فَأُو۟لَـٰٓئِكَ يَقْرَءُونَ كِتَـٰبَهُمْ وَلَا يُظْلَمُونَ فَتِيلًا ﴿٧١﴾
بىز ھەممە ئادەمنى ئۇلارنىڭ نامە - ئەمالى بىلەن چاقىرىدىغان كۈننى (ئېسىڭدا تۇتقىن)، نامە - ئەمالى ئوڭ قولىدىن بېرىلگەنلەر ئۆزلىرىنىڭ نامە - ئەمالىنى ئوقۇيدۇ، ئۇلارغا قىلچە زورلۇق قىلىنمايدۇ
وَمَن كَانَ فِى هَـٰذِهِۦٓ أَعْمَىٰ فَهُوَ فِى ٱلْـَٔاخِرَةِ أَعْمَىٰ وَأَضَلُّ سَبِيلًا ﴿٧٢﴾
بۇ دۇنيادا (دىلى) كۆر بولغان ئادەم ئاخىرەتتىمۇ كۆر ۋە تېخىمۇ گۇمراھ بولىدۇ
وَإِن كَادُوا۟ لَيَفْتِنُونَكَ عَنِ ٱلَّذِىٓ أَوْحَيْنَآ إِلَيْكَ لِتَفْتَرِىَ عَلَيْنَا غَيْرَهُۥ ۖ وَإِذًا لَّٱتَّخَذُوكَ خَلِيلًا ﴿٧٣﴾
ئۇلار سېنى بىز ساڭا ۋەھىي قىلغان قۇرئانغا خىلاپلىق قىلدۇرۇپ، بىزگە قۇرئاندىن غەيرىنى ئىپتىرا قىلدۇرۇشقا تاس قالدى، (ئۇلارنىڭ كۆڭلىدىكىدەك قىلغىنىڭدا) ئەلۋەتتە ئۇلار سېنى (ئۆزلىرىگە) دوست قىلىۋالاتتى
وَلَوْلَآ أَن ثَبَّتْنَـٰكَ لَقَدْ كِدتَّ تَرْكَنُ إِلَيْهِمْ شَيْـًٔا قَلِيلًا ﴿٧٤﴾
سېنى بىز (ھەقتە تۇرۇشتا) مۇستەھكەم قىلمىغان بولساق، ئۇلارغا مايىل بولۇپ كەتكىلى ئىشەنچلىك ھالدا تاس قالغان ئىدىڭ
إِذًا لَّأَذَقْنَـٰكَ ضِعْفَ ٱلْحَيَوٰةِ وَضِعْفَ ٱلْمَمَاتِ ثُمَّ لَا تَجِدُ لَكَ عَلَيْنَا نَصِيرًا ﴿٧٥﴾
ئەگەر شۇنداق قىلىدىغان بولساڭ، ئەلۋەتتە ساڭا دۇنيا ۋە ئاخىرەتتە ھەسسىلەپ ئازاب بەرگەن بولاتتۇق، ئاندىن سەن بىزگە قارشى ياردەم بېرىدىغان ھېچ ئادەم تاپالمايتتىڭ
وَإِن كَادُوا۟ لَيَسْتَفِزُّونَكَ مِنَ ٱلْأَرْضِ لِيُخْرِجُوكَ مِنْهَا ۖ وَإِذًا لَّا يَلْبَثُونَ خِلَـٰفَكَ إِلَّا قَلِيلًا ﴿٧٦﴾
(يەھۇدىيلار) سېنى (مەدىنە) زېمىنىدىن چىقىرىۋېتىش ئۈچۈن، ئۇ يەردىن سېنى خاتىرجەم تۇرغۇزماسلىققا تاس قالدى، شۇنداق بولغاندا، سەن چىقىپ كەتكەندىن كېيىن، ئۇلار ئازغىنا ۋاقىت تۇرالايتتى (يەنى ئازغىنا ۋاقىتتىن كېيىن ھالاك بولاتتى)
سُنَّةَ مَن قَدْ أَرْسَلْنَا قَبْلَكَ مِن رُّسُلِنَا ۖ وَلَا تَجِدُ لِسُنَّتِنَا تَحْوِيلًا ﴿٧٧﴾
(بۇ) سەندىن ئىلگىرى بىز ئەۋەتكەن خۇدانىڭ ئەلچىسىلىرىمىزنىڭ تۇتقان يولىدۇر (يەنى خۇدانىڭ ئەلچىسىلىرىنى ئارىسىدىن ھەيدەپ چىقارغان ھەرقانداق ئۈممەتنى ھالاك قىلىش اﷲ نىڭ خۇدانىڭ ئەلچىسىلىرى ھەققىدە تۇتقان يولىدۇر)، تۇتقان يولىمىزدا ئۆزگىرىش تاپالمايسەن
أَقِمِ ٱلصَّلَوٰةَ لِدُلُوكِ ٱلشَّمْسِ إِلَىٰ غَسَقِ ٱلَّيْلِ وَقُرْءَانَ ٱلْفَجْرِ ۖ إِنَّ قُرْءَانَ ٱلْفَجْرِ كَانَ مَشْهُودًا ﴿٧٨﴾
(ئى مۇھەممەد!) كۈن قايرىلغان ۋاقىتتىن تارتىپ قاراڭغۇ چۈشكەن ۋاقىتقا قەدەر ناماز ئوقۇغىن. بامدات نامىزىنىمۇ ئوقۇغىن، (كۈندۈزنىڭ ۋە كېچىنىڭ پەرىشتىلىرى) چوقۇم بامدات نامىزىدا ھازىر بولىدۇ
وَمِنَ ٱلَّيْلِ فَتَهَجَّدْ بِهِۦ نَافِلَةً لَّكَ عَسَىٰٓ أَن يَبْعَثَكَ رَبُّكَ مَقَامًا مَّحْمُودًا ﴿٧٩﴾
ساڭا نەپلە (ئىبادەت) بولۇشى ئۈچۈن، كېچىنىڭ بىر قىسىمىدا (ناماز ئوقۇشقا، قۇرئان ئوقۇشقا) ئويغانغىن، (ئى مۇھممەد‍! (پەرۋەردىگارىڭنىڭ سېنى مەدھىيىلىنىدىغان ئورۇنغا (يەنى بۈيۈك شاپائەت ماقامىغا) تۇرغۇزۇشى ھەقىقەتتۇر
وَقُل رَّبِّ أَدْخِلْنِى مُدْخَلَ صِدْقٍ وَأَخْرِجْنِى مُخْرَجَ صِدْقٍ وَٱجْعَل لِّى مِن لَّدُنكَ سُلْطَـٰنًا نَّصِيرًا ﴿٨٠﴾

وَلَقَدْ صَرَّفْنَا لِلنَّاسِ فِى هَـٰذَا ٱلْقُرْءَانِ مِن كُلِّ مَثَلٍ فَأَبَىٰٓ أَكْثَرُ ٱلنَّاسِ إِلَّا كُفُورًا ﴿٨٩﴾
بۇ قۇرئاندا ئىنسانلار ئۈچۈن تۈرلۈك مەسىللەرنى خىلمۇ - خىل شەكىلدە بايان قىلدۇق، (روشەن پاكىتلارنىڭ بولىشىغا قارىماي) ئىنسانلارنىڭ تولىسى پەقەت (ھەقىقەتنى) ئىنكار قىلىدۇ
وَقَالُوا۟ لَن نُّؤْمِنَ لَكَ حَتَّىٰ تَفْجُرَ لَنَا مِنَ ٱلْأَرْضِ يَنۢبُوعًا ﴿٩٠﴾
ئۇلار ئېيتتى: «تاكى سەن بىزگە يەر ئاستىدىن بىر بۇلاقنى ئېتىلدۇرۇپ چىقارمىغىچە ساڭا ھەرگىز ئىشەنمەيمىز
أَوْ تَكُونَ لَكَ جَنَّةٌ مِّن نَّخِيلٍ وَعِنَبٍ فَتُفَجِّرَ ٱلْأَنْهَـٰرَ خِلَـٰلَهَا تَفْجِيرًا ﴿٩١﴾
ياكى سېنىڭ خورمىلىق، تاللىق ۋە ئوتتۇرىسىدىن ئۆستەڭلەر ئېقىپ تۇرىدىغان بىر بېغىڭ بولسۇن
أَوْ تُسْقِطَ ٱلسَّمَآءَ كَمَا زَعَمْتَ عَلَيْنَا كِسَفًا أَوْ تَأْتِىَ بِٱللَّهِ وَٱلْمَلَـٰٓئِكَةِ قَبِيلًا ﴿٩٢﴾
ياكى سەن ئېيتقاندەك ئۈستىمىزگە ئاسماننى پارچە - پارچە قىلىپ چۈشۈرگىن، ياكى (خۇدانىڭ ئەلچىسىلىكىڭگە) گۇۋاھچى قىلىپ اﷲ نى ۋە خۇدانىڭ ئەلچى پەرىشتىلىرىنى ئالدىمىزغا كەلتۈرگىن
أَوْ يَكُونَ لَكَ بَيْتٌ مِّن زُخْرُفٍ أَوْ تَرْقَىٰ فِى ٱلسَّمَآءِ وَلَن نُّؤْمِنَ لِرُقِيِّكَ حَتَّىٰ تُنَزِّلَ عَلَيْنَا كِتَـٰبًا نَّقْرَؤُهُۥ ۗ قُلْ سُبْحَانَ رَبِّى هَلْ كُنتُ إِلَّا بَشَرًا رَّسُولًا ﴿٩٣﴾
ياكى سېنىڭ ئالتۇندىن ئۆيۈڭ بولسۇن، ياكى سەن (شوتا بىلەن) ئاسمانغا چىققىن، تاكى (اﷲ تەرىپىدىن سېنىڭ ئۇنىڭ بەندىسى ۋە خۇدانىڭ ئەلچىسىلىكى ئىكەنلىكىڭ يېزىلغان) بىز ئوقۇيالايدىغان بىر مۇقەددەس كىتاب-ۋەھىينى ئېلىپ چۈشمىگىچە سېنىڭ ئاسمانغا چىققانلىقىڭغا ھەرگىز ئىشەنمەيمىز». (ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «اﷲ پاكتۇر، مەن پەقەت بىر خۇدانىڭ ئەلچىسى ئىنسانمەن (اﷲ نىڭ ئىزنى بولمىسا ھېچ ئادەم بىرەر مۆجىزە كەلتۈرەلمەيدۇ)»
وَمَا مَنَعَ ٱلنَّاسَ أَن يُؤْمِنُوٓا۟ إِذْ جَآءَهُمُ ٱلْهُدَىٰٓ إِلَّآ أَن قَالُوٓا۟ أَبَعَثَ ٱللَّهُ بَشَرًا رَّسُولًا ﴿٩٤﴾
ئۇلار (يەنى مۇشرىكلار) نىڭ ئۆزلىرىگە ھىدايەت (يەنى خۇدانىڭ ئەلچىسى بىلەن قۇرئان) كەلگەن چاغدا ئىمان ئېيتماسلىقلىرى «اﷲ ئىنساننى خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتەمدۇ؟» دېگەنلىكلىرى ئۈچۈندۇر
قُل لَّوْ كَانَ فِى ٱلْأَرْضِ مَلَـٰٓئِكَةٌ يَمْشُونَ مُطْمَئِنِّينَ لَنَزَّلْنَا عَلَيْهِم مِّنَ ٱلسَّمَآءِ مَلَكًا رَّسُولًا ﴿٩٥﴾
(ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «مۇبادا زېمىندا (يەنى زېمىن ئەھلىدەك) مېڭىپ يۈرۈيدىغان، ئولتۇراقلاشقان خۇدانىڭ ئەلچى پەرىشتىلىرى بولسا ئىدى، ئەلۋەتتە ئۇلارغا ئاسماندىن پەرىشتىنى خۇدانىڭ ئەلچىسى قىلىپ چۈشۈرەتتۇق»
قُلْ كَفَىٰ بِٱللَّهِ شَهِيدًۢا بَيْنِى وَبَيْنَكُمْ ۚ إِنَّهُۥ كَانَ بِعِبَادِهِۦ خَبِيرًۢا بَصِيرًا ﴿٩٦﴾
(ئى مۇھەممەد‍! (ئېيتقىنكى، «اﷲ مەن بىلەن سىلەرنىڭ ئاراڭلاردا مېنىڭ راستلىقىمغا گۇۋاھ بولۇشقا يېتەرلىكتۇر. اﷲ ئىشەنچلىك ھالدا بەندىلىرىنىڭ (ئەھۋالىنى) بىلىپ تۇرغۇچىدۇر. كۆرۈپ تۇرغۇچىدۇر»
وَمَن يَهْدِ ٱللَّهُ فَهُوَ ٱلْمُهْتَدِ ۖ وَمَن يُضْلِلْ فَلَن تَجِدَ لَهُمْ أَوْلِيَآءَ مِن دُونِهِۦ ۖ وَنَحْشُرُهُمْ يَوْمَ ٱلْقِيَـٰمَةِ عَلَىٰ وُجُوهِهِمْ عُمْيًا وَبُكْمًا وَصُمًّا ۖ مَّأْوَىٰهُمْ جَهَنَّمُ ۖ كُلَّمَا خَبَتْ زِدْنَـٰهُمْ سَعِيرًا ﴿٩٧﴾
كىمكى اﷲ ھىدايەت قىلغان ئىكەن ئۇ ھىدايەت تاپقۇچىدۇر، اﷲ گۇمراھ قىلغان ئادەمگە اﷲ تىن باشقا (ئازابتىن قۇتۇلدۇرىدىغان) ھېچ ياردەمچىلەرنى تاپالمايسەن، قىيامەت كۈنى ئۇلارنى دۈم ياتقۇزۇپ (سۆرەلگەن) كۆر، گاچا، گاس ھالدا يىغىمىز، ئۇلارنىڭ جايى مەڭگۈلۈك ئوت بولىدۇ، (مەڭگۈلۈك ئوتنىڭ) ئوتى پەسلەپ قالسا، ئۇلارغا (ئوتنى) تېخىمۇ يالقۇنجىتىمىز
ذَٰلِكَ جَزَآؤُهُم بِأَنَّهُمْ كَفَرُوا۟ بِـَٔايَـٰتِنَا وَقَالُوٓا۟ أَءِذَا كُنَّا عِظَـٰمًا وَرُفَـٰتًا أَءِنَّا لَمَبْعُوثُونَ خَلْقًا جَدِيدًا ﴿٩٨﴾
ئۇلارنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلىقلىرى ۋە «چىرىگەن سۆڭەك ۋە توپىغا ئايلانغىنىمىزدىن كېيىن راستلا قايتىدىن تېرىلىپ يېڭىدىن ئادەم بولامدۇق؟» دېگەنلىكلىرى ئۈچۈن، ئۇلارغا بېرىلگەن جازا ئەنە شۇدۇر
۞ أَوَلَمْ يَرَوْا۟ أَنَّ ٱللَّهَ ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ قَادِرٌ عَلَىٰٓ أَن يَخْلُقَ مِثْلَهُمْ وَجَعَلَ لَهُمْ أَجَلًا لَّا رَيْبَ فِيهِ فَأَبَى ٱلظَّـٰلِمُونَ إِلَّا كُفُورًا ﴿٩٩﴾
ئاسمانلارنى، زېمىننى ياراتقات اﷲ نىڭ ئۇلارغا ئوخشاش ئىنسانلارنى يارىتالايدىغانلىقىنى ئۇلار بىلمەمدۇ؟ اﷲ ئۇلارنىڭ (ئۆلۈشى ۋە تىرىلىشى ئۈچۈن) شەك - شۈبھىسىز مۇئەييەن ۋاقىت بەلگىلىدى، ھەقىقەتنى رەت قىلغۇچىلار پەقەت كۇفرىغىلا ئۇنايدۇ
قُل لَّوْ أَنتُمْ تَمْلِكُونَ خَزَآئِنَ رَحْمَةِ رَبِّىٓ إِذًا لَّأَمْسَكْتُمْ خَشْيَةَ ٱلْإِنفَاقِ ۚ وَكَانَ ٱلْإِنسَـٰنُ قَتُورًا ﴿١٠٠﴾
«ئەگەر پەرۋەردىگارىمنىڭ رەھمەت خەزىنىلىرى سىلەرنىڭ قولۇڭلاردا بولسا، چىقىمدىن (يەنى تۈگەپ كېتىشىدىن) قورقۇپ، چوقۇم بېخىللىق قىلاتتىڭلار، ئىنسان بېخىل كېلىدۇ» دېگىن
وَلَقَدْ ءَاتَيْنَا مُوسَىٰ تِسْعَ ءَايَـٰتٍۭ بَيِّنَـٰتٍ ۖ فَسْـَٔلْ بَنِىٓ إِسْرَٰٓءِيلَ إِذْ جَآءَهُمْ فَقَالَ لَهُۥ فِرْعَوْنُ إِنِّى لَأَظُنُّكَ يَـٰمُوسَىٰ مَسْحُورًا ﴿١٠١﴾
بىز مۇساغا ئىشەنچلىك ھالدا توققۇز روشەن مۆجىزە ئاتا قىلدۇق، مۇسا پىرئەۋنگە ۋە ئۇنىڭ قەۋمىگە كەلگەن چاغدا بىز مۇساغا: «(ئۇلاردىن) ئىسرائىل ئەۋلادىنى (قويۇۋېتىشنى) سۆرىغىن» (دېدۇق). پىرئەۋن ئۇنىڭغا: «ئى مۇسا! سېنى مەن چوقۇم جادۇ قىلىنغان دەپ ئويلايمەن» دېدى
قَالَ لَقَدْ عَلِمْتَ مَآ أَنزَلَ هَـٰٓؤُلَآءِ إِلَّا رَبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ بَصَآئِرَ وَإِنِّى لَأَظُنُّكَ يَـٰفِرْعَوْنُ مَثْبُورًا ﴿١٠٢﴾
مۇسا: «(ئى پىرئەۋن!) بۇلار (يەنى توققۇز مۆجىزە) نى پەقەت ئاسمانلارنىڭ ۋە زېمىننىڭ پەرۋەردىگارى (مېنىڭ راستلىقىمنى ئىسپاتلايدىغان) دەلىل قىلىپ نازىل قىغانلىقىنى سەن (گەرچە تېلىڭدا ئىقرار قىلمىساڭمۇ دىلىڭدا ئەلۋەتتە) ئوبدان بىلىسەن، مەن سېنى، ئى پىرئەۋن، ئىشەنچلىك ھالدا، ھالاك بولىدۇ دەپ ئويلايمەن» دېدى
فَأَرَادَ أَن يَسْتَفِزَّهُم مِّنَ ٱلْأَرْضِ فَأَغْرَقْنَـٰهُ وَمَن مَّعَهُۥ جَمِيعًا ﴿١٠٣﴾
پىرئەۋن ئۇلارنى (مىسىر) زېمىنىدىن چىقىرىۋەتمەكچى بولدى، بىز پىرئەۋننى ۋە ئۇنىڭ بىلەن بىللە بولغانلارنى - ھەممىسىنى (دېڭىز) غا غەرق قىلىۋەتتۇق
وَقُلْنَا مِنۢ بَعْدِهِۦ لِبَنِىٓ إِسْرَٰٓءِيلَ ٱسْكُنُوا۟ ٱلْأَرْضَ فَإِذَا جَآءَ وَعْدُ ٱلْـَٔاخِرَةِ جِئْنَا بِكُمْ لَفِيفًا ﴿١٠٤﴾
پىرئەۋننى غەرق قىلىۋەتكەندىن كېيىن، بىز ئىسرائىل ئەۋلادىغا: «(مىسىر) زېمىنىدا تۇرۇڭلار، ئاخىرەت ۋەدىسى (يەنى قىيامەت) كەلگەن چاغدا سىلەرنى (قەبرەڭلاردىن مەھشەرگاھقا) (ھەقىقەتنى رەت قىلغۇچىلار بىلەن مۆمىنلەرنى) ئارىلاش ئېلىپ كېلىمىز» دېدۇق
وَبِٱلْحَقِّ أَنزَلْنَـٰهُ وَبِٱلْحَقِّ نَزَلَ ۗ وَمَآ أَرْسَلْنَـٰكَ إِلَّا مُبَشِّرًا وَنَذِيرًا ﴿١٠٥﴾
قۇرئاننى ھەقىقەت ئاساسىدا نازىل قىلدۇق، قۇرئانمۇ ھەقىقەتنى ئۆز ئىچىگە ئالغان ھالدا نازىل بولدى، سېنى بىز پەقەت خوش خەۋەر بەرگۈچى، ئاگاھلاندۇرغۇچى قىلىپ ئەۋەتتۇق
وَقُرْءَانًا فَرَقْنَـٰهُ لِتَقْرَأَهُۥ عَلَى ٱلنَّاسِ عَلَىٰ مُكْثٍ وَنَزَّلْنَـٰهُ تَنزِيلًا ﴿١٠٦﴾
قۇرئاننى كىشىلەرگە دانە - دانە ئوقۇپ بېرىشىڭ ئۈچۈن ئۇنى بۆلۈپ - بۆلۈپ نازىل قىلدۇق، ئۇنى تەدرىجىي نازىل قىلدۇق
قُلْ ءَامِنُوا۟ بِهِۦٓ أَوْ لَا تُؤْمِنُوٓا۟ ۚ إِنَّ ٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ مِن قَبْلِهِۦٓ إِذَا يُتْلَىٰ عَلَيْهِمْ يَخِرُّونَ لِلْأَذْقَانِ سُجَّدًا ﴿١٠٧﴾
«قۇرئانغا ئىشىنىڭلار ياكى ئىشەنمەڭلار (مەيلى، چۈنكى ئىشىنىشىڭلار ئۇنىڭغا كامالەت، ئىشەنمەسلىكىڭلار ئۇنىڭغا نۇقسان ئېلىپ كەلمەيدۇ)» دېگىن. شۈبھىسىزكى، قۇرئان نازىل بولۇشتىن ئىلگىرى مۇقەددەس كىتاب-ۋەھىي بېرىلگەنلەر قۇرئاننى ئاڭلىغان چاغلىرىدا (تەسىرلىنىپ) دەرھال سەجدىگە بارىدۇ
وَيَقُولُونَ سُبْحَـٰنَ رَبِّنَآ إِن كَانَ وَعْدُ رَبِّنَا لَمَفْعُولًا ﴿١٠٨﴾
ئۇلار: «پەرۋەردىگارىمىز پاكتۇر، پەرۋەردىگارىمىزنىڭ ۋەدىسى چوقۇم ئورۇنلىنىدۇ» دەيدۇ
وَيَخِرُّونَ لِلْأَذْقَانِ يَبْكُونَ وَيَزِيدُهُمْ خُشُوعًا ۩ ﴿١٠٩﴾
ئۇلار يىغلىغان ھالدا سەجدە قىلىشقا يىقىلىدۇ (قۇرئاننى ئاڭلاش بىلەن ئۇلار اﷲ قا) تېخىمۇ تەزەررۇ قىلىدۇ
قُلِ ٱدْعُوا۟ ٱللَّهَ أَوِ ٱدْعُوا۟ ٱلرَّحْمَـٰنَ ۖ أَيًّا مَّا تَدْعُوا۟ فَلَهُ ٱلْأَسْمَآءُ ٱلْحُسْنَىٰ ۚ وَلَا تَجْهَرْ بِصَلَاتِكَ وَلَا تُخَافِتْ بِهَا وَٱبْتَغِ بَيْنَ ذَٰلِكَ سَبِيلًا ﴿١١٠﴾
«اﷲ دەپ ئاتىساڭلارمۇ بولىدۇ، رەھمان دەپ ئاتىساڭلارمۇ بولىدۇ، قايسىنى ئاتىساڭلار (ھەممىسى ياخشىدۇر). چۈنكى اﷲ نىڭ نۇرغۇن گۈزەل ئىسىملىرى بار» دېگىن، نامىزىڭدا (قىرائەتنى) (مۇشرىكلار ئاڭلاپ ساڭا ئەزىيەت يەتكۈزمەسلىگى ئۈچۈن) يۇقىرى ئاۋاز بىلەنمۇ ئوقۇمىغىن، (مۆمىنلەر ئاڭلىماي قالماسلىقى ئۈچۈن) پەس ئاۋاز بىلەنمۇ ئوقۇمىغىن. ئۇلارنىڭ ئارىسىدا ئوتتۇرا يول تۇتقىن
وَقُلِ ٱلْحَمْدُ لِلَّهِ ٱلَّذِى لَمْ يَتَّخِذْ وَلَدًا وَلَمْ يَكُن لَّهُۥ شَرِيكٌ فِى ٱلْمُلْكِ وَلَمْ يَكُن لَّهُۥ وَلِىٌّ مِّنَ ٱلذُّلِّ ۖ وَكَبِّرْهُ تَكْبِيرًۢا ﴿١١١﴾
ئېيتقىنكى، «جىمى ھەمدۇسانا بالىسى بولۇشتىن پاك بولغان، سەلتەنىتىدە شېرىكى بولمىغان، خارلىقتىن قۇتۇلۇش ئۈچۈن دوستقا موھتاج بولمىغان (يەنى خار بولمىغان، ياردەمچىگە موھتاج بولمىغان) اﷲ قا خاستۇر! اﷲ نى مۇكەممەل رەۋىشتە ئۇلۇغلىغىن»
