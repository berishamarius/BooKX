بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
حمٓ ﴿١﴾
حامىم
تَنزِيلٌ مِّنَ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ ﴿٢﴾
(بۇ) ناھايىتى شەپقەتلىك، مېھرىبان اﷲ تەرىپىدىن نازىل قىلىنغان (ۋەھيىدۇر)
كِتَـٰبٌ فُصِّلَتْ ءَايَـٰتُهُۥ قُرْءَانًا عَرَبِيًّا لِّقَوْمٍ يَعْلَمُونَ ﴿٣﴾
ئايەتلىرى ئوچۇق بايان قىلىنغان مۇقەددەس كىتاب-ۋەھىيتۇر، بىلىدىغان قەۋم ئۈچۈن نازىل بولغان ئەرەبچە قۇرئاندۇر
بَشِيرًا وَنَذِيرًا فَأَعْرَضَ أَكْثَرُهُمْ فَهُمْ لَا يَسْمَعُونَ ﴿٤﴾
(مۆمىنلەرگە مەڭگۈلۈك باغلار بىلەن) خۇش خەۋەر بەرگۈچىدۇر، (ھەقىقەتنى رەت قىلغۇچىلارنى دوزاخ بىلەن) ئاگاھلاندۇرغۇچىدۇر. ئۇلار (يەنى مۇشرىكلار) نىڭ تولىسى (ئۇنىڭ ئايەتلىرىنى پىكىر قىلىشتىن) يۈز ئۆرۈدى. ئۇلار (ئۇنىڭغا) قۇلاق سالمايدۇ
وَقَالُوا۟ قُلُوبُنَا فِىٓ أَكِنَّةٍ مِّمَّا تَدْعُونَآ إِلَيْهِ وَفِىٓ ءَاذَانِنَا وَقْرٌ وَمِنۢ بَيْنِنَا وَبَيْنِكَ حِجَابٌ فَٱعْمَلْ إِنَّنَا عَـٰمِلُونَ ﴿٥﴾
ئۇلار (خۇدانىڭ ئەلچىسىگە): «بىزنىڭ دىللىرىمىز سەن بىزنى دەۋەت قىلغان نەرسىلەردىن پەردىلەنگەن، قۇلاقلىرىمىز ئېغىر، سەن بىلەن بىزنىڭ ئارىمىزدا پەردە (يەنى دىنىي جەھەتتە ئوخشىماسلىق) بار، سەن ئۆز ئىشىڭنى قىل، بىزمۇ ئىشەنچلىك ھالدا ئۆز ئىشىمىزنى قىلايلى (يەنى سەنمۇ ئۆز دىنىڭ بىلەن بول، بىزمۇ ئۆز دىنىمىز بىلەن بولايلى)» دەيدۇ
قُلْ إِنَّمَآ أَنَا۠ بَشَرٌ مِّثْلُكُمْ يُوحَىٰٓ إِلَىَّ أَنَّمَآ إِلَـٰهُكُمْ إِلَـٰهٌ وَٰحِدٌ فَٱسْتَقِيمُوٓا۟ إِلَيْهِ وَٱسْتَغْفِرُوهُ ۗ وَوَيْلٌ لِّلْمُشْرِكِينَ ﴿٦﴾
سەن (ئۇ مۇشرىكلارغا): «مەن سىلەرگە ئوخشاش ئىنسانمەن، ماڭا، ئىلاھىڭلار بىر ئىلاھتۇر، دەپ ۋەھيى قىلىنىدۇ، ئۇنىڭغا (ئىمان ۋە تائەت - ئىبادەت بىلەن) يۈزلىنىڭلار ۋە ئۇنىڭدىن مەغپىرەت تىلەڭلار!» دېگىن، مۇشرىكلارغا ۋاي!
ٱلَّذِينَ لَا يُؤْتُونَ ٱلزَّكَوٰةَ وَهُم بِٱلْـَٔاخِرَةِ هُمْ كَـٰفِرُونَ ﴿٧﴾
ئۇلار زاكات (پاكلاش ئۇلۇشى) بەرمەيدۇ، ئاخىرەتكە ئىشەنمەيدۇ
إِنَّ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ لَهُمْ أَجْرٌ غَيْرُ مَمْنُونٍ ﴿٨﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارغا ئەلۋەتتە ئۈزلۈكسىز ساۋاب بېرىلىدۇ
۞ قُلْ أَئِنَّكُمْ لَتَكْفُرُونَ بِٱلَّذِى خَلَقَ ٱلْأَرْضَ فِى يَوْمَيْنِ وَتَجْعَلُونَ لَهُۥٓ أَندَادًا ۚ ذَٰلِكَ رَبُّ ٱلْعَـٰلَمِينَ ﴿٩﴾
(سەن ئۇلارغا) «سىلەر زېمىننى ئىككى كۈندە ياراتقان اﷲ نى ئىنكار قىلامسىلەر ۋە ئۇنىڭغا شېرىكلەر كەلتۈرەمسىلەر؟ ئۇ ئالەملەرنىڭ پەرۋەردىگارىدۇر» دېگىن
وَجَعَلَ فِيهَا رَوَٰسِىَ مِن فَوْقِهَا وَبَـٰرَكَ فِيهَا وَقَدَّرَ فِيهَآ أَقْوَٰتَهَا فِىٓ أَرْبَعَةِ أَيَّامٍ سَوَآءً لِّلسَّآئِلِينَ ﴿١٠﴾
ئۇ زېمىننىڭ ئۈستىدە نۇرغۇن تاغلارنى ياراتتى، زېمىننى بەرىكەتلىك قىلدى، ئەھلى زېمىننىڭ رىزقىنى بەلگىلىدى، (زېمىننى يارىتىش بىلەن ئاھالىسىنىڭ رىزقىنى بەلگىلەشنى كەم - زىيادىسىز) باراۋەر تۆت كۈندە (تاماملىدى)، (زېمىننىڭ ۋە ئۇنىڭدىكى نەرسىلەرنىڭ يارىتىلىش مۇددىتىنى) سورىغۇچىلار ئۈچۈن (بۇ جاۋابتۇر)
ثُمَّ ٱسْتَوَىٰٓ إِلَى ٱلسَّمَآءِ وَهِىَ دُخَانٌ فَقَالَ لَهَا وَلِلْأَرْضِ ٱئْتِيَا طَوْعًا أَوْ كَرْهًا قَالَتَآ أَتَيْنَا طَآئِعِينَ ﴿١١﴾
ئاندىن ئۇ ئاسماننى (يارىتىشقا) يۈزلەندى، ھالبۇكى، ئۇ تۇمان ھالىتىدە ئىدى، ئاسمان بىلەن زېمىنغا: «ئىختىيارىي ياكى ئىختىيارسىز ھالدا كېلىڭلار (يەنى ئەمرىمنى قوبۇل قىلىڭلار)» دېدى. ئىككىلىسى «ئىختىيارىي كەلدۇق» دېدى
فَقَضَىٰهُنَّ سَبْعَ سَمَـٰوَاتٍ فِى يَوْمَيْنِ وَأَوْحَىٰ فِى كُلِّ سَمَآءٍ أَمْرَهَا ۚ وَزَيَّنَّا ٱلسَّمَآءَ ٱلدُّنْيَا بِمَصَـٰبِيحَ وَحِفْظًا ۚ ذَٰلِكَ تَقْدِيرُ ٱلْعَزِيزِ ٱلْعَلِيمِ ﴿١٢﴾
اﷲ يەتتە ئاسماننى ئىككى كۈندە ياراتتى. ھەر ئاسماننىڭ ئىشىنى ئۆزىگە بىلدۈردى (يەنى ھەر ئاسماننىڭ ئېھتىياجلىق ئىشلىرىنى ئورۇنلاشتۇردى). ئەڭ تۆۋەنكى ئاسماننى يۇلتۇزلار بىلەن زىننەتلىدۇق ۋە قوغدىدۇق، بۇ ئەنە شۇ غالىب، ھەممىنى بىلگۈچى اﷲ نىڭ تەقدىرىدۇر
فَإِنْ أَعْرَضُوا۟ فَقُلْ أَنذَرْتُكُمْ صَـٰعِقَةً مِّثْلَ صَـٰعِقَةِ عَادٍ وَثَمُودَ ﴿١٣﴾
ئەگەر ئۇلار (بۇ بايانلاردىن كېيىن ئىماندىن) يۈز ئۆرۈسە (سەن ئۇلارغا): «سىلەرنى ئاد ۋە سەمۇد قەۋمىلىرىنىڭ بېشىغا كەلگەن ئازابقا ئوخشاش ئازابتىن ئاگاھلاندۇرىمەن» دېگىن
إِذْ جَآءَتْهُمُ ٱلرُّسُلُ مِنۢ بَيْنِ أَيْدِيهِمْ وَمِنْ خَلْفِهِمْ أَلَّا تَعْبُدُوٓا۟ إِلَّا ٱللَّهَ ۖ قَالُوا۟ لَوْ شَآءَ رَبُّنَا لَأَنزَلَ مَلَـٰٓئِكَةً فَإِنَّا بِمَآ أُرْسِلْتُم بِهِۦ كَـٰفِرُونَ ﴿١٤﴾
ئۆز ۋاقتىدا ئۇلارغا ئالدى - كەينىدىن (يەنى ھەممە تەرەپتىن) خۇدانىڭ ئەلچى-رەسۇللىرى كېلىپ: «بىر اﷲ تىن باشقىغا ئىبادەت قىلماڭلار!» دېسە، ئۇلار: «ئەگەر پەرۋەردىگارىمىز خالىسا، چوقۇم خۇدانىڭ ئەلچى پەرىشتىلىرىنى چۈشۈرگەن بولاتتى، بىز سىلەرنىڭ خۇدانىڭ ئەلچىسىلىكىڭلارنى ئىشەنچلىك ھالدا ئىنكار قىلىمىز» دېدى
فَأَمَّا عَادٌ فَٱسْتَكْبَرُوا۟ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَقَالُوا۟ مَنْ أَشَدُّ مِنَّا قُوَّةً ۖ أَوَلَمْ يَرَوْا۟ أَنَّ ٱللَّهَ ٱلَّذِى خَلَقَهُمْ هُوَ أَشَدُّ مِنْهُمْ قُوَّةً ۖ وَكَانُوا۟ بِـَٔايَـٰتِنَا يَجْحَدُونَ ﴿١٥﴾
ئاد بولسا زېمىندا (اﷲ نىڭ بەندىلىرى ھۇد ۋە ئۇنىڭ بىلەن ئىمان ئېيتقانلارغا) ھەقسىز رەۋىشتە يوغانچىلىق قىلدى. ئۇلار: «كۈچ - قۇۋۋەتتە بىزدىن كىم ئارتۇق؟» دېدى. ئۇلار ئۆزلىرىنى ياراتقان اﷲ نىڭ كۈچ - قۇۋۋەتتە ئۆزلىرىدىن ئۈستۈن ئىكەنلىكىنى بىلمىدىمۇ؟ ئۇلار بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلدى
فَأَرْسَلْنَا عَلَيْهِمْ رِيحًا صَرْصَرًا فِىٓ أَيَّامٍ نَّحِسَاتٍ لِّنُذِيقَهُمْ عَذَابَ ٱلْخِزْىِ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَلَعَذَابُ ٱلْـَٔاخِرَةِ أَخْزَىٰ ۖ وَهُمْ لَا يُنصَرُونَ ﴿١٦﴾
ئۇلارغا بۇ دۇنيادىلا خار قىلغۇچى ئازابنى تېتىتىشىمىز ئۈچۈن، شۇم كۈنلەردە قاتتىق سوغۇق شامالنى ئەۋەتتۇق، ئاخىرەتنىڭ ئازابى تېخىمۇ خار قىلغۇچىدۇر، ئۇلار (ئازابنى دەپئى قىلىدىغان) ياردەمگە ئېرىشەلمەيدۇ
وَأَمَّا ثَمُودُ فَهَدَيْنَـٰهُمْ فَٱسْتَحَبُّوا۟ ٱلْعَمَىٰ عَلَى ٱلْهُدَىٰ فَأَخَذَتْهُمْ صَـٰعِقَةُ ٱلْعَذَابِ ٱلْهُونِ بِمَا كَانُوا۟ يَكْسِبُونَ ﴿١٧﴾
سەمۇدقا بولسا توغرا يولنى كۆرسەتتۇق، ئۇلار ھىدايەتتىن كورلۇق (يەنى گۇمراھلىق) نى ئارتۇق بىلدى. قىلمىشلىرى تۈپەيلىدىن، ئۇلارنى خار قىلغۇچى چاقماق ئازابى ھالاك قىلدى
وَنَجَّيْنَا ٱلَّذِينَ ءَامَنُوا۟ وَكَانُوا۟ يَتَّقُونَ ﴿١٨﴾
ئىمان ئېيتقانلارنى ۋە تەقۋادارلارنى (يەنى سالىھ ئەلەيھىسسالام بىلەن ئۇنىڭغا ئەگەشكەنلەرنى ئازابتىن) قۇتقۇزدۇق
وَيَوْمَ يُحْشَرُ أَعْدَآءُ ٱللَّهِ إِلَى ٱلنَّارِ فَهُمْ يُوزَعُونَ ﴿١٩﴾
ئۇ كۈندە اﷲ نىڭ دۈشمەنلىرى دوزاخقا توپلىنىدۇ، ئالدى - ئارقىسى يىغىلىپ بولغۇچە توختىتىپ قويۇلىدۇ
حَتَّىٰٓ إِذَا مَا جَآءُوهَا شَهِدَ عَلَيْهِمْ سَمْعُهُمْ وَأَبْصَـٰرُهُمْ وَجُلُودُهُم بِمَا كَانُوا۟ يَعْمَلُونَ ﴿٢٠﴾
ئۇلار دوزاخقا كەلگەندە، ئۇلارنىڭ قۇلاقلىرى، كۆزلىرى ۋە تېرىلىرى ئۇلارنىڭ قىلمىشلىرى ھەققىدە گۇۋاھلىق بېرىدۇ
وَقَالُوا۟ لِجُلُودِهِمْ لِمَ شَهِدتُّمْ عَلَيْنَا ۖ قَالُوٓا۟ أَنطَقَنَا ٱللَّهُ ٱلَّذِىٓ أَنطَقَ كُلَّ شَىْءٍ وَهُوَ خَلَقَكُمْ أَوَّلَ مَرَّةٍ وَإِلَيْهِ تُرْجَعُونَ ﴿٢١﴾
ئۇلار ئۆزلىرىنىڭ تېرىلىرىگە: «نېمىشقا بىزنىڭ زىيىنىمىزغا گۇۋاھلىق بەردىڭلار؟» دەيدۇ، تېرىلىرى: «ھەر نەرسىنى سۆزلەتكەن اﷲ بىزنى سۆزلەتتى، ئۇ دەسلەپتە سىلەرنى ياراتتى. ئۇنىڭ دەرگاھىغا قايتۇرۇلىسىلەر» دەيدۇ
وَمَا كُنتُمْ تَسْتَتِرُونَ أَن يَشْهَدَ عَلَيْكُمْ سَمْعُكُمْ وَلَآ أَبْصَـٰرُكُمْ وَلَا جُلُودُكُمْ وَلَـٰكِن ظَنَنتُمْ أَنَّ ٱللَّهَ لَا يَعْلَمُ كَثِيرًا مِّمَّا تَعْمَلُونَ ﴿٢٢﴾
سىلەر (دۇنيادىكى چېغىڭلاردا) قۇلىقىڭلارنىڭ، كۆزۈڭلارنىڭ، تېرىلىرىڭلارنىڭ سىلەرنىڭ زىيىنىڭلارغا گۇۋاھلىق بېرىشىدىن ساقلانمىغان ئىدىڭلار (يەنى يامان ئىشلارنى قىلغان چېغىڭلاردا، رەسۋا بولۇشتىن قورقۇپ، كىشىلەردىن يوشۇرغان ئىدىڭلار، ۋەھالەنكى، ئەزالىرىڭلارنىڭ سىلەرنىڭ زىيىنىڭلارغا گۇۋاھلىق بېرىدىغانلىقىنى گۇمان قىلمىغان ئىدىڭلار)، لېكىن سىلەر قىلمىشىڭلاردىن نۇرغۇن ئىشلارنى «اﷲ بىلمەيدۇ» دەپ گۇمان قىلدىڭلار
وَذَٰلِكُمْ ظَنُّكُمُ ٱلَّذِى ظَنَنتُم بِرَبِّكُمْ أَرْدَىٰكُمْ فَأَصْبَحْتُم مِّنَ ٱلْخَـٰسِرِينَ ﴿٢٣﴾
پەرۋەردىگارىڭلارغا قىلغان ئەنە شۇ (قەبىھ) گۇمانىڭلار سىلەرنى ھالاك قىلدى. شۇنىڭ بىلەن زىيان تارتقۇچىلاردىن بولدۇڭلار
فَإِن يَصْبِرُوا۟ فَٱلنَّارُ مَثْوًى لَّهُمْ ۖ وَإِن يَسْتَعْتِبُوا۟ فَمَا هُم مِّنَ ٱلْمُعْتَبِينَ ﴿٢٤﴾
ئەگەر ئۇلار (ئازابقا) چىدىسا، (چىداملىقنىڭ پايدىسى يوق) دوزاخ ئۇلارنىڭ جايىدۇر، ئەگەر ئۇلار (اﷲ تىن) رازىلىق تىلىسە، ھەرگىز رازىلىققا ئېرىشەلمەيدۇ
۞ وَقَيَّضْنَا لَهُمْ قُرَنَآءَ فَزَيَّنُوا۟ لَهُم مَّا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَحَقَّ عَلَيْهِمُ ٱلْقَوْلُ فِىٓ أُمَمٍ قَدْ خَلَتْ مِن قَبْلِهِم مِّنَ ٱلْجِنِّ وَٱلْإِنسِ ۖ إِنَّهُمْ كَانُوا۟ خَـٰسِرِينَ ﴿٢٥﴾
ئۇلارغا بىز (شەيتان (ئازدۇرغۇچى)لاردىن) دوستلارنى مۇسەللەت قىلدۇق، ئۇلار (يەنى شەيتان (ئازدۇرغۇچى)لار) ئۇلارغا ئالدى - كەينىدىكى نەرسىلەرنى (يەنى ئۇلارنىڭ ھازىرقى ۋە كەلگۈسىدىكى قەبىھ ئەمەللىرىنى) چىرايلىق كۆرسەتتى، ئۇلارغا ئۇلاردىن ئىلگىرى ئۆتكەن ئىنسانلاردىن ۋە جىنلاردىن بولغان (گۇناھكار شەقى) ئۈممەتلەر قاتارىدا ئازاب سۆزى تېگىشلىك بولدى، چۈنكى ئۇلار زىيان تارتقۇچىلار ئىدى
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لَا تَسْمَعُوا۟ لِهَـٰذَا ٱلْقُرْءَانِ وَٱلْغَوْا۟ فِيهِ لَعَلَّكُمْ تَغْلِبُونَ ﴿٢٦﴾
ھەقىقەتنى رەت قىلغۇچىلار: «بۇ قۇرئانغا قۇلاق سالماڭلار، ئۇنى ئېلىشتۇرۇۋېتىڭلار، سىلەر غەلىبە قىلىشىڭلار مۇمكىن» دەيدۇ
فَلَنُذِيقَنَّ ٱلَّذِينَ كَفَرُوا۟ عَذَابًا شَدِيدًا وَلَنَجْزِيَنَّهُمْ أَسْوَأَ ٱلَّذِى كَانُوا۟ يَعْمَلُونَ ﴿٢٧﴾
بىز ھەقىقەتنى رەت قىلغۇچىلارغا چوقۇم قاتتىق ئازابنى تېتىتىمىز، بىز ئۇلارنىڭ يامان قىلمىشلىرىغا ئەڭ قاتتىق جازا بېرىمىز
ذَٰلِكَ جَزَآءُ أَعْدَآءِ ٱللَّهِ ٱلنَّارُ ۖ لَهُمْ فِيهَا دَارُ ٱلْخُلْدِ ۖ جَزَآءًۢ بِمَا كَانُوا۟ بِـَٔايَـٰتِنَا يَجْحَدُونَ ﴿٢٨﴾
اﷲ نىڭ دۈشمەنلىرىگە بېرىلىدىغان جازا ئەنە شۇ دوزاختۇر، ئۇلارنىڭ مەڭگۈلۈك تۇرىدىغان جايى دوزاختۇر، بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلىقلىرى ئۈچۈن (ئۇلارغا) شۇنداق جازا بېرىلىدۇ
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ رَبَّنَآ أَرِنَا ٱلَّذَيْنِ أَضَلَّانَا مِنَ ٱلْجِنِّ وَٱلْإِنسِ نَجْعَلْهُمَا تَحْتَ أَقْدَامِنَا لِيَكُونَا مِنَ ٱلْأَسْفَلِينَ ﴿٢٩﴾
ھەقىقەتنى رەت قىلغۇچىلار: «پەرۋەردىگارىمىز! ئىنسانلاردىن ۋە جىنلاردىن بىزنى ئازدۇرغانلارنى بىزگە كۆرسەتكىن، (ئۇلارنىڭ) ئەڭ خارلاردىن بولۇشى ئۈچۈن ئۇلارنى ئاياغلىرىمىزنىڭ ئاستىدا چەيلەيمىز» دېدى
إِنَّ ٱلَّذِينَ قَالُوا۟ رَبُّنَا ٱللَّهُ ثُمَّ ٱسْتَقَـٰمُوا۟ تَتَنَزَّلُ عَلَيْهِمُ ٱلْمَلَـٰٓئِكَةُ أَلَّا تَخَافُوا۟ وَلَا تَحْزَنُوا۟ وَأَبْشِرُوا۟ بِٱلْجَنَّةِ ٱلَّتِى كُنتُمْ تُوعَدُونَ ﴿٣٠﴾
شۈبھىسىزكى، «پەرۋەردىگارىمىز اﷲ دۇر» دېگەنلەر، ئاندىن توغرا يولدا بولغانلارغا خۇدانىڭ ئەلچى پەرىشتىلىرى چۈشۈپ: «قورقماڭلار، غەم قىلماڭلار، سىلەرگە ۋەدە قىلىنغان مەڭگۈلۈك باغلار ئۈچۈن خۇشال بولۇڭلار، بىز دۇنيادا، ئاخىرەتتە سىلەرنىڭ دوستلىرىڭلارمىز، مەڭگۈلۈك باغلارتە سىلەر ئۈچۈن كۆڭلۈڭلار تارتقان نەرسىلەرنىڭ ھەممىسى ۋە تىلىگەن نەرسەڭلارنىڭ ھەممىسى بار، (ئۇلار) ناھايىتى مەغپىرەت قىلغۇچى، ناھايىتى مېھرىبان (اﷲ) تەرىپىدىن بېرىلگەن زىياپەتتۇر» دەيدۇ
نَحْنُ أَوْلِيَآؤُكُمْ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْـَٔاخِرَةِ ۖ وَلَكُمْ فِيهَا مَا تَشْتَهِىٓ أَنفُسُكُمْ وَلَكُمْ فِيهَا مَا تَدَّعُونَ ﴿٣١﴾
شۈبھىسىزكى، «پەرۋەردىگارىمىز اﷲ دۇر» دېگەنلەر، ئاندىن توغرا يولدا بولغانلارغا خۇدانىڭ ئەلچى پەرىشتىلىرى چۈشۈپ: «قورقماڭلار، غەم قىلماڭلار، سىلەرگە ۋەدە قىلىنغان مەڭگۈلۈك باغلار ئۈچۈن خۇشال بولۇڭلار، بىز دۇنيادا، ئاخىرەتتە سىلەرنىڭ دوستلىرىڭلارمىز، مەڭگۈلۈك باغلارتە سىلەر ئۈچۈن كۆڭلۈڭلار تارتقان نەرسىلەرنىڭ ھەممىسى ۋە تىلىگەن نەرسەڭلارنىڭ ھەممىسى بار، (ئۇلار) ناھايىتى مەغپىرەت قىلغۇچى، ناھايىتى مېھرىبان (اﷲ) تەرىپىدىن بېرىلگەن زىياپەتتۇر» دەيدۇ
نُزُلًا مِّنْ غَفُورٍ رَّحِيمٍ ﴿٣٢﴾
شۈبھىسىزكى، «پەرۋەردىگارىمىز اﷲ دۇر» دېگەنلەر، ئاندىن توغرا يولدا بولغانلارغا خۇدانىڭ ئەلچى پەرىشتىلىرى چۈشۈپ: «قورقماڭلار، غەم قىلماڭلار، سىلەرگە ۋەدە قىلىنغان مەڭگۈلۈك باغلار ئۈچۈن خۇشال بولۇڭلار، بىز دۇنيادا، ئاخىرەتتە سىلەرنىڭ دوستلىرىڭلارمىز، مەڭگۈلۈك باغلارتە سىلەر ئۈچۈن كۆڭلۈڭلار تارتقان نەرسىلەرنىڭ ھەممىسى ۋە تىلىگەن نەرسەڭلارنىڭ ھەممىسى بار، (ئۇلار) ناھايىتى مەغپىرەت قىلغۇچى، ناھايىتى مېھرىبان (اﷲ) تەرىپىدىن بېرىلگەن زىياپەتتۇر» دەيدۇ
وَمَنْ أَحْسَنُ قَوْلًا مِّمَّن دَعَآ إِلَى ٱللَّهِ وَعَمِلَ صَـٰلِحًا وَقَالَ إِنَّنِى مِنَ ٱلْمُسْلِمِينَ ﴿٣٣﴾
اﷲ قا (يەنى اﷲ نىڭ تەۋھىدىگە ۋە تائىتىگە) دەۋەت قىلغان، ياخشى ئەمەللەرنى قىلغان ۋە «مەن ئىشەنچلىك ھالدا مۇسۇلمانلاردىنمەن» دېگەن كىشىدىنمۇ ياخشى سۆزلۈك ئادەم بارمۇ؟
وَلَا تَسْتَوِى ٱلْحَسَنَةُ وَلَا ٱلسَّيِّئَةُ ۚ ٱدْفَعْ بِٱلَّتِى هِىَ أَحْسَنُ فَإِذَا ٱلَّذِى بَيْنَكَ وَبَيْنَهُۥ عَدَٰوَةٌ كَأَنَّهُۥ وَلِىٌّ حَمِيمٌ ﴿٣٤﴾
ياخشى ئىش بىلەن يامان ئىش باراۋەر بولمايدۇ، ياخشى خىسلەت ئارقىلىق (يامان خىسلەتكە) تاقابىل تۇرغىن، (شۇنداق قىلساڭ) سەن بىلەن ئۆزىنىڭ ئارىسىدا ئاداۋەت بار ئادەم گويا سىرداش دوستۇڭدەك بولۇپ قالىدۇ
وَمَا يُلَقَّىٰهَآ إِلَّا ٱلَّذِينَ صَبَرُوا۟ وَمَا يُلَقَّىٰهَآ إِلَّا ذُو حَظٍّ عَظِيمٍ ﴿٣٥﴾
بۇ خىسلەتكە پەقەت سەۋرچان ئادەملەرلا ئېرىشەلەيدۇ، بۇ خىسلەتكە پەقەت بۈيۈك نېسىۋە ئىگىسىلا ئېرىشەلەيدۇ
وَإِمَّا يَنزَغَنَّكَ مِنَ ٱلشَّيْطَـٰنِ نَزْغٌ فَٱسْتَعِذْ بِٱللَّهِ ۖ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٣٦﴾
ئەگەر شەيتان (ئازدۇرغۇچى) سېنى ۋەسۋەسە قىلىپ (يۇقىرىقى خىسلەتنى ئۆزلەشتۈرمەسلىككە كۈشكۈرتسە) اﷲ قا سېغىنىپ (شەيتان (ئازدۇرغۇچى)نىڭ شەررىدىن) پاناھ تىلىگىن، اﷲ ئىشەنچلىك ھالدا (بەندىلىرىنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (ئىشلىرىنى) بىلىپ تۇرغۇچىدۇر
وَمِنْ ءَايَـٰتِهِ ٱلَّيْلُ وَٱلنَّهَارُ وَٱلشَّمْسُ وَٱلْقَمَرُ ۚ لَا تَسْجُدُوا۟ لِلشَّمْسِ وَلَا لِلْقَمَرِ وَٱسْجُدُوا۟ لِلَّهِ ٱلَّذِى خَلَقَهُنَّ إِن كُنتُمْ إِيَّاهُ تَعْبُدُونَ ﴿٣٧﴾
كېچە بىلەن كۈندۈز، كۈن بىلەن ئاي اﷲ نىڭ (بىرلىكىنى ۋە قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر، قۇياشقا سەجدە قىلماڭلار، ئايغىمۇ سەجدە قىلماڭلار، ئۇلارنى ياراتقان اﷲ قا سەجدە قىلىڭلار، ئەگەر پەقەت اﷲ غىلا ئىبادەت قىلىدىغان بولساڭلار
فَإِنِ ٱسْتَكْبَرُوا۟ فَٱلَّذِينَ عِندَ رَبِّكَ يُسَبِّحُونَ لَهُۥ بِٱلَّيْلِ وَٱلنَّهَارِ وَهُمْ لَا يَسْـَٔمُونَ ۩ ﴿٣٨﴾
ئەگەر كۇففارلار چوڭچىلىق قىلىپ (اﷲ قا سەجدە قىلىشتىن) باش تارتسا، پەرۋەردىگارىڭنىڭ دەرگاھىدىكىلەر (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى) كېچە - كۈندۈز اﷲ قا تەسبىھ ئېيتىپ تۇرىدۇ، ئۇلار مالال بولۇپ قالمايدۇ
وَمِنْ ءَايَـٰتِهِۦٓ أَنَّكَ تَرَى ٱلْأَرْضَ خَـٰشِعَةً فَإِذَآ أَنزَلْنَا عَلَيْهَا ٱلْمَآءَ ٱهْتَزَّتْ وَرَبَتْ ۚ إِنَّ ٱلَّذِىٓ أَحْيَاهَا لَمُحْىِ ٱلْمَوْتَىٰٓ ۚ إِنَّهُۥ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٣٩﴾
(بۇمۇ) اﷲ نىڭ (بىرلىكىنى ۋە قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇركى، سەن زېمىننى قاقاس كۆرىسەن، ئۇنىڭغا يامغۇر ياغدۇرغان ۋاقتىمىزدا، زېمىن كۆتۈرۈلىدۇ ۋە كۆپۈشىدۇ، شۈبھىسىزكى، زېمىننى تىرىلدۈرگەن زات ئۆلۈكلەرنى ئەلۋەتتە تىرىلدۈرگۈچىدۇر، ئۇ ئىشەنچلىك ھالدا ھەر نەرسىگە قادىردۇر
إِنَّ ٱلَّذِينَ يُلْحِدُونَ فِىٓ ءَايَـٰتِنَا لَا يَخْفَوْنَ عَلَيْنَآ ۗ أَفَمَن يُلْقَىٰ فِى ٱلنَّارِ خَيْرٌ أَم مَّن يَأْتِىٓ ءَامِنًا يَوْمَ ٱلْقِيَـٰمَةِ ۚ ٱعْمَلُوا۟ مَا شِئْتُمْ ۖ إِنَّهُۥ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٤٠﴾
شۈبھىسىزكى، بىزنىڭ ئايەتلىرىمىزنى بۇرمىلاپ چۈشەندۈرىدىغانلار بىزگە مەخپىي قالمايدۇ، قىيامەت كۈنى دوزاخقا تاشلىنىدىغان ئادەم ياخشىمۇ؟ ياكى قىيامەت كۈنى (اﷲ نىڭ ئازابىدىن) ئەمىن بولغان ھالدا كېلىدىغان ئادەم ياخشىمۇ؟ خالىغىنىڭلارنى قىلىڭلار، شۈبھىسىزكى، اﷲ قىلمىشىڭلارنى كۆرۈپ تۇرغۇچىدۇر
إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِٱلذِّكْرِ لَمَّا جَآءَهُمْ ۖ وَإِنَّهُۥ لَكِتَـٰبٌ عَزِيزٌ ﴿٤١﴾
شۈبھىسىزكى، قۇرئان كەلگەندە ئۇنى ئىنكار قىلغانلار (قاتتىق جازاغا ئۇچرايدۇ)، شۈبھىسىزكى، قۇرئان غالىب مۇقەددەس كىتاب-ۋەھىيتۇر
لَّا يَأْتِيهِ ٱلْبَـٰطِلُ مِنۢ بَيْنِ يَدَيْهِ وَلَا مِنْ خَلْفِهِۦ ۖ تَنزِيلٌ مِّنْ حَكِيمٍ حَمِيدٍ ﴿٤٢﴾
ئۇنىڭغا ئالدىدىنمۇ، ئارقىسىدىنمۇ (يەنى ھېچقايسى تەرىپىدىن) باتىل يۈزلەنمەيدۇ، ئۇ ھېكمەت بىلەن ئىش قىلغۇچى، مەدھىيىگە لايىق اﷲ تەرىپىدىن نازىل قىلىنغاندۇر
مَّا يُقَالُ لَكَ إِلَّا مَا قَدْ قِيلَ لِلرُّسُلِ مِن قَبْلِكَ ۚ إِنَّ رَبَّكَ لَذُو مَغْفِرَةٍ وَذُو عِقَابٍ أَلِيمٍ ﴿٤٣﴾
ساڭا (مۇشرىكلار تەرىپىدىن) ئېيتىلىدىغان سۆزلەر پەقەت ئىلگىرىكى خۇدانىڭ ئەلچى-رەسۇللىرىگە (قەۋملىرى تەرىپىدىن) ئېيتىلغان سۆزلەردۇر، شۈبھىسىزكى، سېنىڭ پەرۋەردىگارىڭ مەغپىرەت ئىگىسىدۇر ۋە قاتتىق ئازاب ئىگىسىدۇر
وَلَوْ جَعَلْنَـٰهُ قُرْءَانًا أَعْجَمِيًّا لَّقَالُوا۟ لَوْلَا فُصِّلَتْ ءَايَـٰتُهُۥٓ ۖ ءَا۬عْجَمِىٌّ وَعَرَبِىٌّ ۗ قُلْ هُوَ لِلَّذِينَ ءَامَنُوا۟ هُدًى وَشِفَآءٌ ۖ وَٱلَّذِينَ لَا يُؤْمِنُونَ فِىٓ ءَاذَانِهِمْ وَقْرٌ وَهُوَ عَلَيْهِمْ عَمًى ۚ أُو۟لَـٰٓئِكَ يُنَادَوْنَ مِن مَّكَانٍۭ بَعِيدٍ ﴿٤٤﴾
مۇبادا بىز ئۇنى ئەجەم تىلىدىكى قۇرئان قىلىپ نازىل قىلغان بولساق، ئۇلار ئەلۋەتتە: «نېمىشقا ئۇنىڭ ئايەتلىرى (بىز چۈشىنىدىغان تىلدا) روشەن بايان قىلىنمىدى؟ خۇدانىڭ ئەلچىسى ئەرەب، مۇقەددەس كىتاب-ۋەھىي ئەجەمچىمۇ؟» دەيتتى. ئۇ ئىمان كەلتۈرگەنلەرگە ھىدايەتتۇر ۋە (دىللاردىكى شەك - شۈبھىگە) شىپادۇر، ئىمان ئېيتمايدىغانلارنىڭ قۇلاقلىرى (قۇرئاننى ئاڭلاشتىن) ئېغىردۇر (يەنى قۇرئاننىڭ دەۋىتىگە قۇلاق سالمىغانلارنىڭ ۋە ئۇنىڭدىن يۈز ئۆرۈگەنلەرنىڭ گويا قۇلاقلىرى ئېغىردۇر)، قۇرئان ئۇلارنىڭ (دىللىرىغا) كورلۇقتۇر (يەنى ئۇلار ھىدايەت نۇرىنى كۆرمەيدۇ)، ئەنە شۇلار يىراقتىن نىدا قىلىنغۇچىلاردۇر (يەنى ئۇلار ھەقنى قوبۇل قىلماسلىقتا ئىمانغا يىراقتىن چاقىرىلغان، مۇساپىنىڭ يىراقلىقىدىن چاقىرىقنى ئاڭلىمىغان كىشىلەرگە ئوخشايدۇ)
وَلَقَدْ ءَاتَيْنَا مُوسَى ٱلْكِتَـٰبَ فَٱخْتُلِفَ فِيهِ ۗ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ لَقُضِىَ بَيْنَهُمْ ۚ وَإِنَّهُمْ لَفِى شَكٍّ مِّنْهُ مُرِيبٍ ﴿٤٥﴾
بىز ئىشەنچلىك ھالدا مۇساغا مۇقەددەس كىتاب-ۋەھىي (يەنى تەۋراتنى) ئاتا قىلدۇق، ئۇ ھەقتە ئىختىلاپ بولدى. ئەگەر پەرۋەردىگارىڭنىڭ (ئۇلارنى جازالاشنى قىيامەتكىچە كېچىكتۈرۈش) ۋەدىسى بولمىسا ئىدى، ئەلۋەتتە ئۇلارنىڭ ئارىسىدا ھۆكۈم چىقىرىلاتتى، ئۇلار ئىشەنچلىك ھالدا قۇرئاندىن زىيادە شەكتىدۇر
مَّنْ عَمِلَ صَـٰلِحًا فَلِنَفْسِهِۦ ۖ وَمَنْ أَسَآءَ فَعَلَيْهَا ۗ وَمَا رَبُّكَ بِظَلَّـٰمٍ لِّلْعَبِيدِ ﴿٤٦﴾
كىمكى ياخشى ئىش قىلىدىكەن، پايدىسى ئۆزىگىدۇر، كىمكى يامان ئىش قىلىدىكەن زىيىنىمۇ ئۆزىگىدۇر. پەرۋەردىگارىڭ بەندىلەرگە زۇلۇم قىلغۇچى ئەمەستۇر
۞ إِلَيْهِ يُرَدُّ عِلْمُ ٱلسَّاعَةِ ۚ وَمَا تَخْرُجُ مِن ثَمَرَٰتٍ مِّنْ أَكْمَامِهَا وَمَا تَحْمِلُ مِنْ أُنثَىٰ وَلَا تَضَعُ إِلَّا بِعِلْمِهِۦ ۚ وَيَوْمَ يُنَادِيهِمْ أَيْنَ شُرَكَآءِى قَالُوٓا۟ ءَاذَنَّـٰكَ مَا مِنَّا مِن شَهِيدٍ ﴿٤٧﴾
قىيامەتنىڭ ۋاقتىنى بىلىش پەقەت اﷲ قا خاستۇر (يەنى قىيامەتنىڭ قاچان بولىدىغانلىقىنى اﷲ تىن باشقا كىشى بىلمەيدۇ). مېۋىلەرنىڭ بوغۇنلىرىدىن چىقىشى، ھەر قانداق چىشىنىڭ ھامىلدار بولۇشى ۋە تۇغۇشى اﷲ نىڭ ئىلمىنىڭ سىرتىدا ئەمەستۇر، ئۇ كۈندە اﷲ ئۇلارغا: «مېنىڭ شېرىكلىرىم (يەنى سىلەر ماڭا شېرىك كەلتۈرگەن نەرسىلەر) قەيەردە؟» دەپ نىدا قىلىدۇ، ئۇلار: «ساڭا مەلۇم قىلىمىزكى، ئارىمىزدا ھېچ كىشى (سېنىڭ شېرىكىڭ بار دەپ) گۇۋاھلىق بېرەلمەيدۇ» دەيدۇ
وَضَلَّ عَنْهُم مَّا كَانُوا۟ يَدْعُونَ مِن قَبْلُ ۖ وَظَنُّوا۟ مَا لَهُم مِّن مَّحِيصٍ ﴿٤٨﴾
ئۇلار ئىلگىرى (يەنى دۇنيادىكى چاغدا) چوقۇنغان بۇتلىرى ئۇلاردىن ئۆزلىرىنى چەتكە ئالىدۇ، ئۇلار (اﷲ نىڭ ئازابىدىن) قاچىدىغان جاي يوق ئىكەنلىكىنى ئېنىق بىلىدۇ
لَّا يَسْـَٔمُ ٱلْإِنسَـٰنُ مِن دُعَآءِ ٱلْخَيْرِ وَإِن مَّسَّهُ ٱلشَّرُّ فَيَـُٔوسٌ قَنُوطٌ ﴿٤٩﴾
ئىنسان (ئۆزىگە) ياخشىلىق تىلەشتىن زېرىكمەيدۇ، ئەگەر ئۇنىڭغا بىرەر يامانلىق يېتىپ قالسا (اﷲ نىڭ رەھمىتىدىن) تولىمۇ ئۈمىدسىزلىنىپ كېتىدۇ
وَلَئِنْ أَذَقْنَـٰهُ رَحْمَةً مِّنَّا مِنۢ بَعْدِ ضَرَّآءَ مَسَّتْهُ لَيَقُولَنَّ هَـٰذَا لِى وَمَآ أَظُنُّ ٱلسَّاعَةَ قَآئِمَةً وَلَئِن رُّجِعْتُ إِلَىٰ رَبِّىٓ إِنَّ لِى عِندَهُۥ لَلْحُسْنَىٰ ۚ فَلَنُنَبِّئَنَّ ٱلَّذِينَ كَفَرُوا۟ بِمَا عَمِلُوا۟ وَلَنُذِيقَنَّهُم مِّنْ عَذَابٍ غَلِيظٍ ﴿٥٠﴾
ئەگەر بىز ئۇنىڭغا كۈلپەتتىن كېيىن رەھمىتىمىزنى تېتىتساق، ئۇ چوقۇم: «بۇ ئۆزۈمنىڭ سەئيى - ئىجتىھاتىمدىن كەلگەن، قىيامەتنىڭ بولىدىغانلىقىغا ئىشەنمەيمەن، مۇبادا (قىيامەت بولۇپ) مەن پەرۋەردىگارىمغا قايتۇرۇلغاندىمۇ ئۇنىڭ دەرگاھىدا چوقۇم مەڭگۈلۈك باغلاركە ئېرىشىمەن» دەيدۇ. ئەلۋەتتە ھەقىقەتنى رەت قىلغۇچىلارغا قىلمىشلىرىنى خەۋەر قىلىمىز ۋە ئۇلارغا قاتتىق ئازابنى تېتىتىمىز
وَإِذَآ أَنْعَمْنَا عَلَى ٱلْإِنسَـٰنِ أَعْرَضَ وَنَـَٔا بِجَانِبِهِۦ وَإِذَا مَسَّهُ ٱلشَّرُّ فَذُو دُعَآءٍ عَرِيضٍ ﴿٥١﴾
ئىنسانغا نېمەت ئاتا قىلساق (پەرۋەردىگارىغا شۈكۈر قىلىشتىن) يۈز ئۆرۈيدۇ، (اﷲ نىڭ ئەمرىگە بويسۇنۇشتىن باش تارتىپ) ھاكاۋۇرلىشىپ كېتىدۇ، ئەگەر ئۇنىڭغا يامانلىق يەتسە (اﷲ قا يۈزلىنىپ) ئۈزلۈكسىز دۇئا قىلىدۇ
قُلْ أَرَءَيْتُمْ إِن كَانَ مِنْ عِندِ ٱللَّهِ ثُمَّ كَفَرْتُم بِهِۦ مَنْ أَضَلُّ مِمَّنْ هُوَ فِى شِقَاقٍۭ بَعِيدٍ ﴿٥٢﴾
(ئى مۇھەممەد! ئېيتقىنكى) «ئېيتىپ بېقىڭلارچۇ (قۇرئان مەن ئېيتقاندەك) اﷲ تەرىپىدىن نازىل بولغان بولسا، سىلەر (ئويلىماستىنلا ئۇنى ئىنكار قىلساڭلار، ھالىڭلار قانداق بولىدۇ). (ھەقتىن) يىراق ئىختىلاپتا بولغان كىشىدىنمۇ گۇمراھ ئادەم بارمۇ؟ (يەنى سىلەردىنمۇ گۇمراھ ئادەم يوق)»
سَنُرِيهِمْ ءَايَـٰتِنَا فِى ٱلْـَٔافَاقِ وَفِىٓ أَنفُسِهِمْ حَتَّىٰ يَتَبَيَّنَ لَهُمْ أَنَّهُ ٱلْحَقُّ ۗ أَوَلَمْ يَكْفِ بِرَبِّكَ أَنَّهُۥ عَلَىٰ كُلِّ شَىْءٍ شَهِيدٌ ﴿٥٣﴾
ئۇلارغا بىز تاكى ئۇلارغا قۇرئاننىڭ ھەق ئىكەنلىكى ئېنىق بولغۇچە، (ئاسمان - زېمىن) ئەتراپىدىكى ۋە ئۆزلىرىدىكى (قۇدرىتىمىزگە دالالەت قىلىدىغان) ئالامەتلەرنى كۆرسىتىمىز، پەرۋەردىگارىڭنىڭ ھەممە نەرسىنى كۆرۈپ تۇرىدىغانلىقى (سېنىڭ راستلىقىڭنىڭ دەلىلى بولۇشقا) ئۇلارغا كۇپايە ئەمەسمۇ
أَلَآ إِنَّهُمْ فِى مِرْيَةٍ مِّن لِّقَآءِ رَبِّهِمْ ۗ أَلَآ إِنَّهُۥ بِكُلِّ شَىْءٍ مُّحِيطٌۢ ﴿٥٤﴾
بىلىڭلاركى، ئۇلار ئىشەنچلىك ھالدا پەرۋەردىگارىغا مۇلاقات بولۇشتىن شەكلىنىدۇ، بىلىڭلاركى، اﷲ ھەممە نەرسىنى ئىشەنچلىك ھالدا تولۇق بىلىپ تۇرغۇچىدۇر
