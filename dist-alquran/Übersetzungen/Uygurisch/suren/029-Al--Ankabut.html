بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
الٓمٓ ﴿١﴾
ئەلىف، لام، مىم
أَحَسِبَ ٱلنَّاسُ أَن يُتْرَكُوٓا۟ أَن يَقُولُوٓا۟ ءَامَنَّا وَهُمْ لَا يُفْتَنُونَ ﴿٢﴾
ئىنسانلار «ئىمان ئېيتتۇق» دەپ قويۇش بىلەنلا سىنالماي تەرك ئېتىلىمىز، دەپ ئويلامدۇ؟
وَلَقَدْ فَتَنَّا ٱلَّذِينَ مِن قَبْلِهِمْ ۖ فَلَيَعْلَمَنَّ ٱللَّهُ ٱلَّذِينَ صَدَقُوا۟ وَلَيَعْلَمَنَّ ٱلْكَـٰذِبِينَ ﴿٣﴾
ئۇلاردىن بۇرۇن ئۆتكەنلەرنى بىز ئىشەنچلىك ھالدا سىنىدۇق، اﷲ (ئىمانىدا) راستچىللارنى چوقۇم بىلىدۇ، (ئىمانىدا) يالغانچىلارنىمۇ چوقۇم بىلىدۇ
أَمْ حَسِبَ ٱلَّذِينَ يَعْمَلُونَ ٱلسَّيِّـَٔاتِ أَن يَسْبِقُونَا ۚ سَآءَ مَا يَحْكُمُونَ ﴿٤﴾
يامان ئىش قىلىدىغانلار بىزدىن (يەنى ئازابىمىزدىن) قېچىپ قۇتۇلالايمىز دەپ ئويلايدۇ. ئۇلارنىڭ چىقارغان ھۆكمى نېمىدېگەن يامان!
مَن كَانَ يَرْجُوا۟ لِقَآءَ ٱللَّهِ فَإِنَّ أَجَلَ ٱللَّهِ لَـَٔاتٍ ۚ وَهُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٥﴾
كىمكى اﷲ قا مۇلاقات بولۇشنى ئۈمىد قىلىدىكەن، (بىلسۇنكى) اﷲ نىڭ (بۇنىڭغا بەلگىلىگەن) ۋاقتى چوقۇم يېتىپ كېلىدۇ، اﷲ (بەندىلىرىنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (بەندىلىرىنىڭ تاشقى ۋە ئىچكى ئەھۋالىنى) بىلىپ تۇرغۇچىدۇر
وَمَن جَـٰهَدَ فَإِنَّمَا يُجَـٰهِدُ لِنَفْسِهِۦٓ ۚ إِنَّ ٱللَّهَ لَغَنِىٌّ عَنِ ٱلْعَـٰلَمِينَ ﴿٦﴾
كىمكى جىھاد قىلىدىكەن، ئۇنىڭ قىلغان جىھادى ئۆزىنىڭ پايدىسى ئۈچۈندۇر، اﷲ ئىشەنچلىك ھالدا تامامى ئەھلى جاھاندىن بىھاجەتتۇر
وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ لَنُكَفِّرَنَّ عَنْهُمْ سَيِّـَٔاتِهِمْ وَلَنَجْزِيَنَّهُمْ أَحْسَنَ ٱلَّذِى كَانُوا۟ يَعْمَلُونَ ﴿٧﴾
شۈبھىسىزكى، ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنىڭ گۇناھلىرىنى يوققا چىقىرىمىز، ئۇلارنى ئەلۋەتتە قىلغان ئەمەللىرىنىڭ ئەڭ ياخشىسى بىىلەن مۇكاپاتلايمىز
وَوَصَّيْنَا ٱلْإِنسَـٰنَ بِوَٰلِدَيْهِ حُسْنًا ۖ وَإِن جَـٰهَدَاكَ لِتُشْرِكَ بِى مَا لَيْسَ لَكَ بِهِۦ عِلْمٌ فَلَا تُطِعْهُمَآ ۚ إِلَىَّ مَرْجِعُكُمْ فَأُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ﴿٨﴾
ئىنساننى ئاتا - ئانىسىغا ياخشىلىق قىلىشقا بۇيرۇدۇق، ئەگەر ئاتا - ئاناڭ سېنى سەن بىلمەيدىغان نەرسىنى ماڭا شېرىك كەلتۈرۈشكە زورلىسا، ئۇلارغا ئىتائەت قىلمىغىن، سىلەر مېنىڭ دەرگاھىمغا قايتىپ بارىسىلەر، سىلەرگە قىمىشلىرىڭلارنى خەۋەر قىلىمەن
وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ لَنُدْخِلَنَّهُمْ فِى ٱلصَّـٰلِحِينَ ﴿٩﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەل قىلغانلارنى چوقۇم ياخشىلار قاتارىغا كىرگۈزىمىز
وَمِنَ ٱلنَّاسِ مَن يَقُولُ ءَامَنَّا بِٱللَّهِ فَإِذَآ أُوذِىَ فِى ٱللَّهِ جَعَلَ فِتْنَةَ ٱلنَّاسِ كَعَذَابِ ٱللَّهِ وَلَئِن جَآءَ نَصْرٌ مِّن رَّبِّكَ لَيَقُولُنَّ إِنَّا كُنَّا مَعَكُمْ ۚ أَوَلَيْسَ ٱللَّهُ بِأَعْلَمَ بِمَا فِى صُدُورِ ٱلْعَـٰلَمِينَ ﴿١٠﴾
بەزى كىشىلەر: «اﷲ قا ئىمان ئېيتتۇق» دەيدۇ، اﷲ نىڭ يولىدا بىرەر ئەزىيەت تارتسا، كىشىلەر تەرىپىدىن يەتكەن كۈلپەتنى اﷲ نىڭ ئازابى بىلەن ئوخشاش بىلىدۇ، ئەگەر پەرۋەردىگارىڭدىن مەدەت كەلسە: «بىز ئىشەنچلىك ھالدا سىلەر بىلەن بىللە ئىدۇق» دەيدۇ. اﷲ جاھان ئەھلىنىڭ دىللىرىدىكىنى ھەممىدىن ئوبدان بىلمەمدۇ؟
وَلَيَعْلَمَنَّ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ وَلَيَعْلَمَنَّ ٱلْمُنَـٰفِقِينَ ﴿١١﴾
اﷲ مۆمىنلەرنى چوقۇم بىلىدۇ، مۇناپىقلارنىمۇ چوقۇم بىلىدۇ
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لِلَّذِينَ ءَامَنُوا۟ ٱتَّبِعُوا۟ سَبِيلَنَا وَلْنَحْمِلْ خَطَـٰيَـٰكُمْ وَمَا هُم بِحَـٰمِلِينَ مِنْ خَطَـٰيَـٰهُم مِّن شَىْءٍ ۖ إِنَّهُمْ لَكَـٰذِبُونَ ﴿١٢﴾
ھەقىقەتنى رەت قىلغۇچىلار مۆمىنلەرگە: «بىزنىڭ يولىمىز (يەنى دىنىمىز) غا ئەگىشىڭلار، سىلەرنىڭ گۇناھىڭلارنى بىز ئۈستىمىزگە ئالىمىز» دېدى. ھەقىقەتنى رەت قىلغۇچىلار مۆمىنلەرنىڭ گۇناھلىرىدىن ئازراقمۇ ئۈستىگە ئالالمايدۇ، ئۇلار ئىشەنچلىك ھالدا يالغانچىلاردۇر
وَلَيَحْمِلُنَّ أَثْقَالَهُمْ وَأَثْقَالًا مَّعَ أَثْقَالِهِمْ ۖ وَلَيُسْـَٔلُنَّ يَوْمَ ٱلْقِيَـٰمَةِ عَمَّا كَانُوا۟ يَفْتَرُونَ ﴿١٣﴾
ئۇلار ئۆزلىرىنىڭ يۈكلىرىنى ۋە ئۇنىڭغا قوشۇپ باشقا يۈكلەرنى (يەنى ئۆزلىرىنىڭ گۇناھلىرىنى ۋە ئازدۇرغانلىرىنىڭ گۇناھلىرىنى) ئۈستىگە ئالىدۇ، قىيامەت كۈنى ئۇلار ئۆزلىرى ئويدۇرۇپ چىققان يالغان سۆزلىرى ئۈچۈن سوراققا تارتىلىدۇ
وَلَقَدْ أَرْسَلْنَا نُوحًا إِلَىٰ قَوْمِهِۦ فَلَبِثَ فِيهِمْ أَلْفَ سَنَةٍ إِلَّا خَمْسِينَ عَامًا فَأَخَذَهُمُ ٱلطُّوفَانُ وَهُمْ ظَـٰلِمُونَ ﴿١٤﴾
بىز ئىشەنچلىك ھالدا نۇھنى قەۋمىگە (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، نۇھ (قەۋمىنى تەۋھىدكە دەۋەت قىلىپ) ئۇلارنىڭ ئىچىدە 50 يىل كەم 1000 يىل تۇردى (نۇھ ئەلەيھىسسالامنىڭ قەۋمى بۇتپەرەس بولۇپ، ئۇنىڭ خۇدانىڭ ئەلچىسىلىكىگە چىنپۈتمىدى). ئۇلار زالىم بولغانلىقلىرى (يەنى كۇفرىدا ۋە گۇمراھلىقتا چىڭ تۇرۇپ ئۆزلىرىگە زۇلۇم قىلغانلىقلىرى) ئۈچۈن، اﷲ ئۇلارنى توپان (بالاسى) بىلەن ھالاك قىلدى
فَأَنجَيْنَـٰهُ وَأَصْحَـٰبَ ٱلسَّفِينَةِ وَجَعَلْنَـٰهَآ ءَايَةً لِّلْعَـٰلَمِينَ ﴿١٥﴾
بىز نۇھنى ۋە كېمىدىكىلەرنى (غەرق بولۇپ كېتىشتىن) قۇتقۇزدۇق، توپاننى پۈتۈن ئەھلى جاھان ئۈچۈن ئىبرەت قىلدۇق
وَإِبْرَٰهِيمَ إِذْ قَالَ لِقَوْمِهِ ٱعْبُدُوا۟ ٱللَّهَ وَٱتَّقُوهُ ۖ ذَٰلِكُمْ خَيْرٌ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿١٦﴾
ئىبراھىمنىمۇ (قەۋمىگە خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق)، ئۆز ۋاقتىدا ئۇ قەۋمىگە ئېيتتى: «بىر اﷲ قا ئىبادەت قىلىڭلارۋە ئۇنىڭدىن قورقۇڭلار، ئەگەر بىلسەڭلار مۇنداق قىلىش سىلەر ئۈچۈن ياخشىدۇر
إِنَّمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ أَوْثَـٰنًا وَتَخْلُقُونَ إِفْكًا ۚ إِنَّ ٱلَّذِينَ تَعْبُدُونَ مِن دُونِ ٱللَّهِ لَا يَمْلِكُونَ لَكُمْ رِزْقًا فَٱبْتَغُوا۟ عِندَ ٱللَّهِ ٱلرِّزْقَ وَٱعْبُدُوهُ وَٱشْكُرُوا۟ لَهُۥٓ ۖ إِلَيْهِ تُرْجَعُونَ ﴿١٧﴾
سىلەر اﷲ نى قويۇپ بۇتلارغىلا ئىبادەت قىلىسىلەر (بۇتلارنى ئۆز قولۇڭلار بىلەن ياسىۋالغانسىلەر، بۇتلارنىڭ قولىدىن پايدا - زىيان كەلمەيدۇ) ۋە يالغاننىلا ئويدۇرىسىلەر، سىلەرنىڭ اﷲ نى قويۇپ چوقۇنۇۋاتقىنىڭلار (يەنى بۇتلار) سىلەرگە رىزىق بېرىشكە قادىر ئەمەس، رىزىقنى اﷲ نىڭ دەرگاھىدىن تەلەپ قىلىڭلار (رىزىق بېرىشكە پەقەت اﷲ نىڭ ئۆزى قادىردۇر)، اﷲ قا ئىبادەت قىلىڭلار ۋە ئۇنىڭغا شۈكۈر قىلىڭلار، اﷲ نىڭ دەرگاھىغا قايتۇرۇلىسىلەر
وَإِن تُكَذِّبُوا۟ فَقَدْ كَذَّبَ أُمَمٌ مِّن قَبْلِكُمْ ۖ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٨﴾
سىلەر (مېنى) ئىنكار قىلساڭلار (ئىنكار قىلغانلىقىڭلار بىلەن ماڭا قىلچەزىيان يەتكۈزەلمەيسىلەر، پەقەت ئۆزۈڭلارغا زىيان يەتكۈزىسىلەر)، سىلەردىن ئىلگىرىكى نۇرغۇن ئۈممەتلەرمۇ) خۇدانىڭ ئەلچىسىلىرىنى (ئىنكار قىلغان) شۇنىڭ بىلەن ئۇلارغا اﷲ ئازابى نازىل بولغان، سىلەرگىمۇ شۇنداق ئازاب نازىل بولىدۇ)، خۇدانىڭ ئەلچىسىنىڭ مەسئۇلىيىتى پەقەت (اﷲ نىڭ ئەمرلىرىنى) چۈشىنىشلىك قىلىپ يەتكۈزۈشتۇر»
أَوَلَمْ يَرَوْا۟ كَيْفَ يُبْدِئُ ٱللَّهُ ٱلْخَلْقَ ثُمَّ يُعِيدُهُۥٓ ۚ إِنَّ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٌ ﴿١٩﴾
ئۇلار (يەنى ئىنكار قىلغۇچىلار) اﷲ نىڭ مەخلۇقاتىنى دەسلەپتە يوقتىن قانداق بارلىققا كەلتۈرگەنلىكىنى، ئاندىن ئۇنى (ئۆلگەندىن كېيىن) تىرىلدۈرىدىغانلىقىنى كۆرمىدىمۇ؟ بۇ اﷲ قا ئىشەنچلىك ھالدا ئاساندۇر (ئۇلار بۇنى قانداقمۇ ئىنكار قىلىدۇ، دەسلەپتە يارىتىشقا قادىر بولغان زات قايتا ئەسلىگە كەلتۈرۈشكە ئەلۋەتتە قادىردۇر)
قُلْ سِيرُوا۟ فِى ٱلْأَرْضِ فَٱنظُرُوا۟ كَيْفَ بَدَأَ ٱلْخَلْقَ ۚ ثُمَّ ٱللَّهُ يُنشِئُ ٱلنَّشْأَةَ ٱلْـَٔاخِرَةَ ۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٢٠﴾
ئېيتقىنكى، «زېمىندا سەير قىلىڭلار، اﷲ نىڭ مەخلۇقاتىنى قانداق قىلىپ (يوقتىن) بار قىلغانلىقىغا قاراڭلار، ئاندىن اﷲ ئۇلارنى (قىيامەتتە) قايتا پەيدا قىلىدۇ، اﷲ ئىشەنچلىك ھالدا ھەممە نەرسىگە قادىردۇر
يُعَذِّبُ مَن يَشَآءُ وَيَرْحَمُ مَن يَشَآءُ ۖ وَإِلَيْهِ تُقْلَبُونَ ﴿٢١﴾
اﷲ خالىغان ئادەمنى جازالايدۇ، خالىغان ئادەمگە رەھمەت قىلىدۇ، (قىيامەتتە) اﷲ نىڭ دەرگاھىغا قايتۇرۇلىسىلەر
وَمَآ أَنتُم بِمُعْجِزِينَ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ۖ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِىٍّ وَلَا نَصِيرٍ ﴿٢٢﴾
سىلەر مەيلى زېمىندا بولسۇن، مەيلى ئاسماندا بولسۇن، اﷲ (نىڭ ئازابى) دىن قېچىپ قۇتۇلالمايسىلەر، سىلەرگە اﷲ تىن بۆلەك دوستمۇ يوق، مەدەتكارمۇ يوق»
وَٱلَّذِينَ كَفَرُوا۟ بِـَٔايَـٰتِ ٱللَّهِ وَلِقَآئِهِۦٓ أُو۟لَـٰٓئِكَ يَئِسُوا۟ مِن رَّحْمَتِى وَأُو۟لَـٰٓئِكَ لَهُمْ عَذَابٌ أَلِيمٌ ﴿٢٣﴾
اﷲ نىڭ ئايەتلىرىنى ۋە ئۇنىڭغا مۇلاقات بولۇشنى (يەنى ئاخىرەتنى) ئىنكار قىلغانلار ئەنە شۇلار مېنىڭ رەھمىتىمدىن نائۈمىد بولدى. ئەنە شۇلار قاتتىق ئازابقا دۇچار بولىدۇ
فَمَا كَانَ جَوَابَ قَوْمِهِۦٓ إِلَّآ أَن قَالُوا۟ ٱقْتُلُوهُ أَوْ حَرِّقُوهُ فَأَنجَىٰهُ ٱللَّهُ مِنَ ٱلنَّارِ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٢٤﴾
ئىبراھىمنىڭ قەۋمىنىڭ جاۋابى: «ئىبراھىمنى ئۆلتۈرۈڭلار ياكى كۆيدۈرۈڭلار!» دېيىشتىن ئىبارەت بولدى. اﷲ ئۇنى ئوتتىن قۇتۇلدۇردى (ئۇلار ئىبراھىم ئەلەيھىسسالامنى ئوتقا تاشلىغاندا، اﷲ ئۇنىڭغا ئوتنى سورۇن ۋە ئامان جاي قىلىپ بەردى). ئىشەنچلىك ھالدا بۇنىڭدا ئىمان ئېيتقان قەۋم ئۈچۈن (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) نۇرغۇن دەلىللەر بار
وَقَالَ إِنَّمَا ٱتَّخَذْتُم مِّن دُونِ ٱللَّهِ أَوْثَـٰنًا مَّوَدَّةَ بَيْنِكُمْ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ ثُمَّ يَوْمَ ٱلْقِيَـٰمَةِ يَكْفُرُ بَعْضُكُم بِبَعْضٍ وَيَلْعَنُ بَعْضُكُم بَعْضًا وَمَأْوَىٰكُمُ ٱلنَّارُ وَمَا لَكُم مِّن نَّـٰصِرِينَ ﴿٢٥﴾
ئىبراھىم ئېيتتى: «بۇ دۇنيادا (بۇتلارغا چوقۇنۇش مەقسىتىدە يىغىلىشىڭلارنىڭ) ئاراڭلاردىكى دوستلۇققا سەۋەب بولۇشى ئۈچۈن، اﷲ نى قويۇپ بۇتلارنى مەبۇد قىلىۋالدىڭلار، ئاندىن قىيامەت كۈنى بەزىڭلارنى بەزىڭلار ئىنكار قىلىسىلەر، بەزىڭلارغا بەزىڭلار لەنەت ئوقۇيسىلەر (يەنى قىيامەت كۈنى مەزكۇر دوستلۇق دۈشمەنلىككە ئايلىنىدۇ، ئەگەشتۈرگۈچىلەر ئەگەشكۈچىلەردىن ئادا - جۇدا بولىدۇ، ئەگەشكۈچىلەر ئەگەشتۈرگۈچىلەرگە لەنەت ئوقۇيدۇ، چۈنكى ئۇلارنىڭ دۇنيادىكى دوستلۇقى اﷲ ئۈچۈن بولغان ئەمەس)، سىلەرنىڭ جايىڭلار دوزاخ بولىدۇ، سىلەرنى (دوزاختىن قۇتۇلدۇرىدىغان) ھېچقانداق ياردەمچى بولمايدۇ»
۞ فَـَٔامَنَ لَهُۥ لُوطٌ ۘ وَقَالَ إِنِّى مُهَاجِرٌ إِلَىٰ رَبِّىٓ ۖ إِنَّهُۥ هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٦﴾
لۇت ئىبراھىمغا ئىمان ئېيتتى. ئىبراھىم ئېيتتى: «مەن چوقۇم پەرۋەردىگارىم تەرىپىگە ھىجرەت قىلىمەن (يەنى اﷲ نىڭ رازىلىقىنى ئىزدەش يۈزىسىدىن ۋەتىنىمنى تەرك ئېتىپ، اﷲ ئەمر قىلغان جايغا ھىجرەت قىلىپ بارىمەن). اﷲ ئىشەنچلىك ھالدا غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر»
وَوَهَبْنَا لَهُۥٓ إِسْحَـٰقَ وَيَعْقُوبَ وَجَعَلْنَا فِى ذُرِّيَّتِهِ ٱلنُّبُوَّةَ وَٱلْكِتَـٰبَ وَءَاتَيْنَـٰهُ أَجْرَهُۥ فِى ٱلدُّنْيَا ۖ وَإِنَّهُۥ فِى ٱلْـَٔاخِرَةِ لَمِنَ ٱلصَّـٰلِحِينَ ﴿٢٧﴾
ئىبراھىمغا ئىسھاقنى، يەئقۇبنى ئاتا قىلدۇق (يەنى ئىبراھىم ئەلەيھىسسالام خۇدالىق ئۈچۈن قەۋمىدىن ئايرىلغاندىن كېيىن، ياخشى پەرزەنت ئىسھاقنى بەردۇق ۋە ئىسھاقنىڭ ئوغلى بولغان نەۋرىسى يەئقۇبنى بەردۇق)، (ئۇنىڭدىن كېيىن) خۇدانىڭ ئەلچىسىلىكنى، مۇقەددەس كىتاب-ۋەھىينى (يەنى ساماۋى مۇقەددەس كىتاب-ۋەھىيلارنى) ئۇنىڭ ئەۋلادىغا خاس قىلدۇق، دۇنيادىكى مۇكاپاتنى ئۇنىڭغا ئاتا قىلدۇق (يەنى جەمئى دىنلاردا ئۇنى ياخشى نامغا ئىگە قىلدۇق)، ئۇ ئىشەنچلىك ھالدا ئاخىرەتتە ياخشى بەندىلەردىندۇر
وَلُوطًا إِذْ قَالَ لِقَوْمِهِۦٓ إِنَّكُمْ لَتَأْتُونَ ٱلْفَـٰحِشَةَ مَا سَبَقَكُم بِهَا مِنْ أَحَدٍ مِّنَ ٱلْعَـٰلَمِينَ ﴿٢٨﴾
لۇتنى (قەۋمىگە خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق)، ئەينى ۋاقىتتا ئۇ قەۋمىگە ئېيتتى: «سىلەر ئىشەنچلىك ھالدا قەبىھ ئىش قىلىۋاتىسىلەر، سىلەردىن ئىلگىرى جاھان ئەھلىدىن بىرىمۇ مۇنداق قەبىھ ئىشنى قىلغان ئەمەس
أَئِنَّكُمْ لَتَأْتُونَ ٱلرِّجَالَ وَتَقْطَعُونَ ٱلسَّبِيلَ وَتَأْتُونَ فِى نَادِيكُمُ ٱلْمُنكَرَ ۖ فَمَا كَانَ جَوَابَ قَوْمِهِۦٓ إِلَّآ أَن قَالُوا۟ ٱئْتِنَا بِعَذَابِ ٱللَّهِ إِن كُنتَ مِنَ ٱلصَّـٰدِقِينَ ﴿٢٩﴾
سىلەر ئىشەنچلىك ھالدا لىۋاتە قىلامسىلەر؟ يوللارنى توسۇپ بۇلاڭچىلىق قىلامسىلەر؟ سورۇنلىرىڭلاردا ئوپئوچۇق يامان ئىشلارنى قىلىۋېرەمسىلەر؟» ئۇنىڭ قەۋمىنىڭ جاۋابى: «ئەگەر سەن راستچىللاردىن بولساڭ، بىزگە اﷲ نىڭ ئازابىنى كەلتۈرگىن» دېيىشتىنلا ئىبارەت بولدى
قَالَ رَبِّ ٱنصُرْنِى عَلَى ٱلْقَوْمِ ٱلْمُفْسِدِينَ ﴿٣٠﴾
لۇت: «پەرۋەردىگارىم! بۇزغۇنچى قەۋمگە قارشى ماڭا ياردەم بەرگىن (يەنى ئۇلارنى ھالاك قىلىپ ماڭا ياردەم بەرگىن، ئۇلار بۇزغۇنچى ئەخمەقلەر بولۇپ، ئۇلارنىڭ تۈزىلىشى ئۈمىد قىلىنمايدۇ، ئۇلار گۇمراھلىققا، بۇزۇقچىلىققا چۆمۈپ كەتكەن)» دېدى
وَلَمَّا جَآءَتْ رُسُلُنَآ إِبْرَٰهِيمَ بِٱلْبُشْرَىٰ قَالُوٓا۟ إِنَّا مُهْلِكُوٓا۟ أَهْلِ هَـٰذِهِ ٱلْقَرْيَةِ ۖ إِنَّ أَهْلَهَا كَانُوا۟ ظَـٰلِمِينَ ﴿٣١﴾
ئەلچىلىرىمىز (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى) ئىبراھىمغا (پەرزەنت بېرىلىشتىن ئىبارەت) خۇش خەۋەر ئېلىپ كەلگەن چاغدا، ئۇلار: «بىز بۇ شەھەر ئاھالىسىنى چوقۇم ھالاك قىلىمىز، ئۇنىڭ ئاھالىسى ئىشەنچلىك ھالدا زالىم ئىدى» دېدى
قَالَ إِنَّ فِيهَا لُوطًا ۚ قَالُوا۟ نَحْنُ أَعْلَمُ بِمَن فِيهَا ۖ لَنُنَجِّيَنَّهُۥ وَأَهْلَهُۥٓ إِلَّا ٱمْرَأَتَهُۥ كَانَتْ مِنَ ٱلْغَـٰبِرِينَ ﴿٣٢﴾
ئىبراھىم: «ئۇ شەھەردە لۇت بارغۇ (يەنى لۇتقا ئوخشاش بىر ياخشى خۇدانىڭ ئەلچىسى تۇرۇۋاتقان شەھەر ئاھالىسىنى قانداقمۇ ھالاك قىىلىسىلەر؟)» دېدى. خۇدانىڭ ئەلچى پەرىشتىلىرى ئېيتتى: «بۇ شەھەردە بار كىشىلەرنى بىز (يەنى لۇتنى ۋە ئۇنىڭغا ئەگەشكەن مۆمىنلەرنى) ئوبدان بىلىمىز، ئۇنى ۋە ئۇنىڭ ئايالىدىن باشقا كىشىلىرىنى ئەلۋەتتە قۇتۇلدۇرىمىز، پەقەت ئايالى (كۇفرىدا قەۋمىگە ھەمنەپەس بولغانلىقى ئۈچۈن) قېلىپ ھالاك بولغۇچىلاردىن بولىدۇ»
وَلَمَّآ أَن جَآءَتْ رُسُلُنَا لُوطًا سِىٓءَ بِهِمْ وَضَاقَ بِهِمْ ذَرْعًا وَقَالُوا۟ لَا تَخَفْ وَلَا تَحْزَنْ ۖ إِنَّا مُنَجُّوكَ وَأَهْلَكَ إِلَّا ٱمْرَأَتَكَ كَانَتْ مِنَ ٱلْغَـٰبِرِينَ ﴿٣٣﴾
بىزنىڭ ئەلچىلىرىمىز لۇتقا كەلگەن چاغدا، لۇت ئۇلارنى قوغداپ قالالمايدىغانلىقىدىن ئەنسىرەپ قايغۇردى ۋە يۈرىكى سىقىلدى، ئۇلار ئېيتتى: «سەن قورقمىغىن ۋە قايغۇرمىغىن، بىز چوقۇم سېنى ۋە ئايالىڭدىن بۆلەك كىشىلىرىڭنى چوقۇم قۇتۇلدۇرىمىز، پەقەت ئۇ قېلىپ ھالاك بولغۇچىلاردىن بولىدۇ»
إِنَّا مُنزِلُونَ عَلَىٰٓ أَهْلِ هَـٰذِهِ ٱلْقَرْيَةِ رِجْزًا مِّنَ ٱلسَّمَآءِ بِمَا كَانُوا۟ يَفْسُقُونَ ﴿٣٤﴾
بۇ شەھەرنىڭ ئاھالىسى اﷲ نىڭ ئىتائىتىدىن چىققانلىقلىرى ئۈچۈن، ئۇلارغا بىز چوقۇم ئاسماندىن ئازاب چۈشۈرىمىز
وَلَقَد تَّرَكْنَا مِنْهَآ ءَايَةًۢ بَيِّنَةً لِّقَوْمٍ يَعْقِلُونَ ﴿٣٥﴾
چۈشۈنىدىغان قەۋم ئۈچۈن (ئىبرەت قىلىپ) ئۇنىڭدىن (يەنى شەھەر خارابىسىنىڭ ئىزلىرىدىن) روشەن نىشاننى ئىشەنچلىك ھالدا قالدۇردۇق
وَإِلَىٰ مَدْيَنَ أَخَاهُمْ شُعَيْبًا فَقَالَ يَـٰقَوْمِ ٱعْبُدُوا۟ ٱللَّهَ وَٱرْجُوا۟ ٱلْيَوْمَ ٱلْـَٔاخِرَ وَلَا تَعْثَوْا۟ فِى ٱلْأَرْضِ مُفْسِدِينَ ﴿٣٦﴾
مەديەن (ئاھالىسىگە) ئۇلارنىڭ قېرىندىشى شۇئەيبنى (خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق)، شۇئەيب: «ئى قەۋمىم! بىر اﷲ قا ئىبادەت قىلىڭلار، ئاخىرەت كۈنىدىن قورقۇڭلار، يەر يۈزىدە بۇزغۇنچىلىق قىلىپ پىتنە - پاسات تېرىماڭلار» دېدى
فَكَذَّبُوهُ فَأَخَذَتْهُمُ ٱلرَّجْفَةُ فَأَصْبَحُوا۟ فِى دَارِهِمْ جَـٰثِمِينَ ﴿٣٧﴾
ئۇلار شۇئەيبنى ئىنكار قىلدى، شۇنىڭ بىلەن ئۇلارغا قاتتىق زىلزىلە يۈزلەندى - دە، ئۇلار ئۆيلىرىدە ئولتۇرغان پېتى قېتىپ قېلىشتى
وَعَادًا وَثَمُودَا۟ وَقَد تَّبَيَّنَ لَكُم مِّن مَّسَـٰكِنِهِمْ ۖ وَزَيَّنَ لَهُمُ ٱلشَّيْطَـٰنُ أَعْمَـٰلَهُمْ فَصَدَّهُمْ عَنِ ٱلسَّبِيلِ وَكَانُوا۟ مُسْتَبْصِرِينَ ﴿٣٨﴾
ئادنى ۋە سەمۇدنى (ھالاك قىلدۇق)، (ئى ئەھلى مەككە!) سىلەرگە ئىشەنچلىك ھالدا ئۇلارنىڭ (ھىجازدىكى، يەمەندىكى) تۇرالغۇ جايلىرىدىن ئۇلارنىڭ ھالاك بولغانلىقى مەلۇم بولدى، شەيتان (ئازدۇرغۇچى) ئۇلارغا (كۇقرىدىن ۋە گۇناھلاردىن ئىبارەت) ئەمەللىرىنى چىرايلىق كۆرسەتتى، ئۇلارنى توغرا يولدىن توستى، ئۇلار ئەقىل ئىگىلىرى ئىدى) لېكىن ئۇلار تەكەببۇرلۇق يۈزىسىدىن ئەقلىنى جايىدا ئىشلەتمىدى)
وَقَـٰرُونَ وَفِرْعَوْنَ وَهَـٰمَـٰنَ ۖ وَلَقَدْ جَآءَهُم مُّوسَىٰ بِٱلْبَيِّنَـٰتِ فَٱسْتَكْبَرُوا۟ فِى ٱلْأَرْضِ وَمَا كَانُوا۟ سَـٰبِقِينَ ﴿٣٩﴾
(دۇنيا - دەپنىسى نۇرغۇن) قارۇننى، پىرئەۋننى ۋە (ئۇنىڭ زۇلۇمدا ياردەمچى بولغان ۋەزىرى) ھاماننى (ھالاك قىلدۇق)، شۈبھىسىزكى، مۇسا ئۇلارغا روشەن مۆجىزىلەرنى ئېلىپ كەلدى، ئۇلار زېمىندا چوڭچىلىق قىلدى (يەنى اﷲ قا ئىبادەت قىلىشتىن، خۇدانىڭ ئەلچىسىگە ئىتائەت قىلىشتىن بويۇنتاۋلىق قىلدى)، ئۇلار (ئازابىمىزدىن) قېچىپ كېتەلمىدى
فَكُلًّا أَخَذْنَا بِذَنۢبِهِۦ ۖ فَمِنْهُم مَّنْ أَرْسَلْنَا عَلَيْهِ حَاصِبًا وَمِنْهُم مَّنْ أَخَذَتْهُ ٱلصَّيْحَةُ وَمِنْهُم مَّنْ خَسَفْنَا بِهِ ٱلْأَرْضَ وَمِنْهُم مَّنْ أَغْرَقْنَا ۚ وَمَا كَانَ ٱللَّهُ لِيَظْلِمَهُمْ وَلَـٰكِن كَانُوٓا۟ أَنفُسَهُمْ يَظْلِمُونَ ﴿٤٠﴾
(بۇ گۇناھكارلارنىڭ) ھەربىرىنى گۇناھى تۈپەيلىدىن جازالىدۇق، ئۇلارنىڭ بەزىسىگە تاش ياغدۇردۇق، ئۇلارنىڭ بەزىسىنى قاتتىق تاۋۇش ھالاك قىلدى، ئۇلارنىڭ بەزىسىنى (مال - مۈلكى بىلەن قوشۇپ) يەرگە يۇتقۇزدۇق، ئۇلارنىڭ بەزىسىنى (سۇدا) غەرق قىلدۇق، اﷲ ئۇلارغا زۇلۇم قىلمىدى، لېكىن ئۇلار ئۆزلىرىگە ئۆزلىرى زۇلۇم قىلدى
مَثَلُ ٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ أَوْلِيَآءَ كَمَثَلِ ٱلْعَنكَبُوتِ ٱتَّخَذَتْ بَيْتًا ۖ وَإِنَّ أَوْهَنَ ٱلْبُيُوتِ لَبَيْتُ ٱلْعَنكَبُوتِ ۖ لَوْ كَانُوا۟ يَعْلَمُونَ ﴿٤١﴾
اﷲ نى قويۇپ بۇتلارنى مەبۇد قىلىۋالغانلارنىڭ مىسالى ئۆي ياسىغان ئۆمۈچۈكنىڭ مىسالىغا ئوخشايدۇ (يەنى ئۆمۈچۈكنىڭ ئۆيى ئۇنى ئىسسىق - سوغۇقتىن قوغدىيالمىغاندەك، ئۇلارنىڭ چوقۇنۇۋاتقان بۇتلىرىمۇ ئۇلارغا ھېچقانداق پايدا - زىيان يەتكۈزەلمەيدۇ)، ئۆيلەرنىڭ ئەڭ ئاجىزى ئۆمۈچۈكنىڭ ئۆيىدۇر. ئەگەر ئۇلار (بۇ ھەقىقەتنى) بىلسە ئىدى (بۇتلارغا چوقۇنمايتتى)
إِنَّ ٱللَّهَ يَعْلَمُ مَا يَدْعُونَ مِن دُونِهِۦ مِن شَىْءٍ ۚ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٤٢﴾
اﷲ ئۇلار اﷲ نى تاشلاپ ئىبادەت قىلىۋاتقان نەرسىنى ئوبدان بىلىدۇ، اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
وَتِلْكَ ٱلْأَمْثَـٰلُ نَضْرِبُهَا لِلنَّاسِ ۖ وَمَا يَعْقِلُهَآ إِلَّا ٱلْعَـٰلِمُونَ ﴿٤٣﴾
بىز ئىنسانلارغا (قۇرئاندا ئۇلارنىڭ زېھنىغا يېقىنلاشتۇرۇش ئۈچۈن) بايان قىلغان بۇ تەمسىللەرنى پەقەت ئالىملارلا چۈشىنەلەيدۇ
خَلَقَ ٱللَّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّلْمُؤْمِنِينَ ﴿٤٤﴾
اﷲ ئاسمانلارنى ۋە زېمىننى ھەق ياراتتى، ئىشەنچلىك ھالدا بۇنىڭدا مۆمىنلەر ئۈچۈن (اﷲ نىڭ بىرلىكىنى كۆرسىتىدىغان) دەلىل بار
ٱتْلُ مَآ أُوحِىَ إِلَيْكَ مِنَ ٱلْكِتَـٰبِ وَأَقِمِ ٱلصَّلَوٰةَ ۖ إِنَّ ٱلصَّلَوٰةَ تَنْهَىٰ عَنِ ٱلْفَحْشَآءِ وَٱلْمُنكَرِ ۗ وَلَذِكْرُ ٱللَّهِ أَكْبَرُ ۗ وَٱللَّهُ يَعْلَمُ مَا تَصْنَعُونَ ﴿٤٥﴾
(ئى مۇھەممەد!) ساڭا ۋەھىي قىلىنغان مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) تىلاۋەت قىلغىن، نامازنى (تەئدىل ئەركان بىلەن) ئوقۇغىن، ناماز ئىشەنچلىك ھالدا قەبىھ ئىشلاردىن ۋە گۇناھلاردىن توسىدۇ، اﷲ نى ياد ئېتىش ھەممىدىن (يەنى ئۇنىڭدىن باشقا ھەممە ئىبادەتتىن) ئۇلۇغدۇر، اﷲ قىلىۋاتقان (ھەممە) ئىشىڭلارنى بىلىپ تۇرىدۇ
۞ وَلَا تُجَـٰدِلُوٓا۟ أَهْلَ ٱلْكِتَـٰبِ إِلَّا بِٱلَّتِى هِىَ أَحْسَنُ إِلَّا ٱلَّذِينَ ظَلَمُوا۟ مِنْهُمْ ۖ وَقُولُوٓا۟ ءَامَنَّا بِٱلَّذِىٓ أُنزِلَ إِلَيْنَا وَأُنزِلَ إِلَيْكُمْ وَإِلَـٰهُنَا وَإِلَـٰهُكُمْ وَٰحِدٌ وَنَحْنُ لَهُۥ مُسْلِمُونَ ﴿٤٦﴾
ئەھلى مۇقەددەس كىتاب-ۋەھىي بىلەن پەقەت ئەڭ چىرايلىق رەۋىشتە مۇنازىرىلىشىڭلار، ئۇلارنىڭ ئىچىدىكى زۇلۇم قىلغۇچىلار (يەنى سىلەرگە قارشى ئۇرۇش قىلغۇچىلار ۋە فىدىيە بېرىشتىن باش تارتقۇچىلار) بۇنىڭدىن مۇستەسنا، ئېيتىڭلاركى، «بىزگە نازىل قىلىنغان مۇقەددەس كىتاب-ۋەھىيقا ۋە سىلەرگە نازىل قىلىنغان مۇقەددەس كىتاب-ۋەھىيقا ئىمان ئېيتتۇق، سىلەرنىڭ ئىلاھىڭلار ۋە بىزنىڭ ئىلاھىمىز بىردۇر، بىز اﷲ قا بويسۇنغۇچىلارمىز»
وَكَذَٰلِكَ أَنزَلْنَآ إِلَيْكَ ٱلْكِتَـٰبَ ۚ فَٱلَّذِينَ ءَاتَيْنَـٰهُمُ ٱلْكِتَـٰبَ يُؤْمِنُونَ بِهِۦ ۖ وَمِنْ هَـٰٓؤُلَآءِ مَن يُؤْمِنُ بِهِۦ ۚ وَمَا يَجْحَدُ بِـَٔايَـٰتِنَآ إِلَّا ٱلْكَـٰفِرُونَ ﴿٤٧﴾
(ئى مۇھەممەد!) شۇنىڭدەك (يەنى سەندىن ئىلگىرىكىلەرگە مۇقەددەس كىتاب-ۋەھىي نازىل قىلغاندەك) ساڭىمۇ مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) نازىل قىلدۇق، بىز مۇقەددەس كىتاب-ۋەھىي بەرگەنلەر (يەنى يەھۇدىيلار ناسارالار) دىن قۇرئانغا ئىشىنىدىغانلار بار، بۇلاردىن (يەنى مەككە ئاھالىسىدىن) مۇ قۇرئانغا ئىشىنىدىغانلار بار، بىزنىڭ ئايەتلىرىمىزنى پەقەت ھەقىقەتنى رەت قىلغۇچىلارلا ئىنكار قىلىدۇ
وَمَا كُنتَ تَتْلُوا۟ مِن قَبْلِهِۦ مِن كِتَـٰبٍ وَلَا تَخُطُّهُۥ بِيَمِينِكَ ۖ إِذًا لَّٱرْتَابَ ٱلْمُبْطِلُونَ ﴿٤٨﴾
ئىلگىرى سەن مۇقەددەس كىتاب-ۋەھىي ئوقۇشنى بىلمەيتتىڭ، خەت يېزىشنىمۇ بىلمەيتتىڭ، (مۇبادا سەن مۇقەددەس كىتاب-ۋەھىي ئوقۇغان، خەت يازىدىغان بولساڭ) ئۇ چاغدا ھەققە قارشى تۇرغۇچىلار چوقۇم گۇمانلىناتتى
بَلْ هُوَ ءَايَـٰتٌۢ بَيِّنَـٰتٌ فِى صُدُورِ ٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ ۚ وَمَا يَجْحَدُ بِـَٔايَـٰتِنَآ إِلَّا ٱلظَّـٰلِمُونَ ﴿٤٩﴾
ئۇنداق ئەمەس (يەنى ئىش زالىملار گۇمان قىلغاندەك ئەمەس)، قۇرئان ئىلىم بېرىلگەنلەرنىڭ كۆڭۈللىرىدە ساقلانغان روشەن ئايەتلەردۇر، بىزنىڭ ئايەتلىرىمىزنى پەقەت زالىملارلا ئىنكار قىلىدۇ
وَقَالُوا۟ لَوْلَآ أُنزِلَ عَلَيْهِ ءَايَـٰتٌ مِّن رَّبِّهِۦ ۖ قُلْ إِنَّمَا ٱلْـَٔايَـٰتُ عِندَ ٱللَّهِ وَإِنَّمَآ أَنَا۠ نَذِيرٌ مُّبِينٌ ﴿٥٠﴾
ئۇلار: «نېمىشقا ئۇنىڭغا پەرۋەردىگارىدىن مۆجىزىلەر نازىل قىلىنمايدۇ» دېدى. ئېيتقىنكى، «مۆجىزىلەر ئىشەنچلىك ھالدا اﷲ نىڭ دەرگاھىدىدۇر (مېنىڭ قولۇمدا ئەمەستۇر)، مەن پەقەت بىر ئاشكارا ئاگاھلاندۇرغۇچىمەن»
أَوَلَمْ يَكْفِهِمْ أَنَّآ أَنزَلْنَا عَلَيْكَ ٱلْكِتَـٰبَ يُتْلَىٰ عَلَيْهِمْ ۚ إِنَّ فِى ذَٰلِكَ لَرَحْمَةً وَذِكْرَىٰ لِقَوْمٍ يُؤْمِنُونَ ﴿٥١﴾
بىزنىڭ ساڭا ئۇلارغا تىلاۋەت قىلىنىپ تۇرىدىغان مۇقەددەس كىتاب-ۋەھىينى نازىل قىلغانلىقىمىز ئۇلارغا (مۆجىزە بولۇشقا) كۇپايە قىلمىدىمۇ؟ بۇ مۇقەددەس كىتاب-ۋەھىيتا ئىمان ئېيتقان قەۋم ئۈچۈن شەك - شۈبھىسىز رەھمەت ۋە ۋەز - نەسىھەت بار
قُلْ كَفَىٰ بِٱللَّهِ بَيْنِى وَبَيْنَكُمْ شَهِيدًا ۖ يَعْلَمُ مَا فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۗ وَٱلَّذِينَ ءَامَنُوا۟ بِٱلْبَـٰطِلِ وَكَفَرُوا۟ بِٱللَّهِ أُو۟لَـٰٓئِكَ هُمُ ٱلْخَـٰسِرُونَ ﴿٥٢﴾
ئېيتقىنكى، «مەن بىلەن سىلەرنىڭ ئاراڭلاردا (مېنىڭ راست خۇدانىڭ ئەلچىسى ئىكەنلىكىمگە) گۇۋاھ بولۇشقا اﷲ كۇپايىدۇر، اﷲ ئاسمانلاردىكى ۋە زېمىندىكى ھەممىنى بىلىپ تۇرىدۇ، بۇتقا ئىشەنگەن، اﷲ نى ئىنكار قىلغان ئادەملەر زىيان تارتقۇچىلاردۇر»
وَيَسْتَعْجِلُونَكَ بِٱلْعَذَابِ ۚ وَلَوْلَآ أَجَلٌ مُّسَمًّى لَّجَآءَهُمُ ٱلْعَذَابُ وَلَيَأْتِيَنَّهُم بَغْتَةً وَهُمْ لَا يَشْعُرُونَ ﴿٥٣﴾
ئۇلار سەندىن ئازابنىڭ چاپسان كېلىشىنى تەلەپ قىلىدۇ، اﷲ (ئۇلارنىڭ ھالاك بولۇشىنىڭ) مۇئەييەن ۋاقتىنى بەلگىلىمىگەن بولسا ئىدى، ئۇلارغا ئازاب ئەلۋەتتە (ئۇلار تەلەپ قىلغان ۋاقىتتىلا) كېلەتتى، ئۇلارغا ئازاب چوقۇم ئۇلار خەۋەرسىز ھالەتتە تۇرغاندا ئۇشتۇمتۇت كېلىدۇ
يَسْتَعْجِلُونَكَ بِٱلْعَذَابِ وَإِنَّ جَهَنَّمَ لَمُحِيطَةٌۢ بِٱلْكَـٰفِرِينَ ﴿٥٤﴾
ئۇلار سەندىن ئازابنىڭ تېز كېلىشىنى تەلەپ قىلىدۇ، شەك - شۈبھىسىزكى، مەڭگۈلۈك ئوت (قىيامەت كۈنى) ھەقىقەتنى رەت قىلغۇچىلارنى قورشاپ تۇرغۇچىدۇر
يَوْمَ يَغْشَىٰهُمُ ٱلْعَذَابُ مِن فَوْقِهِمْ وَمِن تَحْتِ أَرْجُلِهِمْ وَيَقُولُ ذُوقُوا۟ مَا كُنتُمْ تَعْمَلُونَ ﴿٥٥﴾
ئۇ كۈندە ئازاب ئۇلارنى ئۈستىلىرىدىن، ئاياغلىرىنىڭ ئاستىدىن ئورىۋالىدۇ. اﷲ ئۇلارغا: «(دۇنيادا قىلغان ئەمەللىرىڭلارنىڭ) جازاسىنى تېتىڭلار» دەيدۇ
يَـٰعِبَادِىَ ٱلَّذِينَ ءَامَنُوٓا۟ إِنَّ أَرْضِى وَٰسِعَةٌ فَإِيَّـٰىَ فَٱعْبُدُونِ ﴿٥٦﴾
ئى مۆمىن بەندىلىرىم! مېنىڭ زېمىنىم ئىشەنچلىك ھالدا كەڭدۇر، ماڭىلا ئىبادەت قىلىڭلار
كُلُّ نَفْسٍ ذَآئِقَةُ ٱلْمَوْتِ ۖ ثُمَّ إِلَيْنَا تُرْجَعُونَ ﴿٥٧﴾
ھەربىر جاندار ئۆلۈمنىڭ تەمىنى تېتىغۇچىدۇر، ئاندىن بىزنىڭ دەرگاھىمىزغا قايتۇرۇلىسىلەر
وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ لَنُبَوِّئَنَّهُم مِّنَ ٱلْجَنَّةِ غُرَفًا تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا ۚ نِعْمَ أَجْرُ ٱلْعَـٰمِلِينَ ﴿٥٨﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغان كىشىلەرنى مەڭگۈلۈك باغلارنىڭ ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان ئالىي جايلىرىغا ئورۇنلاشتۇرىمىز، ئۇ يەرلەردە ئۇلار مەڭگۈ قالىدۇ، ياخشى ئەمەللەرنى قىلغۇچىلارغا بېرىلگەن مۇكاپات نېمىدېگەن ياخشى!
ٱلَّذِينَ صَبَرُوا۟ وَعَلَىٰ رَبِّهِمْ يَتَوَكَّلُونَ ﴿٥٩﴾
ئۇلار (ھىجرەت قىلىشنىڭ جاپا - مۇشەققەتلىرىگە، اﷲ نىڭ يولىدا يەتكەن ئەزىيەتلەرگە) سەۋر قىلىدۇ، (ھەممە ئىشلىرىدا) پەرۋەردىگارىغا يۆلىنىدۇ
وَكَأَيِّن مِّن دَآبَّةٍ لَّا تَحْمِلُ رِزْقَهَا ٱللَّهُ يَرْزُقُهَا وَإِيَّاكُمْ ۚ وَهُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٦٠﴾
نۇرغۇن ھايۋانلار ئۆز رىزقىنى ئۈستىگە ئالالمايدۇ (يەنى ئۆز رىزقىنى تېپىپ يېيىشتىن ئاجىزدۇر)، ئۇلارنى ۋە سىلەرنى اﷲ رىزىقلاندۇرىدۇ، اﷲ (سۆزۈڭلارنى) ئاڭلاپ تۇرغۇچىدۇر، (ئەھۋالىڭلارنى) بىلىپ تۇرغۇچىدۇر
وَلَئِن سَأَلْتَهُم مَّنْ خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَسَخَّرَ ٱلشَّمْسَ وَٱلْقَمَرَ لَيَقُولُنَّ ٱللَّهُ ۖ فَأَنَّىٰ يُؤْفَكُونَ ﴿٦١﴾
ئەگەر سەن ئۇلاردىن: «ئاسمانلارنى ۋە زېمىننى كىم ياراتتى، كۈننى ۋە ئاينى كىم (بەندىلەرنىڭ مەنپەئىتى ئۈچۈن) بويسۇندۇردى؟» دەپ سورىساڭ، ئۇلار چوقۇم: «اﷲ» دەپ جاۋاب بېرىدۇ، ئۇلار قانداقمۇ (اﷲ نىڭ ئىبادىتىدىن) باش تارتىدۇ!
ٱللَّهُ يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ مِنْ عِبَادِهِۦ وَيَقْدِرُ لَهُۥٓ ۚ إِنَّ ٱللَّهَ بِكُلِّ شَىْءٍ عَلِيمٌ ﴿٦٢﴾
اﷲ بەندىلىرىدىن قايسى ئادەمنىڭ رىزقىنى كەڭ قىلماقچى بولىدىكەن، ئۇنى كەڭ قىلىدۇ، (قايسى بەندىسىنىڭ رىزقىنى تار قىلماقچى بولىدىكەن) ئۇنى تار قىلىدۇ، اﷲ ئىشەنچلىك ھالدا ھەممە نەرسىنى بىلىپ تۇرغۇچىدۇر
وَلَئِن سَأَلْتَهُم مَّن نَّزَّلَ مِنَ ٱلسَّمَآءِ مَآءً فَأَحْيَا بِهِ ٱلْأَرْضَ مِنۢ بَعْدِ مَوْتِهَا لَيَقُولُنَّ ٱللَّهُ ۚ قُلِ ٱلْحَمْدُ لِلَّهِ ۚ بَلْ أَكْثَرُهُمْ لَا يَعْقِلُونَ ﴿٦٣﴾
ئەگەر سەن ئۇلاردىن: «بۇلۇتتىن يامغۇر سۈيىنى چۈشۈرۈپ ئۇنىڭ بىلەن ئۆلگەن زېمىننى تېرىلدۈرگەن كىم؟» دەپ سورىساڭ، ئۇلار چوقۇم «اﷲ» دەپ جاۋاپ بېرىدۇ. ئېيتقىنكى، «(سىلەرگە قارشى پاكىتلار بار بولغانلىقتىن) جىمى ھەمدۇسانا اﷲ قا خاستۇر»، بەلكى ئۇلارنىڭ تولىسى (بۇ سۆزلىرىدىكى زىددىيەتنى) چۈشەنمەيدۇ
وَمَا هَـٰذِهِ ٱلْحَيَوٰةُ ٱلدُّنْيَآ إِلَّا لَهْوٌ وَلَعِبٌ ۚ وَإِنَّ ٱلدَّارَ ٱلْـَٔاخِرَةَ لَهِىَ ٱلْحَيَوَانُ ۚ لَوْ كَانُوا۟ يَعْلَمُونَ ﴿٦٤﴾
دۇنيا تىرىكچىلىكى پەقەت ئويۇن - كۈلكە ۋە غەپلەتتىن ئىبارەتتۇر، شۈبھىسىزكى، ئاخىرەت يۇرتى ھەقىيقى ھايات (يۇرتىدۇر)، ئەگەر ئۇلار بىلسە ئىدى (بۇ دۇنيانى ئۇ دۇنيادىن ئارتۇق كۆرمەيتتى)
فَإِذَا رَكِبُوا۟ فِى ٱلْفُلْكِ دَعَوُا۟ ٱللَّهَ مُخْلِصِينَ لَهُ ٱلدِّينَ فَلَمَّا نَجَّىٰهُمْ إِلَى ٱلْبَرِّ إِذَا هُمْ يُشْرِكُونَ ﴿٦٥﴾
ئۇلار كېمىگە چىقىپ (غەرق بولۇشتىن قورققان) چاغلىرىدا، اﷲ قا كامالى ئىخلاس بىلەن ئىلتىجا قىلىدۇ، اﷲ ئۇلارنى ئامان - ئېسەن قۇرۇقلۇققا چىقارغان چاغدا (بالا - قازادىن قۇتۇلدۇرغان اﷲ نى ئۇنتۇپ)، ناگاھان (اﷲ قا) شېرىك كەلتۈرىدۇ
لِيَكْفُرُوا۟ بِمَآ ءَاتَيْنَـٰهُمْ وَلِيَتَمَتَّعُوا۟ ۖ فَسَوْفَ يَعْلَمُونَ ﴿٦٦﴾
ئۇلار بەرگەن نېمىتىمىز (يەنى دېڭىزدىن قۇتۇلدۇرغانلىقىمىزغا ناشۈكۈرلۈك قىلىپ باقسۇن، قالغان ئۆمۈرلىرىدىن) بەھرىمەن بولسۇن، بۇلار (ئىشنىڭ ئاقىۋىتىنى) كەلگۈسىدە بىلىدۇ
أَوَلَمْ يَرَوْا۟ أَنَّا جَعَلْنَا حَرَمًا ءَامِنًا وَيُتَخَطَّفُ ٱلنَّاسُ مِنْ حَوْلِهِمْ ۚ أَفَبِٱلْبَـٰطِلِ يُؤْمِنُونَ وَبِنِعْمَةِ ٱللَّهِ يَكْفُرُونَ ﴿٦٧﴾
ئۇلار (يەنى مۇشرىكلار) كۆرمەمدۇكى، بىز ھەرەمنى ئامان (جاي) قىلدۇق، ھالبۇكى، ئۇلارنىڭ ئەتراپىدىكى كىشىلەر ئەسىر ئېلىنىۋاتىدۇ ۋە ئۆلتۈرۈلۈۋاتىدۇ، ئۇلار باتىلغا ئىشىنىپ، اﷲ نىڭ نېمىتىنى ئىنكار قىلامدۇ؟
وَمَنْ أَظْلَمُ مِمَّنِ ٱفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا أَوْ كَذَّبَ بِٱلْحَقِّ لَمَّا جَآءَهُۥٓ ۚ أَلَيْسَ فِى جَهَنَّمَ مَثْوًى لِّلْكَـٰفِرِينَ ﴿٦٨﴾
اﷲ قا يالغاننى چاپلىغان ياكى ھەق كەلگەندە ئۇنى ئىنكار قىلغان ئادەمدىنمۇ زالىم ئادەم بارمۇ؟ مەڭگۈلۈك ئوتدە ھەقىقەتنى رەت قىلغۇچىلارغا جاي يوقمىدۇ؟ (ئەلۋەتتە بار)
وَٱلَّذِينَ جَـٰهَدُوا۟ فِينَا لَنَهْدِيَنَّهُمْ سُبُلَنَا ۚ وَإِنَّ ٱللَّهَ لَمَعَ ٱلْمُحْسِنِينَ ﴿٦٩﴾
بىز ئۈچۈن كۈرەش قىلغانلارنى ئەلۋەتتە يولىمىزغا يېتەكلەيمىز، اﷲ ئىشەنچلىك ھالدا ياخشى ئىش قىلغۇچىلار بىلەن بىللىدۇر
