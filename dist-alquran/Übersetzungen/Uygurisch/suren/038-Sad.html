Sad · The Letter "Saad"
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
صٓ ۚ وَٱلْقُرْءَانِ ذِى ٱلذِّكْرِ ﴿١﴾
صاد. شانلىق قۇرئان بىلەن قەسەمكى، (بۇ قۇرئان ئىشەنچلىك ھالدا مۆجىزىدۇر)
بَلِ ٱلَّذِينَ كَفَرُوا۟ فِى عِزَّةٍ وَشِقَاقٍ ﴿٢﴾
بەلكى ھەقىقەتنى رەت قىلغۇچىلار ئىماندىن باش تارتماقتا ۋە (رەسۇلۇللاھقا) مۇخالىپەتچىلىك قىلماقتا
كَمْ أَهْلَكْنَا مِن قَبْلِهِم مِّن قَرْنٍ فَنَادَوا۟ وَّلَاتَ حِينَ مَنَاصٍ ﴿٣﴾
ئۇلاردىن ئىلگىرى (كۇفرى تۈپەيلىدىن) نۇرغۇن ئۈممەتلەرنى ھالاك قىلدۇق، ئۇلار (ئازابنى كۆرگەندە) پەرياد قىلدى، ھالبۇكى، (بۇ) قاچىدىغان ۋاقىت ئەمەس
وَعَجِبُوٓا۟ أَن جَآءَهُم مُّنذِرٌ مِّنْهُمْ ۖ وَقَالَ ٱلْكَـٰفِرُونَ هَـٰذَا سَـٰحِرٌ كَذَّابٌ ﴿٤﴾
ئۇلارغا ئۆزلىرىنىڭ ئىچىدىن بىر ئاگاھلاندۇرغۇچى (يەنى خۇدانىڭ ئەلچىسى) كەلگەنلىكىگە ئۇلار ھەيران قېلىشتى، ھەقىقەتنى رەت قىلغۇچىلار ئېيتتى: «بۇ بىر يالغانچى سېھىرگەردۇر
أَجَعَلَ ٱلْـَٔالِهَةَ إِلَـٰهًا وَٰحِدًا ۖ إِنَّ هَـٰذَا لَشَىْءٌ عُجَابٌ ﴿٥﴾
ئۇ نۇرغۇن ئىلاھلارنى بىر ئىلاھ قىلماقچىمۇ؟ بۇ ئىشەنچلىك ھالدا غەلىتە ئىش»
وَٱنطَلَقَ ٱلْمَلَأُ مِنْهُمْ أَنِ ٱمْشُوا۟ وَٱصْبِرُوا۟ عَلَىٰٓ ءَالِهَتِكُمْ ۖ إِنَّ هَـٰذَا لَشَىْءٌ يُرَادُ ﴿٦﴾
ئۇلارنىڭ (يەنى قۇرەيشنىڭ) كاتتىلىرى (سورۇنلىرىدىن) قوزغىلىپ چىقىپ ئېيتتى: «سىلەر مېڭىڭلار، مەبۇدلىرىڭلارغا ئىبادەت قىلىشتا چىڭ تۇرۇڭلار، بۇ ئىشەنچلىك ھالدا (بىزدىن) ئىرادە قىلىنىدىغان بىر ئىش (يەنى مۇھەممەد سىلەرنى دىنىڭلاردىن تاندۇرۇپ، سىلەرگە ئۆزىنىڭ ئۈستۈنلۈك قازىنىشىنى ئىرادە قىلىدۇ)
مَا سَمِعْنَا بِهَـٰذَا فِى ٱلْمِلَّةِ ٱلْـَٔاخِرَةِ إِنْ هَـٰذَآ إِلَّا ٱخْتِلَـٰقٌ ﴿٧﴾
بۇنى بىز باشقا بىر دىندىن (يەنى ناسارا دىنىدىكىلەردىن) ئاڭلىمىدۇق، بۇ پەقەت ئويدۇرمىدۇر
أَءُنزِلَ عَلَيْهِ ٱلذِّكْرُ مِنۢ بَيْنِنَا ۚ بَلْ هُمْ فِى شَكٍّ مِّن ذِكْرِى ۖ بَل لَّمَّا يَذُوقُوا۟ عَذَابِ ﴿٨﴾
قۇرئان بىزگە نازىل بولماي ئۇنىڭغا نازىل بولدىمۇ؟» بەلكى ئۇلار مېنىڭ زىكرىم (يەنى قۇرئان) دىن شەكتىدۇر، بەلكى ئۇلار تېخى مېنىڭ ئازابىمنى تېتىغىنى يوق
أَمْ عِندَهُمْ خَزَآئِنُ رَحْمَةِ رَبِّكَ ٱلْعَزِيزِ ٱلْوَهَّابِ ﴿٩﴾
ياكى ئۇلاردا غالىب، ناھايىتى كەرەملىك پەرۋەردىگارىڭنىڭ رەھمىتىنىڭ خەزىنىلىرى بارمۇ؟
أَمْ لَهُم مُّلْكُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ۖ فَلْيَرْتَقُوا۟ فِى ٱلْأَسْبَـٰبِ ﴿١٠﴾
ياكى ئۇلار ئاسمانلارنىڭ، زېمىننىڭ ۋە ئۇلارنىڭ ئارىسىدىكى نەرسىلەرنىڭ پادىشاھلىقىغا ئىگىمۇ؟ (ئەگەر ئۇلار ئىگە بولسا، ئۇلار ئۆزلىرىنى ئاسمانغا ئېلىپ چىقىدىغان) شوتىلارغا چىقسۇن
جُندٌ مَّا هُنَالِكَ مَهْزُومٌ مِّنَ ٱلْأَحْزَابِ ﴿١١﴾
ئۇلار (رەسۇلۇللاھقا قارشى) جامائەلەردىن ئۇيۇشقان مەغلۇب قىلىنىدىغان قوشۇندۇر
كَذَّبَتْ قَبْلَهُمْ قَوْمُ نُوحٍ وَعَادٌ وَفِرْعَوْنُ ذُو ٱلْأَوْتَادِ ﴿١٢﴾
ئۇلاردىن (يەنى قۇرەيش مۇشرىكلىرىدىن) ئىلگىرى نۇھ قەۋمى، ئاد (قەۋمى) ۋە قوزۇقلار ئىگىسى پىرئەۋن (قەۋمى) (خۇدانىڭ ئەلچىسىلىرىنى) ئىنكار قىلدى
وَثَمُودُ وَقَوْمُ لُوطٍ وَأَصْحَـٰبُ لْـَٔيْكَةِ ۚ أُو۟لَـٰٓئِكَ ٱلْأَحْزَابُ ﴿١٣﴾
سەمۇد قەۋمى، لۇت قەۋمى ۋە ئەيكىلىكلەر (يەنى شۇئەيب ئەلەيھىسسالام قەۋمى) مۇ (خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلدى)، ئەنە شۇلار (خۇدانىڭ ئەلچى-رەسۇللىرىگە قارشى ئۇيۇشقان) جامائەلەر ئىدى
إِن كُلٌّ إِلَّا كَذَّبَ ٱلرُّسُلَ فَحَقَّ عِقَابِ ﴿١٤﴾
ئۇلار (يەنى جامائەلەر) دىن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلمىغان بىرىمۇ يوق، شۇنىڭ ئۈچۈن ئۇلارغا مېنىڭ ئازابىم تېگىشلىك بولدى
وَمَا يَنظُرُ هَـٰٓؤُلَآءِ إِلَّا صَيْحَةً وَٰحِدَةً مَّا لَهَا مِن فَوَاقٍ ﴿١٥﴾
بۇلار (يەنى مەككە مۇشرىكلىرى) پەقەت بىر ئاۋاز (يەنى بىرىنچى قېتىملىق سۇر) نىلا كۈتىدۇ، ئۇ قايتىلانمايدۇ
وَقَالُوا۟ رَبَّنَا عَجِّل لَّنَا قِطَّنَا قَبْلَ يَوْمِ ٱلْحِسَابِ ﴿١٦﴾
ئۇلار (مەسخىرە قىلىش يولى بىلەن): «پەرۋەردىگارىمىز! (سەن بىزگە ئازابتىن ۋەدە قىلغان) نېسىۋىمىزنى بىزگە بالدۇرراق – قىيامەت كۈنىدىن ئىلگىرى بەرگىن» دېيىشتى
ٱصْبِرْ عَلَىٰ مَا يَقُولُونَ وَٱذْكُرْ عَبْدَنَا دَاوُۥدَ ذَا ٱلْأَيْدِ ۖ إِنَّهُۥٓ أَوَّابٌ ﴿١٧﴾
ئۇلارنىڭ سۆزلىرىگە سەۋر قىلغىن، (دىندا) كۈچلۈك بەندىمىز داۋۇدنى ئەسلىگىن، ئۇ ئىشەنچلىك ھالدا اﷲ قا ئىتائەت قىلغۇچى ئىدى
إِنَّا سَخَّرْنَا ٱلْجِبَالَ مَعَهُۥ يُسَبِّحْنَ بِٱلْعَشِىِّ وَٱلْإِشْرَاقِ ﴿١٨﴾
بىز ئىشەنچلىك ھالدا تاغلارنى (داۋۇدقا) بويسۇندۇرۇپ بەردۇق، ئۇلار داۋۇد بىلەن ئەتىگىنى - ئاخشىمى تەسبىھ ئېيتاتتى
وَٱلطَّيْرَ مَحْشُورَةً ۖ كُلٌّ لَّهُۥٓ أَوَّابٌ ﴿١٩﴾
قۇشلارنىمۇ ئۇنىڭغا بويسۇندۇرۇپ بەردۇق، ئۇلارمۇ (تەرەپ - تەرەپتىن) توپلىنىپ، داۋۇد بىلەن بىللە تەسبىھ ئېيتاتتى، (تاغلارنىڭ، قۇشلارنىڭ) ھەممىسى اﷲ قا ئىتائەت قىلغۇچىدۇر
وَشَدَدْنَا مُلْكَهُۥ وَءَاتَيْنَـٰهُ ٱلْحِكْمَةَ وَفَصْلَ ٱلْخِطَابِ ﴿٢٠﴾
داۋۇدنىڭ سەلتەنىتىنى كۈچەيتتۇق، ئۇنىڭغا ھېكمەت ۋە (ھەممە ئادەم چۈشىنىدىغان) روشەن سۆز - ئىبارىلەرنى ئاتا قىلدۇق
۞ وَهَلْ أَتَىٰكَ نَبَؤُا۟ ٱلْخَصْمِ إِذْ تَسَوَّرُوا۟ ٱلْمِحْرَابَ ﴿٢١﴾
(ئى مۇھەممەد!) دەۋا قىلغۇچىلارنىڭ قىسسىسى ساڭا يەتتىمۇ؟ ئۆز ۋاقتىدا ئۇلار (داۋۇد ئىبادەت قىلىۋاتقان) مەسجىدنىڭ تېمىدىن ئارتىلىپ چۈشۈشتى
إِذْ دَخَلُوا۟ عَلَىٰ دَاوُۥدَ فَفَزِعَ مِنْهُمْ ۖ قَالُوا۟ لَا تَخَفْ ۖ خَصْمَانِ بَغَىٰ بَعْضُنَا عَلَىٰ بَعْضٍ فَٱحْكُم بَيْنَنَا بِٱلْحَقِّ وَلَا تُشْطِطْ وَٱهْدِنَآ إِلَىٰ سَوَآءِ ٱلصِّرَٰطِ ﴿٢٢﴾
ئەينى ۋاقىتتا ئۇلار داۋۇدنىڭ يېنىغا كىردى. داۋۇد ئۇلاردىن قورقتى، ئۇلار (داۋۇدقا): «قورقمىغىن، بىز بىرىمىز - بىرىمىزگە چېقىلغان ئىككى دەۋاگەرمىز، بىزنىڭ ئارىمىزدا ئادىللىق بىلەن ھۆكۈم قىلغىن، ناھەق ھۆكۈم چىقارمىغىن، بىزنى توغرا يولغا باشلىغىن» دېدى
إِنَّ هَـٰذَآ أَخِى لَهُۥ تِسْعٌ وَتِسْعُونَ نَعْجَةً وَلِىَ نَعْجَةٌ وَٰحِدَةٌ فَقَالَ أَكْفِلْنِيهَا وَعَزَّنِى فِى ٱلْخِطَابِ ﴿٢٣﴾
(ئۇلارنىڭ بىرى) «شۈبھىسىزكى، مېنىڭ بۇ قېرىندىشىمنىڭ 99 ساغلىقى بار، مېنىڭ بىر ساغلىقىم بار، قېرىندىشىم ئۇ بىر ساغلىقنىمۇ ماڭا بەرگىن دەيدۇ، (ئارىمىزدا بۇ توغرۇلۇق مۇنازىرە بولۇپ) ئۇ مېنى سۆزدە يېڭىپ قويدى» دېدى
قَالَ لَقَدْ ظَلَمَكَ بِسُؤَالِ نَعْجَتِكَ إِلَىٰ نِعَاجِهِۦ ۖ وَإِنَّ كَثِيرًا مِّنَ ٱلْخُلَطَآءِ لَيَبْغِى بَعْضُهُمْ عَلَىٰ بَعْضٍ إِلَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ وَقَلِيلٌ مَّا هُمْ ۗ وَظَنَّ دَاوُۥدُ أَنَّمَا فَتَنَّـٰهُ فَٱسْتَغْفَرَ رَبَّهُۥ وَخَرَّ رَاكِعًا وَأَنَابَ ۩ ﴿٢٤﴾
داۋۇد ئېيتتى: «ئۇ سېنىڭ ساغلىقىڭنى ئۆزىنىڭ ساغلىقلىرىغا قوشۇۋېلىشنى تەلەپ قىلىپ راستلا سېنى بوزەك قىلىپتۇ، نۇرغۇن شېرىكلەر (يەنى دوستلار)، شۈبھىسىزكى، بىر - بىرىگە چېقىلىدۇ، پەقەت ئىمان ئېيتقان ۋە ياخشى ئەمەل قىلغانلارلا (بىر - بىرىگە چېقىلمايدۇ) بۇلار ئازدۇر». داۋۇد بىزنىڭ ئۇنى سىنىغانلىقىمىزنى بىلدى. پەرۋەردىگارىدىن مەغپىرەت تەلەپ قىلدى، سەجدىگە باردى. (اﷲ قا) تەۋبە قىلدى
فَغَفَرْنَا لَهُۥ ذَٰلِكَ ۖ وَإِنَّ لَهُۥ عِندَنَا لَزُلْفَىٰ وَحُسْنَ مَـَٔابٍ ﴿٢٥﴾
بىز ئۇنىڭ خاتالىقىنى ئەپۇ قىلدۇق، ئۇ بىزنىڭ دەرگاھىمىزدا ئەلۋەتتە يېقىنلىققا ۋە ياخشى ئاقىۋەتكە ئىگە بولىدۇ
يَـٰدَاوُۥدُ إِنَّا جَعَلْنَـٰكَ خَلِيفَةً فِى ٱلْأَرْضِ فَٱحْكُم بَيْنَ ٱلنَّاسِ بِٱلْحَقِّ وَلَا تَتَّبِعِ ٱلْهَوَىٰ فَيُضِلَّكَ عَن سَبِيلِ ٱللَّهِ ۚ إِنَّ ٱلَّذِينَ يَضِلُّونَ عَن سَبِيلِ ٱللَّهِ لَهُمْ عَذَابٌ شَدِيدٌۢ بِمَا نَسُوا۟ يَوْمَ ٱلْحِسَابِ ﴿٢٦﴾
ئى داۋۇد! سېنى بىز ئىشەنچلىك ھالدا يەر يۈزىدە خەلىپە قىلدۇق، كىشىلەرنىڭ ئارىسىدا ئادىللىق بىلەن ھۆكۈم چىقارغىن، نەپسى خاھىشقا ئەگەشمىگىنكى، ئۇ سېنى اﷲ نىڭ يولىدىن ئازدۇرىدۇ، اﷲ نىڭ يولىدىن ئازغانلار ھېساب كۈنى (يەنى قىيامەت كۈنى) نى ئۇنتۇغانلىقلىرى ئۈچۈن ئىشەنچلىك ھالدا قاتتىق ئازابقا دۇچار بولىدۇ
وَمَا خَلَقْنَا ٱلسَّمَآءَ وَٱلْأَرْضَ وَمَا بَيْنَهُمَا بَـٰطِلًا ۚ ذَٰلِكَ ظَنُّ ٱلَّذِينَ كَفَرُوا۟ ۚ فَوَيْلٌ لِّلَّذِينَ كَفَرُوا۟ مِنَ ٱلنَّارِ ﴿٢٧﴾
ئاسماننى، زېمىننى ۋە ئۇلارنىڭ ئارىسىدىكى شەيئىلەرنى بىكار ياراتمىدۇق، ئۇ (يەنى ئۇلارنى بىكار يارىتىلغان دەپ قاراش) ھەقىقەتنى رەت قىلغۇچىلارنىڭ گۇمانىدۇر، ھەقىقەتنى رەت قىلغۇچىلارغا دوزاختىن ۋاي!
أَمْ نَجْعَلُ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ كَٱلْمُفْسِدِينَ فِى ٱلْأَرْضِ أَمْ نَجْعَلُ ٱلْمُتَّقِينَ كَٱلْفُجَّارِ ﴿٢٨﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنى يەر يۈزىدە بۇزغۇنچىلىق قىلغانلار بىلەن ئوخشاش قىلامدۇق؟ ياكى تەقۋادارلارنى فاجىرلارغا ئوخشاش قىلامدۇق؟
كِتَـٰبٌ أَنزَلْنَـٰهُ إِلَيْكَ مُبَـٰرَكٌ لِّيَدَّبَّرُوٓا۟ ءَايَـٰتِهِۦ وَلِيَتَذَكَّرَ أُو۟لُوا۟ ٱلْأَلْبَـٰبِ ﴿٢٩﴾
(بۇ) ئۇلارنىڭ ئايەتلەرنى تەپەككۈر قىلىشلىرى ئۈچۈن، ئەقىل ئىگىلىرىنىڭ ۋەز - نەسىھەت ئېلىشلىرى ئۈچۈن بىز ساڭا نازىل قىلغان مۇبارەك مۇقەددەس كىتاب-ۋەھىيتۇر
وَوَهَبْنَا لِدَاوُۥدَ سُلَيْمَـٰنَ ۚ نِعْمَ ٱلْعَبْدُ ۖ إِنَّهُۥٓ أَوَّابٌ ﴿٣٠﴾
بىز داۋۇدقا سۇلەيماننى ئاتا قىلدۇق، سۇلەيمان نېمىدېگەن ياخشى بەندە! ئۇ (اﷲ قا) ئىشەنچلىك ھالدا ئىتائەت قىلغۇچى ئىدى
إِذْ عُرِضَ عَلَيْهِ بِٱلْعَشِىِّ ٱلصَّـٰفِنَـٰتُ ٱلْجِيَادُ ﴿٣١﴾
ئۆز ۋاقتىدا كەچقۇرۇنلىقى ئۇنىڭغا ياخشى يۈگۈرۈك ئاتلار توغرىلاندى
فَقَالَ إِنِّىٓ أَحْبَبْتُ حُبَّ ٱلْخَيْرِ عَن ذِكْرِ رَبِّى حَتَّىٰ تَوَارَتْ بِٱلْحِجَابِ ﴿٣٢﴾
سۇلەيمان ئېيتتى: «مەن ئىشەنچلىك ھالدا ئاتلارنى پەرۋەردىگارىمنى ئەسلەش ئۈچۈن ياخشى كۆردۈم، ئۇلار (كۆزۈمدىن) غايىب بولغۇچە (ئۇلارنى كۆزدىن كۆچۈرۈش بىلەن بولدۇم)»
رُدُّوهَا عَلَىَّ ۖ فَطَفِقَ مَسْحًۢا بِٱلسُّوقِ وَٱلْأَعْنَاقِ ﴿٣٣﴾
سۇلەيمان ئېيتتى: «ئاتلارنى مېنىڭ ئالدىمغا ئېلىپ كېلىڭلار». (ئاتلار ئېلىپ كېلىنگەندىن كېيىن) سۇلەيمان ئۇلارنىڭ پاچاقلىرىنى، بويۇنلىرىنى سىلاشقا باشلىدى
وَلَقَدْ فَتَنَّا سُلَيْمَـٰنَ وَأَلْقَيْنَا عَلَىٰ كُرْسِيِّهِۦ جَسَدًا ثُمَّ أَنَابَ ﴿٣٤﴾
بىز ئىشەنچلىك ھالدا سۇلەيماننى سىنىدۇق، ئۇنىڭ تەختى ئۈستىگە بىر جەسەتنى تاشلىدۇق، ئاندىن ئۇ (بۇنىڭ سىناق ئىكەنلىكىنى بىلىپ) تەۋبە قىلدى
قَالَ رَبِّ ٱغْفِرْ لِى وَهَبْ لِى مُلْكًا لَّا يَنۢبَغِى لِأَحَدٍ مِّنۢ بَعْدِىٓ ۖ إِنَّكَ أَنتَ ٱلْوَهَّابُ ﴿٣٥﴾
سۇلەيمان ئېيىتتى: «پەرۋەردىگارىم! ماڭا مەغپىرەت قىلغىن، ماڭا مەندىن كېيىن (مەندىن باشقا) ھېچ ئادەمگە مۇيەسسەر بولمايدىغان پادىشاھلىقنى ئاتا قىلغىن، سەن ئىشەنچلىك ھالدا كۆپ ئاتا قىلغۇچىسەن»
فَسَخَّرْنَا لَهُ ٱلرِّيحَ تَجْرِى بِأَمْرِهِۦ رُخَآءً حَيْثُ أَصَابَ ﴿٣٦﴾
بىز سۇلەيمانغا شامالنى بويسۇندۇرۇپ بەردۇق. شامال سۇلەيماننىڭ بۇيرۇقى بويىچە ئۇ خالىغان تەرەپكە لەرزان ماڭاتتى
وَٱلشَّيَـٰطِينَ كُلَّ بَنَّآءٍ وَغَوَّاصٍ ﴿٣٧﴾
ھەمدە ئۇنىڭغا (كاتتا بىنالارنى سالالايدىغان) قۇرغۇچى ۋە (دېڭىزلارغا چۆكۈپ ئۈنچە - مارجانلارنى ئېلىپ چىقالايدىغان) غەۋۋاس شەيتان (ئازدۇرغۇچى)لارنى بويسۇندۇرۇپ بەردۇق
وَءَاخَرِينَ مُقَرَّنِينَ فِى ٱلْأَصْفَادِ ﴿٣٨﴾
يەنە نۇرغۇن شەيتان (ئازدۇرغۇچى)لارغا ئىشكەل سېلىنغاندۇر
هَـٰذَا عَطَآؤُنَا فَٱمْنُنْ أَوْ أَمْسِكْ بِغَيْرِ حِسَابٍ ﴿٣٩﴾
(بىز سۇلەيمانغا ئېيتتۇقكى) «بۇ بىزنىڭ (ساڭا بەرگەن كەڭ) ئاتايىمىز، (ئۇنىڭدىن سەن خالىغان كىشىگە) بەرگىن، ياكى بەرمىگىن، (بۇ ھەقتە سەندىن) ھېساب ئېلىنمايدۇ»
وَإِنَّ لَهُۥ عِندَنَا لَزُلْفَىٰ وَحُسْنَ مَـَٔابٍ ﴿٤٠﴾
ئۇ بىزنىڭ دەرگاھىمىزدا ئىشەنچلىك ھالدا يېقىنلىققا ۋە ياخشى ئاقىۋەتكە ئىگە بولىدۇ
وَٱذْكُرْ عَبْدَنَآ أَيُّوبَ إِذْ نَادَىٰ رَبَّهُۥٓ أَنِّى مَسَّنِىَ ٱلشَّيْطَـٰنُ بِنُصْبٍ وَعَذَابٍ ﴿٤١﴾
بەندىمىز ئەييۇبنى ئەسلىگىن، ئۇ ئۆز ۋاقتىدا پەرۋەردىگارىغا (ئىلتىجا قىلىپ): «شەيتان (ئازدۇرغۇچى) ماڭا ئىشەنچلىك ھالدا جاپا - مۇشەققەت ۋە ئازاب يەتكۈزدى» دەپ نىدا قىلدى
ٱرْكُضْ بِرِجْلِكَ ۖ هَـٰذَا مُغْتَسَلٌۢ بَارِدٌ وَشَرَابٌ ﴿٤٢﴾
ئۇنىڭغا: «پۇتۇڭ بىلەن يەرنى تەپكىن» دېدۇق، ئۇ پۇتى بىلەن يەرنى تېپىۋىدى، سۈزۈك بىر بۇلاق ئېتىلىپ چىقتى. (بىز ئۇنىڭغا) «بۇ يۇيۇنسىمۇ بولىدىغان، ئىچسىمۇ بولىدىغان سوغۇق سۇدۇر» (دېدۇق)
وَوَهَبْنَا لَهُۥٓ أَهْلَهُۥ وَمِثْلَهُم مَّعَهُمْ رَحْمَةً مِّنَّا وَذِكْرَىٰ لِأُو۟لِى ٱلْأَلْبَـٰبِ ﴿٤٣﴾
بىز ئۇنىڭ (كېسەل ۋاقتىدا ھادىسىگە ئۇچراپ يوق بولغان) ئەھلى - ئەۋلادىنى (يېڭىدىن) بەردۇق، ئۇلار بىلەن بىللە بىر باراۋەر ئارتۇق بەردۇق، (بۇ) بىزنىڭ نازىل قىلغان رەھمىتىمىزدۇر، ئەقىل ئىگىلىرى ئۈچۈن ۋەز - نەسىھەتتۇر
وَخُذْ بِيَدِكَ ضِغْثًا فَٱضْرِب بِّهِۦ وَلَا تَحْنَثْ ۗ إِنَّا وَجَدْنَـٰهُ صَابِرًا ۚ نِّعْمَ ٱلْعَبْدُ ۖ إِنَّهُۥٓ أَوَّابٌ ﴿٤٤﴾
(ئۇنىڭغا) «قولۇڭ بىلەن بىر باغلام چۆپنى ئېلىپ، ئۇنىڭ بىلەن (ئايالىڭنى) ئۇرغىن. قەسىمىڭنى بۇزمىغىن» (دېدۇق). بىز ئەييۇبنى ئىشەنچلىك ھالدا سەۋرچان بايقىدۇق، ئۇ نېمىدېگەن ئوبدان بەندە! ئىشەنچلىك ھالدا ئۇ (اﷲ قا) تەۋبە بىلەن يۈزلەنگۈچىدۇر
وَٱذْكُرْ عِبَـٰدَنَآ إِبْرَٰهِيمَ وَإِسْحَـٰقَ وَيَعْقُوبَ أُو۟لِى ٱلْأَيْدِى وَٱلْأَبْصَـٰرِ ﴿٤٥﴾
ئىبادەتتە كۈچلۈك، (دىندا) بەسىرەتلىك بەندىلىرىمىز ئىبراھىم، ئىسھاق ۋە يەئقۇبنى ئەسلىگىن
إِنَّآ أَخْلَصْنَـٰهُم بِخَالِصَةٍ ذِكْرَى ٱلدَّارِ ﴿٤٦﴾
ئۇلارنى بىز پاك - خىسلەتلىك، ئاخىرەتنى ئەسلەپ تۇرىدىغان سەمىمىي كىشلەر قىلدۇق
وَإِنَّهُمْ عِندَنَا لَمِنَ ٱلْمُصْطَفَيْنَ ٱلْأَخْيَارِ ﴿٤٧﴾
ئۇلار بىزنىڭ دەرگاھىمىزدا ئىشەنچلىك ھالدا ئالاھىدە تاللانغان كىشىلەردۇر، ئىشەنچلىك ھالدا ياخشى كىشىلەردۇر
وَٱذْكُرْ إِسْمَـٰعِيلَ وَٱلْيَسَعَ وَذَا ٱلْكِفْلِ ۖ وَكُلٌّ مِّنَ ٱلْأَخْيَارِ ﴿٤٨﴾
ئىسمائىلنى، ئىليەسەنى ۋە زۇلكىفلنى ئەسلىگىن، (ئۇلارنىڭ) ھەممىسى ياخشىلاردىندۇر
هَـٰذَا ذِكْرٌ ۚ وَإِنَّ لِلْمُتَّقِينَ لَحُسْنَ مَـَٔابٍ ﴿٤٩﴾
بۇ ۋەز - نەسىھەتتۇر، تەقۋادارلارغا ئەلۋەتتە ياخشى ئاقىۋەت بار
جَنَّـٰتِ عَدْنٍ مُّفَتَّحَةً لَّهُمُ ٱلْأَبْوَٰبُ ﴿٥٠﴾
(ئۇ) تۇرالغۇ بولغان مەڭگۈلۈك باغلارلەردىن ئىبارەت بولۇپ، ئۇلارغا ئۇنىڭ دەرۋازىلىرى ئوچۇقتۇر
مُتَّكِـِٔينَ فِيهَا يَدْعُونَ فِيهَا بِفَـٰكِهَةٍ كَثِيرَةٍ وَشَرَابٍ ﴿٥١﴾
ئۇلار مەڭگۈلۈك باغلارلەردە تەختلەرگە يۆلىنىپ ئولتۇرىدۇ، ئۇلار مەڭگۈلۈك باغلارلەردە نۇرغۇن مېۋىلەرنى، نۇرغۇن شارابلارنى تەلەپ قىلىپ (ئەكەلدۈرۈپ) تۇرىدۇ
۞ وَعِندَهُمْ قَـٰصِرَٰتُ ٱلطَّرْفِ أَتْرَابٌ ﴿٥٢﴾
ئۇلارنىڭ يانلىرىدا ئەرلىرىدىن غەيرىيگە كۆز سالمايدىغان، ياش قۇرامى ئوخشاش جۈپتىلىرى بولىدۇ
هَـٰذَا مَا تُوعَدُونَ لِيَوْمِ ٱلْحِسَابِ ﴿٥٣﴾
بۇ سىلەرگە قىيامەت كۈندە ۋەدە قىلىنغان نېمەتتۇر
إِنَّ هَـٰذَا لَرِزْقُنَا مَا لَهُۥ مِن نَّفَادٍ ﴿٥٤﴾
بۇ ئىشەنچلىك ھالدا بىزنىڭ ئاتايىمىزدۇركى، ئۇ ھەرگىز تۈگىمەيدۇ
هَـٰذَا ۚ وَإِنَّ لِلطَّـٰغِينَ لَشَرَّ مَـَٔابٍ ﴿٥٥﴾
بۇ (ھەقىقەتتۇر)، ھەقىقەتنى رەت قىلغۇچىلارنىڭ (ئاخىرەتتە بولىدىغان جايى) ئىشەنچلىك ھالدا ئەڭ يامان جايدۇر
جَهَنَّمَ يَصْلَوْنَهَا فَبِئْسَ ٱلْمِهَادُ ﴿٥٦﴾
ئۇ مەڭگۈلۈك ئوتدۇر، ئۇلار مەڭگۈلۈك ئوتگە كىرىدۇ، ئۇ نېمىدېگەن يامان جاي!
هَـٰذَا فَلْيَذُوقُوهُ حَمِيمٌ وَغَسَّاقٌ ﴿٥٧﴾
بۇ (كۆيدۈرگۈچى) قىزىق سۇدۇر، قان - يىرىڭدۇر، ئۇلار بۇنى تېتىسۇن
وَءَاخَرُ مِن شَكْلِهِۦٓ أَزْوَٰجٌ ﴿٥٨﴾
يەنە (ئۇلارغا) مۇشۇ خىلدىكى باشقا تۈرلۈك ئازابلار بار
هَـٰذَا فَوْجٌ مُّقْتَحِمٌ مَّعَكُمْ ۖ لَا مَرْحَبًۢا بِهِمْ ۚ إِنَّهُمْ صَالُوا۟ ٱلنَّارِ ﴿٥٩﴾
(ئەگەشكۈچىلىرى بىلەن دوزاخقا كىرگەن چاغدا، ئەگەشتۈرگۈچىلەرگە ئېيتىلىدۇ) «بۇ جامائە سىلەر بىلەن بىرگە كىرگۈچىدۇر». (ئەگەشتۈرگۈچىلەر ئېيتىدۇ) «ئۇلار قارشى ئېلىنمايدۇ، ئۇلار چوقۇم (بىزگە ئوخشاش) ئازابنى تېتىغۇچىدۇر»
قَالُوا۟ بَلْ أَنتُمْ لَا مَرْحَبًۢا بِكُمْ ۖ أَنتُمْ قَدَّمْتُمُوهُ لَنَا ۖ فَبِئْسَ ٱلْقَرَارُ ﴿٦٠﴾
ئەگەشكۈچىلەر ئېيتىدۇ: «بەلكى سىلەر قارشى ئېلىنمايسىلەر، سىلەر ئازابنى بىزگە ئېلىپ كەلدىڭلار (يەنى بىزنى ئازدۇرۇپ دوزاخقا كىرىشىمىزگە سەۋەبچى بولدۇڭلار)، بۇ نېمىدېگەن يامان جاي!»
قَالُوا۟ رَبَّنَا مَن قَدَّمَ لَنَا هَـٰذَا فَزِدْهُ عَذَابًا ضِعْفًا فِى ٱلنَّارِ ﴿٦١﴾
ئۇلار (يەنى ئەگەشكۈچىلەر): «پەرۋەردىگارىمىز! بىزگە بۇ ئازابنى ئېلىپ كەلگەن ئادەمگە دوزاخ ئازابىنى ئىككى ھەسسە زىيادە قىلغىن» دەيدۇ
وَقَالُوا۟ مَا لَنَا لَا نَرَىٰ رِجَالًا كُنَّا نَعُدُّهُم مِّنَ ٱلْأَشْرَارِ ﴿٦٢﴾
ئۇلار (يەنى ھەقىقەتنى رەت قىلغۇچىلارنىڭ كاتتىباشلىرى) ئېيتىدۇ: «بىز (دۇنيادىكى چاغدا) يامانلاردىن ھېسابلايدىغان ئادەملەرنى (يەنى مۆمىنلەرنى) (دوزاختا) كۆرمەيمىزغۇ؟
أَتَّخَذْنَـٰهُمْ سِخْرِيًّا أَمْ زَاغَتْ عَنْهُمُ ٱلْأَبْصَـٰرُ ﴿٦٣﴾
ئۇلارنى بىز (دۇنيادىكى چاغدا) مەسخىرە قىلاتتۇقمۇ؟ ياكى ئۇلاردىن كۆزلەر ئېغىپ كەتتىمۇ؟ (يەنى ياكى ئۇلار دوزاختا بىز بىلەن بىللە تۇرسىمۇ كۆرمەيۋاتامدۇق؟)»
إِنَّ ذَٰلِكَ لَحَقٌّ تَخَاصُمُ أَهْلِ ٱلنَّارِ ﴿٦٤﴾
ئەھلى دوزاخنىڭ بۇ رەۋىشتە ئۆزئارا جاڭجاللىشىشى چوقۇمدۇر
قُلْ إِنَّمَآ أَنَا۠ مُنذِرٌ ۖ وَمَا مِنْ إِلَـٰهٍ إِلَّا ٱللَّهُ ٱلْوَٰحِدُ ٱلْقَهَّارُ ﴿٦٥﴾
(ئى مۇھەممەد! مەككە ھەقىقەتنى رەت قىلغۇچىلىرىغا) ئېيتقىنكى، «مەن پەقەت بىر ئاگاھلاندۇرغۇچىمەن، غالىب بىر اﷲ تىن باشقا ھېچ ئىلاھ يوقتۇر
رَبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ٱلْعَزِيزُ ٱلْغَفَّـٰرُ ﴿٦٦﴾
(ئۇ اﷲ) ئاسمانلارنىڭ، زېمىننىڭ ۋە ئۇلارنىڭ ئارىسىدىكى شەيئىلەرنىڭ پەرۋەردىگارىدۇر، غالىبتۇر. (ئۆزى خالىغان بەندىلەرنى) ناھايىتى مەغپىرەت قىلغۇچىدۇر»
قُلْ هُوَ نَبَؤٌا۟ عَظِيمٌ ﴿٦٧﴾
(ئۇلارغا) ئېيتقىنكى، «ئۇ (يەنى اﷲ نىڭ بىرلىكىگە دەۋەت قىلغۇچى خۇدانىڭ ئەلچىسى ئىكەنلىكىم) چوڭ خەۋەردۇر
أَنتُمْ عَنْهُ مُعْرِضُونَ ﴿٦٨﴾
سىلەر ئۇنىڭدىن يۈز ئۆرۈيسىلەر
مَا كَانَ لِىَ مِنْ عِلْمٍۭ بِٱلْمَلَإِ ٱلْأَعْلَىٰٓ إِذْ يَخْتَصِمُونَ ﴿٦٩﴾
خۇدانىڭ ئەلچى پەرىشتىلىرى (ئادەم ئەلەيھىسسالام توغرىسىدا) مۇنازىرىلىشىۋاتقان چاغدا، مەن ئۇلارنىڭ ئەھۋالىنى بىلمەيتتىم
إِن يُوحَىٰٓ إِلَىَّ إِلَّآ أَنَّمَآ أَنَا۠ نَذِيرٌ مُّبِينٌ ﴿٧٠﴾
ماڭا پەقەت ۋەھيى قىلىنىدۇكى، مەن روشەن بىر ئاگاھلاندۇرغۇچىمەن»
إِذْ قَالَ رَبُّكَ لِلْمَلَـٰٓئِكَةِ إِنِّى خَـٰلِقٌۢ بَشَرًا مِّن طِينٍ ﴿٧١﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭ خۇدانىڭ ئەلچى پەرىشتىلىرىگە ئېيتتى: «مەن ئىشەنچلىك ھالدا لايدىن بىر ئادەم يارىتىمەن
فَإِذَا سَوَّيْتُهُۥ وَنَفَخْتُ فِيهِ مِن رُّوحِى فَقَعُوا۟ لَهُۥ سَـٰجِدِينَ ﴿٧٢﴾
ئۇنى مەن راۋۇرۇس ياراتقان (يەنى ئۇنى يارىتىپ سۈرەتكە كىرگۈزۈپ، ئەزالىرىنى تولۇق، مۇكەممەل ئىنسان ھالىتىگە كەلتۈرگەن) ۋە ئۇنىڭغا جان كىرگۈزگەن ۋاقتىمدا ئۇنىڭغا سەجدە قىلىڭلار»
فَسَجَدَ ٱلْمَلَـٰٓئِكَةُ كُلُّهُمْ أَجْمَعُونَ ﴿٧٣﴾
خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ بىرىمۇ قالماي، ھەممىسى (ئادەم ئەلەيھىسسالامغا) سەجدە قىلدى
إِلَّآ إِبْلِيسَ ٱسْتَكْبَرَ وَكَانَ مِنَ ٱلْكَـٰفِرِينَ ﴿٧٤﴾
پەقەت ئىبلىسلا بويۇنتاۋلىق قىلدى، ئۇ ھەقىقەتنى رەت قىلغۇچىلاردىن بولۇپ كەتتى
قَالَ يَـٰٓإِبْلِيسُ مَا مَنَعَكَ أَن تَسْجُدَ لِمَا خَلَقْتُ بِيَدَىَّ ۖ أَسْتَكْبَرْتَ أَمْ كُنتَ مِنَ ٱلْعَالِينَ ﴿٧٥﴾
اﷲ ئېيتتى: «ئى ئىبلىس! مەن ئۆز قولۇم بىلەن ياراتقان ئىنسانغا سەجدە قىلىشتىن ساڭا نېمە توسقۇنلۇق قىلدى (ئۇنىڭغا سەجدە قىلماسلىقىڭ تەكەببۇرلۇق قىلغانلىقىڭدىنمۇ؟ ياكى يۇقىرى مەرتىۋىلىكلەردىن بولغانلىقىڭدىنمۇ؟»
قَالَ أَنَا۠ خَيْرٌ مِّنْهُ ۖ خَلَقْتَنِى مِن نَّارٍ وَخَلَقْتَهُۥ مِن طِينٍ ﴿٧٦﴾
ئىبلىس: «مەن ئۇنىڭدىن ئارتۇق، مېنى ئوتتىن ياراتتىڭ، ئۇنى لايدىن ياراتتىڭ» دېدى
قَالَ فَٱخْرُجْ مِنْهَا فَإِنَّكَ رَجِيمٌ ﴿٧٧﴾
اﷲ: «سەن مەڭگۈلۈك باغلارتىن يوقال! سەن ئىشەنچلىك ھالدا قوغلاندى بولدۇڭ، ساڭا ئىشەنچلىك ھالدا قىيامەتكىچە مېنىڭ لەنىتىم بولسۇن» دېدى
وَإِنَّ عَلَيْكَ لَعْنَتِىٓ إِلَىٰ يَوْمِ ٱلدِّينِ ﴿٧٨﴾
اﷲ: «سەن مەڭگۈلۈك باغلارتىن يوقال! سەن ئىشەنچلىك ھالدا قوغلاندى بولدۇڭ، ساڭا ئىشەنچلىك ھالدا قىيامەتكىچە مېنىڭ لەنىتىم بولسۇن» دېدى
قَالَ رَبِّ فَأَنظِرْنِىٓ إِلَىٰ يَوْمِ يُبْعَثُونَ ﴿٧٩﴾
ئىبلىس: «پەرۋەردىگارىم! ماڭا (خالايىق) تىرىلدۈرۈلىدىغان كۈن (يەنى قىيامەت) گىچە مۆھلەت بەرسەڭ (يەنى مېنى ھايات قالدۇرساڭ)» دېدى
قَالَ فَإِنَّكَ مِنَ ٱلْمُنظَرِينَ ﴿٨٠﴾
اﷲ: «جەزمەن ساڭا مەلۇم ۋاقىتقىچە مۆھلەت بېرىلىدۇ» دېدى
إِلَىٰ يَوْمِ ٱلْوَقْتِ ٱلْمَعْلُومِ ﴿٨١﴾
اﷲ: «جەزمەن ساڭا مەلۇم ۋاقىتقىچە مۆھلەت بېرىلىدۇ» دېدى
قَالَ فَبِعِزَّتِكَ لَأُغْوِيَنَّهُمْ أَجْمَعِينَ ﴿٨٢﴾
ئىبلىس ئېيتتى: «ئىززىتىڭ بىلەن قەسەمكى، ئۇلارنىڭ ھەممىسىنى چوقۇم ئازدۇرىمەن
إِلَّا عِبَادَكَ مِنْهُمُ ٱلْمُخْلَصِينَ ﴿٨٣﴾
ئۇلاردىن پەقەت تاللانغان بەندىلىرىڭلا بۇنىڭدىن مۇستەسنا (يەنى ئۇلارنى ئازدۇرالمايمەن)»
قَالَ فَٱلْحَقُّ وَٱلْحَقَّ أَقُولُ ﴿٨٤﴾
اﷲ ئېيتتى: «مېنىڭ سۆزۈم ھەقتۇر، مەن ھەقنى سۆزلەيمەن
لَأَمْلَأَنَّ جَهَنَّمَ مِنكَ وَمِمَّن تَبِعَكَ مِنْهُمْ أَجْمَعِينَ ﴿٨٥﴾
سەن بىلەن ۋە ئۇلاردىن ساڭا ئەگەشكەنلەرنىڭ ھەممىسى بىلەن چوقۇم مەڭگۈلۈك ئوتنى توشقۇزىمەن»
قُلْ مَآ أَسْـَٔلُكُمْ عَلَيْهِ مِنْ أَجْرٍ وَمَآ أَنَا۠ مِنَ ٱلْمُتَكَلِّفِينَ ﴿٨٦﴾
ئېيتقىنكى، «(اﷲ تەرىپىدىن كەلگەن ۋەھيىنى) تەبلىغ قىلغانلىقىمغا سىلەردىن ھېچ ھەق تەلەپ قىلمايمەن، مەن (قۇرئاننى) ئىپتىرا قىلغۇچىلاردىن (يەنى يالغاندىن توقۇغۇچىلاردىن) ئەمەسمەن
إِنْ هُوَ إِلَّا ذِكْرٌ لِّلْعَـٰلَمِينَ ﴿٨٧﴾
قۇرئان پەقەت جاھان ئەھلى ئۈچۈن ۋەز - نەسىھەتتۇر
وَلَتَعْلَمُنَّ نَبَأَهُۥ بَعْدَ حِينٍۭ ﴿٨٨﴾
بىر مۇددەتتىن كېيىن ئۇنىڭ راستلىقىنى چوقۇم بىلىسىلەر»
