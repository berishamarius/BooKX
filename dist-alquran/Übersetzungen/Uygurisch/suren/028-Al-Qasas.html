بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
طسٓمٓ ﴿١﴾
طاسن، مىم
تِلْكَ ءَايَـٰتُ ٱلْكِتَـٰبِ ٱلْمُبِينِ ﴿٢﴾
بۇ، روشەن مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) ئايەتلىرىدۇر
نَتْلُوا۟ عَلَيْكَ مِن نَّبَإِ مُوسَىٰ وَفِرْعَوْنَ بِٱلْحَقِّ لِقَوْمٍ يُؤْمِنُونَ ﴿٣﴾
ئىمان ئېيتقان قەۋم ئۈچۈن ساڭا مۇسا ۋە پىرئەۋن قىسسىسىنى ھەقلىق ئاساستا ئوقۇپ بېرىمىز
إِنَّ فِرْعَوْنَ عَلَا فِى ٱلْأَرْضِ وَجَعَلَ أَهْلَهَا شِيَعًا يَسْتَضْعِفُ طَآئِفَةً مِّنْهُمْ يُذَبِّحُ أَبْنَآءَهُمْ وَيَسْتَحْىِۦ نِسَآءَهُمْ ۚ إِنَّهُۥ كَانَ مِنَ ٱلْمُفْسِدِينَ ﴿٤﴾
شۈبھىسىزكى، پىرئەۋن (مىسىر) زېمىنىدا (زومىگەرلىكتە) ھەددىدىن ئاشتى، ئاھالىسىنى بۆلەكلەرگە بۆلۈپ، ئۇلاردىن بىر تائىپە (يەنى بەنى ئىسرائىل) نى بوزەك قىلدى. ئۇلاردىن ئوغۇللىرىنى ئۆلتۈرۈپ، قىزلىرىنى) خىزمەتكە سېلىش ئۈچۈن) تىرىك قالدۇردى، پىرئەۋن ئىشەنچلىك ھالدا بۇزغۇنچىلاردىن ئىدى
وَنُرِيدُ أَن نَّمُنَّ عَلَى ٱلَّذِينَ ٱسْتُضْعِفُوا۟ فِى ٱلْأَرْضِ وَنَجْعَلَهُمْ أَئِمَّةً وَنَجْعَلَهُمُ ٱلْوَٰرِثِينَ ﴿٥﴾
بىز (مىسىر) زېمىنىدا بوزەك قىلىنغانلارغا مەرھەمەت قىلىشنى ئىرادە قىلىمىز، ئۇلارنى يولباشچىلاردىن قىلىشنى، ئۇلارنى پىرئەۋن ۋە ئۇنىڭ قەۋمىنىڭ (مۈلكىگە) ۋارىس قىلىشنى ئىرادە قىلىمىز
وَنُمَكِّنَ لَهُمْ فِى ٱلْأَرْضِ وَنُرِىَ فِرْعَوْنَ وَهَـٰمَـٰنَ وَجُنُودَهُمَا مِنْهُم مَّا كَانُوا۟ يَحْذَرُونَ ﴿٦﴾
ئۇلارنى مىسىر زېمىنىد اكۈچ - قۇۋۋەتكە ئىگە قىلماقچىمىز، بىز پىرئەۋنگە، (ۋەزىرى) ھامانغا ۋە ئۇلارنىڭ قوشۇنىغا ئۇلار (يەنى بوزەك قىلىنغانلار) دىن قورقىدىغان نەرسىنى (يەنى بەنى ئىسرائىلنى مىسىرغا ھۆكۈمران قىلىشنى) كۆرسىتىمىز
وَأَوْحَيْنَآ إِلَىٰٓ أُمِّ مُوسَىٰٓ أَنْ أَرْضِعِيهِ ۖ فَإِذَا خِفْتِ عَلَيْهِ فَأَلْقِيهِ فِى ٱلْيَمِّ وَلَا تَخَافِى وَلَا تَحْزَنِىٓ ۖ إِنَّا رَآدُّوهُ إِلَيْكِ وَجَاعِلُوهُ مِنَ ٱلْمُرْسَلِينَ ﴿٧﴾
مۇسانىڭ ئانىسىغا ئىلھام بىلەن بىلدۈردۇقكى: «مۇسانى ئېمىتكىن، ئۇنىڭ زىيانكەشلىككە ئۇچرىشىدىن قورقساڭ، ئۇنى (ساندۇققا سېلىپ) دەرياغا (يەنى نىل دەرياسىغا) تاشلىغىن، ئۇنى (ھالاك بولارمىكىن دەپ) قورقمىغىن، (ئۇنىڭ پىراقىدىن) قايغۇرمىغىن، ئۇنى ساڭا چوقۇم قايتۇرىمىز ۋە ئۇنى خۇدانىڭ ئەلچى-رەسۇللىرىدىن قىلىمىز»
فَٱلْتَقَطَهُۥٓ ءَالُ فِرْعَوْنَ لِيَكُونَ لَهُمْ عَدُوًّا وَحَزَنًا ۗ إِنَّ فِرْعَوْنَ وَهَـٰمَـٰنَ وَجُنُودَهُمَا كَانُوا۟ خَـٰطِـِٔينَ ﴿٨﴾
پىرئەۋن نىڭ ئائىلىسىدىكىلەر مۇسانى ئاقىۋەت ئۆزلىرىگە دۈشمەن، ۋە خاپىلىق (نىڭ مەنبەسى) قىلىش ئۈچۈن (نىل دەرياسىدىن) سۈزۈۋالدى. شۈبھىسىزكى، پىرئەۋن، ھامان ۋە ئۇلارنىڭ قوشۇنلىرى خاتالاشقان ئىدى
وَقَالَتِ ٱمْرَأَتُ فِرْعَوْنَ قُرَّتُ عَيْنٍ لِّى وَلَكَ ۖ لَا تَقْتُلُوهُ عَسَىٰٓ أَن يَنفَعَنَآ أَوْ نَتَّخِذَهُۥ وَلَدًا وَهُمْ لَا يَشْعُرُونَ ﴿٩﴾
پىرئەۋن نىڭ ئايالى: «(بۇ بالا) ماڭا ۋە ساڭا كۆزنۇرى، (يەنى خۇشاللىق) بولسۇن، ئۇنى ئۆلتۈرمەڭلار، بەلكى ئۇ بىزگە پايدا يەتكۈزەر ياكى ئۇنى ئوغۇل قىلىۋالارمىز» دېدى. ھالبۇكى، ئۇلار (پىرئەۋن نىڭ ۋە ئۇنىڭ ياردەمچىلىرىنىڭ ھالاكىتىنىڭ مۇسانىڭ قولىدا بولىدىغانلىقىنى) ئۇقمايتتى
وَأَصْبَحَ فُؤَادُ أُمِّ مُوسَىٰ فَـٰرِغًا ۖ إِن كَادَتْ لَتُبْدِى بِهِۦ لَوْلَآ أَن رَّبَطْنَا عَلَىٰ قَلْبِهَا لِتَكُونَ مِنَ ٱلْمُؤْمِنِينَ ﴿١٠﴾
مۇسانىڭ ئانىسى (مۇسانىڭ پىرئەۋتننىڭ قولىغا چۈشۈپ قالغانلىقىنى ئاڭلاپ) ئەس - ھوشىنى يوقاتتى، (ئۇنىڭ اﷲ نىڭ بالىنى قايتۇرۇش ۋەدىسىگە) ئىشەنگۈچىلەردىن بولۇشى ئۈچۈن، ئۇنىڭ كۆڭلىنى خاتىرجەم قىلمىغان بولساق، ئۇ بالىنى ئاشكارىلاپ قويغىلى تاس قالغان ئىدى
وَقَالَتْ لِأُخْتِهِۦ قُصِّيهِ ۖ فَبَصُرَتْ بِهِۦ عَن جُنُبٍ وَهُمْ لَا يَشْعُرُونَ ﴿١١﴾
ئانىسى مۇسانىڭ ھەمشىرىسىگە: «(خەۋەرنى ئۇقۇش ئۈچۈن) مۇساغا ئەگەشكىن» دېدى. ھەمشىرىسى مۇسانى يىراقتىن كۆرۈپ تۇردى. ھالبۇكى، ئۇلار (ئۇنى) تونۇمايتتى
۞ وَحَرَّمْنَا عَلَيْهِ ٱلْمَرَاضِعَ مِن قَبْلُ فَقَالَتْ هَلْ أَدُلُّكُمْ عَلَىٰٓ أَهْلِ بَيْتٍ يَكْفُلُونَهُۥ لَكُمْ وَهُمْ لَهُۥ نَـٰصِحُونَ ﴿١٢﴾
(ئۆز ئانىسى كېلىشتىن) ئىلگىرى مۇسانى سۈت ئەمگۈزگۈچى ئاياللارنىڭ ئېمىتىشىدىن توستۇق، مۇسانىڭ ھەمشىرىسى: «سىلەرگە مۇسانى باقىدىغان ئوبدان بىر ئائىلىنى كۆرسىتىپ قويايمۇ؟ ئۇلار مۇسانى ئىخلاس بىلەن باقىدۇ» دېدى
فَرَدَدْنَـٰهُ إِلَىٰٓ أُمِّهِۦ كَىْ تَقَرَّ عَيْنُهَا وَلَا تَحْزَنَ وَلِتَعْلَمَ أَنَّ وَعْدَ ٱللَّهِ حَقٌّ وَلَـٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿١٣﴾
ئانىسىنىڭ خۇشال بولۇشى، قايغۇرماسلىقى ۋە اﷲ نىڭ ۋەدىسىنىڭ ھەق ئىكەنلىكىنى بىلىشى ئۈچۈن بىز مۇسانى ئۇنىڭغا قايتۇردۇق، لېكىن ئىنسالارنىڭ تولىسى (اﷲ نىڭ ۋەدىسىنىڭ ھەق ئىكەنلىكىنى) بىلمەيدۇ
وَلَمَّا بَلَغَ أَشُدَّهُۥ وَٱسْتَوَىٰٓ ءَاتَيْنَـٰهُ حُكْمًا وَعِلْمًا ۚ وَكَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿١٤﴾
مۇسا ئۆسۈپ يېتىلگەندە، ئەقلى توشقاندا ئۇنىڭغا خۇدانىڭ ئەلچىسىلىكنى ۋە ئىلىمنى ئاتا قىلدۇق، بىز ياخشىلارغا مۇشۇنداق مۇكاپات بېرىمىز
وَدَخَلَ ٱلْمَدِينَةَ عَلَىٰ حِينِ غَفْلَةٍ مِّنْ أَهْلِهَا فَوَجَدَ فِيهَا رَجُلَيْنِ يَقْتَتِلَانِ هَـٰذَا مِن شِيعَتِهِۦ وَهَـٰذَا مِنْ عَدُوِّهِۦ ۖ فَٱسْتَغَـٰثَهُ ٱلَّذِى مِن شِيعَتِهِۦ عَلَى ٱلَّذِى مِنْ عَدُوِّهِۦ فَوَكَزَهُۥ مُوسَىٰ فَقَضَىٰ عَلَيْهِ ۖ قَالَ هَـٰذَا مِنْ عَمَلِ ٱلشَّيْطَـٰنِ ۖ إِنَّهُۥ عَدُوٌّ مُّضِلٌّ مُّبِينٌ ﴿١٥﴾
مۇسا شەھەر ئاھالىسىدىن ھېچ كىشىنى كۆرگىلى بولمايدىغان بىر ۋاقىتتا (يەنى چۈشلۈك ئۇيقىدىكى چاغدا) شەھەرگە كىردى، ئۇ شەھەردە ئىككى كىشىنىڭ ئۇرۇشۇۋاتقانلىقىنى كۆردى. (ئۇرۇشۇۋاتقانلاردىن) بىرى ئۆز قەۋمىدىن بولسا، يەنە بىرى ئۇنىڭ دۈشمىنى ئىدى. ئۆز قەۋمىدىن بولغان ئادەم دۈشمىنىگە قارشى ئۇنىڭدىن ياردەم تىلىدى. مۇسا ئۇنى مۇشت بىلەن بىرنى ئۇرۇپ ئۆلتۈرۈپ قويدى. مۇسا ئېيتتى: «بۇ شەيتان (ئازدۇرغۇچى)نىڭ ئىشىدۇر، شۈبھىسىزكى، شەيتان (ئازدۇرغۇچى) ئازدۇرغۇچى ئاشكارا دۈشمەندۇر»
قَالَ رَبِّ إِنِّى ظَلَمْتُ نَفْسِى فَٱغْفِرْ لِى فَغَفَرَ لَهُۥٓ ۚ إِنَّهُۥ هُوَ ٱلْغَفُورُ ٱلرَّحِيمُ ﴿١٦﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! مەن ئىشەنچلىك ھالدا ئۆزۈمگە زۇلۇم قىلدىم. ماڭا مەغپىرەت قىلغىن». اﷲ ئۇنىڭغا مەغپىرەت قىلدى، اﷲ ئىشەنچلىك ھالدا ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
قَالَ رَبِّ بِمَآ أَنْعَمْتَ عَلَىَّ فَلَنْ أَكُونَ ظَهِيرًا لِّلْمُجْرِمِينَ ﴿١٧﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! (ماڭا مەغپىرەت قىلغانلىق) نېمىتىڭ سەۋەبى بىلەن مەن ھەرگىز گۇناھكارلارغا ياردەمچى بولمايمەن»
فَأَصْبَحَ فِى ٱلْمَدِينَةِ خَآئِفًا يَتَرَقَّبُ فَإِذَا ٱلَّذِى ٱسْتَنصَرَهُۥ بِٱلْأَمْسِ يَسْتَصْرِخُهُۥ ۚ قَالَ لَهُۥ مُوسَىٰٓ إِنَّكَ لَغَوِىٌّ مُّبِينٌ ﴿١٨﴾
مۇسا ئەتىسى ئەتىگەندە قورقۇنچ ئىچىدەئەتراپىغا قاراپ شەھەر ئارىلاپ يۈرەتتى، ناگاھان تۈنۈگۈن ئۇنىڭدىن ياردەم تىلىگەن ئادەم (بىر قىبتى بىلەن ئۇرۇشۇپ تۇرغان ھالدا) مۇسادىن يەنە ياردەم تىلىدى، مۇسا ئۇنىڭغا: «سەن ئىشەنچلىك ھالدا ئاشكارا گۇمراھ ئىكەنسەن» دېدى
فَلَمَّآ أَنْ أَرَادَ أَن يَبْطِشَ بِٱلَّذِى هُوَ عَدُوٌّ لَّهُمَا قَالَ يَـٰمُوسَىٰٓ أَتُرِيدُ أَن تَقْتُلَنِى كَمَا قَتَلْتَ نَفْسًۢا بِٱلْأَمْسِ ۖ إِن تُرِيدُ إِلَّآ أَن تَكُونَ جَبَّارًا فِى ٱلْأَرْضِ وَمَا تُرِيدُ أَن تَكُونَ مِنَ ٱلْمُصْلِحِينَ ﴿١٩﴾
مۇسا ئىككىسىنىڭ (ئورتاق) دۈشمىنى بولغان ئادەم (يەنى قىبتى) گە قول ئۇزاتماقچى بولغاندا، ياردەم تىلىگەن ئادەم: «ئى مۇسا! سەن تۈنۈگۈن بىر ئادەمنى ئۆلتۈرگەندەك مېنى ئۆلتۈرمەكچىمۇسەن، سەن زېمىندا تۈزىگۈچىلەردىن بولۇشنى خالىماي، پەقەت زومىگەر بولۇشنىلا خالايسەن» دېدى
وَجَآءَ رَجُلٌ مِّنْ أَقْصَا ٱلْمَدِينَةِ يَسْعَىٰ قَالَ يَـٰمُوسَىٰٓ إِنَّ ٱلْمَلَأَ يَأْتَمِرُونَ بِكَ لِيَقْتُلُوكَ فَٱخْرُجْ إِنِّى لَكَ مِنَ ٱلنَّـٰصِحِينَ ﴿٢٠﴾
بىر ئادەم شەھەرنىڭ يىراق يېرىدىن ئالدىراپ كېلىپ: «ئى مۇسا! شۈبھىسىزكى، كاتتىلار سېنى ئۆلتۈرۈشكە مەسلىھەت قىلىشىۋاتىدۇ، (شەھەردىن دەرھال) چىقىپ كەتكىن، مەن ئەلۋەتتە ساڭا سادىق كىشىلەردىنمەن» دېدى
فَخَرَجَ مِنْهَا خَآئِفًا يَتَرَقَّبُ ۖ قَالَ رَبِّ نَجِّنِى مِنَ ٱلْقَوْمِ ٱلظَّـٰلِمِينَ ﴿٢١﴾
مۇسا قورقۇنچ ئىچىدە ئەتراپقا باققان ھالدا شەھەردىن چىقتى، ئۇ: «ئى پەرۋەردىگارىم! مېنى زالىم قەۋمدىن قۇتقۇزغىن» دېدى
وَلَمَّا تَوَجَّهَ تِلْقَآءَ مَدْيَنَ قَالَ عَسَىٰ رَبِّىٓ أَن يَهْدِيَنِى سَوَآءَ ٱلسَّبِيلِ ﴿٢٢﴾
ئۇ مەديەن تەرەپكە يۈزلەنگەن چاغدا: «پەرۋەردىگارىم مېنى توغرا يولغا يېتەكلىشى مۇمكىن» دېدى
وَلَمَّا وَرَدَ مَآءَ مَدْيَنَ وَجَدَ عَلَيْهِ أُمَّةً مِّنَ ٱلنَّاسِ يَسْقُونَ وَوَجَدَ مِن دُونِهِمُ ٱمْرَأَتَيْنِ تَذُودَانِ ۖ قَالَ مَا خَطْبُكُمَا ۖ قَالَتَا لَا نَسْقِى حَتَّىٰ يُصْدِرَ ٱلرِّعَآءُ ۖ وَأَبُونَا شَيْخٌ كَبِيرٌ ﴿٢٣﴾
مۇسا مەديەندىكى بىر بۇلاقنىڭ يېنىغا كەلگەندە (ماللىرىنى) سۇغىرىۋاتقان بىر توپ كىشىلەرنى كۆردى. ئۇلاردىن باشقا يەنە (قويلىرىنى سۇدىن) توسۇپ تۇرغان ئىككى ئايالنى كۆردى. مۇسا ئۇلارغا: «سىلەرگە نېمە بولدى؟ (يەنى قويلىرىڭلارنى سۇدىن توسۇپ تۇرۇپ تۇرۇپسىلەرغۇ؟)» دېدى، ئۇلار: «پادىچىلار قويلىرىنى سۇغۇرۇپ بولغاندىن كېيىن، ئاندىن بىز سۇغىرىمىز، ئاتىمىز بولسا ياشىنىپ قالغان بوۋايدۇر» دېدى
فَسَقَىٰ لَهُمَا ثُمَّ تَوَلَّىٰٓ إِلَى ٱلظِّلِّ فَقَالَ رَبِّ إِنِّى لِمَآ أَنزَلْتَ إِلَىَّ مِنْ خَيْرٍ فَقِيرٌ ﴿٢٤﴾
مۇسا ئۇلارنىڭ قويلىرىنى سۇغۇرۇپ بەردى، ئاندىن ئارقىسىغا بۇرۇلۇپ (بىر دەرەخنىڭ) سايىسىگە بېرىپ (ئولتۇرۇپ): «پەرۋەردىگارىم! ماڭا رىزىقتىن نېمىنىلا بەرسەڭ، مەن ئىشەنچلىك ھالدا ئۇنىڭغا موھتاج» دېدى
فَجَآءَتْهُ إِحْدَىٰهُمَا تَمْشِى عَلَى ٱسْتِحْيَآءٍ قَالَتْ إِنَّ أَبِى يَدْعُوكَ لِيَجْزِيَكَ أَجْرَ مَا سَقَيْتَ لَنَا ۚ فَلَمَّا جَآءَهُۥ وَقَصَّ عَلَيْهِ ٱلْقَصَصَ قَالَ لَا تَخَفْ ۖ نَجَوْتَ مِنَ ٱلْقَوْمِ ٱلظَّـٰلِمِينَ ﴿٢٥﴾
ئۇلارنىڭ (يەنى ئۇ ئىككى ئايالنىڭ) بىرى مۇسانىڭ يېنىغا ئۇياتچانلىق بىلەن مېڭىپ كېلىپ: «(قويلىرىمىزنى) سۇغىرىپ بەرگەنلىكىڭنىڭ ھەققىنى بېرىش ئۈچۈن ئاتام سېنى راستلا چاقىرىدۇ» دېدى. مۇسا ئۇنىڭ (يەنى شۇئەيبنىڭ) قېشىغا كېلىپ ئەھۋالنى ھېكايە قىلىپ بەردى، شۇئەيب: «قورقمىغىن، زالىم قەۋمدىن قۇتۇلدۇڭ» دېدى
قَالَتْ إِحْدَىٰهُمَا يَـٰٓأَبَتِ ٱسْتَـْٔجِرْهُ ۖ إِنَّ خَيْرَ مَنِ ٱسْتَـْٔجَرْتَ ٱلْقَوِىُّ ٱلْأَمِينُ ﴿٢٦﴾
ئۇلارنىڭ (يەنى ئاياللارنىڭ) بىرى ئېيتتى: «ئى ئاتا، ئۇنى سەن ئىشلەتكىن، بۇ سەن ئىشلەتكەنلەرنىڭ ئەڭ ياخشىسىدۇر، كۈچلۈك، ئىشەنچلىكتۇر»
قَالَ إِنِّىٓ أُرِيدُ أَنْ أُنكِحَكَ إِحْدَى ٱبْنَتَىَّ هَـٰتَيْنِ عَلَىٰٓ أَن تَأْجُرَنِى ثَمَـٰنِىَ حِجَجٍ ۖ فَإِنْ أَتْمَمْتَ عَشْرًا فَمِنْ عِندِكَ ۖ وَمَآ أُرِيدُ أَنْ أَشُقَّ عَلَيْكَ ۚ سَتَجِدُنِىٓ إِن شَآءَ ٱللَّهُ مِنَ ٱلصَّـٰلِحِينَ ﴿٢٧﴾
ئۇ (يەنى شۇئەيب) ئېيتتى: «ماڭا سەككىز يىل ئىشلەپ بېرىشىڭ بەدىلىگە ساڭا بۇ ئىككى قىزىمنىڭ بىرىنى بېرىشنى خالايمەن، ئەگەر ئون يىلنى توشقۇزۇۋەتسەڭ (ئۇمۇ) ئىختىيارىڭ، (ئون يىلنى شەرت قىلىپ) سېنى مۇشەققەتكە سېلىپ قويۇشنى خالىمايمەن، خۇدا خالىسا، مېنىڭ ياخشى ئادەم ئىكەنلىكىمنى بايقايسەن»
قَالَ ذَٰلِكَ بَيْنِى وَبَيْنَكَ ۖ أَيَّمَا ٱلْأَجَلَيْنِ قَضَيْتُ فَلَا عُدْوَٰنَ عَلَىَّ ۖ وَٱللَّهُ عَلَىٰ مَا نَقُولُ وَكِيلٌ ﴿٢٨﴾
مۇسا ئېيتتى: «بۇ ئىككىمىزنىڭ ئارىمىزدىكى توختامدۇر، ئىككى مۇددەت (يەنى سەككىز يىل بىلەن ئون يىل) نىڭ قايسىسىنى توشقۇزسام توشقۇزاي، ماڭا ئارتۇقچە تەلەپ قويۇشقا بولمايدۇ، اﷲ بىزنىڭ سۆزىمىزگە گۇۋاھتۇر»
۞ فَلَمَّا قَضَىٰ مُوسَى ٱلْأَجَلَ وَسَارَ بِأَهْلِهِۦٓ ءَانَسَ مِن جَانِبِ ٱلطُّورِ نَارًا قَالَ لِأَهْلِهِ ٱمْكُثُوٓا۟ إِنِّىٓ ءَانَسْتُ نَارًا لَّعَلِّىٓ ءَاتِيكُم مِّنْهَا بِخَبَرٍ أَوْ جَذْوَةٍ مِّنَ ٱلنَّارِ لَعَلَّكُمْ تَصْطَلُونَ ﴿٢٩﴾
مۇسا (ئۆز ئارا كېلىشكەن) مۇددەتنى (يەنى كېلىشكەن مۇددەتنىڭ تولۇقراقى بولغان ئون يىلنى) توشقۇزغاندىن كېيىن ئايالىنى ئېلىپ (مىسىرغا قاراپ) يولغا چىقتى، ئۇ تۇر تېغى تەرىپىدە ئوت كۆردى، ئۇ ئايالىغا: «شۈبھىسىزكى، مەن (يىراقتىن) ئوت كۆردۈم، ئۇ يەردىن مەن سىلەرگە بىر خەۋەر ئېلىپ كېلەي، يا سىلەرنىڭ ئىسسىنىشىڭلار ئۈچۈن ئوتتىن بىر پارچە چوغ ئېلىپ كېلەي» دېدى (مۇسا ئەلەيھىسسالام ئوتنىڭ يېنىغا كېلىپ، ئۇنىڭ ئوت ئەمەس نۇر ئىكەنلىكىنى بىلدى)
فَلَمَّآ أَتَىٰهَا نُودِىَ مِن شَـٰطِئِ ٱلْوَادِ ٱلْأَيْمَنِ فِى ٱلْبُقْعَةِ ٱلْمُبَـٰرَكَةِ مِنَ ٱلشَّجَرَةِ أَن يَـٰمُوسَىٰٓ إِنِّىٓ أَنَا ٱللَّهُ رَبُّ ٱلْعَـٰلَمِينَ ﴿٣٠﴾
مۇسا ئوتنىڭ يېنىغا كەلگەندە ۋادىنىڭ ئوڭ تەرىپىنىڭ گىرۋىكىگە توغرا كېلىدىغان مۇبارەك جايدىكى دەرەخ تەرەپتىن (مۇنداق) نىدا ئاڭلاندى: «ئى مۇسا، (ساڭا سۆز قىلىۋاتقان) مەن، ئىشەنچلىك ھالدا، ئالەملەرنىڭ پەرۋەردىگارى اﷲ دۇرمەن
وَأَنْ أَلْقِ عَصَاكَ ۖ فَلَمَّا رَءَاهَا تَهْتَزُّ كَأَنَّهَا جَآنٌّ وَلَّىٰ مُدْبِرًا وَلَمْ يُعَقِّبْ ۚ يَـٰمُوسَىٰٓ أَقْبِلْ وَلَا تَخَفْ ۖ إِنَّكَ مِنَ ٱلْـَٔامِنِينَ ﴿٣١﴾
سەن ھاساڭنى تاشلىغىن». مۇسا ھاسىسىنىڭ گويا ئەجدىھادەك تېز ھەرىكەتلىنىۋاتقانلىقىنى كۆرگەندە ئارقىسىغا بۇرۇلۇپ قاچتى، كەينىگە قارىمىدى، (نىدا قىلىندىكى) «ئى مۇسا! ئالدىڭغا ماڭغىن، قورقمىغىن، سەن ئىشەنچلىك ھالدا (قۇرقۇنچتىن) ئەمىن بولغۇچىلاردىنسەن
ٱسْلُكْ يَدَكَ فِى جَيْبِكَ تَخْرُجْ بَيْضَآءَ مِنْ غَيْرِ سُوٓءٍ وَٱضْمُمْ إِلَيْكَ جَنَاحَكَ مِنَ ٱلرَّهْبِ ۖ فَذَٰنِكَ بُرْهَـٰنَانِ مِن رَّبِّكَ إِلَىٰ فِرْعَوْنَ وَمَلَإِي۟هِۦٓ ۚ إِنَّهُمْ كَانُوا۟ قَوْمًا فَـٰسِقِينَ ﴿٣٢﴾
قولۇڭنى قوينۇڭغا سالغىن، ئۇ ھېچقانداق ئىللەتسىز (بىر پارچە ئايدەك پارلاپ) ئاپئاق بولۇپ چىقىدۇ، قورقۇنچنى دەپئى قىلىش ئۈچۈن، قولۇڭنى يىغىۋالغىن (يەنى قوينۇڭغا سېلىۋالغىن)، بۇ ئىككىسى پىرئەۋنگە (ۋە ئۇنىڭ قەۋمىنىڭ) چوڭلىرىغا پەرۋەردىگارىڭ تەرىپىدىن كەلگەن پاكىتتۇر، ئۇلار ئىشەنچلىك ھالدا (بىزنىڭ ئىتائىتىمىزدىن چىققۇچى) پاسىق قەۋم ئىدى»
قَالَ رَبِّ إِنِّى قَتَلْتُ مِنْهُمْ نَفْسًا فَأَخَافُ أَن يَقْتُلُونِ ﴿٣٣﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! مەن ئىشەنچلىك ھالدا ئۇلاردىن بىر ئادەمنى ئۆلتۈرۈپ قويدۇم، شۇنىڭ ئۈچۈن ئۇلارنىڭ مېنى ئۆلتۈرۈشىدىن قورقىمەن
وَأَخِى هَـٰرُونُ هُوَ أَفْصَحُ مِنِّى لِسَانًا فَأَرْسِلْهُ مَعِىَ رِدْءًا يُصَدِّقُنِىٓ ۖ إِنِّىٓ أَخَافُ أَن يُكَذِّبُونِ ﴿٣٤﴾
قېرىندىشىم ھارۇننىڭ تىلى مېنىڭدىن راۋاندۇر، ئۇنى مەن بىلەن (خۇدانىڭ ئەلچىسىلىكىمنى) تەستىقلايدىغان ياردەمچى قىلىپ ئەۋەتكىن، مەن ئىشەنچلىك ھالدا ئۇلارنىڭ مېنى ئىنكار قىلىشىدىن قورقىمەن»
قَالَ سَنَشُدُّ عَضُدَكَ بِأَخِيكَ وَنَجْعَلُ لَكُمَا سُلْطَـٰنًا فَلَا يَصِلُونَ إِلَيْكُمَا ۚ بِـَٔايَـٰتِنَآ أَنتُمَا وَمَنِ ٱتَّبَعَكُمَا ٱلْغَـٰلِبُونَ ﴿٣٥﴾
اﷲ ئېيتتى: «قېرىندىشىڭ ئارقىلىق سېنى كۈچەيتىمىز، ئىككىڭلارنى غالىب قىلىمىز، شۇنىڭ ئۈچۈن ئۇلار ئىككىڭلارغا زىيانكەشلىك قىلالمايدۇ، سىلەر ئىككىڭلارۋە ئىككىڭلارغا ئەگەشكەنلەر بىزنىڭ ئايەتلىرىمىز ئارقىلىق غەلىبە قىلىسىلەر»
فَلَمَّا جَآءَهُم مُّوسَىٰ بِـَٔايَـٰتِنَا بَيِّنَـٰتٍ قَالُوا۟ مَا هَـٰذَآ إِلَّا سِحْرٌ مُّفْتَرًى وَمَا سَمِعْنَا بِهَـٰذَا فِىٓ ءَابَآئِنَا ٱلْأَوَّلِينَ ﴿٣٦﴾
مۇسا ئۇلارغا بىزنىڭ روشەن مۆجىزىلىرىمىزنى ئېلىپ كەلگەندە، ئۇلار: «بۇ ئويدۇرۇپ چىقىرىلغان سېھىردۇر، بىزنىڭ ئاتا - بوۋىلىرىمىزنىڭ (زامانىدا) بۇنداق نەرسىلەرنىڭ بارلىقىنى ئاڭلىمىغان ئىدۇق» دېدى
وَقَالَ مُوسَىٰ رَبِّىٓ أَعْلَمُ بِمَن جَآءَ بِٱلْهُدَىٰ مِنْ عِندِهِۦ وَمَن تَكُونُ لَهُۥ عَـٰقِبَةُ ٱلدَّارِ ۖ إِنَّهُۥ لَا يُفْلِحُ ٱلظَّـٰلِمُونَ ﴿٣٧﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! ئۆزىنىڭ دەرگاھىدىن كىمنىڭ ھىدايەت ئېلىپ كەلگەنلىكىنى، ئاخىرەتنىڭ ياخشى ئاقىۋىتىگە كىمنىڭ ئىگە بولىدىغانلىقىنى ئوبدان بىلىدۇ، زالىملار (يەنى اﷲ قا يالغاننى چاپلىغۇچى فاجىرلار) چوقۇم مۇۋەپپەقىيەت قازىنالمايدۇ»
وَقَالَ فِرْعَوْنُ يَـٰٓأَيُّهَا ٱلْمَلَأُ مَا عَلِمْتُ لَكُم مِّنْ إِلَـٰهٍ غَيْرِى فَأَوْقِدْ لِى يَـٰهَـٰمَـٰنُ عَلَى ٱلطِّينِ فَٱجْعَل لِّى صَرْحًا لَّعَلِّىٓ أَطَّلِعُ إِلَىٰٓ إِلَـٰهِ مُوسَىٰ وَإِنِّى لَأَظُنُّهُۥ مِنَ ٱلْكَـٰذِبِينَ ﴿٣٨﴾
پىرئەۋن ئېيتتى: «ئى كاتتىلار! مەن ئۆزۈمدىن باشقا يەنە بىر ئىلاھنىڭ بارلىقىنى بىلمەيمەن. ئى ھامان، مېنىڭ ئۈچۈن پىششىق خىش پىشۇرۇپ ئېگىز بىر بىنا سالغىن، مەن مۇسانىڭ ئىلاھىنى كۆرۈشۈم مۇمكىن، مەن ئىشەنچلىك ھالدا ئۇنى (ئاسماندا بىر پەرۋەردىگار بار دېگەن دەۋاسىدا) يالغانچىلاردىن گۇمان قىلىمەن»
وَٱسْتَكْبَرَ هُوَ وَجُنُودُهُۥ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَظَنُّوٓا۟ أَنَّهُمْ إِلَيْنَا لَا يُرْجَعُونَ ﴿٣٩﴾
پىرئەۋن ۋە ئۇنىڭ قوشۇنى (مىسىر) زېمىنىدا زۇلۇم قىلىپ) ئىمان ئېيتىشقا) بويۇنتاۋلىق قىلدى. ئۇلار ئۆزلىرىنى بىزتەرەپكە قايتۇرۇلمايمىز) يەنى قىيامەت ۋە قايتا تىرىلىش يوق، ھېساب ئېلىنىشمۇ، جازاغا تارتىلىشمۇيوق) دەپ گۇمان قىلدى
فَأَخَذْنَـٰهُ وَجُنُودَهُۥ فَنَبَذْنَـٰهُمْ فِى ٱلْيَمِّ ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلظَّـٰلِمِينَ ﴿٤٠﴾
بىز پىرئەۋننى ۋە ئۇنىڭ قوشۇنلىرىنى جازالىدۇق، ئۇلارنى دېڭىزغا تاشلىۋەتتۇق (يەنى ئۇلارنى دېڭىزدا غەرق قىلىۋەتتۇق، ئۇلاردىن بىر ئادەممۇ ساق قالغىنى يوق)، (ئى مۇھەممەد!) زالىملارنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىغا (ئىبرەت كۆزى بىلەن) قارىغىن
وَجَعَلْنَـٰهُمْ أَئِمَّةً يَدْعُونَ إِلَى ٱلنَّارِ ۖ وَيَوْمَ ٱلْقِيَـٰمَةِ لَا يُنصَرُونَ ﴿٤١﴾
ئۇلارنى. بىز دوزاخقا ئۈندەيدىغان پېشۋالار قىلدۇق (يەنى ئۇلارنى بىز دۇنيادا گۇمراھلار ئەگىشىدىغان كۇفرىنىڭ باشلىقلىرى قىلدۇق)، قىيامەت كۈنى ئۇلار ياردەمگە ئېرىشەلمەيدۇ (يەنى قىيامەت كۈنى ئۇلاردىن ئازابنى دەپئى قىلىدىغان مەدەتكار بولمايدۇ)
وَأَتْبَعْنَـٰهُمْ فِى هَـٰذِهِ ٱلدُّنْيَا لَعْنَةً ۖ وَيَوْمَ ٱلْقِيَـٰمَةِ هُم مِّنَ ٱلْمَقْبُوحِينَ ﴿٤٢﴾
ئۇلارغا بۇ دۇنيادا لەنەتنى ئەگەشتۈردۇق (يەنى ئۇلارغا بۇ دۇنيادا اﷲ، خۇدانىڭ ئەلچى پەرىشتىلىرى ۋە مۆمىنلەر لەنەت ئوقۇيدۇ)، قىيامەت كۈنى ئۇلار اﷲ نىڭ رەھمىتىدىن يىراق قىلىنغۇچىلاردۇر
وَلَقَدْ ءَاتَيْنَا مُوسَى ٱلْكِتَـٰبَ مِنۢ بَعْدِ مَآ أَهْلَكْنَا ٱلْقُرُونَ ٱلْأُولَىٰ بَصَآئِرَ لِلنَّاسِ وَهُدًى وَرَحْمَةً لَّعَلَّهُمْ يَتَذَكَّرُونَ ﴿٤٣﴾
شۈبھىسىزكى، بىز (نۇھنىڭ قەۋمى ئاد، سەمۇد ۋە لۇتنىڭ قەۋمى قاتارلىق) ئىلگىرىكى ئۈممەتلەرنى ھالاك قىلغاندىن كېيىن، كىشىلەر (يەنى بەنى ئىسرائىل) نىڭ دىللىرىغا (ھەقىقەتنى كۆرىدىغان) نۇر بېرىپ، ئۇلارغا ھىدايەت ۋە رەھمەت قىلدۇق، ئۇلار ۋەز - نەسىھەت ئالسۇن دەپ، مۇساغا مۇقەددەس كىتاب-ۋەھىينى (يەنى تەۋراتنى) بەردۇق
وَمَا كُنتَ بِجَانِبِ ٱلْغَرْبِىِّ إِذْ قَضَيْنَآ إِلَىٰ مُوسَى ٱلْأَمْرَ وَمَا كُنتَ مِنَ ٱلشَّـٰهِدِينَ ﴿٤٤﴾
(ئى مۇھەممەد!) بىز غەربىي تاغنىڭ بىر تەرىپىدە مۇساغا ۋەھيى نازىل قىلغىنىمىزدا سەن يوق ئىدىڭ، (بۇنى) ئۆز كۆزۈڭ بىلەنمۇ كۆرمىگەن ئىدىڭ
وَلَـٰكِنَّآ أَنشَأْنَا قُرُونًا فَتَطَاوَلَ عَلَيْهِمُ ٱلْعُمُرُ ۚ وَمَا كُنتَ ثَاوِيًا فِىٓ أَهْلِ مَدْيَنَ تَتْلُوا۟ عَلَيْهِمْ ءَايَـٰتِنَا وَلَـٰكِنَّا كُنَّا مُرْسِلِينَ ﴿٤٥﴾
لېكىن بىز نۇرغۇن ئۈممەتلەرنى ياراتتۇق، ئۇلارنىڭ ئۆمرى ئۇزۇن بولدى، سەن مەديەن ئاھالىسىنىڭ ئىچىدە ئۇلارغا بىزنىڭ ئايەتلىرىمىزنى تىلاۋەت قىلىپ تۇرغۇچى بولغىنىڭ يوق، لېكىن بىز (سېنى ئۇلاردىن باشقا بىر قەۋمگە خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق
وَمَا كُنتَ بِجَانِبِ ٱلطُّورِ إِذْ نَادَيْنَا وَلَـٰكِن رَّحْمَةً مِّن رَّبِّكَ لِتُنذِرَ قَوْمًا مَّآ أَتَىٰهُم مِّن نَّذِيرٍ مِّن قَبْلِكَ لَعَلَّهُمْ يَتَذَكَّرُونَ ﴿٤٦﴾
بىز تۇر تېغىنىڭ بىر تەرىپىدە (مۇساغا) سۆز قىلغىنىمىزدا سەن يوق ئىدىڭ، لېكىن بۇ سەندىن ئىلگىرى ھېچبىر ئاگاھلاندۇرغۇچى كەلمىگەن قەۋمنى ئاگاھلاندۇرۇشۇڭ ئۈچۈن، ئۇلارنىڭ ۋەز - نەسىھەت ئېلىشى ئۈچۈن، پەرۋەردىگارىڭ تەرىپىدىن نازىل قىلىنغان رەھمەتتۇر
وَلَوْلَآ أَن تُصِيبَهُم مُّصِيبَةٌۢ بِمَا قَدَّمَتْ أَيْدِيهِمْ فَيَقُولُوا۟ رَبَّنَا لَوْلَآ أَرْسَلْتَ إِلَيْنَا رَسُولًا فَنَتَّبِعَ ءَايَـٰتِكَ وَنَكُونَ مِنَ ٱلْمُؤْمِنِينَ ﴿٤٧﴾
ئۇلارنىڭ قىلغان ئىشلىرى (يەنى كۇفرى ۋە گۇناھلىرى) تۈپەيلىدىن، ئۇلارغا بىرەر ئازاب كەلسە، «پەرۋەردىگارىمىز! بىزگە بىرەر خۇدانىڭ ئەلچىسى ئەۋەتكەن بولساڭ ئىدىڭ، سېنىڭ ئايەتلىرىڭگە ئەگىشەتتۇق ۋە مۆمىنلەردىن بولاتتۇق» دېمەسلىكلىرى ئۈچۈن (سېنى ئۇلارغا خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق)
فَلَمَّا جَآءَهُمُ ٱلْحَقُّ مِنْ عِندِنَا قَالُوا۟ لَوْلَآ أُوتِىَ مِثْلَ مَآ أُوتِىَ مُوسَىٰٓ ۚ أَوَلَمْ يَكْفُرُوا۟ بِمَآ أُوتِىَ مُوسَىٰ مِن قَبْلُ ۖ قَالُوا۟ سِحْرَانِ تَظَـٰهَرَا وَقَالُوٓا۟ إِنَّا بِكُلٍّ كَـٰفِرُونَ ﴿٤٨﴾
ئۇلارغا بىز ۋەھى قىلغان ھەقىقەت نازىل بولغان چاغدا، ئۇلار: «ئۇنىڭغا (يەنى مۇھەممەد ئەلەيھىسسالامغا) نېمىشقا مۇساغا بېرىلگەن مۆجىزە بېرىلمىدى» دېدى. ئۇلار ئىلگىرى مۇساغا بېرىلگەن مۆجىزىنى ئىنكار قىلمىدىمۇ؟ ئۇلار: «(تەۋرات بىلەن قۇرئان) بىر - بىرىنى تەستىق قىلىدىغان سېھىردۇر» دېدى. ئۇلار: «بىز ئىشەنچلىك ھالدا (ئىككى مۇقەددەس كىتاب-ۋەھىينىڭ) ھەربىرىنى ئىنكار قىلغۇچىمىز» دېدى
قُلْ فَأْتُوا۟ بِكِتَـٰبٍ مِّنْ عِندِ ٱللَّهِ هُوَ أَهْدَىٰ مِنْهُمَآ أَتَّبِعْهُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٤٩﴾
ئېيتقىنكى، «ئەگەر (ئۇ ئىككى مۇقەددەس كىتاب-ۋەھىي سېھىردۇر دېگەن سۆزۈڭلاردا) راستچىل بولىدىغان بولساڭلار، ئۇ ئىككى مۇقەددەس كىتاب-ۋەھىيقا قارىغاندا، اﷲ تەرىپىدىن نازىل بولغان تېخىمۇ توغرا بىر مۇقەددەس كىتاب-ۋەھىي كەلتۈرۈپ بېقىڭلار، مەن ئۇنىڭغا ئەگىشەي»
فَإِن لَّمْ يَسْتَجِيبُوا۟ لَكَ فَٱعْلَمْ أَنَّمَا يَتَّبِعُونَ أَهْوَآءَهُمْ ۚ وَمَنْ أَضَلُّ مِمَّنِ ٱتَّبَعَ هَوَىٰهُ بِغَيْرِ هُدًى مِّنَ ٱللَّهِ ۚ إِنَّ ٱللَّهَ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّـٰلِمِينَ ﴿٥٠﴾
ئەگەر ئۇلار ساڭا جاۋاپ بەرمىسە، بىلگىنكى، ئۇلار پەقەت ئۆزلىرىنىڭ نەپسى خاھىشىغىلا ئەگىشىدۇ، اﷲ ۋەھىي قىلغان توغرا يولنى قويۇپ نەپسى خاھىشىغا ئەگەشكەن كىشىدىنمۇ ئازغۇن ئادەم بارمۇ؟ اﷲ ئىشەنچلىك ھالدا زالىم قەۋمنى ھىدايەت قىلمايدۇ
۞ وَلَقَدْ وَصَّلْنَا لَهُمُ ٱلْقَوْلَ لَعَلَّهُمْ يَتَذَكَّرُونَ ﴿٥١﴾
ۋەز - نەسىھەت ئالسۇن دەپ، شۈبھىسىزكى، ئۇلارغا قۇرئاننى ئۈزۈلدۈرمەي چۈشۈردۇق
ٱلَّذِينَ ءَاتَيْنَـٰهُمُ ٱلْكِتَـٰبَ مِن قَبْلِهِۦ هُم بِهِۦ يُؤْمِنُونَ ﴿٥٢﴾
قۇرئاندىن ئىلگىرى بىز مۇقەددەس كىتاب-ۋەھىي (يەنى ئىنجىل، تەۋرات) نازىل قىلغانلار (يەنى ناسارا ۋە يەھۇدىيلارنىڭ مۆمىنلىرى) قۇرئانغا ئىشىنىدۇ
وَإِذَا يُتْلَىٰ عَلَيْهِمْ قَالُوٓا۟ ءَامَنَّا بِهِۦٓ إِنَّهُ ٱلْحَقُّ مِن رَّبِّنَآ إِنَّا كُنَّا مِن قَبْلِهِۦ مُسْلِمِينَ ﴿٥٣﴾
ئۇلارغا قۇرئان تىلاۋەت قىلىنغان چاغدا، ئۇلار: «بىز قۇرئانغا ئىشەندۇق. شۈبھىسىزكى، ئۇ پەرۋەردىگارىمىز تەرىپىدىن نازىل بولغان ھەقىقەتتۇر. بىز ئىشەنچلىك ھالدا بۇنىڭدىن بۇرۇن مۇسۇلمان ئىدۇق» دەيدۇ
أُو۟لَـٰٓئِكَ يُؤْتَوْنَ أَجْرَهُم مَّرَّتَيْنِ بِمَا صَبَرُوا۟ وَيَدْرَءُونَ بِٱلْحَسَنَةِ ٱلسَّيِّئَةَ وَمِمَّا رَزَقْنَـٰهُمْ يُنفِقُونَ ﴿٥٤﴾
ئۇلارنىڭ قىلغان سەۋر - تاقىتى ئۈچۈن ئۇلارغا قوش ساۋاب بېرىلىدۇ، ئۇلار ياخشىلىق ئارقىلىق يامانلىقنى دەپئى قىلىدۇ، بىز ئۇلارغا رىزىق قىلىپ بەرگەن نەرسىلەردىن ياخشىلىق يوللىرىغا سەرپ قىلىدۇ
وَإِذَا سَمِعُوا۟ ٱللَّغْوَ أَعْرَضُوا۟ عَنْهُ وَقَالُوا۟ لَنَآ أَعْمَـٰلُنَا وَلَكُمْ أَعْمَـٰلُكُمْ سَلَـٰمٌ عَلَيْكُمْ لَا نَبْتَغِى ٱلْجَـٰهِلِينَ ﴿٥٥﴾
ئۇلار بىھۇدە سۆزلەرنى ئاڭلىغاندا ئۇنىڭدىن يۈز ئۆرۈپ (يەنى قۇلاق سالماي): «بىزنىڭ ئەمەللىرىمىز ئۆزىمىز ئۈچۈن، سىلەرنىڭ ئەمەللىرىڭلارمۇ ئۆزۈڭلار ئۈچۈن، سىلەرگە ئامانلىق بولسۇن، بىز نادانلاردىن دوستلۇق تىلىمەيمىز» دەيدۇ
إِنَّكَ لَا تَهْدِى مَنْ أَحْبَبْتَ وَلَـٰكِنَّ ٱللَّهَ يَهْدِى مَن يَشَآءُ ۚ وَهُوَ أَعْلَمُ بِٱلْمُهْتَدِينَ ﴿٥٦﴾
شۈبھىسىزكى، سەن خالىغان ئادەمىڭنى ھىدايەت قىلالمايسەن، لېكىن اﷲ ئۆزى خالىغان ئادەمنى ھىدايەت قىلىدۇ، اﷲ ھىدايەت تاپقۇچىلارنى ئوبدان بىلىدۇ
وَقَالُوٓا۟ إِن نَّتَّبِعِ ٱلْهُدَىٰ مَعَكَ نُتَخَطَّفْ مِنْ أَرْضِنَآ ۚ أَوَلَمْ نُمَكِّن لَّهُمْ حَرَمًا ءَامِنًا يُجْبَىٰٓ إِلَيْهِ ثَمَرَٰتُ كُلِّ شَىْءٍ رِّزْقًا مِّن لَّدُنَّا وَلَـٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿٥٧﴾
ئۇلار: «ئەگەر بىز سەن بىلەن توغرا يولغا ئەگىشىدىغان بولساق، زېمىنىمىزدىن ھەيدەپ چىقىرىلىمىز» دېدى. ئۇلارنى بىز تېچ ھەرەمگە يەرلەشتۈرمىدۇقمۇ؟ ھەرەمگە تۈرلۈك مېۋىلەرنىڭ ھەممىسى كەلتۈرۈلىدۇ، (بۇ) بىزنىڭ دەرگاھىمىزدىن چۈشۈرۈلگەن رىزىقتۇر، لېكىن ئۇلارنىڭ تولىسى بىلمەيدۇ
وَكَمْ أَهْلَكْنَا مِن قَرْيَةٍۭ بَطِرَتْ مَعِيشَتَهَا ۖ فَتِلْكَ مَسَـٰكِنُهُمْ لَمْ تُسْكَن مِّنۢ بَعْدِهِمْ إِلَّا قَلِيلًا ۖ وَكُنَّا نَحْنُ ٱلْوَٰرِثِينَ ﴿٥٨﴾
بىز نۇرغۇن شەھەرنى ھالاك قىلدۇق، ئۇلارنىڭ ئاھالىسى كەيىپ - ساپالىق تۇرمۇش كۆچۈرەتتى، بۇ ئۇلارنىڭ تۇرالغۇ جايلىرىدۇر، ئۇلار ھالاك بولغاندىن كېيىن (ئۇ جايلاردا) ئازغىنا ۋاقىتتىن تاشقىرى ئادەم تۇرغان ئەمەس، ئۇلار (نىڭ مال - مۈلكى ۋە يۇرتىغا) بىز ۋارىسلىق قىلدۇق
وَمَا كَانَ رَبُّكَ مُهْلِكَ ٱلْقُرَىٰ حَتَّىٰ يَبْعَثَ فِىٓ أُمِّهَا رَسُولًا يَتْلُوا۟ عَلَيْهِمْ ءَايَـٰتِنَا ۚ وَمَا كُنَّا مُهْلِكِى ٱلْقُرَىٰٓ إِلَّا وَأَهْلُهَا ظَـٰلِمُونَ ﴿٥٩﴾
پەرۋەردىگارىڭ شەھەرلەرنىڭ مەركىزىگە، ئۇلارنىڭ ئاھالىسىگە بىزنىڭ ئايەتلىرىمىزنى ئوقۇپ بېرىدىغان بىر خۇدانىڭ ئەلچىسى ئەۋەتمىگىچە شەھەرلەرنى ھالاك قىلغۇچى بولمىدۇق، شەھەرلەرنىڭ ئاھالىسى زالىم بولمىغىچە شەھەرلەرنى ھالاك قىلغۇچىمۇ بولمىدۇق
وَمَآ أُوتِيتُم مِّن شَىْءٍ فَمَتَـٰعُ ٱلْحَيَوٰةِ ٱلدُّنْيَا وَزِينَتُهَا ۚ وَمَا عِندَ ٱللَّهِ خَيْرٌ وَأَبْقَىٰٓ ۚ أَفَلَا تَعْقِلُونَ ﴿٦٠﴾
سىلەرگە قانداق نەرسە بېرىلمىسۇن، (ئۇ) دۇنيا تىرىكچىلىكىدىكى پايدىلىنىدىغان نەرسىدۇر ۋە دۇنيا تىرىكچىلىكىنىڭ زىننىتىدۇر، اﷲ نىڭ دەرگاھىدىكى ساۋاب ئەڭ ياخشىدۇر، ئەڭ باقىيدۇر، چۈشەنمەمسىلەر؟
أَفَمَن وَعَدْنَـٰهُ وَعْدًا حَسَنًا فَهُوَ لَـٰقِيهِ كَمَن مَّتَّعْنَـٰهُ مَتَـٰعَ ٱلْحَيَوٰةِ ٱلدُّنْيَا ثُمَّ هُوَ يَوْمَ ٱلْقِيَـٰمَةِ مِنَ ٱلْمُحْضَرِينَ ﴿٦١﴾
بىز ياخشى ۋەدىنى قىلغان (يەنى مەڭگۈلۈك باغلارنى ۋەدە قىلغان) ئادەم ئۇنى تاپىدۇ، (ئۇ) دۇنيا تىرىكچىلىكىدىكى پايدىلىنىدىغان نەرسىلەر بىلەن بىز پايدىلاندۇرغان، ئاندىن قىيامەت كۈنى ئازابقا دۇچار بولغان ئادەم بىلەن ئوخشاشمۇ
وَيَوْمَ يُنَادِيهِمْ فَيَقُولُ أَيْنَ شُرَكَآءِىَ ٱلَّذِينَ كُنتُمْ تَزْعُمُونَ ﴿٦٢﴾
شۇ كۈنى اﷲ ئۇلارغا: «سىلەر گۇمان قىلغان مېنىڭ شېرىكلىرىم قەيەردە؟» دەپ نىدا قىلىدۇ
قَالَ ٱلَّذِينَ حَقَّ عَلَيْهِمُ ٱلْقَوْلُ رَبَّنَا هَـٰٓؤُلَآءِ ٱلَّذِينَ أَغْوَيْنَآ أَغْوَيْنَـٰهُمْ كَمَا غَوَيْنَا ۖ تَبَرَّأْنَآ إِلَيْكَ ۖ مَا كَانُوٓا۟ إِيَّانَا يَعْبُدُونَ ﴿٦٣﴾
ئۆزلىرىگە ئازاب تېگىشلىك بولغانلار (يەنى مۇشرىكلارنىڭ چوڭلىرى): «پەرۋەردىگارىمىز! بىز (سېنىڭ يولۇڭدىن) ئازدۇرغانلارنى (يەنى ئەگەشكۈچىلىرىمىزنى) ئۆزىمىز ئازغاندەك ئازدۇردۇق، (پەرۋەردىگارىمىز!) ئۇلاردىن يىراقلىشىپ ساڭا يېقىنلاشتۇق، ئۇلار بىزگە ئىبادەت قىلمايتتى (پەقەت نەپسى خاھىشلىرىغا ئىبادەت قىلاتتى)» دېيىشىدۇ
وَقِيلَ ٱدْعُوا۟ شُرَكَآءَكُمْ فَدَعَوْهُمْ فَلَمْ يَسْتَجِيبُوا۟ لَهُمْ وَرَأَوُا۟ ٱلْعَذَابَ ۚ لَوْ أَنَّهُمْ كَانُوا۟ يَهْتَدُونَ ﴿٦٤﴾
ئۇلارغا: «شېرىك كەلتۈرگەن مەبۇدلىرىڭلارنى چاقىرىڭلار» دېيىلگەن ھامان، ئۇلار شېرىك كەلتۈرگەن مەبۇدلىرىنى چاقىرىدۇ، ئۇلاردىن جاۋاب كەلمەيدۇ، ئۇلار ئازابنى كۆرگەن چاغدا، (ھەقىقەتنى رەت قىلغۇچى بولغانلىقلىرىغا نادامەت قىلىپ) (دۇنيادىكى ۋاقتىدا) ھىدايەت تاپقان بولۇشلىرىنى ئارزۇ قىلىدۇ
وَيَوْمَ يُنَادِيهِمْ فَيَقُولُ مَاذَآ أَجَبْتُمُ ٱلْمُرْسَلِينَ ﴿٦٥﴾
ئۇ كۈندە اﷲ ئۇلارغا نىدا قىلىپ ئېيتىدۇ: «خۇدانىڭ ئەلچى-رەسۇللىرىگەنېمە بىلەن جاۋاپ بەردىڭلار؟»
فَعَمِيَتْ عَلَيْهِمُ ٱلْأَنۢبَآءُ يَوْمَئِذٍ فَهُمْ لَا يَتَسَآءَلُونَ ﴿٦٦﴾
ئۇ كۈندە ئۇلارغا بارلىق خەۋەرلەر قاراڭغۇ بولىدۇ، ئۇلار ئۆز ئارا سوراشمايدۇ
فَأَمَّا مَن تَابَ وَءَامَنَ وَعَمِلَ صَـٰلِحًا فَعَسَىٰٓ أَن يَكُونَ مِنَ ٱلْمُفْلِحِينَ ﴿٦٧﴾
تەۋبە قىلغان، ئىمان ئېيتقان ۋە ياخشى ئەمەل قىلغان ئادەمگە كەلسەك، ئۇنىڭ مەقسەتكە ئېرىشكۈچىلەردىن بولۇشى ئۈمىدلىكتۇر
وَرَبُّكَ يَخْلُقُ مَا يَشَآءُ وَيَخْتَارُ ۗ مَا كَانَ لَهُمُ ٱلْخِيَرَةُ ۚ سُبْحَـٰنَ ٱللَّهِ وَتَعَـٰلَىٰ عَمَّا يُشْرِكُونَ ﴿٦٨﴾
اﷲ (مەخلۇقاتىدىن) خالىغىنىنى يارىتىدۇ، خالىغان ئادەمنى (خۇدانىڭ ئەلچىسىلىككە) تاللايدۇ، تاللاش ھوقوقى ئۇلاردا ئەمەس، اﷲ پاكتۇر، ئۇلارنىڭ شېرىك كەلتۈرگەنلىرىدىن ئۈستۈندۇر
وَرَبُّكَ يَعْلَمُ مَا تُكِنُّ صُدُورُهُمْ وَمَا يُعْلِنُونَ ﴿٦٩﴾
پەرۋەردىگارىڭ ئۇلارنىڭ دىللىرىدا يوشۇرۇن تۇتقانلىرىنى ۋە (ئېغىزلىرىدا) دېگەنلىرىنى بىلىپ تۇرىدۇ
وَهُوَ ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ۖ لَهُ ٱلْحَمْدُ فِى ٱلْأُولَىٰ وَٱلْـَٔاخِرَةِ ۖ وَلَهُ ٱلْحُكْمُ وَإِلَيْهِ تُرْجَعُونَ ﴿٧٠﴾
ئۇ اﷲ دۇر، ئۇنىڭدىن بۆلەك ھېچ مەبۇد (بەرھەق) يوقتۇر، دۇنيا ۋە ئاخىرەتتە جىمى ھەمدۇسانا اﷲ قا مەنسۇپتۇر، ھۆكۈم چىقىرىش اﷲ قا مەنسۇپتۇر، سىلەر اﷲ نىڭ دەرگاھىغا قايتۇرۇلىسىلەر
قُلْ أَرَءَيْتُمْ إِن جَعَلَ ٱللَّهُ عَلَيْكُمُ ٱلَّيْلَ سَرْمَدًا إِلَىٰ يَوْمِ ٱلْقِيَـٰمَةِ مَنْ إِلَـٰهٌ غَيْرُ ٱللَّهِ يَأْتِيكُم بِضِيَآءٍ ۖ أَفَلَا تَسْمَعُونَ ﴿٧١﴾
«ئېيتىپ بېقىڭلارچۇ! ئەگەر اﷲ كېچىنى سىلەر ئۈچۈن قىيامەت كۈنىگىچە ئۇزۇن سوزىدىغان بولسا، اﷲ تىن بۆلەك قايسى ئىلاھ سىلەرگە يورۇقلۇق ئېلىپ كېلەلەيدۇ، سىلەر ئاڭلىمامسىلەر؟» دېگىن
قُلْ أَرَءَيْتُمْ إِن جَعَلَ ٱللَّهُ عَلَيْكُمُ ٱلنَّهَارَ سَرْمَدًا إِلَىٰ يَوْمِ ٱلْقِيَـٰمَةِ مَنْ إِلَـٰهٌ غَيْرُ ٱللَّهِ يَأْتِيكُم بِلَيْلٍ تَسْكُنُونَ فِيهِ ۖ أَفَلَا تُبْصِرُونَ ﴿٧٢﴾
«ئېيتىپ بېقىڭلارچۇ؟ ئەگەر اﷲ كۈندۈزنى سىلەر ئۈچۈن قىيامەت كۈنىگىچە سوزىدىغان بولسا، اﷲ تىن بۆلەك قايسى ئىلاھ سىلەرگە ئارام ئېلىشىڭلار ئۈچۈن كېچىنى ئېلىپ كېلەلەيدۇ؟ سىلەر كۆرمەمسىلەر؟» دېگىن
وَمِن رَّحْمَتِهِۦ جَعَلَ لَكُمُ ٱلَّيْلَ وَٱلنَّهَارَ لِتَسْكُنُوا۟ فِيهِ وَلِتَبْتَغُوا۟ مِن فَضْلِهِۦ وَلَعَلَّكُمْ تَشْكُرُونَ ﴿٧٣﴾
سىلەرنى كېچىدە ئارام ئالسۇن، كۈندۈزدە (ھاياتلىق يولىدا ھەرىكەتلىنىپ) اﷲ نىڭ پەزلىنى تەلەپ قىلسۇن ۋە (اﷲ نىڭ نېمەتلىرىگە) شۈكۈر قىلسۇن دەپ، سىلەر ئۈچۈن كېچە بىلەن كۈندۈزنى يارىتىشى اﷲ نىڭ رەھمىتىدىندۇر
وَيَوْمَ يُنَادِيهِمْ فَيَقُولُ أَيْنَ شُرَكَآءِىَ ٱلَّذِينَ كُنتُمْ تَزْعُمُونَ ﴿٧٤﴾
ئۇ كۈندە اﷲ ئۇلارغا نىدا قىلىپ: «سىلەر گۇمان قىلغان مېنىڭ شېرىكلىرىم قەيەردە؟» دەيدۇ
وَنَزَعْنَا مِن كُلِّ أُمَّةٍ شَهِيدًا فَقُلْنَا هَاتُوا۟ بُرْهَـٰنَكُمْ فَعَلِمُوٓا۟ أَنَّ ٱلْحَقَّ لِلَّهِ وَضَلَّ عَنْهُم مَّا كَانُوا۟ يَفْتَرُونَ ﴿٧٥﴾
بىز ھەر ئۈممەتتىن) ئۇلارنىڭ ئەمەللىرىگە گۇۋاھلىق بېرىدىغان) بىر گۇۋاھچى (يەنى خۇدانىڭ ئەلچىسى) نى چىقىرىمىز، پاكىتىڭلارنى كەلتۈرۈڭلار دەيمىز، ئۇلار ھەقىقەتنىڭ اﷲ قا مەنسۇپ ئىكەنلىكىنى ئاندىن بىلىدۇ، ئۇلارنىڭ ئويدۇرۇپ چىققان نەرسىلىرى (يەنى مەبۇدلىرى) ئۆزلىرىنى ئۇلاردىن چەتكە ئالىدۇ
۞ إِنَّ قَـٰرُونَ كَانَ مِن قَوْمِ مُوسَىٰ فَبَغَىٰ عَلَيْهِمْ ۖ وَءَاتَيْنَـٰهُ مِنَ ٱلْكُنُوزِ مَآ إِنَّ مَفَاتِحَهُۥ لَتَنُوٓأُ بِٱلْعُصْبَةِ أُو۟لِى ٱلْقُوَّةِ إِذْ قَالَ لَهُۥ قَوْمُهُۥ لَا تَفْرَحْ ۖ إِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْفَرِحِينَ ﴿٧٦﴾
قارۇن ئىشەنچلىك ھالدا مۇسانىڭ قەۋمىدىن ئىدى. قارۇن ئۇلارغا يوغانلىق قىلدى، قارۇنغا خەزىنىلەردىن شۇ قەدەر بەرگەن ئىدۇقكى، ئۇلارنىڭ ئاچقۇچلىرىنى (كۆتۈرۈش) كۈچلۈك بىر جامائەگىمۇ ئىشەنچلىك ھالدا ئېغىرلىق قىلاتتى، ئەينى ۋاقىتتا قارۇنغا قەۋمى ئېيتتى: «كۆرەڭلەپ كەتمە، اﷲ ئىشەنچلىك ھالدا كۆرەڭلەپ كەتكۈچىلەرنى دوست تۇتمايدۇ
وَٱبْتَغِ فِيمَآ ءَاتَىٰكَ ٱللَّهُ ٱلدَّارَ ٱلْـَٔاخِرَةَ ۖ وَلَا تَنسَ نَصِيبَكَ مِنَ ٱلدُّنْيَا ۖ وَأَحْسِن كَمَآ أَحْسَنَ ٱللَّهُ إِلَيْكَ ۖ وَلَا تَبْغِ ٱلْفَسَادَ فِى ٱلْأَرْضِ ۖ إِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْمُفْسِدِينَ ﴿٧٧﴾
اﷲ ساڭا بەرگەن بايلىق بىلەن ئاخىرەت يۇرتىنى تىلىگىن، دۇنيادىكى نېسىۋەڭنىمۇ ئۇنتۇمىغىن، اﷲ ساڭا ياخشىلىق قىلغاندەك، سەن (اﷲ نىڭ بەندىلىرىگىمۇ) ياخشىلىق قىلغىن، يەر يۈزىدە بۇزغۇنچىلىقنى تىلىمىگىن، اﷲ ئىشەنچلىك ھالدا بۇزغۇنچىلىق قىلغۇچىلارنى دوست تۇتمايدۇ»
قَالَ إِنَّمَآ أُوتِيتُهُۥ عَلَىٰ عِلْمٍ عِندِىٓ ۚ أَوَلَمْ يَعْلَمْ أَنَّ ٱللَّهَ قَدْ أَهْلَكَ مِن قَبْلِهِۦ مِنَ ٱلْقُرُونِ مَنْ هُوَ أَشَدُّ مِنْهُ قُوَّةً وَأَكْثَرُ جَمْعًا ۚ وَلَا يُسْـَٔلُ عَن ذُنُوبِهِمُ ٱلْمُجْرِمُونَ ﴿٧٨﴾
قارۇن: «مېنىڭ ئالاھىدە بىلىمىم بولغانلىقتىن بۇ بايلىققا ئېرىشتىم» دېدى. قارۇندىن ئىلگىرى ئۆتكەن ئۈممەتلەردىن ئۇنىڭغا قارىغاندا تېخىمۇ كۈچلۈك، توپلىغان (مېلى) تېخىمۇ كۆپ بولغان كىشىلەرنى اﷲ نىڭ ھالاك قىلغانلىقىنى ئۇ بىلمىدىمۇ؟ گۇناھكارلارنىڭ گۇناھلىرىنىڭ سورىلىشى (اﷲ بۇنى بىلگەنلىكى ئۈچۈن) ھاجەتسىزدۇر
فَخَرَجَ عَلَىٰ قَوْمِهِۦ فِى زِينَتِهِۦ ۖ قَالَ ٱلَّذِينَ يُرِيدُونَ ٱلْحَيَوٰةَ ٱلدُّنْيَا يَـٰلَيْتَ لَنَا مِثْلَ مَآ أُوتِىَ قَـٰرُونُ إِنَّهُۥ لَذُو حَظٍّ عَظِيمٍ ﴿٧٩﴾
قارۇن ئۆز قەۋمىنىڭ ئالدىغا بارلىق زىننىتى بىلەن ھەشەمەتلىك ھالدا چىقتى. دۇنيا تىرىكچىلىكىنى كۆزلەيدىغانلار: «كاشكى بىزگە قارۇنغا بېرىلگەن بايلىق بېرىلسىچۇ، ئۇ ئىشەنچلىك ھالدا (دۇنيالىقتىن) چوڭ نېسىۋىگە ئىگە ئىكەن» دېدى
وَقَالَ ٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ وَيْلَكُمْ ثَوَابُ ٱللَّهِ خَيْرٌ لِّمَنْ ءَامَنَ وَعَمِلَ صَـٰلِحًا وَلَا يُلَقَّىٰهَآ إِلَّا ٱلصَّـٰبِرُونَ ﴿٨٠﴾
ئىلىملىك كىشىلەر: «ۋاي سىلەرگە! ئىمان ئېيتقان ۋە ياخشى ئەمەلنى قىلغان كىشىگە اﷲ نىڭ ساۋابى ياخشىدۇر. ئۇ ساۋاب پەقەت سەۋر قىلغۇچىلارغىلا بېرىلىدۇ» دېدى
فَخَسَفْنَا بِهِۦ وَبِدَارِهِ ٱلْأَرْضَ فَمَا كَانَ لَهُۥ مِن فِئَةٍ يَنصُرُونَهُۥ مِن دُونِ ٱللَّهِ وَمَا كَانَ مِنَ ٱلْمُنتَصِرِينَ ﴿٨١﴾
قارۇننى ئۇنىڭ ئۆيى بىلەن قوشۇپ يەرگە يۇتقۇزدۇق، ئۇنىڭدىن اﷲ نىڭ ئازابىنى دەپئى قىلىدىغان بىر جامائە بولمىدى. قارۇن ئۆزىنى قوغدىيالمىدى
وَأَصْبَحَ ٱلَّذِينَ تَمَنَّوْا۟ مَكَانَهُۥ بِٱلْأَمْسِ يَقُولُونَ وَيْكَأَنَّ ٱللَّهَ يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ مِنْ عِبَادِهِۦ وَيَقْدِرُ ۖ لَوْلَآ أَن مَّنَّ ٱللَّهُ عَلَيْنَا لَخَسَفَ بِنَا ۖ وَيْكَأَنَّهُۥ لَا يُفْلِحُ ٱلْكَـٰفِرُونَ ﴿٨٢﴾
تۈنۈگۈن تېخى قارۇننىڭ دەرىجىسىنى ئارزۇ قىلغانلار دېيىشتىكى، «پاھ كۆردۈڭمۇ؟ اﷲ بەندىلىرىدىن خالىغان ئادەمنىڭ رىزقىنى كەڭ قىلىدىكەن، (خالىغان ئادەمنىڭ رىزقىنى) تار قىلىدىكەن، اﷲ بىزگە مەرھەمەت قىلمىغان (يەنى ئازۇ قىلغىنىمىزنى بەرگەن) بولسا ئىدى، بىزنى ئەلۋەتتە يەر يۇتاتتى، ۋاي كۆردۈڭمۇ؟ ھەقىقەتنى رەت قىلغۇچىلار نىجات تاپمايدۇ»
تِلْكَ ٱلدَّارُ ٱلْـَٔاخِرَةُ نَجْعَلُهَا لِلَّذِينَ لَا يُرِيدُونَ عُلُوًّا فِى ٱلْأَرْضِ وَلَا فَسَادًا ۚ وَٱلْعَـٰقِبَةُ لِلْمُتَّقِينَ ﴿٨٣﴾
ئەنە شۇ ئاخىرەت يۇرتىنى يەر يۈزىدە چوڭچىلىق قىلىشنى ۋە بۇزغۇنچىلىق قىلىشنى كۆزلىمەيدىغانلارغا خاس قىلدۇق، (ياخشى) ئاقىۋەت تەقۋادارلارغا مەنسۇپتۇر
مَن جَآءَ بِٱلْحَسَنَةِ فَلَهُۥ خَيْرٌ مِّنْهَا ۖ وَمَن جَآءَ بِٱلسَّيِّئَةِ فَلَا يُجْزَى ٱلَّذِينَ عَمِلُوا۟ ٱلسَّيِّـَٔاتِ إِلَّا مَا كَانُوا۟ يَعْمَلُونَ ﴿٨٤﴾
كىمكى ياخشى ئىش قىلغان ئىكەن، ئۇ قىلغان ئىشلىرىدىنمۇ ياخشى مۇكاپاتقا (يەنى قىلغان ياخشىلىقىدىن نەچچە ھەسسە ئارتۇق ساۋابقا) ئېرىشىدۇ، كىمكى يامان ئىش قىلىدىكەن، يامان ئىش قىلغانلارغا پەقەت قىلغان ئىشلىرىغا يارىشا جازا بېرىلىدۇ
إِنَّ ٱلَّذِى فَرَضَ عَلَيْكَ ٱلْقُرْءَانَ لَرَآدُّكَ إِلَىٰ مَعَادٍ ۚ قُل رَّبِّىٓ أَعْلَمُ مَن جَآءَ بِٱلْهُدَىٰ وَمَنْ هُوَ فِى ضَلَـٰلٍ مُّبِينٍ ﴿٨٥﴾
ساڭا قۇرئاننى (يەنى ئۇنى تەبلىغ قىلىشنى ۋە ئۇنىڭ ئەھكامىغا ئەمەل قىلىشنى) پەرز قىلغان زات سېنى ئەلۋەتتە قايتىدىغان يەرگە (يەنى مەككىگە) قايتۇرىدۇ، ئېيتقىنكى، «كىمنىڭ ھىدايەت ئېلىپ كەلگەنلىكىنى، كىمنىڭ ئوپئوچۇق گۇمراھلىقتا ئىكەنلىكىنى پەرۋەردىگارىم ھەممىدىن ئوبدان بىلىدۇ»
وَمَا كُنتَ تَرْجُوٓا۟ أَن يُلْقَىٰٓ إِلَيْكَ ٱلْكِتَـٰبُ إِلَّا رَحْمَةً مِّن رَّبِّكَ ۖ فَلَا تَكُونَنَّ ظَهِيرًا لِّلْكَـٰفِرِينَ ﴿٨٦﴾
سەن (خۇدانىڭ ئەلچىسى بولۇشتىن ئىلگىرى) قۇرئاننىڭ ساڭا نازىل قىلىنىشىنى ئۈمىد قىلمىغان ئىدىڭ، پەقەت پەرۋەردىگارىڭ رەھمەت قىلىپ (ساڭا ئۇنى نازىل قىلدى)، ھەقىقەتنى رەت قىلغۇچىلارغا ياردەمچى بولمىغىن
وَلَا يَصُدُّنَّكَ عَنْ ءَايَـٰتِ ٱللَّهِ بَعْدَ إِذْ أُنزِلَتْ إِلَيْكَ ۖ وَٱدْعُ إِلَىٰ رَبِّكَ ۖ وَلَا تَكُونَنَّ مِنَ ٱلْمُشْرِكِينَ ﴿٨٧﴾
ساڭا اﷲ نىڭ ئايەتلىرى نازىل قىلىنغاندىن كېيىن، ئۇ ئايەتلەرگە (ئەمەل قىلىش) تىن ھەقىقەتنى رەت قىلغۇچىلار سېنى توسمىسۇن، پەرۋەردىگارىڭغا (يەنى ئۇنى بىردەپ بىلىشكە ۋە ئۇنىڭغا ئىبادەت قىلىشقا) دەۋەت قىلغىن، ھەرگىزمۇ مۇشرىكلاردىن بولمىغىن (يەنى ئۇلارنىڭ نەپسى خاھىشلىرىغا ماسلاشمىغىن)
وَلَا تَدْعُ مَعَ ٱللَّهِ إِلَـٰهًا ءَاخَرَ ۘ لَآ إِلَـٰهَ إِلَّا هُوَ ۚ كُلُّ شَىْءٍ هَالِكٌ إِلَّا وَجْهَهُۥ ۚ لَهُ ٱلْحُكْمُ وَإِلَيْهِ تُرْجَعُونَ ﴿٨٨﴾
اﷲ قا قوشۇپ باشقا بىر ئىلاھقا ئىبادەت قىلما، اﷲ تىن باشقا ھېچ ئىلاھ يوقتۇر، اﷲ نىڭ زاتىدىن باشقا بارلىق نەرسە يوقىلىدۇ، (مەخلۇقاتتا ئىجرا بولىدىغان) ھۆكۈم پەقەت اﷲ قا خاستۇر، سىلەر اﷲ نىڭ دەرگاھىغا قايتۇرۇلىسىلەر
