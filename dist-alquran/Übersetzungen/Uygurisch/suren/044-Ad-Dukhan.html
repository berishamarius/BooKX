حا، مىم
وَٱلْكِتَـٰبِ ٱلْمُبِينِ ﴿٢﴾
روشەن قۇرئان بىلەن قەسەم قىلىمەن
إِنَّآ أَنزَلْنَـٰهُ فِى لَيْلَةٍ مُّبَـٰرَكَةٍ ۚ إِنَّا كُنَّا مُنذِرِينَ ﴿٣﴾
بىز قۇرئاننى ئىشەنچلىك ھالدا مۇبارەك كېچىدە (يەنى شەبى قەدرى كېچىسىدە) نازىل قىلدۇق، بىز ئىشەنچلىك ھالدا ئىنسانلارنى (قۇرئان بىلەن) ئاگاھلاندۇرغۇچى بولدۇق
فِيهَا يُفْرَقُ كُلُّ أَمْرٍ حَكِيمٍ ﴿٤﴾
ئۇ كېچىدە ھەر بىر ھېكمەتلىك (يەنى ھېكمەتنىڭ تەقەززاسى بويىچە قىلىنغان) ئىش (يەنى بەندىلەرنىڭ رىزقى، ئەجىلى ۋە باشقا ئەھۋالى) ئايرىلىدۇ
أَمْرًا مِّنْ عِندِنَآ ۚ إِنَّا كُنَّا مُرْسِلِينَ ﴿٥﴾
(شۇ كېچە تەقدىر قىلىنغان ئىشلارنىڭ ھەممىسى) بىزنىڭ دەرگاھىمىزدىن بولغان ئىشتۇر، بىز ئىشەنچلىك ھالدا (ئىنسانلارغا خۇدانىڭ ئەلچى-رەسۇللىرىنى) ئەۋەتكۈچى بولدۇق
رَحْمَةً مِّن رَّبِّكَ ۚ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٦﴾
(شۇ) پەرۋەردىگارىڭنىڭ رەھمىتىدىندۇر. ئۇ ئىشەنچلىك ھالدا (بەندىلەرنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر. (ئەھۋالىنى) بىلىپ تۇرغۇچىدۇر
رَبِّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَآ ۖ إِن كُنتُم مُّوقِنِينَ ﴿٧﴾
ئەگەر سىلەر ھەقىقىي ئىشەنگۈچى بولساڭلار، (ئۇ) ئاسمانلارنىڭ، زېمىننىڭ ۋە ئۇلارنىڭ ئارىسىدىكى نەرسىلەرنىڭ پەرۋەردىگارىدۇر
لَآ إِلَـٰهَ إِلَّا هُوَ يُحْىِۦ وَيُمِيتُ ۖ رَبُّكُمْ وَرَبُّ ءَابَآئِكُمُ ٱلْأَوَّلِينَ ﴿٨﴾
ئۇنىڭدىن باشقا ھېچ ئىلاھ يوقتۇر. ئۇ (ئۆلۈكنى) تىرىلدۈرەلەيدۇ، (تىرىكنىڭ) جېنىنى ئالالايدۇ، ئۇ سىلەرنىڭ پەرۋەردىگارىڭلاردۇر، ئىلگىرىكى ئاتا - بوۋاڭلارنىڭ پەرۋەردىگارىدۇر
بَلْ هُمْ فِى شَكٍّ يَلْعَبُونَ ﴿٩﴾
ئۇنداق ئەمەس (يەنى ئۇلار ھەقىقىي ئىشەنگۈچى ئەمەس)، بەلكى ئۇلار (قىيامەتتىن) شەكتە بولۇپ (ھەقنى ئىستىھزا قىلىپ) ئوينايدۇ
فَٱرْتَقِبْ يَوْمَ تَأْتِى ٱلسَّمَآءُ بِدُخَانٍ مُّبِينٍ ﴿١٠﴾
(ئى مۇھەممەد!) سەن ئاسمان (ھەممە ئادەم) ئوچۇق (كۆرىدىغان) تۈتۈننى كەلتۈرىدىغان كۈندە (ئۇلارغا بولىدىغان ئازابنى) كۈتكىن
يَغْشَى ٱلنَّاسَ ۖ هَـٰذَا عَذَابٌ أَلِيمٌ ﴿١١﴾
(ئۇ تۈتۈن) كىشىلەرنى ئورىۋالىدۇ، بۇ قاتتىق ئازابتۇر
رَّبَّنَا ٱكْشِفْ عَنَّا ٱلْعَذَابَ إِنَّا مُؤْمِنُونَ ﴿١٢﴾
(ئۇلار) «پەرۋەردىگارىمىز! بىزدىن ئازابنى كۆتۈرۈۋەتكىن، بىز ئىشەنچلىك ھالدا ئىمان كەلتۈرگۈچىدۇرمىز» (دەيدۇ)
أَنَّىٰ لَهُمُ ٱلذِّكْرَىٰ وَقَدْ جَآءَهُمْ رَسُولٌ مُّبِينٌ ﴿١٣﴾
(ئازاب كۆتۈرۈلۈپ كەتكەندە) ئۇلار قانداقمۇ ئىبرەت ئالسۇن، ھالبۇكى، ئۇلارغا روشەن خۇدانىڭ ئەلچىسى كەلدى (شۇنداق تۇرۇقلۇق ئۇلار ئىمان ئېيتمىدى)
ثُمَّ تَوَلَّوْا۟ عَنْهُ وَقَالُوا۟ مُعَلَّمٌ مَّجْنُونٌ ﴿١٤﴾
ئاندىن ئۇلار ئۇنىڭدىن يۈز ئۆرۈدى ۋە: «ئۇ ئۆگىتىلگەن (يەنى قۇرئاننى كىشىلەردىن ئۆگەنگەن) دۇر، مەجنۇندۇر» دېدى
إِنَّا كَاشِفُوا۟ ٱلْعَذَابِ قَلِيلًا ۚ إِنَّكُمْ عَآئِدُونَ ﴿١٥﴾
بىز ئىشەنچلىك ھالدا ئازابنى (سىلەردىن) ئازغىنا ۋاقىت كۆتۈرۈۋېتىمىز، (ئاندىن) سىلەر ئىشەنچلىك ھالدا (ئىلگىرىكى ھەقىقەتنى رەت قىلغۇچىلىق ھالىتىڭلارغا) قايتىۋالىسىلەر
يَوْمَ نَبْطِشُ ٱلْبَطْشَةَ ٱلْكُبْرَىٰٓ إِنَّا مُنتَقِمُونَ ﴿١٦﴾
ئۇلارنى قاتتىق تۇتقان كۈنىمىزدە چوقۇم جازالايمىز
۞ وَلَقَدْ فَتَنَّا قَبْلَهُمْ قَوْمَ فِرْعَوْنَ وَجَآءَهُمْ رَسُولٌ كَرِيمٌ ﴿١٧﴾
ئۇلاردىن ئىلگىرى بىز ئىشەنچلىك ھالدا پىرئەۋننىڭ قەۋمىنى سىنىدۇق، ئۇلارغا ئېسىل بىر خۇدانىڭ ئەلچىسى كەلدى
أَنْ أَدُّوٓا۟ إِلَىَّ عِبَادَ ٱللَّهِ ۖ إِنِّى لَكُمْ رَسُولٌ أَمِينٌ ﴿١٨﴾
(مۇسا ئۇلارغا ئېيتتى) «ماڭا اﷲ نىڭ بەندىلىرىنى (يەنى بەنى ئىسرائىلنى) تاپشۇرۇپ بېرىڭلار، مەن سىلەرگە ئىشەنچلىك ھالدا ئىشەنچلىك خۇدانىڭ ئەلچىسى
وَأَن لَّا تَعْلُوا۟ عَلَى ٱللَّهِ ۖ إِنِّىٓ ءَاتِيكُم بِسُلْطَـٰنٍ مُّبِينٍ ﴿١٩﴾
اﷲ قا تەكەببۇرلۇق قىلماڭلار، مەن سىلەرگە ئىشەنچلىك ھالدا روشەن مۆجىزە بىلەن كەلدىم
وَإِنِّى عُذْتُ بِرَبِّى وَرَبِّكُمْ أَن تَرْجُمُونِ ﴿٢٠﴾
مەن ئىشەنچلىك ھالدا مېنىڭ پەرۋەردىگارىم ۋە سىلەرنىڭ پەرۋەردىگارىڭلارغا سېغىنىپ، سىلەرنىڭ مېنى ئۆلتۈرۈشۈڭلاردىن پاناھ تىلەيمەن
وَإِن لَّمْ تُؤْمِنُوا۟ لِى فَٱعْتَزِلُونِ ﴿٢١﴾
ئەگەر ماڭا ئىمان كەلتۈرمىسەڭلار مېنى (ئۆز ھالىمغا) قويۇۋېتىڭلار»
فَدَعَا رَبَّهُۥٓ أَنَّ هَـٰٓؤُلَآءِ قَوْمٌ مُّجْرِمُونَ ﴿٢٢﴾
مۇسا پەرۋەردىگارىغا: «بۇلار گۇناھكار قەۋمدۇر» دەپ دۇئا قىلدى
فَأَسْرِ بِعِبَادِى لَيْلًا إِنَّكُم مُّتَّبَعُونَ ﴿٢٣﴾
(اﷲ ئېيتتى) «مېنىڭ بەندىلىرىمنى كېچىدە ئېلىپ چىقىپ كەتكىن، سىلەر ئىشەنچلىك ھالدا قوغلىنىسىلەر
وَٱتْرُكِ ٱلْبَحْرَ رَهْوًا ۖ إِنَّهُمْ جُندٌ مُّغْرَقُونَ ﴿٢٤﴾
دېڭىزنى تىپتىنچ (يەنى سەن ئۆتۈپ بولغاندىن كېيىن قۇرۇق ھالىتى بويىچە) قويغىن، ئۇلار ئىشەنچلىك ھالدا غەرق قىلىنغۇچى قوشۇندۇر»
كَمْ تَرَكُوا۟ مِن جَنَّـٰتٍ وَعُيُونٍ ﴿٢٥﴾
ئۇلار (غەرق بولۇپ) نۇرغۇن باغلارنى، بۇلاقلارنى، زىرائەتلەرنى، چىرايلىق جايلارنى قالدۇردى
وَزُرُوعٍ وَمَقَامٍ كَرِيمٍ ﴿٢٦﴾
ئۇلار (غەرق بولۇپ) نۇرغۇن باغلارنى، بۇلاقلارنى، زىرائەتلەرنى، چىرايلىق جايلارنى قالدۇردى
وَنَعْمَةٍ كَانُوا۟ فِيهَا فَـٰكِهِينَ ﴿٢٧﴾
ئۇلار بەھرىمەن بولۇۋاتقان نېمەتلەرنى قالدۇردى
كَذَٰلِكَ ۖ وَأَوْرَثْنَـٰهَا قَوْمًا ءَاخَرِينَ ﴿٢٨﴾
شۇنداق قىلىپ، ئۇلارنى باشقىلارغا (يەنى بەنى ئىسرائىلغا) مىراس قىلىپ بەردۇق
فَمَا بَكَتْ عَلَيْهِمُ ٱلسَّمَآءُ وَٱلْأَرْضُ وَمَا كَانُوا۟ مُنظَرِينَ ﴿٢٩﴾
ئۇلارغا ئاسمانمۇ، زېمىنمۇ يىغلىمىدى، ئۇلارغا مۆھلەتمۇ بېرىلمىدى
وَلَقَدْ نَجَّيْنَا بَنِىٓ إِسْرَٰٓءِيلَ مِنَ ٱلْعَذَابِ ٱلْمُهِينِ ﴿٣٠﴾
شەك - شۈبھىسىزكى، بىز ئىسرائىل ئەۋلادىنى خار قىلغۇچى ئازابتىن – پىرئەۋن (ئازابى) دىن قۇتقۇزدۇق، پىرئەۋن ئىشەنچلىك ھالدا مۇتەكەببىر ئىدى، ھەددىدىن ئاشقۇچىلاردىن ئىدى
مِن فِرْعَوْنَ ۚ إِنَّهُۥ كَانَ عَالِيًا مِّنَ ٱلْمُسْرِفِينَ ﴿٣١﴾
شەك - شۈبھىسىزكى، بىز ئىسرائىل ئەۋلادىنى خار قىلغۇچى ئازابتىن – پىرئەۋن (ئازابى) دىن قۇتقۇزدۇق، پىرئەۋن ئىشەنچلىك ھالدا مۇتەكەببىر ئىدى، ھەددىدىن ئاشقۇچىلاردىن ئىدى
وَلَقَدِ ٱخْتَرْنَـٰهُمْ عَلَىٰ عِلْمٍ عَلَى ٱلْعَـٰلَمِينَ ﴿٣٢﴾
ئۇلارنى (يەنى ئۇلارنىڭ ئەينى زاماندا مۇشۇنداق شەرەپكە لايىق ئىكەنلىكىنى) بىلىپ تۇرۇپ (ئۆز زامانىدىكى) جاھان ئەھلىدىن ئارتۇق قىلدۇق
وَءَاتَيْنَـٰهُم مِّنَ ٱلْـَٔايَـٰتِ مَا فِيهِ بَلَـٰٓؤٌا۟ مُّبِينٌ ﴿٣٣﴾
ئۇلارغا (يەنى تەپەككۇر قىلغۇچىلارغا) روشەن سىناق بار نۇرغۇن مۆجىزىلەرنى ئاتا قىلدۇق
إِنَّ هَـٰٓؤُلَآءِ لَيَقُولُونَ ﴿٣٤﴾
بۇلار (يەنى قۇرەيش كۇففارلىرى) چوقۇم ئېيتىدۇ:
إِنْ هِىَ إِلَّا مَوْتَتُنَا ٱلْأُولَىٰ وَمَا نَحْنُ بِمُنشَرِينَ ﴿٣٥﴾
«پەقەت بىرلا قېتىم ئۆلىمىز، قايتا تىرىلمەيمىز، ئەگەر (ئاخىرەت بار دېگەن سۆزۈڭلاردا) راستچىل بولساڭلار، بىزنىڭ ئاتا - بوۋىلىرىمىزنى (تىرىلدۈرۈپ) ئەكېلىپ بېقىڭلار»

ئۇلار (يەنى مۇشرىكلار) كۈچلۈكمۇ؟ ياكى تۈببە قەۋمىمۇ؟ ئۇلاردىن ئىلگىرى ئۆتكەن (نۇرغۇن) قەۋملەرنى ھالاك قىلدۇق. چۈنكى ئۇلار گۇناھكار ئىدى

ئۇ ئېرىتىلگەن مىستەك (قىزىق) دۇر، ئۇ قورساقلاردا قايناقسۇدەك قاينايدۇ
كَغَلْىِ ٱلْحَمِيمِ ﴿٤٦﴾
ئۇ ئېرىتىلگەن مىستەك (قىزىق) دۇر، ئۇ قورساقلاردا قايناقسۇدەك قاينايدۇ
خُذُوهُ فَٱعْتِلُوهُ إِلَىٰ سَوَآءِ ٱلْجَحِيمِ ﴿٤٧﴾
(دوزاخقا مۇئەككەل خۇدانىڭ ئەلچى پەرىشتىلىرىگە) «ئۇنى تۇتۇپ سۆرەپ دوزاخنىڭ ئوتتۇرىسىغا ئېلىپ بېرىڭلار، ئاندىن ئۇنىڭ بېشىغا قايناقسۇ قۇيۇپ ئازابلاڭلار» دېيىلىدۇ
ثُمَّ صُبُّوا۟ فَوْقَ رَأْسِهِۦ مِنْ عَذَابِ ٱلْحَمِيمِ ﴿٤٨﴾
(دوزاخقا مۇئەككەل خۇدانىڭ ئەلچى پەرىشتىلىرىگە) «ئۇنى تۇتۇپ سۆرەپ دوزاخنىڭ ئوتتۇرىسىغا ئېلىپ بېرىڭلار، ئاندىن ئۇنىڭ بېشىغا قايناقسۇ قۇيۇپ ئازابلاڭلار» دېيىلىدۇ
ذُقْ إِنَّكَ أَنتَ ٱلْعَزِيزُ ٱلْكَرِيمُ ﴿٤٩﴾
(ئۇنىڭغا خارلاش ۋە مەسخىرە قىلىش يۈزىسىدىن) «(بۇ ئازابنى) تېتىغىن، سەن ئىشەنچلىك ھالدا ئىززەتلىك ئۇلۇغ زات ئىدىڭ
إِنَّ هَـٰذَا مَا كُنتُم بِهِۦ تَمْتَرُونَ ﴿٥٠﴾
بۇ (ئازاب) ئىشەنچلىك ھالدا (ھايات ۋاقتىڭلاردا) سىلەر شەكلەنگەن نەرسىدۇر» دېيىلىدۇ
إِنَّ ٱلْمُتَّقِينَ فِى مَقَامٍ أَمِينٍ ﴿٥١﴾
تەقۋادارلار ئىشەنچلىك ھالدا بىخەتەر جايدا بولىدۇ
فِى جَنَّـٰتٍ وَعُيُونٍ ﴿٥٢﴾
باغلاردا، بۇلاقلارنىڭ ئارىسىدا بولىدۇ
يَلْبَسُونَ مِن سُندُسٍ وَإِسْتَبْرَقٍ مُّتَقَـٰبِلِينَ ﴿٥٣﴾
ئۇلار قېلىن، يۇپقا يىپەك كىيىملەرنى كىيىپ بىر - بىرىگە قاراپ ئولتۇرۇشىدۇ
كَذَٰلِكَ وَزَوَّجْنَـٰهُم بِحُورٍ عِينٍ ﴿٥٤﴾
(ئۇلارنى تۈرلۈك ھۆرمەتلەر بىلەن) مۇشۇنداق ئىكرام قىلدۇق، شەھلا كۆزلۈك ھۆرلەرنى ئۇلارغا جۈپ قىلىمىز
يَدْعُونَ فِيهَا بِكُلِّ فَـٰكِهَةٍ ءَامِنِينَ ﴿٥٥﴾
ئۇلار مەڭگۈلۈك باغلارتە (مەيدە ئېغىرلىشىش ۋە كېسەللىكلەردىن) ئەمىن بولغان ھالدا (خىزمەتچىلەردىن) ھەممە مېۋىلەرنى (كەلتۈرۈشنى) تەلەپ قىلىدۇ
لَا يَذُوقُونَ فِيهَا ٱلْمَوْتَ إِلَّا ٱلْمَوْتَةَ ٱلْأُولَىٰ ۖ وَوَقَىٰهُمْ عَذَابَ ٱلْجَحِيمِ ﴿٥٦﴾
ئۇلار ئىلگىرىكى (يەنى دۇنيادىكى) ئۆلۈمدىن باشقا (مەڭگۈلۈك باغلارتە) ھېچقانداق ئۆلۈمنى تېتىمايدۇ، اﷲ ئۇلارنى دوزاخ ئازابىدىن ساقلايدۇ
فَضْلًا مِّن رَّبِّكَ ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٥٧﴾
(بۇ) پەرۋەردىگارىڭنىڭ مەرھەمىتىدىندۇر، بۇ زور مۇۋەپپەقىيەتتۇر
فَإِنَّمَا يَسَّرْنَـٰهُ بِلِسَانِكَ لَعَلَّهُمْ يَتَذَكَّرُونَ ﴿٥٨﴾
ئۇلارنىڭ ۋەز - نەسىھەت ئېلىشى ئۈچۈن قۇرئاننى سېنىڭ تىلىڭ بىلەن ئاسانلاشتۇرۇپ بەردۇق
فَٱرْتَقِبْ إِنَّهُم مُّرْتَقِبُونَ ﴿٥٩﴾
(ئى مۇھەممەد! ئۇلارغا چۈشىدىغان ئازابنى) كۈتكىن، شۈبھىسىزكى، ئۇلار (سېنىڭ ھالاك بولۇشۇڭنى) كۈتكۈچىدۇر
