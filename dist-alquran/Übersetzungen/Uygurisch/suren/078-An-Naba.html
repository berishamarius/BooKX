بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
عَمَّ يَتَسَآءَلُونَ ﴿١﴾
(مۇشرىكلار ئۆزئارا نېمە توغرۇلۇق) سورىشىدۇ؟
عَنِ ٱلنَّبَإِ ٱلْعَظِيمِ ﴿٢﴾
ئۇلار چوڭ بىر خەۋەر (يەنى قىيامەت توغرۇلۇق) سورىشىدۇ
ٱلَّذِى هُمْ فِيهِ مُخْتَلِفُونَ ﴿٣﴾
ئۇلار شۇ (يەنى قىيامەت) توغرىسىدا ئىختىلاپ قىلىشقۇچىلاردۇر
كَلَّا سَيَعْلَمُونَ ﴿٤﴾
ھەرگىز ئۇنداق ئەمەس (قىيامەت ئۇلارنىڭ ئويلىغىنىدەك يالغان ئەمەس)، ئۇلار (ھەقىقىي ئەھۋالنى) كەلگۈسىدە بىلىدۇ
ثُمَّ كَلَّا سَيَعْلَمُونَ ﴿٥﴾
ھەرگىز ئۇنداق ئەمەس، ئۇلار (بېشىغا كەلگەن ئازابنى) كەلگۈسىدە بىلىدۇ
أَلَمْ نَجْعَلِ ٱلْأَرْضَ مِهَـٰدًا ﴿٦﴾
(زېمىننىڭ ئۈستىدە مۇقىملىشىشىڭلار ۋە كەڭ تۈزلەڭلىكلەرگە زىرائەت تېرىپ پايدىلىنىشىڭلار ئۈچۈن) زېمىننى بىسات قىلمىدۇقمۇ؟
وَٱلْجِبَالَ أَوْتَادًا ﴿٧﴾
(زېمىننىڭ تەۋرەپ كەتمەسلىكى ئۈچۈن) تاغلارنى قوزۇق قىلمىدۇقمۇ؟
وَخَلَقْنَـٰكُمْ أَزْوَٰجًا ﴿٨﴾
سىلەرنى ئەر - ئايال قىلىپ ياراتتۇق
وَجَعَلْنَا نَوْمَكُمْ سُبَاتًا ﴿٩﴾
ئۇيقۇنى سىلەرنىڭ (بەدىنىڭلار ئۈچۈن) راھەت قىلدۇق
وَجَعَلْنَا ٱلَّيْلَ لِبَاسًا ﴿١٠﴾
كېچىنى لىباس قىلدۇق
وَجَعَلْنَا ٱلنَّهَارَ مَعَاشًا ﴿١١﴾
كۈندۈزنى (سىلەر) تىرىكچىلىك قىلىدىغان (ۋاقىت) قىلدۇق
وَبَنَيْنَا فَوْقَكُمْ سَبْعًا شِدَادًا ﴿١٢﴾
ئۈستۈڭلاردا مۇستەھكەم يەتتە ئاسماننى بىنا قىلدۇق
وَجَعَلْنَا سِرَاجًا وَهَّاجًا ﴿١٣﴾
(سىلەر ئۈچۈن) يېنىپ تۇرغان چىراغنى (كۈننى) ياراتتۇق
وَأَنزَلْنَا مِنَ ٱلْمُعْصِرَٰتِ مَآءً ثَجَّاجًا ﴿١٤﴾
ئاشلىقلارنى، ئوت - چۆپلەرنى، دەرەخلىرى قويۇق باغچىلارنى ئۆستۈرۈش ئۈچۈن بۇلۇتلاردىن مول ياغدۇرۇپ بەردۇق
لِّنُخْرِجَ بِهِۦ حَبًّا وَنَبَاتًا ﴿١٥﴾
ئاشلىقلارنى، ئوت - چۆپلەرنى، دەرەخلىرى قويۇق باغچىلارنى ئۆستۈرۈش ئۈچۈن بۇلۇتلاردىن مول ياغدۇرۇپ بەردۇق
وَجَنَّـٰتٍ أَلْفَافًا ﴿١٦﴾
ئاشلىقلارنى، ئوت - چۆپلەرنى، دەرەخلىرى قويۇق باغچىلارنى ئۆستۈرۈش ئۈچۈن بۇلۇتلاردىن مول ياغدۇرۇپ بەردۇق
إِنَّ يَوْمَ ٱلْفَصْلِ كَانَ مِيقَـٰتًا ﴿١٧﴾
قىيامەت كۈنى ئىشەنچلىك ھالدا (مۇكاپاتلاش ۋە جازالاش) ۋاقتىدۇر
يَوْمَ يُنفَخُ فِى ٱلصُّورِ فَتَأْتُونَ أَفْوَاجًا ﴿١٨﴾
ئۇ كۈنى سۇر چېلىنىدۇ، سىلەر توپ - توپ بولۇپ كېلىسىلەر
وَفُتِحَتِ ٱلسَّمَآءُ فَكَانَتْ أَبْوَٰبًا ﴿١٩﴾
ئاسمان ئېچىلىپ نۇرغۇن ئىشىكلەر پەيدا بولىدۇ
وَسُيِّرَتِ ٱلْجِبَالُ فَكَانَتْ سَرَابًا ﴿٢٠﴾
تاغلار ھەرىكەتلەندۈرۈلۈپ سەررابتەك بولۇپ قالىدۇ (يەنى تاغلار توزۇپ يىراقتىن قارىغۇچىغا سۇدەك تۇيۇلىدۇ)
إِنَّ جَهَنَّمَ كَانَتْ مِرْصَادًا ﴿٢١﴾
مەڭگۈلۈك ئوت ئىشەنچلىك ھالدا (كۇففارلارنى) كۈتۈپ تۇرىدۇ
لِّلطَّـٰغِينَ مَـَٔابًا ﴿٢٢﴾
مەڭگۈلۈك ئوت، شۈبھىسىزكى، گۇناھكارلارنىڭ جايىدۇر
لَّـٰبِثِينَ فِيهَآ أَحْقَابًا ﴿٢٣﴾
ئۇلار مەڭگۈلۈك ئوتدە ئۇزاق مۇددەت قالىدۇ
لَّا يَذُوقُونَ فِيهَا بَرْدًا وَلَا شَرَابًا ﴿٢٤﴾
ئۇلار مەڭگۈلۈك ئوتدە (مەڭگۈلۈك ئوتنىڭ ھارارىتىنى يىنىكلىتىلگەن) بىرەر سۆرۈنلۈككە ۋە (تەشنالىقنى قاندۇرىدىغان) بىرەر ئىچىملىككە ئېرىشەلمەيدۇ
إِلَّا حَمِيمًا وَغَسَّاقًا ﴿٢٥﴾
ئۇلارنىڭ ئېرىشىدىغىنى پەقەت يۇقىرى ھارارەتلىك قايناقسۇ ۋە يىرىڭدىنلا ئىبارەت بولىدۇ
جَزَآءً وِفَاقًا ﴿٢٦﴾
بۇ (ئۇلارغا بېرىلىدىغان) مۇۋاپىق جازادۇر
إِنَّهُمْ كَانُوا۟ لَا يَرْجُونَ حِسَابًا ﴿٢٧﴾
چۈنكى ئۇلار (قىلمىشلىرىدىن) ھېساب ئېلىنىشتىن قورقمايتتى
وَكَذَّبُوا۟ بِـَٔايَـٰتِنَا كِذَّابًا ﴿٢٨﴾
ھەمدە بىزنىڭ ئايەتلىرىمىزنى قاتتىق ئىنكار قىلاتتى
وَكُلَّ شَىْءٍ أَحْصَيْنَـٰهُ كِتَـٰبًا ﴿٢٩﴾
ھەر نەرسىنى تولۇق خاتىرىلەپ قويىمىز
فَذُوقُوا۟ فَلَن نَّزِيدَكُمْ إِلَّا عَذَابًا ﴿٣٠﴾
(ئۇلارغا ئېيتىمىزكى) «تېتىڭلار! سىلەرگە پەقەت ئازابنىلا زىيادە قىلىمىز»
إِنَّ لِلْمُتَّقِينَ مَفَازًا ﴿٣١﴾
شۈبھىسىزكى، تەقۋادارلار نېمەتكە ئېرىشىدۇ
حَدَآئِقَ وَأَعْنَـٰبًا ﴿٣٢﴾
(ئۇ نېمەت) باغلار، ئۈزۈملەردۇر
وَكَوَاعِبَ أَتْرَابًا ﴿٣٣﴾
تەڭتۇش قىزلاردۇر
وَكَأْسًا دِهَاقًا ﴿٣٤﴾
(شاراب بىلەن) تولدۇرۇلغان جاملاردۇر
لَّا يَسْمَعُونَ فِيهَا لَغْوًا وَلَا كِذَّٰبًا ﴿٣٥﴾
ئۇلار مەڭگۈلۈك باغلارتە يالغان سۆز، بىھۇدە سۆز ئاڭلىمايدۇ
جَزَآءً مِّن رَّبِّكَ عَطَآءً حِسَابًا ﴿٣٦﴾
(ئەنە شۇلار) پەرۋەردىگارىڭنىڭ بەرگەن مۇكاپاتى، يېتەرلىك ئاتاسىدۇر
رَّبِّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ٱلرَّحْمَـٰنِ ۖ لَا يَمْلِكُونَ مِنْهُ خِطَابًا ﴿٣٧﴾
ئۇ ئاسمانلارنىڭ، زېمىننىڭ ۋە ئۇلارنىڭ ئارىسىدىكى شەيئىلەرنىڭ پەرۋەردىگارىدۇر. ناھايىتى شەپقەتلىكتۇر. (ئۇ كۈندە ھېچ كىشى) ئۇنىڭغا سۆز قىلىشقا قادىر بولالمايدۇ
يَوْمَ يَقُومُ ٱلرُّوحُ وَٱلْمَلَـٰٓئِكَةُ صَفًّا ۖ لَّا يَتَكَلَّمُونَ إِلَّا مَنْ أَذِنَ لَهُ ٱلرَّحْمَـٰنُ وَقَالَ صَوَابًا ﴿٣٨﴾
جىبرىئىل ۋە خۇدانىڭ ئەلچى پەرىشتىلىرى سەپ بولۇپ تۇرۇپ كېتىدىغان كۈندە (يەنى قىيامەتتە)، ئۇلار (نىڭ ئارىسىدا) مېھرىبان اﷲ نىڭ ئىزنىگە ئېرىشكەن ۋە توغرا سۆزنى قىلغانلاردىن باشقا ھېچ ئەھەدىنىڭ سۆز قىلىشىغا بولمايدۇ
ذَٰلِكَ ٱلْيَوْمُ ٱلْحَقُّ ۖ فَمَن شَآءَ ٱتَّخَذَ إِلَىٰ رَبِّهِۦ مَـَٔابًا ﴿٣٩﴾
ئەنە شۇ چوقۇم بولىدىغان كۈندۇر. كىمكى (ئىمان بىلەن، ياخشى ئەمەل بىلەن) پەرۋەردىگارىغا قايتىدىغان يولنى تۇتۇشنى خالىسا (تۇتسۇن)
إِنَّآ أَنذَرْنَـٰكُمْ عَذَابًا قَرِيبًا يَوْمَ يَنظُرُ ٱلْمَرْءُ مَا قَدَّمَتْ يَدَاهُ وَيَقُولُ ٱلْكَافِرُ يَـٰلَيْتَنِى كُنتُ تُرَٰبًۢا ﴿٤٠﴾
بىز سىلەرنى ئىشەنچلىك ھالدا (يۈز بېرىشى) يېقىن ئازابتىن (يەنى ئاخىرەت ئازابىدىن) ئاگاھلاندۇردۇق، ئۇ كۈندە كىشى ئىلگىرى قىلغان ئەمەللىرىنى كۆرىدۇ. ھەقىقەتنى رەت قىلغۇچى: «كاشكى مەن تۇپراق بولۇپ كەتسەمچۇ!» دەيدۇ
