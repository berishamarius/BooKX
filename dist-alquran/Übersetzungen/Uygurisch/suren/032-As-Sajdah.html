السجدة
As-Sajdah · The Prostration
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
الٓمٓ ﴿١﴾
ئەلىف، لام، مىم
تَنزِيلُ ٱلْكِتَـٰبِ لَا رَيْبَ فِيهِ مِن رَّبِّ ٱلْعَـٰلَمِينَ ﴿٢﴾
(بۇ) مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) نىڭ ئالەملەرنىڭ پەرۋەردىگارى تەرىپىدىن نازىل قىلىنغانلىقىدا ھېچ شەك يوقتۇر
أَمْ يَقُولُونَ ٱفْتَرَىٰهُ ۚ بَلْ هُوَ ٱلْحَقُّ مِن رَّبِّكَ لِتُنذِرَ قَوْمًا مَّآ أَتَىٰهُم مِّن نَّذِيرٍ مِّن قَبْلِكَ لَعَلَّهُمْ يَهْتَدُونَ ﴿٣﴾
مۇشرىكلار: «مۇھەممەد ئۇنى ئۆزى توقۇدى» دەيدۇ. ھەرگىز ئۇنداق ئەمەس، بەلكى ئۇ سەندىن ئىلگىرى ھېچقانداق بىر خۇدانىڭ ئەلچىسى كەلمىگەن بىر قەۋمنى ھىدايەت تاپسۇن دەپ، ئۇلارنى ئاگاھلاندۇرۇشۇڭ ئۈچۈن، پەرۋەردىگارىڭ تەرىپىدىن كەلگەن ھەقىقەتتۇر
ٱللَّهُ ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَمَا بَيْنَهُمَا فِى سِتَّةِ أَيَّامٍ ثُمَّ ٱسْتَوَىٰ عَلَى ٱلْعَرْشِ ۖ مَا لَكُم مِّن دُونِهِۦ مِن وَلِىٍّ وَلَا شَفِيعٍ ۚ أَفَلَا تَتَذَكَّرُونَ ﴿٤﴾
اﷲ ئاسمانلارنى، زېمىننى ۋە ئۇلارنىڭ ئارىسىدىكى نەرسىلەرنى ئالتە كۈندە ياراتتى، ئاندىن ئەرش ئۈستىدە (ئۆزىگە لايىق رەۋىشتە) قارار ئالدى. سىلەرگە اﷲ تىن باشقا ھېچقانداق ئىگە يوق، شاپائەت قىلغۇچىمۇ يوق (بۇنى) پىكىر قىلىپ (ئىمان ئېيتمامسىلەر؟)
يُدَبِّرُ ٱلْأَمْرَ مِنَ ٱلسَّمَآءِ إِلَى ٱلْأَرْضِ ثُمَّ يَعْرُجُ إِلَيْهِ فِى يَوْمٍ كَانَ مِقْدَارُهُۥٓ أَلْفَ سَنَةٍ مِّمَّا تَعُدُّونَ ﴿٥﴾
اﷲ ئاسماندىن زېمىنغىچە بولغان (مەخلۇقاتىنىڭ) ئىشلىرىنى ئىدارە قىلىپ تۇرىدۇ، ئاندىن ئۇ ئىشلار ئۇزۇنلۇقى سىلەر ساناۋاتقان مىڭ يىلدەك كېلىدىغان بىر كۈندە اﷲ نىڭ دەرگاھىغا ئۆرلەيدۇ
ذَٰلِكَ عَـٰلِمُ ٱلْغَيْبِ وَٱلشَّهَـٰدَةِ ٱلْعَزِيزُ ٱلرَّحِيمُ ﴿٦﴾
ئۇ (اﷲ) يوشۇرۇننى ۋە ئاشكارىنى بىلگۈچىدۇر، غالىبتۇر. (بەندىلىرىگە) ناھايىتى مېھرىباندۇر
ٱلَّذِىٓ أَحْسَنَ كُلَّ شَىْءٍ خَلَقَهُۥ ۖ وَبَدَأَ خَلْقَ ٱلْإِنسَـٰنِ مِن طِينٍ ﴿٧﴾
ئۇ ھەممە نەرسىنى چىرايلىق (يەنى ھېكمىتىگە مۇۋاپىق، جايىدا) ياراتتى. دەسلەپتە ئىنساننى (يەنى ئىنسانلارنىڭ ئاتىسى ئادەم ئەلەيھىسسالامنى) لايدىن ياراتتى
ثُمَّ جَعَلَ نَسْلَهُۥ مِن سُلَـٰلَةٍ مِّن مَّآءٍ مَّهِينٍ ﴿٨﴾
ئاندىن ئۇنىڭ نەسلىنى ئەرزىمەس سۇنىڭ جەۋھىرى (يەنى مەنىي) دىن ياراتتى
ثُمَّ سَوَّىٰهُ وَنَفَخَ فِيهِ مِن رُّوحِهِۦ ۖ وَجَعَلَ لَكُمُ ٱلسَّمْعَ وَٱلْأَبْصَـٰرَ وَٱلْأَفْـِٔدَةَ ۚ قَلِيلًا مَّا تَشْكُرُونَ ﴿٩﴾
ئاندىن ئۇنى راۋۇرۇس يارىتىپ، ئۇنىڭغا ئۆزىگە مەنسۇپ روھنى كىرگۈزدى (يەنى جاننى كىرگۈزدى). يەنە سىلەر ئۈچۈن قۇلاقلارنى، كۆزلەرنى ۋە يۈرەكلەرنى ياراتتى، سىلەر ئاز شۈكۈر قىلىسىلەر
وَقَالُوٓا۟ أَءِذَا ضَلَلْنَا فِى ٱلْأَرْضِ أَءِنَّا لَفِى خَلْقٍ جَدِيدٍۭ ۚ بَلْ هُم بِلِقَآءِ رَبِّهِمْ كَـٰفِرُونَ ﴿١٠﴾
ئۇلار: «يەر ئاستىدا يوقالغاندىن كېيىن (يەنى گۆردە توپىغا ئايلىنىپ كەتكەندىن كېيىن) راستلا قايتا يارىتىلامدۇق؟» دەيدۇ، بەلكى ئۇلار پەرۋەردىگارىغا مۇلاقات بولۇشنى ئىنكار قىلغۇچىلاردۇر
۞ قُلْ يَتَوَفَّىٰكُم مَّلَكُ ٱلْمَوْتِ ٱلَّذِى وُكِّلَ بِكُمْ ثُمَّ إِلَىٰ رَبِّكُمْ تُرْجَعُونَ ﴿١١﴾
ئېيتقىنكى، «سىلەرنىڭ (جېنىڭلارنى ئېلىشقا) مۇئەككەل قىلىنغان پەرىشتە سىلەرنىڭ جېنىڭلارنى ئالىدۇ، ئاندىن پەرۋەردىگارىڭلارنىڭ دەرگاھىغا قايتۇرۇلىسىلەر»
وَلَوْ تَرَىٰٓ إِذِ ٱلْمُجْرِمُونَ نَاكِسُوا۟ رُءُوسِهِمْ عِندَ رَبِّهِمْ رَبَّنَآ أَبْصَرْنَا وَسَمِعْنَا فَٱرْجِعْنَا نَعْمَلْ صَـٰلِحًا إِنَّا مُوقِنُونَ ﴿١٢﴾
ئەگەر سەن گۇناھكارلارنىڭ پەرۋەردىگارىنىڭ دەرگاھىدا باشلىرىنى ساڭگىلىتىپ تۇرغانلىقىنى كۆرىدىغان بولساڭ (ئەلۋەتتە قورقۇنچلۇق ھالەتنى كۆرگەن بولاتتىڭ)، ئۇلار ئېيتىدۇ: «پەرۋەردىگارىمىز! (ئىشنىڭ ھەقىقىتىنى) كۆردۇق، ئاڭلىدۇق. بىزنى ياخشى ئەمەل قىلىش ئۈچۈن (دۇنياغا) قايتۇرغىن، بىز (ئەمدى) ئىشەنچلىك ھالدا (سېنىڭ ۋەدەڭنىڭ ھەق ئىكەنلىكىگە) ئىشەنگۈچىلەرمىز»
وَلَوْ شِئْنَا لَـَٔاتَيْنَا كُلَّ نَفْسٍ هُدَىٰهَا وَلَـٰكِنْ حَقَّ ٱلْقَوْلُ مِنِّى لَأَمْلَأَنَّ جَهَنَّمَ مِنَ ٱلْجِنَّةِ وَٱلنَّاسِ أَجْمَعِينَ ﴿١٣﴾
ئەگەر بىز خالىساق، ھەر بىر كىشىگە ئەلۋەتتە ھىدايەت بېرەتتۇق، لېكىن مېنىڭ (گۇناھكارلارنى ئازاب قىلىشتىن ئىبارەت) ھۆكمىم بېكىپ كەتتى. مەن چوقۇم جىنلار ۋە ئىنسانلار (يەنى ئۇلارنىڭ گۇناھكارلىرى) - ھەممىسى بىلەن مەڭگۈلۈك ئوتنى تولدۇرىمەن
فَذُوقُوا۟ بِمَا نَسِيتُمْ لِقَآءَ يَوْمِكُمْ هَـٰذَآ إِنَّا نَسِينَـٰكُمْ ۖ وَذُوقُوا۟ عَذَابَ ٱلْخُلْدِ بِمَا كُنتُمْ تَعْمَلُونَ ﴿١٤﴾
(ئەھلى دوزاخقا ئېيتىلىدۇ) «مۇشۇ كۈندىكى ئۇچرىشىشىڭلارنى ئۇنتۇپ قالغانلىقىڭلار ئۈچۈن (بۇ خورلىغۇچى ئازابنى) تېتىڭلار، بىز ئىشەنچلىك ھالدا سىلەرنى ئۇنتۇدۇق، قىلمىشىڭلار تۈپەيلىدىن مەڭگۈلۈك ئازابنى تېتىڭلار»
إِنَّمَا يُؤْمِنُ بِـَٔايَـٰتِنَا ٱلَّذِينَ إِذَا ذُكِّرُوا۟ بِهَا خَرُّوا۟ سُجَّدًا وَسَبَّحُوا۟ بِحَمْدِ رَبِّهِمْ وَهُمْ لَا يَسْتَكْبِرُونَ ۩ ﴿١٥﴾
بىزنىڭ ئايەتلىرىمىزگە ئىشەنچلىك ھالدا شۇنداق كىشىلەر ئىمان ئېيتىدۇكى، ئۇلارغا شۇ ئايەتلەر بىلەن ۋەز - نەسىھەت قىلىنسا (اﷲ نىڭ ئايەتلىرىنى ئۇلۇغلاش يۈزىسىدىن اﷲ قا) سەجدە قىلغان ھالدا يىقىلىدۇ، پەرۋەردىگارىغا تەسبىھ ئېيتىدۇ، ھەمدۇ ئېيتىدۇ. ئۇلار (پەرۋەردىگارىنىڭ تائەت - ئىبادىتىدىن) چوڭچىلىق قىلىپ باش تارتمايدۇ
تَتَجَافَىٰ جُنُوبُهُمْ عَنِ ٱلْمَضَاجِعِ يَدْعُونَ رَبَّهُمْ خَوْفًا وَطَمَعًا وَمِمَّا رَزَقْنَـٰهُمْ يُنفِقُونَ ﴿١٦﴾
ئۇلارنىڭ يانلىرى ئورۇن - كۆرپىدىن يىراق بولىدۇ (يەنى ئۇلار كېچىسى ئىبادەت قىلىپ ئاز ئۇخلايدۇ)، ئۇلار پەرۋەردىگارىنىڭ (ئازابىدىن) قورقۇپ، (رەھمىتىنى) ئۈمىد قىلىپ ئۇنىڭغا دۇئا قىلىدۇ، ئۇلارغا بىز رىزىق قىلىپ بەرگەن نەرسىلەردىن (ياخشىلىق يوللىرىغا) سەرپ قىلىدۇ
فَلَا تَعْلَمُ نَفْسٌ مَّآ أُخْفِىَ لَهُم مِّن قُرَّةِ أَعْيُنٍ جَزَآءًۢ بِمَا كَانُوا۟ يَعْمَلُونَ ﴿١٧﴾
ئۇلارنىڭ قىلغان ئەمەللىرىگە مۇكاپات يۈزىسىدىن اﷲ نىڭ ھۇزۇرىدا ساقلانغان ۋە ئۇلارنى خۇشال قىلىدىغان كاتتا نېمەتنى ھېچكىم بىلمەيدۇ
أَفَمَن كَانَ مُؤْمِنًا كَمَن كَانَ فَاسِقًا ۚ لَّا يَسْتَوُۥنَ ﴿١٨﴾
مۆمىن ئادەم پاسىق ئادەمگە ئوخشامدۇ؟ ئۇلار (ئاخىرەتلىك ساۋابتا) باراۋەر ئەمەس
أَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فَلَهُمْ جَنَّـٰتُ ٱلْمَأْوَىٰ نُزُلًۢا بِمَا كَانُوا۟ يَعْمَلُونَ ﴿١٩﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارغا كەلسەك، ئۇلارنىڭ قارارگاھى مەڭگۈلۈك باغلار بولىدۇ، (ئۇ) ئۇلارنىڭ قىلغان ئەمەللىرىنى مۇكاپاتلاش يۈزىسىدىن ئۇلا رئۈچۈن تەييارلانغان جايدۇر
وَأَمَّا ٱلَّذِينَ فَسَقُوا۟ فَمَأْوَىٰهُمُ ٱلنَّارُ ۖ كُلَّمَآ أَرَادُوٓا۟ أَن يَخْرُجُوا۟ مِنْهَآ أُعِيدُوا۟ فِيهَا وَقِيلَ لَهُمْ ذُوقُوا۟ عَذَابَ ٱلنَّارِ ٱلَّذِى كُنتُم بِهِۦ تُكَذِّبُونَ ﴿٢٠﴾
اﷲ نىڭ ئىتائىتىدىن چىققانلارغا كەلسەك، ئۇلارنىڭ جايى دوزاخ بولىدۇ، ھەر قاچان ئۇلار دوزاختىن چىقماقچى بولسا، ئۇلار دوزاخقا قايتۇرۇلىدۇ. ئۇلارغا: «سىلەر ئىنكار قىلغان دوزاخنىڭ ئازابىنى تېتىڭلار» دېيىلىدۇ
وَلَنُذِيقَنَّهُم مِّنَ ٱلْعَذَابِ ٱلْأَدْنَىٰ دُونَ ٱلْعَذَابِ ٱلْأَكْبَرِ لَعَلَّهُمْ يَرْجِعُونَ ﴿٢١﴾
ئۇلارنىڭ تەۋبە قىلىشى ئۈچۈن بىز ئۇلارغا چوقۇم ئەڭ چوڭ ئازابتىن (يەنى ئاخىرەت ئازابىدىن) ئىلگىرى ئەڭ يېقىنقى ئازاب (يەنى دۇنيا ئازابى) نى تېتىتىمىز
وَمَنْ أَظْلَمُ مِمَّن ذُكِّرَ بِـَٔايَـٰتِ رَبِّهِۦ ثُمَّ أَعْرَضَ عَنْهَآ ۚ إِنَّا مِنَ ٱلْمُجْرِمِينَ مُنتَقِمُونَ ﴿٢٢﴾
پەرۋەردىگارىنىڭ ئايەتلىرى بىلەن ۋەز - نەسىھەت قىلىنغان، ئاندىن ئۇنىڭدىن يۈز ئۆرۈگەن ئادەمدىنمۇ زالىم ئادەم بارمۇ؟ بىز گۇناھكارلارنى چوقۇم جازالايمىز
وَلَقَدْ ءَاتَيْنَا مُوسَى ٱلْكِتَـٰبَ فَلَا تَكُن فِى مِرْيَةٍ مِّن لِّقَآئِهِۦ ۖ وَجَعَلْنَـٰهُ هُدًى لِّبَنِىٓ إِسْرَٰٓءِيلَ ﴿٢٣﴾
شەك - شۈبھىسىزكى، بىز مۇساغا مۇقەددەس كىتاب-ۋەھىي (يەنى تەۋراتنى) ئاتا قىلدۇق. (ئى مۇھەممەد!) سەن (مىراج كېچىسىدە، يا قىيامەت كۈنىدە) مۇساغا مۇلاقات بولۇشتىن شەكلەنمىگىن، بىز تەۋراتنى ئىسرائىل ئەۋلادىغا يېتەكچى قىلدۇق
وَجَعَلْنَا مِنْهُمْ أَئِمَّةً يَهْدُونَ بِأَمْرِنَا لَمَّا صَبَرُوا۟ ۖ وَكَانُوا۟ بِـَٔايَـٰتِنَا يُوقِنُونَ ﴿٢٤﴾
ئۇلار (كۈلپەتلەرگە) سەۋر قىلغان، بىزنىڭ ئايەتلىرىمىزگە جەزمەن ئىشەنگەن چاغدا، ئۇلاردىن بىر قىسىم كىشىلەرنى بىزنىڭ ئەمرىمىز بىلەن توغرا يول كۆرسىتىدىغان پېشۋالار قىلدۇق
إِنَّ رَبَّكَ هُوَ يَفْصِلُ بَيْنَهُمْ يَوْمَ ٱلْقِيَـٰمَةِ فِيمَا كَانُوا۟ فِيهِ يَخْتَلِفُونَ ﴿٢٥﴾
سېنىڭ پەرۋەردىگارىڭ قىيامەت كۈنى ئۇلارنىڭ ئارىسىدا ئۇلار ئىختىلاپ قىلىشقان (دىنىي) ئىشلار ئۈستىدە ئەلۋەتتە ھۆكۈم چىقىرىدۇ
أَوَلَمْ يَهْدِ لَهُمْ كَمْ أَهْلَكْنَا مِن قَبْلِهِم مِّنَ ٱلْقُرُونِ يَمْشُونَ فِى مَسَـٰكِنِهِمْ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ ۖ أَفَلَا يَسْمَعُونَ ﴿٢٦﴾
ئۇلاردىن ئىلگىرى ئۆتكەن قانچىلىغان ئۈممەتلەرنى ھالاك قىلغانلىقىمىزنى ئۇلار بىلمەمدۇ؟ ئۇلار (يەنى مەككىلىكلەر) ئۇلار (يەنى ھالاك بولغانلار) نىڭ يۇرتىدىن ئۆتۈپ تۇرىدۇ، بۇنىڭدا (بىزنىڭ قۇدرىتىمىزنىڭ چوڭلۇقىنى كۆرسىتىدىغان) نۇرغۇن دەلىللەر بار، ئۇلار ئاڭلىمامدۇ؟
أَوَلَمْ يَرَوْا۟ أَنَّا نَسُوقُ ٱلْمَآءَ إِلَى ٱلْأَرْضِ ٱلْجُرُزِ فَنُخْرِجُ بِهِۦ زَرْعًا تَأْكُلُ مِنْهُ أَنْعَـٰمُهُمْ وَأَنفُسُهُمْ ۖ أَفَلَا يُبْصِرُونَ ﴿٢٧﴾
ئۇلار كۆرمەمدۇكى، بىز گىياھسىز زېمىنغا يامغۇر سۈيىنى سۈرۈپ ئېلىپ بارىمىز، ئۇنىڭ بىلەن ئۇلارنىڭ ھايۋاناتلىرى ۋە ئۆزلىرى يەيدىغان زىرائەتلەرنى ئۈندۈرۈپ چىقىرىمىز، ئۇلار (بۇنى) كۆرمەمدۇ؟
وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْفَتْحُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٢٨﴾
ئۇلار (يەنى مەككە كۇففارلىرى مەسخىرە قىلىش يۈزىسىدىن): «(بىز بىلەن سىلەرنىڭ ئاراڭلاردا چىقىرىلىدىغان) بۇ ھۆكۈم قاچان ئەمەلگە ئاشىدۇ؟ ئەگەر سىلەر (دەۋايىڭلاردا) راستچىل بولساڭلار (ئېيتىپ بېرىڭلار)» دەيدۇ
قُلْ يَوْمَ ٱلْفَتْحِ لَا يَنفَعُ ٱلَّذِينَ كَفَرُوٓا۟ إِيمَـٰنُهُمْ وَلَا هُمْ يُنظَرُونَ ﴿٢٩﴾
ئېيتقىنكى، «ھۆكۈم چىقىرىلىدىغان كۈندە ھەقىقەتنى رەت قىلغۇچىلارنىڭ ئېيتقان ئىمانىنىڭ پايدىسى بولمايدۇ، ئۇلارغا مۆھلەتمۇ بېرىلمەيدۇ»
فَأَعْرِضْ عَنْهُمْ وَٱنتَظِرْ إِنَّهُم مُّنتَظِرُونَ ﴿٣٠﴾
سەن ئۇلاردىن يۈز ئۆرۈگىن (يەنى سۆزلىرىگە پەرۋا قىلىپ كەتمىگىن). سەن (اﷲ نىڭ ئۇلارغا چۈشىدىغان ئازابىنى) كۈتكىن، ئۇلارمۇ ئىشەنچلىك ھالدا (سىلەرگە چۈشىدىغان نەرسىلەرنى) كۈتكۈچىلەردۇر
