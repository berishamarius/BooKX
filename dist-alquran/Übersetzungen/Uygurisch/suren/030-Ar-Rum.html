رۇملۇقلار يېقىن بىر جايدا يېڭىلدى، ئۇلار يېڭىلگەندىن كېيىن بىر قانچە يىل ئىچىدە يېڭىدۇ، ئىلگىرى ۋە كېيىن ھەممە ئىش اﷲ نىڭ باشقۇرۇشىدىدۇر، بۇ كۈندە مۆمىنلەر اﷲ نىڭ بەرگەن ياردىمى بىلەن خۇشال بولىدۇ، اﷲ خالىغان كىشىگە ياردەم بېرىدۇ، اﷲ غالىبتۇر، (مۆمىنلەرگە) ناھايىتى مېھرىباندۇر
وَعْدَ ٱللَّهِ ۖ لَا يُخْلِفُ ٱللَّهُ وَعْدَهُۥ وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَعْلَمُونَ ﴿٦﴾
اﷲ (ئۇلارنى غالىب قىلىدىغانلىقىنى) ۋەدە قىلدى. اﷲ ۋەدىسىگە خىلاپلىق قىلمايدۇ) چۈنكى اﷲ نىڭ ۋەدىسى ھەقتۇر، سۆزى راستتۇر)، لېكىن كىشىلەرنىڭ تولىسى (بۇنى) بىلمەيدۇ
يَعْلَمُونَ ظَـٰهِرًا مِّنَ ٱلْحَيَوٰةِ ٱلدُّنْيَا وَهُمْ عَنِ ٱلْـَٔاخِرَةِ هُمْ غَـٰفِلُونَ ﴿٧﴾
ئۇلار ھاياتىي دۇنيانىڭ تاشقى كۆرۈنۈشىنىلا بىلىدۇ، ئۇلار ئاخىرەتتىن (يەنى ئاخىرەت ئىشىدا ئويلىنىشتىن ۋە ئاخىرەت ئۈچۈن ئىشلەشتىن) غەپلەتتىدۇر
أَوَلَمْ يَتَفَكَّرُوا۟ فِىٓ أَنفُسِهِم ۗ مَّا خَلَقَ ٱللَّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَمَا بَيْنَهُمَآ إِلَّا بِٱلْحَقِّ وَأَجَلٍ مُّسَمًّى ۗ وَإِنَّ كَثِيرًا مِّنَ ٱلنَّاسِ بِلِقَآئِ رَبِّهِمْ لَكَـٰفِرُونَ ﴿٨﴾
ئۇلار (ئاقىۋىتىنىڭ قانداق بولىدىغانلىقىنى بىلىشلىرى ئۈچۈن) ئۆزلىرى ئۈستىدە پىكىر يۈرگۈزمەمدۇ؟ اﷲ ئاسمانلارنى، زېمىننى ۋە ئۇلارنىڭ ئارىسىدىكى (يۇلتۇز قاتارلىق) نەرسىلەرنى پەقەت ھەق ۋە مۇئەييەن ۋاقىتلىق قىلىپ ياراتتى، شەك – شۈبھىسىزكى، نۇرغۇن كىشىلەر پەرۋەردىگارىغا مۇلاقات بولۇشنى (قايتا تىرىلىدىغانلىقىنى، قىلغان ئەمەلىگە يارىشا مۇكاپاتلىنىدىغانلىقىنى ياكى جازالىنىدىغانلىقىنى) ئىنكار قىلغۇچىدۇر
أَوَلَمْ يَسِيرُوا۟ فِى ٱلْأَرْضِ فَيَنظُرُوا۟ كَيْفَ كَانَ عَـٰقِبَةُ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَانُوٓا۟ أَشَدَّ مِنْهُمْ قُوَّةً وَأَثَارُوا۟ ٱلْأَرْضَ وَعَمَرُوهَآ أَكْثَرَ مِمَّا عَمَرُوهَا وَجَآءَتْهُمْ رُسُلُهُم بِٱلْبَيِّنَـٰتِ ۖ فَمَا كَانَ ٱللَّهُ لِيَظْلِمَهُمْ وَلَـٰكِن كَانُوٓا۟ أَنفُسَهُمْ يَظْلِمُونَ ﴿٩﴾
ئۇلار يەر يۈزىدە كېزىپ ئۆزلىرىدىن ئىلگىرى ئۆتكەنلەرنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۆزەتمىدىمۇ؟ ئۇلار (يەنى ئۆتكەنلەر) قۇۋۋەتتە بۇلاردىن كۈچلۈك ئىدى. ئۇلار زىرائەت تېرىشتا، ئىمارەت سېلىشتا بۇلار (يەنى قۇرەيشلەر) دىن ئۈستۈن ئىدى. ئۇلارغا خۇدانىڭ ئەلچىسىلىرى مۆجىزىلەرنى ئېلىپ كەلدى (ئۇلار مۆجىزىلەرنى ئىنكار قىلدى). اﷲ ئۇلارغا زۇلۇم قىلمىدى (يەنى اﷲ ئۇلارنى گۇناھسىز ھالاك قىلغىنى يوق)، لېكىن ئۇلار ئۆزلىرى (كۇفرى ئۈچۈن ۋە خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ مۆجىزىلىرىنى ئىنكار قىلغانلىقلىرى ئۈچۈن) ئۆزلىرىگە زۇلۇم قىلدى (شۇنىڭ بىلەن ئۇلار ھالاك بولۇشقا تېگىشلىك بولدى)
ثُمَّ كَانَ عَـٰقِبَةَ ٱلَّذِينَ أَسَـٰٓـُٔوا۟ ٱلسُّوٓأَىٰٓ أَن كَذَّبُوا۟ بِـَٔايَـٰتِ ٱللَّهِ وَكَانُوا۟ بِهَا يَسْتَهْزِءُونَ ﴿١٠﴾
ئاندىن يامان ئىش قىلغانلارنىڭ ئاقىۋىتى ئەڭ يامان بولىدۇ، چۈنكى ئۇلار اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغان ۋە مەسخىرە قىلغان كىشىلەردۇر
ٱللَّهُ يَبْدَؤُا۟ ٱلْخَلْقَ ثُمَّ يُعِيدُهُۥ ثُمَّ إِلَيْهِ تُرْجَعُونَ ﴿١١﴾
اﷲ مەخلۇقاتنى دەسلەپتە (يوقتىن) بار قىلىدۇ. ئاندىن ئۇنى (ئۆلگەندىن كېيىن) تىرىلدۈرىدۇ، ئاندىن اﷲ نىڭ دەرگاھىغا (ھېساب ئۈچۈن) قايتۇرۇلىسىلەر
وَيَوْمَ تَقُومُ ٱلسَّاعَةُ يُبْلِسُ ٱلْمُجْرِمُونَ ﴿١٢﴾
قىيامەت قايىم بولغان كۈندە گۇناھكارلار ئۈمىدسىزلىنىپ كېتىدۇ
وَلَمْ يَكُن لَّهُم مِّن شُرَكَآئِهِمْ شُفَعَـٰٓؤُا۟ وَكَانُوا۟ بِشُرَكَآئِهِمْ كَـٰفِرِينَ ﴿١٣﴾
ئۇلارغا شېرىك كەلتۈرگەن بۇتلىرىدىن شاپائەت قىلغۇچىلار بولمايدۇ، ئۇلار شېرىك كەلتۈرگەن بۇتلىرىدىن تانىدۇ
وَيَوْمَ تَقُومُ ٱلسَّاعَةُ يَوْمَئِذٍ يَتَفَرَّقُونَ ﴿١٤﴾
قىيامەت قايىم بولغان كۈندە (مۆمىنلەر بىلەن ھەقىقەتنى رەت قىلغۇچىلار) بىر - بىرىدىن ئايرىلىدۇ
فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فَهُمْ فِى رَوْضَةٍ يُحْبَرُونَ ﴿١٥﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلار باغچىدا (يەنى مەڭگۈلۈك باغلارنىڭ باغچىلىرىدا) يايرىتىلىدۇ
وَأَمَّا ٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِـَٔايَـٰتِنَا وَلِقَآئِ ٱلْـَٔاخِرَةِ فَأُو۟لَـٰٓئِكَ فِى ٱلْعَذَابِ مُحْضَرُونَ ﴿١٦﴾
بىزنىڭ ئايەتلىرىمىزنى ۋە ئاخىرەتتىكى مۇلاقاتنى ئىنكار قىلغان ۋە يالغانغا چىقارغانلارغا كەلسەك، ئۇلار داۋاملىق ئازابتا قالىدۇ
فَسُبْحَـٰنَ ٱللَّهِ حِينَ تُمْسُونَ وَحِينَ تُصْبِحُونَ ﴿١٧﴾
سىلەر ئاخشامدا ۋە ئەتىگەندە اﷲ قا تەسبىھ ئېيتىڭلار
وَلَهُ ٱلْحَمْدُ فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَعَشِيًّا وَحِينَ تُظْهِرُونَ ﴿١٨﴾
ئاسمانلاردىكى، زېمىندىكى ھەمدۇسانا، كەچ ۋە پېشىن ۋاقىتلىرىدىكى ھەمدۇسانانىڭ ھەممىسى اﷲ قا خاستۇر
يُخْرِجُ ٱلْحَىَّ مِنَ ٱلْمَيِّتِ وَيُخْرِجُ ٱلْمَيِّتَ مِنَ ٱلْحَىِّ وَيُحْىِ ٱلْأَرْضَ بَعْدَ مَوْتِهَا ۚ وَكَذَٰلِكَ تُخْرَجُونَ ﴿١٩﴾
اﷲ جانسىز نەرسىلەردىن جانلىق نەرسىلەرنى چىقىرىدۇ، جانلىق نەرسىلەردىن جانسىز نەرسىلەرنى چىقىرىدۇ، ئۆلگەن زېمىننى تىرىلدۈرىدۇ (يەنى قۇرغاق زېمىننى ئۆسۈملۈكلەر بىلەن كۆكەرتىدۇ)، سىلەرمۇ مۇشۇنداق تىرىلدۈرۈلىسىلەر
وَمِنْ ءَايَـٰتِهِۦٓ أَنْ خَلَقَكُم مِّن تُرَابٍ ثُمَّ إِذَآ أَنتُم بَشَرٌ تَنتَشِرُونَ ﴿٢٠﴾
اﷲ نىڭ سىلەرنى تۇپراقتىن ياراتقانلىقى، ئاندىن سىلەرنىڭ ئىنسان بولۇپ زېمىندا تارىلىپ يۈرۈشۈڭلار اﷲ نىڭ (كامالى قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر
وَمِنْ ءَايَـٰتِهِۦٓ أَنْ خَلَقَ لَكُم مِّنْ أَنفُسِكُمْ أَزْوَٰجًا لِّتَسْكُنُوٓا۟ إِلَيْهَا وَجَعَلَ بَيْنَكُم مَّوَدَّةً وَرَحْمَةً ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يَتَفَكَّرُونَ ﴿٢١﴾
ئاياللار بىلەن ئۇنسى – ئۈلپەت ئېلىشىڭلار ئۈچۈن (اﷲ نىڭ) ئۇلارنى سىلەرنىڭ ئۆز تىپىڭلاردىن ياراتقانلىقى، ئاراڭلاردا (يەنى ئەر – خوتۇن ئارىسىدا) مېھر - مۇھەببەت ئورناتقانلىقى اﷲ نىڭ (كامالى قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر، پىكىر يۈرگۈزىدىغان قەۋم ئۈچۈن، شەك - شۈبھىسىزكى، بۇنىڭدا نۇرغۇن ئىبرەتلەر بار
وَمِنْ ءَايَـٰتِهِۦ خَلْقُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَٱخْتِلَـٰفُ أَلْسِنَتِكُمْ وَأَلْوَٰنِكُمْ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّلْعَـٰلِمِينَ ﴿٢٢﴾
اﷲ نىڭ ئاسمانلارنى، زېمىننى ياراتقانلىقى، تىللىرىڭلارنىڭ، رەڭگىلىرىڭلارنىڭ خىلمۇخىل بولۇشى اﷲ نىڭ (كامالى قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر، بۇنىڭدا بىلىملىك كىشىلەر ئۈچۈن ئىشەنچلىك ھالدا نۇرغۇن ئالامەتلەر بار
وَمِنْ ءَايَـٰتِهِۦ مَنَامُكُم بِٱلَّيْلِ وَٱلنَّهَارِ وَٱبْتِغَآؤُكُم مِّن فَضْلِهِۦٓ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يَسْمَعُونَ ﴿٢٣﴾
سىلەرنىڭ كېچىسى ۋە كۈندۈزى ئۇخلىشىڭلار، (كۈندۈزى) اﷲ نىڭ مەرھەمىتىدىن (رىزىق) تەلەپ قىلىپ (ھەرىكەتلىنىشىڭلار) اﷲ نىڭ (كامالى قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر، شەك - شۈبھىسىزكى، بۇنىڭدا ئاڭلايدىغان قەۋم ئۈچۈن نۇرغۇن ئىبرەتلەر بار
وَمِنْ ءَايَـٰتِهِۦ يُرِيكُمُ ٱلْبَرْقَ خَوْفًا وَطَمَعًا وَيُنَزِّلُ مِنَ ٱلسَّمَآءِ مَآءً فَيُحْىِۦ بِهِ ٱلْأَرْضَ بَعْدَ مَوْتِهَآ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يَعْقِلُونَ ﴿٢٤﴾
اﷲ نىڭ سىلەرگە چاقماقنى قورقۇنچ ۋە ئۈمىد قىلىپ كۆرسىتىشى، بۇلۇتتىن يامغۇر ياغدۇرۇپ ئۇنىڭ بىلەن ئۆلگەن زېمىننى تىرىلدۈرۈشى اﷲ نىڭ (كامالى قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر، چۈشىنەلەيدىغان قەۋم ئۈچۈن بۇنىڭدا، شەك - شۈبھىسىزكى، نۇرغۇن ئىبرەتلەر بار
وَمِنْ ءَايَـٰتِهِۦٓ أَن تَقُومَ ٱلسَّمَآءُ وَٱلْأَرْضُ بِأَمْرِهِۦ ۚ ثُمَّ إِذَا دَعَاكُمْ دَعْوَةً مِّنَ ٱلْأَرْضِ إِذَآ أَنتُمْ تَخْرُجُونَ ﴿٢٥﴾
اﷲ نىڭ ئاسمان زېمىننى ئۆز ئەمرى بىلەن (مۇئەللەق) تۇرغۇزۇشى اﷲ نىڭ (كامالى قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر، ئاندىن اﷲ (قەبرىلەردە كۆمۈلۈپ ياتقان) سىلەرنى يەر ئاستىدىن چاقىرسا، دەرھال چىقىسىلەر
وَلَهُۥ مَن فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۖ كُلٌّ لَّهُۥ قَـٰنِتُونَ ﴿٢٦﴾
ئاسمانلاردىكى ۋە زېمىندىكى ھەممە اﷲ نىڭدۇر (يەنى اﷲ نىڭ مۈلكىدۇر، اﷲ نىڭ مەخلۇقاتىدۇر ۋە اﷲ نىڭ تەسەررۇپى ئاستىدىدۇر)، ھەممە اﷲ قا بويسۇنغۇچىدۇر
وَهُوَ ٱلَّذِى يَبْدَؤُا۟ ٱلْخَلْقَ ثُمَّ يُعِيدُهُۥ وَهُوَ أَهْوَنُ عَلَيْهِ ۚ وَلَهُ ٱلْمَثَلُ ٱلْأَعْلَىٰ فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۚ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٧﴾
مەخلۇقاتنى دەسلەپتە (يوقتىن) بار قىلىدىغان، ئاندىن ئۇنى (ئۆلگەندىن كېيىن) تىرىلىدۈرىدىغان اﷲ ئەنە شۇدۇر، ئۇ (يەنى مەخلۇقاتنى تىرىلدۈرۈش) اﷲ قا (ئۇنى دەسلەپتە يوقتىن بار قىلغانغا قارىغاندا) ئوڭايدۇر، ئاسمانلاردا ۋە زېمىندا ئەڭ ئالىي سۈپەت اﷲ قا خاستۇر، اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
ضَرَبَ لَكُم مَّثَلًا مِّنْ أَنفُسِكُمْ ۖ هَل لَّكُم مِّن مَّا مَلَكَتْ أَيْمَـٰنُكُم مِّن شُرَكَآءَ فِى مَا رَزَقْنَـٰكُمْ فَأَنتُمْ فِيهِ سَوَآءٌ تَخَافُونَهُمْ كَخِيفَتِكُمْ أَنفُسَكُمْ ۚ كَذَٰلِكَ نُفَصِّلُ ٱلْـَٔايَـٰتِ لِقَوْمٍ يَعْقِلُونَ ﴿٢٨﴾
اﷲ سىلەرگە ئۆزۈڭلاردىن بىر مىسال كەلتۈرىدۇ، بىز سىلەرگە رىزىق قىلىپ بەرگەن مال – مۈلۈككە قۇللىرىڭلارنىڭ شېرىك بولۇشىغا، ئۇ بارىدا ئۇلارنىڭ سىلەر بىلەن باراۋەر بولۇشىغا، ئۇلاردىن ئۆزلىرىڭلار (يەنى ئازاد كىشىلەر) دىن قورققاندەك قورقۇشقا رازىمۇسىلەر؟ (بۇنىڭغا رازى بولمىغان ئىكەنسىلەر، مەخلۇقاتنىڭ اﷲ قا شېرىك بولۇشىغا قانداق رازى بولدۇڭلار؟) چۈشىنىدىغان قەۋم ئۈچۈن ئايەتلەرنى مۇشۇنداق تەپسىلىي بايان قىلىمىز
بَلِ ٱتَّبَعَ ٱلَّذِينَ ظَلَمُوٓا۟ أَهْوَآءَهُم بِغَيْرِ عِلْمٍ ۖ فَمَن يَهْدِى مَنْ أَضَلَّ ٱللَّهُ ۖ وَمَا لَهُم مِّن نَّـٰصِرِينَ ﴿٢٩﴾
ئۇنداق ئەمەس (يەنى اﷲ قا شېرىك كەلتۈرۈشتە ئۇلارنىڭ ھېچقانداق ئۆزرىسى يوق)، (ئۆزلىرىگە) زۇلۇم قىلغانلار بىلىمسىزلىكتىن نەپسى خاھىشىغا ئەگەشتى. اﷲ گۇمراھ قىلغان ئادەمنى كىم ھىدايەت قىلالايدۇ، ئۇلارغا ھەرگىز ياردەم قىلغۇچى بولمايدۇ
فَأَقِمْ وَجْهَكَ لِلدِّينِ حَنِيفًا ۚ فِطْرَتَ ٱللَّهِ ٱلَّتِى فَطَرَ ٱلنَّاسَ عَلَيْهَا ۚ لَا تَبْدِيلَ لِخَلْقِ ٱللَّهِ ۚ ذَٰلِكَ ٱلدِّينُ ٱلْقَيِّمُ وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَعْلَمُونَ ﴿٣٠﴾
باتىل دىنلاردىن بۇرۇلۇپ ئىسلام دىنىغا يۈزلەنگىن، اﷲ نىڭ دىنىغا (ئەگەشكىنكى) اﷲ ئىنسانلارنى شۇ دىن بىلەن ياراتقان، اﷲ نىڭ ياراتقىنىدا ئۆزگىرىش بولمايدۇ، بۇ توغرا دىندۇر، لېكىن ئىنسانلارنىڭ تولىسى بىلمەيدۇ
۞ مُنِيبِينَ إِلَيْهِ وَٱتَّقُوهُ وَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَلَا تَكُونُوا۟ مِنَ ٱلْمُشْرِكِينَ ﴿٣١﴾
اﷲ قا تەۋبە بىلەن قايتىڭلار، اﷲ تىن قورقۇڭلار، نامازنى ئادا قىلىڭلار، مۇشرىكلاردىن بولماڭلار
مِنَ ٱلَّذِينَ فَرَّقُوا۟ دِينَهُمْ وَكَانُوا۟ شِيَعًا ۖ كُلُّ حِزْبٍۭ بِمَا لَدَيْهِمْ فَرِحُونَ ﴿٣٢﴾
سىلەر دىندا ئىختىلاپ قىلىشىپ، پىرقە - پىرقە بولۇپ، ھەرپىرقە ئۆز دىنى بىلەن خوشاللىنىدىغانلاردىن (بولماڭلار)
وَإِذَا مَسَّ ٱلنَّاسَ ضُرٌّ دَعَوْا۟ رَبَّهُم مُّنِيبِينَ إِلَيْهِ ثُمَّ إِذَآ أَذَاقَهُم مِّنْهُ رَحْمَةً إِذَا فَرِيقٌ مِّنْهُم بِرَبِّهِمْ يُشْرِكُونَ ﴿٣٣﴾
ئىنسانلارغا بىرەر زىيان - زەخمەت يەتكەن چاغدا، ئۇلار پەرۋەردىگارىغا ئىخلاس بىلەن ئىلتىجا قىلىدۇ، ئاندىن اﷲ ئۇلارغا رەھمىتىنى تېتىتقان چاغدا، ئۇلاردىن بىر پىرقە دەرھال پەرۋەردىگارىغا شېرىك كەلتۈرىدۇ
لِيَكْفُرُوا۟ بِمَآ ءَاتَيْنَـٰهُمْ ۚ فَتَمَتَّعُوا۟ فَسَوْفَ تَعْلَمُونَ ﴿٣٤﴾
ئۇلار بىزنىڭ بەرگەن نېمەتلىرىمىزگە نا شۈكۈرلۈك قىلىپ باقسۇن، (ئى مۇشرىكلار!) (دۇنيا تىرىكچىلىكىدە سىلەرگە بەرگەن نېمەتلىرىمىزدىن) بەھرىمەن بولۇڭلار، سىلەر (پانىي دۇنيانىڭ نېمىتىدىن بەھرىمەن بولغانلىقىڭلارنىڭ ئاقىۋىتىنى) كەلگۈسىدە بىلىسىلەر
أَمْ أَنزَلْنَا عَلَيْهِمْ سُلْطَـٰنًا فَهُوَ يَتَكَلَّمُ بِمَا كَانُوا۟ بِهِۦ يُشْرِكُونَ ﴿٣٥﴾
ياكى ئۇلارغا ئۇلارنىڭ بۇتلارغا چوقۇنىشىنىڭ توغرىلىقىنى سۆزلەيدىغان بىرەر مۇقەددەس كىتاب-ۋەھىي نازىل قىلدۇقمۇ؟ (مۇنداق بولغىنى يوق)
وَإِذَآ أَذَقْنَا ٱلنَّاسَ رَحْمَةً فَرِحُوا۟ بِهَا ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌۢ بِمَا قَدَّمَتْ أَيْدِيهِمْ إِذَا هُمْ يَقْنَطُونَ ﴿٣٦﴾
ئىنسانلارغا رەھمەتنى تېتىتساق، ئۇلار بۇنىڭدىن خۇشاللىنىپ كېتىدۇ، ئەگەر ئۇلارغا قىلمىشلىرى تۈپەيلىدىن بىرەر كۆڭۈلسىزلىك يەتسە، دەرھال ئۈمىدسىزلىنىپ كېتىدۇ
أَوَلَمْ يَرَوْا۟ أَنَّ ٱللَّهَ يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ وَيَقْدِرُ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٣٧﴾
ئۇلار بىلمەمدۇكى، اﷲ خالىغان ئادەمنىڭ رىزقىنى كەڭ قىلىدۇ، (خالىغان ئادەمنىڭ رىزقىنى) تار قىلىدۇ، ئىشەنچلىك ھالدا بۇنىڭدا (رىزىق بەرگەن اﷲ نىڭ ھېكمىتىگە) ئىشىنىدىغان قەۋم ئۈچۈن (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) نۇرغۇن ئالامەتلەر بار
فَـَٔاتِ ذَا ٱلْقُرْبَىٰ حَقَّهُۥ وَٱلْمِسْكِينَ وَٱبْنَ ٱلسَّبِيلِ ۚ ذَٰلِكَ خَيْرٌ لِّلَّذِينَ يُرِيدُونَ وَجْهَ ٱللَّهِ ۖ وَأُو۟لَـٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿٣٨﴾
خىش - ئەقرىبالارغا، مىسكىنلەرگە، ئىبن سەبىللەرگە (سىلە - رەھىم ۋە ياخشىلىقتىن تېگىشلىك) ھەققىنى بەرگىن، اﷲ نىڭ رازىلىقىنى كۆزلەيدىغانلار ئۈچۈن مۇنداق قىلىش ياخشىدۇر، ئەنە شۇلار مەقسەتكە ئېرىشكۈچىلەردۇر
وَمَآ ءَاتَيْتُم مِّن رِّبًا لِّيَرْبُوَا۟ فِىٓ أَمْوَٰلِ ٱلنَّاسِ فَلَا يَرْبُوا۟ عِندَ ٱللَّهِ ۖ وَمَآ ءَاتَيْتُم مِّن زَكَوٰةٍ تُرِيدُونَ وَجْهَ ٱللَّهِ فَأُو۟لَـٰٓئِكَ هُمُ ٱلْمُضْعِفُونَ ﴿٣٩﴾
سىلەر كىشىلەرنىڭ پۇل - مېلى ئىچىدە ئۆستۈرۈش ئۈچۈن بىرەر پۇل - مال بەرسەڭلار، اﷲ نىڭ دەرگاھىدا ئۇ ئۆسمەيدۇ (يەنى كىشىلەر بەرگىنىمدىن جىقراق قايتۇرسۇن دېگەن نىيەت بىلەن بېرىلگەن سوغىنىڭ اﷲ نىڭ دەرگاھىدا ساۋابى بولمايدۇ)، اﷲ نىڭ رازىلىقىنى كۆزلەپ بەرگەن سەدىقەڭلار (ياكى ئېھسانىڭلار) غا ھەسسىلەپ ساۋاب بېرىلىدۇ
ٱللَّهُ ٱلَّذِى خَلَقَكُمْ ثُمَّ رَزَقَكُمْ ثُمَّ يُمِيتُكُمْ ثُمَّ يُحْيِيكُمْ ۖ هَلْ مِن شُرَكَآئِكُم مَّن يَفْعَلُ مِن ذَٰلِكُم مِّن شَىْءٍ ۚ سُبْحَـٰنَهُۥ وَتَعَـٰلَىٰ عَمَّا يُشْرِكُونَ ﴿٤٠﴾
اﷲ سىلەرنى خەلق ئەتتى، ئاندىن سىلەرگە رىزىق بەردى. ئاندىن سىلەرنى قەبزى روھ قىلىدۇ، ئاندىن سىلەرنى تىرىلدۈرىدۇ، شېرىك كەلتۈرگەنلىرىڭلاردىن قايسىسى بۇ ئىشلارنىڭ قايسىبىرىنى قىلالايدۇ، اﷲ پاكتۇر، ئۇلارنىڭ شېرىك كەلتۈرگەن نەرسىلىرىدىن ئالىيدۇر
ظَهَرَ ٱلْفَسَادُ فِى ٱلْبَرِّ وَٱلْبَحْرِ بِمَا كَسَبَتْ أَيْدِى ٱلنَّاسِ لِيُذِيقَهُم بَعْضَ ٱلَّذِى عَمِلُوا۟ لَعَلَّهُمْ يَرْجِعُونَ ﴿٤١﴾
ئىنسانلارنىڭ قىلغان گۇناھلىرى تۈپەيلىدىن، قۇرۇقلۇقتا ۋە دېڭىزدا ئاپەت يۈزبەردى، اﷲ ئۇلارنى تەۋبە قىلسۇن دەپ قىلمىشلىرىنىڭ بىر قىسمىنىڭ (جازاسىنى) ئۇلارغا تېتىتتى
قُلْ سِيرُوا۟ فِى ٱلْأَرْضِ فَٱنظُرُوا۟ كَيْفَ كَانَ عَـٰقِبَةُ ٱلَّذِينَ مِن قَبْلُ ۚ كَانَ أَكْثَرُهُم مُّشْرِكِينَ ﴿٤٢﴾
(ئى مۇھەممەد! مۇشرىكلارغا) ئېيتقىنكى، يەر يۈزىدە سەير قىلىڭلار، ئۆتكەنكىلەرنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىغا قاراڭلار (ئۆزلىرىگە زۇلۇم قىلغانلارنىڭ جايلىرىغا قاراڭلاركى، ئۇلارنىڭ خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلغانلىقلىرىنىڭ ئاقىۋىتى قانداق بولغان، اﷲ تائالا ئۇلارنىڭ دىيارلىرىنى خاراپ قىلىپ، ئۇلارنى ئىبرەت ئالغۇچىلار ئۈچۈن ئىبرەت قىلدى)، ئۇلارنىڭ تولىسى مۇشرىك ئىدى
فَأَقِمْ وَجْهَكَ لِلدِّينِ ٱلْقَيِّمِ مِن قَبْلِ أَن يَأْتِىَ يَوْمٌ لَّا مَرَدَّ لَهُۥ مِنَ ٱللَّهِ ۖ يَوْمَئِذٍ يَصَّدَّعُونَ ﴿٤٣﴾
اﷲ تەرىپىدىن كېلىدىغان، قارشى تۇرغىلى بولمايدىغان كۈن (يەنى قىيامەت) كېلىشتىن ئىلگىرى توغرا دىنغا يۈزلەنگىن، بۇ كۈندە ئۇلار (نىڭ بىر پىرقىسى مەڭگۈلۈك باغلاركە، بىر پىرقىسى مەڭگۈلۈك ئوتگە) ئايرىلىدۇ
مَن كَفَرَ فَعَلَيْهِ كُفْرُهُۥ ۖ وَمَنْ عَمِلَ صَـٰلِحًا فَلِأَنفُسِهِمْ يَمْهَدُونَ ﴿٤٤﴾
كىمكى ھەقىقەتنى رەت قىلغۇچى بولسا (دوزاختا ئەبەدىي قېلىپ) كۇفرىنىڭ جازاسىنى ئۆزى تارتىدۇ، كىملەركى ياخشى ئەمەل قىلسا، ئۆزلىرى ئۈچۈن (ئاخىرەتتە) جاي راستلايدۇ
لِيَجْزِىَ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ مِن فَضْلِهِۦٓ ۚ إِنَّهُۥ لَا يُحِبُّ ٱلْكَـٰفِرِينَ ﴿٤٥﴾
اﷲ ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنى مەرھەمىتىدىن مۇكاپاتلايدۇ، اﷲ ئىشەنچلىك ھالدا ھەقىقەتنى رەت قىلغۇچىلارنى دوست تۇتمايدۇ
وَمِنْ ءَايَـٰتِهِۦٓ أَن يُرْسِلَ ٱلرِّيَاحَ مُبَشِّرَٰتٍ وَلِيُذِيقَكُم مِّن رَّحْمَتِهِۦ وَلِتَجْرِىَ ٱلْفُلْكُ بِأَمْرِهِۦ وَلِتَبْتَغُوا۟ مِن فَضْلِهِۦ وَلَعَلَّكُمْ تَشْكُرُونَ ﴿٤٦﴾
اﷲ نىڭ سىلەرگە ئۆزىنىڭ رەھمىتىنى تېتىتىش ئۈچۈن، ئەمرى بويىچە كېمىلەرنى دېڭىزلاردا ماڭدۇرۇش ئۈچۈن، ئۆزىنىڭ مەرھەمىتىدىن تىلىشىڭلار ئۈچۈن، شۈكۈر قىلىشىڭلار ئۈچۈن، شاماللارنى يامغۇر بىلەن خۇش خەۋەر بەرگۈچى قىلىپ ئەۋەتىشى اﷲ نىڭ (كامالى قۇدرىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر
وَلَقَدْ أَرْسَلْنَا مِن قَبْلِكَ رُسُلًا إِلَىٰ قَوْمِهِمْ فَجَآءُوهُم بِٱلْبَيِّنَـٰتِ فَٱنتَقَمْنَا مِنَ ٱلَّذِينَ أَجْرَمُوا۟ ۖ وَكَانَ حَقًّا عَلَيْنَا نَصْرُ ٱلْمُؤْمِنِينَ ﴿٤٧﴾
(ئى مۇھەممەد!) شۈبھىسىزكى، سەندىن ئىلگىرى نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرىنى قەۋمىگە (خۇدانىڭ ئەلچىسى قىلپ) ئەۋەتتۇق، ئۇ خۇدانىڭ ئەلچى-رەسۇللىرى ئۇلارغا (ئۆزلىرىنىڭ راست خۇدانىڭ ئەلچىسىلىكلىرىنى ئىسپاتلايدىغان) نۇرغۇن روشەن مۆجىزىلەرنى ئېلىپ كەلدى (ئۇلار مۆجىزىلەرنى ئىنكار قىلىپ ئاسىيلىق قىلدى)، ئاسىيلىق قىلغانلارنى جازالىدۇق، مۆمىنلەرگە ياردەم قىلىش بىزگە تېگىشلىك بولدى
ٱللَّهُ ٱلَّذِى يُرْسِلُ ٱلرِّيَـٰحَ فَتُثِيرُ سَحَابًا فَيَبْسُطُهُۥ فِى ٱلسَّمَآءِ كَيْفَ يَشَآءُ وَيَجْعَلُهُۥ كِسَفًا فَتَرَى ٱلْوَدْقَ يَخْرُجُ مِنْ خِلَـٰلِهِۦ ۖ فَإِذَآ أَصَابَ بِهِۦ مَن يَشَآءُ مِنْ عِبَادِهِۦٓ إِذَا هُمْ يَسْتَبْشِرُونَ ﴿٤٨﴾
اﷲ شاماللارنى ئەۋەتىپ بۇلۇتنى قوزغايدۇ، ئاندىن ئۇنى ئاسماندا خالىغىنى بويىچە تارقىتىدۇ، (گاھى) ئۇنى تارقاق پارچىلارغا بۆلىدۇ، شۇنىڭ بىلەن يامغۇرنىڭ بۇلۇتنىڭ ئارىسىدىن چۈشۈۋاتقانلىقىنى كۆرىسەن، اﷲ يامغۇرنى خالىغان بەندىلىرىنىڭ ئۈستىگە ياغدۇرغان چاغدا، ئۇلار دەرھال خۇشال بولۇپ كېتىدۇ
وَإِن كَانُوا۟ مِن قَبْلِ أَن يُنَزَّلَ عَلَيْهِم مِّن قَبْلِهِۦ لَمُبْلِسِينَ ﴿٤٩﴾
ھالبۇكى، ئۇلار يامغۇر يېغىشتىن بۇرۇن ئىشەنچلىك ھالدا ئۈمىدسىزلىنىپ كەتكەن ئىدى
فَٱنظُرْ إِلَىٰٓ ءَاثَـٰرِ رَحْمَتِ ٱللَّهِ كَيْفَ يُحْىِ ٱلْأَرْضَ بَعْدَ مَوْتِهَآ ۚ إِنَّ ذَٰلِكَ لَمُحْىِ ٱلْمَوْتَىٰ ۖ وَهُوَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٥٠﴾
اﷲ نىڭ رەھمىتىنىڭ نەتىجىلىرىگە قارىغىنكى، ئۇ زېمىننى ئۆلگەندىن كېيىن قانداق تىرىلدۈرىدۇ، شۈبھىسىزكى، اﷲ ئىشەنچلىك ھالدا ئۆلۈكلەرنى تىرىلدۈرگۈچىدۇر، اﷲ ھەممە نەرسىگە قادىردۇر
وَلَئِنْ أَرْسَلْنَا رِيحًا فَرَأَوْهُ مُصْفَرًّا لَّظَلُّوا۟ مِنۢ بَعْدِهِۦ يَكْفُرُونَ ﴿٥١﴾
ئەگەر بىز (زىرائەتكە زىيانلىق) شامالنى ئەۋەتسەك، ئۇلار (شامالنىڭ تەسىرىدىن) زىرائەتنىڭ سارغىيىپ كەتكەنلىكىنى كۆرسە، شۇنىڭدىن كېيىن ئۇلار (اﷲ نىڭ ئىلگىرىكى نېمەتلىرىگە) كۇفرىلىق قىلىدۇ
فَإِنَّكَ لَا تُسْمِعُ ٱلْمَوْتَىٰ وَلَا تُسْمِعُ ٱلصُّمَّ ٱلدُّعَآءَ إِذَا وَلَّوْا۟ مُدْبِرِينَ ﴿٥٢﴾
شۈبھىسىزكى، سەن ئۆلۈكلەرگە (سۆزۈڭنى) ئاڭلىتالمايسەن، ئارقىسىغا قاراپ يۈز ئۆرۈگەن گاسلارغىمۇ چاقىرىقىڭنى ئاڭلىتالمايسەن
وَمَآ أَنتَ بِهَـٰدِ ٱلْعُمْىِ عَن ضَلَـٰلَتِهِمْ ۖ إِن تُسْمِعُ إِلَّا مَن يُؤْمِنُ بِـَٔايَـٰتِنَا فَهُم مُّسْلِمُونَ ﴿٥٣﴾
كورلارنىمۇ گۇمراھلىقتىن قۇتقۇزۇپ توغرا يولغا سالالمايسەن، سەن پەقەت بىزنىڭ ئايەتلىرىمىزگە ئىشىنىدىغانلارغىلا (سۆزۈڭنى) ئاڭلىتالايسەن، ئۇلار مۇسۇلمانلاردۇر
۞ ٱللَّهُ ٱلَّذِى خَلَقَكُم مِّن ضَعْفٍ ثُمَّ جَعَلَ مِنۢ بَعْدِ ضَعْفٍ قُوَّةً ثُمَّ جَعَلَ مِنۢ بَعْدِ قُوَّةٍ ضَعْفًا وَشَيْبَةً ۚ يَخْلُقُ مَا يَشَآءُ ۖ وَهُوَ ٱلْعَلِيمُ ٱلْقَدِيرُ ﴿٥٤﴾
اﷲ سىلەرنى ئاجىز ياراتتى، ئاندىن سىلەرنى ئاجىزلىقتىن كۈچلۈك قىلدى، ئاندىن (سىلەرنى) كۈچلۈكلۈكتىن ئاجىز قىلدى ۋە (سىلەرگە) قېرىلىقنى يەتكۈزدى، اﷲ نېمىنى خالىسا شۇنى يارىتىدۇ، ئۇ ھەممىنى بىلگۈچىدۇر، ھەممىگە قادىردۇر
وَيَوْمَ تَقُومُ ٱلسَّاعَةُ يُقْسِمُ ٱلْمُجْرِمُونَ مَا لَبِثُوا۟ غَيْرَ سَاعَةٍ ۚ كَذَٰلِكَ كَانُوا۟ يُؤْفَكُونَ ﴿٥٥﴾
قىيامەت قايىم بولغان كۈندە مۇشرىكلار (دۇنيادا) پەقەت ئازغىنا ۋاقىت تۇرغانلىقىغا قەسەم قىلىدۇ، ئۇلار دۇنيادىكى چاغلىرىدىمۇ مۇشۇنداق يالغانچىلىق قىلاتتى
وَقَالَ ٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ وَٱلْإِيمَـٰنَ لَقَدْ لَبِثْتُمْ فِى كِتَـٰبِ ٱللَّهِ إِلَىٰ يَوْمِ ٱلْبَعْثِ ۖ فَهَـٰذَا يَوْمُ ٱلْبَعْثِ وَلَـٰكِنَّكُمْ كُنتُمْ لَا تَعْلَمُونَ ﴿٥٦﴾
ئىلمى بار، ئىمانى بار ئادەملەر ئېيتتى: «سىلەر اﷲ نىڭ تەقدىرى بويىچە (دۇنيادا ۋە قەبرىلەردە) قىيامەت كۈنىگىچە قالدىڭلار، مانا بۇ قىيامەت كۈنىدۇر، لېكىن سىلەر بىلمەيتتىڭلار (يەنى قىيامەتنىڭ ھەقلىقىغا ئىشەنمەيتتىڭلار)»
فَيَوْمَئِذٍ لَّا يَنفَعُ ٱلَّذِينَ ظَلَمُوا۟ مَعْذِرَتُهُمْ وَلَا هُمْ يُسْتَعْتَبُونَ ﴿٥٧﴾
بۇ كۈندە زالىملارنىڭ (ئېيتقان) ئۆزرىسىنىڭ پايدىسى بولمايدۇ، ئۇلاردىن ئۇلارنىڭ (تەۋبە قىلىشى يا ئىتائەت قىلىشى بىلەن) اﷲ نى رازى قىلىشمۇ تەلەپ قىلىنمايدۇ
وَلَقَدْ ضَرَبْنَا لِلنَّاسِ فِى هَـٰذَا ٱلْقُرْءَانِ مِن كُلِّ مَثَلٍ ۚ وَلَئِن جِئْتَهُم بِـَٔايَةٍ لَّيَقُولَنَّ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ أَنتُمْ إِلَّا مُبْطِلُونَ ﴿٥٨﴾
شەك - شۈبھىسىزكى، بۇ قۇرئاندا كىشىلەرگە تۈرلۈك مىساللارنى بايان قىلدۇق، ئەگەر سەن ئۇلارغا (قۇرئان ئايەتلىرىدىن) بىرەر ئايەتنى كەلتۈرسەڭ، ھەقىقەتنى رەت قىلغۇچىلار (دىللىرى قەساۋەتلىشىپ كەتكەنلىكتىن) چوقۇم ئېيتىدۇكى، «سىلەر پەقەت يالغانچىلار سىلەر»
كَذَٰلِكَ يَطْبَعُ ٱللَّهُ عَلَىٰ قُلُوبِ ٱلَّذِينَ لَا يَعْلَمُونَ ﴿٥٩﴾
(اﷲ نىڭ بىرلىكىنى) بىلمەيدىغان (كۇففار) لارنىڭ دىللىرىنى مۇشۇنداق پېچەتلەيمىز
فَٱصْبِرْ إِنَّ وَعْدَ ٱللَّهِ حَقٌّ ۖ وَلَا يَسْتَخِفَّنَّكَ ٱلَّذِينَ لَا يُوقِنُونَ ﴿٦٠﴾
(ئى مۇھەممەد!) (مۇشرىكلارنىڭ ئەزىيەتلىرىگە) سەۋر قىلغىن، اﷲ نىڭ (ساڭا نۇسرەت ئاتا قىلىش ۋە دىنىڭنى ئۈستۈنلۈككە ئىگە قىلىش) ۋەدىسى ئىشەنچلىك ھالدا ھەقتۇر، (ئاخىرەتكە) ئىشەنمەيدىغانلار سېنى دىققەت قىلىپ قويمىسۇن
