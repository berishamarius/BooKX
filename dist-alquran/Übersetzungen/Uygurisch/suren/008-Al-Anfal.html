بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
يَسْـَٔلُونَكَ عَنِ ٱلْأَنفَالِ ۖ قُلِ ٱلْأَنفَالُ لِلَّهِ وَٱلرَّسُولِ ۖ فَٱتَّقُوا۟ ٱللَّهَ وَأَصْلِحُوا۟ ذَاتَ بَيْنِكُمْ ۖ وَأَطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥٓ إِن كُنتُم مُّؤْمِنِينَ ﴿١﴾
(ئى مۇھەممەد! ساھابىلىرىڭ) غەنىيمەتلەر (نى قانداق تەقسىم قىلىشىڭ) توغرۇلۇق سەندىن سورىشىدۇ، (ئۇلارغا) ئېيتقىنكى، «غەنىيمەتلەر (توغرىسىدا ھۆكۈم چىقىرىش) اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە خاستۇر، اﷲ تىن قورقۇڭلار، (ئىختىلاپ قىلىشماي ئىتتىپاق بولۇپ) ئاراڭلارنى تۈزەڭلار، ئەگەر سىلەر مۆمىن بولساڭلار، اﷲ قا ۋە اﷲ نىڭ پەيغەمبىرىگە ئىتائەت قىلىڭلار»
إِنَّمَا ٱلْمُؤْمِنُونَ ٱلَّذِينَ إِذَا ذُكِرَ ٱللَّهُ وَجِلَتْ قُلُوبُهُمْ وَإِذَا تُلِيَتْ عَلَيْهِمْ ءَايَـٰتُهُۥ زَادَتْهُمْ إِيمَـٰنًا وَعَلَىٰ رَبِّهِمْ يَتَوَكَّلُونَ ﴿٢﴾
پەقەت اﷲ ياد ئېتىلسە دىللىرىدا قورقۇنچ پەيدا بولىدىغان، اﷲ نىڭ ئايەتلىرى تىلاۋەت قىلىنسا ئىمانى كۈچىيىدىغان، پەرۋەردىگارىغا تەۋەككۈل قىلىدىغان كىشىلەرلا (كامىل) مۆمىنلەردۇر
ٱلَّذِينَ يُقِيمُونَ ٱلصَّلَوٰةَ وَمِمَّا رَزَقْنَـٰهُمْ يُنفِقُونَ ﴿٣﴾
ئۇلار (مۇكەممەل رەۋىشتە) ناماز ئوقۇيدۇ، بىز ئۇلارغا رىزىق قىلىپ بەرگەن پۇل - مالدىن (خۇدا يولىدا) سەرپ قىلىدۇ
أُو۟لَـٰٓئِكَ هُمُ ٱلْمُؤْمِنُونَ حَقًّا ۚ لَّهُمْ دَرَجَـٰتٌ عِندَ رَبِّهِمْ وَمَغْفِرَةٌ وَرِزْقٌ كَرِيمٌ ﴿٤﴾
ئەنە شۇلار ھەقىقىي مۆمىنلەردۇر، ئۇلار پەرۋەردىگارىنىڭ دەرگاھىدا يۇقىرى مەرتىۋىلەرگە، مەغپىرەتكە ۋە ئېسىل رىزىققا (يەنى مەڭگۈلۈك باغلارنىڭ تۈگىمەس نېمەتلىرىگە) ئېرىشىدۇ
كَمَآ أَخْرَجَكَ رَبُّكَ مِنۢ بَيْتِكَ بِٱلْحَقِّ وَإِنَّ فَرِيقًا مِّنَ ٱلْمُؤْمِنِينَ لَكَـٰرِهُونَ ﴿٥﴾
(غەنىيمەت تەقسىماتى توغرىسىدىكى ھۆكۈمنىڭ اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە خاس قىلىنغانلىقىنى ئۇلارنىڭ ياقتۇرمىغانلىقى) پەرۋەردىگارىڭنىڭ سېنى ھەقلىق رەۋىشتە ئۆيۈڭدىن (يەنى ئۆيۈڭ جايلاشقان مەدىنىدىن ئۇرۇشقا) چىقارغانلىقىنى (ئۇلارنىڭ ياقتۇرمىغانلىقىغا) ئوخشايدۇ. مۆمىنلەردىن بىر گۇرۇھ ئادەم (دۈشمەنگە قارشى ئۇرۇشقا چىقىشنى) ئەلۋەتتە يامان كۆرىدۇ
يُجَـٰدِلُونَكَ فِى ٱلْحَقِّ بَعْدَ مَا تَبَيَّنَ كَأَنَّمَا يُسَاقُونَ إِلَى ٱلْمَوْتِ وَهُمْ يَنظُرُونَ ﴿٦﴾
ئۇلار گويا (ئالامەتلىرى) كۆرۈنۈپ تۇرۇۋاتقان ئۆلۈمگە ھەيدىلىۋاتقاندەك، ھەقىقەت ئايدىڭلاشقاندىن كېيىن، ھەقىقەت (يەنى ئۇرۇشقا چىقىش مەسىلىسى) ئۈستىدە سەن بىلەن مۇنازىرىلىشىدۇ
وَإِذْ يَعِدُكُمُ ٱللَّهُ إِحْدَى ٱلطَّآئِفَتَيْنِ أَنَّهَا لَكُمْ وَتَوَدُّونَ أَنَّ غَيْرَ ذَاتِ ٱلشَّوْكَةِ تَكُونُ لَكُمْ وَيُرِيدُ ٱللَّهُ أَن يُحِقَّ ٱلْحَقَّ بِكَلِمَـٰتِهِۦ وَيَقْطَعَ دَابِرَ ٱلْكَـٰفِرِينَ ﴿٧﴾
ئۆز ۋاقتىدا اﷲ ئىككى گۇرۇھ (بىرى مۇشرىكلار كارۋىنى، يەنى بىرى مۇشرىكلار قوشۇنى) دىن بىرىنىڭ قولۇڭلارغا كەلتۈرۈلۈشىنى ۋەدە قىلدى، سىلەر قورالسىز گۇرۇھنىڭ (يەنى كارۋاننىڭ) قولۇڭلارغا كەلتۈرۈلۈشىنى ياقتۇردۇڭلار. اﷲ ئۆز سۆزلىرى ئارقىلىق ھەقنى ھەق قىلىشنى (يەنى ئىسلام دىنىنى ئۈستۈن قىلىشنى)، ھەقىقەتنى رەت قىلغۇچىلارنىڭ يىلتىزىنى قۇرۇتۇشنى خالايدۇ
لِيُحِقَّ ٱلْحَقَّ وَيُبْطِلَ ٱلْبَـٰطِلَ وَلَوْ كَرِهَ ٱلْمُجْرِمُونَ ﴿٨﴾
اﷲ گۇناھكار ئادەملەرنىڭ ياقتۇرمىغىنىغا قارىماي، ھەقنى ھەق (يەنى ئىسلامنى ئاشكارا) قىلىدۇ، باتىل (يەنى كۇفرى) نى بەربات قىلىدۇ
إِذْ تَسْتَغِيثُونَ رَبَّكُمْ فَٱسْتَجَابَ لَكُمْ أَنِّى مُمِدُّكُم بِأَلْفٍ مِّنَ ٱلْمَلَـٰٓئِكَةِ مُرْدِفِينَ ﴿٩﴾
ئۆز ۋاقتىدا (دۇئا قىلىپ) پەرۋەردىگارىڭلاردىن ياردەم تىلىدىڭلار، اﷲ: «سىلەرگە ئاقىمۇ - ئارقا (چۈشىدىغان) مىڭ پەرىشتە بىلەن ياردەم بېرىمەن» دەپ دۇئايىڭلارنى ئىجابەت قىلدى
وَمَا جَعَلَهُ ٱللَّهُ إِلَّا بُشْرَىٰ وَلِتَطْمَئِنَّ بِهِۦ قُلُوبُكُمْ ۚ وَمَا ٱلنَّصْرُ إِلَّا مِنْ عِندِ ٱللَّهِ ۚ إِنَّ ٱللَّهَ عَزِيزٌ حَكِيمٌ ﴿١٠﴾
اﷲ سىلەرگە پەقەت (غەلىبە قىلىدىغانلىقىڭلارغا) بېشارەت بېرىش ۋە كۆڭلۈڭلارنى ئارام تاپقۇزۇش ئۈچۈنلا ياردەم بەردى. ياردەم پەقەت اﷲ تەرىپىدىنلا كېلىدۇ. اﷲ ئىشەنچلىك ھالدا غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
إِذْ يُغَشِّيكُمُ ٱلنُّعَاسَ أَمَنَةً مِّنْهُ وَيُنَزِّلُ عَلَيْكُم مِّنَ ٱلسَّمَآءِ مَآءً لِّيُطَهِّرَكُم بِهِۦ وَيُذْهِبَ عَنكُمْ رِجْزَ ٱلشَّيْطَـٰنِ وَلِيَرْبِطَ عَلَىٰ قُلُوبِكُمْ وَيُثَبِّتَ بِهِ ٱلْأَقْدَامَ ﴿١١﴾
ئۆز ۋاقتىدا اﷲ تىنچلاندۇرۇش يۈزىسىدىن سىلەرگە ئۇيقۇ بېغىشلىدى؛ (تاھارەت ئېلىپ، غۇسلى قىلىپ) پاڭ بولۇشۇڭلار ئۈچۈن، سىلەردىن شەيتان (ئازدۇرغۇچى)نىڭ ۋەسۋەسىسىنى كەتكۈزۈش ئۈچۈن، (اﷲ نىڭ ياردىمىگە ئىشەنچ قىلىپ) كۆڭلۈڭلارنىڭ توق تۇرۇشى ئۈچۈن، قەدىمىڭلارنىڭ (قۇمغا پېتىپ كەتمەي) مەزمۇت تۇرۇشى ئۈچۈن، اﷲ سىلەرگە بۇلۇتتىن يامغۇر ياغدۇرۇپ بەردى
إِذْ يُوحِى رَبُّكَ إِلَى ٱلْمَلَـٰٓئِكَةِ أَنِّى مَعَكُمْ فَثَبِّتُوا۟ ٱلَّذِينَ ءَامَنُوا۟ ۚ سَأُلْقِى فِى قُلُوبِ ٱلَّذِينَ كَفَرُوا۟ ٱلرُّعْبَ فَٱضْرِبُوا۟ فَوْقَ ٱلْأَعْنَاقِ وَٱضْرِبُوا۟ مِنْهُمْ كُلَّ بَنَانٍ ﴿١٢﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭ خۇدانىڭ ئەلچى پەرىشتىلىرىگە: «مەن سىلەر بىلەن بىللە، مۆمىنلەرنى (ئۇرۇش مەيدانىدا) ساباتلىق قىلىڭلار، ھەقىقەتنى رەت قىلغۇچىلارنىڭ دىللىرىغا قورقۇنچ سالىمەن» دەپ ۋەھيى قىلدى. (قىلىچ بىلەن ھەقىقەتنى رەت قىلغۇچىلارنىڭ) گەدەنلىرىگە چېپىڭلار (يەنى باشلىرىنى كېسىڭلار)، ئۇلارنىڭ بارماقلىرىغا (ھەممە ئەزاسىغا چېپىڭلار
ذَٰلِكَ بِأَنَّهُمْ شَآقُّوا۟ ٱللَّهَ وَرَسُولَهُۥ ۚ وَمَن يُشَاقِقِ ٱللَّهَ وَرَسُولَهُۥ فَإِنَّ ٱللَّهَ شَدِيدُ ٱلْعِقَابِ ﴿١٣﴾
بۇ شۇنىڭ ئۈچۈنكى، ئۇلار اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە قارشىلىق قىلدى. كىمكى اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە قارشىلىق قىلىدىكەن، اﷲ ئۇنى قاتتىق جازالايدۇ
ذَٰلِكُمْ فَذُوقُوهُ وَأَنَّ لِلْكَـٰفِرِينَ عَذَابَ ٱلنَّارِ ﴿١٤﴾
(ئى ھەقىقەتنى رەت قىلغۇچىلار جامائەسى! بۇ دۇنيادا) مۇشۇنداق جازانى تېتىڭلار. ھەقىقەتنى رەت قىلغۇچىلار (ئاخىرەتتە) دوزاخ ئازابىغا دۇچار بولىدۇ
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا لَقِيتُمُ ٱلَّذِينَ كَفَرُوا۟ زَحْفًا فَلَا تُوَلُّوهُمُ ٱلْأَدْبَارَ ﴿١٥﴾
ئى مۆمىنلەر! ھەقىقەتنى رەت قىلغۇچىلارنىڭ ھۇجۇمىغا دۇچ كەلگەن چېغىڭلاردا، ئۇلارغا ئارقاڭلارنى قىلماڭلار (يەنى قاچماڭلار)
وَمَن يُوَلِّهِمْ يَوْمَئِذٍ دُبُرَهُۥٓ إِلَّا مُتَحَرِّفًا لِّقِتَالٍ أَوْ مُتَحَيِّزًا إِلَىٰ فِئَةٍ فَقَدْ بَآءَ بِغَضَبٍ مِّنَ ٱللَّهِ وَمَأْوَىٰهُ جَهَنَّمُ ۖ وَبِئْسَ ٱلْمَصِيرُ ﴿١٦﴾
كىمكى قايتا ئۇرۇشۇش ياكى (ياردەم تىلەش) ئۈچۈن مۇسۇلمانلار جامائەسىگە قوشۇلۇش مەقسىتىدە بىر تەرەپكە يۆتكىلىش قىلماستىن، بەلكى دۈشمەنگە ئارقىسىنى قىلىدىكەن (يەنى قاچىدىكەن)، ئۇ ئىشەنچلىك ھالدا اﷲ نىڭ غەزىپىگە ئۇچرايدۇ، ئۇنىڭ جايى دوزاخ بولىدۇ. دوزاخ نېمىدېگەن يامان جاي!
فَلَمْ تَقْتُلُوهُمْ وَلَـٰكِنَّ ٱللَّهَ قَتَلَهُمْ ۚ وَمَا رَمَيْتَ إِذْ رَمَيْتَ وَلَـٰكِنَّ ٱللَّهَ رَمَىٰ ۚ وَلِيُبْلِىَ ٱلْمُؤْمِنِينَ مِنْهُ بَلَآءً حَسَنًا ۚ إِنَّ ٱللَّهَ سَمِيعٌ عَلِيمٌ ﴿١٧﴾
(ئى مۇسۇلمانلار! بەدرىدە) ئۇلارنى (يەنى مۇشرىكلارنى) سىلەر (ئۆز كۈچۈڭلار بىلەن) ئۆلتۈرگىنىڭلار يوق، بەلكى ئەمەلدە ئۇلارنى (سىلەرگە ياردەم بېرىپ، ئۇلارنىڭ دىللىرىدا قورقۇنچ پەيدا قىلىش بىلەن) اﷲ ئۆلتۈردى، (ئى مۇھەممەد بىر سىقىم توپىنى مۇشرىكلارغا) ئاتقىنىڭدا سەن ئاتمىدىڭ، بەلكى ئەمەلدە ئۇنى (مۇشرىكلارنىڭ كۆزلىرىگە) اﷲ ئاتتى. (اﷲ نىڭ مۇنداق قىلىشى) مۆمىنلەرگە (ساۋاب، غەلىبە، غەنىيمەتلەردىن ئىبارەت) چىرايلىق ئىنئاملارنى ئاتا قىلىش ئۈچۈن ئىدى. اﷲ ئىشەنچلىك ھالدا (ئۇلارنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (نىيەتلىرىنى، ئەھۋاللىرىنى) بىلىپ تۇرغۇچىدۇر
ذَٰلِكُمْ وَأَنَّ ٱللَّهَ مُوهِنُ كَيْدِ ٱلْكَـٰفِرِينَ ﴿١٨﴾
بۇ (اﷲ نىڭ مۆمىنلەرگە بەرگەن ئىنئامىدۇر). اﷲ ھەقىقەتنى رەت قىلغۇچىلارنىڭ ھىيلىسىنى چوقۇم مەغلۇپ قىلغۇچىدۇر
إِن تَسْتَفْتِحُوا۟ فَقَدْ جَآءَكُمُ ٱلْفَتْحُ ۖ وَإِن تَنتَهُوا۟ فَهُوَ خَيْرٌ لَّكُمْ ۖ وَإِن تَعُودُوا۟ نَعُدْ وَلَن تُغْنِىَ عَنكُمْ فِئَتُكُمْ شَيْـًٔا وَلَوْ كَثُرَتْ وَأَنَّ ٱللَّهَ مَعَ ٱلْمُؤْمِنِينَ ﴿١٩﴾
(ئى كۇففارلار جامائەسى!) ئەگەر سىلەر غەلىبىنى تىلىسەڭلار، غەلىبە ئاللىقاچان سىلەرگە كەلدى (يەنى ئۇ سىلەرنىڭ زىيىنىڭلارغا كەلدى). ئەگەر (خۇدانىڭ ئەلچىسى بىلەن دۈشمەنلىشىشتىن) يانساڭلار، بۇ سىلەر ئۈچۈن ياخشىدۇر، ئەگەر ئۇنىڭ بىلەن يەنە ئۇرۇشساڭلار، بىز ئۇنىڭغا يەنە ياردەم بېرىمىز، قوشۇنۇڭلار كۆپ بولغان تەقدىردىمۇ، ئۇ سىلەردىن ھېچ نەرسىنى دەپئى قىلىپ بېرەلمەيدۇ، اﷲ ئىشەنچلىك ھالدا مۆمىنلەر بىلەن بىللىدۇر
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ وَلَا تَوَلَّوْا۟ عَنْهُ وَأَنتُمْ تَسْمَعُونَ ﴿٢٠﴾
ئى مۆمىنلەر! اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە ئىتائەت قىلىڭلار، سىلەر (قۇرئاننى) ئاڭلاپ تۇرۇپ ئۇنىڭدىن يۈز ئۆرۈمەڭلار
وَلَا تَكُونُوا۟ كَٱلَّذِينَ قَالُوا۟ سَمِعْنَا وَهُمْ لَا يَسْمَعُونَ ﴿٢١﴾
ئاڭلىماي تۇرۇپ ئاڭلىدۇق دېگەن كىشىلەرنى (يەنى ھەقىقەتنى رەت قىلغۇچىلارنى) دورىماڭلار
۞ إِنَّ شَرَّ ٱلدَّوَآبِّ عِندَ ٱللَّهِ ٱلصُّمُّ ٱلْبُكْمُ ٱلَّذِينَ لَا يَعْقِلُونَ ﴿٢٢﴾
شۈبھىسىزكى، اﷲ نىڭ نەزىرىدە ھايۋانلارنىڭ ئەڭ يامىنى (ھەقنى ئاڭلىماي) گاس بولۇۋالغان، (ھەقنى سۆزلىمەي) گاچا بولۇۋالغان، (ياخشى - ياماننى پەرق ئەتمەي) ئەقلىنى يوقاتقان كىشىلەردۇر
وَلَوْ عَلِمَ ٱللَّهُ فِيهِمْ خَيْرًا لَّأَسْمَعَهُمْ ۖ وَلَوْ أَسْمَعَهُمْ لَتَوَلَّوا۟ وَّهُم مُّعْرِضُونَ ﴿٢٣﴾
ئەگەر اﷲ ئۇلاردا بىرەر ياخشىلىق بار دەپ بىلسە ئىدى، ئەلۋەتتە ئۇلارغا (ھەقنى) ئاڭلىتاتتى؛ ئۇلارغا ئاڭلاتقان تەقدىردىمۇ، ئۇلار ھەقتىن ئەلۋەتتە يۈز ئۆرۈگەن ھالدا باش تارتاتتى
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱسْتَجِيبُوا۟ لِلَّهِ وَلِلرَّسُولِ إِذَا دَعَاكُمْ لِمَا يُحْيِيكُمْ ۖ وَٱعْلَمُوٓا۟ أَنَّ ٱللَّهَ يَحُولُ بَيْنَ ٱلْمَرْءِ وَقَلْبِهِۦ وَأَنَّهُۥٓ إِلَيْهِ تُحْشَرُونَ ﴿٢٤﴾
ئى مۆمىنلەر! اﷲ ۋە ئۇنىڭ پەيغەمبىرى سىلەرنى تىرىلدۈرىدىغان (يەنى ئەبەدىي ھاياتقا ئېرىشتۈرىدىغان) ئىمانغا دەۋەت قىلسا، ئۇنى قوبۇل قىلىڭلار. بىلىڭلاركى، اﷲ كىشى بىلەن ئۇنىڭ قەلبى ئارىسىدا توسالغۇ بولالايدۇ (يەنى كىشىنىڭ دىلىنى ئىگىسىنىڭ خاھىشى بويىچە ئەمەس، بەلكى ئۆز خاھىشى بويىچە تەسەررۇپ قىلىدۇ). (قىيامەت كۈنى) سىلەر اﷲ نىڭ دەرگاھىغا يىغىلىسىلەر
وَٱتَّقُوا۟ فِتْنَةً لَّا تُصِيبَنَّ ٱلَّذِينَ ظَلَمُوا۟ مِنكُمْ خَآصَّةً ۖ وَٱعْلَمُوٓا۟ أَنَّ ٱللَّهَ شَدِيدُ ٱلْعِقَابِ ﴿٢٥﴾
سىلەرنىڭ ئاراڭلاردىكى زۇلۇم قىلغانلارنىڭ بېشىغا كېلىش بىلەنلا چەكلىنىپ قالمايدىغان بالا - قازادىن ساقلىنىڭلار، بىلىڭلاركى، اﷲ نىڭ ئازابى قاتتىقتۇر
وَٱذْكُرُوٓا۟ إِذْ أَنتُمْ قَلِيلٌ مُّسْتَضْعَفُونَ فِى ٱلْأَرْضِ تَخَافُونَ أَن يَتَخَطَّفَكُمُ ٱلنَّاسُ فَـَٔاوَىٰكُمْ وَأَيَّدَكُم بِنَصْرِهِۦ وَرَزَقَكُم مِّنَ ٱلطَّيِّبَـٰتِ لَعَلَّكُمْ تَشْكُرُونَ ﴿٢٦﴾
ياد ئېيتىڭلاركى، ئۆز ۋاقتىدا سىلەر (مەككە) زېمىنىدا ئاز سانلىق بولۇپ، بوزەك قىلىنغان ئىدىڭلار، كىشىلەر (يەنى مۇشرىكلار) نىڭ تالان - تاراج قىلىشىدىن قورقاتتىڭلار، شۈكۈر قىلىشىڭلار ئۈچۈن اﷲ سىلەرنى (مەدىنىدە) يەرلەشتۈردى، (بەدرى ئۇرۇشىدا) ئۆز ياردىمى بىلەن سىلەرنى كۈچلەندۈردى. سىلەرگە ھالال نەرسىلەر (يەنى غەنىيمەتلەر) نى رىزىق قىلىپ بەردى
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَخُونُوا۟ ٱللَّهَ وَٱلرَّسُولَ وَتَخُونُوٓا۟ أَمَـٰنَـٰتِكُمْ وَأَنتُمْ تَعْلَمُونَ ﴿٢٧﴾
ئى مۆمىنلەر! اﷲ قا، خۇدانىڭ ئەلچىسىگە خىيانەت قىلماڭلار، سىلەرگە قويۇلغان ئامانەتلەرگە بىلىپ تۇرۇپ خىيانەت قىلماڭلار
وَٱعْلَمُوٓا۟ أَنَّمَآ أَمْوَٰلُكُمْ وَأَوْلَـٰدُكُمْ فِتْنَةٌ وَأَنَّ ٱللَّهَ عِندَهُۥٓ أَجْرٌ عَظِيمٌ ﴿٢٨﴾
بىلىڭلاركى، سىلەرنىڭ ماللىرىڭلار، بالىلىرىڭلار سىلەر ئۈچۈن بىر تۈرلۈك سىناقتۇر، اﷲ نىڭ دەرگاھىدا چوڭ ساۋاب بار
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِن تَتَّقُوا۟ ٱللَّهَ يَجْعَل لَّكُمْ فُرْقَانًا وَيُكَفِّرْ عَنكُمْ سَيِّـَٔاتِكُمْ وَيَغْفِرْ لَكُمْ ۗ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٢٩﴾
ئى مۆمىنلەر! ئەگەر اﷲ تىن قورقساڭلار، اﷲ سىلەرگە ھەق بىلەن باتىلنى ئايرىيدىغان ھىدايەت ئاتا قىلىدۇ، گۇناھىڭلارنى كەچۈرىدۇ، سىلەرگە مەغپىرەت قىلىدۇ. اﷲ كاتتا پەزل ئىگىسىدۇر
وَإِذْ يَمْكُرُ بِكَ ٱلَّذِينَ كَفَرُوا۟ لِيُثْبِتُوكَ أَوْ يَقْتُلُوكَ أَوْ يُخْرِجُوكَ ۚ وَيَمْكُرُونَ وَيَمْكُرُ ٱللَّهُ ۖ وَٱللَّهُ خَيْرُ ٱلْمَـٰكِرِينَ ﴿٣٠﴾
(ئى مۇھەممەد!) ئۆز ۋاقتىدا ھەقىقەتنى رەت قىلغۇچىلار سېنى قاماققا ئېلىش، ياكى ئۆلتۈرۈش ۋە ياكى (مەككىدىن) ھەيدەپ چىقىرىش ئۈچۈن مىكىر ئىشلەتتى، ئۇلار مىكىر ئىشلىتىدۇ، اﷲ ئۇلارنىڭ مىكرىنى بەربات قىلىدۇ، اﷲ نىڭ تەدبىرى ئۇلارنىڭ مىكرىدىن ئۈنۈملۈكتۇر
وَإِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُنَا قَالُوا۟ قَدْ سَمِعْنَا لَوْ نَشَآءُ لَقُلْنَا مِثْلَ هَـٰذَآ ۙ إِنْ هَـٰذَآ إِلَّآ أَسَـٰطِيرُ ٱلْأَوَّلِينَ ﴿٣١﴾
ئۇلارغا بىزنىڭ ئايەتلىرىمىز تىلاۋەت قىلىنسا، ئۇلار: «بىز (بۇ سۆزلەرنى) ئاللىقاچان ئاڭلىغان ئىدۇق، ئەگەر خالىساق ئۇنىڭغا ئوخشاش سۆزلەرنى بىزمۇ ئەلۋەتتە قىلالايتتۇق، بۇ پەقەت بۇرۇنقىلاردىن قالغان ئەپسانىلاردۇر» دېدى
وَإِذْ قَالُوا۟ ٱللَّهُمَّ إِن كَانَ هَـٰذَا هُوَ ٱلْحَقَّ مِنْ عِندِكَ فَأَمْطِرْ عَلَيْنَا حِجَارَةً مِّنَ ٱلسَّمَآءِ أَوِ ٱئْتِنَا بِعَذَابٍ أَلِيمٍ ﴿٣٢﴾
ئۆز ۋاقتىدا ئۇلار: «ئەگەر بۇ (يەنى قۇرئان) سېنىڭ تەرىپىڭدىن نازىل بولغان ھەقىقەت بولىدىغان بولسا، ئاسماندىن ئۈستىمىزگە تاش ياغدۇرغىن، ياكى بىزنى قاتتىق ئازاب بىلەن جازالىغىن!» دېدى
وَمَا كَانَ ٱللَّهُ لِيُعَذِّبَهُمْ وَأَنتَ فِيهِمْ ۚ وَمَا كَانَ ٱللَّهُ مُعَذِّبَهُمْ وَهُمْ يَسْتَغْفِرُونَ ﴿٣٣﴾
سەن ئۇلارنىڭ ئىچىدە تۇرغان چېغىڭدا (سېنى ھۆرمەتلەش يۈزىسىدىن) اﷲ ئۇلارغا ئازاب قىلمايدۇ، ئۇلار ئىستىغپار ئېيتىپ تۇرغان چاغدا اﷲ ئۇلارغا ئازاب قىلمايدۇ
وَمَا لَهُمْ أَلَّا يُعَذِّبَهُمُ ٱللَّهُ وَهُمْ يَصُدُّونَ عَنِ ٱلْمَسْجِدِ ٱلْحَرَامِ وَمَا كَانُوٓا۟ أَوْلِيَآءَهُۥٓ ۚ إِنْ أَوْلِيَآؤُهُۥٓ إِلَّا ٱلْمُتَّقُونَ وَلَـٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿٣٤﴾
ئۇلار (باشقىلارنىڭ) مەسجىدى ھەرەمغا كىرىشىنى توسۇۋاتسا، اﷲ ئۇلارغا نېمىشقا ئازاب قىلمىسۇن؟ ئۇلار مەسجىدى ھەرەمنىڭ ئىگىلىرى ئەمەس، مەسجىدى ھەرەمنىڭ ئىگىلىرى تەقۋادارلاردىن باشقىلار ئەمەستۇر، لېكىن (بۇنى) ئۇلارنىڭ تولىسى بىلمەيدۇ
وَمَا كَانَ صَلَاتُهُمْ عِندَ ٱلْبَيْتِ إِلَّا مُكَآءً وَتَصْدِيَةً ۚ فَذُوقُوا۟ ٱلْعَذَابَ بِمَا كُنتُمْ تَكْفُرُونَ ﴿٣٥﴾
ئۇلارنىڭ مەسجىدى ھەرەمنىڭ يېنىدىكى ئىبادىتى پەقەت ئىسقىرتماق، چاۋاك چالماقتىنلا ئىبارەت بولدى. (ئى ھەقىقەتنى رەت قىلغۇچىلار!) ھەقىقەتنى رەت قىلغۇچى بولغانلىقىڭلار تۈپەيلىدىن (دۇنيا ۋە ئاخىرەتنىڭ) ئازابىنى تېتىڭلار
إِنَّ ٱلَّذِينَ كَفَرُوا۟ يُنفِقُونَ أَمْوَٰلَهُمْ لِيَصُدُّوا۟ عَن سَبِيلِ ٱللَّهِ ۚ فَسَيُنفِقُونَهَا ثُمَّ تَكُونُ عَلَيْهِمْ حَسْرَةً ثُمَّ يُغْلَبُونَ ۗ وَٱلَّذِينَ كَفَرُوٓا۟ إِلَىٰ جَهَنَّمَ يُحْشَرُونَ ﴿٣٦﴾
شۈبھىسىزكى، ھەقىقەتنى رەت قىلغۇچىلار مال - مۈلۈكلىرىنى (باشقىلارنى) اﷲ نىڭ يولىدىن توسۇش ئۈچۈن سەرپ قىلىدۇ، سەرپ قىلىنغاندىن كېيىن ئۇلارنىڭ مال - مۈلۈكلىرى ئۇلارغا ھەسرەت ئېلىپ كېلىدۇ، ئاخىر ئۇلار يېڭىلىدۇ. ھەقىقەتنى رەت قىلغۇچىلار مەڭگۈلۈك ئوتگىلا ھەيدەپ كىرگۈزۈلىدۇ
لِيَمِيزَ ٱللَّهُ ٱلْخَبِيثَ مِنَ ٱلطَّيِّبِ وَيَجْعَلَ ٱلْخَبِيثَ بَعْضَهُۥ عَلَىٰ بَعْضٍ فَيَرْكُمَهُۥ جَمِيعًا فَيَجْعَلَهُۥ فِى جَهَنَّمَ ۚ أُو۟لَـٰٓئِكَ هُمُ ٱلْخَـٰسِرُونَ ﴿٣٧﴾
اﷲ نىڭ (ئۇلارنى دۇنيادا مەغلۇپ قىلىشى، ئاخىرەتتە مەڭگۈلۈك ئوتگە ھەيدىشى) ھەقىقەتنى رەت قىلغۇچىنى مۆمىندىن پەرقلەندۈرۈش ئۈچۈندۇر، اﷲ ھەممە ھەقىقەتنى رەت قىلغۇچىلارنى ئۈستى - ئۈستىگە دۆۋىلەپ توپلىغاندىن كېيىن، دوزاخقا تاشلايدۇ، ئەنە شۇلار زىيان تارتقۇچىلاردۇر
قُل لِّلَّذِينَ كَفَرُوٓا۟ إِن يَنتَهُوا۟ يُغْفَرْ لَهُم مَّا قَدْ سَلَفَ وَإِن يَعُودُوا۟ فَقَدْ مَضَتْ سُنَّتُ ٱلْأَوَّلِينَ ﴿٣٨﴾
(ئى مۇھەممەد!) ھەقىقەتنى رەت قىلغۇچىلارغا ئېيتقىنكى، ئەگەر ئۇلار (كۇفرىدىن، خۇدانىڭ ئەلچىسى بىلەن دۈشمەنلىشىشتىن) يانسا، ئۇلارنىڭ ئۆتكەنكى خۇدانىڭ ئەلچىسىلىرىمنى ئىنكار قىلغانلارنى ھالاك قىلىش مېنىڭ تۇتقان يولۇم بولۇپ كەلدى (ئۇلارنىمۇ شۇنداق قىلىمەن)
وَقَـٰتِلُوهُمْ حَتَّىٰ لَا تَكُونَ فِتْنَةٌ وَيَكُونَ ٱلدِّينُ كُلُّهُۥ لِلَّهِ ۚ فَإِنِ ٱنتَهَوْا۟ فَإِنَّ ٱللَّهَ بِمَا يَعْمَلُونَ بَصِيرٌ ﴿٣٩﴾
پىتنە تۈگىگەن، دىن پۈتۈنلەي اﷲ ئۈچۈن بولغانغا قەدەر ئۇلار بىلەن ئۇرۇشۇڭلار؛ ئەگەر ئۇلار (كۇفرىدىن) يانسا، اﷲ ئۇلارنىڭ قىلمىشلىرىنى كۆرىدۇ
وَإِن تَوَلَّوْا۟ فَٱعْلَمُوٓا۟ أَنَّ ٱللَّهَ مَوْلَىٰكُمْ ۚ نِعْمَ ٱلْمَوْلَىٰ وَنِعْمَ ٱلنَّصِيرُ ﴿٤٠﴾
ئەگەر ئۇلار (ئىماندىن) يۈز ئۆرۈسە، (ئى مۆمىنلەر!) بىلىڭلاركى، اﷲ سىلەرنىڭ مەدەتكارىڭلاردۇر، اﷲ نېمىدېگەن ياخشى مەدەتكار، نېمىدېگەن ياخشى ياردەمچى!
۞ وَٱعْلَمُوٓا۟ أَنَّمَا غَنِمْتُم مِّن شَىْءٍ فَأَنَّ لِلَّهِ خُمُسَهُۥ وَلِلرَّسُولِ وَلِذِى ٱلْقُرْبَىٰ وَٱلْيَتَـٰمَىٰ وَٱلْمَسَـٰكِينِ وَٱبْنِ ٱلسَّبِيلِ إِن كُنتُمْ ءَامَنتُم بِٱللَّهِ وَمَآ أَنزَلْنَا عَلَىٰ عَبْدِنَا يَوْمَ ٱلْفُرْقَانِ يَوْمَ ٱلْتَقَى ٱلْجَمْعَانِ ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٤١﴾
(ئى مۆمىنلەر!) بىلىڭلاركى، سىلەر ئالغان غەنىيمەتنىڭ بەشتىن بىرى اﷲ قا، خۇدانىڭ ئەلچىسىگە، خۇدانىڭ ئەلچىسىنىڭ خىش - ئەقرىبالىرىغا، يېتىملەرگە، مىسكىنلەرگە، ئىبن سەبىللەرگە خاستۇر. ئەگەر سىلەر اﷲ قا، ئىككى گۇرۇھ (يەنى مۇسۇلمانلار بىلەن ھەقىقەتنى رەت قىلغۇچىلار) ئۇچرىشىپ ھەق بىلەن ناھەق ئايرىلغان كۈندە (يەنى بەدرى ئۇرۇشى بولغان كۈندە) بەندىمىزگە (يەنى مۇھەممەد ئەلەيھىسسالامغا) بىز نازىل قىلغان نەرسىلەر (يەنى ئايەتلەر ۋە خۇدانىڭ ئەلچى پەرىشتىلىرى) گە ئىشەنسەڭلار (اﷲ نىڭ غەنىيمەت توغرىسىدىكى ھۆكمىنىڭ ئەنە شۇ ئىكەنلىكىنى بىلىڭلار). اﷲ ھەر نەرسىگە قادىردۇر
إِذْ أَنتُم بِٱلْعُدْوَةِ ٱلدُّنْيَا وَهُم بِٱلْعُدْوَةِ ٱلْقُصْوَىٰ وَٱلرَّكْبُ أَسْفَلَ مِنكُمْ ۚ وَلَوْ تَوَاعَدتُّمْ لَٱخْتَلَفْتُمْ فِى ٱلْمِيعَـٰدِ ۙ وَلَـٰكِن لِّيَقْضِىَ ٱللَّهُ أَمْرًا كَانَ مَفْعُولًا لِّيَهْلِكَ مَنْ هَلَكَ عَنۢ بَيِّنَةٍ وَيَحْيَىٰ مَنْ حَىَّ عَنۢ بَيِّنَةٍ ۗ وَإِنَّ ٱللَّهَ لَسَمِيعٌ عَلِيمٌ ﴿٤٢﴾
ئۆز ۋاقتىدا سىلەر ۋادىنىڭ (مەدىنىگە) يېقىن تەرىپىدە ئىدىڭلار، ئۇلار ۋادىنىڭ (مەدىنىگە) يىراق تەرىپىدە ئىدى. (قۇرەيشنىڭ) سودا كارۋىنى بولسا سىلەرنىڭ تۆۋىنىڭلاردا ئىدى. (مۇشرىكلار بىلەن ئۇچرىشىشنى) ۋەدىلەشكەن بولساڭلار، سىلەر (ئۆزۈڭلارنىڭ ئازلىقىنى، مۇشرىكلارنىڭ كۆپلۈكىنى كۆرۈپ) ئەلۋەتتە بۇنىڭغا خىلاپلىق قىلغان بولاتتىڭلار، لېكىن اﷲ بولۇشقا تېگىشلىك ئىشنى (يەنى مۇسۇلمانلارنى ئەزىز، مۇشرىكلارنى خار قىلىشنى) ئەمەلگە ئاشۇرۇش ئۈچۈن (سىلەرنى بەدرىدە مۇشرىكلار بىلەن ۋەدىسىز ئۇچراشتۇردى). اﷲ نىڭ مۇنداق قىلىشى ھالاك بولىدىغانلارنىڭ روشەن دەلىلنى كۆرۈپ ئاندىن ھالاك بولۇشى، ياشايدىغانلارنىڭ روشەن دەلىلنى كۆرۈپ ئاندىن ياشىشى ئۈچۈن ئىدى. اﷲ (بەندىلىرىنىڭ سۆزلىرىنى)، ئەلۋەتتە، ئاڭلاپ تۇرغۇچىدۇر، (نىيەتلىرىنى) بىلىپ تۇرغۇچىدۇر
إِذْ يُرِيكَهُمُ ٱللَّهُ فِى مَنَامِكَ قَلِيلًا ۖ وَلَوْ أَرَىٰكَهُمْ كَثِيرًا لَّفَشِلْتُمْ وَلَتَنَـٰزَعْتُمْ فِى ٱلْأَمْرِ وَلَـٰكِنَّ ٱللَّهَ سَلَّمَ ۗ إِنَّهُۥ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ ﴿٤٣﴾
(ئى مۇھەممەد!) ئۆز ۋاقتىدا، اﷲ چۈشۈڭدە ساڭا ئۇلارنى (يەنى دۈشمەنلىرىڭنى) ئاز كۆرسەتتى، ئەگەر ساڭا ئۇلارنى كۆپ كۆرسەتكەن بولسا، سىلەرگە چوقۇم قورقۇنچ چۈشەتتى ۋە جەڭ ئىشىدا چوقۇم دەتالاش قىلاتتىڭلار، لېكىن اﷲ (سىلەرنى مۇنداق قىلىشتىن) ساقلىدى. اﷲ دىللاردىكىنى ئەلۋەتتە بىلىپ تۇرغۇچىدۇر
وَإِذْ يُرِيكُمُوهُمْ إِذِ ٱلْتَقَيْتُمْ فِىٓ أَعْيُنِكُمْ قَلِيلًا وَيُقَلِّلُكُمْ فِىٓ أَعْيُنِهِمْ لِيَقْضِىَ ٱللَّهُ أَمْرًا كَانَ مَفْعُولًا ۗ وَإِلَى ٱللَّهِ تُرْجَعُ ٱلْأُمُورُ ﴿٤٤﴾
اﷲ بولۇشقا تېگىشلىك ئىشنى ئىشقا ئاشۇرۇش ئۈچۈن، دۈشمەن بىلەن ئۇچراشقان چېغىڭلاردا، (دۈشمەنلەرگە قارشى تۇرۇشقا جۈرئەت قىلسۇن دەپ) ئۇلارنى سىلەرنىڭ كۆزۈڭلارغا ئاز كۆرسەتتى ۋە سىلەرنى ئۇلارنىڭ كۆزلىرىگە (سىلەرگە قارشى تەييارلىقتا بولمىسۇن دەپ) ئاز كۆرسەتتى ھەممە ئىش اﷲ قا قايتۇرۇلىدۇ (اﷲ ئۆزى خالىغانچە ھۆكۈم قىلىدۇ)
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا لَقِيتُمْ فِئَةً فَٱثْبُتُوا۟ وَٱذْكُرُوا۟ ٱللَّهَ كَثِيرًا لَّعَلَّكُمْ تُفْلِحُونَ ﴿٤٥﴾
ئى مۆمىنلەر! (مۇشرىكلاردىن) بىر جامائەگە (يەنى دۈشمەن قوشۇنىغا) ئۇچراشقان چېغىڭلاردا ساباتلىق كۆرسىتىڭلار، مۇۋەپپەقىيەت قازىنىشىڭلار ئۈچۈن اﷲ نى كۆپ ياد ئېتىڭلار
وَأَطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ وَلَا تَنَـٰزَعُوا۟ فَتَفْشَلُوا۟ وَتَذْهَبَ رِيحُكُمْ ۖ وَٱصْبِرُوٓا۟ ۚ إِنَّ ٱللَّهَ مَعَ ٱلصَّـٰبِرِينَ ﴿٤٦﴾
(پۈتۈن سۆز ھەرىكەتلىرىڭلاردا) اﷲ قا ۋە ئۇنىڭ پەيغەمبىرىگە ئىتائەت قىلىڭلار، ئىختىلاپ قىلىشماڭلار، بولمىسا، (دۈشمەن بىلەن ئۇچرىشىشتىن) قورقۇپ قالىسىلەر، كۈچ - قۇۋۋىتىڭلار كېتىپ قالىدۇ؛ سەۋر قىلىڭلار، اﷲ ئىشەنچلىك ھالدا سەۋر قىلغۇچىلار بىلەن بىللىدۇر
وَلَا تَكُونُوا۟ كَٱلَّذِينَ خَرَجُوا۟ مِن دِيَـٰرِهِم بَطَرًا وَرِئَآءَ ٱلنَّاسِ وَيَصُدُّونَ عَن سَبِيلِ ٱللَّهِ ۚ وَٱللَّهُ بِمَا يَعْمَلُونَ مُحِيطٌ ﴿٤٧﴾
يۇرتلىرىدىن چوڭچىلىق ۋە رىيا بىلەن چىققان كىشىلەر (يەنى قۇرەيش مۇشرىكلىرى) گە ئوخشاش بولماڭلار، ئۇلار (كىشىلەرنى) اﷲ نىڭ يولىدىن توسىدۇ، اﷲ ئۇلارنىڭ ھەممە ئىشلىرىنى بىلگۈچىدۇر
وَإِذْ زَيَّنَ لَهُمُ ٱلشَّيْطَـٰنُ أَعْمَـٰلَهُمْ وَقَالَ لَا غَالِبَ لَكُمُ ٱلْيَوْمَ مِنَ ٱلنَّاسِ وَإِنِّى جَارٌ لَّكُمْ ۖ فَلَمَّا تَرَآءَتِ ٱلْفِئَتَانِ نَكَصَ عَلَىٰ عَقِبَيْهِ وَقَالَ إِنِّى بَرِىٓءٌ مِّنكُمْ إِنِّىٓ أَرَىٰ مَا لَا تَرَوْنَ إِنِّىٓ أَخَافُ ٱللَّهَ ۚ وَٱللَّهُ شَدِيدُ ٱلْعِقَابِ ﴿٤٨﴾
ئۆز ۋاقتىدا شەيتان (ئازدۇرغۇچى) ئۇلارغا (قەبىھ) ئەمەللىرىنى چىرايلىق كۆرسىتىپ: «ھېچقانداق كىشى سىلەرنى يېڭەلمەيدۇ، مەن ئىشەنچلىك ھالدا سىلەرگە مەدەتكار» دېگەن ئىدى. ئىككى قوشۇن ئۇچراشقان چاغدا شەيتان (ئازدۇرغۇچى) ئارقىسىغا چېكىندى ۋە: «مەن سىلەردىن ئادا - جۇدامەن، سىلەر كۆرمەيۋاتقاننى ئىشەنچلىك ھالدا كۆرۈپ تۇرۇۋاتىمەن، مەن راستلا اﷲ تىن قورقىمەن، اﷲ نىڭ ئازابى قاتتىقتۇر» دېدى
إِذْ يَقُولُ ٱلْمُنَـٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ غَرَّ هَـٰٓؤُلَآءِ دِينُهُمْ ۗ وَمَن يَتَوَكَّلْ عَلَى ٱللَّهِ فَإِنَّ ٱللَّهَ عَزِيزٌ حَكِيمٌ ﴿٤٩﴾
ئۆز ۋاقتىدا مۇناپىقلار ۋە دىللىرىدا كېسىلى (يەنى شەك) بارلار: «بۇلارنى (يەنى مۇسۇلمانلارنى) ئۇلارنىڭ دىنى ئالدىدى (يەنى مۇسۇلمانلار ئۆز دىنىغا مەغرۇر بولۇپ، ئۆزلىرىنى يېڭىلمەس دەپ گۇمان قىلىپ، ئاز سانلىق تۇرۇقلۇق كۆپ سانلىقلار بىلەن ئۇرۇشۇشقا چىقتى)» دېدى. كىمكى اﷲ قا يۆلىنىدىكەن (اﷲ ئۇنىڭغا ئەلۋەتتە ياردەم بېرىدۇ). چۈنكى اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
وَلَوْ تَرَىٰٓ إِذْ يَتَوَفَّى ٱلَّذِينَ كَفَرُوا۟ ۙ ٱلْمَلَـٰٓئِكَةُ يَضْرِبُونَ وُجُوهَهُمْ وَأَدْبَـٰرَهُمْ وَذُوقُوا۟ عَذَابَ ٱلْحَرِيقِ ﴿٥٠﴾
خۇدانىڭ ئەلچى پەرىشتىلىرى (بەدرى ئۇرۇشىدا) ھەقىقەتنى رەت قىلغۇچىلارنىڭ جانلىرىنى ئېلىۋاتقاندا، ئۇلارنىڭ يۈزلىرىگە ۋە ئارقىلىرىغا ئۇرۇۋاتقانلىقىنى كۆرسەڭ ئىدىڭ (ئەلۋەتتە قورقۇنچلۇق ھالنى كۆرگەن بولاتتىڭ). (خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارغا) «دوزاخنىڭ كۆيدۈرگۈچى ئازابىنى تېتىڭلار!» (دەيتتى)
ذَٰلِكَ بِمَا قَدَّمَتْ أَيْدِيكُمْ وَأَنَّ ٱللَّهَ لَيْسَ بِظَلَّـٰمٍ لِّلْعَبِيدِ ﴿٥١﴾
بۇ (ئازاب) سىلەرنىڭ قىلمىش جىنايىتىڭلار تۈپەيلىدىندۇر، اﷲ بەندىلىرىگە زۇلۇم قىلغۇچى ئەمەستۇر
كَدَأْبِ ءَالِ فِرْعَوْنَ ۙ وَٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَفَرُوا۟ بِـَٔايَـٰتِ ٱللَّهِ فَأَخَذَهُمُ ٱللَّهُ بِذُنُوبِهِمْ ۗ إِنَّ ٱللَّهَ قَوِىٌّ شَدِيدُ ٱلْعِقَابِ ﴿٥٢﴾
(بۇ ھەقىقەتنى رەت قىلغۇچىلارنىڭ گۇناھ قىلىش ئادىتى) پىرئەۋن جامائەسى ۋە ئۇلاردىن بۇرۇنقىلارنىڭ ئادىتىگە ئوخشايدۇ، ئۇلار اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلدى، گۇناھلىرى تۈپەيلىدىن، اﷲ ئۇلارنى ھالاك قىلدى. اﷲ ئىشەنچلىك ھالدا كۈچلۈكتۇر، اﷲ نىڭ ئازابى قاتتىقتۇر
ذَٰلِكَ بِأَنَّ ٱللَّهَ لَمْ يَكُ مُغَيِّرًا نِّعْمَةً أَنْعَمَهَا عَلَىٰ قَوْمٍ حَتَّىٰ يُغَيِّرُوا۟ مَا بِأَنفُسِهِمْ ۙ وَأَنَّ ٱللَّهَ سَمِيعٌ عَلِيمٌ ﴿٥٣﴾
بۇ (يەنى ئۇلارغا نازىل بولغان ئازاب) شۇنىڭ ئۈچۈنكى، بىرەر قەۋم ئۆزلىرىنىڭ ھالىتىنى (كۇفرى ۋە گۇناھ بىلەن) ئۆزگەرتمىگۈچە اﷲ ئۇلارغا بەرگەن نېمىتىنى ئۆزگەرتىۋەتمەيدۇ، اﷲ (ئۇلارنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (قىلمىشلىرىنى) بىلىپ تۇرغۇچىدۇر
كَدَأْبِ ءَالِ فِرْعَوْنَ ۙ وَٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَذَّبُوا۟ بِـَٔايَـٰتِ رَبِّهِمْ فَأَهْلَكْنَـٰهُم بِذُنُوبِهِمْ وَأَغْرَقْنَآ ءَالَ فِرْعَوْنَ ۚ وَكُلٌّ كَانُوا۟ ظَـٰلِمِينَ ﴿٥٤﴾
(ئۇلارنىڭ ئادىتى) پىرئەۋن جامائەسىنىڭ ۋە ئۇلاردىن بۇرۇنقىلارنىڭ ئادىتىگە ئوخشايدۇكى، ئۇلار پەرۋەردىگارىنىڭ ئايەتلىرىنى ئىنكار قىلغان ئىدى. گۇناھلىرى تۈپەيلىدىن ئۇلارنى ھالاك قىلدۇق، پىرئەۋن جامائەسىنى (دېڭىزغا) غەرق قىلدۇق، ئۇلارنىڭ ھەممىسى (كۇفرىلىق ۋە گۇناھ قىلىپ، ئۆزلىرىنى ئازابقا دۇچار قىلىش بىلەن) ئۆزلىرىگە زۇلۇم قىلغۇچى (قەۋم) بولدى
إِنَّ شَرَّ ٱلدَّوَآبِّ عِندَ ٱللَّهِ ٱلَّذِينَ كَفَرُوا۟ فَهُمْ لَا يُؤْمِنُونَ ﴿٥٥﴾
اﷲ نىڭ نەزىرىدە ھايۋانلارنىڭ ئەڭ يامىنى كۇفرىدا چىڭ تۇرغانلاردۇركى، ئۇلار ئىشەنچلىك ھالدا ئىمان ئېيتمايدۇ
ٱلَّذِينَ عَـٰهَدتَّ مِنْهُمْ ثُمَّ يَنقُضُونَ عَهْدَهُمْ فِى كُلِّ مَرَّةٍ وَهُمْ لَا يَتَّقُونَ ﴿٥٦﴾
(ئى مۇھەممەد!) سەن ئۇلارنىڭ ئارىسىدىكى مەلۇم كىشىلەر بىلەن (مۇشرىكلارغا ياردەم بەرمەسلىكلىرى ئۈستىدە) مۇئاھىدە تۈزدۈڭ، ئۇلار بولسا ھەر قېتىمدا مۇئاھىدىسىنى بۇزىدۇ، ئۇلار اﷲ تىن قورقمايدۇ
فَإِمَّا تَثْقَفَنَّهُمْ فِى ٱلْحَرْبِ فَشَرِّدْ بِهِم مَّنْ خَلْفَهُمْ لَعَلَّهُمْ يَذَّكَّرُونَ ﴿٥٧﴾
ئەگەر ئۇرۇشتا ئۇلار ئۈستىدىن غەلىبە قىلساڭ، ئارقىدىكىلىرىنىڭ قېچىشى ۋە ئىبرەت ئېلىشى ئۈچۈن ئۇلارنى قاتتىق جازالىغىن (يەنى ئۇلارنى باشقىلارغا ئىبرەت قىلغىنكى، ئۇلارنىڭ سەن بىلەن ئۇرۇش قىلىشقا دەرمانى قالمىسۇن)
وَإِمَّا تَخَافَنَّ مِن قَوْمٍ خِيَانَةً فَٱنۢبِذْ إِلَيْهِمْ عَلَىٰ سَوَآءٍ ۚ إِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْخَآئِنِينَ ﴿٥٨﴾
ئەگەر سەن (مۇئاھىدە تۈزگەن) قەۋمدىن خىيانەت (ئالامەتلىرىنى) سەزسەڭ، مۇئاھىدىسىنى ئۇلارغا ئوچۇق - يورۇقلۇق بىلەن تاشلاپ بەرگىن (يەنى سەن بىلەن مۇئاھىدە تۈزۈشكەن قەۋمدىن خىيانەت شەپىسى كۆرۈلسە، ئۇلارغا تۇيۇقسىز ھۇجۇم قىلماستىن، مۇئاھىدىنىڭ ئەمەلدىن قالغانلىقىنى ئالدى بىلەن ئۇلارغا ئۇقتۇرۇپ قويغىن). اﷲ ئىشەنچلىك ھالدا خائىنلارنى دوست تۇتمايدۇ
وَلَا يَحْسَبَنَّ ٱلَّذِينَ كَفَرُوا۟ سَبَقُوٓا۟ ۚ إِنَّهُمْ لَا يُعْجِزُونَ ﴿٥٩﴾
(بەدرى ئۇرۇشىدا ئۆلتۈرۈلمەي قالغان) ھەقىقەتنى رەت قىلغۇچىلار قۇتۇلدۇق دەپ ئويلىمىسۇن، ئۇلار (پەرۋەردىگارىنىڭ جازاسىدىن) ئىشەنچلىك ھالدا قېچىپ قۇتۇلالمايدۇ
وَأَعِدُّوا۟ لَهُم مَّا ٱسْتَطَعْتُم مِّن قُوَّةٍ وَمِن رِّبَاطِ ٱلْخَيْلِ تُرْهِبُونَ بِهِۦ عَدُوَّ ٱللَّهِ وَعَدُوَّكُمْ وَءَاخَرِينَ مِن دُونِهِمْ لَا تَعْلَمُونَهُمُ ٱللَّهُ يَعْلَمُهُمْ ۚ وَمَا تُنفِقُوا۟ مِن شَىْءٍ فِى سَبِيلِ ٱللَّهِ يُوَفَّ إِلَيْكُمْ وَأَنتُمْ لَا تُظْلَمُونَ ﴿٦٠﴾
دۈشمەنلىرىڭلار (بىلەن ئۇرۇش قىلىش) ئۈچۈن، قولۇڭلاردىن كېلىشىچە قورال كۈچى، جەڭ ئېتى تەييارلاڭلار، بۇنىڭ بىلەن، اﷲ نىڭ دۈشمىنىنى، ئۆزۈڭلارنىڭ دۈشمىنىڭلارنى ۋە ئۇلاردىن باشقا دۈشمەنلەرنى قورقۇتىسىلەر، ئۇلارنى سىلەر تونۇمايسىلەر، اﷲ تونۇيدۇ، سىلەرنىڭ اﷲ يولىدا سەرپ قىلغىنىڭلار مەيلى نېمە بولسا بولسۇن، سىلەرگە ئۇنىڭ ساۋابى تولۇق بېرىلىدۇ، سىلەرگە زۇلۇم قىلىنمايدۇ (يەنى بۇ ساۋابتىن ھېچ نەرسە كېمەيتىلمەيدۇ)
۞ وَإِن جَنَحُوا۟ لِلسَّلْمِ فَٱجْنَحْ لَهَا وَتَوَكَّلْ عَلَى ٱللَّهِ ۚ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٦١﴾
ئەگەر ئۇلار تىنچلىققا مايىل بولسا، سەنمۇ تىنچلىققا مايىل بولغىن، اﷲ قا تەۋەككۈل قىلغىن. اﷲ (ئۇلارنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (نىيەتلىرىنى) بىلىپ تۇرغۇچىدۇر
وَإِن يُرِيدُوٓا۟ أَن يَخْدَعُوكَ فَإِنَّ حَسْبَكَ ٱللَّهُ ۚ هُوَ ٱلَّذِىٓ أَيَّدَكَ بِنَصْرِهِۦ وَبِٱلْمُؤْمِنِينَ ﴿٦٢﴾
ئەگەر ئۇلار (تەييارلىق قىلىۋېلىش ئۈچۈن سۈلھى بىلەن) سېنى ئالدىماقچى بولسا، اﷲ (ئۇلارنىڭ شەررىدىن) شەكسىز ساڭا كۇپايە قىلغۇچىدۇر. اﷲ سېنى ئۆزنىڭ ياردىمى بىلەن ۋە مۆمىنلەر بىلەن كۈچلەندۈرىدۇ
وَأَلَّفَ بَيْنَ قُلُوبِهِمْ ۚ لَوْ أَنفَقْتَ مَا فِى ٱلْأَرْضِ جَمِيعًا مَّآ أَلَّفْتَ بَيْنَ قُلُوبِهِمْ وَلَـٰكِنَّ ٱللَّهَ أَلَّفَ بَيْنَهُمْ ۚ إِنَّهُۥ عَزِيزٌ حَكِيمٌ ﴿٦٣﴾
اﷲ مۆمىنلەرنىڭ دىللىرىنى بىرلەشتۈردى. سەن يەر يۈزىدىكى پۈتۈن بايلىقنى سەرپ قىلىپمۇ ئۇلارنىڭ دىللىرىنى بىرلەشتۈرەلمەيتىڭ؛ لېكىن اﷲ (ئۆزىنىڭ قۇدرەت كامىلەسى بىلەن) ئۇلارنى ئىناق قىلدى. شۈبھىسىزكى، اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
يَـٰٓأَيُّهَا ٱلنَّبِىُّ حَسْبُكَ ٱللَّهُ وَمَنِ ٱتَّبَعَكَ مِنَ ٱلْمُؤْمِنِينَ ﴿٦٤﴾
ئى خۇدانىڭ ئەلچىسى! (يالغۇز) اﷲ ساڭا ۋە ساڭا ئەگەشكەن مۆمىنلەرگە كۇپايە قىلغۇچىدۇر
يَـٰٓأَيُّهَا ٱلنَّبِىُّ حَرِّضِ ٱلْمُؤْمِنِينَ عَلَى ٱلْقِتَالِ ۚ إِن يَكُن مِّنكُمْ عِشْرُونَ صَـٰبِرُونَ يَغْلِبُوا۟ مِا۟ئَتَيْنِ ۚ وَإِن يَكُن مِّنكُم مِّا۟ئَةٌ يَغْلِبُوٓا۟ أَلْفًا مِّنَ ٱلَّذِينَ كَفَرُوا۟ بِأَنَّهُمْ قَوْمٌ لَّا يَفْقَهُونَ ﴿٦٥﴾
ئى خۇدانىڭ ئەلچىسى! مۆمىنلەرنى (مۇشرىكلار بىلەن) ئۇرۇش قىلىشقا رىغبەتلەندۈرگىن، ئەگەر سىلەردە چىداملىق يىگىرمە ئادەم بولىدىغان بولسا، (دۈشمەندىن) 200 نى يېڭەلەيدۇ؛ ئەگەر سىلەردە (چىداملىق) 100 ئادەم بولىدىغان بولسا، ھەقىقەتنى رەت قىلغۇچىلاردىن 1000 نى يېڭەلەيدۇ؛ چۈنكى ئۇلار (جاھىل) قەۋم بولۇپ، (اﷲ نىڭ ھېكمىتىنى) چۈشەنمەيدۇ
ٱلْـَٔـٰنَ خَفَّفَ ٱللَّهُ عَنكُمْ وَعَلِمَ أَنَّ فِيكُمْ ضَعْفًا ۚ فَإِن يَكُن مِّنكُم مِّا۟ئَةٌ صَابِرَةٌ يَغْلِبُوا۟ مِا۟ئَتَيْنِ ۚ وَإِن يَكُن مِّنكُمْ أَلْفٌ يَغْلِبُوٓا۟ أَلْفَيْنِ بِإِذْنِ ٱللَّهِ ۗ وَٱللَّهُ مَعَ ٱلصَّـٰبِرِينَ ﴿٦٦﴾
اﷲ ئەمدى سىلەرنىڭ (يۈكۈڭلارنى) يىنىكلەتتى، سىلەرنىڭ ئاجىزلىقىڭلارنى بىلدى، ئەگەر سىلەردە چىداملىق 100 ئادەم بولىدىغان بولسا، 200 (دۈشمەن) نى يېڭەلەيدۇ، ئەگەر سىلەردە (چىداملىق) 1000 ئادەم بولىدىغان بولسا، اﷲ نىڭ ئىزنى بىلەن 2000 (دۈشمەن) نى يېڭەلەيدۇ. اﷲ چىداملىقلار بىلەن بىللىدۇر
مَا كَانَ لِنَبِىٍّ أَن يَكُونَ لَهُۥٓ أَسْرَىٰ حَتَّىٰ يُثْخِنَ فِى ٱلْأَرْضِ ۚ تُرِيدُونَ عَرَضَ ٱلدُّنْيَا وَٱللَّهُ يُرِيدُ ٱلْـَٔاخِرَةَ ۗ وَٱللَّهُ عَزِيزٌ حَكِيمٌ ﴿٦٧﴾
خۇدانىڭ ئەلچىسىگە زېمىندا دۈشمەننى كۆپرەك ئۆلتۈرمەي تۇرۇپ (يەنى مۇشرىكلىكنىڭ ھەيۋىسىنى يوقىتىپ، ئىسلامغا قۇۋۋەت بەرمەي تۇرۇپ)، ئەسىرلەردىن فىدىيە ئېلىش لايىق ئەمەس ئىدى. (ئى مۆمىنلەر! سىلەر فىدىيە ئېلىش بىلەن) دۇنيا مەنپەئىتىنى كۆزلەيسىلەر، اﷲ سىلەرگە ئاخىرەتنى (يەنى ئاخىرەتنىڭ ساۋابىنى) تىلەيدۇ. اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
لَّوْلَا كِتَـٰبٌ مِّنَ ٱللَّهِ سَبَقَ لَمَسَّكُمْ فِيمَآ أَخَذْتُمْ عَذَابٌ عَظِيمٌ ﴿٦٨﴾
اﷲ نىڭ (ئوچۇق چەكلەنمىگەن ئىشنى قىلغانلارنى جازالىماسلىق دېگەن) ھۆكمى ئەزىلىسى بولمىسا ئىدى، (ئەسىرلەردىن) فىدىيە ئالغانلىقىڭلار ئۈچۈن ئەلۋەتتە زور ئازابقا دۇچار بولاتتىڭلار
فَكُلُوا۟ مِمَّا غَنِمْتُمْ حَلَـٰلًا طَيِّبًا ۚ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ غَفُورٌ رَّحِيمٌ ﴿٦٩﴾
(ئى جىھاد قىلغۇچىلار جامائەسى!) غەنىيمەت ئالغان نەرسەڭلارنى ھالال ۋە پاك بىلىپ يەڭلار، اﷲ (نىڭ ئەمرىگە مۇخالىپەتچىلىك قىلىش) دىن قورقۇڭلار. اﷲ ئىشەنچلىك ھالدا مەغپىرەت قىلغۇچىدۇر، (بەندىلىرىگە) مېھرىباندۇر
يَـٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّمَن فِىٓ أَيْدِيكُم مِّنَ ٱلْأَسْرَىٰٓ إِن يَعْلَمِ ٱللَّهُ فِى قُلُوبِكُمْ خَيْرًا يُؤْتِكُمْ خَيْرًا مِّمَّآ أُخِذَ مِنكُمْ وَيَغْفِرْ لَكُمْ ۗ وَٱللَّهُ غَفُورٌ رَّحِيمٌ ﴿٧٠﴾
ئى خۇدانىڭ ئەلچىسى! قولۇڭلاردىكى ئەسىرلەرگە ئېيتقىنكى، ئەگەر اﷲ دىلىڭلاردا ئىمان بارلىقىنى بىلسە (يەنى دىلىڭلاردا ئىمان بولسا، اﷲ ئۇنى بىلىدۇ)، سىلەرگە ئۆزۈڭلاردىن ئېلىنغان فىدىيىدىن ياخشىراق نەرسىنى بېرىدۇ، (ئىلگىرىكى گۇناھلىرىڭلارنى) مەغپىرەت قىلىدۇ. اﷲ (تەۋبە قىلغۇچىلارنى) مەغپىرەت قىلغۇچىدۇر، (بەندىلىرىگە) مېھرىباندۇر
وَإِن يُرِيدُوا۟ خِيَانَتَكَ فَقَدْ خَانُوا۟ ٱللَّهَ مِن قَبْلُ فَأَمْكَنَ مِنْهُمْ ۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌ ﴿٧١﴾
(ئى مۇھەممەد!) ئەگەر ئۇلار (يەنى مۇشۇ ئەسەرلەر ئىماننى دەۋا قىلىش ئارقىلىق) ساڭا خىيانەت قىلماقچى بولغان بولسا، ئۇنداقتا ئىلگىرى ئۇلار (كۇفرى بىلەن، ئەھدىنى بۇزۇش بىلەن) اﷲ قا خىيانەت قىلغان ئىدى، اﷲ سېنى ئۇلارنىڭ ئۈستىدىن غالىب قىلدى. اﷲ (بولغۇسى ھەممە ئىشنى) بىلگۈچى، ئۆز ھېكمىتى بويىچە ئىش قىلغۇچىدۇر
إِنَّ ٱلَّذِينَ ءَامَنُوا۟ وَهَاجَرُوا۟ وَجَـٰهَدُوا۟ بِأَمْوَٰلِهِمْ وَأَنفُسِهِمْ فِى سَبِيلِ ٱللَّهِ وَٱلَّذِينَ ءَاوَوا۟ وَّنَصَرُوٓا۟ أُو۟لَـٰٓئِكَ بَعْضُهُمْ أَوْلِيَآءُ بَعْضٍ ۚ وَٱلَّذِينَ ءَامَنُوا۟ وَلَمْ يُهَاجِرُوا۟ مَا لَكُم مِّن وَلَـٰيَتِهِم مِّن شَىْءٍ حَتَّىٰ يُهَاجِرُوا۟ ۚ وَإِنِ ٱسْتَنصَرُوكُمْ فِى ٱلدِّينِ فَعَلَيْكُمُ ٱلنَّصْرُ إِلَّا عَلَىٰ قَوْمٍۭ بَيْنَكُمْ وَبَيْنَهُم مِّيثَـٰقٌ ۗ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٧٢﴾
ئىمان ئېيتقانلار، (اﷲ نى ۋە اﷲ نىڭ پەيغەمبىرىنى سۆيۈش يۈزىسىدىن يۇرتىنى تاشلاپ) ھىجرەت قىلغان، پۇل - ماللىرى ۋە جانلىرى بىلەن اﷲ يولىدا جىھاد قىلغانلار (يەنى مۇھاجىرلار)، (يۇرتىدا مۇھاجىرلارغا) جاي بەرگەنلەر ۋە ياردەم كۆرسەتكەنلەر (يەنى ئەنسارلار) - ئەنە شۇلار ئەلۋەتتە بىر - بىرىگە ئىگىدۇر (يەنى بىر - بىرىگە ياردەمچىدۇر، بىر - بىرىگە مىراسخوردۇر). ئىمان ئېيتقان ئەمما (مەدىنىگە) ھىجرەت قىلمىغانلار بىلەن سىلەرنىڭ ئوتتۇراڭلاردا تاكى ئۇلار ھىجرەت قىلغانغا قەدەر ئۆزئارا ئىگە بولۇش (يەنى ياردەملىشىش، مىراسخور بولۇش) بولمايدۇ؛ ئەگەر دىن ئىشىدا ئۇلار سىلەردىن ياردەم تەلەپ قىلسا، ئۇلارنىڭ دۈشمىنى بىلەن سىلەرنىڭ ئاراڭلاردا كېلىشىم بولمىسىلا، ئۇلارغا ياردەم بېرىشىڭلار كېرەك، اﷲ سىلەرنىڭ قىلىۋاتقان ئىشىڭلارنى كۆرۈپ تۇرغۇچىدۇر
وَٱلَّذِينَ كَفَرُوا۟ بَعْضُهُمْ أَوْلِيَآءُ بَعْضٍ ۚ إِلَّا تَفْعَلُوهُ تَكُن فِتْنَةٌ فِى ٱلْأَرْضِ وَفَسَادٌ كَبِيرٌ ﴿٧٣﴾
ھەقىقەتنى رەت قىلغۇچىلار (ياردەمدە ۋە مىراستا) بىر - بىرىگە ئىگىدۇر، ئەگەر بۇ ھۆكۈمگە ئەمەل قىلمىساڭلار، يەر يۈزىدە پىتنە ۋە چوڭ قالايمىقانچىلىق تۇغۇلىدۇ
وَٱلَّذِينَ ءَامَنُوا۟ وَهَاجَرُوا۟ وَجَـٰهَدُوا۟ فِى سَبِيلِ ٱللَّهِ وَٱلَّذِينَ ءَاوَوا۟ وَّنَصَرُوٓا۟ أُو۟لَـٰٓئِكَ هُمُ ٱلْمُؤْمِنُونَ حَقًّا ۚ لَّهُم مَّغْفِرَةٌ وَرِزْقٌ كَرِيمٌ ﴿٧٤﴾
ئىمان ئېيتقانلار، (مەدىنىگە) ھىجرەت قىلغانلار، اﷲ نىڭ يولىدا جىھاد قىلغانلار، (يۇرتىدا مۇھاجىرلارغا) جاي بەرگەنلەر ۋە ياردەم كۆرسەتكەنلەر (يەنى ئەنسارلار) - ئەنە شۇلار ھەقىقىي مۆمىنلەردۇر، ئۇلار (نىڭ گۇناھى) مەغپىرەت قىلىنىدۇ، ئۇلار (مەڭگۈلۈك باغلارتە) ئېسىل رىزىققا ئىگە بولىدۇ
وَٱلَّذِينَ ءَامَنُوا۟ مِنۢ بَعْدُ وَهَاجَرُوا۟ وَجَـٰهَدُوا۟ مَعَكُمْ فَأُو۟لَـٰٓئِكَ مِنكُمْ ۚ وَأُو۟لُوا۟ ٱلْأَرْحَامِ بَعْضُهُمْ أَوْلَىٰ بِبَعْضٍ فِى كِتَـٰبِ ٱللَّهِ ۗ إِنَّ ٱللَّهَ بِكُلِّ شَىْءٍ عَلِيمٌۢ ﴿٧٥﴾
كېيىن ئىمان ئېيتىپ (مەدىنىگە) ھىجرەت قىلغانلار ۋە سىلەر بىلەن بىللە جىھاد قىلغانلار سىلەردىن سانىلىدۇ. اﷲ نىڭ ھۆكمىدە، ئۇرۇق - تۇغقانلار بىر - بىرىگە مىراسخور بولۇشقا (ياتلاردىن) ئەڭ ھەقلىقتۇر، شۈبھىسىزكى، اﷲ ھەر شەيئىنى بىلىپ تۇرغۇچىدۇر
