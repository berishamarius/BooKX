ئەلىف، لام، مىم
ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْحَىُّ ٱلْقَيُّومُ ﴿٢﴾
اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، اﷲ (ھەمىشە) تىرىكتۇر، مەخلۇقاتنى ئىدارە قىلىپ تۇرغۇچىدۇر
نَزَّلَ عَلَيْكَ ٱلْكِتَـٰبَ بِٱلْحَقِّ مُصَدِّقًا لِّمَا بَيْنَ يَدَيْهِ وَأَنزَلَ ٱلتَّوْرَىٰةَ وَٱلْإِنجِيلَ ﴿٣﴾
(ئى مۇھەممەد!) ئۆزىدىن ئىلگىرىكى مۇقەددەس كىتاب-ۋەھىيلارنى تەستىق قىلغۇچى ھەق مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) (اﷲ) ساڭا نازىل قىلدى. ئىلگىرى، كىشىلەرگە يول كۆرسەتكۈچى قىلىپ تەۋرات بىلەن ئىنجىلنى نازىل قىلغان ئىدى
مِن قَبْلُ هُدًى لِّلنَّاسِ وَأَنزَلَ ٱلْفُرْقَانَ ۗ إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِـَٔايَـٰتِ ٱللَّهِ لَهُمْ عَذَابٌ شَدِيدٌ ۗ وَٱللَّهُ عَزِيزٌ ذُو ٱنتِقَامٍ ﴿٤﴾
ھەق بىلەن باتىلنى ئايرىغۇچى مۇقەددەس كىتاب-ۋەھىينى (يەنى بارلىق ساماۋى مۇقەددەس كىتاب-ۋەھىيلارنى) نازىل قىلدى. شۈبھىسىزكى، اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغانلار (ئاخىرەتتە) قاتتىق ئازابقا دۇچار بولىدۇ. اﷲ غالىبتۇر، (يامانلارنى) جازالىغۇچىدۇر
إِنَّ ٱللَّهَ لَا يَخْفَىٰ عَلَيْهِ شَىْءٌ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ﴿٥﴾
شۈبھىسىزكى، ئاسمان ۋە زېمىندىكى ھېچ نەرسە اﷲ قا مەخپىي ئەمەس
هُوَ ٱلَّذِى يُصَوِّرُكُمْ فِى ٱلْأَرْحَامِ كَيْفَ يَشَآءُ ۚ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٦﴾
ئۇ سىلەرنى بەچچىدانلاردا (يەنى ئاناڭلارنىڭ قورسىقىدىكى چېغىڭلاردا) ئۆزى خالىغان شەكىلگە كىرگۈزىدۇ. ئۇنىڭدىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
هُوَ ٱلَّذِىٓ أَنزَلَ عَلَيْكَ ٱلْكِتَـٰبَ مِنْهُ ءَايَـٰتٌ مُّحْكَمَـٰتٌ هُنَّ أُمُّ ٱلْكِتَـٰبِ وَأُخَرُ مُتَشَـٰبِهَـٰتٌ ۖ فَأَمَّا ٱلَّذِينَ فِى قُلُوبِهِمْ زَيْغٌ فَيَتَّبِعُونَ مَا تَشَـٰبَهَ مِنْهُ ٱبْتِغَآءَ ٱلْفِتْنَةِ وَٱبْتِغَآءَ تَأْوِيلِهِۦ ۗ وَمَا يَعْلَمُ تَأْوِيلَهُۥٓ إِلَّا ٱللَّهُ ۗ وَٱلرَّٰسِخُونَ فِى ٱلْعِلْمِ يَقُولُونَ ءَامَنَّا بِهِۦ كُلٌّ مِّنْ عِندِ رَبِّنَا ۗ وَمَا يَذَّكَّرُ إِلَّآ أُو۟لُوا۟ ٱلْأَلْبَـٰبِ ﴿٧﴾
(ئى مۇھەممەد!) ئۇ (اﷲ) ساڭا مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) نازىل قىلدى. ئۇنىڭدا مەھكەم (يەنى مەنىسى ئوچۇق) ئايەتلەر باركى، ئۇلار مۇقەددەس كىتاب-ۋەھىينىڭ (يەنى پۈتۈن قۇرئاننىڭ) ئاساسىدۇر؛ يەنە باشقا مۇتەشابىھ (يەنى مەنىسى مۇئەييەن ئەمەس) ئايەتلەر باردۇر. دىللىرىدا ئەگرىلىك بار (يەنى گۇمراھلىققا مايىل) كىشىلەر پىتنە قوزغاش ۋە ئۆز رايى بويىچە مەنە بېرىش ئۈچۈن، مۇتەشابىھ ئايەتلەرگە ئەگىشىدۇ (يەنى مۇتەشابىھ ئايەتلەرنى ئۆز نەپسى خاھىشى بويىچە چۈشەندۈرىدۇ). بۇنداق ئايەتلەرنىڭ (ھەقىقىي) مەنىسىنى پەقەت اﷲ بىلىدۇ. ئىلىمدا توشقانلار ئېيتىدۇ: «ئۇنىڭغا ئىشەندۇق، ھەممىسى پەرۋەردىگارىمىز تەرىپىدىن نازىل بولغان. (بۇنى) پەقەت ئەقىل ئىگىلىرىلا چۈشىنىدۇ
رَبَّنَا لَا تُزِغْ قُلُوبَنَا بَعْدَ إِذْ هَدَيْتَنَا وَهَبْ لَنَا مِن لَّدُنكَ رَحْمَةً ۚ إِنَّكَ أَنتَ ٱلْوَهَّابُ ﴿٨﴾
پەرۋەردىگارىمىز! بىزنى ھىدايەت قىلغىنىڭدىن كېيىن دىللىرىمىزنى توغرا يولدىن بۇرىۋەتمىگىن، بىزگە دەرگاھىڭدىن رەھمەت بېغىشلىغىن. شۈبھىسىزكى، سەن (بەندىلىرىڭگە ئاتالارنى) بەكمۇ بېغىشلىغۇچىسەن
رَبَّنَآ إِنَّكَ جَامِعُ ٱلنَّاسِ لِيَوْمٍ لَّا رَيْبَ فِيهِ ۚ إِنَّ ٱللَّهَ لَا يُخْلِفُ ٱلْمِيعَادَ ﴿٩﴾
پەرۋەردىگارىمىز! شۈبھىسىزكى، كېلىشى شەكسىز بولغان كۈندە (يەنى ھېساب ئېلىنىدىغان قىيامەت كۈنىدە) كىشىلەرنى توپلىغۇچىسەن. شۈبھىسىزكى، اﷲ ۋەدىسىگە خىلاپلىق قىلمايدۇ»
إِنَّ ٱلَّذِينَ كَفَرُوا۟ لَن تُغْنِىَ عَنْهُمْ أَمْوَٰلُهُمْ وَلَآ أَوْلَـٰدُهُم مِّنَ ٱللَّهِ شَيْـًٔا ۖ وَأُو۟لَـٰٓئِكَ هُمْ وَقُودُ ٱلنَّارِ ﴿١٠﴾
ھەقىقەتنى رەت قىلغۇچىلارنىڭ ماللىرى، بالىلىرى اﷲ نىڭ ئازابى ئالدىدا ئىشەنچلىك ھالدا ھېچ نەرسىگە دال بولالمايدۇ. ئۇلار (يەنى كۇففارلار) دوزاخنىڭ يېقىلغۇسىدۇر
كَدَأْبِ ءَالِ فِرْعَوْنَ وَٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَذَّبُوا۟ بِـَٔايَـٰتِنَا فَأَخَذَهُمُ ٱللَّهُ بِذُنُوبِهِمْ ۗ وَٱللَّهُ شَدِيدُ ٱلْعِقَابِ ﴿١١﴾
(بۇ كۇففارلارنىڭ ھالى) پىرئەۋن تەۋەلىرىنىڭ ۋە ئۇلاردىن ئىلگىرىكىلەرنىڭ ھالىغا ئوخشايدۇ. ئۇلار بىزنىڭ ئايەتلىرىمىزنى يالغانغا چىقاردى، گۇناھى تۈپەيلىدىن اﷲ ئۇلارنى جازالىدى، اﷲ نىڭ جازاسى قاتتىقتۇر
قُل لِّلَّذِينَ كَفَرُوا۟ سَتُغْلَبُونَ وَتُحْشَرُونَ إِلَىٰ جَهَنَّمَ ۚ وَبِئْسَ ٱلْمِهَادُ ﴿١٢﴾
(ئى مۇھەممەد!) ھەقىقەتنى رەت قىلغۇچىلارغا ئېيتقىنكى، «سىلەر (دۇنيادا) مەغلۇپ قىلىنىسىلەر ۋە (ئاخىرەتتە) مەڭگۈلۈك ئوتگە توپلىنىسىلەر، مەڭگۈلۈك ئوت نېمىدېگەن يامان تۆشەك»
قَدْ كَانَ لَكُمْ ءَايَةٌ فِى فِئَتَيْنِ ٱلْتَقَتَا ۖ فِئَةٌ تُقَـٰتِلُ فِى سَبِيلِ ٱللَّهِ وَأُخْرَىٰ كَافِرَةٌ يَرَوْنَهُم مِّثْلَيْهِمْ رَأْىَ ٱلْعَيْنِ ۚ وَٱللَّهُ يُؤَيِّدُ بِنَصْرِهِۦ مَن يَشَآءُ ۗ إِنَّ فِى ذَٰلِكَ لَعِبْرَةً لِّأُو۟لِى ٱلْأَبْصَـٰرِ ﴿١٣﴾
(ئى يەھۇدىيلار جامائەسى! بەدرىدە جەڭ ئۈچۈن) ئۇچراشقان ئىككى گۇرۇھتا سىلەر ئۈچۈن ئىبرەت بار، بىرى اﷲ يولىدا ئۇرۇشۇۋاتقان گۇرۇھتۇر، يەنى بىر مۇسۇلمانلارنى (سانىنى) ئۆزلىرىدەك ئىككى باراۋەر ئوپئوچۇق كۆرگەن ھەقىقەتنى رەت قىلغۇچىلار گۇرۇھىدۇر (يەنى اﷲ ھەقىقەتنى رەت قىلغۇچىلارغا قورقۇنچ سېلىش بىلەن ئۇلارنى ئۇرۇشتىن قول يىغدۇرۇش ئۈچۈن، ھەقىقەتنى رەت قىلغۇچىلارغا مۇسۇلمانلارنىڭ سانىنى جىق كۆرسەتتى). اﷲ خالىغان كىشىگە ياردەم بېرىپ قوللايدۇ. بۇنىڭدا ئىشەنچلىك ھالدا ئەقىل ئىگىلىرى ئۈچۈن ئىبرەت بار
زُيِّنَ لِلنَّاسِ حُبُّ ٱلشَّهَوَٰتِ مِنَ ٱلنِّسَآءِ وَٱلْبَنِينَ وَٱلْقَنَـٰطِيرِ ٱلْمُقَنطَرَةِ مِنَ ٱلذَّهَبِ وَٱلْفِضَّةِ وَٱلْخَيْلِ ٱلْمُسَوَّمَةِ وَٱلْأَنْعَـٰمِ وَٱلْحَرْثِ ۗ ذَٰلِكَ مَتَـٰعُ ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَٱللَّهُ عِندَهُۥ حُسْنُ ٱلْمَـَٔابِ ﴿١٤﴾
ئاياللار، ئوغۇللار، ئالتۇن - كۈمۈشتىن توپلانغان كۆپ ماللار، ئارغىماقلار، چارۋىلار ۋە ئېكىنلەردىن ئىبارەت كۆڭۈل تارتىدىغان نەرسىلەرنىڭ مۇھەببىتى ئىنسانلارغا چىرايلىق كۆرسىتىلدى. ئۇلار دۇنيا تىرىكچىلىكىدە مەنپەئەتلىنىدىغان (باقاسى يوق) شەيئىلەردۇر؛ اﷲ نىڭ دەرگاھىدا بولسا قايتىدىغان گۈزەل جاي (يەنى مەڭگۈلۈك باغلار) باردۇر (شۇنىڭ ئۈچۈن باشقىغا ئەمەس، مەڭگۈلۈك باغلاركە قىزىقىش كېرەك)
۞ قُلْ أَؤُنَبِّئُكُم بِخَيْرٍ مِّن ذَٰلِكُمْ ۚ لِلَّذِينَ ٱتَّقَوْا۟ عِندَ رَبِّهِمْ جَنَّـٰتٌ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا وَأَزْوَٰجٌ مُّطَهَّرَةٌ وَرِضْوَٰنٌ مِّنَ ٱللَّهِ ۗ وَٱللَّهُ بَصِيرٌۢ بِٱلْعِبَادِ ﴿١٥﴾
(ئى مۇھەممەد! قەۋمىڭگە) ئېيتقىنكى، «سىلەرگە ئۇلاردىنمۇ (يەنى دۇنيانىڭ زىبۇزىننەتلىرى ۋە نېمەتلىرىدىنمۇ) ياخشى بولغان نەرسىلەرنى ئېيتىپ بېرەيمۇ؟ تەقۋادارلار ئۈچۈن پەرۋەردىگارى ھۇزۇرىدا ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان مەڭگۈلۈك باغلارلەر بولۇپ، (ئۇلار) مەڭگۈلۈك باغلارلەردە مەڭگۈ قالىدۇ، (مەڭگۈلۈك باغلارلەردە) پاك جۈپتىلەر بار، (تەقۋادارلار ئۈچۈن) يەنە اﷲ نىڭ رەزاسى بار». اﷲ بەندىلىرىنى (يەنى ئۇلارنىڭ ھەممە ئىشلىرىنى) كۆرۈپ تۇرغۇچىدۇر
ٱلَّذِينَ يَقُولُونَ رَبَّنَآ إِنَّنَآ ءَامَنَّا فَٱغْفِرْ لَنَا ذُنُوبَنَا وَقِنَا عَذَابَ ٱلنَّارِ ﴿١٦﴾
ئۇلار (يەنى تەقۋادارلار): «پەرۋەردىگارىمىز! بىز شۈبھىسىز ئىمان ئېيتتۇق، بىزنىڭ گۇناھلىرىمىزنى مەغپىرەت قىلغىن، بىزنى دوزاخ ئازابىدىن ساقلىغىن» دەيدۇ
ٱلصَّـٰبِرِينَ وَٱلصَّـٰدِقِينَ وَٱلْقَـٰنِتِينَ وَٱلْمُنفِقِينَ وَٱلْمُسْتَغْفِرِينَ بِٱلْأَسْحَارِ ﴿١٧﴾
ئۇلار (يەنى تەقۋادارلار) سەۋر قىلغۇچىلاردۇر، راستچىللاردۇر، (اﷲ قا) ئىتائەت قىلغۇچىلاردۇر، (ياخشىلىق يوللىرىغا پۇل - ماللىرىنى) سەرپ قىلغۇچىلاردۇر ۋە سەھەرلەردە ئىستىغپار ئېيتقۇچىلاردۇر
شَهِدَ ٱللَّهُ أَنَّهُۥ لَآ إِلَـٰهَ إِلَّا هُوَ وَٱلْمَلَـٰٓئِكَةُ وَأُو۟لُوا۟ ٱلْعِلْمِ قَآئِمًۢا بِٱلْقِسْطِ ۚ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿١٨﴾
اﷲ ئادالەتنى بەرپا قىلغان ھالدا گۇۋاھلىق بەردىكى، ئۇنىڭدىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر. خۇدانىڭ ئەلچى پەرىشتىلىرىمۇ، ئىلىم ئەھلىلىرىمۇ شۇنداق گۇۋاھلىق بەردى: ئۇنىڭدىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر. ئۇ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
إِنَّ ٱلدِّينَ عِندَ ٱللَّهِ ٱلْإِسْلَـٰمُ ۗ وَمَا ٱخْتَلَفَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَـٰبَ إِلَّا مِنۢ بَعْدِ مَا جَآءَهُمُ ٱلْعِلْمُ بَغْيًۢا بَيْنَهُمْ ۗ وَمَن يَكْفُرْ بِـَٔايَـٰتِ ٱللَّهِ فَإِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ﴿١٩﴾
ئىشەنچلىك ھالدا اﷲ نىڭ نەزىرىدە مەقبۇل دىن ئىسلام (دىنى) دۇر. مۇقەددەس كىتاب-ۋەھىي بېرىلگەنلەر (يەنى يەھۇدىيلار ۋە ناسارالار) پەقەت ئۆزلىرىگە ئىلىم كەلگەندىن كېيىنلا ئۆزئارا ھەسەت قىلىشىپ ئىختىلاپ قىلىشتى (يەنى ئىسلامنىڭ ھەقلىقىنى، مۇھەممەد ئەلەيھىسسالامنىڭ خۇدانىڭ ئەلچىسىلىكىنى روشەن پاكىتلار ئارقىلىق بىلىپ تۇرۇپ، كۆرەلمەسلىكتىن ئىنكار قىلىشتى)، كىمكى اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلىدىكەن (اﷲ ئۇنى يېقىندا جازالايدۇ)، اﷲ تېز ھېساب ئالغۇچىدۇر
فَإِنْ حَآجُّوكَ فَقُلْ أَسْلَمْتُ وَجْهِىَ لِلَّهِ وَمَنِ ٱتَّبَعَنِ ۗ وَقُل لِّلَّذِينَ أُوتُوا۟ ٱلْكِتَـٰبَ وَٱلْأُمِّيِّـۧنَ ءَأَسْلَمْتُمْ ۚ فَإِنْ أَسْلَمُوا۟ فَقَدِ ٱهْتَدَوا۟ ۖ وَّإِن تَوَلَّوْا۟ فَإِنَّمَا عَلَيْكَ ٱلْبَلَـٰغُ ۗ وَٱللَّهُ بَصِيرٌۢ بِٱلْعِبَادِ ﴿٢٠﴾
(ئى مۇھەممەد!) ئەگەر ئۇلار سەن بىلەن (دىن بارىسىدا) مۇنازىرەلەشسە: «ماڭا ئەگەشكەنلەر بىلەن بىرلىكتە، مەن ئۆزۈمنى اﷲ قا تاپشۇردۇم (يەنى اﷲ نىڭ ئەمرلىرىگە بويسۇنغۇچىمىز)» دېگىن. مۇقەددەس كىتاب-ۋەھىي بېرىلگەنلەرگە (يەنى يەھۇدىيلار ۋە ناسارالارغا) ۋە ئۈممىلەرگە (يەنى ئەرەب مۇشرىكلىرىغا): «مۇسۇلمان بولدۇڭلارمۇ؟» دېگىن، ئەگەر ئۇلار مۇسۇلمان بولسا، توغرا يول تاپقان بولىدۇ. ئەگەر ئۇلار يۈز ئۆرۈسە (ساڭا ھېچ زىيىنى يوق)، سېنىڭ پەقەت تەبلىغ قىلىش مەسئۇلىيىتىڭ بار، اﷲ بەندىلىرىنى (يەنى ھەممە ئەھۋالىنى) كۆرۈپ تۇرغۇچىدۇر
إِنَّ ٱلَّذِينَ يَكْفُرُونَ بِـَٔايَـٰتِ ٱللَّهِ وَيَقْتُلُونَ ٱلنَّبِيِّـۧنَ بِغَيْرِ حَقٍّ وَيَقْتُلُونَ ٱلَّذِينَ يَأْمُرُونَ بِٱلْقِسْطِ مِنَ ٱلنَّاسِ فَبَشِّرْهُم بِعَذَابٍ أَلِيمٍ ﴿٢١﴾
اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلىدىغانلار، خۇدانىڭ ئەلچى-رەسۇللىرىنى ناھەق ئۆلتۈرىدىغانلار، كىشىلەرنىڭ ئىچىدىكى ئادالەتكە بۇيرۇيدىغانلار (يەنى اﷲ قا دەۋەت قىلغۇچىلار) نى ئۆلتۈرىدىغانلارغا چوقۇم قاتتىق ئازاب بىلەن خۇش خەۋەر بەرگىن
أُو۟لَـٰٓئِكَ ٱلَّذِينَ حَبِطَتْ أَعْمَـٰلُهُمْ فِى ٱلدُّنْيَا وَٱلْـَٔاخِرَةِ وَمَا لَهُم مِّن نَّـٰصِرِينَ ﴿٢٢﴾
ئەنە شۇلارنىڭ دۇنيا ۋە ئاخىرەتتىكى ئەمەللىرى بىكار بولدى، ئۇلارغا ھېچ ياردەم قىلغۇچى يوقتۇر
أَلَمْ تَرَ إِلَى ٱلَّذِينَ أُوتُوا۟ نَصِيبًا مِّنَ ٱلْكِتَـٰبِ يُدْعَوْنَ إِلَىٰ كِتَـٰبِ ٱللَّهِ لِيَحْكُمَ بَيْنَهُمْ ثُمَّ يَتَوَلَّىٰ فَرِيقٌ مِّنْهُمْ وَهُم مُّعْرِضُونَ ﴿٢٣﴾
مۇقەددەس كىتاب-ۋەھىيتن نېسىۋە بېرىلگەنلەر (يەنى يەھۇدىيلارنىڭ ئۆلىمالىرى) نى كۆرمىدىڭمۇ؟ ئۇلار ئۆز ئارىسىدا (دەتالاش قىلىشقان ئىشلار توغرىسىدا) اﷲ نىڭ مۇقەددەس كىتاب-ۋەھىيى بويىچە ھۆكۈم چىقىرىشقا چاقىرىلسا، ئۇلارنىڭ بىر تۈركۈمى بۇنىڭدىن (يەنى اﷲ نىڭ ھۆكىمىنى قوبۇل قىلىشتىن) يۈز ئۆرۈيدۇ، ئۇلار ھەقتىن يۈز ئۆرۈگۈچىلەردۇر (يەنى ھەقتىن يۈز ئۆرۈش ئۇلارنىڭ تەبىئىتىگە سىڭىپ كەتكەن)
ذَٰلِكَ بِأَنَّهُمْ قَالُوا۟ لَن تَمَسَّنَا ٱلنَّارُ إِلَّآ أَيَّامًا مَّعْدُودَٰتٍ ۖ وَغَرَّهُمْ فِى دِينِهِم مَّا كَانُوا۟ يَفْتَرُونَ ﴿٢٤﴾
بۇ ئۇلارنىڭ «دوزاخ ئوتى بىزنى ساناقلىق كۈنلەرلا (يەنى موزايغا چوقۇنغان 40 كۈنلۈك ۋاقىتلا) كۆيدۈرىدۇ» دېگەنلىكلىرى ۋە دىنىي جەھەتتە ئويدۇرۇپ چىقارغان نەرسىلىرىنىڭ ئۆزلىرىنى ئالدىغانلىقى ئۈچۈندۇر
فَكَيْفَ إِذَا جَمَعْنَـٰهُمْ لِيَوْمٍ لَّا رَيْبَ فِيهِ وَوُفِّيَتْ كُلُّ نَفْسٍ مَّا كَسَبَتْ وَهُمْ لَا يُظْلَمُونَ ﴿٢٥﴾
كېلىشى شەكسىز بولغان كۈندە (يەنى قىيامەت كۈنىدە) ئۇلارنى توپلايمىز، ھەر بىر كىشىنىڭ قىلغان ئىشىنىڭ جازاسى ۋە مۇكاپاتى تولۇق بېرىلىدىغان، ئۇلارغا (ئازابنى زىيادە قىلىۋېتىش ياكى ساۋابىنى كېمەيتىۋېتىش بىلەن) زۇلۇم قىلىنمايدىغان شۇ چاغدا (ئۇلارنىڭ ھالى) قانداق بولىدۇ؟
قُلِ ٱللَّهُمَّ مَـٰلِكَ ٱلْمُلْكِ تُؤْتِى ٱلْمُلْكَ مَن تَشَآءُ وَتَنزِعُ ٱلْمُلْكَ مِمَّن تَشَآءُ وَتُعِزُّ مَن تَشَآءُ وَتُذِلُّ مَن تَشَآءُ ۖ بِيَدِكَ ٱلْخَيْرُ ۖ إِنَّكَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٢٦﴾
ئېيتىقىنكى، «پادىشاھلىقنىڭ ئىگىسى بولغان ئى اﷲ! خالىغان ئادەمگە پادىشاھلىقنى بېرىسەن، خالىغان ئادەمدىن پادىشاھلىقنى تارتىپ ئالىسەن؛ خالىغان ئادەمنى ئەزىز قىلىسەن، خالىغان ئادەمنى خار قىلىسەن؛ ھەممە ياخشىلىق (نىڭ خەزىنىسى يالغۇز) سېنىڭ قولۇڭدىدۇر. سەن ئىشەنچلىك ھالدا ھەممىگە قادىرسەن
تُولِجُ ٱلَّيْلَ فِى ٱلنَّهَارِ وَتُولِجُ ٱلنَّهَارَ فِى ٱلَّيْلِ ۖ وَتُخْرِجُ ٱلْحَىَّ مِنَ ٱلْمَيِّتِ وَتُخْرِجُ ٱلْمَيِّتَ مِنَ ٱلْحَىِّ ۖ وَتَرْزُقُ مَن تَشَآءُ بِغَيْرِ حِسَابٍ ﴿٢٧﴾
كېچىنى كۈندۈزگە كىرگۈزىسەن، كۈندۈزنى كېچىگە كىرگۈزىسەن (شۇنىڭ بىلەن، كېچە - كۈندۈزنىڭ ئۇزۇن - قىسقا بولۇشى پەسىللەر بويىچە نۆۋەتلىشىپ تۇرىدۇ)؛ جانلىقنى (يەنى تىرىك بولغان ئىنساننى، ھايۋاننى) جانسىزدىن (يەنى جانلىق بولغان ئابىمەنىيدىن) چىقىرىسەن، جانسىزنى (يەنى جانسىز بولغان ئابىمەنىينى) جانلىقتىن (يەنى جانلىق بولغان ئىنساندىن، ھايۋاندىن) چىقىرىسەن؛ ئۆزۈڭ خالىغان كىشىگە ھېسابسىز رىزىق بېرىسەن»
لَّا يَتَّخِذِ ٱلْمُؤْمِنُونَ ٱلْكَـٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ ۖ وَمَن يَفْعَلْ ذَٰلِكَ فَلَيْسَ مِنَ ٱللَّهِ فِى شَىْءٍ إِلَّآ أَن تَتَّقُوا۟ مِنْهُمْ تُقَىٰةً ۗ وَيُحَذِّرُكُمُ ٱللَّهُ نَفْسَهُۥ ۗ وَإِلَى ٱللَّهِ ٱلْمَصِيرُ ﴿٢٨﴾
مۆمىنلەر مۆمىنلەرنى قويۇپ ھەقىقەتنى رەت قىلغۇچىلارنى دوست تۇتمىسۇن، كىمكى شۇنداق قىلىدىكەن، ئۇ اﷲ نىڭ دوستلۇقىغا ئېرىشەلمەيدۇ، ئۇلاردىن قورقۇپ دوستلۇق ئىزھار قىلىشىڭلار بۇنىڭدىن مۇستەسنا. اﷲ سىلەرنى ئۆزىنىڭ ئازابىدىن قورقۇتىدۇ، ئاخىر قايتىدىغان جاي اﷲ نىڭ دەرگاھىدۇر
قُلْ إِن تُخْفُوا۟ مَا فِى صُدُورِكُمْ أَوْ تُبْدُوهُ يَعْلَمْهُ ٱللَّهُ ۗ وَيَعْلَمُ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٢٩﴾
ئېيتقىنكى، «دىلىڭلاردىكىنى مەيلى يوشۇرۇڭلار، مەيلى ئاشكارىلاڭلار، اﷲ ئۇنى بىلىپ تۇرىدۇ. اﷲ ئاسمانلاردىكى ۋە زېمىندىكى نەرسىلەرنىمۇ بىلىپ تۇرىدۇ. اﷲ ھەر نەرسىگە قادىردۇر»
يَوْمَ تَجِدُ كُلُّ نَفْسٍ مَّا عَمِلَتْ مِنْ خَيْرٍ مُّحْضَرًا وَمَا عَمِلَتْ مِن سُوٓءٍ تَوَدُّ لَوْ أَنَّ بَيْنَهَا وَبَيْنَهُۥٓ أَمَدًۢا بَعِيدًا ۗ وَيُحَذِّرُكُمُ ٱللَّهُ نَفْسَهُۥ ۗ وَٱللَّهُ رَءُوفٌۢ بِٱلْعِبَادِ ﴿٣٠﴾
ئۇ كۈندە (يەنى قىيامەت كۈنىدە) ھەر ئادەمنىڭ قىلغان ياخشى ئىشلىرى، يامان ئىشلىرى ئۆزىنىڭ ئالدىدا ئايان بولىدۇ، (ئۇلار يامان ئىشلىرىنى كۆرمەسلىكنى)، ئۆزى بىلەن يامان ئىشلىرىنىڭ ئارىسىدا (مەشرىق بىلەن مەغرىبنىڭ ئارىسىدەك) ئۇزۇن مۇساپە بولۇشىنى ئارزۇ قىلىدۇ. اﷲ سىلەرنى ئۆزىنىڭ ئازابىدىن قورقۇتىدۇ. اﷲ بەندىلىرىگە مېھرىباندۇر
قُلْ إِن كُنتُمْ تُحِبُّونَ ٱللَّهَ فَٱتَّبِعُونِى يُحْبِبْكُمُ ٱللَّهُ وَيَغْفِرْ لَكُمْ ذُنُوبَكُمْ ۗ وَٱللَّهُ غَفُورٌ رَّحِيمٌ ﴿٣١﴾
(ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «ئەگەر سىلەر اﷲ نى دوست تۇتساڭلار، ماڭا ئەگىشىڭلاركى، اﷲ سىلەرنى دوست تۇتىدۇ، (ئۆتكەنكى) گۇناھىڭلارنى مەغپىرەت قىلىدۇ. اﷲ ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر»
قُلْ أَطِيعُوا۟ ٱللَّهَ وَٱلرَّسُولَ ۖ فَإِن تَوَلَّوْا۟ فَإِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْكَـٰفِرِينَ ﴿٣٢﴾
ئېيتقىنكى، «اﷲ قا ۋە خۇدانىڭ ئەلچىسىگە ئىتائەت قىلىڭلار». ئەگەر ئۇلار (ئىتائەت قىلىشتىن) يۈز ئۆرۈسە، اﷲ ھەقىقەتنى رەت قىلغۇچىلارنى (يەنى اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغانلارنى) دوست تۇتمايدۇ
۞ إِنَّ ٱللَّهَ ٱصْطَفَىٰٓ ءَادَمَ وَنُوحًا وَءَالَ إِبْرَٰهِيمَ وَءَالَ عِمْرَٰنَ عَلَى ٱلْعَـٰلَمِينَ ﴿٣٣﴾
اﷲ ئىشەنچلىك ھالدا ئادەمنى، نۇھنى، ئىبراھىم ئەۋلادىنى، ئىمران ئەۋلادىنى (خۇدانىڭ ئەلچىسىلىككە) تاللاپ، (ئۇلارنى ئۆزلىرىنىڭ زامانىدىكى) جاھان ئەھلىدىن ئارتۇق قىلدى
ذُرِّيَّةًۢ بَعْضُهَا مِنۢ بَعْضٍ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌ ﴿٣٤﴾
ئۇ ئەۋلادلار بىر - بىرى بىلەن قانداشتۇر، اﷲ (بەندىلەرنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (بەندىلەرنىڭ نىيەتلىرىنى) بىلىپ تۇرغۇچىدۇر
إِذْ قَالَتِ ٱمْرَأَتُ عِمْرَٰنَ رَبِّ إِنِّى نَذَرْتُ لَكَ مَا فِى بَطْنِى مُحَرَّرًا فَتَقَبَّلْ مِنِّىٓ ۖ إِنَّكَ أَنتَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٣٥﴾
ئۆز ۋاقتىدا ئىمراننىڭ ئايالى ئېيتتى: «پەرۋەردىگارىم! مەن قورسىقىمدىكى پەرزەنتىمنى (دۇنيا ئىشلىرىدىن) ئازاد قىلىنغان ھالدا چوقۇم سېنىڭ خىزمىتىڭگە ئاتىدىم، (بۇ نەزرەمنى) قوبۇل قىلغىن، سەن ئىشەنچلىك ھالدا (دۇئايىمنى) ئاڭلاپ تۇرغۇچىسەن، (نىيىتىمنى) بىلىپ تۇرغۇچىسەن»
فَلَمَّا وَضَعَتْهَا قَالَتْ رَبِّ إِنِّى وَضَعْتُهَآ أُنثَىٰ وَٱللَّهُ أَعْلَمُ بِمَا وَضَعَتْ وَلَيْسَ ٱلذَّكَرُ كَٱلْأُنثَىٰ ۖ وَإِنِّى سَمَّيْتُهَا مَرْيَمَ وَإِنِّىٓ أُعِيذُهَا بِكَ وَذُرِّيَّتَهَا مِنَ ٱلشَّيْطَـٰنِ ٱلرَّجِيمِ ﴿٣٦﴾
ئۇ قىز تۇغقان ۋاقتىدا: «ئى پەرۋەردىگارىم! قىز تۇغدۇم» دېدى، ئۇنىڭ نېمە تۇغقانلىقىنى اﷲ ئوبدان بىلىدۇ. (ئۇ تەلەپ قىلغان) ئوغۇل (ئاتا قىلىنغان) قىزغا ئوخشاش ئەمەس (بەلكى بۇ قىز ئارتۇقتۇر). (ئىمراننىڭ ئايالى ئېيتتىكى) «مەن ئۇنى مەريەم ئاتىدىم، مەن ئىشەنچلىك ھالدا ئۇنىڭ ئۈچۈن، ئۇنىڭ ئەۋلادلىرى ئۈچۈن ساڭا سېغىنىپ، قوغلاندى شەيتان (ئازدۇرغۇچى)دىن پاناھ تىلەيمەن»
فَتَقَبَّلَهَا رَبُّهَا بِقَبُولٍ حَسَنٍ وَأَنۢبَتَهَا نَبَاتًا حَسَنًا وَكَفَّلَهَا زَكَرِيَّا ۖ كُلَّمَا دَخَلَ عَلَيْهَا زَكَرِيَّا ٱلْمِحْرَابَ وَجَدَ عِندَهَا رِزْقًا ۖ قَالَ يَـٰمَرْيَمُ أَنَّىٰ لَكِ هَـٰذَا ۖ قَالَتْ هُوَ مِنْ عِندِ ٱللَّهِ ۖ إِنَّ ٱللَّهَ يَرْزُقُ مَن يَشَآءُ بِغَيْرِ حِسَابٍ ﴿٣٧﴾
ئۇنى پەرۋەردىگارى چىرايلىقچە قوبۇل قىلدى، ئۇنى ئوبدان ئۆستۈردى، ئۇنىڭغا زەكەرىيانى كېپىل قىلدى (يەنى ئۇنىڭ خىزمىتىنى قىلىشقا زەكەرىيانى قويدى). زەكەرىيا ھەر قېتىم (ئۇنىڭ) ئىبادەتگاھىغا ئۇنى كۆرگىلى كىرسە، ئۇنىڭ يېنىدا بىرەر يېمەكلىك ئۇچرىتاتتى. ئۇ: «ئى مەريەم! بۇ ساڭا قەيەردىن كەلدى؟» دەيتتى، مەريەم: «ئۇ، اﷲ تەرىپىدىن كەلدى» دەيتتى. اﷲ ئىشەنچلىك ھالدا خالىغان ئادەمگە (جاپا - مۇشەققەتسىزلا) ھېسابسىز رىزىق بېرىدۇ
هُنَالِكَ دَعَا زَكَرِيَّا رَبَّهُۥ ۖ قَالَ رَبِّ هَبْ لِى مِن لَّدُنكَ ذُرِّيَّةً طَيِّبَةً ۖ إِنَّكَ سَمِيعُ ٱلدُّعَآءِ ﴿٣٨﴾
شۇ ۋاقىتتا (يەنى مەريەمنىڭ كارامىتىنى كۆرگەن چاغدا) زەكەرىيا پەرۋەردىگارىغا دۇئا قىلىپ: «ئى پەرۋەردىگارىم! ماڭا ئۆز دەرگاھىڭدىن بىر ياخشى پەرزەنت ئاتا قىلغىن، ئىشەنچلىك ھالدا دۇئانى ئىشىتكۈچىسەن» دېدى
فَنَادَتْهُ ٱلْمَلَـٰٓئِكَةُ وَهُوَ قَآئِمٌ يُصَلِّى فِى ٱلْمِحْرَابِ أَنَّ ٱللَّهَ يُبَشِّرُكَ بِيَحْيَىٰ مُصَدِّقًۢا بِكَلِمَةٍ مِّنَ ٱللَّهِ وَسَيِّدًا وَحَصُورًا وَنَبِيًّا مِّنَ ٱلصَّـٰلِحِينَ ﴿٣٩﴾
ئۇ ئىبادەتگاھىدا ناماز ئوقۇۋاتقاندا خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇنىڭغا: «ساڭا اﷲ نىڭ كەلىمىسى (يەنى ئىسا) نى تەستىق قىلغۇچى، سەييىد، پەرھىزكار ۋە ياخشىلاردىن بولغان يەھيا ئىسىملىك بىر خۇدانىڭ ئەلچىسى بىلەن خۇش خەۋەر بېرىدۇ» دەپ نىدا قىلدى
قَالَ رَبِّ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَقَدْ بَلَغَنِىَ ٱلْكِبَرُ وَٱمْرَأَتِى عَاقِرٌ ۖ قَالَ كَذَٰلِكَ ٱللَّهُ يَفْعَلُ مَا يَشَآءُ ﴿٤٠﴾
ئۇ: «پەرۋەردىگارىم! مەن قېرىپ قالغان تۇرسام، ئايالىم تۇغۇتتىن قالغان تۇرسا قانداقمۇ مېنىڭ ئوغلۇم بولسۇن؟» دېدى. خۇدانىڭ ئەلچى پەرىشتىلىرى: «اﷲ شۇنىڭدەك خالىغىنىنى قىلىدۇ (يەنى قۇدرىتى ئىلاھىيە ھېچ نەرسىدىن ئاجىز ئەمەس)» دېدى
قَالَ رَبِّ ٱجْعَل لِّىٓ ءَايَةً ۖ قَالَ ءَايَتُكَ أَلَّا تُكَلِّمَ ٱلنَّاسَ ثَلَـٰثَةَ أَيَّامٍ إِلَّا رَمْزًا ۗ وَٱذْكُر رَّبَّكَ كَثِيرًا وَسَبِّحْ بِٱلْعَشِىِّ وَٱلْإِبْكَـٰرِ ﴿٤١﴾
ئۇ: ‹‹پەرۋەردىگارىم! ماڭا (ئايالىمنىڭ ھامىلىدار بولغانلىقىغا) بىر نىشانە ئاتا قىلغىن» دېدى. اﷲ: ‹‹ساڭا بولىدىغان نىشانە شۇكى، ئۈچ كۈنگىچە كىشىلەرگە سۆز قىلالمايسەن (يەنى زىكرى تەسبىھ ئېيتىشتىن باشقا سۆزگە تىلىڭ كەلمەيدۇ). پەقەت ئىشارەتلا قىلىسەن، (نېمىتىمگە شۈكۈر قىلىش يۈزىسىدىن) پەرۋەردىگارىڭنى كۆپ ياد ئەت، كەچتە، ئەتىگەندە تەسبىھ ئېيت» دېدى
وَإِذْ قَالَتِ ٱلْمَلَـٰٓئِكَةُ يَـٰمَرْيَمُ إِنَّ ٱللَّهَ ٱصْطَفَىٰكِ وَطَهَّرَكِ وَٱصْطَفَىٰكِ عَلَىٰ نِسَآءِ ٱلْعَـٰلَمِينَ ﴿٤٢﴾
خۇدانىڭ ئەلچى پەرىشتىلىرى ئېيتتى: «ئى مەريەم! اﷲ ئىشەنچلىك ھالدا سېنى (پۈتۈن ئاياللار ئىچىدىن) تاللىدى، سېنى پاك قىلدى، سېنى پۈتۈن جاھان ئاياللىرىدىن ئارتۇق قىلدى
يَـٰمَرْيَمُ ٱقْنُتِى لِرَبِّكِ وَٱسْجُدِى وَٱرْكَعِى مَعَ ٱلرَّٰكِعِينَ ﴿٤٣﴾
ئى مەريەم! پەرۋەردىگارىڭغا ئىتائەت قىل، سەجدە قىل ۋە رۇكۇ قىلغۇچىلار بىلەن بىللە رۇكۇ قىل»
ذَٰلِكَ مِنْ أَنۢبَآءِ ٱلْغَيْبِ نُوحِيهِ إِلَيْكَ ۚ وَمَا كُنتَ لَدَيْهِمْ إِذْ يُلْقُونَ أَقْلَـٰمَهُمْ أَيُّهُمْ يَكْفُلُ مَرْيَمَ وَمَا كُنتَ لَدَيْهِمْ إِذْ يَخْتَصِمُونَ ﴿٤٤﴾
(ئى مۇھەممەد!) بۇ - ساڭا ۋەھيى قىلىۋاتقىنىمىز - غەيب خەۋەرلىرىدىندۇر. مەريەمنى ئۇلارنىڭ قايسىسى تەربىيىسىگە ئېلىش (مەسىلىسىدە چەك تاشلاش يۈزىسىدىن) قەلەملىرىنى (سۇغا) تاشلىغان چاغدا، سەن ئۇلارنىڭ يېنىدا يوق ئىدىڭ، (مەريەمنىڭ كىمنىڭ تەربىيىسىدە بولۇشىنى) ئۆزئارا جاڭجال قىلىشقانلىرىدا سەن ئۇلارنىڭ يېنىدا يوق ئىدىڭ
إِذْ قَالَتِ ٱلْمَلَـٰٓئِكَةُ يَـٰمَرْيَمُ إِنَّ ٱللَّهَ يُبَشِّرُكِ بِكَلِمَةٍ مِّنْهُ ٱسْمُهُ ٱلْمَسِيحُ عِيسَى ٱبْنُ مَرْيَمَ وَجِيهًا فِى ٱلدُّنْيَا وَٱلْـَٔاخِرَةِ وَمِنَ ٱلْمُقَرَّبِينَ ﴿٤٥﴾
ئۆز ۋاقتىدا خۇدانىڭ ئەلچى پەرىشتىلىرى ئېيتتى: «ئى مەريەم! اﷲ ساڭا (ئاتىنىڭ ۋاسىتىسىز) اﷲ نىڭ بىر كەلىمىسى (دىن تۆرەلگەن بىر بوۋاق) بىلەن خۇش خەۋەر بېرىدۇكى، ئۇنىڭ ئىسمى، مەسىھ مەريەم ئوغلى ئىسادۇر، ئۇ دۇنيا ۋە ئاخىرەتتە ئابرۇيلۇق ۋە اﷲ قا يېقىنلاردىن بولىدۇ
وَيُكَلِّمُ ٱلنَّاسَ فِى ٱلْمَهْدِ وَكَهْلًا وَمِنَ ٱلصَّـٰلِحِينَ ﴿٤٦﴾
ئۇ بۆشۈكتىمۇ (يەنى بوۋاقلىق چېغىدىمۇ)، ئوتتۇرا ياش بولغاندىمۇ كىشىلەرگە (خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ سۆزىنى) سۆزلەيدۇ ۋە (تەقۋادارلىقتا كامىل) ياخشى ئادەملەردىن بولىدۇ»
قَالَتْ رَبِّ أَنَّىٰ يَكُونُ لِى وَلَدٌ وَلَمْ يَمْسَسْنِى بَشَرٌ ۖ قَالَ كَذَٰلِكِ ٱللَّهُ يَخْلُقُ مَا يَشَآءُ ۚ إِذَا قَضَىٰٓ أَمْرًا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ﴿٤٧﴾
مەريەم: «پەرۋەردىگارىم! ماڭا ئىنساننىڭ قولى تەگمىگەن تۇرسا (يەنى ئەرلىك بولمىسام) قانداق بالام بولىدۇ؟» دېدى. پەرىشتە: «اﷲ خالىغىنىنى شۇنداق يارىتىدۇ (يەنى ئاتا - ئانا، ئارقىلىقمۇ ۋە ئۇنىڭسىزمۇ يارىتىدۇ). بىرەر ئىشنىڭ بولۇشىنى ئىرادە قىلسا، ۋۇجۇدقا كېلىدۇ» دېدى
وَيُعَلِّمُهُ ٱلْكِتَـٰبَ وَٱلْحِكْمَةَ وَٱلتَّوْرَىٰةَ وَٱلْإِنجِيلَ ﴿٤٨﴾
اﷲ ئۇنىڭغا خەتنى، ھېكمەتنى (يەنى سۆز ھەرىكەتتە توغرا بولۇشىنى)، تەۋراتنى، ئىنجىلنى ئۆگىتىدۇ
وَرَسُولًا إِلَىٰ بَنِىٓ إِسْرَٰٓءِيلَ أَنِّى قَدْ جِئْتُكُم بِـَٔايَةٍ مِّن رَّبِّكُمْ ۖ أَنِّىٓ أَخْلُقُ لَكُم مِّنَ ٱلطِّينِ كَهَيْـَٔةِ ٱلطَّيْرِ فَأَنفُخُ فِيهِ فَيَكُونُ طَيْرًۢا بِإِذْنِ ٱللَّهِ ۖ وَأُبْرِئُ ٱلْأَكْمَهَ وَٱلْأَبْرَصَ وَأُحْىِ ٱلْمَوْتَىٰ بِإِذْنِ ٱللَّهِ ۖ وَأُنَبِّئُكُم بِمَا تَأْكُلُونَ وَمَا تَدَّخِرُونَ فِى بُيُوتِكُمْ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ ﴿٤٩﴾
ئۇنى بەنى ئىسرائىلغا خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتىدۇ. (ئۇ دەيدۇ) «شۈبھىسىزكى، سىلەرگە مەن رەببىڭلار تەرىپىدىن بولغان (خۇدانىڭ ئەلچىسىلىكىمنىڭ راستلىقىنى كۆرسىتىدىغان) بىر مۆجىزە ئېلىپ كەلدىم، مەن سىلەرگە لايدىن قۇشنىڭ شەكلىدەك بىر نەرسە ياسايمەن، ئاندىن ئۇنىڭغا پۈۋلەيمەن - دە، اﷲ نىڭ ئىزنى بىلەن، ئۇ قۇش بولىدۇ. تۇغما كورنى، بەرەس كېسىلىنى ساقايتىمەن، اﷲ نىڭ ئىزنى بىلەن، ئۆلۈكلەرنى تىرىلدۈرىمەن، سىلەر يەيدىغان ۋە ئۆيۈڭلاردا ساقلايدىغان يېمەكلىكلەردىن خەۋەر بېرىمەن (يەنى سىلەرنىڭ يوشۇرۇن ئەھۋالىڭلارنى ئېيتىپ بېرەلەيمەن). شۈبھىسىزكى، ئەگەر سىلەر اﷲ نىڭ مۆجىزىلىرىگە ئىشەنگۈچى بولساڭلار، بۇنىڭدا (يەنى مەن كەلتۈرگەن مۆجىزىلەردە) سىلەر ئۈچۈن ئەلۋەتتە (مېنىڭ راستلىقىمنى كۆرسىتىدىغان روشەن) ئالامەت بار
وَمُصَدِّقًا لِّمَا بَيْنَ يَدَىَّ مِنَ ٱلتَّوْرَىٰةِ وَلِأُحِلَّ لَكُم بَعْضَ ٱلَّذِى حُرِّمَ عَلَيْكُمْ ۚ وَجِئْتُكُم بِـَٔايَةٍ مِّن رَّبِّكُمْ فَٱتَّقُوا۟ ٱللَّهَ وَأَطِيعُونِ ﴿٥٠﴾
مەن ئىلگىرى كەلگەن تەۋراتنى تەستىق قىلغان ھالدا (كەلدىم)، سىلەرگە پەرۋەردىگارىڭلار تەرىپىدىن دەلىل ئېلىپ كەلدىم. اﷲ تىن قورقۇڭلار ۋە (مېنىڭ ئەمرىمگە) ئىتائەت قىلىڭلار
إِنَّ ٱللَّهَ رَبِّى وَرَبُّكُمْ فَٱعْبُدُوهُ ۗ هَـٰذَا صِرَٰطٌ مُّسْتَقِيمٌ ﴿٥١﴾
اﷲ ئىشەنچلىك ھالدا مېنىڭ پەرۋەردىگارىمدۇر، سىلەرنىڭمۇ پەرۋەردىگارىڭلاردۇر، ئۇنىڭغا ئىبادەت قىلىڭلار، بۇ توغرا يولدۇر»
۞ فَلَمَّآ أَحَسَّ عِيسَىٰ مِنْهُمُ ٱلْكُفْرَ قَالَ مَنْ أَنصَارِىٓ إِلَى ٱللَّهِ ۖ قَالَ ٱلْحَوَارِيُّونَ نَحْنُ أَنصَارُ ٱللَّهِ ءَامَنَّا بِٱللَّهِ وَٱشْهَدْ بِأَنَّا مُسْلِمُونَ ﴿٥٢﴾
ئىسا ئۇلارنى (يەنى يەھۇدىيلار) دىن كۇفرىنى (كۇفرىدا چىڭ تۇرۇپ، ئۇنى ئۆلتۈرۈش نىيىتىگە كەلگەنلىكىنى) سەزگەن چاغدا: «اﷲ ئۈچۈن (اﷲ قا دەۋەت قىلىش يولىدا) ماڭا كىملەر ياردەم بېرىدۇ؟» دېدى. ھەۋارىلار (يەنى ئۇنىڭ تەۋەلىرىدىن ھەقىقىي مۆمىنلەر) ئېيتتى: «اﷲ ئۈچۈن ساڭا بىز ياردەم بېرىمىز، اﷲ قا ئىمان ئېيتتۇق. گۇۋاھ بولغىنكى، بىز (خۇدانىڭ ئەلچىسىلىكىڭگە) بويسۇنغۇچىلارمىز
رَبَّنَآ ءَامَنَّا بِمَآ أَنزَلْتَ وَٱتَّبَعْنَا ٱلرَّسُولَ فَٱكْتُبْنَا مَعَ ٱلشَّـٰهِدِينَ ﴿٥٣﴾
پەرۋەردىگارىمىز! سەن نازىل قىلغان مۇقەددەس كىتاب-ۋەھىيقا ئىشەندۇق، خۇدانىڭ ئەلچىسىگە ئەگەشتۇق، بىزنى (ھىدايىتىڭگە) شاھىت بولغانلار قاتارىدا قىلغىن»
وَمَكَرُوا۟ وَمَكَرَ ٱللَّهُ ۖ وَٱللَّهُ خَيْرُ ٱلْمَـٰكِرِينَ ﴿٥٤﴾
ئۇلار مىكىر ئىشلەتتى (يەنى ئىسا ئەلەيھىسسالامنى ئۆلتۈرمەكچى بولدى)، اﷲ ئۇلارنىڭ مىكرىنى بەربات قىلدى (ئىسا ئەلەيھىسسالامنى قۇتقۇزۇپ ئاسمانغا ئېلىپ چىقىپ كەتتى). اﷲ مىكىر قىلغۇچىلارنىڭ جازاسىنى ئوبدان بەرگۈچىدۇر
إِذْ قَالَ ٱللَّهُ يَـٰعِيسَىٰٓ إِنِّى مُتَوَفِّيكَ وَرَافِعُكَ إِلَىَّ وَمُطَهِّرُكَ مِنَ ٱلَّذِينَ كَفَرُوا۟ وَجَاعِلُ ٱلَّذِينَ ٱتَّبَعُوكَ فَوْقَ ٱلَّذِينَ كَفَرُوٓا۟ إِلَىٰ يَوْمِ ٱلْقِيَـٰمَةِ ۖ ثُمَّ إِلَىَّ مَرْجِعُكُمْ فَأَحْكُمُ بَيْنَكُمْ فِيمَا كُنتُمْ فِيهِ تَخْتَلِفُونَ ﴿٥٥﴾
ئۆز ۋاقتىدا اﷲ ئېيتتى: «ئى ئىسا! مەن سېنى (ئەجىلىڭ يەتكەندە) قەبزى روھ قىلىمەن، سېنى دەرگاھىمغا كۆتۈرىمەن (يەنى ئاسمانغا ئېلىپ چىقىمەن). سېنى ھەقىقەتنى رەت قىلغۇچىلاردىن پاك قىلىمەن (يەنى سېنى ئۆلتۈرمەكچى بولغان يامانلارنىڭ شەررىدىن ساقلايمەن)، ساڭا ئەگەشكەنلەرنى قىيامەتكىچە ھەقىقەتنى رەت قىلغۇچىلاردىن ئۈستۈن قىلىمەن. ئاندىن مېنىڭ دەرگاھىمغا قايتىسىلەر، سىلەر ئىختىلاپ قىلىشقان ئىش (يەنى ئىسانىڭ ئىشى) ئۈستىدە ئاراڭلاردا مەن ھۆكۈم چىقىرىمەن
فَأَمَّا ٱلَّذِينَ كَفَرُوا۟ فَأُعَذِّبُهُمْ عَذَابًا شَدِيدًا فِى ٱلدُّنْيَا وَٱلْـَٔاخِرَةِ وَمَا لَهُم مِّن نَّـٰصِرِينَ ﴿٥٦﴾
ھەقىقەتنى رەت قىلغۇچىلارغا بولسا دۇنيا ۋە ئاخىرەتتە قاتتىق ئازاب قىلىمەن، ئۇلارغا (ئۇلاردىن اﷲ نىڭ ئازابىنى توسىدىغان) ھېچ مەدەتكار بولمايدۇ»
وَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فَيُوَفِّيهِمْ أُجُورَهُمْ ۗ وَٱللَّهُ لَا يُحِبُّ ٱلظَّـٰلِمِينَ ﴿٥٧﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەل قىلغانلارغا كەلسەك، اﷲ ئۇلارنىڭ (قىلغان ياخشى ئەمەللىرىنىڭ) ئەجرىنى تولۇق بېرىدۇ. اﷲ زالىملارنى دوست تۇتمايدۇ (زالىملارنى دوست تۇتمايدىغان زات قانداقمۇ بەندىلىرىگە زۇلۇم قىلسۇن؟)
ذَٰلِكَ نَتْلُوهُ عَلَيْكَ مِنَ ٱلْـَٔايَـٰتِ وَٱلذِّكْرِ ٱلْحَكِيمِ ﴿٥٨﴾
(ئى مۇھەممەد!) بۇ بولسا ساڭا بىز ئوقۇپ بېرىۋاتقان ئايەتلەردۇر ۋە ھېكمەتلىك قۇرئاندۇر
إِنَّ مَثَلَ عِيسَىٰ عِندَ ٱللَّهِ كَمَثَلِ ءَادَمَ ۖ خَلَقَهُۥ مِن تُرَابٍ ثُمَّ قَالَ لَهُۥ كُن فَيَكُونُ ﴿٥٩﴾
شۈبھىسىزكى، اﷲ نىڭ نەزىرىدە ئىسانىڭ مىسالى (ئۇ ئاتىسىز يارىتىلغانلىقتىن) ئادەمنىڭ (يەنى ئادەم ئەلەيھىسسالامنىڭ) مىسالىغا ئوخشايدۇ. ئادەمنى اﷲ (ئاتا - ئانىسىز) تۇپراقتىن ياراتتى، ئاندىن ئۇنىڭغا: «ۋۇجۇدقا كەل» دېدى - دە، ئۇ ۋۇجۇدقا كەلدى (ئىسانىڭ ئىشى ئادەمنىڭ ئىشىدىن ئەجەبلىنەرلىك ئەمەس)
ٱلْحَقُّ مِن رَّبِّكَ فَلَا تَكُن مِّنَ ٱلْمُمْتَرِينَ ﴿٦٠﴾
(ئىسا ھەققىدىكى بۇ سۆز) اﷲ تەرىپىدىن نازىل بولغان ھەقىقەتتۇر، سەن شەك قىلغۇچىلاردىن بولمىغىن
فَمَنْ حَآجَّكَ فِيهِ مِنۢ بَعْدِ مَا جَآءَكَ مِنَ ٱلْعِلْمِ فَقُلْ تَعَالَوْا۟ نَدْعُ أَبْنَآءَنَا وَأَبْنَآءَكُمْ وَنِسَآءَنَا وَنِسَآءَكُمْ وَأَنفُسَنَا وَأَنفُسَكُمْ ثُمَّ نَبْتَهِلْ فَنَجْعَل لَّعْنَتَ ٱللَّهِ عَلَى ٱلْكَـٰذِبِينَ ﴿٦١﴾
سەن ئىسا توغرۇلۇق ھەقىقىي مەلۇماتقا ئىگە بولغىنىڭدىن كېيىن، كىملەركى سەن بىلەن مۇنازىرىلەشسە، سەن ئۇلارغا: «كېلىڭلار، ئوغۇللىرىمىزنى ۋە ئوغۇللىرىڭلارنى، ئاياللىرىمىزنى ۋە ئاياللىرىڭلارنى، ئۆزلىرىمىزنى ۋە ئۆزلىرىڭلارنى يىغىپ، اﷲ نىڭ لەنىتى يالغانچىلارغا بولسۇن، دەپ اﷲ قا يالۋۇرۇپ دۇئا قىلايلى» دېگىن
إِنَّ هَـٰذَا لَهُوَ ٱلْقَصَصُ ٱلْحَقُّ ۚ وَمَا مِنْ إِلَـٰهٍ إِلَّا ٱللَّهُ ۚ وَإِنَّ ٱللَّهَ لَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٦٢﴾
شۈبھىسىزكى، بۇ ئەلۋەتتە راست قىسسىدۇر. بىر اﷲ تىن باشقا ھېچ ئىلاھ يوقتۇر، اﷲ ئىشەنچلىك ھالدا غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
فَإِن تَوَلَّوْا۟ فَإِنَّ ٱللَّهَ عَلِيمٌۢ بِٱلْمُفْسِدِينَ ﴿٦٣﴾
ئەگەر ئۇلار (ئىماندىن) يۈز ئۆرۈسە (ئۇلار بۇزغۇنچىلاردۇر)، شۈبھىسىزكى، اﷲ بۇزغۇنچىلارنى بىلىپ تۇرىدۇ
قُلْ يَـٰٓأَهْلَ ٱلْكِتَـٰبِ تَعَالَوْا۟ إِلَىٰ كَلِمَةٍ سَوَآءٍۭ بَيْنَنَا وَبَيْنَكُمْ أَلَّا نَعْبُدَ إِلَّا ٱللَّهَ وَلَا نُشْرِكَ بِهِۦ شَيْـًٔا وَلَا يَتَّخِذَ بَعْضُنَا بَعْضًا أَرْبَابًا مِّن دُونِ ٱللَّهِ ۚ فَإِن تَوَلَّوْا۟ فَقُولُوا۟ ٱشْهَدُوا۟ بِأَنَّا مُسْلِمُونَ ﴿٦٤﴾
«ئى ئەھلى مۇقەددەس كىتاب-ۋەھىي! (يەنى يەھۇدىيلار ۋە ناسارالار) پەقەت اﷲ غىلا ئىبادەت قىلىش، اﷲ قا ھېچ نەرسىنى شېرىك كەلتۈرمەسلىك، اﷲ نى قويۇپ بىر - بىرىمىزنى خۇدا قىلىۋالماسلىقتەك ھەممىمىزگە ئورتاق بولغان بىر سۆزگە (يەنى بىرخىل ئەقىدىگە) ئەمەل قىلايلى» دېگىن، ئەگەر ئۇلار يۈز ئۆرۈسە (يەنى بۇنى قوبۇل قىلمىسا): «(ئى يەھۇدىيلار ۋە ناسارالار جامائەسى!) بىزنىڭ مۇسۇلمان ئىكەنلىكىمىزگە گۇۋاھ بولۇڭلار
يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تُحَآجُّونَ فِىٓ إِبْرَٰهِيمَ وَمَآ أُنزِلَتِ ٱلتَّوْرَىٰةُ وَٱلْإِنجِيلُ إِلَّا مِنۢ بَعْدِهِۦٓ ۚ أَفَلَا تَعْقِلُونَ ﴿٦٥﴾
ئى ئەھلى مۇقەددەس كىتاب-ۋەھىي! (يەنى يەھۇدىيلار ۋە ناسارالار) نېمە ئۈچۈن ئىبراھىم توغرىسىدا مۇنازىرىلىشىسىلەر؟ (ئىبراھىمنى ئۆزۈڭلارنىڭ دىنىدا دەپ گۇمان قىلىسىلەر؟) ھالبۇكى، تەۋرات بىلەن ئىنجىل ئىبراھىمدىن كېيىن نازىل بولدى. (سۆزۈڭلارنىڭ ئاساسسىز ئىكەنلىكىنى) چۈشەنمەمسىلەر؟
هَـٰٓأَنتُمْ هَـٰٓؤُلَآءِ حَـٰجَجْتُمْ فِيمَا لَكُم بِهِۦ عِلْمٌ فَلِمَ تُحَآجُّونَ فِيمَا لَيْسَ لَكُم بِهِۦ عِلْمٌ ۚ وَٱللَّهُ يَعْلَمُ وَأَنتُمْ لَا تَعْلَمُونَ ﴿٦٦﴾
(ئى يەھۇدىيلار ۋە ناسارالار جامائەسى!) سىلەر شۇنداق كىشىلەر سىلەركى، بىلىدىغان نەرسەڭلار (ئىسا) ئۈستىدە مۇنازىرىلەشتىڭلار، ئەمدى سىلەر بىلمەيدىغان نەرسەڭلار (ئىبراھىم ۋە ئۇنىڭ دىنى) ئۈستىدم نېمىشقا مۇنازىرىلىشىسىلەر؟ (ئىبراھىمنىڭ ئىشىدىكى ھەقىقەتنى) اﷲ بىلىدۇ، سىلەر بىلمەيسىلەر» دەڭلار
مَا كَانَ إِبْرَٰهِيمُ يَهُودِيًّا وَلَا نَصْرَانِيًّا وَلَـٰكِن كَانَ حَنِيفًا مُّسْلِمًا وَمَا كَانَ مِنَ ٱلْمُشْرِكِينَ ﴿٦٧﴾
ئىبراھىم يەھۇدىيمۇ ئەمەس، ناسارامۇ ئەمەس ۋە لېكىن توغرا دىنغا ئېتىقاد قىلغۇچى، اﷲ قا بويسۇنغۇچى ئىدى، مۇشرىكلاردىن ئەمەس ئىدى
إِنَّ أَوْلَى ٱلنَّاسِ بِإِبْرَٰهِيمَ لَلَّذِينَ ٱتَّبَعُوهُ وَهَـٰذَا ٱلنَّبِىُّ وَٱلَّذِينَ ءَامَنُوا۟ ۗ وَٱللَّهُ وَلِىُّ ٱلْمُؤْمِنِينَ ﴿٦٨﴾
ئىبراھىمغا ئەڭ يېقىن بولغانلار، شۈبھىسىزكى، (ئىبراھىمنىڭ زامانىدا ۋە ئۇنىڭدىن كېيىن) ئۇنىڭغا (يەنى ئۇنىڭ يولىغا) ئەگەشكەنلەردۇر، بۇ خۇدانىڭ ئەلچىسى (يەنى مۇھەممەد ئەلەيھىسسالام) دۇر، (بۇ خۇدانىڭ ئەلچىسىنىڭ ئۈممىتى بولغان) مۆمىنلەردۇر. اﷲ مۆمىنلەرنىڭ مەدەتكارىدۇر
وَدَّت طَّآئِفَةٌ مِّنْ أَهْلِ ٱلْكِتَـٰبِ لَوْ يُضِلُّونَكُمْ وَمَا يُضِلُّونَ إِلَّآ أَنفُسَهُمْ وَمَا يَشْعُرُونَ ﴿٦٩﴾
ئەھلى مۇقەددەس كىتاب-ۋەھىيتىن بىر تۈركۈم كىشىلەر سىلەرنى ئازدۇرۇشنى ئۈمىد قىلىدۇ، ئۇلار پەقەت ئۆزلىرىنىلا ئازدۇرىدۇ (بۇنىڭ ۋابالى ئۇلارغا ھەسسىلەپ ئازاب قىلىنىش بىلەن ئۆزلىرىگىلا قايتىدۇ). (لېكىن) ئۇلار (بۇنى) تۇيمايدۇ
يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَكْفُرُونَ بِـَٔايَـٰتِ ٱللَّهِ وَأَنتُمْ تَشْهَدُونَ ﴿٧٠﴾
ئى ئەھلى مۇقەددەس كىتاب-ۋەھىي! اﷲ نىڭ ئايەتلىرىنى (يەنى مۇھەممەد ئەلەيھىسسالامغا نازىل قىلىنغان قۇرئاننىڭ ھەقلىقىنى) بىلىپ تۇرۇپ نېمىشقا ئىنكار قىلىسىلەر؟
يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَلْبِسُونَ ٱلْحَقَّ بِٱلْبَـٰطِلِ وَتَكْتُمُونَ ٱلْحَقَّ وَأَنتُمْ تَعْلَمُونَ ﴿٧١﴾
ئى ئەھلى مۇقەددەس كىتاب-ۋەھىي! نېمە ئۈچۈن ھەقنى باتىلغا ئارىلاشتۇرىسىلەر ۋە بىلىپ تۇرۇپ ھەقنى (يەنى مۇھەممەد ئەلەيھىسسالامنىڭ مۇقەددەس كىتاب-ۋەھىيلىرىڭلاردىكى سۈپىتىنى) يوشۇرىسىلەر؟
وَقَالَت طَّآئِفَةٌ مِّنْ أَهْلِ ٱلْكِتَـٰبِ ءَامِنُوا۟ بِٱلَّذِىٓ أُنزِلَ عَلَى ٱلَّذِينَ ءَامَنُوا۟ وَجْهَ ٱلنَّهَارِ وَٱكْفُرُوٓا۟ ءَاخِرَهُۥ لَعَلَّهُمْ يَرْجِعُونَ ﴿٧٢﴾
ئەھلى مۇقەددەس كىتاب-ۋەھىيتىن بىر تۈركۈمى: «مۆمىنلەرگە نازىل قىلىنغان ۋەھيىگە ئەتىگەندە ئىمان ئېيتىڭلار، كەچقۇرۇنلۇقى يېنىۋېلىڭلار، (شۇنداق قىلساڭلار) ئۇلار (ئىماندىن) قايتىشى مۇمكىن. پەقەت سىلەرنىڭ دىنىڭلارغا ئەگەشكەنلەرگىلا ئىشىنىڭلار» دېدى. (ئى مۇھەممەد!) ئېيتقىنكى، «توغرا يول اﷲ نىڭ يولىدۇر». (يەھۇدىيلار بى - بىرىگە ئېيتىدۇ) بىرەرسىگە سىلەرگە بېرىلگەنگە ئوخشاش ۋەھىي بېرىلىشتىن ياكى ئۇلارنىڭ (يەنى مۇسۇلمانلارنىڭ) (قىيامەت كۈنى) پەرۋەردىگارىڭلارنىڭ دەرگاھىدا سىلەر بىلەن مۇنازىرىلىشىشىدىن (يەنى مۇنازىرىلىشىپ سىلەرنى يېڭىپ قويۇشىدىن قورقۇپ) پەقەت دىنىڭلارغا ئەگەشكەن ئادەمگىلا ئىشىنىڭلار (دىنىڭلاردا بولمىغان ھېچ ئادەمنىڭ خۇدانىڭ ئەلچىسىلىكىنى ئېتىراپ قىلماڭلار. ئەگەر سىلەر مۇھەممەدنىڭ خۇدانىڭ ئەلچىسىلىكىنى ئىقرار قىلىپ ئۇنىڭ دىنىغا كىرمىسەڭلار، بۇ قىيامەت كۈنى سىلەرگە قارشى پاكىت بولىدۇ). (ئى مۇھەممەد!) ئېيتقىنكى، «پەزل - كەرەم (يەنى خۇدانىڭ ئەلچىسىلىك) اﷲ نىڭ ئىلكىدىدۇر. اﷲ ئۇنى خالىغان كىشىگە بېرىدۇ». اﷲ (نىڭ پەزلى) كەڭدۇر، (اﷲ) ھەممىنى بىلگۈچىدۇر
وَلَا تُؤْمِنُوٓا۟ إِلَّا لِمَن تَبِعَ دِينَكُمْ قُلْ إِنَّ ٱلْهُدَىٰ هُدَى ٱللَّهِ أَن يُؤْتَىٰٓ أَحَدٌ مِّثْلَ مَآ أُوتِيتُمْ أَوْ يُحَآجُّوكُمْ عِندَ رَبِّكُمْ ۗ قُلْ إِنَّ ٱلْفَضْلَ بِيَدِ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۗ وَٱللَّهُ وَٰسِعٌ عَلِيمٌ ﴿٧٣﴾
ئەھلى مۇقەددەس كىتاب-ۋەھىيتىن بىر تۈركۈمى: «مۆمىنلەرگە نازىل قىلىنغان ۋەھيىگە ئەتىگەندە ئىمان ئېيتىڭلار، كەچقۇرۇنلۇقى يېنىۋېلىڭلار، (شۇنداق قىلساڭلار) ئۇلار (ئىماندىن) قايتىشى مۇمكىن. پەقەت سىلەرنىڭ دىنىڭلارغا ئەگەشكەنلەرگىلا ئىشىنىڭلار» دېدى. (ئى مۇھەممەد!) ئېيتقىنكى، «توغرا يول اﷲ نىڭ يولىدۇر». (يەھۇدىيلار بى - بىرىگە ئېيتىدۇ) بىرەرسىگە سىلەرگە بېرىلگەنگە ئوخشاش ۋەھىي بېرىلىشتىن ياكى ئۇلارنىڭ (يەنى مۇسۇلمانلارنىڭ) (قىيامەت كۈنى) پەرۋەردىگارىڭلارنىڭ دەرگاھىدا سىلەر بىلەن مۇنازىرىلىشىشىدىن (يەنى مۇنازىرىلىشىپ سىلەرنى يېڭىپ قويۇشىدىن قورقۇپ) پەقەت دىنىڭلارغا ئەگەشكەن ئادەمگىلا ئىشىنىڭلار (دىنىڭلاردا بولمىغان ھېچ ئادەمنىڭ خۇدانىڭ ئەلچىسىلىكىنى ئېتىراپ قىلماڭلار. ئەگەر سىلەر مۇھەممەدنىڭ خۇدانىڭ ئەلچىسىلىكىنى ئىقرار قىلىپ ئۇنىڭ دىنىغا كىرمىسەڭلار، بۇ قىيامەت كۈنى سىلەرگە قارشى پاكىت بولىدۇ). (ئى مۇھەممەد!) ئېيتقىنكى، «پەزل - كەرەم (يەنى خۇدانىڭ ئەلچىسىلىك) اﷲ نىڭ ئىلكىدىدۇر. اﷲ ئۇنى خالىغان كىشىگە بېرىدۇ». اﷲ (نىڭ پەزلى) كەڭدۇر، (اﷲ) ھەممىنى بىلگۈچىدۇر
يَخْتَصُّ بِرَحْمَتِهِۦ مَن يَشَآءُ ۗ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٧٤﴾
اﷲ رەھمىتىنى ئۆزى خالىغان كىشىگە خاس قىلىدۇ، اﷲ ئۇلۇغ پەزل ئىگىسىدۇر
۞ وَمِنْ أَهْلِ ٱلْكِتَـٰبِ مَنْ إِن تَأْمَنْهُ بِقِنطَارٍ يُؤَدِّهِۦٓ إِلَيْكَ وَمِنْهُم مَّنْ إِن تَأْمَنْهُ بِدِينَارٍ لَّا يُؤَدِّهِۦٓ إِلَيْكَ إِلَّا مَا دُمْتَ عَلَيْهِ قَآئِمًا ۗ ذَٰلِكَ بِأَنَّهُمْ قَالُوا۟ لَيْسَ عَلَيْنَا فِى ٱلْأُمِّيِّـۧنَ سَبِيلٌ وَيَقُولُونَ عَلَى ٱللَّهِ ٱلْكَذِبَ وَهُمْ يَعْلَمُونَ ﴿٧٥﴾
ئەھلى مۇقەددەس كىتاب-ۋەھىيتىن بەزى كىشىلەر باركى، ئۇنىڭدا كۆپ مال ئامانەت قويساڭمۇ ساڭا ئۇنى قايتۇرۇپ بېرىدۇ؛ ئۇلاردىن يەنە بەزى كىشىلەرمۇ باركى، ئۇنىڭدا بىر دىنار ئامانەت قويساڭمۇ ئۇنىڭ بېشىدا تۇرمىغىچە (يەنى قايتا - قايتا سۈيلىمىگىچە) ئۇنى ساڭا قايتۇرۇپ بەرمەيدۇ. بۇ شۇنىڭ ئۈچۈنكى، ئۇلار: «ئۈممىلەرنىڭ مېلىغا خىيانەت قىلساق بىزگە گۇناھ بولمايدۇ (يەنى بىزنىڭ دىنىمىزدا بولمىغانلارنىڭ ماللىرى بىزگە ھارام ئەمەس)» دېدى، ئۇلار بىلىپ تۇرۇپ اﷲ نامىدىن يالغاننى توقۇيدۇ
بَلَىٰ مَنْ أَوْفَىٰ بِعَهْدِهِۦ وَٱتَّقَىٰ فَإِنَّ ٱللَّهَ يُحِبُّ ٱلْمُتَّقِينَ ﴿٧٦﴾
ئۇنداق ئەمەس (يەنى ئۇلار باشقا دىندىكىلەرگە زۇلۇم قىلسا گۇناھ بولىدۇ). ئەھدىگە ۋاپا قىلغان ۋە گۇناھتىن ساقلانغانلار (اﷲ دوست تۇتقان كىشىلەردۇر)، چۈنكى اﷲ تەقۋادارلارنى ئىشەنچلىك ھالدا دوست تۇتىدۇ
إِنَّ ٱلَّذِينَ يَشْتَرُونَ بِعَهْدِ ٱللَّهِ وَأَيْمَـٰنِهِمْ ثَمَنًا قَلِيلًا أُو۟لَـٰٓئِكَ لَا خَلَـٰقَ لَهُمْ فِى ٱلْـَٔاخِرَةِ وَلَا يُكَلِّمُهُمُ ٱللَّهُ وَلَا يَنظُرُ إِلَيْهِمْ يَوْمَ ٱلْقِيَـٰمَةِ وَلَا يُزَكِّيهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌ ﴿٧٧﴾
اﷲ قا بەرگەن ئەھدىنى ۋە قەسەملىرىنى ئازغىنا بەدەلگە تېگىشىدىغانلار ئاخىرەتتە (اﷲ نىڭ رەھمىتىدىن) ھېچقانداق نېسىۋىگە ئېرىشەلمەيدۇ، قىيامەت كۈنى اﷲ ئۇلارغا (ئۇلارنى خۇش قىلىدىغان) سۆز قىلمايدۇ، ئۇلارغا (رەھمەت كۆزى بىلەن) قارىمايدۇ، ئۇلارنى (گۇناھلىرىدىن) پاكلىمايدۇ، ئۇلار قاتتىق ئازابقا قالىدۇ
وَإِنَّ مِنْهُمْ لَفَرِيقًا يَلْوُۥنَ أَلْسِنَتَهُم بِٱلْكِتَـٰبِ لِتَحْسَبُوهُ مِنَ ٱلْكِتَـٰبِ وَمَا هُوَ مِنَ ٱلْكِتَـٰبِ وَيَقُولُونَ هُوَ مِنْ عِندِ ٱللَّهِ وَمَا هُوَ مِنْ عِندِ ٱللَّهِ وَيَقُولُونَ عَلَى ٱللَّهِ ٱلْكَذِبَ وَهُمْ يَعْلَمُونَ ﴿٧٨﴾
ئۇلاردىن (يەنى يەھۇدىيلاردىن) بىر تۈركۈمى مۇقەددەس كىتاب-ۋەھىيتا بولمىغان نەرسىلەرنى، سىلەرنىڭ مۇقەددەس كىتاب-ۋەھىيتا بار ئىكەن دەپ ئويلىشىڭلار ئۈچۈن، تىللىرىنى ئەگرى - بۈگرى قىلىپ ئوقۇيدۇ ۋە ئۇنى اﷲ تەرىپىدىن نازىل بولغان دەيدۇ. ھالبۇكى، ئۇ اﷲ تەرىپىدىن نازىل بولغان ئەمەس، ئۇلار بىلىپ تۇرۇپ اﷲ نامىدىن يالغاننى توقۇيدۇ
مَا كَانَ لِبَشَرٍ أَن يُؤْتِيَهُ ٱللَّهُ ٱلْكِتَـٰبَ وَٱلْحُكْمَ وَٱلنُّبُوَّةَ ثُمَّ يَقُولَ لِلنَّاسِ كُونُوا۟ عِبَادًا لِّى مِن دُونِ ٱللَّهِ وَلَـٰكِن كُونُوا۟ رَبَّـٰنِيِّـۧنَ بِمَا كُنتُمْ تُعَلِّمُونَ ٱلْكِتَـٰبَ وَبِمَا كُنتُمْ تَدْرُسُونَ ﴿٧٩﴾
اﷲ بىرەر ئىنسانغا مۇقەددەس كىتاب-ۋەھىينى، ھېكمەتنى، خۇدانىڭ ئەلچىسىلىكنى بەرگەندىن كېيىن، ئۇنىڭ كىشىلەرگە: «اﷲ نى قويۇپ ماڭا بەندە بولۇڭلار» دېيىشى لايىق بولمايدۇ (يەنى خۇدانىڭ ئەلچىسىدىن مۇنداق گەپنىڭ سادىر بولۇشىنى تەسەۋۋۇر قىلغىلى بولمايدۇ) ۋە لېكىن (ئۇ): «مۇقەددەس كىتاب-ۋەھىينى ئۆگەتكەنلىكىڭلار، ئوقۇغانلىقىڭلار ئۈچۈن رەببانى بولۇڭلار» (دەيدۇ)
وَلَا يَأْمُرَكُمْ أَن تَتَّخِذُوا۟ ٱلْمَلَـٰٓئِكَةَ وَٱلنَّبِيِّـۧنَ أَرْبَابًا ۗ أَيَأْمُرُكُم بِٱلْكُفْرِ بَعْدَ إِذْ أَنتُم مُّسْلِمُونَ ﴿٨٠﴾
شۇنىڭدەك ئۇنىڭ خۇدانىڭ ئەلچى پەرىشتىلىرىنى، خۇدانىڭ ئەلچى-رەسۇللىرىنى رەب قىلىۋېلىشقا بۇيرۇشى لايىق بولمايدۇ. مۇسۇلمان بولغىنىڭلاردىن كېيىن پەيغەمبىرىڭلار سىلەرنى كۇفرىغا بۇيرۇمدۇ؟
وَإِذْ أَخَذَ ٱللَّهُ مِيثَـٰقَ ٱلنَّبِيِّـۧنَ لَمَآ ءَاتَيْتُكُم مِّن كِتَـٰبٍ وَحِكْمَةٍ ثُمَّ جَآءَكُمْ رَسُولٌ مُّصَدِّقٌ لِّمَا مَعَكُمْ لَتُؤْمِنُنَّ بِهِۦ وَلَتَنصُرُنَّهُۥ ۚ قَالَ ءَأَقْرَرْتُمْ وَأَخَذْتُمْ عَلَىٰ ذَٰلِكُمْ إِصْرِى ۖ قَالُوٓا۟ أَقْرَرْنَا ۚ قَالَ فَٱشْهَدُوا۟ وَأَنَا۠ مَعَكُم مِّنَ ٱلشَّـٰهِدِينَ ﴿٨١﴾
ئۆز ۋاقتىدىلا اﷲ خۇدانىڭ ئەلچى-رەسۇللىرىدىن چىن ئەھدە ئېلىپ: «سىلەرگە مەن مۇقەددەس كىتاب-ۋەھىينى ۋە ھېكمەتنى ئاتا قىلدىم، كېيىن سىلەرگە، سىلەردىكى نەرسىلەرنى (يەنى مۇقەددەس كىتاب-ۋەھىي بىلەن ھېكمەتنى) ئېتىراپ قىلغۇچى بىر خۇدانىڭ ئەلچىسى (يەنى مۇھەممەد ئەلەيھىسسالام) كەلسە ئۇنىڭغا، ئەلۋەتتە، ئىمان ئېيتىشىڭلار كېرەك ۋە ئۇنىڭغا ئەلۋەتتە ياردەم بېرىشىڭلار كېرەك» (دېدى). اﷲ: «(بۇ ئەھدىنى) ئىقرار (يەنى ئېتىراپ) قىلدىڭلارمۇ؟ شۇنىڭ (يەنە بۇ ئىش) ئۈچۈن مېنىڭ ئەھدىمنى قوبۇل قىلدىڭلارمۇ؟» دېدى. ئۇلار: «ئېتىراپ قىلدۇق» دېدى. اﷲ: «(ئۆزۈڭلارنىڭ ۋە تەۋەلىرىڭلارنىڭ ئېتىراپ قىلغانلىقىغا) گۇۋاھ بولۇڭلار، مەنمۇ سىلەر بىلەن بىللە گۇۋاھ بولغۇچىلاردىندۇرمەن» دېدى
فَمَن تَوَلَّىٰ بَعْدَ ذَٰلِكَ فَأُو۟لَـٰٓئِكَ هُمُ ٱلْفَـٰسِقُونَ ﴿٨٢﴾
كىمكى شۇنىڭدىن كېيىن يۈز ئۆرۈسە (يەنى ئەھدىنى بۇزسا)، ئۇلار اﷲ نىڭ ئىتائىتىدىن چىققۇچىلاردۇر
أَفَغَيْرَ دِينِ ٱللَّهِ يَبْغُونَ وَلَهُۥٓ أَسْلَمَ مَن فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ طَوْعًا وَكَرْهًا وَإِلَيْهِ يُرْجَعُونَ ﴿٨٣﴾
ئۇلار (يەنى يەھۇدىيلار ۋە ناسارالار) اﷲ نىڭ دىنىدىن (يەنى ئىسلام دىنىدىن) باشقا دىننى تىلەمدۇ؟ ھالبۇكى، ئاسمانلاردىكىلەر ئىختىيارىي ۋە ئىختىيارسىز رەۋىشتە اﷲ قا بويسۇنغان تۇرسا، ئۇلار (قىيامەت كۈنى) اﷲ تەرىپىگە قايتۇرۇلىدۇ
قُلْ ءَامَنَّا بِٱللَّهِ وَمَآ أُنزِلَ عَلَيْنَا وَمَآ أُنزِلَ عَلَىٰٓ إِبْرَٰهِيمَ وَإِسْمَـٰعِيلَ وَإِسْحَـٰقَ وَيَعْقُوبَ وَٱلْأَسْبَاطِ وَمَآ أُوتِىَ مُوسَىٰ وَعِيسَىٰ وَٱلنَّبِيُّونَ مِن رَّبِّهِمْ لَا نُفَرِّقُ بَيْنَ أَحَدٍ مِّنْهُمْ وَنَحْنُ لَهُۥ مُسْلِمُونَ ﴿٨٤﴾
(ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «بىز اﷲ قا ئىمان ئېيتتۇق، بىزگە نازىل قىلىنغانغا (يەنى قۇرئانغا)، ئىبراھىمغا، ئىسمائىلغا، ئىسھاققا، يەئقۇبقا ۋە ئۇنىڭ ئەۋلادلىرىغا نازىل قىلىنغان ۋەھيىگە، مۇساغا بېرىلگەنگە (يەنى ئىنجىلغا) ۋە خۇدانىڭ ئەلچى-رەسۇللىرىگە پەرۋەردىگارى تەرىپىدىن بېرىلگەنگە (يەنى مۇقەددەس كىتاب-ۋەھىيلارغا) ئىمان ئېيتتۇق، ئۇلارنىڭ ھېچبىرىنى ئايرىۋەتمەيمىز (يەنى ئۇلارنىڭ بەزىسىگە ئىمان ئېيتىپ، بەزىسىنى ئىنكار قىلمايمىز)، بىز اﷲ قا بويسۇنغۇچىلارمىز»
وَمَن يَبْتَغِ غَيْرَ ٱلْإِسْلَـٰمِ دِينًا فَلَن يُقْبَلَ مِنْهُ وَهُوَ فِى ٱلْـَٔاخِرَةِ مِنَ ٱلْخَـٰسِرِينَ ﴿٨٥﴾
كىمكى ئىسلام (دىنىدىن) غەيرىي دىننى تىلەيدىكەن، ھەرگىز ئۇ (يەنى ئۇنىڭ دىنى) قوبۇل قىلىنمايدۇ، ئۇ ئاخىرەتتە زىيان تارتقۇچىدۇر
كَيْفَ يَهْدِى ٱللَّهُ قَوْمًا كَفَرُوا۟ بَعْدَ إِيمَـٰنِهِمْ وَشَهِدُوٓا۟ أَنَّ ٱلرَّسُولَ حَقٌّ وَجَآءَهُمُ ٱلْبَيِّنَـٰتُ ۚ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّـٰلِمِينَ ﴿٨٦﴾
ئىمان ئېيتقاندىن، خۇدانىڭ ئەلچىسىنىڭ ھەق ئىكەنلىكىگە شاھىت بولغاندىن ۋە ئۆزلىرىگە روشەن دەلىللەر كەلگەندىن كېيىن ھەقىقەتنى رەت قىلغۇچى بولغان قەۋمنى اﷲ قانداقمۇ ھىدايەت قىلسۇن! اﷲ زالىم قەۋمنى ھىدايەت قىلمايدۇ
أُو۟لَـٰٓئِكَ جَزَآؤُهُمْ أَنَّ عَلَيْهِمْ لَعْنَةَ ٱللَّهِ وَٱلْمَلَـٰٓئِكَةِ وَٱلنَّاسِ أَجْمَعِينَ ﴿٨٧﴾
اﷲ نىڭ، خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ ۋە ئىنسانلارنىڭ ھەممىسىنىڭ لەنىتىگە ئۇچراش ئۇلارغا بېرىلگەن جازادۇر
خَـٰلِدِينَ فِيهَا لَا يُخَفَّفُ عَنْهُمُ ٱلْعَذَابُ وَلَا هُمْ يُنظَرُونَ ﴿٨٨﴾
ئۇلار دوزاختا مەڭگۈ قالىدۇ. ئۇلاردىن ئازاب يېنىكلىتىلمەيدۇ. كېچىكتۈرۈلمەيدۇ
إِلَّا ٱلَّذِينَ تَابُوا۟ مِنۢ بَعْدِ ذَٰلِكَ وَأَصْلَحُوا۟ فَإِنَّ ٱللَّهَ غَفُورٌ رَّحِيمٌ ﴿٨٩﴾
پەقەت تەۋبە قىلغانلار ۋە (بۇزۇق ئەمەللىرىنى) تۈزەتكەنلەر بۇنىڭدىن مۇستەسنا، اﷲ (مۇنداقلارغا) مەغپىرەت قىلغۇچىدۇر ۋە مېھرىباندۇر
إِنَّ ٱلَّذِينَ كَفَرُوا۟ بَعْدَ إِيمَـٰنِهِمْ ثُمَّ ٱزْدَادُوا۟ كُفْرًا لَّن تُقْبَلَ تَوْبَتُهُمْ وَأُو۟لَـٰٓئِكَ هُمُ ٱلضَّآلُّونَ ﴿٩٠﴾
شۈبھىسىزكى، ئىمان ئېيتقاندىن كېيىن ھەقىقەتنى رەت قىلغۇچى بولغانلار، ئاندىن كۇفرىنى تېخىمۇ ئاشۇرغانلارنىڭ تەۋبىسى ھەرگىزمۇ قوبۇل قىلىنمايدۇ. ئەنە شۇلار گۇمراھلاردۇر
إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَمَاتُوا۟ وَهُمْ كُفَّارٌ فَلَن يُقْبَلَ مِنْ أَحَدِهِم مِّلْءُ ٱلْأَرْضِ ذَهَبًا وَلَوِ ٱفْتَدَىٰ بِهِۦٓ ۗ أُو۟لَـٰٓئِكَ لَهُمْ عَذَابٌ أَلِيمٌ وَمَا لَهُم مِّن نَّـٰصِرِينَ ﴿٩١﴾
شۈبھىسىزكى، ھەقىقەتنى رەت قىلغۇچى بولۇپ (تەۋبە قىلماستىن) كۇفرى بىلەن ئۆلگەنلەرنىڭ ھېچبىرىدىن يەر يۈزىگە بىر كېلىدىغان ئالتۇن فىدىيە بەرسىمۇ قوبۇل قىلىنمايدۇ، ئەنە شۇلار قاتتىق ئازابقا قالىدۇ. ئۇلارغا (ئۇلارنى اﷲ نىڭ ئازابىدىن قۇتۇلدۇرىدىغان) ھېچبىر مەدەتكار بولمايدۇ
لَن تَنَالُوا۟ ٱلْبِرَّ حَتَّىٰ تُنفِقُوا۟ مِمَّا تُحِبُّونَ ۚ وَمَا تُنفِقُوا۟ مِن شَىْءٍ فَإِنَّ ٱللَّهَ بِهِۦ عَلِيمٌ ﴿٩٢﴾
ياخشى كۆرگەن نەرسەڭلاردىن سەرپ قىلمىغۇچە (يەنى ماللىرىڭلارنىڭ ياخشىسىنى سەدىقە قىلمىغۇچە) ھەرگىز ياخشىلىققا (يەنى مەڭگۈلۈك باغلاركە) ئېرىشەلمەيسىلەر، (اﷲ نىڭ يولىدا) نېمىنى سەرپ قىلماڭلار. اﷲ ئۇنى بىلىپ تۇرغۇچىدۇر
۞ كُلُّ ٱلطَّعَامِ كَانَ حِلًّا لِّبَنِىٓ إِسْرَٰٓءِيلَ إِلَّا مَا حَرَّمَ إِسْرَٰٓءِيلُ عَلَىٰ نَفْسِهِۦ مِن قَبْلِ أَن تُنَزَّلَ ٱلتَّوْرَىٰةُ ۗ قُلْ فَأْتُوا۟ بِٱلتَّوْرَىٰةِ فَٱتْلُوهَآ إِن كُنتُمْ صَـٰدِقِينَ ﴿٩٣﴾
تەۋرات نازىل قىلىنىشتىن ئىلگىرى، ئىسرائىل (يەنى يەئقۇب ئەلەيھىسسالام) ئۆزىگە ھارام قىلغان يېمەكلىكلەر (يەنى تۆگىنىڭ گۆشى ۋە سۈتى) دىن باشقا يېمەكلىكلەرنىڭ ھەممىسى ئىسرائىل ئەۋلادىغا ھالال ئىدى. (ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «ئەگەر راستچىل بولساڭلار، تەۋراتنى ئېلىپ كېلىپ ئوقۇپ بېقىڭلار»
فَمَنِ ٱفْتَرَىٰ عَلَى ٱللَّهِ ٱلْكَذِبَ مِنۢ بَعْدِ ذَٰلِكَ فَأُو۟لَـٰٓئِكَ هُمُ ٱلظَّـٰلِمُونَ ﴿٩٤﴾
شۇنىڭدىن كېيىن (يەنى روشەن دەلىل مەيدانغا كەلگەندىن كېيىن)، كىمكى اﷲ نامىدىن يالغاننى توقۇيدىكەن، ئۇلار زالىملاردۇر
قُلْ صَدَقَ ٱللَّهُ ۗ فَٱتَّبِعُوا۟ مِلَّةَ إِبْرَٰهِيمَ حَنِيفًا وَمَا كَانَ مِنَ ٱلْمُشْرِكِينَ ﴿٩٥﴾
ئېيتقىنكى، «اﷲ راست ئېيتتى، (ئىسلام دىنىدىن ئىبارەت) ئىبراھىمنىڭ توغرا دىنىغا ئەگىشىڭلار، ئۇ (يەنى ئىبراھىم) مۇشرىكلاردىن ئەمەس ئىدى»
إِنَّ أَوَّلَ بَيْتٍ وُضِعَ لِلنَّاسِ لَلَّذِى بِبَكَّةَ مُبَارَكًا وَهُدًى لِّلْعَـٰلَمِينَ ﴿٩٦﴾
ئىشەنچلىك ھالدا ئىنسانلارغا (ئىبادەت ئۈچۈن) تۇنجى سېلىنغان ئۆي (يەنى بەيتۇللا) مەككىدىدۇر، مۇبارەكتۇر، جاھان ئەھلىگە ھىدايەتتۇر
فِيهِ ءَايَـٰتٌۢ بَيِّنَـٰتٌ مَّقَامُ إِبْرَٰهِيمَ ۖ وَمَن دَخَلَهُۥ كَانَ ءَامِنًا ۗ وَلِلَّهِ عَلَى ٱلنَّاسِ حِجُّ ٱلْبَيْتِ مَنِ ٱسْتَطَاعَ إِلَيْهِ سَبِيلًا ۚ وَمَن كَفَرَ فَإِنَّ ٱللَّهَ غَنِىٌّ عَنِ ٱلْعَـٰلَمِينَ ﴿٩٧﴾
ئۇنىڭدا ئوچۇق ئالامەتلەر باركى، ماقامى ئىبراھىم شۇلارنىڭ بىرى، بەيتۇللاغا كىرگەن ئادەم ئەمىن بولىدۇ. قادىر بولالىغان كىشىلەرنىڭ اﷲ ئۈچۈن كەبىنى زىيارەت قىلىشى ئۇلارغا پەرز قىلىندى. كىمكى ئىنكار قىلىدىكەن (يەنى ھەجنى تەرك ئېتىدىكەن، زىيىنى ئۆزىگە)، شۈبھىسىزكى، اﷲ ئەھلى جاھاندىن (يەنى ئۇلارنىڭ ئىبادىتىدىن) بىھاجەتتۇر
قُلْ يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَكْفُرُونَ بِـَٔايَـٰتِ ٱللَّهِ وَٱللَّهُ شَهِيدٌ عَلَىٰ مَا تَعْمَلُونَ ﴿٩٨﴾
ئى مۇھەممەد! ئېيتقىنكى: «ئى ئەھلى مۇقەددەس كىتاب-ۋەھىي! (يەنى يەھۇدىيلار، ناسارالار) نېمە ئۈچۈن اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلىسىلەر؟ ۋەھالەنكى، اﷲ قىلمىشىڭلارغا شاھىتتۇر»
قُلْ يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لِمَ تَصُدُّونَ عَن سَبِيلِ ٱللَّهِ مَنْ ءَامَنَ تَبْغُونَهَا عِوَجًا وَأَنتُمْ شُهَدَآءُ ۗ وَمَا ٱللَّهُ بِغَـٰفِلٍ عَمَّا تَعْمَلُونَ ﴿٩٩﴾
(ئى مۇھەممەد!) ئېيتقىنكى، «ئى ئەھلى مۇقەددەس كىتاب-ۋەھىي! (يەنى يەھۇدىيلار، ناسارالار) (ئىسلام دىنىنىڭ ھەقلىقىغا) شاھىت بولۇپ تۇرۇپ نېمە ئۈچۈن اﷲ نىڭ يولىنى ئەگرى كۆرسىتىشنى قەستلەپ، مۆمىنلەرنى ئۇنىڭدىن توسىسىلەر؟ اﷲ قىلمىشىڭلاردىن غاپىل ئەمەستۇر»
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِن تُطِيعُوا۟ فَرِيقًا مِّنَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَـٰبَ يَرُدُّوكُم بَعْدَ إِيمَـٰنِكُمْ كَـٰفِرِينَ ﴿١٠٠﴾
ئى مۆمىنلەر! ئەگەر ئەھلى مۇقەددەس كىتاب-ۋەھىيتىن بىر پىرقىغە ئىتائەت قىلساڭلار، مۆمىن بولغىنىڭلاردىن كېيىن، ئۇلار سىلەرنى ھەقىقەتنى رەت قىلغۇچى قىلىپ قويىدۇ
وَكَيْفَ تَكْفُرُونَ وَأَنتُمْ تُتْلَىٰ عَلَيْكُمْ ءَايَـٰتُ ٱللَّهِ وَفِيكُمْ رَسُولُهُۥ ۗ وَمَن يَعْتَصِم بِٱللَّهِ فَقَدْ هُدِىَ إِلَىٰ صِرَٰطٍ مُّسْتَقِيمٍ ﴿١٠١﴾
سىلەرگە اﷲ نىڭ ئايەتلىرى ئوقۇلۇۋاتقان (يەنى ۋەھيى ئۈزۈلۈپ قالماي اﷲ نىڭ ئايەتلىرى نازىل بولۇۋاتقان) ۋە ئاراڭلاردا اﷲ نىڭ پەيغەمبىرى (ھايات) تۇرغان تۇرسا، قانداقمۇ ھەقىقەتنى رەت قىلغۇچى بولۇسىلەر؟ كىمكى اﷲ قا (يەنى اﷲ نىڭ دىنىغا) مەھكەم يېپىشىدىكەن، ئۇ توغرا يولغا يېتەكلەنگەن بولىدۇ
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ حَقَّ تُقَاتِهِۦ وَلَا تَمُوتُنَّ إِلَّا وَأَنتُم مُّسْلِمُونَ ﴿١٠٢﴾
ئى مۆمىنلەر! اﷲ قا لايىق رەۋىشتە تەقۋادارلىق قىلىڭلار، پەقەت مۇسۇلمانلىق ھالىتىڭلار بىلەنلا ۋاپات بولۇڭلار
وَٱعْتَصِمُوا۟ بِحَبْلِ ٱللَّهِ جَمِيعًا وَلَا تَفَرَّقُوا۟ ۚ وَٱذْكُرُوا۟ نِعْمَتَ ٱللَّهِ عَلَيْكُمْ إِذْ كُنتُمْ أَعْدَآءً فَأَلَّفَ بَيْنَ قُلُوبِكُمْ فَأَصْبَحْتُم بِنِعْمَتِهِۦٓ إِخْوَٰنًا وَكُنتُمْ عَلَىٰ شَفَا حُفْرَةٍ مِّنَ ٱلنَّارِ فَأَنقَذَكُم مِّنْهَا ۗ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَـٰتِهِۦ لَعَلَّكُمْ تَهْتَدُونَ ﴿١٠٣﴾
ھەممىڭلار اﷲ نىڭ ئاغامچىسىغا (يەنى اﷲ نىڭ دىنىغا) مەھكەم يېپىشىڭلار، ئايرىلماڭلار (يەنى سىلەردىن ئىلگىرى يەھۇدىيلار ۋە ناسارالار ئىختىلاپ قىلىشقاندەك، دىندا ئىختىلاپ قىلىشماڭلار). اﷲ نىڭ سىلەرگە بەرگەن نېمىتىنى ئەسلەڭلار، ئۆز ۋاقتىدا سىلەر ئۆزئارا دۈشمەن ئىدىڭلار، اﷲ دىلىڭلارنى بىرلەشتۈردى، اﷲ نىڭ نېمىتى بىلەن ئۆزئارا قېرىنداش بولدۇڭلار، سىلەر دوزاخ چۇقۇرىنىڭ گىرۋىكىدە ئىدىڭلار، اﷲ سىلەرنى (ئىسلام ئارقىلىق) ئۇنىڭدىن قۇتقۇزدى. سىلەرنىڭ ھىدايەت تېپىشىڭلار ئۈچۈن، اﷲ ئايەتلىرىنى سىلەرگە شۇنداق بايان قىلىدۇ
وَلْتَكُن مِّنكُمْ أُمَّةٌ يَدْعُونَ إِلَى ٱلْخَيْرِ وَيَأْمُرُونَ بِٱلْمَعْرُوفِ وَيَنْهَوْنَ عَنِ ٱلْمُنكَرِ ۚ وَأُو۟لَـٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿١٠٤﴾
سىلەرنىڭ ئاراڭلاردا خەيرلىك ئىشلارغا دەۋەت قىلىدىغان، ياخشى ئىشلارغا بۇيرۇپ، يامان ئىشلارنى مەنئى قىلىدىغان بىر جامائە بولسۇن؛ ئەنە شۇلار مەقسىتىگە ئېرىشكۈچىلەردۇر
وَلَا تَكُونُوا۟ كَٱلَّذِينَ تَفَرَّقُوا۟ وَٱخْتَلَفُوا۟ مِنۢ بَعْدِ مَا جَآءَهُمُ ٱلْبَيِّنَـٰتُ ۚ وَأُو۟لَـٰٓئِكَ لَهُمْ عَذَابٌ عَظِيمٌ ﴿١٠٥﴾
روشەن دەلىللەر كەلگەندىن كېيىن ئايرىلىپ كېتىشكەن ۋە ئىختىلاپ قىلىشقان كىشىلەر (يەنى يەھۇدىيلار ۋە ناسارالار) دەك بولماڭلار، ئەنە شۇلار چوڭ ئازابقا دۇچار بولىدۇ
يَوْمَ تَبْيَضُّ وُجُوهٌ وَتَسْوَدُّ وُجُوهٌ ۚ فَأَمَّا ٱلَّذِينَ ٱسْوَدَّتْ وُجُوهُهُمْ أَكَفَرْتُم بَعْدَ إِيمَـٰنِكُمْ فَذُوقُوا۟ ٱلْعَذَابَ بِمَا كُنتُمْ تَكْفُرُونَ ﴿١٠٦﴾
شۇ كۈندە (يەنى قىيامەت كۈنىدە) بەزى يۈزلەر (يەنى مۆمىنلەرنىڭ يۈزلىرى) ئاقىرىدۇ، بەزى يۈزلەر (يەنى كۇففارلارنىڭ يۈزلىرى) قارىيىدۇ. يۈزى قارايغانلارغا (خۇدانىڭ ئەلچى پەرىشتىلىرى ئېيتىدۇكى) «ئىمان ئېيتقىنىڭلاردىن كېيىن ھەقىقەتنى رەت قىلغۇچى بولدۇڭلارمۇ؟ ھەقىقەتنى رەت قىلغۇچى بولغانلىقىڭلار ئۈچۈن ئازابنى تېتىڭلار»
وَأَمَّا ٱلَّذِينَ ٱبْيَضَّتْ وُجُوهُهُمْ فَفِى رَحْمَةِ ٱللَّهِ هُمْ فِيهَا خَـٰلِدُونَ ﴿١٠٧﴾
يۈزى ئاقارغانلار اﷲ نىڭ رەھمىتىدە (يەنى مەڭگۈلۈك باغلارتە) بولىدۇ، ئۇلار ئۇنىڭدا (يەنى مەڭگۈلۈك باغلارتە) مەڭگۈ قالىدۇ
تِلْكَ ءَايَـٰتُ ٱللَّهِ نَتْلُوهَا عَلَيْكَ بِٱلْحَقِّ ۗ وَمَا ٱللَّهُ يُرِيدُ ظُلْمًا لِّلْعَـٰلَمِينَ ﴿١٠٨﴾
بۇ، اﷲ نىڭ ئايەتلىرىدۇر، ئۇلارنى (ئى مۇھەممەد!) ساڭا راست ئوقۇپ بېرىمىز، اﷲ جاھان ئەھلىگە زۇلۇم قىلىشنى خالىمايدۇ
وَلِلَّهِ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ وَإِلَى ٱللَّهِ تُرْجَعُ ٱلْأُمُورُ ﴿١٠٩﴾
ئاسمانلار ۋە زېمىندىكى شەيئىلەر اﷲ نىڭ (مۈلكى، مەخلۇقاتى ۋە بەندىلىرى) دۇر، ھەممە ئىشلار اﷲ قا قايتۇرۇلىدۇ
كُنتُمْ خَيْرَ أُمَّةٍ أُخْرِجَتْ لِلنَّاسِ تَأْمُرُونَ بِٱلْمَعْرُوفِ وَتَنْهَوْنَ عَنِ ٱلْمُنكَرِ وَتُؤْمِنُونَ بِٱللَّهِ ۗ وَلَوْ ءَامَنَ أَهْلُ ٱلْكِتَـٰبِ لَكَانَ خَيْرًا لَّهُم ۚ مِّنْهُمُ ٱلْمُؤْمِنُونَ وَأَكْثَرُهُمُ ٱلْفَـٰسِقُونَ ﴿١١٠﴾
(ئى مۇھەممەد ئۈممىتى!) سىلەر ئىنسانلار مەنپەئىتى ئۈچۈن ئوتتۇرىغا چىقىرىلغان ياخشىلىققا بۇيرۇپ يامانلىقتىن توسىدىغان اﷲ قا ئىمان ئېيتىدىغان ئەڭ ياخشى ئۈممەتىسلەر. ئەگەر ئەھلى مۇقەددەس كىتاب-ۋەھىي (يەنى يەھۇدىيلار، ناسارالار) ئىمان ئېيتسا (يەنى مۇھەممەد ئەلەيھىسسالامغا نازىل قىلىنغان ۋەھيىگە ئىشەنسە)، ئۇلار ئۈچۈن (دۇنيا ۋە ئاخىرەتتە) ئەلۋەتتە ياخشى بولاتتى. ئۇلارنىڭ ئىچىدە مۆمىنلەرمۇ بار، ئۇلارنىڭ تولىسى پاسىقلاردۇر
لَن يَضُرُّوكُمْ إِلَّآ أَذًى ۖ وَإِن يُقَـٰتِلُوكُمْ يُوَلُّوكُمُ ٱلْأَدْبَارَ ثُمَّ لَا يُنصَرُونَ ﴿١١١﴾
ئۇلار سىلەرگە ئازار بېرىشتىن باشقا ھېچقانداق زىيان - زەخمەت يەتكۈزەلمەيدۇ؛ ئەگەر سىلەر بىلەن ئۇرۇشسا، سىلەرگە ئارقىسىنى قىلىپ قاچىدۇ، ئاندىن ئۇلارغا ياردەم بېرىلمەيدۇ
ضُرِبَتْ عَلَيْهِمُ ٱلذِّلَّةُ أَيْنَ مَا ثُقِفُوٓا۟ إِلَّا بِحَبْلٍ مِّنَ ٱللَّهِ وَحَبْلٍ مِّنَ ٱلنَّاسِ وَبَآءُو بِغَضَبٍ مِّنَ ٱللَّهِ وَضُرِبَتْ عَلَيْهِمُ ٱلْمَسْكَنَةُ ۚ ذَٰلِكَ بِأَنَّهُمْ كَانُوا۟ يَكْفُرُونَ بِـَٔايَـٰتِ ٱللَّهِ وَيَقْتُلُونَ ٱلْأَنۢبِيَآءَ بِغَيْرِ حَقٍّ ۚ ذَٰلِكَ بِمَا عَصَوا۟ وَّكَانُوا۟ يَعْتَدُونَ ﴿١١٢﴾
ئۇلارنىڭ قەيەردە بولمىسۇن اﷲ نىڭ ۋە مۇسۇلمانلارنىڭ ھىمايىسىدە بولغانلىرىدىن باشقىلىرى خارلىققا پېتىپ قالدى، اﷲ نىڭ غەزىپىگە ئۇچرىدى. موھتاجلىققا چۆمدى. بۇ ئۇلارنىڭ اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغانلىقلىرى ۋە خۇدانىڭ ئەلچى-رەسۇللىرىنى ناھەق ئۆلتۈرگەنلىكلىرى ئۈچۈندۇر، ئۇلارنىڭ اﷲ نىڭ پەرمانىدىن چىقىپ كەتكەنلىكلىرى ۋە اﷲ نىڭ چەكلىمىسىدىن ئېشىپ كەتكەنلىكلىرى ئۈچۈندۇر
۞ لَيْسُوا۟ سَوَآءً ۗ مِّنْ أَهْلِ ٱلْكِتَـٰبِ أُمَّةٌ قَآئِمَةٌ يَتْلُونَ ءَايَـٰتِ ٱللَّهِ ءَانَآءَ ٱلَّيْلِ وَهُمْ يَسْجُدُونَ ﴿١١٣﴾
ئۇلار بىرخىل ئەمەس، ئەھلى مۇقەددەس كىتاب-ۋەھىيتىن (يەنى يەھۇدىيلار، ناسارالاردىن) كېچىلىرى اﷲ نىڭ ئايەتلىرىنى تىلاۋەت قىلىپ تۇرىدىغان ۋە سەجدە قىلىپ تۇرىدىغان بىر جامائە كىشىلەر بار
يُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْـَٔاخِرِ وَيَأْمُرُونَ بِٱلْمَعْرُوفِ وَيَنْهَوْنَ عَنِ ٱلْمُنكَرِ وَيُسَـٰرِعُونَ فِى ٱلْخَيْرَٰتِ وَأُو۟لَـٰٓئِكَ مِنَ ٱلصَّـٰلِحِينَ ﴿١١٤﴾
ئۇلار اﷲ قا ۋە ئاخىرەت كۈنىگە ئىشىنىدۇ، ياخشى ئىشلارغا بۇيرۇيدۇ، يامان ئىشلاردىن توسىدۇ، خەيرلىك ئىشلارنى قىلىشقا ئالدىرايدۇ؛ ئەنە شۇلار ياخشىلاردىندۇر
وَمَا يَفْعَلُوا۟ مِنْ خَيْرٍ فَلَن يُكْفَرُوهُ ۗ وَٱللَّهُ عَلِيمٌۢ بِٱلْمُتَّقِينَ ﴿١١٥﴾
ئۇلار ھەر قانداق ياخشىلىقنى قىلسۇن، ئۇ بىكار كەتمەيدۇ. اﷲ تەقۋادارلارنى ئوبدان بىلىدۇ
إِنَّ ٱلَّذِينَ كَفَرُوا۟ لَن تُغْنِىَ عَنْهُمْ أَمْوَٰلُهُمْ وَلَآ أَوْلَـٰدُهُم مِّنَ ٱللَّهِ شَيْـًٔا ۖ وَأُو۟لَـٰٓئِكَ أَصْحَـٰبُ ٱلنَّارِ ۚ هُمْ فِيهَا خَـٰلِدُونَ ﴿١١٦﴾
شۈبھىسىزكى، ھەقىقەتنى رەت قىلغۇچىلارنىڭ ماللىرى، بالىلىرى اﷲ نىڭ ئازابى ئالدىدا ھەرگىز ھېچ نەرسىگە دال بولالمايدۇ؛ ئۇلار دوزاخ ئەھلىدۇر، دوزاختا مەڭگۈ قالىدۇ
مَثَلُ مَا يُنفِقُونَ فِى هَـٰذِهِ ٱلْحَيَوٰةِ ٱلدُّنْيَا كَمَثَلِ رِيحٍ فِيهَا صِرٌّ أَصَابَتْ حَرْثَ قَوْمٍ ظَلَمُوٓا۟ أَنفُسَهُمْ فَأَهْلَكَتْهُ ۚ وَمَا ظَلَمَهُمُ ٱللَّهُ وَلَـٰكِنْ أَنفُسَهُمْ يَظْلِمُونَ ﴿١١٧﴾
ئۇلارنىڭ بۇ دۇنيادا (رىيا قىلىپ) بەرگەن سەدىقىلىرى ئۆزلىرىگە زۇلۇم قىلغان بىر قەۋمنىڭ زىرائىتىگە تېگىپ ئۇنى نابۇت قىلىۋەتكەن سوغۇق شامالغا ئوخشايدۇ (ئىگىلىرىنىڭ گۇناھى تۈپەيلىدىن نابۇت قىلىنغان بۇ زىرائەتكە ئوخشاش، اﷲ ئۇلارنىڭ ياخشى ئەمەللىرىنى بەربات قىلىدۇ). ئۇلارغا اﷲ زۇلۇم قىلمىدى ۋە لېكىن ئۇلار ئۆزلىرىگە ئۆزلىرى زۇلۇم قىلدى
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ بِطَانَةً مِّن دُونِكُمْ لَا يَأْلُونَكُمْ خَبَالًا وَدُّوا۟ مَا عَنِتُّمْ قَدْ بَدَتِ ٱلْبَغْضَآءُ مِنْ أَفْوَٰهِهِمْ وَمَا تُخْفِى صُدُورُهُمْ أَكْبَرُ ۚ قَدْ بَيَّنَّا لَكُمُ ٱلْـَٔايَـٰتِ ۖ إِن كُنتُمْ تَعْقِلُونَ ﴿١١٨﴾
ئى مۆمىنلەر! ئۆزۈڭلاردىن بولمىغانلارنى (يەنى مۇناپىقلارنى) سىرداش قىلىۋالماڭلار، ئۇلار سىلەرگە بۇزغۇنچىلىق قىلىشتا بوشاڭلىق قىلىپ قالمايدۇ. ئۇلار سىلەرنىڭ مۇشەققەتتە قېلىشىڭلارنى ئارزۇ قىلىدۇ، ئۇلارنىڭ دۈشمەنلىكى ئېغىزلىرىدىن مەلۇم بولۇپ تۇرىدۇ، ئۇلارنىڭ دىللىرىدا يوشۇرغان دۈشمەنلىكى تېخىمۇ چوڭدۇر. شۈبھىسىزكى، ئەگەر چۈشىنىدىغان بولساڭلار، سىلەرگە نۇرغۇن ئايەتلەرنى بايان قىلدۇق
هَـٰٓأَنتُمْ أُو۟لَآءِ تُحِبُّونَهُمْ وَلَا يُحِبُّونَكُمْ وَتُؤْمِنُونَ بِٱلْكِتَـٰبِ كُلِّهِۦ وَإِذَا لَقُوكُمْ قَالُوٓا۟ ءَامَنَّا وَإِذَا خَلَوْا۟ عَضُّوا۟ عَلَيْكُمُ ٱلْأَنَامِلَ مِنَ ٱلْغَيْظِ ۚ قُلْ مُوتُوا۟ بِغَيْظِكُمْ ۗ إِنَّ ٱللَّهَ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ ﴿١١٩﴾
(ئى مۆمىنلەر جامائەسى!) سىلەر ئۇلارنى دوست تۇتىسىلەر، ئۇلار سىلەرنى دوست تۇتمايدۇ (سىلەرگە بولغان دۈشمەنلىكنى يوشۇرۇن تۇتىدۇ)، سىلەر (ھەممە ساماۋى) مۇقەددەس كىتاب-ۋەھىيقا ئىشىنىسىلەر (شۇنداق تۇرۇقلۇق ئۇلار سىلەرنى ئۆچ كۆرىدۇ)، ئۇلار سىلەر بىلەن ئۇچراشقاندا، ئىمان ئېيتتۇق، دەيدۇ. ئۆزلىرى يالغۇز قالغاندا سىلەرگە بولغان ئاچچىقىدىن بارماقلىرىنى چىشلەيدۇ. (ئى مۇھەممەد!) ئېيتىقىنكى، «ئاچچىقىڭلار بىلەن ئۆلۈڭلار! (يەنى اﷲ ئۆلگىنىڭلارغا قەدەر ئاچچىقىڭلارنى داۋاملاشتۇرسۇن!)» اﷲ ئىشەنچلىك ھالدا دىللاردىكىنى بىلگۈچىدۇر

وَإِذْ غَدَوْتَ مِنْ أَهْلِكَ تُبَوِّئُ ٱلْمُؤْمِنِينَ مَقَـٰعِدَ لِلْقِتَالِ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌ ﴿١٢١﴾
(ئى مۇھەممەد!) ئۆز ۋاقتىدا سەن ئەتىگەندە ئۆيۈڭدىن (ئۇھۇدقا) چىقىپ مۆمىنلەرنى ئۇرۇش سەپلىرىگە ئورۇنلاشتۇردۇڭ، اﷲ (سۆزۈڭلارنى) ئاڭلاپ تۇرغۇچىدۇر، (ئەھۋالىڭلارنى) بىلىپ تۇرغۇچىدۇر
إِذْ هَمَّت طَّآئِفَتَانِ مِنكُمْ أَن تَفْشَلَا وَٱللَّهُ وَلِيُّهُمَا ۗ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٢٢﴾
ئۆز ۋاقتىدا سىلەردىن ئىككى گۇرۇھ (يەنى سەلەمە ئايمىقى بىلەن بەنى ھارىسە ئايمىقى) ئاجىزلىق كۆرسەتمەكچى (يەنى قورقۇپ ئۇرۇش مەيدانىدىن قايتماقچى) بولدى؛ اﷲ ئۇلارنىڭ مەدەتكارى ئىدى. (مۆمىنلەر جىمى ئەھۋالى ۋە جىمى ئىشلىرىدا) اﷲ غىلا تەۋەككۈل قىلسۇن!
وَلَقَدْ نَصَرَكُمُ ٱللَّهُ بِبَدْرٍ وَأَنتُمْ أَذِلَّةٌ ۖ فَٱتَّقُوا۟ ٱللَّهَ لَعَلَّكُمْ تَشْكُرُونَ ﴿١٢٣﴾
اﷲ سىلەرگە بەدرىدە نۇسرەت ئاتا قىلدى، ھالبۇكى، سىلەر كۈچسىز ئىدىڭلار (سانىڭلار ۋە قورالىڭلار ئاز ئىدى). اﷲ نىڭ سىلەرگە بەرگەن نۇسرىتىگە شۈكۈر قىلىش يۈزىسىدىن اﷲ تىن قورقۇڭلار
إِذْ تَقُولُ لِلْمُؤْمِنِينَ أَلَن يَكْفِيَكُمْ أَن يُمِدَّكُمْ رَبُّكُم بِثَلَـٰثَةِ ءَالَـٰفٍ مِّنَ ٱلْمَلَـٰٓئِكَةِ مُنزَلِينَ ﴿١٢٤﴾

بَلَىٰٓ ۚ إِن تَصْبِرُوا۟ وَتَتَّقُوا۟ وَيَأْتُوكُم مِّن فَوْرِهِمْ هَـٰذَا يُمْدِدْكُمْ رَبُّكُم بِخَمْسَةِ ءَالَـٰفٍ مِّنَ ٱلْمَلَـٰٓئِكَةِ مُسَوِّمِينَ ﴿١٢٥﴾
شۇنداق، ئۇ يېتىدۇ. ئەگەر (جەڭدە) سەۋر قىلساڭلار، (اﷲ تىن) قورقۇپ (ئۇنىڭ ئەمرىگە ئىتائەت قىلساڭلار)، دۈشمەن دەرھال ھۇجۇم قىلىپ كەلگەندە پەرۋەردىگارىڭلار سىلەرگە بەش مىڭ نىشانلىق پەرىشتە بىلەن ياردەم بېرىدۇ
وَمَا جَعَلَهُ ٱللَّهُ إِلَّا بُشْرَىٰ لَكُمْ وَلِتَطْمَئِنَّ قُلُوبُكُم بِهِۦ ۗ وَمَا ٱلنَّصْرُ إِلَّا مِنْ عِندِ ٱللَّهِ ٱلْعَزِيزِ ٱلْحَكِيمِ ﴿١٢٦﴾
اﷲ ئۇنى (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى ئارقىلىق ياردەم بېرىشنى) پەقەت سىلەرگە خۇش خەۋەر بولسۇن ۋە دىلىڭلار شۇنىڭ بىلەن خاتىرجەم بولسۇن (شۇنىڭ بىلەن دۈشمەننىڭ سانىنىڭ كۆپلۈكىدىن قورقمىسۇن) دەپ قىلدى. ياردەم پەقەت غالىب، ھېكمەت بىلەن ئىش قىلغۇچى اﷲ تەرىپىدىندۇر
لِيَقْطَعَ طَرَفًا مِّنَ ٱلَّذِينَ كَفَرُوٓا۟ أَوْ يَكْبِتَهُمْ فَيَنقَلِبُوا۟ خَآئِبِينَ ﴿١٢٧﴾
(اﷲ نىڭ ياردىمى) ھەقىقەتنى رەت قىلغۇچىلارنىڭ بىر تۈركۈمىنى ھالاك قىلىش ئۈچۈن ياكى ئۇلارنى رەسۋا قىلىپ، ئۇلارنىڭ ئۈمىدسىز ھالدا مەغلۇپ بولۇشلىرى ئۈچۈندۇر
لَيْسَ لَكَ مِنَ ٱلْأَمْرِ شَىْءٌ أَوْ يَتُوبَ عَلَيْهِمْ أَوْ يُعَذِّبَهُمْ فَإِنَّهُمْ ظَـٰلِمُونَ ﴿١٢٨﴾
(ئى مۇھەممەد!) ھېچ ئىش (يەنى بەندىلەرنىڭ ئىشلىرىنىڭ تەدبىرى) سېنىڭ ئىختىيارىڭدا ئەمەس، اﷲ يا ئۇلارنىڭ (مۇسۇلمان بولسا) تەۋبىسىنى قوبۇل قىلىدۇ، يا ئۇلار زالىم بولغانلىقى ئۈچۈن (كۇفرىدا چىڭ تۇرسا) ئۇلارنى ئازابقا دۇچار قىلىدۇ
وَلِلَّهِ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ يَغْفِرُ لِمَن يَشَآءُ وَيُعَذِّبُ مَن يَشَآءُ ۚ وَٱللَّهُ غَفُورٌ رَّحِيمٌ ﴿١٢٩﴾
ئاسمانلاردىكى ۋە زېمىندىكى نەرسىلەرنىڭ ھەممىسى اﷲ نىڭ مۈلكىدۇر. اﷲ خالىغان كىشىگە مەغپىرەت قىلىدۇ، خالىغان كىشىگە ئازاب قىلىدۇ. اﷲ مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَأْكُلُوا۟ ٱلرِّبَوٰٓا۟ أَضْعَـٰفًا مُّضَـٰعَفَةً ۖ وَٱتَّقُوا۟ ٱللَّهَ لَعَلَّكُمْ تُفْلِحُونَ ﴿١٣٠﴾
ئى مۆمىنلەر! جازانىنى قاتمۇقات ئاشۇرۇپ يېمەڭلار، مەقسىدىڭلارغا ئېرىشىش ئۈچۈن اﷲ تىن (نەھيى قىلغان ئىشلارنى تەرك ئېتىپ) قورقۇڭلار
وَٱتَّقُوا۟ ٱلنَّارَ ٱلَّتِىٓ أُعِدَّتْ لِلْكَـٰفِرِينَ ﴿١٣١﴾
ھەقىقەتنى رەت قىلغۇچىلار ئۈچۈن تەييارلانغان دوزاختىن (يەنى دوزاخقا كىرىشكە سەۋەب بولىدىغان يامان ئىشلاردىن) ساقلىنىڭلار
وَأَطِيعُوا۟ ٱللَّهَ وَٱلرَّسُولَ لَعَلَّكُمْ تُرْحَمُونَ ﴿١٣٢﴾
سىلەرگە رەھمەت قىلىنىشى ئۈچۈن، اﷲ قا ۋە خۇدانىڭ ئەلچىسىگە ئىتائەت قىلىڭلار
۞ وَسَارِعُوٓا۟ إِلَىٰ مَغْفِرَةٍ مِّن رَّبِّكُمْ وَجَنَّةٍ عَرْضُهَا ٱلسَّمَـٰوَٰتُ وَٱلْأَرْضُ أُعِدَّتْ لِلْمُتَّقِينَ ﴿١٣٣﴾
پەرۋەردىگارىڭلارنىڭ مەغپىرىتىگە ۋە تەقۋادارلار ئۈچۈن تەييارلانغان، كەڭلىكى ئاسمان - زېمىنچە كېلىدىغان مەڭگۈلۈك باغلاركە ئالدىراڭلار
ٱلَّذِينَ يُنفِقُونَ فِى ٱلسَّرَّآءِ وَٱلضَّرَّآءِ وَٱلْكَـٰظِمِينَ ٱلْغَيْظَ وَٱلْعَافِينَ عَنِ ٱلنَّاسِ ۗ وَٱللَّهُ يُحِبُّ ٱلْمُحْسِنِينَ ﴿١٣٤﴾
تەقۋادارلار كەڭچىلىكتىمۇ، قىسىنچىلىقتىمۇ اﷲ يولىدا (پۇل - مال) سەرپ قىلىدىغانلار، (ئۆچ ئېلىشقا قادىر تۇرۇقلۇق) ئاچچىقىنى يۇتىدىغانلار، (يامانلىق قىلغان ياكى بوزەك قىلغان) كىشىلەرنى كەچۈرىدىغانلاردۇر. اﷲ ياخشىلىق قىلغۇچىلارنى دوست تۇتىدۇ
وَٱلَّذِينَ إِذَا فَعَلُوا۟ فَـٰحِشَةً أَوْ ظَلَمُوٓا۟ أَنفُسَهُمْ ذَكَرُوا۟ ٱللَّهَ فَٱسْتَغْفَرُوا۟ لِذُنُوبِهِمْ وَمَن يَغْفِرُ ٱلذُّنُوبَ إِلَّا ٱللَّهُ وَلَمْ يُصِرُّوا۟ عَلَىٰ مَا فَعَلُوا۟ وَهُمْ يَعْلَمُونَ ﴿١٣٥﴾

أُو۟لَـٰٓئِكَ جَزَآؤُهُم مَّغْفِرَةٌ مِّن رَّبِّهِمْ وَجَنَّـٰتٌ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا ۚ وَنِعْمَ أَجْرُ ٱلْعَـٰمِلِينَ ﴿١٣٦﴾

وَلِيُمَحِّصَ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ وَيَمْحَقَ ٱلْكَـٰفِرِينَ ﴿١٤١﴾
مۆمىنلەرنى (ئۇلارغا يەتكەن مۇسىبەت ئارقىلىق) پاكلىشى ئۈچۈندۇر، ھەقىقەتنى رەت قىلغۇچىلارنى يوق قىلىشى ئۈچۈندۇر
أَمْ حَسِبْتُمْ أَن تَدْخُلُوا۟ ٱلْجَنَّةَ وَلَمَّا يَعْلَمِ ٱللَّهُ ٱلَّذِينَ جَـٰهَدُوا۟ مِنكُمْ وَيَعْلَمَ ٱلصَّـٰبِرِينَ ﴿١٤٢﴾
(ئى مۆمىنلەر جامائەسى!) اﷲ سىلەردىن (ھەقىقىي) جىھاد قىلغانلارنى ۋە (جىھادنىڭ جاپا - مۇشەققەتلىرىگە) چىدىغۇچىلارنى بىلمەي (يەنى ئايرىماي تۇرۇپ)، (پەقەت ئاغزىڭلاردا ئىمان ئېيتتۇق دېيىش بىلەنلا) مەڭگۈلۈك باغلاركە كىرىشنى ئويلامسىلەر؟
وَلَقَدْ كُنتُمْ تَمَنَّوْنَ ٱلْمَوْتَ مِن قَبْلِ أَن تَلْقَوْهُ فَقَدْ رَأَيْتُمُوهُ وَأَنتُمْ تَنظُرُونَ ﴿١٤٣﴾
سىلەر ئۆلۈم (يەنى دۈشمەن) بىلەن ئۇچرىشىشتىن ئىلگىرى ئۇنى ئارزۇ قىلاتتىڭلار، مانا ئۇنى ئۆز كۆزۈڭلار بىلەن كۆردۈڭلار (ئەمدى نېمىشقا مەغلۇپ بولىسىلەر؟)
وَمَا مُحَمَّدٌ إِلَّا رَسُولٌ قَدْ خَلَتْ مِن قَبْلِهِ ٱلرُّسُلُ ۚ أَفَإِي۟ن مَّاتَ أَوْ قُتِلَ ٱنقَلَبْتُمْ عَلَىٰٓ أَعْقَـٰبِكُمْ ۚ وَمَن يَنقَلِبْ عَلَىٰ عَقِبَيْهِ فَلَن يَضُرَّ ٱللَّهَ شَيْـًٔا ۗ وَسَيَجْزِى ٱللَّهُ ٱلشَّـٰكِرِينَ ﴿١٤٤﴾
مۇھەممەد پەقەت بىر خۇدانىڭ ئەلچىسىدۇر، ئۇنىڭدىن بۇرۇن كۆپ خۇدانىڭ ئەلچى-رەسۇللىرى ئۆتتى (ئۇلارنىڭ بەزىلىرى ئۆز ئەجىلى بىلەن ئۆلگەن، بەزىلىرى باشقىلار تەرىپىدىن ئۆلتۈرۈلگەن)؛ ئۇ ۋاپات بولسا ياكى ئۆلتۈرۈلسە، ئارقاڭلارغا يېنىۋالامسىلەر؟ (يەنى ئىماندىن يېنىپ مۇرتەد بولامسىلەر؟) كىمكى ئارقىسىغا يېنىۋالىدىكەن (يەنى مۇرتەد بولىدىكەن)، ئۇ اﷲ قا قىلچە زىيان يەتكۈزەلمەيدۇ. اﷲ شۈكۈر قىلغۇچىلارنى مۇكاپاتلايدۇ
وَمَا كَانَ لِنَفْسٍ أَن تَمُوتَ إِلَّا بِإِذْنِ ٱللَّهِ كِتَـٰبًا مُّؤَجَّلًا ۗ وَمَن يُرِدْ ثَوَابَ ٱلدُّنْيَا نُؤْتِهِۦ مِنْهَا وَمَن يُرِدْ ثَوَابَ ٱلْـَٔاخِرَةِ نُؤْتِهِۦ مِنْهَا ۚ وَسَنَجْزِى ٱلشَّـٰكِرِينَ ﴿١٤٥﴾
ھېچ كىشى اﷲ نىڭ ئىزنىسىز ئۆلمەيدۇ، اﷲ ھەر ئادەمنىڭ ئەجىلىنى پۈتۈۋەتكەن (ئادەمنىڭ ئۆمرىنى قورقۇنچاقلىق ئۇزارتالمايدۇ، باتۇرلۇق قىسقارتالمايدۇ). كىمكى دۇنيا ساۋابىنى كۆزلىسە، ئۇنىڭغا دۇنيانىڭ ساۋابىنى بېرىمىز (ئۇنىڭ ئاخىرەتلىكتىن نېسىۋىسى بولمايدۇ)؛ كىمكى ئاخىرەتنىڭ ساۋابىنى كۆزلىسە، ئۇنىڭغا ئاخىرەتنىڭ ساۋابىنى بېرىمىز، شۈكۈر قىلغۇچىلارنى مۇكاپاتلايمىز
وَكَأَيِّن مِّن نَّبِىٍّ قَـٰتَلَ مَعَهُۥ رِبِّيُّونَ كَثِيرٌ فَمَا وَهَنُوا۟ لِمَآ أَصَابَهُمْ فِى سَبِيلِ ٱللَّهِ وَمَا ضَعُفُوا۟ وَمَا ٱسْتَكَانُوا۟ ۗ وَٱللَّهُ يُحِبُّ ٱلصَّـٰبِرِينَ ﴿١٤٦﴾
نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرى بىلەن كۆپلىگەن خۇدا جۇي ئۆلىمالار بىرلىكتە جەڭ قىلدى. ئۇلار اﷲ يولىدا يەتكەن كۈلپەتلەردىن روھسىزلانمىدى، بوشاشمىدى، باش ئەگمىدى، اﷲ ئۆزىنىڭ يولىدا دۇچ كەلگەن قىيىنچىلىقلارغا چىداشلىق بەرگۈچىلەرنى دوست تۇتىدۇ
وَمَا كَانَ قَوْلَهُمْ إِلَّآ أَن قَالُوا۟ رَبَّنَا ٱغْفِرْ لَنَا ذُنُوبَنَا وَإِسْرَافَنَا فِىٓ أَمْرِنَا وَثَبِّتْ أَقْدَامَنَا وَٱنصُرْنَا عَلَى ٱلْقَوْمِ ٱلْكَـٰفِرِينَ ﴿١٤٧﴾
ئۇلارنىڭ سۆزى پەقەت: «پەرۋەردىگارىمىز! گۇناھلىرىمىزنى، ئىشىمىزدا چەكتىن ئاشقانلىقىمىزنى مەغپىرەت قىلغىن، قەدەملىرىمىزنى (جەڭ مەيدانىدا) مۇستەھكەم قىلغىن ۋە ھەقىقەتنى رەت قىلغۇچى قەۋمگە قارشى بىزگە ياردەم بەرگىن» دېگەندىن باشقا بولمىدى
فَـَٔاتَىٰهُمُ ٱللَّهُ ثَوَابَ ٱلدُّنْيَا وَحُسْنَ ثَوَابِ ٱلْـَٔاخِرَةِ ۗ وَٱللَّهُ يُحِبُّ ٱلْمُحْسِنِينَ ﴿١٤٨﴾
اﷲ ئۇلارغا دۇنيانىڭ ساۋابىنى (يەنى دۈشمەنلەر ئۈستىدىن غەلىبە قىلىشنى) ۋە ئاخىرەتنىڭ ياخشى ساۋابىنى (يەنى مەڭگۈلۈك باغلار ۋە ئۇنىڭ نېمەتلىرىنى) ئاتا قىلدى، اﷲ ياخشىلىق قىلغۇچىلارنى دوست تۇتىدۇ
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِن تُطِيعُوا۟ ٱلَّذِينَ كَفَرُوا۟ يَرُدُّوكُمْ عَلَىٰٓ أَعْقَـٰبِكُمْ فَتَنقَلِبُوا۟ خَـٰسِرِينَ ﴿١٤٩﴾
ئى مۆمىنلەر! ئەگەر سىلەر ھەقىقەتنى رەت قىلغۇچىلارغا ئىتائەت قىلساڭلار، ئۇلار سىلەرنى ئارقاڭلارغا (يەنى كۇفرىغا) ياندۇرىدۇ، شۇنىڭ بىلەن زىيان تارتىپ قالىسىلەر
بَلِ ٱللَّهُ مَوْلَىٰكُمْ ۖ وَهُوَ خَيْرُ ٱلنَّـٰصِرِينَ ﴿١٥٠﴾
ئۇنداق ئەمەس (يەنى ئۇلار سىلەرنىڭ ئىتائەت قىلىدىغان مەدەتكارىڭلار ئەمەس)، اﷲ سىلەرنىڭ مەدەتكارىڭلاردۇر (ئۇنىڭغا ئىتائەت قىلىڭلار)، ئۇ ياردەم قىلغۇچىلارنىڭ ئەڭ ياخشىسىدۇر
سَنُلْقِى فِى قُلُوبِ ٱلَّذِينَ كَفَرُوا۟ ٱلرُّعْبَ بِمَآ أَشْرَكُوا۟ بِٱللَّهِ مَا لَمْ يُنَزِّلْ بِهِۦ سُلْطَـٰنًا ۖ وَمَأْوَىٰهُمُ ٱلنَّارُ ۚ وَبِئْسَ مَثْوَى ٱلظَّـٰلِمِينَ ﴿١٥١﴾
(ئىلاھلىقى) ھەققىدە ھېچبىر دەلىل چۈشۈرمىگەن نەرسىلەرنى (يەنى بۇتلارنى) اﷲ قا شېرىك قىلغانلىقلىرى ئۈچۈن، ھەقىقەتنى رەت قىلغۇچىلارنىڭ دىللىرىغا قورقۇنچ سالىمىز، ئۇلارنىڭ بارىدىغان جايى دوزاختۇر، زالىملارنىڭ جايى (يەنى دوزاخ) ئىشەنچلىك ھالدا يامان جايدۇر
وَلَقَدْ صَدَقَكُمُ ٱللَّهُ وَعْدَهُۥٓ إِذْ تَحُسُّونَهُم بِإِذْنِهِۦ ۖ حَتَّىٰٓ إِذَا فَشِلْتُمْ وَتَنَـٰزَعْتُمْ فِى ٱلْأَمْرِ وَعَصَيْتُم مِّنۢ بَعْدِ مَآ أَرَىٰكُم مَّا تُحِبُّونَ ۚ مِنكُم مَّن يُرِيدُ ٱلدُّنْيَا وَمِنكُم مَّن يُرِيدُ ٱلْـَٔاخِرَةَ ۚ ثُمَّ صَرَفَكُمْ عَنْهُمْ لِيَبْتَلِيَكُمْ ۖ وَلَقَدْ عَفَا عَنكُمْ ۗ وَٱللَّهُ ذُو فَضْلٍ عَلَى ٱلْمُؤْمِنِينَ ﴿١٥٢﴾
اﷲ سىلەرگە قىلغان (دۈشمىنىڭلارغا قارشى نۇسرەت ئاتا قىلىشتىن ئىبارەت) ۋەدىسىدە تۇردى؛ اﷲ نىڭ ئىزنى بىلەن ئۇلارنى قىرىۋاتاتتىڭلار، اﷲ سىلەر ياقتۇرىدىغان غەلىبىنى كۆرسەتكەندىن كېيىن زەئىپلىشىپ قالدىڭلار، خۇدانىڭ ئەلچىسىنىڭ ئەمرى توغرىسىدا جاڭجاللاشتىڭلار، (خۇدانىڭ ئەلچىسىنىڭ ئەمرىگە) بويسۇنمىدىڭلار، بەزىلىرىڭلار دۇنيانى كۆزلىدىڭلار، بەزىلىرىڭلار ئاخىرەتنى كۆزلىدىڭلار، ئاندىن اﷲ سىناش ئۈچۈن سىلەردىن دۈشمەننى قايتۇردى (يەنى سىلەرنى غەلىبىدىن مەغلۇبىيەتكە يۈزلەندۈردى) ۋە سىلەرنى ئىشەنچلىك ھالدا ئەپۇ قىلدى. اﷲ مۆمىنلەرگە مەرھەمەت قىلغۇچىدۇر
۞ إِذْ تُصْعِدُونَ وَلَا تَلْوُۥنَ عَلَىٰٓ أَحَدٍ وَٱلرَّسُولُ يَدْعُوكُمْ فِىٓ أُخْرَىٰكُمْ فَأَثَـٰبَكُمْ غَمًّۢا بِغَمٍّ لِّكَيْلَا تَحْزَنُوا۟ عَلَىٰ مَا فَاتَكُمْ وَلَا مَآ أَصَـٰبَكُمْ ۗ وَٱللَّهُ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٥٣﴾
ئۆز ۋاقتىدا خۇدانىڭ ئەلچىسى (ئى اﷲ نىڭ بەندىلىرى! مېنىڭ تەرىپىمگە كېلىڭلار، مەن رەسۇلۇللاھتۇرمەن، كىمكى جىھاد مەيدانىغا قايتسا، ئۇ مەڭگۈلۈك باغلاركە كىرىدۇ دەپ) ئارقاڭلاردىن سىلەرنى چاقىرىپ تۇراتتى، سىلەر ھېچ كىشىگە قارىماي قاچاتتىڭلار، (بۇنىڭدىن كېيىن) قولۇڭلاردىن كەتكەن نەرسىگە (يەنى غەنىيمەتكە)، بېشىڭلارغا كەلگەن مۇسىبەتكە (يەنى مەغلۇبىيەتكە) قايغۇرماسلىقىڭلار ئۈچۈن، اﷲ سىلەرگە غەم ئۈستىگە غەم بەردى (يەنى سىلەرنى غەنىيمەتتىن مەھرۇم قىلغاننىڭ ئۈستىگە مەغلۇبىيەت بىلەن جازالىدى). اﷲ قىلمىشىڭلاردىن خەۋەرداردۇر
ثُمَّ أَنزَلَ عَلَيْكُم مِّنۢ بَعْدِ ٱلْغَمِّ أَمَنَةً نُّعَاسًا يَغْشَىٰ طَآئِفَةً مِّنكُمْ ۖ وَطَآئِفَةٌ قَدْ أَهَمَّتْهُمْ أَنفُسُهُمْ يَظُنُّونَ بِٱللَّهِ غَيْرَ ٱلْحَقِّ ظَنَّ ٱلْجَـٰهِلِيَّةِ ۖ يَقُولُونَ هَل لَّنَا مِنَ ٱلْأَمْرِ مِن شَىْءٍ ۗ قُلْ إِنَّ ٱلْأَمْرَ كُلَّهُۥ لِلَّهِ ۗ يُخْفُونَ فِىٓ أَنفُسِهِم مَّا لَا يُبْدُونَ لَكَ ۖ يَقُولُونَ لَوْ كَانَ لَنَا مِنَ ٱلْأَمْرِ شَىْءٌ مَّا قُتِلْنَا هَـٰهُنَا ۗ قُل لَّوْ كُنتُمْ فِى بُيُوتِكُمْ لَبَرَزَ ٱلَّذِينَ كُتِبَ عَلَيْهِمُ ٱلْقَتْلُ إِلَىٰ مَضَاجِعِهِمْ ۖ وَلِيَبْتَلِىَ ٱللَّهُ مَا فِى صُدُورِكُمْ وَلِيُمَحِّصَ مَا فِى قُلُوبِكُمْ ۗ وَٱللَّهُ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ ﴿١٥٤﴾
غەمدىن كېيىن اﷲ سىلەرگە ئەمىنلىك چۈشۈرۈپ بەردى، سىلەرنىڭ بىر قىسمىڭلارنى (يەنى ھەقىقىي مۆمىنلەرنى) مۈگدەك باستى، يەنە بىر قىسمىڭلار (يەنى مۇناپىقلار) ئۆزلىرى بىلەن بولۇپ كەتتى (يەنى كۇففارلار قايتا ھۇجۇم قىلارمۇ، دەپ جان قايغۇسى بىلەن بولۇپ كەتتى)، بۇلار، جاھىلىيەت دەۋرى كىشىلىرىگە ئوخشاش، اﷲ قا قارىتا ناتوغرا گۇمانلاردا بولدى، ئۇلار: «بىزگە اﷲ ۋەدە قىلغان غەلىبىدىن نېسىۋە بارمۇ؟» دەيدۇ. ئېيتىقىنكى، «ھەممە ئىش ئىشەنچلىك ھالدا اﷲ نىڭ ئىلكىدە». ئۇلار ساڭا ئاشكارا قىلمايدىغان نەرسىلەرنى ئۆز ئىچىدە يوشۇرۇشىدۇ؛ ئۇلار: «ئەگەر بىزگە غەلىبىدىن ئازراق نېسىۋە بولسا ئىدى، بۇ يەردە ئۆلتۈرەلمەيتتۇق» دەيدۇ. (ئى مۇھەممەد!) ئېيتقىنكى، «ئەگەر سىلەر ئۆيلىرىڭلاردا بولغان تەقدىردىمۇ (ئاراڭلاردىكى) ئۆلتۈرۈلۈشى پۈتۈۋېتىلگەن كىشىلەر چوقۇم ئۆزلىرىنىڭ ئۆلىدىغان جايلىرىغا بېرىشىدۇ (اﷲ نىڭ قازاسىدىن قېچىپ قۇتۇلغىلى بولمايدۇ). (اﷲ نىڭ مۇنداق قىلىشى) كۆڭلۈڭلاردىكىنى سىناش ۋە قەلبىڭلاردىكى نەرسىلەرنى پاكلاش ئۈچۈندۇر. اﷲ كۆڭۈللەردىكىنى (ياخشىلىق بولسۇن، يامانلىق بولسۇن) بىلگۈچىدۇر»
إِنَّ ٱلَّذِينَ تَوَلَّوْا۟ مِنكُمْ يَوْمَ ٱلْتَقَى ٱلْجَمْعَانِ إِنَّمَا ٱسْتَزَلَّهُمُ ٱلشَّيْطَـٰنُ بِبَعْضِ مَا كَسَبُوا۟ ۖ وَلَقَدْ عَفَا ٱللَّهُ عَنْهُمْ ۗ إِنَّ ٱللَّهَ غَفُورٌ حَلِيمٌ ﴿١٥٥﴾
ئىككى قوشۇن (يەنى مۇسۇلمانلار بىلەن مۇشرىكلار) بىر - بىرى بىلەن ئۇچراشقان كۈندە، ئىچىڭلاردىكى (جەڭدىن) ئىشەنچلىك ھالدا يۈز ئۆرۈگەن كىشىلەرنى، بەزى قىلمىشلىرى (يەنى خۇدانىڭ ئەلچىسىنىڭ ئەمرىگە مۇخالىپەتچىلىك قىلغانلىقى) تۈپەيلىدىن، شەيتان (ئازدۇرغۇچى) تېيىلدۇرۇشقا ئورۇندى؛ اﷲ ئىشەنچلىك ھالدا ئۇلارنى ئەپۇ قىلدى. اﷲ ئىشەنچلىك ھالدا مەغپىرەت قىلغۇچىدۇر، ناھايىتى ھەلىمدۇر (يەنى ئاسىيلىق قىلغانلارنى جازالاشقا ئالدىراپ كەتمەيدۇ)
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَكُونُوا۟ كَٱلَّذِينَ كَفَرُوا۟ وَقَالُوا۟ لِإِخْوَٰنِهِمْ إِذَا ضَرَبُوا۟ فِى ٱلْأَرْضِ أَوْ كَانُوا۟ غُزًّى لَّوْ كَانُوا۟ عِندَنَا مَا مَاتُوا۟ وَمَا قُتِلُوا۟ لِيَجْعَلَ ٱللَّهُ ذَٰلِكَ حَسْرَةً فِى قُلُوبِهِمْ ۗ وَٱللَّهُ يُحْىِۦ وَيُمِيتُ ۗ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿١٥٦﴾
ئى مۆمىنلەر! سەپەرگە چىققان ياكى غازاتقا چىققان قېرىنداشلىرى ھەققىدە، ئۇلار يېنىمىزدا بولغاندا ئۆلمەيتتى ۋە ئۆلتۈرۈلمەيتتى، دېگەن ھەقىقەتنى رەت قىلغۇچىلارغا ئوخشاش بولماڭلار، اﷲ بۇنى ئۇلارنىڭ دىللىرىدا ھەسرەت قىلسۇن. اﷲ تىرىلدۈرىدۇ ۋە ئۆلتۈرىدۇ (ئۇلارنىڭ غازاتقا چىقماسلىقى ئۇلارنى ئۆلۈمدىن قۇتۇلدۇرالمايدۇ). اﷲ قىلمىشىڭلارنى كۆرۈپ تۇرغۇچىدۇر
وَلَئِن قُتِلْتُمْ فِى سَبِيلِ ٱللَّهِ أَوْ مُتُّمْ لَمَغْفِرَةٌ مِّنَ ٱللَّهِ وَرَحْمَةٌ خَيْرٌ مِّمَّا يَجْمَعُونَ ﴿١٥٧﴾
شۈبھىسىزكى، ئەگەر سىلەر اﷲ يولىدا (يەنى غازاتتا) ئۆلتۈرۈلسەڭلار ياكى (غازاتقا كېتىپ بېرىپ ئۆز ئەجىلىڭلار بىلەن) ئۆلسەڭلار، سىلەرگە اﷲ تەرىپىدىن بولغان مەغپىرەت ۋە رەھمەت، ئەلۋەتتە، ئۇلارنىڭ توپلايدىغان دۇنياسىدىن ئارتۇقتۇر
وَلَئِن مُّتُّمْ أَوْ قُتِلْتُمْ لَإِلَى ٱللَّهِ تُحْشَرُونَ ﴿١٥٨﴾
ئەگەر سىلەر (ئۆز ئەجىلىڭلار بىلەن) ئۆلسەڭلار ياكى (جەڭ مەيدانىدا) ئۆلتۈرۈلسەڭلار، (ئاخىرەتتە) چوقۇم اﷲ نىڭ دەرگاھىغا توپلىنىسىلەر (اﷲ سىلەرنى ئەمەلىڭلارغا قاراپ مۇكاپاتلايدۇ)
فَبِمَا رَحْمَةٍ مِّنَ ٱللَّهِ لِنتَ لَهُمْ ۖ وَلَوْ كُنتَ فَظًّا غَلِيظَ ٱلْقَلْبِ لَٱنفَضُّوا۟ مِنْ حَوْلِكَ ۖ فَٱعْفُ عَنْهُمْ وَٱسْتَغْفِرْ لَهُمْ وَشَاوِرْهُمْ فِى ٱلْأَمْرِ ۖ فَإِذَا عَزَمْتَ فَتَوَكَّلْ عَلَى ٱللَّهِ ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُتَوَكِّلِينَ ﴿١٥٩﴾
اﷲ نىڭ رەھمىتى بىلەن سەن ئۇلارغا مۇلايىم بولدۇڭ؛ ئەگەر قوپال، باغرى قاتتىق بولغان بولساڭ، ئۇلار چۆرەڭدىن تارقاپ كېتەتتى؛ ئۇلارنى ئەپۇ قىل، ئۇلار ئۈچۈن مەغپىرەت تىلىگىن، ئىشتا ئۇلار بىلەن كېڭەشكىن؛ (كېڭەشكەندىن كېيىن) بىر ئىشقا بەل باغلىساڭ، اﷲ قا تەۋەككۈل قىلغىن. اﷲ ئىشەنچلىك ھالدا تەۋەككۈل قىلغۇچىلارنى دوست تۇتىدۇ
إِن يَنصُرْكُمُ ٱللَّهُ فَلَا غَالِبَ لَكُمْ ۖ وَإِن يَخْذُلْكُمْ فَمَن ذَا ٱلَّذِى يَنصُرُكُم مِّنۢ بَعْدِهِۦ ۗ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٦٠﴾
ئەگەر اﷲ سىلەرگە ياردەم بەرسە، سىلەرنى ھېچ كىشى يېڭەلمەيدۇ. ئەگەر اﷲ سىلەرگە ياردىمىنى تەرك ئەتسە، اﷲ تىن باشقا سىلەرگە كىم ياردەم بېرىدۇ؟ مۆمىنلەر يالغۇز اﷲ قا تەۋەككۈل قىلسۇن! (يەنى ھەممە ئىشنى اﷲ قا تاپشۇرسۇن)
وَمَا كَانَ لِنَبِىٍّ أَن يَغُلَّ ۚ وَمَن يَغْلُلْ يَأْتِ بِمَا غَلَّ يَوْمَ ٱلْقِيَـٰمَةِ ۚ ثُمَّ تُوَفَّىٰ كُلُّ نَفْسٍ مَّا كَسَبَتْ وَهُمْ لَا يُظْلَمُونَ ﴿١٦١﴾
ھېچقانداق بىر خۇدانىڭ ئەلچىسى غەنىيمەتكە خىيانەت قىلمايدۇ (يەنى بۇنى تەسەۋۋۇر قىلغىلى بولمايدۇ). كىمكى غەنىيمەتكە خىيانەت قىلىدىكەن، قىيامەت كۈنى ئۇ خىيانەت قىلغان نەرسىسى بىلەن (يەنى خىيانەت قىلغان نەرسىسى گەجگىسىگە ئارتىلغان ھالدا) كېلىدۇ. ھەر كىشنىڭ قىلغان ئەمەلىگە تولۇق مۇكاپات، (ياكى) تولۇق جازا بېرىلىدۇ، ئۇلارغا (ياخشىلارنىڭ ساۋابىنى كېمەيتىۋېتىپ، ئاسىيلارنىڭ جازاسىنى ئاشۇرۇۋېتىپ) زۇلۇم قىلىنمايدۇ
أَفَمَنِ ٱتَّبَعَ رِضْوَٰنَ ٱللَّهِ كَمَنۢ بَآءَ بِسَخَطٍ مِّنَ ٱللَّهِ وَمَأْوَىٰهُ جَهَنَّمُ ۚ وَبِئْسَ ٱلْمَصِيرُ ﴿١٦٢﴾
اﷲ نىڭ رازىلىقىنى ئىزدىگەن كىشى اﷲ نىڭ غەزىپىگە ئۇچراشقا تېگىشلىك بولغان كىشىگە ئوخشامدۇ؟ اﷲ نىڭ غەزىپىگە ئۇچراشقا تېگىشلىك بولغان كىشىنىڭ بارىدىغان جايى مەڭگۈلۈك ئوتدۇر، مەڭگۈلۈك ئوت نېمىدېگەن يامان جاي!
هُمْ دَرَجَـٰتٌ عِندَ ٱللَّهِ ۗ وَٱللَّهُ بَصِيرٌۢ بِمَا يَعْمَلُونَ ﴿١٦٣﴾
اﷲ نىڭ دەرگاھىدا ئۇلارنىڭ دەرىجىسى تۈرلۈك بولىدۇ. اﷲ ئۇلارنىڭ قىلمىشلىرىنى كۆرۈپ تۇرغۇچىدۇر
لَقَدْ مَنَّ ٱللَّهُ عَلَى ٱلْمُؤْمِنِينَ إِذْ بَعَثَ فِيهِمْ رَسُولًا مِّنْ أَنفُسِهِمْ يَتْلُوا۟ عَلَيْهِمْ ءَايَـٰتِهِۦ وَيُزَكِّيهِمْ وَيُعَلِّمُهُمُ ٱلْكِتَـٰبَ وَٱلْحِكْمَةَ وَإِن كَانُوا۟ مِن قَبْلُ لَفِى ضَلَـٰلٍ مُّبِينٍ ﴿١٦٤﴾
اﷲ مۆمىنلەرگە اﷲ نىڭ ئايەتلىرىنى تىلاۋەت قىلىدىغان، ئۇلارنى (گۇناھلاردىن) پاك قىلىدىغان، ئۇلارغا مۇقەددەس كىتاب-ۋەھىينى ۋە ھېكمەتنى (يەنى قۇرئان ۋە سۈننەتنى) ئۆگىتىدىغان، ئۆزلىرىدىن بولغان بىر خۇدانىڭ ئەلچىسى ئەۋەتىپ، ئۇلارغا چوڭ ئېھسان قىلدى؛ ھالبۇكى، ئۇلار بۇرۇن ئوچۇق گۇمراھلىقتا ئىدى
أَوَلَمَّآ أَصَـٰبَتْكُم مُّصِيبَةٌ قَدْ أَصَبْتُم مِّثْلَيْهَا قُلْتُمْ أَنَّىٰ هَـٰذَا ۖ قُلْ هُوَ مِنْ عِندِ أَنفُسِكُمْ ۗ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿١٦٥﴾
سىلەرگە يەتكەن مۇسىبەت سىلەر دۈشمەنگە يەتكۈزگەن مۇسىبەتنىڭ يېرىمىغا تەڭ كېلىدىغان تۇرسا (يەنى ئۇھۇد غازىتىدا سىلەردىن 70 كىشى شېھىت بولغان تۇرسا، بەدرى غازىتىدا بولسا مۇشرىكلاردىن 70 كىشىنى ئۆلتۈرگەن، 70 كىشىنى ئەسىر ئالغان تۇرساڭلار)، سىلەر يەنە: «بۇ مۇسىبەت قەيەردىن كەلدى؟» دېدىڭلار. ئېيتقىنكى، «ئۇ سىلەرنىڭ ئۆزلىرىڭلاردىن (يەنى خۇدانىڭ ئەلچىسىنىڭ ئەمرىگە خىلاپلىق قىلغانلىقىڭلار ۋە غەنمەتكە ھېرىس بولغانلىقىڭلاردىن) بولدى». اﷲ ئىشەنچلىك ھالدا ھەر نەرسىگە قادىردۇر
وَمَآ أَصَـٰبَكُمْ يَوْمَ ٱلْتَقَى ٱلْجَمْعَانِ فَبِإِذْنِ ٱللَّهِ وَلِيَعْلَمَ ٱلْمُؤْمِنِينَ ﴿١٦٦﴾
ئىككى قوشۇن ئۇچراشقان كۈندە (يەنى مۇسۇلمانلار بىلەن مۇشرىكلار توقۇنۇشقان ئۇھۇد جېڭىدە) سىلەرگە كەلگەن مۇسىبەت اﷲ نىڭ ئىرادىسى بىلەن كەلگەندۇر، بۇ ھەقىقىي مۆمىنلەرنى بىلىش (يەنى ئايرىش) ئۈچۈندۇر
وَلِيَعْلَمَ ٱلَّذِينَ نَافَقُوا۟ ۚ وَقِيلَ لَهُمْ تَعَالَوْا۟ قَـٰتِلُوا۟ فِى سَبِيلِ ٱللَّهِ أَوِ ٱدْفَعُوا۟ ۖ قَالُوا۟ لَوْ نَعْلَمُ قِتَالًا لَّٱتَّبَعْنَـٰكُمْ ۗ هُمْ لِلْكُفْرِ يَوْمَئِذٍ أَقْرَبُ مِنْهُمْ لِلْإِيمَـٰنِ ۚ يَقُولُونَ بِأَفْوَٰهِهِم مَّا لَيْسَ فِى قُلُوبِهِمْ ۗ وَٱللَّهُ أَعْلَمُ بِمَا يَكْتُمُونَ ﴿١٦٧﴾
ۋە مۇناپىقلارنى بىلىش (يەنى ئايرىش) ئۈچۈندۇركى، ئۇلارغا: «كېلىڭلار، اﷲ يولىدا ئۇرۇش قىلىڭلار ياكى (ئۆزۈڭلارنى) قوغداڭلار» دېيىلسە، ئۇلار: «ئۇرۇشتىن خەۋەر تاپقان بولساق، ئەلۋەتتە، سىلەرگە ئەگىشەتتۇق (يەنى سىلەر بىلەن بىر سەپتە تۇرۇپ ئۇرۇش قىلاتتۇق)» دېدى. ئۇ كۈندە ئۇلار ئىمانغا قارىغاندا كۇفرىغا يېقىن ئىدى. ئۇلار كۆڭۈللىرىدە يوق نەرسىلەرنى ئېغىزلىرىدا دېدى، ئۇلار يوشۇررغاننى (يەنى نىفاق بىلەن مۇشرىكلىكنى) اﷲ ئوبدان بىلىدۇ
ٱلَّذِينَ قَالُوا۟ لِإِخْوَٰنِهِمْ وَقَعَدُوا۟ لَوْ أَطَاعُونَا مَا قُتِلُوا۟ ۗ قُلْ فَٱدْرَءُوا۟ عَنْ أَنفُسِكُمُ ٱلْمَوْتَ إِن كُنتُمْ صَـٰدِقِينَ ﴿١٦٨﴾
(بۇ يەنە اﷲ نىڭ) ئۇرۇشقا چىقىشنى خالىمىغان (مۇناپىقلارنى ئايرىشى ئۈچۈندۇركى، ئۇلار ئۆيلىرىدە) ئولتۇرۇپ قېرىنداشلىرىغا: «ئەگەر ئۇلار (يەنى مۆمىنلەر) بىزگە بويسۇنغان بولسا (يەنى نەسىھىتىمىزنى ئاڭلاپ ئۇرۇشتىن قايتقان بولسا)، ئۆلتۈرۈلمەيتتى» دېدى. (ئى مۇھەممەد! بۇ مۇناپىقلارغا) ئېيتقىنكى، «(ئەگەر ئۇرۇشقا چىقماسلىق ئۆلۈمدىن قۇتۇلدۇرىدىغان بولسا) ئۆلۈمنى ئۆزۈڭلاردىن دەپئى قىلىپ بېقىڭلار، ئەگەر (دەۋايىڭلاردا) راستچىل بولساڭلار»
وَلَا تَحْسَبَنَّ ٱلَّذِينَ قُتِلُوا۟ فِى سَبِيلِ ٱللَّهِ أَمْوَٰتًۢا ۚ بَلْ أَحْيَآءٌ عِندَ رَبِّهِمْ يُرْزَقُونَ ﴿١٦٩﴾
اﷲ نىڭ يولىدا شېھىت بولغانلارنى ئۆلۈك دەپ گۇمان قىلمىغىن، بەلكى ئۇلار تىرىك بولۇپ، اﷲ نىڭ دەرگاھىدىكى رىزىقتىن بەھرىمەن قىلىنىدۇ (يەنى مەڭگۈلۈك باغلارنىڭ نېمەتلىرىدىن ئەتىگەن - ئاخشامدا مەڭگۈلۈك رىزىقلاندۇرۇپ تۇرۇلىدۇ)
فَرِحِينَ بِمَآ ءَاتَىٰهُمُ ٱللَّهُ مِن فَضْلِهِۦ وَيَسْتَبْشِرُونَ بِٱلَّذِينَ لَمْ يَلْحَقُوا۟ بِهِم مِّنْ خَلْفِهِمْ أَلَّا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُونَ ﴿١٧٠﴾
ئۇلار اﷲ نىڭ ئۆزلىرىگە بەرگەن پەزلىدىن خۇرسەندۇر، ئۆزلىرىنىڭ ئارقىسىدىن تېخى يېتىپ كەلمىگەن (يەنى شېھىت بولماي تىرىك قالغان) قېرىنداشلىرىغا (ئاخىرەتتە) نە قورقۇنچ، (دۇنيادىن ئايرىلغانلىقىغا) نە قايغۇ يوق ئىكەنلىكى بىلەن خۇش خەۋەر بېرىشنى تىلەيدۇ
۞ يَسْتَبْشِرُونَ بِنِعْمَةٍ مِّنَ ٱللَّهِ وَفَضْلٍ وَأَنَّ ٱللَّهَ لَا يُضِيعُ أَجْرَ ٱلْمُؤْمِنِينَ ﴿١٧١﴾
ئۇلار اﷲ تەرىپىدىن بېرىلگەن نېمەت ۋە پەزلنى، مۆمىنلەرنىڭ ئەجرىنى اﷲ نىڭ بىكار قىلىۋەتمەيدىغانلىقى بىلەن خۇش خەۋەر بېرىشنى تىلەيدۇ
ٱلَّذِينَ ٱسْتَجَابُوا۟ لِلَّهِ وَٱلرَّسُولِ مِنۢ بَعْدِ مَآ أَصَابَهُمُ ٱلْقَرْحُ ۚ لِلَّذِينَ أَحْسَنُوا۟ مِنْهُمْ وَٱتَّقَوْا۟ أَجْرٌ عَظِيمٌ ﴿١٧٢﴾
(ئۇھۇد غازىتىدا) يارىلانغاندىن كېيىن اﷲ نىڭ ۋە خۇدانىڭ ئەلچىسىنىڭ چاقىرىقىغا ئاۋاز قوشقانلار، ئۇلار (يەنى مۆمىنلەر) دىن ياخشى ئەمەللەرنى قىلغانلار ۋە تەقۋادارلىق قىلغانلار چوڭ ساۋابقا ئېرىشىدۇ
ٱلَّذِينَ قَالَ لَهُمُ ٱلنَّاسُ إِنَّ ٱلنَّاسَ قَدْ جَمَعُوا۟ لَكُمْ فَٱخْشَوْهُمْ فَزَادَهُمْ إِيمَـٰنًا وَقَالُوا۟ حَسْبُنَا ٱللَّهُ وَنِعْمَ ٱلْوَكِيلُ ﴿١٧٣﴾
(مۇشرىكلار تەرەپدارى بولغان) ئادەملەر ئۇلارغا: «شەك - شۈبھىسىزكى، كىشىلەر (يەنى قۇرەيشلەر) سىلەرگە قارشى قوشۇن توپلىدى، ئۇلاردىن قورقۇڭلار» دېدى. بۇ سۆز ئۇلارنىڭ ئىمانىنى كۈچەيتتى. ئۇلار: «بىزگە اﷲ كۇپايە، اﷲ نېمىدېگەن ياخشى ھامىي!» دېدى
فَٱنقَلَبُوا۟ بِنِعْمَةٍ مِّنَ ٱللَّهِ وَفَضْلٍ لَّمْ يَمْسَسْهُمْ سُوٓءٌ وَٱتَّبَعُوا۟ رِضْوَٰنَ ٱللَّهِ ۗ وَٱللَّهُ ذُو فَضْلٍ عَظِيمٍ ﴿١٧٤﴾
ئۇلار اﷲ نىڭ نېمىتى ۋە پەزلى بىلەن قايتىپ كەلدى، ھېچقانداق زىيان - زەخمەتكە ئۇچرىمىدى، ئۇلار اﷲ نىڭ رازىلىقىنى ئىزدىدى. اﷲ ئۇلۇغ پەزل ئىگىسىدۇر
إِنَّمَا ذَٰلِكُمُ ٱلشَّيْطَـٰنُ يُخَوِّفُ أَوْلِيَآءَهُۥ فَلَا تَخَافُوهُمْ وَخَافُونِ إِن كُنتُم مُّؤْمِنِينَ ﴿١٧٥﴾
ئەنە شۇ شەيتان (ئازدۇرغۇچى) ئۆز دوستلىرىنى (يەنى كۇففارلارنى) قورقىتىدۇ. ئەگەر مۆمىن بولساڭلار، ئۇلاردىن قورقماڭلار، مەندىن قورقۇڭلار
وَلَا يَحْزُنكَ ٱلَّذِينَ يُسَـٰرِعُونَ فِى ٱلْكُفْرِ ۚ إِنَّهُمْ لَن يَضُرُّوا۟ ٱللَّهَ شَيْـًٔا ۗ يُرِيدُ ٱللَّهُ أَلَّا يَجْعَلَ لَهُمْ حَظًّا فِى ٱلْـَٔاخِرَةِ ۖ وَلَهُمْ عَذَابٌ عَظِيمٌ ﴿١٧٦﴾
(ئى مۇھەممەد!) كۇفرىغا ئۆزلىرىنى ئاتىغانلار (يەنى سۆزلىرى ۋە ھەرىكەتلىرى بىلەن كۇفرىغا شاپاشلايدىغان مۇناپىقلار) سېنى غەمكىن قىلمىسۇن، ئۇلار ئىشەنچلىك ھالدا (كۇفرى بىلەن) اﷲ قا قىلچە زىيان يەتكۈزەلمەيدۇ. اﷲ ئاخىرەتتە ئۇلارغا (ساۋابتىن) ھېچبىر نېسىۋە بەرمەسلىكنى خالايدۇ؛ ئۇلار چوڭ ئازابقا دۇچار بولىدۇ
إِنَّ ٱلَّذِينَ ٱشْتَرَوُا۟ ٱلْكُفْرَ بِٱلْإِيمَـٰنِ لَن يَضُرُّوا۟ ٱللَّهَ شَيْـًٔا وَلَهُمْ عَذَابٌ أَلِيمٌ ﴿١٧٧﴾
ئىماننى كۇفرىغا تېگىشكەنلەر (يەنى مەزكۇر مۇناپىقلار) (كۇفرى بىلەن) اﷲ قا قىلچە زىيان يەتكۈزەلمەيدۇ، ئۇلار قاتتىق ئازابقا دۇچار بولىدۇ
وَلَا يَحْسَبَنَّ ٱلَّذِينَ كَفَرُوٓا۟ أَنَّمَا نُمْلِى لَهُمْ خَيْرٌ لِّأَنفُسِهِمْ ۚ إِنَّمَا نُمْلِى لَهُمْ لِيَزْدَادُوٓا۟ إِثْمًا ۚ وَلَهُمْ عَذَابٌ مُّهِينٌ ﴿١٧٨﴾
ھەقىقەتنى رەت قىلغۇچىلار ئۆزلىرىگە (جازا بەرمەستىن) مۆھلەت بېرىشىمىزنى (يەنى ئۆمرىنى ئۇزۇن قىلىشىمىزنى) ھەرگىز ئۆزلىرى ئۈچۈن پايدىلىق دەپ گۇمان قىلمىسۇن، ئۇلارغا مۆھلەت بېرىشىمىز (ۋە ئەجىلىنى كېچىكتۈرۈشىمىز) پەقەت ئۇلارنىڭ گۇناھىنىڭ كۆپىيىشى ئۈچۈندۇر. ئۇلار (ئاخىرەتتە) خار قىلغۇچى ئازابقا دۇچار بولىدۇ
مَّا كَانَ ٱللَّهُ لِيَذَرَ ٱلْمُؤْمِنِينَ عَلَىٰ مَآ أَنتُمْ عَلَيْهِ حَتَّىٰ يَمِيزَ ٱلْخَبِيثَ مِنَ ٱلطَّيِّبِ ۗ وَمَا كَانَ ٱللَّهُ لِيُطْلِعَكُمْ عَلَى ٱلْغَيْبِ وَلَـٰكِنَّ ٱللَّهَ يَجْتَبِى مِن رُّسُلِهِۦ مَن يَشَآءُ ۖ فَـَٔامِنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۚ وَإِن تُؤْمِنُوا۟ وَتَتَّقُوا۟ فَلَكُمْ أَجْرٌ عَظِيمٌ ﴿١٧٩﴾
اﷲ ياماننى (يەنى مۇناپىقنى) ياخشىدىن (يەنى مۆمىندىن) ئايرىمىغۇچە مۆمىنلەرنى (يەنى سىلەرنى) ھازىرقى پېتىڭلارچە قويمايدۇ. اﷲ سىلەرنى غەيبتىنمۇ خەۋەردار قىلمايدۇ ۋە لېكىن اﷲ خۇدانىڭ ئەلچىسىلىرىدىن خالىغىنىنى تاللاپ، ئۇنىڭغا (مۇھەممەد ئەلەيھىسسالامغا مۇناپىقلارنىڭ ھالىنى بىلدۈرگەندەك) غەيبنى بىلدۈرىدۇ؛ اﷲ قا ۋە ئۇنىڭ خۇدانىڭ ئەلچىسىلىرىگە ئىمان ئېيتىڭلار، ئەگەر سىلەر ئىمان ئېيتساڭلار ۋە (پەرۋەردىگارىڭلاردىن) قورقساڭلار، چوڭ ساۋابقا ئېرىشىسىلەر
وَلَا يَحْسَبَنَّ ٱلَّذِينَ يَبْخَلُونَ بِمَآ ءَاتَىٰهُمُ ٱللَّهُ مِن فَضْلِهِۦ هُوَ خَيْرًا لَّهُم ۖ بَلْ هُوَ شَرٌّ لَّهُمْ ۖ سَيُطَوَّقُونَ مَا بَخِلُوا۟ بِهِۦ يَوْمَ ٱلْقِيَـٰمَةِ ۗ وَلِلَّهِ مِيرَٰثُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۗ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌ ﴿١٨٠﴾
اﷲ ئۆز پەزلىدىن بەرگەن نەرسىلەرگە (يەنى پۇل - مالغا) بېخىللىق قىلىدىغانلار بېخىللىقنى ئۆزلىرى ئۈچۈن پايدىلىق دەپ گۇمان قىلمىسۇن، ئەمەلدە بۇ ئۇلار ئۈچۈن زىيانلىقتۇر؛ ئۇلارنىڭ بېخىللىق قىلغان نەرسىسى قىيامەت كۈنى ئۇلارنىڭ بوينىغا تاقاق قىلىپ سېلىنىدۇ. ئاسمانلارنىڭ ۋە يەرنىڭ مىراسى اﷲ نىڭدۇر (يەنى كائىناتتىكى ھەممە نەرسە اﷲ نىڭ مۈلكى بولۇپ، ئۇلار پانىي بولغاندىن كېيىن اﷲ نىڭ دەرگاھىغا قايتىدۇ). اﷲ قىلغان ئەمەللىرىڭلاردىن خەۋەرداردۇر
لَّقَدْ سَمِعَ ٱللَّهُ قَوْلَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ فَقِيرٌ وَنَحْنُ أَغْنِيَآءُ ۘ سَنَكْتُبُ مَا قَالُوا۟ وَقَتْلَهُمُ ٱلْأَنۢبِيَآءَ بِغَيْرِ حَقٍّ وَنَقُولُ ذُوقُوا۟ عَذَابَ ٱلْحَرِيقِ ﴿١٨١﴾
«اﷲ بولسا ئىشەنچلىك ھالدا پېقىر، بىز بولساق باي» دېگەن كىشىلەر (يەنى يەھۇدىيلار) نىڭ سۆزىنى اﷲ ئىشەنچلىك ھالدا ئاڭلىدى، ئۇلارنىڭ سۆزلىرىنى ۋە خۇدانىڭ ئەلچى-رەسۇللىرىنى ناھەق ئۆلتۈرگەنلىكلىرىنى خاتىرىلەپ قويىمىز (يەنى نامە - ئەمالىغا مۇئەككەل خۇدانىڭ ئەلچى پەرىشتىلىرىگە بۇنى ئۇلارنىڭ نامە - ئەمالىغا يېزىشنى بۇيرۇيمىز) ھەمدە ئاخىرەتتە خۇدانىڭ ئەلچى پەرىشتىلىرى تىلى ئارقىلىق، كۆيدۈرگۈچى ئازابنى تېتىڭلار، دەيمىز
ذَٰلِكَ بِمَا قَدَّمَتْ أَيْدِيكُمْ وَأَنَّ ٱللَّهَ لَيْسَ بِظَلَّامٍ لِّلْعَبِيدِ ﴿١٨٢﴾
بۇ سىلەرنىڭ قىلمىشىڭلار تۈپەيلىدىندۇر، اﷲ بەندىلىرىگە زۇلۇم قىلغۇچى ئەمەستۇر
ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ عَهِدَ إِلَيْنَآ أَلَّا نُؤْمِنَ لِرَسُولٍ حَتَّىٰ يَأْتِيَنَا بِقُرْبَانٍ تَأْكُلُهُ ٱلنَّارُ ۗ قُلْ قَدْ جَآءَكُمْ رُسُلٌ مِّن قَبْلِى بِٱلْبَيِّنَـٰتِ وَبِٱلَّذِى قُلْتُمْ فَلِمَ قَتَلْتُمُوهُمْ إِن كُنتُمْ صَـٰدِقِينَ ﴿١٨٣﴾
ئۇلار: «(تەۋراتتا) اﷲ بىزنى (ئاسماندىن چۈشكەن) ئوت يەپ كېتىدىغان بىر قۇربانلىق كەلتۈرمىگىچە ئىشەنچلىك ھالدا ھېچبىر خۇدانىڭ ئەلچىسىگە ئىمان كەلتۈرمەسلىككە بۇيرۇدى» دېدى (بۇ ئىپتىرا بولۇپ، اﷲ ئۇلارنى بۇيرۇغان ئەمەس). (ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «سىلەرگە مۆجىزىلەر ۋە سىلەر ئېيتقان مۆجىزە بىلەن ئىشەنچلىك ھالدا نۇرغۇن خۇدانىڭ ئەلچىسى كەلدى؛ ئەگەر سىلەر (دەۋايىڭلاردا) راستچىل بولساڭلار، ئۇلارنى نېمىشقا ئۆلتۈردۈڭلار؟»
فَإِن كَذَّبُوكَ فَقَدْ كُذِّبَ رُسُلٌ مِّن قَبْلِكَ جَآءُو بِٱلْبَيِّنَـٰتِ وَٱلزُّبُرِ وَٱلْكِتَـٰبِ ٱلْمُنِيرِ ﴿١٨٤﴾
(ئى مۇھەممەد!) ئۇلار سېنى يالغانغا چىقارغان بولسا (قايغۇرۇپ كەتمىگىن)، سەندىن ئىلگىرى ئۆتكەن، مۆجىزىلەرنى، مۇقەددەس كىتاب-ۋەھىيلارنى (يەنى ئىبراھىم ئەلەيھىسسالامغا نازىل قىلىنغان سەھىپىلەرگە ئوخشاش ساماۋى مۇقەددەس كىتاب-ۋەھىيلارنى) ۋە نۇرلۇق مۇقەددەس كىتاب-ۋەھىينى (يەنى تەۋرات، ئىنجىلغا ئوخشاشلارنى) ئېلىپ كەلگەن نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرىمۇ يالغانغا چىقىرىلغان ئىدى
كُلُّ نَفْسٍ ذَآئِقَةُ ٱلْمَوْتِ ۗ وَإِنَّمَا تُوَفَّوْنَ أُجُورَكُمْ يَوْمَ ٱلْقِيَـٰمَةِ ۖ فَمَن زُحْزِحَ عَنِ ٱلنَّارِ وَأُدْخِلَ ٱلْجَنَّةَ فَقَدْ فَازَ ۗ وَمَا ٱلْحَيَوٰةُ ٱلدُّنْيَآ إِلَّا مَتَـٰعُ ٱلْغُرُورِ ﴿١٨٥﴾
ھەر بىرجان ئىگىسى ئۆلۈمنىڭ تەمىنى تېتىغۇچىدۇر. سىلەرنىڭ ئەجرىڭلار (يەنى ئەمەللىرىڭلارنىڭ ساۋابى) قىيامەت كۈنى تولۇق بېرىلىدۇ، كىمكى دوزاختىن يىراق قىلىنغان ۋە مەڭگۈلۈك باغلاركە كىرگۈزۈلگەن ئىكەن، ئۇ مۇرادىغا يەتكەن بولىدۇ. دۇنيا تىرىكچىلىكى پەقەت ئالدايدىغان مالدىنلا ئىبارەتتۇر (يەنى ئالدايدىغان نەرسە بولۇپ، ئازغىنا بەھرىمەن بولغاندىن كېيىن تۈگەيدۇ)
۞ لَتُبْلَوُنَّ فِىٓ أَمْوَٰلِكُمْ وَأَنفُسِكُمْ وَلَتَسْمَعُنَّ مِنَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ وَمِنَ ٱلَّذِينَ أَشْرَكُوٓا۟ أَذًى كَثِيرًا ۚ وَإِن تَصْبِرُوا۟ وَتَتَّقُوا۟ فَإِنَّ ذَٰلِكَ مِنْ عَزْمِ ٱلْأُمُورِ ﴿١٨٦﴾
سىلەر ماللىرىڭلاردا (سەدىقىغە بۇيرۇلۇش ۋە ئاپەت يېتىشى بىلەن) ۋە جانلىرىڭلاردا (قەتل قىلىنىش، ئەسىر ئېلىنىش، كېسەل بولۇش بىلەن) چوقۇم سىنىلىسىلەر، سىلەردىن ئىلگىرى مۇقەددەس كىتاب-ۋەھىي بېرىلگەنلەر (يەنى يەھۇدىيلار ۋە ناسارالار) دىن، مۇشرىكلاردىن چوقۇم نۇرغۇن يامان سۆز ئاڭلايسىلەر، ئەگەر (يۇقىرىقى ئەھۋاللار يۈز بەرگەندە) سەۋر قىلساڭلار (سۆزۈڭلار ۋە ھەرىكىتىڭلاردا اﷲ تىن) قورقساڭلار، ئۇنداقتا بۇ ئىشەنچلىك ھالدا ئىرادە بىلەن قىلىنىشقا تېگىشلىك ئىشلاردىندۇر
وَإِذْ أَخَذَ ٱللَّهُ مِيثَـٰقَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَـٰبَ لَتُبَيِّنُنَّهُۥ لِلنَّاسِ وَلَا تَكْتُمُونَهُۥ فَنَبَذُوهُ وَرَآءَ ظُهُورِهِمْ وَٱشْتَرَوْا۟ بِهِۦ ثَمَنًا قَلِيلًا ۖ فَبِئْسَ مَا يَشْتَرُونَ ﴿١٨٧﴾
ئۆز ۋاقتىدا، اﷲ مۇقەددەس كىتاب-ۋەھىي بېرىلگەنلەر (يەنى يەھۇدىيلار) دىن مۇقەددەس كىتاب-ۋەھىينى (يەنى مۇقەددەس كىتاب-ۋەھىيتىكى اﷲ نىڭ ئەھكاملىرىنى) كىشىلەرگە چوقۇم بايان قىلىپ بېرىشكە ۋە يوشۇرماسلىققا ئەھدە ئالدى، لېكىن بۇنى (يەنى ئەھدىنى) ئۇلار ئارقىسىغا چۆرۈۋەتتى ۋە ئۇنى ئازغىنا بەدەلگە (يەنى دۇنيانىڭ ئەرزىمەس نەرسىسىگە) تېگىشتى. ئۇلارنىڭ تېگىشىپ ئالغىنى نېمىدېگەن يامان!
لَا تَحْسَبَنَّ ٱلَّذِينَ يَفْرَحُونَ بِمَآ أَتَوا۟ وَّيُحِبُّونَ أَن يُحْمَدُوا۟ بِمَا لَمْ يَفْعَلُوا۟ فَلَا تَحْسَبَنَّهُم بِمَفَازَةٍ مِّنَ ٱلْعَذَابِ ۖ وَلَهُمْ عَذَابٌ أَلِيمٌ ﴿١٨٨﴾
(ئى مۇھەممەد!) قىلغان ئىشلىرىدىن (يەنى ھەقنى ياپقانلىقىدىن) خۇشال بولىدىغان، قىلمىغان ئىشلىرى بىلەن مەدھىيىلىنىشنى ياخشى كۆرىدىغان كىشىلەرنى ھەرگىز اﷲ نىڭ ئازابىدىن قۇتۇلىدۇ دەپ گۇمان قىلمىغىن، ئۇلار قاتتىق ئازابقا دۇچار بولىدۇ
وَلِلَّهِ مُلْكُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿١٨٩﴾
ئاسمانلارنىڭ ۋە زېمىننىڭ پادىشاھلىقى اﷲ قا خاستۇر، اﷲ ھەممە نەرسىگە قادىردۇر
إِنَّ فِى خَلْقِ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَٱخْتِلَـٰفِ ٱلَّيْلِ وَٱلنَّهَارِ لَـَٔايَـٰتٍ لِّأُو۟لِى ٱلْأَلْبَـٰبِ ﴿١٩٠﴾
شۈبھىسىزكى، ئاسمانلارنىڭ ۋە زېمىننىڭ يارىتىلىشىدا، كېچە بىلەن كۈندۈزنىڭ نۆۋەتلىشىپ تۇرۇشىدا ئەقىل ئىگىلىرى ئۈچۈن، ئەلۋەتتە، روشەن دەلىللەر بار
ٱلَّذِينَ يَذْكُرُونَ ٱللَّهَ قِيَـٰمًا وَقُعُودًا وَعَلَىٰ جُنُوبِهِمْ وَيَتَفَكَّرُونَ فِى خَلْقِ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ رَبَّنَا مَا خَلَقْتَ هَـٰذَا بَـٰطِلًا سُبْحَـٰنَكَ فَقِنَا عَذَابَ ٱلنَّارِ ﴿١٩١﴾
ئۇلار ئۆرە تۇرغاندىمۇ، ئولتۇرغاندىمۇ، ياتقاندىمۇ اﷲ نى ئەسلەپ تۇرىدۇ، ئاسمانلارنىڭ ۋە زېمىننىڭ يارىتىلىشى توغرىسىدا پىكىر يۈرگۈزىدۇ. (ئۇلار ئېيتىدۇ) «پەرۋەردىگارىمىز! بۇنى بىكار ياراتمىدىڭ، سەن پاكتۇرسەن، بىزنى دوزاخ ئازابىدىن ساقلىغىن
رَبَّنَآ إِنَّكَ مَن تُدْخِلِ ٱلنَّارَ فَقَدْ أَخْزَيْتَهُۥ ۖ وَمَا لِلظَّـٰلِمِينَ مِنْ أَنصَارٍ ﴿١٩٢﴾
پەرۋەردىگارىمىز! سەن كىمنىكى دوزاخقا كىرگۈزىدىكەنسەن، ئۇنى ئەلۋەتتە خار قىلغان بولىسەن. زالىملارغا ھېچقانداق ياردەمچى بولمايدۇ
رَّبَّنَآ إِنَّنَا سَمِعْنَا مُنَادِيًا يُنَادِى لِلْإِيمَـٰنِ أَنْ ءَامِنُوا۟ بِرَبِّكُمْ فَـَٔامَنَّا ۚ رَبَّنَا فَٱغْفِرْ لَنَا ذُنُوبَنَا وَكَفِّرْ عَنَّا سَيِّـَٔاتِنَا وَتَوَفَّنَا مَعَ ٱلْأَبْرَارِ ﴿١٩٣﴾
پەرۋەردىگارىمىز! بىز ئىشەنچلىك ھالدا بىر چاقىرغۇچىنىڭ (يەنى مۇھەممەد ئەلەيھىسسالامنىڭ) 'رەببىڭلارغا ئىمان ئېيتىڭلار' دەپ ئىمانغا چاقىرغانلىقىنى ئاڭلىدۇق، ئىمان ئېيتتۇق، پەرۋەردىگارىمىز! بىزنىڭ گۇناھلىرىمىزنى مەغپىرەت قىلغىن، يامانلىقلىرىمىزنى يوققا چىقارغىن، بىزنى ياخشىلارنىڭ قاتارىدا قەبزى روھ قىلغىن
رَبَّنَا وَءَاتِنَا مَا وَعَدتَّنَا عَلَىٰ رُسُلِكَ وَلَا تُخْزِنَا يَوْمَ ٱلْقِيَـٰمَةِ ۗ إِنَّكَ لَا تُخْلِفُ ٱلْمِيعَادَ ﴿١٩٤﴾
پەرۋەردىگارىمىز! بىزگە خۇدانىڭ ئەلچىسىلىرىڭ ئارقىلىق ۋەدە قىلغان نەرسىنى (يەنى ئىتائەت قىلغانلارغا خاس بولغان مەڭگۈلۈك باغلارنى) بەرگىن، قىيامەت كۈنى بىزنى رەسۋا قىلمىغىن. سەن ئىشەنچلىك ھالدا ۋەدەڭگە خىلاپلىق قىلمايسەن»
فَٱسْتَجَابَ لَهُمْ رَبُّهُمْ أَنِّى لَآ أُضِيعُ عَمَلَ عَـٰمِلٍ مِّنكُم مِّن ذَكَرٍ أَوْ أُنثَىٰ ۖ بَعْضُكُم مِّنۢ بَعْضٍ ۖ فَٱلَّذِينَ هَاجَرُوا۟ وَأُخْرِجُوا۟ مِن دِيَـٰرِهِمْ وَأُوذُوا۟ فِى سَبِيلِى وَقَـٰتَلُوا۟ وَقُتِلُوا۟ لَأُكَفِّرَنَّ عَنْهُمْ سَيِّـَٔاتِهِمْ وَلَأُدْخِلَنَّهُمْ جَنَّـٰتٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ ثَوَابًا مِّنْ عِندِ ٱللَّهِ ۗ وَٱللَّهُ عِندَهُۥ حُسْنُ ٱلثَّوَابِ ﴿١٩٥﴾
ئۇلارنىڭ دۇئاسىنى پەرۋەردىگارى ئىجابەت قىلدى: «مەن سىلەردىن ئەر بولسۇن، ئايال بولسۇن، ھەر قانداق بىر ياخشى ئىش قىلغۇچىنىڭ قىلغان ئەمەلىنى بىكار قىلىۋەتمەيمەن، سىلەر بىر - بىرىڭلاردىن تۆرەلگەن. ھىجرەت قىلغانلار، يۇرتلىرىدىن ھەيدەپ چىقىرىلغانلار، مېنىڭ يولۇمدا (يەنى اﷲ نىڭ دىنى ئۈچۈن) ئەزىيەت تارتقانلار، ئۇرۇشقا قاتناشقانلار، (يەنى مېنىڭ يولۇمدا ئۇرۇشقانلار) ۋە مېنىڭ يولۇمدا ئۆلتۈرۈلگەنلەرنىڭ گۇناھلىرىنى (مەغپىرىتىم ۋە رەھمىتىم بىلەن) ئەلۋەتتە، ئۇلارنى ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان مەڭگۈلۈك باغلارلەرگە كىرگۈزىمەن». بۇ (ئۇلارنىڭ ياخشى ئەمەللىرى ئۈچۈن) اﷲ تەرىپىدىن بېرىلگەن مۇكاپاتتۇر. اﷲ نىڭ دەرگاھىدا ياخشى مۇكاپات (يەنى مەڭگۈلۈك باغلار) بار
لَا يَغُرَّنَّكَ تَقَلُّبُ ٱلَّذِينَ كَفَرُوا۟ فِى ٱلْبِلَـٰدِ ﴿١٩٦﴾
ھەقىقەتنى رەت قىلغۇچىلارنىڭ (پۇل - مال، ئابرۇي، مەرتىۋە ئىزدەش يۈزىسىدىن) شەھەرلەردە كېزىپ يۈرۈشلىرى سېنى ئالدىمىسۇن
مَتَـٰعٌ قَلِيلٌ ثُمَّ مَأْوَىٰهُمْ جَهَنَّمُ ۚ وَبِئْسَ ٱلْمِهَادُ ﴿١٩٧﴾
بۇ ئازغىنا بەھرىمەن بولۇشتۇر، كەلگۈسىدە ئۇلارنىڭ جايى مەڭگۈلۈك ئوت بولىدۇ. ئۇ نېمىدېگەن يامان تۆشەك!
لَـٰكِنِ ٱلَّذِينَ ٱتَّقَوْا۟ رَبَّهُمْ لَهُمْ جَنَّـٰتٌ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا نُزُلًا مِّنْ عِندِ ٱللَّهِ ۗ وَمَا عِندَ ٱللَّهِ خَيْرٌ لِّلْأَبْرَارِ ﴿١٩٨﴾
پەرۋەردىگارىغا تەقۋادارلىق قىلغان كىشىلەر ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان مەڭگۈلۈك باغلارلەردىن بەھرىمەن بولۇپ، ئۇ يەردە مەڭگۈ قالىدۇ. بۇ، اﷲ تەرىپىدىن بولغان كۈتۈۋېلىشتۇر. اﷲ نىڭ دەرگاھىدىكى نەرسىلەر (يەنى مۇكاپات ۋە ھۆرمەت) ياخشى بەندىلەر ئۈچۈن (كۇففارلار بەھرىمەن بولغان باقاسى يوق ئازغىنا راھەتتىن) ئارتۇقتۇر
وَإِنَّ مِنْ أَهْلِ ٱلْكِتَـٰبِ لَمَن يُؤْمِنُ بِٱللَّهِ وَمَآ أُنزِلَ إِلَيْكُمْ وَمَآ أُنزِلَ إِلَيْهِمْ خَـٰشِعِينَ لِلَّهِ لَا يَشْتَرُونَ بِـَٔايَـٰتِ ٱللَّهِ ثَمَنًا قَلِيلًا ۗ أُو۟لَـٰٓئِكَ لَهُمْ أَجْرُهُمْ عِندَ رَبِّهِمْ ۗ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ﴿١٩٩﴾
ئەھلى مۇقەددەس كىتاب-ۋەھىي ئارىسىدا ئىشەنچلىك ھالدا اﷲ قا ئىمان كەلتۈرىدىغان، سىلەرگە نازىل بولغان مۇقەددەس كىتاب-ۋەھىيقا (يەنى قۇرئانغا)، ئۆزلىرىگە نازىل بولغان مۇقەددەس كىتاب-ۋەھىيقا (يەنى تەۋراتقا ۋە ئىنجىلغا) ئىشىنىدىغانلار بار؛ ئۇلار اﷲ تىن قورقىدۇ، اﷲ نىڭ ئايەتلىرىنى ئازغىنا بەدەلگە تېگىشمەيدۇ (يەنى ئۇلارنىڭ مۇقەددەس كىتاب-ۋەھىيلىرىدىكى مۇھەممەد ئەلەيھىسسالامنىڭ سۈپەتلىرى ۋە شەرىئەت ئەھكاملىرىنى، راھىبلارغا ئوخشاش، دۇنيانىڭ ئەرزىمەس نەرسىلىرىنى دەپ ئۆزگەرتىۋەتمەيدۇ)، ئۇلارغا پەرۋەردىگارىنىڭ دەرگاھىدا ساۋاب بېرىلىدۇ. اﷲ ئىشەنچلىك ھالدا تېز ھېساب ئالغۇچىدۇر
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱصْبِرُوا۟ وَصَابِرُوا۟ وَرَابِطُوا۟ وَٱتَّقُوا۟ ٱللَّهَ لَعَلَّكُمْ تُفْلِحُونَ ﴿٢٠٠﴾
ئى مۆمىنلەر! تائەت - ئىبادەتنىڭ مۇشەققەتلىرىگە ۋە سىلەرگە يەتكەن ئېغىرچىلىقلارغا سەۋر قىلىڭلار، دۈشمەنلەرگە زىيادە چىداملىق بولۇڭلار، (چېگرالىرىڭلارنى ساقلاپ) جىھادقا تەييار تۇرۇڭلار، مەقسىتىڭلارغا يېتىش ئۈچۈن اﷲ تىن قورقۇڭلار (يەنى ئۇنىڭ ئەمرىگە مۇخالىپەتچىلىك قىلماڭلار)
