قۇرئان ئاتا - بوۋىلىرى ئاگاھلاندۇرۇلمىغان (يەنى ئۇلارغا خۇدانىڭ ئەلچىسى كەلمىگەن)، غاپىل قالغان بىر قەۋمنى ئاگاھلاندۇرۇشۇڭ ئۈچۈن، غالىب، ناھايىتى مېھرىبان اﷲ تەرىپىدىن نازىل قىلىنغاندۇر
لَقَدْ حَقَّ ٱلْقَوْلُ عَلَىٰٓ أَكْثَرِهِمْ فَهُمْ لَا يُؤْمِنُونَ ﴿٧﴾
ئۇلارنىڭ كۆپچىلىكىگە (ئازاب توغرىسىدىكى) ھۆكۈم ئىشەنچلىك ھالدا تېگىشلىك بولدى. شۇنىڭ ئۈچۈن ئۇلار ئىمان ئېيتمايدۇ
إِنَّا جَعَلْنَا فِىٓ أَعْنَـٰقِهِمْ أَغْلَـٰلًا فَهِىَ إِلَى ٱلْأَذْقَانِ فَهُم مُّقْمَحُونَ ﴿٨﴾
بىز ئىشەنچلىك ھالدا ئۇلارنىڭ بويۇنلىرىغا ئېڭەكلىرىگىچە تاقاقلارنى تاقىدۇق، شۇنىڭ ئۈچۈن ئۇلار باشلىرىنى ئىگەلمەيدۇ (يەنى ئۇلار ئىمانغا بويسۇنمايدۇ ۋە ئىمانغا باش ئەگمەيدۇ)
وَجَعَلْنَا مِنۢ بَيْنِ أَيْدِيهِمْ سَدًّا وَمِنْ خَلْفِهِمْ سَدًّا فَأَغْشَيْنَـٰهُمْ فَهُمْ لَا يُبْصِرُونَ ﴿٩﴾
ئۇلارنىڭ ئالدىدا بىر توسۇق، كەينىدە بىر توسۇق قىلدۇق، ئۇلارنىڭ كۆزلىرىنى پەردىلىدۇق، شۇنىڭ ئۈچۈن ئۇلار (ئىماننىڭ يوللىرى توسۇلۇپ كەتكەنلىكى ئۈچۈن) ھېچ نەرسىنى كۆرەلمەيدۇ
وَسَوَآءٌ عَلَيْهِمْ ءَأَنذَرْتَهُمْ أَمْ لَمْ تُنذِرْهُمْ لَا يُؤْمِنُونَ ﴿١٠﴾
ئۇلارنى ئاگاھلاندۇرامسەن، ئاگاھلاندۇرمامسەن، ئۇلارغا بەرىبىر ئوخشاش، ئىمان ئېيتمايدۇ
إِنَّمَا تُنذِرُ مَنِ ٱتَّبَعَ ٱلذِّكْرَ وَخَشِىَ ٱلرَّحْمَـٰنَ بِٱلْغَيْبِ ۖ فَبَشِّرْهُ بِمَغْفِرَةٍ وَأَجْرٍ كَرِيمٍ ﴿١١﴾
سەن پەقەت قۇرئانغا ئەگەشكەن، مەرھەمەتلىك اﷲ نى كۆرمەي تۇرۇپ قورققان كىشىنى ئاگاھلاندۇرالايسەن (يەنى ئاگاھلاندۇرۇشۇڭنىڭ پايدىسى بولىدۇ)، ئۇنىڭغا (اﷲ نىڭ ئۇنىڭ گۇناھلىرىنى) مەغپىرەت قىلىدىغانلىقى ۋە ئېسىل ساۋاب (يەنى مەڭگۈلۈك باغلار ئاتا قىلىدىغانلىقى) بىلەن خۇش خەۋەر بەرگىن
إِنَّا نَحْنُ نُحْىِ ٱلْمَوْتَىٰ وَنَكْتُبُ مَا قَدَّمُوا۟ وَءَاثَـٰرَهُمْ ۚ وَكُلَّ شَىْءٍ أَحْصَيْنَـٰهُ فِىٓ إِمَامٍ مُّبِينٍ ﴿١٢﴾
بىز، شۈبھىسىزكى، ئۆلۈكلەرنى تىرىلدۈرىمىز، ئۇلارنىڭ (دۇنيادا) قىلغان (ياخشى - يامان) ئەمەللىرىنى ۋە ئىشلىرىنى خاتىرىلەپ قويىمىز، ھەممە شەيئىنى روشەن دەپتەردە (يەنى لەۋھۇلمەھپۇزدا) تولۇق خاتىرىلەپ قويغانمىز
وَٱضْرِبْ لَهُم مَّثَلًا أَصْحَـٰبَ ٱلْقَرْيَةِ إِذْ جَآءَهَا ٱلْمُرْسَلُونَ ﴿١٣﴾
ئۇلارغا (يەنى كۇففارلارغا) شەھەر (يەنى ئەنتاكىيە) ئاھالىسىنى مىسال قىلىپ كەلتۈرگىن، ئۆز ۋاقتىدا ئۇلارغا خۇدانىڭ ئەلچى-رەسۇللىرى كەلگەن ئىدى
إِذْ أَرْسَلْنَآ إِلَيْهِمُ ٱثْنَيْنِ فَكَذَّبُوهُمَا فَعَزَّزْنَا بِثَالِثٍ فَقَالُوٓا۟ إِنَّآ إِلَيْكُم مُّرْسَلُونَ ﴿١٤﴾
ئەينى زاماندا ئۇلارغا ئىككى خۇدانىڭ ئەلچىسى ئەۋەتتۇق. ئۇلار ئۇ ئىككى خۇدانىڭ ئەلچىسىنى ئىنكار قىلدى. ئۈچىنچى خۇدانىڭ ئەلچىسىنى ئەۋەتىش بىلەن (ئۇلارغا) ياردەم قىلدۇق، خۇدانىڭ ئەلچى-رەسۇللىرى (ئۇلارغا): «بىز ئىشەنچلىك ھالدا (اﷲ نىڭ) سىلەرگە (يەنى سىلەرنى ھىدايەت قىلىشقا) ئەۋەتىلگەن ئەلچىلىرىمىز» دېدى
قَالُوا۟ مَآ أَنتُمْ إِلَّا بَشَرٌ مِّثْلُنَا وَمَآ أَنزَلَ ٱلرَّحْمَـٰنُ مِن شَىْءٍ إِنْ أَنتُمْ إِلَّا تَكْذِبُونَ ﴿١٥﴾
ئۇلار: «سىلەر پەقەت بىزگە ئوخشاش ئىنسان، مېھرىبان اﷲ ھېچ نەرسە نازىل قىلغىنى يوق، سىلەر پەقەت يالغانچىسىلەر» دېدى
قَالُوا۟ رَبُّنَا يَعْلَمُ إِنَّآ إِلَيْكُمْ لَمُرْسَلُونَ ﴿١٦﴾
ئەلچىلەر: «پەرۋەردىگارىمىز بىلىدۇكى، بىز شەك - شۈبھىسىز سىلەرگە ئەۋەتىلگەن ئەلچىلەرمىز
وَمَا عَلَيْنَآ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٧﴾
بىزنىڭ مەسئۇلىيىتىمىز پەقەت (روشەن دەلىللەر بىلەن) چۈشىنىشلىك قىلىپ تەبلىغ قىلىشتۇر» دېدى
قَالُوٓا۟ إِنَّا تَطَيَّرْنَا بِكُمْ ۖ لَئِن لَّمْ تَنتَهُوا۟ لَنَرْجُمَنَّكُمْ وَلَيَمَسَّنَّكُم مِّنَّا عَذَابٌ أَلِيمٌ ﴿١٨﴾
ئۇلار (ئەلچىلەرگە): «بىز سىلەردىن شۇم پال ئالدۇق. ئەگەر سىلەر (دەۋىتىڭلارنى) توختاتمىساڭلار، سىلەرنى چوقۇم تاش - كېسەك قىلىپ ئۆلتۈرىمىز ۋە بىزنىڭ قاتتىق جازايىمىزغا ئۇچرايسىلەر» دېدى
قَالُوا۟ طَـٰٓئِرُكُم مَّعَكُمْ ۚ أَئِن ذُكِّرْتُم ۚ بَلْ أَنتُمْ قَوْمٌ مُّسْرِفُونَ ﴿١٩﴾
ئەلچىلەر: «(كۇفرىڭلار سەۋەبىدىن) شۇملۇقۇڭلار ئۆزۈڭلار بىلەن بىللىدۇر، سىلەرگە ۋەز - نەسىھەت قىلىنسا (شۇم پال ئالامسىلەر؟) ھەرگىز ئۇنداق ئەمەس (يەنى ئىش سىلەر گۇمان قىلغاندەك ئەمەس)، سىلەر (اﷲ قا شېرىك كەلتۈرۈش بىلەن) ھەددىدىن ئاشقۇچى قەۋمسىلەر» دېدى
وَجَآءَ مِنْ أَقْصَا ٱلْمَدِينَةِ رَجُلٌ يَسْعَىٰ قَالَ يَـٰقَوْمِ ٱتَّبِعُوا۟ ٱلْمُرْسَلِينَ ﴿٢٠﴾
شەھەرنىڭ يىراق جايىدىن بىر كىشى (يەنى ھەبىب نەجار) يۈگۈرۈپ كېلىپ ئېيتتى: «ئى قەۋمىم! خۇدانىڭ ئەلچى-رەسۇللىرىگە ئەگىشىڭلار
ٱتَّبِعُوا۟ مَن لَّا يَسْـَٔلُكُمْ أَجْرًا وَهُم مُّهْتَدُونَ ﴿٢١﴾
توغرا يولدا بولغان، (ئىمانغا دەۋەت قىلغانلىقىغا) سىلەردىن ھەق سورىمايدىغان كىشىلەرگە ئەگىشىڭلار
وَمَا لِىَ لَآ أَعْبُدُ ٱلَّذِى فَطَرَنِى وَإِلَيْهِ تُرْجَعُونَ ﴿٢٢﴾
مەن نېمىشقا مېنى ياراتقان ۋە سىلەر (ئۆلگەندىن كېيىن) دەرگاھىغا قايتىدىغان اﷲ قا ئىبادەت قىلماي؟
ءَأَتَّخِذُ مِن دُونِهِۦٓ ءَالِهَةً إِن يُرِدْنِ ٱلرَّحْمَـٰنُ بِضُرٍّ لَّا تُغْنِ عَنِّى شَفَـٰعَتُهُمْ شَيْـًٔا وَلَا يُنقِذُونِ ﴿٢٣﴾
مەن اﷲ تىن غەيرىينى ئىلاھ قىلىۋالىدىغان بولسام، ئەگەر مېھرىبان اﷲ ماڭا بىرەر زىيان - زەخمەت يەتكۈزمەكچى بولسا، ئۇلارنىڭ شاپائىتى مەندىن ھېچ نەرسىنى دەپئى قىلالمايدۇ ھەمدە مېنى قۇتقۇزالمايدۇ
إِنِّىٓ إِذًا لَّفِى ضَلَـٰلٍ مُّبِينٍ ﴿٢٤﴾
ئەگەر ئۇنداق بولسا (يەنى ئاللادىن غەيرىينى ئىلاھ قىلىۋالىدىغان بولسام)، مەن ئىشەنچلىك ھالدا ئوپئوچۇق گۇمراھلىقتا بولىمەن
إِنِّىٓ ءَامَنتُ بِرَبِّكُمْ فَٱسْمَعُونِ ﴿٢٥﴾
مەن ئىشەنچلىك ھالدا (سىلەرنى ياراتقان) پەرۋەردىگارىڭلارغا ئىمان ئېيتتىم (ئىمانىمغا) قۇلاق سېلىڭلار (ۋە ئۇنىڭغا گۇۋاھ بولۇڭلار)»

ئۆلۈك (يەنى قۇرغاق) زېمىننى (يامغۇر بىلەن) تىرىلدۈرۈپ، ئۇنىڭدىن ئۇلار يەيدىغان ئاشلىقنى ئۆستۈرگەنلىكىمىز ئۇلارغا (اﷲ نىڭ قۇدرىتى كامىلەسىنى كۆرسىتىدىغان) ئالامەتتۇر
وَجَعَلْنَا فِيهَا جَنَّـٰتٍ مِّن نَّخِيلٍ وَأَعْنَـٰبٍ وَفَجَّرْنَا فِيهَا مِنَ ٱلْعُيُونِ ﴿٣٤﴾
بىز زېمىندا نۇرغۇنلىغان خورما باغلىرىنى، ئۈزۈم باغلىرىنى بەرپا قىلدۇق، شۇ زېمىندا بۇلاقلارنى ئېقىتتۇق
لِيَأْكُلُوا۟ مِن ثَمَرِهِۦ وَمَا عَمِلَتْهُ أَيْدِيهِمْ ۖ أَفَلَا يَشْكُرُونَ ﴿٣٥﴾
ئۇلار ئۇنىڭ (يەنى باغلارنىڭ) مېۋىسىنى يېسۇن، (بۇ مېۋىلەرنى) ئۇلارنىڭ قوللىرى ياراتقان ئەمەس، ئۇلار (اﷲ نىڭ نېمەتلىرىگە) شۈكۈر قىلمامدۇ؟
سُبْحَـٰنَ ٱلَّذِى خَلَقَ ٱلْأَزْوَٰجَ كُلَّهَا مِمَّا تُنۢبِتُ ٱلْأَرْضُ وَمِنْ أَنفُسِهِمْ وَمِمَّا لَا يَعْلَمُونَ ﴿٣٦﴾
اﷲ (پۈتۈن ئەيىب - نۇقسانلاردىن) پاكتۇر، ئۇ پۈتۈن شەيئىلەرنى جۈپ ياراتتى. زېمىندىن ئۈنۈپ چىقىدىغان نەرسىلەرنىڭ، ئۇلارنىڭ ئۆزلىرىنىڭ ۋە ئۇلار بىلمەيدىغان نەرسىلەر (يەنى ئاجايىپ مەخلۇقاتلار) نىڭ ھەممىسىنىڭ جۈپتى بار
وَءَايَةٌ لَّهُمُ ٱلَّيْلُ نَسْلَخُ مِنْهُ ٱلنَّهَارَ فَإِذَا هُم مُّظْلِمُونَ ﴿٣٧﴾
ئۇلارغا (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) بىر ئالامەت كېچىدىن ئىبارەتكى، ئۇنىڭدىن كۈندۈزنى ئايرىۋېتىمىز - دە (يەنى كۈندۈزنىڭ نۇرىنى يوق قىلىۋېتىمىز - دە)، ئۇلار ناگاھان قاراڭغۇدا قالىدۇ
وَٱلشَّمْسُ تَجْرِى لِمُسْتَقَرٍّ لَّهَا ۚ ذَٰلِكَ تَقْدِيرُ ٱلْعَزِيزِ ٱلْعَلِيمِ ﴿٣٨﴾
كۈن بەلگىلەنگەن جايىغا قاراپ سەير قىلىدۇ، بۇ، غالىب، ھەممىنى بىلگۈچى اﷲ نىڭ ئالدىنئالا بەلگىلىگەن ئىشىدۇر
وَٱلْقَمَرَ قَدَّرْنَـٰهُ مَنَازِلَ حَتَّىٰ عَادَ كَٱلْعُرْجُونِ ٱلْقَدِيمِ ﴿٣٩﴾
ئايغا سەير قىلىدىغان مەنزىللەرنى بەلگىلىدۇق، ئۇ (ئاخىرقى مەنزىلگە يېتىپ بارغاندا) خورمىنىڭ قۇرۇپ قالغان شېخىغا ئوخشاش بولۇپ قالىدۇ
لَا ٱلشَّمْسُ يَنۢبَغِى لَهَآ أَن تُدْرِكَ ٱلْقَمَرَ وَلَا ٱلَّيْلُ سَابِقُ ٱلنَّهَارِ ۚ وَكُلٌّ فِى فَلَكٍ يَسْبَحُونَ ﴿٤٠﴾
كۈننىڭ ئايغا يېتىۋېلىشى (يەنى ئىككىسىنىڭ جەم بولۇپ قېلىشى)، كېچىنىڭ كۈندۈزدىن ئېشىپ كېتىشى (يەنى ۋاقتى كەلمەستىن كۈندۈزنىڭ ئورنىنى ئېلىشى) مۇمكىن ئەمەس، ھەر بىرى پەلەكتە ئۈزۈپ تۇرىدۇ
وَءَايَةٌ لَّهُمْ أَنَّا حَمَلْنَا ذُرِّيَّتَهُمْ فِى ٱلْفُلْكِ ٱلْمَشْحُونِ ﴿٤١﴾
ئۇلارغا (بىزنىڭ كامالى قۇدرىتىمىزنى كۆرسىتىدىغان) ئالامەت شۇكى، بىز ئۇلارنىڭ ئاتا - بوۋىلىرىنى (نۇھ ئەلەيھىسسالامنىڭ نەرسە - كېرەك ۋە ھايۋاناتلار بىلەن) توشقۇزۇلغان كېمىسىگە چۈشۈردۇق
وَخَلَقْنَا لَهُم مِّن مِّثْلِهِۦ مَا يَرْكَبُونَ ﴿٤٢﴾
بىز ئۇلار ئۈچۈن شۇ كېمىگە ئوخشاش ئۇلار چۈشىدىغان نەرسىلەرنى ياراتتۇق
وَإِن نَّشَأْ نُغْرِقْهُمْ فَلَا صَرِيخَ لَهُمْ وَلَا هُمْ يُنقَذُونَ ﴿٤٣﴾
ئەگەر بىز خالىساق ئۇلارنى غەرق قىلىۋېتىمىز، ئۇلارغا ياردەم بەرگۈچىمۇ بولمايدۇ ۋە ئۇلارنى قۇتۇلدۇرغۇچىمۇ بولمايدۇ
إِلَّا رَحْمَةً مِّنَّا وَمَتَـٰعًا إِلَىٰ حِينٍ ﴿٤٤﴾
پەقەت بىز رەھىم قىلغانلىقىمىز ۋە ئۇلارنى مۇئەييەن مۇددەتكىچە (ھاياتتىن) بەھرىمەن قىلىدىغانلىقىمىز ئۈچۈن (ئۇلار قۇتۇلدۇرۇلىدۇ)
وَإِذَا قِيلَ لَهُمُ ٱتَّقُوا۟ مَا بَيْنَ أَيْدِيكُمْ وَمَا خَلْفَكُمْ لَعَلَّكُمْ تُرْحَمُونَ ﴿٤٥﴾
ئۇلارغا: «رەھمەتكە ئېرىشىشىڭلار ئۈچۈن ئالدىڭلاردىكىدىن (يەنى دۇنيا ئازابىدىن) ۋە ئارقاڭلاردىكىدىن (يەنى ئاخىرەت ئازابىدىن) قورقۇڭلار» دېيىلسە (ئۇلار بۇنىڭدىن يۈز ئۆرۈيدۇ)
وَمَا تَأْتِيهِم مِّنْ ءَايَةٍ مِّنْ ءَايَـٰتِ رَبِّهِمْ إِلَّا كَانُوا۟ عَنْهَا مُعْرِضِينَ ﴿٤٦﴾
پەرۋەردىگارىنىڭ (خۇدانىڭ ئەلچىسىنىڭ راستلىقىنى ئىسپاتلايدىغان) ئايەتلىرىدىن بىر ئايەت كەلسىلا، ئۇلار بۇنىڭدىن يۈز ئۆرۈيدۇ
وَإِذَا قِيلَ لَهُمْ أَنفِقُوا۟ مِمَّا رَزَقَكُمُ ٱللَّهُ قَالَ ٱلَّذِينَ كَفَرُوا۟ لِلَّذِينَ ءَامَنُوٓا۟ أَنُطْعِمُ مَن لَّوْ يَشَآءُ ٱللَّهُ أَطْعَمَهُۥٓ إِنْ أَنتُمْ إِلَّا فِى ضَلَـٰلٍ مُّبِينٍ ﴿٤٧﴾
ئەگەر ئۇلارغا: «اﷲ سىلەرگە رىزىق قىلىپ بەرگەن نەرسىلەردىن (پېقىرلارغا) سەدىقە قىلىپ بېرىڭلار» دېيىلسە، ھەقىقەتنى رەت قىلغۇچىلار مۆمىنلەرگە (مەسخىرە قىلىش يۈزىسىدىن): «ئەگەر اﷲ خالىسا تەمىنلەيدىغان ئادەمنى بىز تەمىنلەمدۇق؟ سىلەر پەقەت روشەن گۇمراھلىقتاسىلەر» دەيدۇ
وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٤٨﴾
ئۇلار: «ئەگەر راستچىل بولساڭلار، بۇ ۋەدە قاچان ئىشقا ئاشىدۇ؟» دەيدۇ
مَا يَنظُرُونَ إِلَّا صَيْحَةً وَٰحِدَةً تَأْخُذُهُمْ وَهُمْ يَخِصِّمُونَ ﴿٤٩﴾
ئۇلار پەقەت بىر ئاۋازنىلا (يەنى ئىسراپىل ئەلەيھىسسالامنىڭ بىرىنچى قېتىملىق چالغان سۇرىنىلا) كۈتىدۇ، ئۇلار (دۇنيادا سودا - سېتىق ۋە شۇنىڭغا ئوخشاش ئىشلار ئۈستىدە) جاڭجاللىشىپ تۇرغان ھالەتلىرىدە، ئۇ ئاۋاز ئۇلارنى ھالاك قىلىدۇ
فَلَا يَسْتَطِيعُونَ تَوْصِيَةً وَلَآ إِلَىٰٓ أَهْلِهِمْ يَرْجِعُونَ ﴿٥٠﴾
ئۇلار بىر - بىرىگە ۋەسىيەت قالدۇرۇشقىمۇ، ئائىلىسىگە قايتىشقىمۇ قادىر بولالمايدۇ
وَنُفِخَ فِى ٱلصُّورِ فَإِذَا هُم مِّنَ ٱلْأَجْدَاثِ إِلَىٰ رَبِّهِمْ يَنسِلُونَ ﴿٥١﴾
سۇر چېلىنغان ھامان ئۇلار قەبرىلىرىدىن چىقىپ پەرۋەردىگارى تەرەپكە يۈگۈرىدۇ
قَالُوا۟ يَـٰوَيْلَنَا مَنۢ بَعَثَنَا مِن مَّرْقَدِنَا ۜ ۗ هَـٰذَا مَا وَعَدَ ٱلرَّحْمَـٰنُ وَصَدَقَ ٱلْمُرْسَلُونَ ﴿٥٢﴾
ئۇلار: «ۋاي ئىسىت! بىزنى ئۇخلاۋاتقان يېرىمىزدىن (يەنى قەبرىمىزدىن) كىم ئويغاتتى؟» دەيدۇ (چۈنكى ئۇلار ئىككى سۇر ئارىسىدا ئۇيقۇدا بولۇپ ئازابتىن خالى بولىدۇ)، (خۇدانىڭ ئەلچى پەرىشتىلىرى يا مۆمىنلەر ئۇنىڭغا جاۋابەن) «مېھرىبان اﷲ نىڭ ۋەدە قىلغىنى مۇشۇ (كۈن)، خۇدانىڭ ئەلچى-رەسۇللىرى راست ئېيتقان» دەيدۇ
إِن كَانَتْ إِلَّا صَيْحَةً وَٰحِدَةً فَإِذَا هُمْ جَمِيعٌ لَّدَيْنَا مُحْضَرُونَ ﴿٥٣﴾
پەقەت بىر ئاۋاز بىلەنلا ئۇلارنىڭ ھەممىسى ھۇزۇرىمىزغا ھازىر قىلىنىدۇ
فَٱلْيَوْمَ لَا تُظْلَمُ نَفْسٌ شَيْـًٔا وَلَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٥٤﴾
ئۇ كۈندە ھېچبىر ئىنسانغا قىلچە زۇلۇم قىلىنمايدۇ، سىلەرگە پەقەت قىلمىشىڭلارغا يارىشا جازا بېرىلىدۇ
إِنَّ أَصْحَـٰبَ ٱلْجَنَّةِ ٱلْيَوْمَ فِى شُغُلٍ فَـٰكِهُونَ ﴿٥٥﴾
شۈبھىسىزكى، ئۇ كۈندە مەڭگۈلۈك باغلار ئەھلىلىرى نېمەت ئىچىدىدۇر
هُمْ وَأَزْوَٰجُهُمْ فِى ظِلَـٰلٍ عَلَى ٱلْأَرَآئِكِ مُتَّكِـُٔونَ ﴿٥٦﴾
ئۇلار ۋە ئۇلارنىڭ جۈپتىلىرى (مەڭگۈلۈك باغلارلەرنىڭ) سايىلىرى ئاستىدا تەختلەرگە يۆلەنگەن ھالدا تۇرىدۇ
لَهُمْ فِيهَا فَـٰكِهَةٌ وَلَهُم مَّا يَدَّعُونَ ﴿٥٧﴾
ئۇلار ئۈچۈن مەڭگۈلۈك باغلارتە (تۈرلۈك) مېۋىلەر بار، ئۇلار ئۈچۈن مەڭگۈلۈك باغلارتە (ئۇلارنىڭ) كۆڭلى تارتقان نەرسىلەر بولىدۇ
سَلَـٰمٌ قَوْلًا مِّن رَّبٍّ رَّحِيمٍ ﴿٥٨﴾
مېھرىبان پەرۋەردىگارى تەرىپىدىن (ئۇلارغا) سالام دېيىلىدۇ
وَٱمْتَـٰزُوا۟ ٱلْيَوْمَ أَيُّهَا ٱلْمُجْرِمُونَ ﴿٥٩﴾
(گۇناھكارلارغا) «ئى گۇناھكارلار! بۈگۈن (بىر تەرەپكە) ئايرىلىڭلار» (دېيىلىدۇ)
۞ أَلَمْ أَعْهَدْ إِلَيْكُمْ يَـٰبَنِىٓ ءَادَمَ أَن لَّا تَعْبُدُوا۟ ٱلشَّيْطَـٰنَ ۖ إِنَّهُۥ لَكُمْ عَدُوٌّ مُّبِينٌ ﴿٦٠﴾
سىلەرگە (خۇدانىڭ ئەلچىسىلىرىم ئارقىلىق): «ئى ئادەم بالىلىرى! شەيتان (ئازدۇرغۇچى)غا چوقۇنماڭلار، ئۇ ئىشەنچلىك ھالدا سىلەرگە ئاشكارا دۈشمەندۇر، ماڭىلا ئىبادەت قىلىڭلار، بۇ توغرا يولدۇر» دەپ تەۋسىيە قىلمىدىممۇ؟
وَأَنِ ٱعْبُدُونِى ۚ هَـٰذَا صِرَٰطٌ مُّسْتَقِيمٌ ﴿٦١﴾
سىلەرگە (خۇدانىڭ ئەلچىسىلىرىم ئارقىلىق): «ئى ئادەم بالىلىرى! شەيتان (ئازدۇرغۇچى)غا چوقۇنماڭلار، ئۇ ئىشەنچلىك ھالدا سىلەرگە ئاشكارا دۈشمەندۇر، ماڭىلا ئىبادەت قىلىڭلار، بۇ توغرا يولدۇر» دەپ تەۋسىيە قىلمىدىممۇ؟
وَلَقَدْ أَضَلَّ مِنكُمْ جِبِلًّا كَثِيرًا ۖ أَفَلَمْ تَكُونُوا۟ تَعْقِلُونَ ﴿٦٢﴾
شۈبھىسىزكى، شەيتان (ئازدۇرغۇچى) سىلەردىن نۇرغۇن كىشىلەرنى ئازدۇردى، (بۇنى) چۈشەنمەمسىلەر؟
هَـٰذِهِۦ جَهَنَّمُ ٱلَّتِى كُنتُمْ تُوعَدُونَ ﴿٦٣﴾
سىلەرگە ۋەدە قىلىنغان مەڭگۈلۈك ئوت مانا شۇ
ٱصْلَوْهَا ٱلْيَوْمَ بِمَا كُنتُمْ تَكْفُرُونَ ﴿٦٤﴾
سىلەر (دۇنيادىكى چېغىڭلاردا) ھەقىقەتنى رەت قىلغۇچى بولغانلىقىڭلار ئۈچۈن مەڭگۈلۈك ئوتگە كىرىڭلار
ٱلْيَوْمَ نَخْتِمُ عَلَىٰٓ أَفْوَٰهِهِمْ وَتُكَلِّمُنَآ أَيْدِيهِمْ وَتَشْهَدُ أَرْجُلُهُم بِمَا كَانُوا۟ يَكْسِبُونَ ﴿٦٥﴾
شۇ كۈندە ئۇلارنىڭ ئېغىزلىرىنى پېچەتلىۋېتىمىز، ئۇلارنىڭ قىلمىشلىرىنى قوللىرى بىزگە سۆزلەپ بېرىدۇ، پۇتلىرى گۇۋاھلىق بېرىدۇ (يەنى ھەر ئەزا ئۇنىڭدىن سادىر بولغان ئىشنى سۆزلەپ بېرىدۇ)
وَلَوْ نَشَآءُ لَطَمَسْنَا عَلَىٰٓ أَعْيُنِهِمْ فَٱسْتَبَقُوا۟ ٱلصِّرَٰطَ فَأَنَّىٰ يُبْصِرُونَ ﴿٦٦﴾
ئەگەر بىز خالىساق، ئۇلارنىڭ كۆزلىرىنى ئەلۋەتتە يوق قىلىۋېتەتتۇق، ئاندىن ئۇلار يول مېڭىشقا ئالدىرايتتى، لېكىن ئۇلار قانداقمۇ كۆرەلىسۇن؟
وَلَوْ نَشَآءُ لَمَسَخْنَـٰهُمْ عَلَىٰ مَكَانَتِهِمْ فَمَا ٱسْتَطَـٰعُوا۟ مُضِيًّا وَلَا يَرْجِعُونَ ﴿٦٧﴾
ئەگەر بىز خالىساق، ئۇلارنى ئۆيلىرىدىلا سۈرىتىنى مۇبەددەل (يەنى مايمۇن، چوشقا ياكى تاش) قىلىۋېتەتتۇق، شۇنىڭ بىلەن ئۇلار ئالدىغىمۇ ماڭالمايتتى، ئارقىغىمۇ قايتالمايتتى
وَمَن نُّعَمِّرْهُ نُنَكِّسْهُ فِى ٱلْخَلْقِ ۖ أَفَلَا يَعْقِلُونَ ﴿٦٨﴾
بىز ئۆمرىنى ئۇزۇن قىلغان ئادەمنىڭ تېنىنى ئاجىزلىتىۋېتىمىز (يەنى ياشلىق، قىرانلىق ۋاقتىدىكىدىن ئاجىزلايدۇ، قېرىيدۇ)، ئۇلار (بۇنى) چۈشەنمەمدۇ؟
وَمَا عَلَّمْنَـٰهُ ٱلشِّعْرَ وَمَا يَنۢبَغِى لَهُۥٓ ۚ إِنْ هُوَ إِلَّا ذِكْرٌ وَقُرْءَانٌ مُّبِينٌ ﴿٦٩﴾
ئۇنىڭغا (يەنى خۇدانىڭ ئەلچىسىگە) بىز شېئىر تەلىم بەرمىدۇق، ئۇنىڭغا شېئىر مۇناسىپمۇ ئەمەس. ئۇ پەقەت تىرىكلەرنى ئاگاھلاندۇرۇش، ھەقىقەتنى رەت قىلغۇچىلارغا ئازابنىڭ تېگىشلىك ئىكەنلىكىنى (بىلدۈرۈش ئۈچۈن نازىل بولغان) ۋەز - نەسىھەتتۇر ۋە روشەن قۇرئاندۇر
لِّيُنذِرَ مَن كَانَ حَيًّا وَيَحِقَّ ٱلْقَوْلُ عَلَى ٱلْكَـٰفِرِينَ ﴿٧٠﴾
ئۇنىڭغا (يەنى خۇدانىڭ ئەلچىسىگە) بىز شېئىر تەلىم بەرمىدۇق، ئۇنىڭغا شېئىر مۇناسىپمۇ ئەمەس. ئۇ پەقەت تىرىكلەرنى ئاگاھلاندۇرۇش، ھەقىقەتنى رەت قىلغۇچىلارغا ئازابنىڭ تېگىشلىك ئىكەنلىكىنى (بىلدۈرۈش ئۈچۈن نازىل بولغان) ۋەز - نەسىھەتتۇر ۋە روشەن قۇرئاندۇر
أَوَلَمْ يَرَوْا۟ أَنَّا خَلَقْنَا لَهُم مِّمَّا عَمِلَتْ أَيْدِينَآ أَنْعَـٰمًا فَهُمْ لَهَا مَـٰلِكُونَ ﴿٧١﴾
ئۇلار بىلمەمدۇكى، ئۇلار ئۈچۈن ھايۋانلارنى (قۇدرەت) قولىمىز بىلەن خەلق ئەتتۇق، ئۇلار ھايۋانلارنى باشقۇرۇپ تۇرغۇچىلاردۇر
وَذَلَّلْنَـٰهَا لَهُمْ فَمِنْهَا رَكُوبُهُمْ وَمِنْهَا يَأْكُلُونَ ﴿٧٢﴾
ھايۋانلارنى ئۇلارغا بويسۇندۇرۇپ بەردۇق، ئۇلار ئۇ ھايۋانلارنىڭ بەزىسىنى مىنىدۇ، بەزىسىنى يەيدۇ
وَلَهُمْ فِيهَا مَنَـٰفِعُ وَمَشَارِبُ ۖ أَفَلَا يَشْكُرُونَ ﴿٧٣﴾
ئۇلار شۇ ھايۋانلاردىن پايدىلىنىدۇ ۋە سۈتلىرىنى ئىچىدۇ، ئۇلار (پەرۋەردىگارىنىڭ بۇ نېمەتلىرىگە) شۈكۈر قىلمامدۇ؟
وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لَّعَلَّهُمْ يُنصَرُونَ ﴿٧٤﴾
ئۇلار ياردەمگە ئېرىشىشنى ئۈمىد قىلىپ، اﷲ نى قويۇپ باشقا مەبۇدلارنى ئىلاھ قىلىۋالدى
لَا يَسْتَطِيعُونَ نَصْرَهُمْ وَهُمْ لَهُمْ جُندٌ مُّحْضَرُونَ ﴿٧٥﴾
ئۇ مەبۇدلار ئۇلارغا ياردەم بېرەلمەيدۇ. بۇلار شۇ مەبۇدلار ئۈچۈن (يەنى ئۇلارنىڭ خىزمىتى ئۈچۈن) ھازىرلانغان قوشۇندۇر
فَلَا يَحْزُنكَ قَوْلُهُمْ ۘ إِنَّا نَعْلَمُ مَا يُسِرُّونَ وَمَا يُعْلِنُونَ ﴿٧٦﴾
(ئى، مۇھەممەد!) ئۇلار (يەنى مۇشرىكلار) نىڭ سۆزى (يەنى سېنى ئىنكار قىلغانلىقى) سېنى غەمكىن قىلمىسۇن. بىز ئىشەنچلىك ھالدا ئۇلارنىڭ دىللىرىدىكىنى ۋە ئاشكارا قىلغان (سۆز ۋە ھەرىكەتلىرى) نى بىلىپ تۇرىمىز
أَوَلَمْ يَرَ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ مِن نُّطْفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ ﴿٧٧﴾
ئىنسان بىلمەمدۇكى، بىز ئىشەنچلىك ھالدا ئۇنى ئابىمەنىيدىن ياراتتۇق. ئەمدى ئۇ (پەرۋەردىگارىغا) ئاشكارا خۇسۇمەتچى بولۇپ قالدى
وَضَرَبَ لَنَا مَثَلًا وَنَسِىَ خَلْقَهُۥ ۖ قَالَ مَن يُحْىِ ٱلْعِظَـٰمَ وَهِىَ رَمِيمٌ ﴿٧٨﴾
ئۇ بىزگە (ئىنساننىڭ قايتا تىرىلىشىنى يىراق ساناپ، چىرىگەن سۆڭەكلەرنى) مىسال قىلىپ كۆرسەتتى. ئۆزىنىڭ يارىتىلغانلىقىنى بولسا ئۇنتۇدى، ئۇ: «چىرىپ كەتكەن سۆڭەكلەرنى كىم تىرىلدۈرەلەيدۇ؟» دېدى
قُلْ يُحْيِيهَا ٱلَّذِىٓ أَنشَأَهَآ أَوَّلَ مَرَّةٍ ۖ وَهُوَ بِكُلِّ خَلْقٍ عَلِيمٌ ﴿٧٩﴾
ئېيتقىنكى، «ئۇنى ئەڭ دەسلەپتە ياراتقان زات تىرىلدۈرىدۇ، ئۇ ھەر بىر مەخلۇقنى بىلگۈچىدۇر»
ٱلَّذِى جَعَلَ لَكُم مِّنَ ٱلشَّجَرِ ٱلْأَخْضَرِ نَارًا فَإِذَآ أَنتُم مِّنْهُ تُوقِدُونَ ﴿٨٠﴾
اﷲ سىلەر ئۈچۈن يېشىل دەرەختىن ئوت پەيدا قىلدى. سىلەر ئۇنىڭ بىلەن ئوت ياقىسىلەر
أَوَلَيْسَ ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِقَـٰدِرٍ عَلَىٰٓ أَن يَخْلُقَ مِثْلَهُم ۚ بَلَىٰ وَهُوَ ٱلْخَلَّـٰقُ ٱلْعَلِيمُ ﴿٨١﴾
ئاسمانلارنى ۋە زېمىننى ياراتقان زات ئۇلارنىڭ ئوخشىشىنى يارىتىشقا قادىر ئەمەسمۇ؟ ئۇ بۇنىڭغا قادىر، اﷲ ماھىر ياراتقۇچىدۇر، ھەممىنى بىلگۈچىدۇر
إِنَّمَآ أَمْرُهُۥٓ إِذَآ أَرَادَ شَيْـًٔا أَن يَقُولَ لَهُۥ كُن فَيَكُونُ ﴿٨٢﴾
اﷲ بىرەر شەيئىنى (يارىتىشنى) ئىرادە قىلسا، ئۇنىڭغا «ۋۇجۇتقا كەل» دەيدۇ - دە، ئۇ ۋۇجۇتقا كېلىدۇ
فَسُبْحَـٰنَ ٱلَّذِى بِيَدِهِۦ مَلَكُوتُ كُلِّ شَىْءٍ وَإِلَيْهِ تُرْجَعُونَ ﴿٨٣﴾
اﷲ پاكتۇركى، ھەر بىر نەرسىنىڭ ئىگىلىك ھوقوقى ئۇنىڭ قولىدىدۇر، (ئۆلگەندىن كېيىن) ئۇنىڭ دەرگاھىغا قايتۇرۇلىسىلەر
