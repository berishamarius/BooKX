الحجر
Al-Hijr · ھىجر
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
الٓر ۚ تِلْكَ ءَايَـٰتُ ٱلْكِتَـٰبِ وَقُرْءَانٍ مُّبِينٍ ﴿١﴾
ئەلىف، لام، را. بۇ، مۇقەددەس كىتاب-ۋەھىينىڭ - روشەن قۇرئاننىڭ ئايەتلىرىدۇر
رُّبَمَا يَوَدُّ ٱلَّذِينَ كَفَرُوا۟ لَوْ كَانُوا۟ مُسْلِمِينَ ﴿٢﴾
ھەقىقەتنى رەت قىلغۇچىلار (قىيامەتنىڭ دەھشىتىنى كۆرگەندە، دۇنيادىكى چېغىمىزدا) مۇسۇلمان بولغان بولساقچۇ، دەپ ئارمان قىلىدۇ
ذَرْهُمْ يَأْكُلُوا۟ وَيَتَمَتَّعُوا۟ وَيُلْهِهِمُ ٱلْأَمَلُ ۖ فَسَوْفَ يَعْلَمُونَ ﴿٣﴾
(ئى مۇھەممەد!) ئۇلارنى (يەنى كۇففارلارنى مەيلىگە) قويۇۋەت، يەپ - ئىچىپ، ئويناپ - كۈلۈپ يۈرسۇن، ئۇلار بىھۇدە ئارزۇلار بىلەن بولىۋەرمىسۇن، ئۇلار ئۇزاققا قالماي (ئۆز قىلمىشلىرىنىڭ يامانلىقىنى) بىلىدۇ
وَمَآ أَهْلَكْنَا مِن قَرْيَةٍ إِلَّا وَلَهَا كِتَابٌ مَّعْلُومٌ ﴿٤﴾
بىز قانداق بىر شەھەرنى ھالاك قىلمايلى، پەقەت ئۇنىڭ (ھالاك بولىدىغان) مۇئەييەن ۋاقتى بولىدۇ
مَّا تَسْبِقُ مِنْ أُمَّةٍ أَجَلَهَا وَمَا يَسْتَـْٔخِرُونَ ﴿٥﴾
ھەرقانداق ئۈممەتنىڭ (ھالاك بولىدىغان ۋاقتى) ئىلگىرى سۈرۈلمەيدۇ، ھەم كېچىكتۈرۈلمەيدۇ
وَقَالُوا۟ يَـٰٓأَيُّهَا ٱلَّذِى نُزِّلَ عَلَيْهِ ٱلذِّكْرُ إِنَّكَ لَمَجْنُونٌ ﴿٦﴾
ئۇلار ئېيتتى: «ئى قۇرئان نازىل قىلىنغان ئادەم! سەن ئىشەنچلىك ھالدا مەجنۇنسەن
لَّوْ مَا تَأْتِينَا بِٱلْمَلَـٰٓئِكَةِ إِن كُنتَ مِنَ ٱلصَّـٰدِقِينَ ﴿٧﴾
ئەگەر سەن (مەن رەسۇلۇللاھ دېگەن سۆزۈڭدە) راستچىللاردىن بولساڭ، (خۇدانىڭ ئەلچىسىلىكىڭگە گۇۋاھلىق بېرىدىغان) خۇدانىڭ ئەلچى پەرىشتىلىرىنى ئالدىمىزغا ئېلىپ كەلسەڭچۇ؟»
مَا نُنَزِّلُ ٱلْمَلَـٰٓئِكَةَ إِلَّا بِٱلْحَقِّ وَمَا كَانُوٓا۟ إِذًا مُّنظَرِينَ ﴿٨﴾
خۇدانىڭ ئەلچى پەرىشتىلىرىنى پەقەت ھەق ئاساسىدا (يەنى ئازاب نازىل قىلىشىمىز مۇقەررەر بولغان چاغدىلا) نازىل قىلىمىز، بۇ چاغدا ئۇلارغا مۆھلەت بېرىلمەيدۇ (يەنى ئۇلارنىڭ ئازابى كېچىكتۈرۈلمەيدۇ)
إِنَّا نَحْنُ نَزَّلْنَا ٱلذِّكْرَ وَإِنَّا لَهُۥ لَحَـٰفِظُونَ ﴿٩﴾
قۇرئاننى ئىشەنچلىك ھالدا بىز نازىل قىلدۇق ۋە چوقۇم ئۇنى قوغدايمىز
وَلَقَدْ أَرْسَلْنَا مِن قَبْلِكَ فِى شِيَعِ ٱلْأَوَّلِينَ ﴿١٠﴾
شۈبھىسىزكى، سېنىڭدىن ئىلگىرىكى بۇرۇنقى ھەرقايسى تائىپىلەرگە نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتتۇق
وَمَا يَأْتِيهِم مِّن رَّسُولٍ إِلَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿١١﴾
ئۇلار ئۆزلىرىگە خۇدانىڭ ئەلچىسى كەلسىلا ئۇنى مەسخىرە قىلىشتى
كَذَٰلِكَ نَسْلُكُهُۥ فِى قُلُوبِ ٱلْمُجْرِمِينَ ﴿١٢﴾
شۇنداق مەسخىرە قىلىشنى بىز گۇناھكارلارنىڭ دىللىرىغا سالىمىز
لَا يُؤْمِنُونَ بِهِۦ ۖ وَقَدْ خَلَتْ سُنَّةُ ٱلْأَوَّلِينَ ﴿١٣﴾
ئۇلار قۇرئانغا ئىشەنمەيدۇ، بۇ بۇرۇندىن ئادەت بولۇپ قالغان (يەنى بۇرۇندىن تارتىپ خۇدانىڭ ئەلچىسىلىرنى ئىنكار قىلغان قەۋملەرنى اﷲ نىڭ ھالاك قىلىشى ئادەت بولۇپ كېلىۋاتقان) ئىش
وَلَوْ فَتَحْنَا عَلَيْهِم بَابًا مِّنَ ٱلسَّمَآءِ فَظَلُّوا۟ فِيهِ يَعْرُجُونَ ﴿١٤﴾
بىز ئۇلارغا ئاسماندىن بىر ئىشىك ئېچىپ بەرگىن، ئۇلار ئۇنىڭدىن ئاسمانغا ئۆرلەپ (ئالەمى ئەرۋاھنى، خۇدانىڭ ئەلچى پەرىشتىلىرىنى كۆرگەن تەقدىردىمۇ)
لَقَالُوٓا۟ إِنَّمَا سُكِّرَتْ أَبْصَـٰرُنَا بَلْ نَحْنُ قَوْمٌ مَّسْحُورُونَ ﴿١٥﴾
ئۇلار چوقۇم: «كۆزلىرىمىز باغلىنىپ قالدى، بەلكىم سېھىرلەنگەن قەۋم بولۇپ قالساق كېرەك» دېيىشىدۇ
وَلَقَدْ جَعَلْنَا فِى ٱلسَّمَآءِ بُرُوجًا وَزَيَّنَّـٰهَا لِلنَّـٰظِرِينَ ﴿١٦﴾
بىز ئاسماندا بۇرۇجلارنى ياراتتۇق ۋە كۆزەتكۈچىلەر ئۈچۈن ئاسماننى (يۇلتۇزلار بىلەن) بېزىدۇق
وَحَفِظْنَـٰهَا مِن كُلِّ شَيْطَـٰنٍ رَّجِيمٍ ﴿١٧﴾
ھەر بىر قوغلاندى شەيتان (ئازدۇرغۇچى)دىن ئاسماننى قوغدىدۇق
إِلَّا مَنِ ٱسْتَرَقَ ٱلسَّمْعَ فَأَتْبَعَهُۥ شِهَابٌ مُّبِينٌ ﴿١٨﴾
لېكىن (ئاسمان خەۋەرلىرىگە) ئوغرىلىقچە قۇلاق سالماقچى بولغان شەيتان (ئازدۇرغۇچى)نى روشەن ئاقار يۇلتۇز قوغلاپ يېتىدۇ (يەنى ھالاك قىلىدۇ)
وَٱلْأَرْضَ مَدَدْنَـٰهَا وَأَلْقَيْنَا فِيهَا رَوَٰسِىَ وَأَنۢبَتْنَا فِيهَا مِن كُلِّ شَىْءٍ مَّوْزُونٍ ﴿١٩﴾
يەر يۈزىنى يېيىپ كەڭ قىلدۇق، ئۇنىڭغا تاغلارنى ئورناتتۇق، يەر يۈزىدە ھەرخىل تەكشى ئۆلچەملىك ئۆسۈملۈكلەرنى ئۆستۈردۇق
وَجَعَلْنَا لَكُمْ فِيهَا مَعَـٰيِشَ وَمَن لَّسْتُمْ لَهُۥ بِرَٰزِقِينَ ﴿٢٠﴾
يەر يۈزىدە سىلەر ئۈچۈن ۋە سىلەر رىزقىنى بېرەلمەيدىغانلار (يەنى خادىملار ۋە ھايۋانلار) ئۈچۈن تۇرمۇش لازىمەتلىكلىرىنى ياراتتۇق (ئۇلارنىڭ يېمەك - ئىچمەكلىرىنى سىلەر ئەمەس، بىز يارىتىمىز)
وَإِن مِّن شَىْءٍ إِلَّا عِندَنَا خَزَآئِنُهُۥ وَمَا نُنَزِّلُهُۥٓ إِلَّا بِقَدَرٍ مَّعْلُومٍ ﴿٢١﴾
قانداق نەرسە بولمىسۇن، ئۇنىڭ خەزىنىسى (يەنى خەزىنىنىڭ ئاچقۇچى) بىزنىڭ دەرگاھىمىزدىدۇر، بىز ئۇنى پەقەت مەلۇم مىقداردىلا چۈشۈرىمىز
وَأَرْسَلْنَا ٱلرِّيَـٰحَ لَوَٰقِحَ فَأَنزَلْنَا مِنَ ٱلسَّمَآءِ مَآءً فَأَسْقَيْنَـٰكُمُوهُ وَمَآ أَنتُمْ لَهُۥ بِخَـٰزِنِينَ ﴿٢٢﴾
بىز يامغۇرنى ھەيدەيدىغان شاماللارنى ئەۋەتتۇق، بىز بۇلۇتتىن يامغۇر ياغدۇرۇپ سىلەرنى سۇغاردۇق، سىلەر ھەرگىز ئۇنى ساقلاشقا قادىر ئەمەسسىلەر
وَإِنَّا لَنَحْنُ نُحْىِۦ وَنُمِيتُ وَنَحْنُ ٱلْوَٰرِثُونَ ﴿٢٣﴾
بىز (ھەممىنى) تىرىلدۈرىمىز، (ھەممىنى) ئۆلتۈرىمىز ۋە (ھەممىگە) ۋارىسلىق قىلىمىز
وَلَقَدْ عَلِمْنَا ٱلْمُسْتَقْدِمِينَ مِنكُمْ وَلَقَدْ عَلِمْنَا ٱلْمُسْتَـْٔخِرِينَ ﴿٢٤﴾
بىز سىلەردىن ئىلگىرى ئۆتكەنلەرنى ئىشەنچلىك ھالدا بىلىمىز ۋە كېيىن كېلىدىغانلارنىمۇ ئىشەنچلىك ھالدا بىلىمىز
وَإِنَّ رَبَّكَ هُوَ يَحْشُرُهُمْ ۚ إِنَّهُۥ حَكِيمٌ عَلِيمٌ ﴿٢٥﴾
پەرۋەردىگارىڭ ئىشەنچلىك ھالدا (ھېساب ئېلىش ۋە جازالاش ئۈچۈن) ئۇلارنى يىغىدۇ. اﷲ ئىشەنچلىك ھالدا ھېكمەت بىلەن ئىش قىلغۇچىدۇر، ھەممىنى بىلگۈچىدۇر
وَلَقَدْ خَلَقْنَا ٱلْإِنسَـٰنَ مِن صَلْصَـٰلٍ مِّنْ حَمَإٍ مَّسْنُونٍ ﴿٢٦﴾
شەك - شۈبھىسىزكى، بىز ئىنسان (يەنى ئادەم ئەلەيھىسسالامنى) نى قارا لايدىن ياسىلىپ شەكىلگە كىرگۈزۈلگەن (چەكسە جىرىڭلايدىغان) قۇرۇق لايدىن ياراتتۇق
وَٱلْجَآنَّ خَلَقْنَـٰهُ مِن قَبْلُ مِن نَّارِ ٱلسَّمُومِ ﴿٢٧﴾
جىنلار (يەنى شەيتان (ئازدۇرغۇچى)لار ۋە ئۇلارنىڭ باشلىقى ئىبلىس) نى ئىلگىرى تۈتۈنسىز ئوت يالقۇنىدىن ياراتقان ئىدۇق
وَإِذْ قَالَ رَبُّكَ لِلْمَلَـٰٓئِكَةِ إِنِّى خَـٰلِقٌۢ بَشَرًا مِّن صَلْصَـٰلٍ مِّنْ حَمَإٍ مَّسْنُونٍ ﴿٢٨﴾
پەرۋەردىگارىڭ ئۆز ۋاقتىدا خۇدانىڭ ئەلچى پەرىشتىلىرىگە ئېيتتى: «مەن ئىشەنچلىك ھالدا قارا لايدىن ياسىلىپ شەكىلگە كىرگۈزۈلگەن (چەكسە جىرىڭلايدىغان) قۇرۇق لايدىن ئىنسان (يەنى ئادەم ئەلەيھىسسالام) نى يارىتىمەن
فَإِذَا سَوَّيْتُهُۥ وَنَفَخْتُ فِيهِ مِن رُّوحِى فَقَعُوا۟ لَهُۥ سَـٰجِدِينَ ﴿٢٩﴾
مەن ئۇنى تولۇق ياراتقان (يەنى ئۇنى يارىتىپ سۈرەتكە كىرگۈزۈپ، ئەزالىرى تولۇق ۋە مۇكەممەل ئىنسان ھالىتىگە كەلتۈرگەن) ۋە ئۇنىڭغا جان كىرگۈزگەن چېغىمدا، سىلەر ئۇنىڭغا سەجدە قىلىڭلار)
فَسَجَدَ ٱلْمَلَـٰٓئِكَةُ كُلُّهُمْ أَجْمَعُونَ ﴿٣٠﴾
خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ بىرىمۇ ئايرىلىپ قالماي ھەممىسى (ئادەم ئەلەيھىسسالامغا) سەجدە قىلدى
إِلَّآ إِبْلِيسَ أَبَىٰٓ أَن يَكُونَ مَعَ ٱلسَّـٰجِدِينَ ﴿٣١﴾
پەقەت ئىبلىسلا سەجدە قىلىشتىن باش تارتتى
قَالَ يَـٰٓإِبْلِيسُ مَا لَكَ أَلَّا تَكُونَ مَعَ ٱلسَّـٰجِدِينَ ﴿٣٢﴾
اﷲ: «ئى ئىبلىس! سەن نېمىشقا سەجدە قىلمايسەن؟» دېدى
قَالَ لَمْ أَكُن لِّأَسْجُدَ لِبَشَرٍ خَلَقْتَهُۥ مِن صَلْصَـٰلٍ مِّنْ حَمَإٍ مَّسْنُونٍ ﴿٣٣﴾
ئىبلىس: «قارا لايدىن ياساپ شەكىلگە كىرگۈزۈپ (چەكسە جىرىڭلايدىغان) قۇرۇق لايدىن يارىتىلغان ئىنسان مېنىڭ سەجدە قىلىشىمغا لايىق ئەمەس» دېدى
قَالَ فَٱخْرُجْ مِنْهَا فَإِنَّكَ رَجِيمٌ ﴿٣٤﴾
اﷲ ئېيتتى: «سەن مەڭگۈلۈك باغلارتىن يوقال، ئىشەنچلىك ھالدا قوغلاندى بولدۇڭ
وَإِنَّ عَلَيْكَ ٱللَّعْنَةَ إِلَىٰ يَوْمِ ٱلدِّينِ ﴿٣٥﴾
ساڭا قىيامەت كۈنىگىچە ئىشەنچلىك ھالدا لەنەت بولسۇن»
قَالَ رَبِّ فَأَنظِرْنِىٓ إِلَىٰ يَوْمِ يُبْعَثُونَ ﴿٣٦﴾
ئىبلىس: «پەرۋەردىگارىم! ماڭا خالايىقلار تىرىلدۈرۈلىدىغان كۈن (يەنى قىيامەت كۈنى) گىچە مۆھلەت بەرگىن (مېنى ھايات قالدۇرغىن)» دېدى
قَالَ فَإِنَّكَ مِنَ ٱلْمُنظَرِينَ ﴿٣٧﴾
اﷲ: «ساڭا ئىشەنچلىك ھالدا مەلۇم ۋاقىتقىچە مۆھلەت بېرىلىدۇ» دېدى
إِلَىٰ يَوْمِ ٱلْوَقْتِ ٱلْمَعْلُومِ ﴿٣٨﴾
اﷲ: «ساڭا ئىشەنچلىك ھالدا مەلۇم ۋاقىتقىچە مۆھلەت بېرىلىدۇ» دېدى
قَالَ رَبِّ بِمَآ أَغْوَيْتَنِى لَأُزَيِّنَنَّ لَهُمْ فِى ٱلْأَرْضِ وَلَأُغْوِيَنَّهُمْ أَجْمَعِينَ ﴿٣٩﴾
ئىبلىس ئېيتتى: «پەرۋەردىگارىم! سەن مېنى گۇمراھ قىغانلىقىڭ ئۈچۈن، يەر يۈزىدە ئادەم بالىلىرىغا (گۇناھلارنى) ئەلۋەتتە چىرايلىق كۆرسىتىمەن، ئۇلارنىڭ ھەممىسىنى ئەلۋەتتە ئازدۇرىمەن
إِلَّا عِبَادَكَ مِنْهُمُ ٱلْمُخْلَصِينَ ﴿٤٠﴾
ئۇلاردىن پەقەت تاللانغان بەندىلىرىڭلا بۇنىڭدىن مۇستەسنا (يەنى ئۇلارنى ئازدۇرالمايمەن)»
قَالَ هَـٰذَا صِرَٰطٌ عَلَىَّ مُسْتَقِيمٌ ﴿٤١﴾
اﷲ ئېيتتى: «بۇ مەن ساقلاشقا تېگىشلىك توغرا يولدۇر
إِنَّ عِبَادِى لَيْسَ لَكَ عَلَيْهِمْ سُلْطَـٰنٌ إِلَّا مَنِ ٱتَّبَعَكَ مِنَ ٱلْغَاوِينَ ﴿٤٢﴾
مېنىڭ بەندىلىرىمگە (يەنى ئۇلارنى ئازدۇرۇشقا) سېنىڭ كۈچۈڭ يەتمەيدۇ، پەقەت گۇمراھلاردىن ساڭا ئەگەشكەنلەرگىلا (سېنىڭ كۈچۈڭ يېتىدۇ)»
وَإِنَّ جَهَنَّمَ لَمَوْعِدُهُمْ أَجْمَعِينَ ﴿٤٣﴾
شۈبھىسىزكى، دوزاخ ئۇلار (يەنى ئىبلىس ۋە ئۇنىڭ تەۋەلىرى) نىڭ ھەممىسىگە ۋەدە قىلىنغان جايدۇر
لَهَا سَبْعَةُ أَبْوَٰبٍ لِّكُلِّ بَابٍ مِّنْهُمْ جُزْءٌ مَّقْسُومٌ ﴿٤٤﴾
مەڭگۈلۈك ئوتنىڭ يەتتە دەرۋازىسى بولۇپ، ئۇلارنىڭ ھەربىرىدىن كىرىدىغان مۇئەييەن بىر بۆلۈك (ئادەم) بار
إِنَّ ٱلْمُتَّقِينَ فِى جَنَّـٰتٍ وَعُيُونٍ ﴿٤٥﴾
تەقۋادارلار ئىشەنچلىك ھالدا (ئاخىرەتتە ياپيېشىل) باغ - بوستانلاردىن، سۇ، ھەسەل ۋە مەي ئېقىپ تۇرىدىغان) بۇلاقلاردىن بەھرىمەن بولىدۇ
ٱدْخُلُوهَا بِسَلَـٰمٍ ءَامِنِينَ ﴿٤٦﴾
(ئۇلارغا) مەڭگۈلۈك باغلاركە ئامان - ئېسەن، تىنچ كىرىڭلار! دېيىلىدۇ
وَنَزَعْنَا مَا فِى صُدُورِهِم مِّنْ غِلٍّ إِخْوَٰنًا عَلَىٰ سُرُرٍ مُّتَقَـٰبِلِينَ ﴿٤٧﴾
ئۇلارنىڭ دىللىرىدىكى دۈشمەنلىكنى چىقىرىپ تاشلايمىز، ئۇلار ئۆزئارا قېرىنداش بولغان ھالدا تەختلەر ئۈستىدە بىر - بىرىگە قارىشىپ ئولتۇرىدۇ
لَا يَمَسُّهُمْ فِيهَا نَصَبٌ وَمَا هُم مِّنْهَا بِمُخْرَجِينَ ﴿٤٨﴾
مەڭگۈلۈك باغلارتە ئۇلارغا چارچاش بولمايدۇ، ئۇلار مەڭگۈلۈك باغلارتىن چىقىرىۋېتىلمەيدۇ
۞ نَبِّئْ عِبَادِىٓ أَنِّىٓ أَنَا ٱلْغَفُورُ ٱلرَّحِيمُ ﴿٤٩﴾
(ئى مۇھەممەد!) مېنىڭ (مۆمىن) بەندىلىرىمگە خەۋەر قىلغىنكى، مەن تولىمۇ مەغپىرەت قىلغۇچىمەن، ناھايىتى رەھىم قىلغۇچىمەن
وَأَنَّ عَذَابِى هُوَ ٱلْعَذَابُ ٱلْأَلِيمُ ﴿٥٠﴾
مېنىڭ ئازابىم قاتتىقتۇر
وَنَبِّئْهُمْ عَن ضَيْفِ إِبْرَٰهِيمَ ﴿٥١﴾
ئۇلارغا ئىبراھىمنىڭ مېھمانلىرى ھەققىدىكى (ھېكايىنى) ئېيتىپ بەرگىن
إِذْ دَخَلُوا۟ عَلَيْهِ فَقَالُوا۟ سَلَـٰمًا قَالَ إِنَّا مِنكُمْ وَجِلُونَ ﴿٥٢﴾
ئۆز ۋاقتىدا خۇدانىڭ ئەلچى پەرىشتىلىرى ئىبراھىمنىڭ يېنىغا كىرىپ سالام بەردى، ئىبراھىم: «بىز ئىشەنچلىك ھالدا سىلەردىن قورقۇۋاتىمىز» دېدى
قَالُوا۟ لَا تَوْجَلْ إِنَّا نُبَشِّرُكَ بِغُلَـٰمٍ عَلِيمٍ ﴿٥٣﴾
ئۇلار: «قورقمىغىن، (بىز پەرۋەردىگارىڭنىڭ ئەلچىلىرىمىز)، ئىشەنچلىك ھالدا ساڭا (ئۆسۈپ چوڭ بولغىنىدا) ناھايىتى بىلىملىك (بولىدىغان) بىر ئوغۇل (يەنى ئىسھاق) بىلەن خۇش خەۋەر بېرىمىز» دېدى
قَالَ أَبَشَّرْتُمُونِى عَلَىٰٓ أَن مَّسَّنِىَ ٱلْكِبَرُ فَبِمَ تُبَشِّرُونَ ﴿٥٤﴾
ئىبراھىم: «مەن قېرىغان تۇرسام، ماڭا (بالا بىلەن) خۇش خەۋەر بەردىڭلارمۇ؟ ماڭا نېمە بىلەن خۇش خەۋەر بېرىۋاتىسىلەر؟» دېدى
قَالُوا۟ بَشَّرْنَـٰكَ بِٱلْحَقِّ فَلَا تَكُن مِّنَ ٱلْقَـٰنِطِينَ ﴿٥٥﴾
ئۇلار: «بىز ساڭا راست خۇش خەۋەر بەردۇق، نائۈمىد بولمىغىن» دېدى
قَالَ وَمَن يَقْنَطُ مِن رَّحْمَةِ رَبِّهِۦٓ إِلَّا ٱلضَّآلُّونَ ﴿٥٦﴾
ئىبراھىم: «پەرۋەردىگارىڭنىڭ رەھمىتىدىن گۇمراھلاردىن باشقا كىممۇ نائۈمىد بولار؟» دېدى
قَالَ فَمَا خَطْبُكُمْ أَيُّهَا ٱلْمُرْسَلُونَ ﴿٥٧﴾
ئىبراھىم: «ئى ئەلچىلەر! نېمە ئىش بىلەن كەلدىڭلار؟» دېدى
قَالُوٓا۟ إِنَّآ أُرْسِلْنَآ إِلَىٰ قَوْمٍ مُّجْرِمِينَ ﴿٥٨﴾
ئۇلار ئېيتتى: «بىز ئىشەنچلىك ھالدا گۇناھكار بىز قەۋم (نى جازالاش) ئۈچۈن ئەۋەتىلدۇق
إِلَّآ ءَالَ لُوطٍ إِنَّا لَمُنَجُّوهُمْ أَجْمَعِينَ ﴿٥٩﴾
پەقەت لۇتنىڭ تەۋەلىرىلا بۇنىڭدىن مۇستەسنا، بىز ئۇلار (يەنى لۇتنىڭ تەۋەلىرى ۋە ئۇنىڭ ئائىلىسىدىكىلەر) نىڭ ھەممىسىنى (ئازابتىن) چوقۇم قۇتۇلدۇرىمىز
إِلَّا ٱمْرَأَتَهُۥ قَدَّرْنَآ ۙ إِنَّهَا لَمِنَ ٱلْغَـٰبِرِينَ ﴿٦٠﴾
پەقەت لۇتنىڭ خوتۇنىلا بۇنىڭدىن مۇستەسنا، بىز ئۇنىڭ چوقۇم باشقىلار بىلەن بىللە ئازابلىنىدىغانلىقىنى ئالدىن بېكىتتۇق»
فَلَمَّا جَآءَ ءَالَ لُوطٍ ٱلْمُرْسَلُونَ ﴿٦١﴾
ئەلچىلەر لۇتنىڭ ئائىلىسىگە كەلگەن چاغدا
قَالَ إِنَّكُمْ قَوْمٌ مُّنكَرُونَ ﴿٦٢﴾
لۇت: «سىلەر ئىشەنچلىك ھالدا ناتونۇش ئادەملەرغۇ؟» دېدى
قَالُوا۟ بَلْ جِئْنَـٰكَ بِمَا كَانُوا۟ فِيهِ يَمْتَرُونَ ﴿٦٣﴾
ئۇلار ئېيتتى: «ئۇنداق ئەمەس (بىز اﷲ نىڭ ئەلچىلىرىمىز)، بىز سېنىڭ ئالدىڭغا ئۇلار (يەنى سېنىڭ قەۋمىڭ) شەكلىنىۋاتقان ئازابنى ئېلىپ كەلدۇق
وَأَتَيْنَـٰكَ بِٱلْحَقِّ وَإِنَّا لَصَـٰدِقُونَ ﴿٦٤﴾
بىز ساڭا ھەقىقەتنى ئېلىپ كەلدۇق، بىز ئەلۋەتتە (سۆزىمىزدە) راستچىلمىز
فَأَسْرِ بِأَهْلِكَ بِقِطْعٍ مِّنَ ٱلَّيْلِ وَٱتَّبِعْ أَدْبَـٰرَهُمْ وَلَا يَلْتَفِتْ مِنكُمْ أَحَدٌ وَٱمْضُوا۟ حَيْثُ تُؤْمَرُونَ ﴿٦٥﴾
كېچىنىڭ ئاخىرىدا ئائىلەڭدىكىلەرنى ئېلىپ يولغا چىققىن، ئۆزۈڭ ئۇلارنىڭ ئارقىسىدا ماڭغىن، سىلەردىن ھېچبىر ئادەم ئارقىسىغا قارىمىسۇن، سىلەر بۇيرۇلغان يەرگە يەتكۈچە ئالغا قاراپ مېڭىۋېرىڭلار»،
وَقَضَيْنَآ إِلَيْهِ ذَٰلِكَ ٱلْأَمْرَ أَنَّ دَابِرَ هَـٰٓؤُلَآءِ مَقْطُوعٌ مُّصْبِحِينَ ﴿٦٦﴾
سەھەردە بۇلار (يەنى لۇتنىڭ قەۋم) نىڭ يىلتىزى قۇرۇتۇلىدۇ (يەنى تەلتۆكۈس ھالاك قىلىنىدۇ) دېگەن ھۆكۈمنى لۇتقا ۋەھىي قىلدۇق
وَجَآءَ أَهْلُ ٱلْمَدِينَةِ يَسْتَبْشِرُونَ ﴿٦٧﴾
(سەدۇم) شەھرىنىڭ ئاھالىسى (يەنى لۇتنىڭ قەۋمى) خۇشال بولۇشۇپ يېتىپ كېلىشتى
قَالَ إِنَّ هَـٰٓؤُلَآءِ ضَيْفِى فَلَا تَفْضَحُونِ ﴿٦٨﴾
لۇت ئېيتتى: «بۇلار مېنىڭ مېھمىنىم، (ئۇلارغا چېقىلىپ قويۇپ ئۇلارنىڭ ئالدىدا) مېنى رەسۋا قىلماڭلار
وَٱتَّقُوا۟ ٱللَّهَ وَلَا تُخْزُونِ ﴿٦٩﴾
اﷲ تىن قورقۇڭلار (مېھمىنىمنى خارلاش ئارقىلىق) مېنى خارلىماڭلار»
قَالُوٓا۟ أَوَلَمْ نَنْهَكَ عَنِ ٱلْعَـٰلَمِينَ ﴿٧٠﴾
ئۇلار: «بىز سېنى كەلگەن مۇساپىرلارنى ھىمايە قىلىشتىن توسمىغانمىدۇق؟» دېدى
قَالَ هَـٰٓؤُلَآءِ بَنَاتِىٓ إِن كُنتُمْ فَـٰعِلِينَ ﴿٧١﴾
لۇت: «بۇلار مېنىڭ قىزلىرىم (يەنى قەۋمىمنىڭ قىزلىرى) دۇر، ئەگەر (قازائى شەھۋەت) قىلماقچى بولساڭلار (ئۇلارنى نىكاھلاپ ئېلىڭلار)» دېدى
لَعَمْرُكَ إِنَّهُمْ لَفِى سَكْرَتِهِمْ يَعْمَهُونَ ﴿٧٢﴾
(ئى مۇھەممەد!) سېنىڭ ھاياتىڭ بىلەن قەسەمكى، ئۇلار ئەلۋەتتە گۇمراھلىقلىرىدا تېڭىرقاپ يۈرۈشىدۇ
فَأَخَذَتْهُمُ ٱلصَّيْحَةُ مُشْرِقِينَ ﴿٧٣﴾
كۈن چىقار ۋاقتىدا ئۇلارغا قاتتىق چۇقان يۈزلەندى
فَجَعَلْنَا عَـٰلِيَهَا سَافِلَهَا وَأَمْطَرْنَا عَلَيْهِمْ حِجَارَةً مِّن سِجِّيلٍ ﴿٧٤﴾
بىز ئۇ شەھەرنى ئاستىن - ئۈستۈن قىلىۋەتتۇق (يەنى كۆمتۈرۈۋەتتۇق) ۋە ئۇلارنىڭ ئۈستىىگە (يامغۇردەك) ساپال تاش ياغدۇردۇق
إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّلْمُتَوَسِّمِينَ ﴿٧٥﴾
ئۇنىڭدا (يەنى ئۇلارغا نازىل بولغان ئازابتا) كۆزەتكۈچىلەر ئۈچۈن ئىشەنچلىك ھالدا نۇرغۇن ئىبرەتلەر بار
وَإِنَّهَا لَبِسَبِيلٍ مُّقِيمٍ ﴿٧٦﴾
ئۇ شەھەر (قۇرەيشلەر شامغا بارىدىغان) يول ئۈستىدە ئىشەنچلىك ھالدا ھېلىمۇ مەۋجۇت تۇرماقتا
إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّلْمُؤْمِنِينَ ﴿٧٧﴾
بۇنىڭدا مۆمىنلەر ئۈچۈن ئىشەنچلىك ھالدا ئىبرەت بار
وَإِن كَانَ أَصْحَـٰبُ ٱلْأَيْكَةِ لَظَـٰلِمِينَ ﴿٧٨﴾
ئەيكەلىكلەر، شۈبھىسىزكى، زالىم ئىدى (يەنى كۇفرىدا ھەددىدىن ئاشقانلار ئىدى)
فَٱنتَقَمْنَا مِنْهُمْ وَإِنَّهُمَا لَبِإِمَامٍ مُّبِينٍ ﴿٧٩﴾
(شۇئەيبنىڭ خۇدانىڭ ئەلچىسىلىكىنى ئىنكار قىلغانلىقلىرى ئۈچۈن) ئۇلارنى ھالاك قىلدۇق، ئۇلار (يەنى بۇ ئىككى شەھەر) نىڭ ھەر ئىككىلىسى، شۈبھىسىزكى، ئاشكارا يول ئۈستىدىدۇر (ئى مەككە ئاھالىسى! ئۇلاردىن ئىبرەت ئالمامسىلەر؟)
وَلَقَدْ كَذَّبَ أَصْحَـٰبُ ٱلْحِجْرِ ٱلْمُرْسَلِينَ ﴿٨٠﴾
ھىجرىلىكلەر ئىشەنچلىك ھالدا خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلدى (يەنى ئۇلارنىڭ سالىھ ئەلەيھىسسالامنى ئىنكار قىلغانلىقى باشقا خۇدانىڭ ئەلچى-رەسۇللىرىنىمۇ ئىنكار قىلغانلىقىدۇر)
وَءَاتَيْنَـٰهُمْ ءَايَـٰتِنَا فَكَانُوا۟ عَنْهَا مُعْرِضِينَ ﴿٨١﴾
ئۇلارغا مۆجىزىلىرىمىزنى كەلتۈردۇق (يەنى تاشتىن چىشى تۆگە چىقىرىشتەك قۇدرىتىمىزگە دالالەت قىلىدىغان مۆجىزىلەرنى ئۇلارغا كۆرسەتتۇق)، ئۇلار بۇ (مۆجىزە) لەردىن يۈز ئۆرۈدى
وَكَانُوا۟ يَنْحِتُونَ مِنَ ٱلْجِبَالِ بُيُوتًا ءَامِنِينَ ﴿٨٢﴾
ئۇلار (ئازابتىن) ئەمىن بولغان ھالدا تاغلارنى تېشىپ (بۇ بىزنى اﷲ نىڭ ئازابىدىن ساقلايدۇ دەپ) ئۆي سالاتتى
فَأَخَذَتْهُمُ ٱلصَّيْحَةُ مُصْبِحِينَ ﴿٨٣﴾
لېكىن قاتتىق چۇقان سەھەردىلا ئۇلارنى ھالاك قىلدى
فَمَآ أَغْنَىٰ عَنْهُم مَّا كَانُوا۟ يَكْسِبُونَ ﴿٨٤﴾
ئۇلارنىڭ قىلغان ئەمگەكلىرى ئۆزلىرىگە ئەسقاتمىدى (يەنى ئۇلارنىڭ سالغان بىنالىرى ئۇلارنى اﷲ نىڭ ئازابىدىن ساقلاپ قالالمىدى)
وَمَا خَلَقْنَا ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَمَا بَيْنَهُمَآ إِلَّا بِٱلْحَقِّ ۗ وَإِنَّ ٱلسَّاعَةَ لَـَٔاتِيَةٌ ۖ فَٱصْفَحِ ٱلصَّفْحَ ٱلْجَمِيلَ ﴿٨٥﴾
بىز ئاسمانلارنى، زېمىننى ۋە ئۇلاردىكى نەرسىلەرنى پەقەت ھەق ئاساسىدىلا ياراتتۇق، قىيامەتنىڭ بولۇشى چوقۇمدۇر، (ئى مۇھەممەد! نادانلار بىلەن تەڭ بولمىغىن، ئۇلارغا) چىرايلىق مۇئامىلىدە بولغىن
إِنَّ رَبَّكَ هُوَ ٱلْخَلَّـٰقُ ٱلْعَلِيمُ ﴿٨٦﴾
پەرۋەردىگارىڭ ئىشەنچلىك ھالدا ھەممىنى ياراتقۇچىدۇر، (بەندىلەرنىڭ ئەھۋالىنى) بىلىپ تۇرغۇچىدۇر
وَلَقَدْ ءَاتَيْنَـٰكَ سَبْعًا مِّنَ ٱلْمَثَانِى وَٱلْقُرْءَانَ ٱلْعَظِيمَ ﴿٨٧﴾
(نامازدا) تەكرارلىنىپ تۇرىدىغان يەتتە ئايەتنى (يەنى سۈرە فاتىھەنى) ۋە ئۇلۇغ قۇرئاننى ئىشەنچلىك ھالدا ساڭا ئاتا قىلدۇق
لَا تَمُدَّنَّ عَيْنَيْكَ إِلَىٰ مَا مَتَّعْنَا بِهِۦٓ أَزْوَٰجًا مِّنْهُمْ وَلَا تَحْزَنْ عَلَيْهِمْ وَٱخْفِضْ جَنَاحَكَ لِلْمُؤْمِنِينَ ﴿٨٨﴾
بەزى ھەقىقەتنى رەت قىلغۇچىلارنىڭ بىز بەھرىمەن قىلغان نەرسىلىرىگە كۆز سالمىغىن، ئۇلار (نىڭ ئىمان ئېيتمىغانلىقىدىن) قايغۇرمىغىن، مۆمىنلەرگە كەمتەر بولغىن
وَقُلْ إِنِّىٓ أَنَا ٱلنَّذِيرُ ٱلْمُبِينُ ﴿٨٩﴾
«مەن ئىشەنچلىك ھالدا (سىلەرنى اﷲ نىڭ ئازابىنىڭ نازىل بولۇشىدىن) ئاشكارا ئاگاھلاندۇرغۇچىمەن» دېگىن
كَمَآ أَنزَلْنَا عَلَى ٱلْمُقْتَسِمِينَ ﴿٩٠﴾
پارچىلىۋەتكۈچىلەرگە (يەنى يەھۇدىيلار ۋە ناسارالارغا تەۋرات بىلەن ئىنجىلنى) نازىل قىلغاندەك (ساڭا سۈرە فاتىھە بىلەن ئۇلۇغ قۇرئاننى نازىل قىلدۇق)
ٱلَّذِينَ جَعَلُوا۟ ٱلْقُرْءَانَ عِضِينَ ﴿٩١﴾
ئۇلار (يەنى يەھۇدىيلار ۋە ناسارالار) قۇرئاننى پارچە - پەرچە قىلىۋەتتى (يەنى قۇرئاننىڭ بەزىسىگە ئىشىنىپ، بەزىسىنى ئىنكار قىلدى)
فَوَرَبِّكَ لَنَسْـَٔلَنَّهُمْ أَجْمَعِينَ ﴿٩٢﴾
پەرۋەردىگارىڭنىڭ نامى بىلەن قەسەمكى، ئۇلار (يەنى خالايىق) نىڭ ھەممىسىنىڭ (بۇ دۇنيادا) قىلغان ئىشلىرىنى چوقۇم سورايمىز
عَمَّا كَانُوا۟ يَعْمَلُونَ ﴿٩٣﴾
پەرۋەردىگارىڭنىڭ نامى بىلەن قەسەمكى، ئۇلار (يەنى خالايىق) نىڭ ھەممىسىنىڭ (بۇ دۇنيادا) قىلغان ئىشلىرىنى چوقۇم سورايمىز
فَٱصْدَعْ بِمَا تُؤْمَرُ وَأَعْرِضْ عَنِ ٱلْمُشْرِكِينَ ﴿٩٤﴾
ساڭا بۇيرۇلغاننى (يەنى پەرۋەردىگارىڭنىڭ ئەمرىنى) ئاشكارا ئوتتۇرىغا قويغىن، مۇشرىكلار (نىڭ مەسخىرەلىرىگە) پەرۋا قىلمىغىن
إِنَّا كَفَيْنَـٰكَ ٱلْمُسْتَهْزِءِينَ ﴿٩٥﴾
سېنى مەسخىرە قىلغۇچىلارغا بىز چوقۇم تېتىيمىز
ٱلَّذِينَ يَجْعَلُونَ مَعَ ٱللَّهِ إِلَـٰهًا ءَاخَرَ ۚ فَسَوْفَ يَعْلَمُونَ ﴿٩٦﴾
ئۇلار باشقا مەبۇدلارنى اﷲ قا شېرىك قىلىدۇ، ئۇلار ئۇزاققا قالماي (ئىشنىڭ ئاقىۋىتىنى) بىلىدۇ
وَلَقَدْ نَعْلَمُ أَنَّكَ يَضِيقُ صَدْرُكَ بِمَا يَقُولُونَ ﴿٩٧﴾
بىز ئۇلارنىڭ سۆزلىرىدىن يۈرىكىڭنىڭ سېقىلىۋاتقانلىقىنى ئوبدان بىلىمىز
فَسَبِّحْ بِحَمْدِ رَبِّكَ وَكُن مِّنَ ٱلسَّـٰجِدِينَ ﴿٩٨﴾
پەرۋەردىگارىڭغا ھەمدۇ ۋە تەسبىھ ئېيتقىن، ھەم سەجدە قىلغۇچىلاردىن بولغىن
وَٱعْبُدْ رَبَّكَ حَتَّىٰ يَأْتِيَكَ ٱلْيَقِينُ ﴿٩٩﴾
سەن ئۆزۈڭگە ئۆلۈم كەلگەنگە (يەنى ئەجىلىڭ يەتكەنگە) قەدەر پەرۋەردىگارىڭغا ئىبادەت قىلغىن
