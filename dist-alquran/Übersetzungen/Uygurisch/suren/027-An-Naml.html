طسٓ ۚ تِلْكَ ءَايَـٰتُ ٱلْقُرْءَانِ وَكِتَابٍ مُّبِينٍ ﴿١﴾
طا، سىن. بۇ، قۇرئاننىڭ ۋە روشەن مۇقەددەس كىتاب-ۋەھىينىڭ ئايەتلىرىدۇر
هُدًى وَبُشْرَىٰ لِلْمُؤْمِنِينَ ﴿٢﴾
مۆمىنلەر ئۈچۈن ھىدايەتتۇر ۋە خۇش خەۋەردۇر
ٱلَّذِينَ يُقِيمُونَ ٱلصَّلَوٰةَ وَيُؤْتُونَ ٱلزَّكَوٰةَ وَهُم بِٱلْـَٔاخِرَةِ هُمْ يُوقِنُونَ ﴿٣﴾
(شۇنداق مۆمىنلەركى) ئۇلار نامازنى ئادا قىلىدۇ، زاكات (پاكلاش ئۇلۇشى) بېرىدۇ ۋە ئاخىرەتكە جەزمەن ئىشىنىدۇ
إِنَّ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْـَٔاخِرَةِ زَيَّنَّا لَهُمْ أَعْمَـٰلَهُمْ فَهُمْ يَعْمَهُونَ ﴿٤﴾
ئاخىرەتكە ئىشەنمەيدىغانلارغا ئۇلارنىڭ ئەمەللىرىنى ئىشەنچلىك ھالدا چىرايلىق كۆرسەتتۇق، ئۇلار (گۇمراھلىقلىرىدا) تېڭىرقاپ يۈرۈشىدۇ
أُو۟لَـٰٓئِكَ ٱلَّذِينَ لَهُمْ سُوٓءُ ٱلْعَذَابِ وَهُمْ فِى ٱلْـَٔاخِرَةِ هُمُ ٱلْأَخْسَرُونَ ﴿٥﴾
ئۇلار (دۇنيادا) قاتتىق ئازابقا دۇچار بولىدۇ. ئاخىرەتتە ئۇلار ئەڭ زىيان تارتقۇچىلاردۇر
وَإِنَّكَ لَتُلَقَّى ٱلْقُرْءَانَ مِن لَّدُنْ حَكِيمٍ عَلِيمٍ ﴿٦﴾
ساڭا قۇرئان ئەلۋەتتە ھېكمەت بىلەن ئىش قىلغۇچى، ھەممىنى بىلگۈچى اﷲ تەرىپىدىن بېرىلىدۇ
إِذْ قَالَ مُوسَىٰ لِأَهْلِهِۦٓ إِنِّىٓ ءَانَسْتُ نَارًا سَـَٔاتِيكُم مِّنْهَا بِخَبَرٍ أَوْ ءَاتِيكُم بِشِهَابٍ قَبَسٍ لَّعَلَّكُمْ تَصْطَلُونَ ﴿٧﴾
ئۆز ۋاقتىدا مۇسا ئەھلىيىسىگە: «مەن ئىشەنچلىك ھالدا ئوت كۆردۈم، مەن ئوتنىڭ يېنىغا بېرىپ (يول توغرۇلۇق) بىرەر خەۋەر ئۇقۇپ كېلەي، ياكى سىلەرنىڭ ئىسسىنىشىڭلار ئۈچۈن ئوتتىن بىر چوغ ئېلىپ كېلەي» دېدى
فَلَمَّا جَآءَهَا نُودِىَ أَنۢ بُورِكَ مَن فِى ٱلنَّارِ وَمَنْ حَوْلَهَا وَسُبْحَـٰنَ ٱللَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٨﴾
مۇسا ئوتنىڭ يېنىغا كەلگەندە (مۇنداق) نىدا ئاڭلاندى: «اﷲ ئوتنىڭ يېنىدىكى كىشىنى ۋە ئوتنىڭ ئەتراپىدىكىلەرنى مۇبارەك قىلدى. ئالەملەرنىڭ پەرۋەردىگارى اﷲ پاكتۇر
يَـٰمُوسَىٰٓ إِنَّهُۥٓ أَنَا ٱللَّهُ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٩﴾
ئى مۇسا! شۈبھىسىزكى، مەن غالىب، ھېكمەت بىلەن ئىش قىلغۇچى اﷲ دۇرمەن
وَأَلْقِ عَصَاكَ ۚ فَلَمَّا رَءَاهَا تَهْتَزُّ كَأَنَّهَا جَآنٌّ وَلَّىٰ مُدْبِرًا وَلَمْ يُعَقِّبْ ۚ يَـٰمُوسَىٰ لَا تَخَفْ إِنِّى لَا يَخَافُ لَدَىَّ ٱلْمُرْسَلُونَ ﴿١٠﴾
ھاساڭنى تاشلىغىن!» مۇسا ھاسىسىنىڭ گويا ئەجدىھادەك تېز ھەرىكەتلىنىۋاتقانلىقىنى كۆرگەندە، ئارقىسىغا بۇرۇلۇپ چېكىندى، كەينىگە (يەنى قورققىنىدىن ئەجدىھاغا) قارىيالمىدى. (اﷲ ئېيتتى) «ئى مۇسا! قورقمىغىن، مېنىڭ ھۇزۇرۇمدا خۇدانىڭ ئەلچى-رەسۇللىرى ئەلۋەتتە قورقمايدۇ
إِلَّا مَن ظَلَمَ ثُمَّ بَدَّلَ حُسْنًۢا بَعْدَ سُوٓءٍ فَإِنِّى غَفُورٌ رَّحِيمٌ ﴿١١﴾
لېكىن كىمكى (ئۆزىگە) زۇلۇم قىلسا، ئاندىن يامان ئەمەللىرىنى ياخشى ئەمەلگە ئۆزگەرتسە، مەن (ئۇنىڭغا) ناھايىتى مەغپىرەت قىلغۇچىدۇرمەن، ناھايىتى مېھرىباندۇر
وَأَدْخِلْ يَدَكَ فِى جَيْبِكَ تَخْرُجْ بَيْضَآءَ مِنْ غَيْرِ سُوٓءٍ ۖ فِى تِسْعِ ءَايَـٰتٍ إِلَىٰ فِرْعَوْنَ وَقَوْمِهِۦٓ ۚ إِنَّهُمْ كَانُوا۟ قَوْمًا فَـٰسِقِينَ ﴿١٢﴾
قولۇڭنى قوينۇڭغا سالغىن، ئۇ ھېچقانداق ئىللەتسىز ئاپئاق بولۇپ چىقىدۇ، (بۇ) مېنىڭ پىرئەۋن ۋە ئۇنىڭ قەۋمىگە ئېلىپ بېرىشىڭ ئۈچۈن بەرگەن توققۇز مۆجىزەمنىڭ ئىچىدىدۇر، ئۇلار ئىشەنچلىك ھالدا پاسىق قەۋم ئىدى»
فَلَمَّا جَآءَتْهُمْ ءَايَـٰتُنَا مُبْصِرَةً قَالُوا۟ هَـٰذَا سِحْرٌ مُّبِينٌ ﴿١٣﴾
ئۇلارغا بىزنىڭ نۇرغۇن روشەن ئايەتلىرىمىز نازىل بولغان چاغدا، ئۇلار: «بۇ روشەن سېھىردۇر» دېدى
وَجَحَدُوا۟ بِهَا وَٱسْتَيْقَنَتْهَآ أَنفُسُهُمْ ظُلْمًا وَعُلُوًّا ۚ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُفْسِدِينَ ﴿١٤﴾
ئۇلار ئۇ ئايەتلەرنى ئىچىدە ئېتىراپ قىلدى، لېكىن ئۇلار ئۇنى زۇلۇم ۋە تەكەببۇرلۇق قىلىش يۈزىسىدىن ئىنكار قىلدى. بۇزغۇنچىلارنىڭ ئاقىۋىتىنىڭ قانداق بولىدىغانلىقىغا قارىغىن
وَلَقَدْ ءَاتَيْنَا دَاوُۥدَ وَسُلَيْمَـٰنَ عِلْمًا ۖ وَقَالَا ٱلْحَمْدُ لِلَّهِ ٱلَّذِى فَضَّلَنَا عَلَىٰ كَثِيرٍ مِّنْ عِبَادِهِ ٱلْمُؤْمِنِينَ ﴿١٥﴾
بىز ئىشەنچلىك ھالدا داۋۇدقا، سۇلايمانغا (دۇنيا ۋە دىن ئىلىملىرىدىن كەڭ) ئىلىم ئاتا قىلدۇق، ئۇلار ئېيتتى: «جىمى ھەمدۇسانا بىزنى نۇرغۇن بەندىلىرىدىن ئارتۇق قىلغان اﷲ قا خاستۇر!»
وَوَرِثَ سُلَيْمَـٰنُ دَاوُۥدَ ۖ وَقَالَ يَـٰٓأَيُّهَا ٱلنَّاسُ عُلِّمْنَا مَنطِقَ ٱلطَّيْرِ وَأُوتِينَا مِن كُلِّ شَىْءٍ ۖ إِنَّ هَـٰذَا لَهُوَ ٱلْفَضْلُ ٱلْمُبِينُ ﴿١٦﴾
سۇلايمان (خۇدانىڭ ئەلچىسىلىكتە، ئىلىمدە، پادىشاھلىقتا ئاتىسى) داۋۇدقا ۋارىسلىق قىلدى. ئۇ: «ئى ئىنسانلار! بىزگە قۇشلارنىڭ تىلى تەلىم بېرىلدى. (دۇنيانىڭ نېمەتلىرىدىن) ھەممە نەرسە ئاتا قىلىندى، بۇ ئەلۋەتتە (اﷲ نىڭ) روشەن ئېھسانىدۇر» دېدى
وَحُشِرَ لِسُلَيْمَـٰنَ جُنُودُهُۥ مِنَ ٱلْجِنِّ وَٱلْإِنسِ وَٱلطَّيْرِ فَهُمْ يُوزَعُونَ ﴿١٧﴾
سۇلايماننىڭ جىنلاردىن، ئىنسانلاردىن ۋە قۇشلاردىن بولغان قوشۇنلىرى توپلاندى (سۇلايمان ئەلەيھىسسالام ئۇلارنىڭ ئالدىدا دەبدەبە بىلەن ماڭاتتى). ئۇلار تەرتىپلىك ئورۇنلاشتۇرۇلغان ئىدى
حَتَّىٰٓ إِذَآ أَتَوْا۟ عَلَىٰ وَادِ ٱلنَّمْلِ قَالَتْ نَمْلَةٌ يَـٰٓأَيُّهَا ٱلنَّمْلُ ٱدْخُلُوا۟ مَسَـٰكِنَكُمْ لَا يَحْطِمَنَّكُمْ سُلَيْمَـٰنُ وَجُنُودُهُۥ وَهُمْ لَا يَشْعُرُونَ ﴿١٨﴾
ئۇلار (شامدىكى بىر) چۈمۈلىلەر ۋادىسىغا يېتىپ كەلگەندە، بىر چۈمۈلە ئېيتتى: «ئى چۈمۈلىلەر! ئۇۋىلىرىڭلارغا كىرىپ كېتىڭلار، سۇلايمان ۋە ئۇنىڭ قوشۇنى ئۇقماستىن سىلەرنى يەنچىۋەتمىسۇن»
فَتَبَسَّمَ ضَاحِكًا مِّن قَوْلِهَا وَقَالَ رَبِّ أَوْزِعْنِىٓ أَنْ أَشْكُرَ نِعْمَتَكَ ٱلَّتِىٓ أَنْعَمْتَ عَلَىَّ وَعَلَىٰ وَٰلِدَىَّ وَأَنْ أَعْمَلَ صَـٰلِحًا تَرْضَىٰهُ وَأَدْخِلْنِى بِرَحْمَتِكَ فِى عِبَادِكَ ٱلصَّـٰلِحِينَ ﴿١٩﴾
سۇلايمان چۈمۈلىنىڭ سۆزىدىن تەبەسسۇم قىلىپ كۈلدى ۋە ئېيتتى: «پەرۋەردىگارىم! سەن مېنى ماڭا ۋە ئاتا - ئانامغا بەرگەن نېمىتىڭگە شۈكۈر قىلىشقا، سەن رازى بولىدىغان ياخشى ئەمەلنى قىلىشقا مۇۋەپپەق قىلغىن، رەھمىتىڭ بىلەن مېنى ياخشى بەندىلىرىڭ قاتارىغا كىرگۈزگىن»
وَتَفَقَّدَ ٱلطَّيْرَ فَقَالَ مَا لِىَ لَآ أَرَى ٱلْهُدْهُدَ أَمْ كَانَ مِنَ ٱلْغَآئِبِينَ ﴿٢٠﴾
ئۇ قۇشلارنى كۆزدىن كەچۈرگەندىن كېيىن ئېيتتى: «ماڭا نېمە بولدى؟ ھۆپۆپنى كۆرمەيمەنغۇ! يا ئۇ يوقاپ كەتتىمۇ
لَأُعَذِّبَنَّهُۥ عَذَابًا شَدِيدًا أَوْ لَأَا۟ذْبَحَنَّهُۥٓ أَوْ لَيَأْتِيَنِّى بِسُلْطَـٰنٍ مُّبِينٍ ﴿٢١﴾
ئۇنى چوقۇم قاتتىق جازالايمەن، يا ئۇنى چوقۇم بوغۇزلايمەن، يا چوقۇم (يوقاپ كەتكەنلىكىنى ئاقلايدىغان) بىر روشەن دەلىل كەلتۈرىدۇ»
فَمَكَثَ غَيْرَ بَعِيدٍ فَقَالَ أَحَطتُ بِمَا لَمْ تُحِطْ بِهِۦ وَجِئْتُكَ مِن سَبَإٍۭ بِنَبَإٍ يَقِينٍ ﴿٢٢﴾
ئۇزاق ئۆتمەي ھۆپۆپ كەلدى، ئۇ ئېيتتى: «مەن سەن بىلمىگەن ئىشنى بىلىپ (كەلدىم)، ساڭا مەن سەبەدىن (يەنى يەمەندىكى سەبە شەھىرىدىن) بىر مۇھىم خەۋەر ئېلىپ كەلدىم
إِنِّى وَجَدتُّ ٱمْرَأَةً تَمْلِكُهُمْ وَأُوتِيَتْ مِن كُلِّ شَىْءٍ وَلَهَا عَرْشٌ عَظِيمٌ ﴿٢٣﴾
مەن ئىشەنچلىك ھالدا ئۇلارغا (يەنى سەبە ئاھالىسىگە) بىر ئايالنىڭ (يەنى بىلقىسنىڭ) پادىشاھلىق قىلىۋاتقانلىقىنى كۆردۈم، ئۇنىڭغا (ئۆزىگە ۋە سەلتەنەتىگە كېرەكلىك) ھەممە نەرسە بېرىلگەن ئىكەن، ئۇ چوڭ ئەرشكە ئىگە ئىكەن
وَجَدتُّهَا وَقَوْمَهَا يَسْجُدُونَ لِلشَّمْسِ مِن دُونِ ٱللَّهِ وَزَيَّنَ لَهُمُ ٱلشَّيْطَـٰنُ أَعْمَـٰلَهُمْ فَصَدَّهُمْ عَنِ ٱلسَّبِيلِ فَهُمْ لَا يَهْتَدُونَ ﴿٢٤﴾
ئۇنىڭ ۋە قەۋمىنىڭ اﷲ نى قويۇپ قۇياشقا چوقۇنىدىغانلىقىنى بايقىدىم، شەيتان (ئازدۇرغۇچى) ئۇلارغا قىلمىشلىرىنى (يەنى اﷲ نى قويۇپ، قۇياشقا چوقۇنغانلىقلىرىنى) چىرايلىق كۆرسەتتى، ئۇلارنى توغرا يولدىن توستى، ئۇلار ھىدايەت تاپمايدۇ
أَلَّا يَسْجُدُوا۟ لِلَّهِ ٱلَّذِى يُخْرِجُ ٱلْخَبْءَ فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَيَعْلَمُ مَا تُخْفُونَ وَمَا تُعْلِنُونَ ﴿٢٥﴾
ئۇلار ئاسمانلاردىكى ۋە زېمىندىكى سىرلەرنى ئاشكارىلىغۇچى، سىلەرنىڭ يوشۇرۇن ۋە ئاشكارا ئىشىڭلارنى بىلىپ تۇرغۇچى اﷲ قا سەجدە قىلمايدۇ
ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ رَبُّ ٱلْعَرْشِ ٱلْعَظِيمِ ۩ ﴿٢٦﴾
اﷲ تىن باشقا ھېچ ئىلاھ يوقتۇر، ئۇ بۈيۈك ئەرشنىڭ پەرۋەردىگارىدۇر»
۞ قَالَ سَنَنظُرُ أَصَدَقْتَ أَمْ كُنتَ مِنَ ٱلْكَـٰذِبِينَ ﴿٢٧﴾
سۇلايمان ئېيتتى: «سەن راست دەۋاتامسەن، يا يالغانمۇ؟ قارايمىز
ٱذْهَب بِّكِتَـٰبِى هَـٰذَا فَأَلْقِهْ إِلَيْهِمْ ثُمَّ تَوَلَّ عَنْهُمْ فَٱنظُرْ مَاذَا يَرْجِعُونَ ﴿٢٨﴾
بۇ خېتىمنى ئېلىپ بېرىپ ئۇلارغا تاشلىغىن، ئاندىن ئۇلاردىن نېرىراق (جايدا يوشۇرۇنۇپ) تۇر، ئۇلارنىڭ قانداق جاۋاپ قايتۇرىدىغانلىقىغا قارىغىن»
قَالَتْ يَـٰٓأَيُّهَا ٱلْمَلَؤُا۟ إِنِّىٓ أُلْقِىَ إِلَىَّ كِتَـٰبٌ كَرِيمٌ ﴿٢٩﴾
ئۇ (يەنى بىلقىس) ئېيتتى: «ئى ئۇلۇغلار! ماڭا ئىشەنچلىك ھالدا سۇلايماندىن بىر پارچە قىممەتلىك خەت كەلدى. (ئۇنىڭ مەزمۇنى شۇكى) ناھايىتى شەپقەتلىك ۋە مېھرىبان اﷲ نىڭ ئىسمى بىلەن باشلايمەن
إِنَّهُۥ مِن سُلَيْمَـٰنَ وَإِنَّهُۥ بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ ﴿٣٠﴾
ئۇ (يەنى بىلقىس) ئېيتتى: «ئى ئۇلۇغلار! ماڭا ئىشەنچلىك ھالدا سۇلايماندىن بىر پارچە قىممەتلىك خەت كەلدى. (ئۇنىڭ مەزمۇنى شۇكى) ناھايىتى شەپقەتلىك ۋە مېھرىبان اﷲ نىڭ ئىسمى بىلەن باشلايمەن
أَلَّا تَعْلُوا۟ عَلَىَّ وَأْتُونِى مُسْلِمِينَ ﴿٣١﴾
سىلەر ماڭا ھاكاۋۇرلۇق قىلماڭلار، مېنىڭ ئالدىمغا مۇسۇلمان بولغان ھالدا كېلىڭلار»
قَالَتْ يَـٰٓأَيُّهَا ٱلْمَلَؤُا۟ أَفْتُونِى فِىٓ أَمْرِى مَا كُنتُ قَاطِعَةً أَمْرًا حَتَّىٰ تَشْهَدُونِ ﴿٣٢﴾
ئۇ (يەنى بىلقىس) ئېيتتى: «ئى ئۇلۇغلار! مېنىڭ (بۇ) ئىشىمدا مەسلىھەت بېرىڭلار، سىلەرنى ئۈستىدە قويماي تۇرۇپ ھېچ ئىشنى بېكىتكىنىم يوق»
قَالُوا۟ نَحْنُ أُو۟لُوا۟ قُوَّةٍ وَأُو۟لُوا۟ بَأْسٍ شَدِيدٍ وَٱلْأَمْرُ إِلَيْكِ فَٱنظُرِى مَاذَا تَأْمُرِينَ ﴿٣٣﴾
ئۇلار ئېيتتى: «بىز كۈچلۈك ۋە جەڭگىۋارمىز، ئىش سېنىڭ ئىختىيارىڭدىدۇر، (بىزنى) نېمىگە بۇيرۇيدىغانلىقىڭنى (ئويلاپ) كۆرگىن»
قَالَتْ إِنَّ ٱلْمُلُوكَ إِذَا دَخَلُوا۟ قَرْيَةً أَفْسَدُوهَا وَجَعَلُوٓا۟ أَعِزَّةَ أَهْلِهَآ أَذِلَّةً ۖ وَكَذَٰلِكَ يَفْعَلُونَ ﴿٣٤﴾
ئۇ (يەنى بىلقىس) ئېيتتى: «شۈبھىسىزكى، پادىشاھلار بىرەر شەھەرگە ھۇجۇم قىلىپ كىرسە، ئۇنى خاراب قىلىدۇ، شەھەرنىڭ مۆتىۋەر ئادەملىرىنى (ئۆلتۈرۈش، ئەسىرگە ئېلىش ۋە سۈرگۈن قىلىش بىلەن) خار قىلىدۇ، ئۇلارمۇ شۇنداق قىلىدۇ
وَإِنِّى مُرْسِلَةٌ إِلَيْهِم بِهَدِيَّةٍ فَنَاظِرَةٌۢ بِمَ يَرْجِعُ ٱلْمُرْسَلُونَ ﴿٣٥﴾
مەن چوقۇم ئۇلارغا سوۋغا ئەۋەتىمەن، ئەلچىلەرنىڭ نېمە خەۋەر ئېلىپ كېلىدىغانلىقىغا قارايمەن»
فَلَمَّا جَآءَ سُلَيْمَـٰنَ قَالَ أَتُمِدُّونَنِ بِمَالٍ فَمَآ ءَاتَىٰنِۦَ ٱللَّهُ خَيْرٌ مِّمَّآ ءَاتَىٰكُم بَلْ أَنتُم بِهَدِيَّتِكُمْ تَفْرَحُونَ ﴿٣٦﴾
ئەلچى سۇلايماننىڭ ئالدىغا يېتىپ كەلگەندە، سۇلايمان ئېيتتى: «سىلەر ماڭا مال بىلەن ياردەم قىلماقچىمۇ؟ اﷲ نىڭ ماڭا بەرگەنلىرى سىلەرنىڭ بەرگەنلىرىڭلاردىن ياخشىدۇر، بەلكى سىلەر سوغاڭلار بىلەن كۆرەڭلەپ كېتىسىلەر
ٱرْجِعْ إِلَيْهِمْ فَلَنَأْتِيَنَّهُم بِجُنُودٍ لَّا قِبَلَ لَهُم بِهَا وَلَنُخْرِجَنَّهُم مِّنْهَآ أَذِلَّةً وَهُمْ صَـٰغِرُونَ ﴿٣٧﴾
سەن قايتىپ كەتكىن، ئۇلارنىڭ ئۈستىگە ئۇلار تاقابىل تۇرالمايدىغان بىر قوشۇن بىلەن چوقۇم بارىمىز، ئۇلارنى يۇرتىدىن خار، كەمسىتىلگەن ھالدا چوقۇم چىقىرىۋېتىمىز»
قَالَ يَـٰٓأَيُّهَا ٱلْمَلَؤُا۟ أَيُّكُمْ يَأْتِينِى بِعَرْشِهَا قَبْلَ أَن يَأْتُونِى مُسْلِمِينَ ﴿٣٨﴾
سۇلايمان ئېيتتى: «ئى ئۇلۇغلار! ئۇلار مېنىڭ قېشىمغا مۇسۇلمان بولۇپ كېلىشتىن بۇرۇن، (سىلەردىن) كىم ئۇنىڭ تەختىنى ئېلىپ كېلىدۇ»
قَالَ عِفْرِيتٌ مِّنَ ٱلْجِنِّ أَنَا۠ ءَاتِيكَ بِهِۦ قَبْلَ أَن تَقُومَ مِن مَّقَامِكَ ۖ وَإِنِّى عَلَيْهِ لَقَوِىٌّ أَمِينٌ ﴿٣٩﴾
جىندىن بولغان ئىفرىت ئېيتتى: «مەن ئۇنى ئورنۇڭدىن تۇرۇشتىن بۇرۇن ساڭا ئېلىپ كېلىمەن، مەن ئۇنى ئېلىپ كېلىشكە ئەلۋەتتە قادىرمەن، ئىشەنچلىكمەن»
قَالَ ٱلَّذِى عِندَهُۥ عِلْمٌ مِّنَ ٱلْكِتَـٰبِ أَنَا۠ ءَاتِيكَ بِهِۦ قَبْلَ أَن يَرْتَدَّ إِلَيْكَ طَرْفُكَ ۚ فَلَمَّا رَءَاهُ مُسْتَقِرًّا عِندَهُۥ قَالَ هَـٰذَا مِن فَضْلِ رَبِّى لِيَبْلُوَنِىٓ ءَأَشْكُرُ أَمْ أَكْفُرُ ۖ وَمَن شَكَرَ فَإِنَّمَا يَشْكُرُ لِنَفْسِهِۦ ۖ وَمَن كَفَرَ فَإِنَّ رَبِّى غَنِىٌّ كَرِيمٌ ﴿٤٠﴾
(نازىل بولغان) مۇقەددەس كىتاب-ۋەھىينى چوڭقۇر بىلىدىغان زات (يەنى ئاسەف ئىبن بۇرخىيا): «ئۇنى مەن ساڭا كۆزۈڭنى يۇمۇپ ئاچقۇچە ئېلىپ كېلىمەن» دېدى (ئۇ دۇئا قىلىۋېدى، تەخت دەرھال ئالدىدا ھازىر بولدى). سۇلايمان تەختنىڭ يېنىدا تۇرغانلىقىنى كۆرگەندە: «بۇ پەرۋەردىگارىمنىڭ (ماڭا قىلغان) ئېھسانىدۇر، ئۇ شۈكۈر قىلامدىم، يا تۇز كورلۇق قىلامدىم، بۇنىڭ بىلەن مېنى سىنىدى، كىمكى شۈكۈر قىلىدىكەن، ئۇ ئۆزىنىڭ پايدىسى ئۈچۈن شۈكۈر قىلىدۇ، كىمكى تۇزكورلۇق قىلىدىكەن، (بىلىش كېرەككى) ئىشەنچلىك ھالدا پەرۋەردىگارىم (ئۇنىڭ شۈكۈر قىلىشىدىن بىھاجەتتۇر)، (اﷲ نىڭ) كەرىمى كەڭدۇر» دېدى
قَالَ نَكِّرُوا۟ لَهَا عَرْشَهَا نَنظُرْ أَتَهْتَدِىٓ أَمْ تَكُونُ مِنَ ٱلَّذِينَ لَا يَهْتَدُونَ ﴿٤١﴾
سۇلايمان (بىلقىسنىڭ يېتىپ كېلىشى يېقىنلاشقان ۋاقىتتا ئۇنىڭ ئەقلىنى سىناش يۈزىسىدىن) ئېيتتى: «ئۇنىڭ (يەنى بىلقىسنىڭ) تەختىنىڭ (شەكلىنى) ئۆزگەرتىڭلار، ئۇ تەختىنى تونۇمدۇ، تونىمامدۇ، قارايمىز»
فَلَمَّا جَآءَتْ قِيلَ أَهَـٰكَذَا عَرْشُكِ ۖ قَالَتْ كَأَنَّهُۥ هُوَ ۚ وَأُوتِينَا ٱلْعِلْمَ مِن قَبْلِهَا وَكُنَّا مُسْلِمِينَ ﴿٤٢﴾
بىلقىس كەلگەندە (ئۇنىڭغا): «سېنىڭ تەختىڭ مۇشۇنداقمۇ؟» دېيىلدى. ئۇ: «شۇدەك تۇرىدۇ» دېدى. (سۇلايمان اﷲ نىڭ نېمىتىنى سۆزلەش يۈزىسىدىن ئېيتتى) ئۇنىڭدىن (يەنى بىلقىستىن) بۇرۇن بىزگە (اﷲ نى ۋە اﷲ نىڭ قۇدرىتىنى بىلدۈرىدىغان) ئىلىم بېرىلدى، بىز مۇسۇلمان بولدۇق
وَصَدَّهَا مَا كَانَت تَّعْبُدُ مِن دُونِ ٱللَّهِ ۖ إِنَّهَا كَانَتْ مِن قَوْمٍ كَـٰفِرِينَ ﴿٤٣﴾
ئۇنى اﷲ نى قويۇپ چوقۇنغان نەرسىلىرى (اﷲ قا ئىبادەت قىلىشتىن) توستى. چۈنكى ئۇ ھەقىقەتنى رەت قىلغۇچى قەۋمدىن ئىدى
قِيلَ لَهَا ٱدْخُلِى ٱلصَّرْحَ ۖ فَلَمَّا رَأَتْهُ حَسِبَتْهُ لُجَّةً وَكَشَفَتْ عَن سَاقَيْهَا ۚ قَالَ إِنَّهُۥ صَرْحٌ مُّمَرَّدٌ مِّن قَوَارِيرَ ۗ قَالَتْ رَبِّ إِنِّى ظَلَمْتُ نَفْسِى وَأَسْلَمْتُ مَعَ سُلَيْمَـٰنَ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٤٤﴾
ئۇنىڭغا (يەنى بىلقىسقا): «سارايغا كىرگىن» دېيىلدى، ئۇ (يەنى بىلقىس) ساراينى چوڭ سۇ دەپ گۇمانلىنىپ (كىيىمىنى كۆتۈرۈپ) ئىككى پاچىقىنى ئاچتى، سۇلايمان ئېيتتى: «شۈبھىسىزكى، ئۇ ئەينەكتىن ياسالغان سارايدۇر». ئۇ (يەنى بىلقىس) ئېيتتى: «پەرۋەردىگارىم، مەن ئىشەنچلىك ھالدا (اﷲ قا شېرىك كەلتۈرۈپ، قۇياشقا چوقۇنۇش بىلەن) ئۆزۈمگە زۇلۇم قىلدىم، سۇلايمان بىلەن بىللە ئالەملەرنىڭ پەرۋەردىگارى اﷲ قا بويسۇندۇم (يەنى سۇلايماننىڭ دىنىغا ئەگىشىپ ئىسلامغا كىردىم)»
وَلَقَدْ أَرْسَلْنَآ إِلَىٰ ثَمُودَ أَخَاهُمْ صَـٰلِحًا أَنِ ٱعْبُدُوا۟ ٱللَّهَ فَإِذَا هُمْ فَرِيقَانِ يَخْتَصِمُونَ ﴿٤٥﴾
بىز ئىشەنچلىك ھالدا سەمۇدقا (يەنى سەمۇد قەبىلىسىگە نەسەب جەھەتتىن) قېرىندىشى سالىھنى (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، (سالىھ) «اﷲ قا ئىبادەت قىلىڭلار» (دېدى)، ناگاھان ئۇلار جېدەللىشىدىغان (يەنى دىن بارىسىدا جىدەللىشىدىغان مۆمىنلەر ۋە ھەقىقەتنى رەت قىلغۇچىلاردىن ئىبارەت) ئىككى گۇرۇھقا بۆلۈنۈپ كەتتى
قَالَ يَـٰقَوْمِ لِمَ تَسْتَعْجِلُونَ بِٱلسَّيِّئَةِ قَبْلَ ٱلْحَسَنَةِ ۖ لَوْلَا تَسْتَغْفِرُونَ ٱللَّهَ لَعَلَّكُمْ تُرْحَمُونَ ﴿٤٦﴾
سالىھ ئېيتتى: «ئى قەۋمىم، سىلەر نېمىشقا رەھمەتتىن بۇرۇن ئازابنىڭ كېلىشىنى تىلەيسىلەر؟ سىلەر رەھمەتكە ئېرىشىشىڭلار ئۈچۈن نېمىشقا اﷲ تىن مەغپىرەت تەلەپ قىلمايسىلەر؟»
قَالُوا۟ ٱطَّيَّرْنَا بِكَ وَبِمَن مَّعَكَ ۚ قَالَ طَـٰٓئِرُكُمْ عِندَ ٱللَّهِ ۖ بَلْ أَنتُمْ قَوْمٌ تُفْتَنُونَ ﴿٤٧﴾
ئۇلار ئېيتتى: «(بىزگە كەلگەن قەھەتچىلىك) سېنىڭ ۋە سەن بىلەن بولغان كىشىلەرنىڭ شۇملۇقىدىن كەلدى». سالىھ ئېيتتى: «سىلەرگە كېلىدىغان ياخشى يامانلىق اﷲ تەرىپىدىن كېلىدۇ، بەلكى سىلەرنى اﷲ سىنايدۇ»
وَكَانَ فِى ٱلْمَدِينَةِ تِسْعَةُ رَهْطٍ يُفْسِدُونَ فِى ٱلْأَرْضِ وَلَا يُصْلِحُونَ ﴿٤٨﴾
شەھەردە (يەنى ھىجرىدە) يەر يۈزىدە بۇزغۇنچىلىق قىلىدىغان، ئىسلاھ قىلمايدىغان توققۇز نەپەر كىشى بار ئىدى
قَالُوا۟ تَقَاسَمُوا۟ بِٱللَّهِ لَنُبَيِّتَنَّهُۥ وَأَهْلَهُۥ ثُمَّ لَنَقُولَنَّ لِوَلِيِّهِۦ مَا شَهِدْنَا مَهْلِكَ أَهْلِهِۦ وَإِنَّا لَصَـٰدِقُونَ ﴿٤٩﴾
ئۇلار: «ئۆز ئارا اﷲ بىلەن قەسەم قىلىڭلار» دېدى. ئۇلار ئېيتتى: «بىر كېچىدە سالىھنى ۋە ئۇنىڭ تەۋەلىرىنى چوقۇم ئۆلتۈرەيلى، ئاندىن چوقۇم ئۇنىڭ (يەنى ئۇلارنىڭ) ئىگىسىگە، ئۇ ئۆلتۈرۈلگەن چاغدا بىز ئۈستىدە ئەمەس ئىدۇق، بىز ئىشەنچلىك ھالدا راستچىلمىز، دەيلى»
وَمَكَرُوا۟ مَكْرًا وَمَكَرْنَا مَكْرًا وَهُمْ لَا يَشْعُرُونَ ﴿٥٠﴾
ئۇلار (سالىھقا قارشى) سۇيىقەست پىلانلىدى. ئۇلارنى (ئۇلارنىڭ ھالاك بولۇشىنى تېزلىتىش ئۈچۈن) سۇيىقەستى ئۈچۈن تۇيۇقسىز جازالىدۇق
فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ مَكْرِهِمْ أَنَّا دَمَّرْنَـٰهُمْ وَقَوْمَهُمْ أَجْمَعِينَ ﴿٥١﴾
ئۇلارنىڭ سۇيىقەستىنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىغا قارىغىنكى، ئۇلارنى ۋە ئۇلارنىڭ قەۋمىنى پۈتۈنلەي ھالاك قىلدۇق
فَتِلْكَ بُيُوتُهُمْ خَاوِيَةًۢ بِمَا ظَلَمُوٓا۟ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّقَوْمٍ يَعْلَمُونَ ﴿٥٢﴾

شۈبھىسىزكى، پەرۋەردىگارىڭ ئۇلار (يەنى بەنى ئىسرائىل ئەۋلادى) نىڭ ئارىسىدا (قىيامەت كۈنى) ئادىل ھۆكۈم چىقىرىدۇ، اﷲ غالىبتۇر، (بەندىلىرىنىڭ ئىشلىرىنى) بىلگۈچىدۇر
فَتَوَكَّلْ عَلَى ٱللَّهِ ۖ إِنَّكَ عَلَى ٱلْحَقِّ ٱلْمُبِينِ ﴿٧٩﴾
اﷲ قا تەۋەككۈل قىلغىن (يەنى ئىشىڭنى اﷲ قا تاپشۇرغىن، ھەممە ئىشىڭدا اﷲ قا يۆلەنگىن، اﷲ ساڭا مەدەتكاردۇر)، (ئى مۇھەممەد) شۈبھىسىزكى، سەن روشەن ھەق (دىن) دىسەن
إِنَّكَ لَا تُسْمِعُ ٱلْمَوْتَىٰ وَلَا تُسْمِعُ ٱلصُّمَّ ٱلدُّعَآءَ إِذَا وَلَّوْا۟ مُدْبِرِينَ ﴿٨٠﴾
سەن ئۆلۈكلەرگە ۋە يۈز ئۆرۈگەن گاسلارغا (يەنى دىللىرىنىڭ ئۆلۈكلىكىدە ئۆلۈكلەرگە ۋە ھەقنى ئاڭلىماسلىقتا گاسلارغا ئوخشايدىغان كۇففارلارغا) دەۋەتنى ئاڭلىتالمايسەن
وَمَآ أَنتَ بِهَـٰدِى ٱلْعُمْىِ عَن ضَلَـٰلَتِهِمْ ۖ إِن تُسْمِعُ إِلَّا مَن يُؤْمِنُ بِـَٔايَـٰتِنَا فَهُم مُّسْلِمُونَ ﴿٨١﴾
سەن (دىلى) كورلارنى گۇمراھلىقتىن ئايرىپ، ھىدايەت قىلالمايسەن، سەن (دەۋىتىڭنى) پەقەت بىزنىڭ ئايەتلىرىمىزگە ئىمان ئېيتىپ مۇسۇلمان بولغانلارغىلا ئاڭلىتالايسەن
۞ وَإِذَا وَقَعَ ٱلْقَوْلُ عَلَيْهِمْ أَخْرَجْنَا لَهُمْ دَآبَّةً مِّنَ ٱلْأَرْضِ تُكَلِّمُهُمْ أَنَّ ٱلنَّاسَ كَانُوا۟ بِـَٔايَـٰتِنَا لَا يُوقِنُونَ ﴿٨٢﴾
ئۇلار (يەنى كۇففارلار) غا ئالدىنئالا ئېيتىلغان سۆز ئىشقا ئاشقاندا (يەنى ئازاپ ۋە قىيامەت يېقىنلاشقاندا)، ئۇلارغا زېمىندىن بىر تۈرلۈك ھايۋاننى چىقىرىمىزكى، ئۇ ئۇلارغا ئىنسانلارنىڭ بىزنىڭ ئايەتلىرىمىزگە ئىشەنمىگەنلىكىنى سۆزلەيدۇ
وَيَوْمَ نَحْشُرُ مِن كُلِّ أُمَّةٍ فَوْجًا مِّمَّن يُكَذِّبُ بِـَٔايَـٰتِنَا فَهُمْ يُوزَعُونَ ﴿٨٣﴾
ئۇ كۈندە ھەر ئۈممەتنىڭ ئىچىدىن بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلاردىن بىر توپ ئادەمنى) ھېساب ئېلىش ۋە جازالاش ئۈچۈن) يىغىمىز (ئالدى - كەينى يىغىلغۇچە) ئۇلار توختىتىپ تۇرۇلىدۇ
حَتَّىٰٓ إِذَا جَآءُو قَالَ أَكَذَّبْتُم بِـَٔايَـٰتِى وَلَمْ تُحِيطُوا۟ بِهَا عِلْمًا أَمَّاذَا كُنتُمْ تَعْمَلُونَ ﴿٨٤﴾
ئۇلار) اﷲ نىڭ دەرگاھىغا) كەلگەن چاغلىرىدا، اﷲ ئۇلارغا: «سىلەر تولۇق بىلمەي تۇرۇپ مېنىڭ ئايەتلىرىمنى ئىنكار قىلدىڭلارمۇ؟ سىلەر (ئەمرى قىلىنغان نەرسىلەردىن) زادى نېمىلەرنى قىلدىڭلار؟» دەيدۇ
وَوَقَعَ ٱلْقَوْلُ عَلَيْهِم بِمَا ظَلَمُوا۟ فَهُمْ لَا يَنطِقُونَ ﴿٨٥﴾
ئۇلار) ئۆزلىرىگە) زۇلۇم قىلغانلىقلىرى ئۈچۈن، ئۇلارغا ئازاب ھەقلىق بولدى. ئۇلار (ئۆزرىلىرى بولمىغانلىقى ئۈچۈن) سۆز قىلالمايدۇ
أَلَمْ يَرَوْا۟ أَنَّا جَعَلْنَا ٱلَّيْلَ لِيَسْكُنُوا۟ فِيهِ وَٱلنَّهَارَ مُبْصِرًا ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٨٦﴾
ئۇلارغا كېچىنى ئارام ئالىدىغان (ۋاقىت) قىلغانلىقىمىزنى، كۈندۈزنى (ھاياتلىق يولىدا ھەرىكەتلىنىش ئۈچۈن) يورۇق قىلغانلىقىمىزنى ئۇلار ئۇقمامدۇ؟ بۇنىڭدا ئىمان ئېيتىدىغان قەۋم ئۈچۈن (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) نۇرغۇن ئالامەتلەر بار
وَيَوْمَ يُنفَخُ فِى ٱلصُّورِ فَفَزِعَ مَن فِى ٱلسَّمَـٰوَٰتِ وَمَن فِى ٱلْأَرْضِ إِلَّا مَن شَآءَ ٱللَّهُ ۚ وَكُلٌّ أَتَوْهُ دَٰخِرِينَ ﴿٨٧﴾
ئۇ كۈندە سۇر چىلىنىدۇ، اﷲ خالىغانلاردىن (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى، خۇدانىڭ ئەلچى-رەسۇللىرى ۋە شېھىتلاردىن) باشقا ئاسمانلاردىكى ۋە زېمىندىكى ھەممىنى قورقۇنچ باسىدۇ، ھەممە اﷲ قا بويسۇنغان ھالدا كېلىدۇ
وَتَرَى ٱلْجِبَالَ تَحْسَبُهَا جَامِدَةً وَهِىَ تَمُرُّ مَرَّ ٱلسَّحَابِ ۚ صُنْعَ ٱللَّهِ ٱلَّذِىٓ أَتْقَنَ كُلَّ شَىْءٍ ۚ إِنَّهُۥ خَبِيرٌۢ بِمَا تَفْعَلُونَ ﴿٨٨﴾
تاغلارنى تۇرغۇن ھالەتتە گۇمان قىلىسەن، ھالبۇكى، ئۇلار بۇلۇتتەك چۆگىلەپ تۇرىدۇ، (بۇ) ھەممە نەرسىنى پۇختا ياراتقان اﷲ نىڭ ھۈنىرىدۇر، اﷲ ئىشەنچلىك ھالدا سىلەرنىڭ قىلمىشلىرىڭلاردىن تولۇق خەۋەرداردۇر
مَن جَآءَ بِٱلْحَسَنَةِ فَلَهُۥ خَيْرٌ مِّنْهَا وَهُم مِّن فَزَعٍ يَوْمَئِذٍ ءَامِنُونَ ﴿٨٩﴾
ياخشى ئىش قىلغانلار قىلغان ياخشىلىقىدىن ئوبدانراق مۇكاپاتقا ئېرىشىدۇ، ئۇلار بۇ كۈندە قورقۇنجىدىن ئەمىن بولىدۇ
وَمَن جَآءَ بِٱلسَّيِّئَةِ فَكُبَّتْ وُجُوهُهُمْ فِى ٱلنَّارِ هَلْ تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٩٠﴾
يامان ئىش قىلغانلار دوزاخقا يۈزى بىلەن تاشلىنىدۇ، سىلەر پەقەت قىلمىشىڭلارغا يارىشا جازالىنىسىلەر
إِنَّمَآ أُمِرْتُ أَنْ أَعْبُدَ رَبَّ هَـٰذِهِ ٱلْبَلْدَةِ ٱلَّذِى حَرَّمَهَا وَلَهُۥ كُلُّ شَىْءٍ ۖ وَأُمِرْتُ أَنْ أَكُونَ مِنَ ٱلْمُسْلِمِينَ ﴿٩١﴾
(ئى مۇھەممەد! ئېيتقىنكى) «مەن پەقەت اﷲ ھۈرمەتلىك قىلغان بۇ شەھەر (يەنى مەككە مۇكەررەمە) نىڭ پەرۋەردىگارىغا ئىبادەت قىلىشقا بۇيرۇلدۇم، ھەممە نەرسە اﷲ نىڭ مۈلكىدۇر، مەن مۇسۇلمانلاردىن بولۇشقا، قۇرئان ئوقۇشقا بۇيرۇلدۇم». كىمكى ھىدايەت تاپىدىكەن، ئۇ ئۆزىنىڭ پايدىسى ئۈچۈندۇر، كىمكى ئازىدىكەن (ئۇنىڭ ۋابالى ئۆزىگە بولىدۇ) سەن (ئۇنىڭغا) ئېيتقىنكى، «مەن پەقەت ئاگاھلاندۇرغۇچىلاردىنمەن»
وَأَنْ أَتْلُوَا۟ ٱلْقُرْءَانَ ۖ فَمَنِ ٱهْتَدَىٰ فَإِنَّمَا يَهْتَدِى لِنَفْسِهِۦ ۖ وَمَن ضَلَّ فَقُلْ إِنَّمَآ أَنَا۠ مِنَ ٱلْمُنذِرِينَ ﴿٩٢﴾
(ئى مۇھەممەد! ئېيتقىنكى) «مەن پەقەت اﷲ ھۈرمەتلىك قىلغان بۇ شەھەر (يەنى مەككە مۇكەررەمە) نىڭ پەرۋەردىگارىغا ئىبادەت قىلىشقا بۇيرۇلدۇم، ھەممە نەرسە اﷲ نىڭ مۈلكىدۇر، مەن مۇسۇلمانلاردىن بولۇشقا، قۇرئان ئوقۇشقا بۇيرۇلدۇم». كىمكى ھىدايەت تاپىدىكەن، ئۇ ئۆزىنىڭ پايدىسى ئۈچۈندۇر، كىمكى ئازىدىكەن (ئۇنىڭ ۋابالى ئۆزىگە بولىدۇ) سەن (ئۇنىڭغا) ئېيتقىنكى، «مەن پەقەت ئاگاھلاندۇرغۇچىلاردىنمەن»
وَقُلِ ٱلْحَمْدُ لِلَّهِ سَيُرِيكُمْ ءَايَـٰتِهِۦ فَتَعْرِفُونَهَا ۚ وَمَا رَبُّكَ بِغَـٰفِلٍ عَمَّا تَعْمَلُونَ ﴿٩٣﴾
(ئى مۇھەممەد!) ئېيتقىنكى، «جىمى ھەمدۇسانا اﷲ قا خاستۇر، اﷲ سىلەرگە (قۇدرىتىنى كۆرسىتىپ بېرىدىغان) ئالامەتلىرىنى كۆرسىتىدۇ، سىلەر ئۇلارنى (تونۇش پايدا بەرمەيدىغان چاغدا) تونۇيسىلەر، پەرۋەردىگارىڭ سىلەرنىڭ قىلغان ئەمەللىرىڭلاردىن غاپىل ئەمەستۇر
