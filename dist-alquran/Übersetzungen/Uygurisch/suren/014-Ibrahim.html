ابراهيم
Ibrahim · Abraham
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
الٓر ۚ كِتَـٰبٌ أَنزَلْنَـٰهُ إِلَيْكَ لِتُخْرِجَ ٱلنَّاسَ مِنَ ٱلظُّلُمَـٰتِ إِلَى ٱلنُّورِ بِإِذْنِ رَبِّهِمْ إِلَىٰ صِرَٰطِ ٱلْعَزِيزِ ٱلْحَمِيدِ ﴿١﴾
ئەلىق. لام. را. (بۇ قۇرئان شۇنداق) مۇقەددەس كىتاب-ۋەھىيتۇركى، (ئى مۇھەممەد!) ئۇنى ساڭا كىشىلەرنى پەرۋەردىگارىنىڭ ئىزنى بىلەن قاراڭغۇلۇقتىن (يەنى كۇفرىدىن) يورۇقلۇققا (يەنى ئىمانغا) چىقىرىشىڭ ئۈچۈن، غالىب، (ھەممە تىللاردا) مەدھىيىلەنگەن اﷲ نىڭ يولىغا (باشلىشىڭ ئۈچۈن) نازىل قىلدۇق
ٱللَّهِ ٱلَّذِى لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَوَيْلٌ لِّلْكَـٰفِرِينَ مِنْ عَذَابٍ شَدِيدٍ ﴿٢﴾
ئاسمانلاردىكى ۋە زېمىندىكى ھەممە نەرسە اﷲ نىڭ مۈلكىدۇر، قاتتىق ئازابتىن ھەقىقەتنى رەت قىلغۇچىلارغا ۋاي!
ٱلَّذِينَ يَسْتَحِبُّونَ ٱلْحَيَوٰةَ ٱلدُّنْيَا عَلَى ٱلْـَٔاخِرَةِ وَيَصُدُّونَ عَن سَبِيلِ ٱللَّهِ وَيَبْغُونَهَا عِوَجًا ۚ أُو۟لَـٰٓئِكَ فِى ضَلَـٰلٍۭ بَعِيدٍ ﴿٣﴾
ئۇلار بۇ دۇنيانى ئاخىرەتتىن ئارتۇق. كۆرىدۇ (كىشىلەرنى) اﷲ نىڭ يولىدىن (يەنى ئىسلام دىنىدىن) توسىدۇ، اﷲ نىڭ يولىنىڭ ئەگرى بولۇشىنى تىلەيدۇ، ئۇلار (ھەقتىن) يىراق بولغان گۇمراھلىقتىدۇر
وَمَآ أَرْسَلْنَا مِن رَّسُولٍ إِلَّا بِلِسَانِ قَوْمِهِۦ لِيُبَيِّنَ لَهُمْ ۖ فَيُضِلُّ ٱللَّهُ مَن يَشَآءُ وَيَهْدِى مَن يَشَآءُ ۚ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٤﴾
قانداقلا بىر خۇدانىڭ ئەلچىسىنى ئەۋەتمەيلى، ئۇنى بىز ئۆز قەۋمىگە (اﷲ نىڭ شەرىئىتىنى) بايان قىلىش ئۈچۈن، پەقەت ئۆز قەۋمىنىڭ تىلى بىلەن (سۆزلەيدىغان) قىلىپ ئەۋەتتۇق، اﷲ خالىغان ئادەمنى گۇمراھ قىلىدۇ، خالىغان ئادەمنى ھىدايەت قىلىدۇ (يەنى خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ ۋەزىپىسى پەقەت تەبلىغ قىلىش بولۇپ، ھىدايەت قىلىش ئىشى اﷲ نىڭ ئىلگىدىدۇر). اﷲ غالىبتۇر ھېكمەت بىلەن ئىش قىلغۇچىدۇر
وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔايَـٰتِنَآ أَنْ أَخْرِجْ قَوْمَكَ مِنَ ٱلظُّلُمَـٰتِ إِلَى ٱلنُّورِ وَذَكِّرْهُم بِأَيَّىٰمِ ٱللَّهِ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّكُلِّ صَبَّارٍ شَكُورٍ ﴿٥﴾
بىز مۇسانى مۆجىزىلىرىمىز بىلەن ئىشەنچلىك ھالدا (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، (ئۇنىڭغا دېدۇقكى) «قەۋمىڭنى قاراڭغۇلۇقتىن (يەنى كۇفرىدىن) يورۇقلۇققا. (يەنى ئىمانغا) چىقارغىن، ئۇلارغا اﷲ نىڭ نېمەتلىرىنى ئەسلەتكىن». بۇنىڭدا (بالاغا) سەۋر قىلغۇچى، (نېمەتكە) شۈكۈر قىلغۇچى ھەربىر (بەندە) ئۈچۈن ئەلۋەتتە نۇرغۇن ئىبرەتلەر بار
وَإِذْ قَالَ مُوسَىٰ لِقَوْمِهِ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ أَنجَىٰكُم مِّنْ ءَالِ فِرْعَوْنَ يَسُومُونَكُمْ سُوٓءَ ٱلْعَذَابِ وَيُذَبِّحُونَ أَبْنَآءَكُمْ وَيَسْتَحْيُونَ نِسَآءَكُمْ ۚ وَفِى ذَٰلِكُم بَلَآءٌ مِّن رَّبِّكُمْ عَظِيمٌ ﴿٦﴾
مۇسا ئۆز قەۋمىگە ئۆز ۋاقتىدا: «اﷲ نىڭ سىلەرگە بەرگەن نېمىتىنى ئەسلەڭلار. اﷲ ئۆز ۋاقتىدا سىلەرنى پىرئەۋن قەۋمىدىن قۇتقۇزدى، ئۇلار سىلەرنى قاتتىق قىينايتتى، ئوغۇللىرىڭلارنى بوغۇزلايتتى، خوتۇن كىشىلىرىڭلارنى (خىزمەت قىلدۇرۇشقا) قالدۇراتتى. بۇنىڭدا پەرۋەردىگارىڭلار تەرىپىدىن (سىلەرگە) چوڭ سىناق بار ئىدى» دېدى
وَإِذْ تَأَذَّنَ رَبُّكُمْ لَئِن شَكَرْتُمْ لَأَزِيدَنَّكُمْ ۖ وَلَئِن كَفَرْتُمْ إِنَّ عَذَابِى لَشَدِيدٌ ﴿٧﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭلار: «نېمىتىگە شۈكۈر قىلساڭلار، (ئۇنى) تېخىمۇ زىيادە قىلىمەن. ئەگەر (كۇفرانى نېمەت) قىلساڭلار، مېنىڭ ئازابىم، ئەلۋەتتە، بەكمۇ قاتتىق بولىدۇ» دەپ جاكالىدى
وَقَالَ مُوسَىٰٓ إِن تَكْفُرُوٓا۟ أَنتُمْ وَمَن فِى ٱلْأَرْضِ جَمِيعًا فَإِنَّ ٱللَّهَ لَغَنِىٌّ حَمِيدٌ ﴿٨﴾
مۇسا (ئۆز قەۋمىگە): «ئەگەر سىلەر ۋە پۈتۈن يەر يۈزىدىكى كىشىلەر ھەقىقەتنى رەت قىلغۇچى بولۇپ كەتسەڭلارمۇ (اﷲ قا قىلچە زىيان يەتكۈزەلمەيسىلەر)، چۈنكى اﷲ ئەلۋەتتە (ھەممىدىن) بىھاجەتتۇر، مەدھىيىلەشكە لايىقتۇر» دېدى
أَلَمْ يَأْتِكُمْ نَبَؤُا۟ ٱلَّذِينَ مِن قَبْلِكُمْ قَوْمِ نُوحٍ وَعَادٍ وَثَمُودَ ۛ وَٱلَّذِينَ مِنۢ بَعْدِهِمْ ۛ لَا يَعْلَمُهُمْ إِلَّا ٱللَّهُ ۚ جَآءَتْهُمْ رُسُلُهُم بِٱلْبَيِّنَـٰتِ فَرَدُّوٓا۟ أَيْدِيَهُمْ فِىٓ أَفْوَٰهِهِمْ وَقَالُوٓا۟ إِنَّا كَفَرْنَا بِمَآ أُرْسِلْتُم بِهِۦ وَإِنَّا لَفِى شَكٍّ مِّمَّا تَدْعُونَنَآ إِلَيْهِ مُرِيبٍ ﴿٩﴾
سىلەردىن ئىلگىرى ئۆتكەن نۇھ، ئاد ۋە سەمۇد قەۋملىرىنىڭ ۋە ئۇلاردىن كېيىن كەلگەن، (سانىنى) پەقەت اﷲ بىلىدىغان ئۈممەتلەرنىڭ خەۋىرى سىلەرگە يەتمىدىمۇ؟ ئۇلارغا خۇدانىڭ ئەلچىسىلىرى مۆجىزىلەر بىلەن كەلدى، ئۇلار قوللىرى بىلەن خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ ئېغىزلىرىغا ئىشارەت قىلىپ (يەنى ئۇلارنىڭ سۈكۈت قىلىشىنى كۆرسىتىپ): «بىز سىلەرنىڭ خۇدانىڭ ئەلچىسىلىكىڭلارنى ئىشەنچلىك ھالدا ئىنكار قىلىمىز، بىز سىلە دەۋەت قىلىۋاتقان ئىشقا ئىشەنچلىك ھالدا زور گۇماندىمىز» دېدى
۞ قَالَتْ رُسُلُهُمْ أَفِى ٱللَّهِ شَكٌّ فَاطِرِ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۖ يَدْعُوكُمْ لِيَغْفِرَ لَكُم مِّن ذُنُوبِكُمْ وَيُؤَخِّرَكُمْ إِلَىٰٓ أَجَلٍ مُّسَمًّى ۚ قَالُوٓا۟ إِنْ أَنتُمْ إِلَّا بَشَرٌ مِّثْلُنَا تُرِيدُونَ أَن تَصُدُّونَا عَمَّا كَانَ يَعْبُدُ ءَابَآؤُنَا فَأْتُونَا بِسُلْطَـٰنٍ مُّبِينٍ ﴿١٠﴾
خۇدانىڭ ئەلچىسىلىرى: «ئاسمانلارنى ۋە زېمىننى ياراتقۇچى اﷲ نىڭ (بىرلىكىدە) شەك بارمۇ؟ اﷲ گۇناھلىرىڭلارنى مەغپىرەت قىلىش ئۈچۈن، سىلەرنى (ئىمانغا) دەۋەت قىلىدۇ، سىلەرنى (جازالاشنى) مەلۇم مۇددەتكىچە (يەنى ئەجىلىڭلار يەتكۈچە) تەخىر قىلىدۇ» دېدى. ئۇلار: «سىلەر پەقەت بىزگە ئوخشاش ئىنسان، ئاتا - بوۋىلىرىمىز ئىبادەت قىلىپ كېلىۋاتقان (بۇتلار) غا ئىبادەت قىلىشىمىزدىن توسماقچى بولىسىلەر، (راستلىقىڭلار ئۈچۈن) بىزگە ئېنىق پاكىت كۆرسىتىڭلار» دېدى
قَالَتْ لَهُمْ رُسُلُهُمْ إِن نَّحْنُ إِلَّا بَشَرٌ مِّثْلُكُمْ وَلَـٰكِنَّ ٱللَّهَ يَمُنُّ عَلَىٰ مَن يَشَآءُ مِنْ عِبَادِهِۦ ۖ وَمَا كَانَ لَنَآ أَن نَّأْتِيَكُم بِسُلْطَـٰنٍ إِلَّا بِإِذْنِ ٱللَّهِ ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١١﴾
خۇدانىڭ ئەلچىسىلىرى ئۇلارغا: «بىز پەقەت سىلەرگە ئوخشاش ئىنسانمىز، لېكىن اﷲ بەندىلىرىدىن خالىغان كىشىلەرگە (خۇدانىڭ ئەلچىسىلىكىنى) مەرھەمەت قىلىدۇ، اﷲ نىڭ ئىزنىسىز سىلەرگە ھېچقانداق پاكىت كۆرسىتەلمەيمىز، مۆمىنلەر (ھەممە ئىشتا) يالغۇز بىر اﷲ غىلا تەۋەككۈل قىلسۇن (يەنى ھەممە ئىشلىرىنى اﷲ قا تاپشۇرسۇن)
وَمَا لَنَآ أَلَّا نَتَوَكَّلَ عَلَى ٱللَّهِ وَقَدْ هَدَىٰنَا سُبُلَنَا ۚ وَلَنَصْبِرَنَّ عَلَىٰ مَآ ءَاذَيْتُمُونَا ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُتَوَكِّلُونَ ﴿١٢﴾
اﷲ بىزنى (بىزنى تونۇيدىغان) يوللىرىمىزغا يېتەكلىگەن تۇرسا، نېمىشقا اﷲ قا تەۋەككۈل قىلمايلى؟ بىزگە يەتكۈزگەن ئەزىيەتلىرىڭلارغا، ئەلۋەتتە، سەۋر قىلىمىز، تەۋەككۈل قىلغۇچىلار پەقەت اﷲ قا تەۋەككۈل قىلسۇن» دېدى
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لِرُسُلِهِمْ لَنُخْرِجَنَّكُم مِّنْ أَرْضِنَآ أَوْ لَتَعُودُنَّ فِى مِلَّتِنَا ۖ فَأَوْحَىٰٓ إِلَيْهِمْ رَبُّهُمْ لَنُهْلِكَنَّ ٱلظَّـٰلِمِينَ ﴿١٣﴾
ھەقىقەتنى رەت قىلغۇچىلار خۇدانىڭ ئەلچىسىلىرىگە: «سىلەرنى زېمىندىن چوقۇم ھەيدەپ چىقىرىمىز، ياكى چوقۇم بىزنىڭ دىنىمىزغا قايتىشىڭلار كېرەك» دېدى. ئۇلارغا پەرۋەردىگارى (مۇنداق) ۋەھيى قىلدى: «زالىملارنى چوقۇم ھالاك قىلىمىز
وَلَنُسْكِنَنَّكُمُ ٱلْأَرْضَ مِنۢ بَعْدِهِمْ ۚ ذَٰلِكَ لِمَنْ خَافَ مَقَامِى وَخَافَ وَعِيدِ ﴿١٤﴾
ئۇلارنى ھالاك قىلغاندىن كېيىن زېمىندا چوقۇم سىلەرنى تۇرغۇزىمىز، بۇ (يەنى غەلىبە) مېنىڭ ئالدىمدا تۇرۇشتىن (يەنى سوراققا تارتىلىشتىن) ۋە ئازابىمدىن قورققانلار ئۈچۈندۇر»
وَٱسْتَفْتَحُوا۟ وَخَابَ كُلُّ جَبَّارٍ عَنِيدٍ ﴿١٥﴾
خۇدانىڭ ئەلچى-رەسۇللىرى اﷲ تىن مەدەت تىلىدى، ھەر بىر ئۇچىغا چىققان تەرسا، شەپقەتسىز (يەنى اﷲ قا ئىتائەت قىلىشقا گەدەنكەشلىك قىلىپ ھەقىقەتتىن باش تارتقۇچى) ھالاك بولدى
مِّن وَرَآئِهِۦ جَهَنَّمُ وَيُسْقَىٰ مِن مَّآءٍ صَدِيدٍ ﴿١٦﴾
ئۇنىڭ ئالدىدا مەڭگۈلۈك ئوت تۇرىدۇ، مەڭگۈلۈك ئوتدە ئۇ يىرىڭ ئارىلاش سۇ بىلەن سۇغۇرۇلىدۇ
يَتَجَرَّعُهُۥ وَلَا يَكَادُ يُسِيغُهُۥ وَيَأْتِيهِ ٱلْمَوْتُ مِن كُلِّ مَكَانٍ وَمَا هُوَ بِمَيِّتٍ ۖ وَمِن وَرَآئِهِۦ عَذَابٌ غَلِيظٌ ﴿١٧﴾
(ئۇنىڭ ئاچچىقلىقىدىن) ئۇنى يۇتۇملاپ ئېچىپ تەسلىكتە يۇتىدۇ، ئۆلۈم ئۇنى قورىشىۋالىدۇ. لېكىن ئۇ ھەرگىز ئۆلمەيدۇ، شۇنىڭدىن كېيىن تېخىمۇ قاتتىق ئازابقا دۇچار بولىدۇ
مَّثَلُ ٱلَّذِينَ كَفَرُوا۟ بِرَبِّهِمْ ۖ أَعْمَـٰلُهُمْ كَرَمَادٍ ٱشْتَدَّتْ بِهِ ٱلرِّيحُ فِى يَوْمٍ عَاصِفٍ ۖ لَّا يَقْدِرُونَ مِمَّا كَسَبُوا۟ عَلَىٰ شَىْءٍ ۚ ذَٰلِكَ هُوَ ٱلضَّلَـٰلُ ٱلْبَعِيدُ ﴿١٨﴾
پەرۋەردىگارنى ئىنكار قىلغانلارنىڭ قىلغان (ياخشى) ئەمەللىرى بورانلىق كۈندە شامال ئۇچۇرۇپ كەتكەن بىر دۆۋە كۈلگە ئوخشىمايدۇ، ھەقىقەتنى رەت قىلغۇچىلار (بۇ دۇنيادا) قىلغان ياخشى ئەمەللىرى ئۈچۈن ئازراقمۇ شاۋابقا ئېرىشەلمەيدۇ، بۇ چۇڭقۇر ئېزىشتۇر (يەنى چوڭ زىياندۇر)
أَلَمْ تَرَ أَنَّ ٱللَّهَ خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ ۚ إِن يَشَأْ يُذْهِبْكُمْ وَيَأْتِ بِخَلْقٍ جَدِيدٍ ﴿١٩﴾
ئاسمانلارنى ۋە زېمىننى اﷲ نىڭ ھەق ئاساستا ياراتقانلىقىنى (يەنى بىكار ياراتمىغانلىقىنى) كۆرمىدىڭمۇ؟ ئەگەر اﷲ خالىسا، (ئى ئىنسانلار!) سىلەرنى يوق قىلىۋېتىپ (ئورنۇڭلارغا سىلەردىن ياخشى) يېڭى ئادەملەرنى يارىتىدۇ
وَمَا ذَٰلِكَ عَلَى ٱللَّهِ بِعَزِيزٍ ﴿٢٠﴾
بۇ اﷲ قا قىيىن ئەمەس
وَبَرَزُوا۟ لِلَّهِ جَمِيعًا فَقَالَ ٱلضُّعَفَـٰٓؤُا۟ لِلَّذِينَ ٱسْتَكْبَرُوٓا۟ إِنَّا كُنَّا لَكُمْ تَبَعًا فَهَلْ أَنتُم مُّغْنُونَ عَنَّا مِنْ عَذَابِ ٱللَّهِ مِن شَىْءٍ ۚ قَالُوا۟ لَوْ هَدَىٰنَا ٱللَّهُ لَهَدَيْنَـٰكُمْ ۖ سَوَآءٌ عَلَيْنَآ أَجَزِعْنَآ أَمْ صَبَرْنَا مَا لَنَا مِن مَّحِيصٍ ﴿٢١﴾
ئۇلار (يەنى خالايىقلار) نىڭ ھەممىسى (قىيامەت كۈنى قەبرىلىرىدىن) چىقىپ، اﷲ قا مۇلاقات بولىدۇ، ئاجىزلىرى (يەنى ئەگەشكۈچىلىرى) مۇتەكەببىرلىرىگە (يەنى ئەگەشتۈرگۈچىلىرىگە): «شۈبھىسىزكى، بىز سىلەرنىڭ ئەگەشكۈچىلىرىڭلار ئىدۇق، سىلەر بىزگە بولۇۋاتقان اﷲ نىڭ ئازابىدىن ئازراق دەپئى قىلالامسىلەر؟» دەيدۇ. ئۇلار: «اﷲ بىزنى ھىدايەت قىلغان بولسا ئىدى، بىز ئەلۋەتتە سىلەرنى ھىدايەت قىلاتتۇق، بىز مەيلى نالە - پەرياد قىلايلى، مەيلى سەۋر قىلايلى، بەربىر ئوخشاش، بىزگە ھېچقانداق قاچىدىغان جاي يوق» دەيدۇ
وَقَالَ ٱلشَّيْطَـٰنُ لَمَّا قُضِىَ ٱلْأَمْرُ إِنَّ ٱللَّهَ وَعَدَكُمْ وَعْدَ ٱلْحَقِّ وَوَعَدتُّكُمْ فَأَخْلَفْتُكُمْ ۖ وَمَا كَانَ لِىَ عَلَيْكُم مِّن سُلْطَـٰنٍ إِلَّآ أَن دَعَوْتُكُمْ فَٱسْتَجَبْتُمْ لِى ۖ فَلَا تَلُومُونِى وَلُومُوٓا۟ أَنفُسَكُم ۖ مَّآ أَنَا۠ بِمُصْرِخِكُمْ وَمَآ أَنتُم بِمُصْرِخِىَّ ۖ إِنِّى كَفَرْتُ بِمَآ أَشْرَكْتُمُونِ مِن قَبْلُ ۗ إِنَّ ٱلظَّـٰلِمِينَ لَهُمْ عَذَابٌ أَلِيمٌ ﴿٢٢﴾
ئىش پۈتكەندە (يەنى ھېساب تۈگەپ، مەڭگۈلۈك باغلارىلەر بىلەن دوزىخىلار ئايرىلىپ بولغاندا)، شەيتان (ئازدۇرغۇچى): «اﷲ ئىشەنچلىك ھالدا سىلەرگە (ئىتائەت قىلغۇچىنى مۇكاپاتلاش، ئاسىيلىق قىلغۇچىنى جازالاشتىن ئىبارەت) راست ۋەدىنى قىلغان ئىدى، (ۋەدىسىگە ۋاپا قىلدى). مەن سىلەرگە (ئۆلگەندىن كېيىن تىرىلىش ساۋاب، جازا دېگەنلەر يوق، دەپ يالغان) ۋەدە قىلغان ئىدىم، (ۋەدەمگە) خىلاپلىق قىلدىم، سىلەرگە مېنىڭ (كۇفرىغا، گۇناھقا زورلىغۇدەك) ھۆكۈمرانلىقىم بولغىنى يوق، مەن سىلەرنى پەقەت (گۇمراھلىققا) دەۋەت قىلدىم، سىلەر (دەۋىتىمنى) قوبۇل قىلدىڭلار، شۇنىڭ ئۈچۈن مېنى ئەيىبلىمەڭلار، ئۆزۈڭلارنى ئەيىبلەڭلار، مەن سىلەرگە ياردەم بېرىپ سىلەرنى قۇتقۇزالمايمەن، سىلەرمۇ ماڭا ياردەم بېرىپ مېنى قۇتقۇزالمايسىلەر، مەن سىلەرنىڭ بۇرۇن مېنى اﷲ قا (ئىبادەتتە) شېرىك قىلغانلىقىڭلارنى ئېتىراپ قىلمايمەن، شۈبھىسىزكى، زالىملار چوقۇم قاتتىق ئازابقا دۇچار بولىدۇ» دەيدۇ
وَأُدْخِلَ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ جَنَّـٰتٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا بِإِذْنِ رَبِّهِمْ ۖ تَحِيَّتُهُمْ فِيهَا سَلَـٰمٌ ﴿٢٣﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلار، پەرۋەردىگارىنىڭ ئىزنى بىلەن، ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان مەڭگۈلۈك باغلارلەرگە كىرىدۇ، ئۇ مەڭگۈلۈك باغلارلەردە مەڭگۈ قالىدۇ، (خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارنى ھۆرمەتلەپ) سىلەرگە ئامانلىق بولسۇن، دەپ سالام بېرىدۇ
أَلَمْ تَرَ كَيْفَ ضَرَبَ ٱللَّهُ مَثَلًا كَلِمَةً طَيِّبَةً كَشَجَرَةٍ طَيِّبَةٍ أَصْلُهَا ثَابِتٌ وَفَرْعُهَا فِى ٱلسَّمَآءِ ﴿٢٤﴾
اﷲ نىڭ مۇنداق بىر تەمسىل كۆرسەتكەنلىكىنى كۆرمىدىڭمۇ. كەلىمە تەييىبە (يەنى ياخشى سۆز، ئىمان كەلىمىسى) يىلتىزى يەرنىڭ ئاستىدا بولغان، شېخى ئاسمانغا تاقاشقان
تُؤْتِىٓ أُكُلَهَا كُلَّ حِينٍۭ بِإِذْنِ رَبِّهَا ۗ وَيَضْرِبُ ٱللَّهُ ٱلْأَمْثَالَ لِلنَّاسِ لَعَلَّهُمْ يَتَذَكَّرُونَ ﴿٢٥﴾
پەرۋەردىگارىنىڭ ئىزنى بىلەن ۋاقتى - ۋاقتىدا مېۋە بېرىپ تۇرىدىغان ئېسىل دەرەخقە ئوخشايدۇ. اﷲ كىشىلەرگە ۋەز - نەسىھەت ئالسۇن دەپ، ئۇلارغا نۇرغۇن تەمسىللەرنى كەلتۈرىدۇ
وَمَثَلُ كَلِمَةٍ خَبِيثَةٍ كَشَجَرَةٍ خَبِيثَةٍ ٱجْتُثَّتْ مِن فَوْقِ ٱلْأَرْضِ مَا لَهَا مِن قَرَارٍ ﴿٢٦﴾
يامان سۆز (يەنى كۇفرى كەلىمىسى) زېمىندىن قومۇرۇپ تاشلانغان، ھېچقانداق قارارى يوق ناچار دەرەخقە ئوخشايدۇ
يُثَبِّتُ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ بِٱلْقَوْلِ ٱلثَّابِتِ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْـَٔاخِرَةِ ۖ وَيُضِلُّ ٱللَّهُ ٱلظَّـٰلِمِينَ ۚ وَيَفْعَلُ ٱللَّهُ مَا يَشَآءُ ﴿٢٧﴾
اﷲ مۆمىنلەرنى مۇستەھكەم ئىمان بىلەن دۇنيادا ۋە ئاخىرەتتە مەھكەم تۇرغۇزىدۇ. اﷲ زالىملا - نى گۇمراھ قىلىدۇ، اﷲ خالىغىنىنى قىلىدۇ
۞ أَلَمْ تَرَ إِلَى ٱلَّذِينَ بَدَّلُوا۟ نِعْمَتَ ٱللَّهِ كُفْرًا وَأَحَلُّوا۟ قَوْمَهُمْ دَارَ ٱلْبَوَارِ ﴿٢٨﴾
اﷲ نىڭ بەرگەن نېمىتىگە كۇفرىلىق قىلغان ۋە ئۆز قەۋمىنى (ئازدۇرۇش بىلەن) ھالاكەت مەۋقەسىگە چۈشۈرۈپ قويغانلارنى كۆرمىدىڭمۇ؟
جَهَنَّمَ يَصْلَوْنَهَا ۖ وَبِئْسَ ٱلْقَرَارُ ﴿٢٩﴾
(ھالاكەت مەۋقەسى) مەڭگۈلۈك ئوتدۇركى، ئۇلار مەڭگۈلۈك ئوتگە كىرىدۇ، ئۇ نېمىدېگەن يامان جاي!
وَجَعَلُوا۟ لِلَّهِ أَندَادًا لِّيُضِلُّوا۟ عَن سَبِيلِهِۦ ۗ قُلْ تَمَتَّعُوا۟ فَإِنَّ مَصِيرَكُمْ إِلَى ٱلنَّارِ ﴿٣٠﴾
ئۇلار (كىشىلەرنى) اﷲ نىڭ يولىدىن (يەنى ئىسلام دىنىدىن) ئازدۇرۇش ئۈچۈن، اﷲ قا نۇرغۇن بۇتلارنى شېرىك قىلىۋالدى. ئۇلارغا: «(دۇنيانىڭ نېمەتلىرىدىن) بەھرىمەن بولدۇڭلار، شۈبھىسىزكى، سىلەرنىڭ بارىدىغان جايىڭلار دوزاختۇر» دېگىن
قُل لِّعِبَادِىَ ٱلَّذِينَ ءَامَنُوا۟ يُقِيمُوا۟ ٱلصَّلَوٰةَ وَيُنفِقُوا۟ مِمَّا رَزَقْنَـٰهُمْ سِرًّا وَعَلَانِيَةً مِّن قَبْلِ أَن يَأْتِىَ يَوْمٌ لَّا بَيْعٌ فِيهِ وَلَا خِلَـٰلٌ ﴿٣١﴾
مۆمىن بەندىلىرىمگە ئېيتقىنكى، ئۇلار نامازنى ئادا قىلسۇن، ئالدى - ساتتىمۇ، دوستلۇقمۇ بولمايدىغان كۈن (يەنى قىيامەت كۈنى) كېلىشتىن بۇرۇن، بىز ئۇلارغا بەرگەن مال - مۈلۈكتىن يوشۇرۇن ۋە ئاشكارا يوسۇندا سەدىقە قىلسۇن
ٱللَّهُ ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَأَنزَلَ مِنَ ٱلسَّمَآءِ مَآءً فَأَخْرَجَ بِهِۦ مِنَ ٱلثَّمَرَٰتِ رِزْقًا لَّكُمْ ۖ وَسَخَّرَ لَكُمُ ٱلْفُلْكَ لِتَجْرِىَ فِى ٱلْبَحْرِ بِأَمْرِهِۦ ۖ وَسَخَّرَ لَكُمُ ٱلْأَنْهَـٰرَ ﴿٣٢﴾
اﷲ ئاسمانلارنى ۋە زېمىننى ياراتتى، بۇلۇتتىن يامغۇر ياغدۇرۇپ بەردى، يامغۇر سۈيى بىلەن سىلەرگە رىزىق قىلىپ نۇرغۇن مېۋىلەرنى ئۆستۈرۈپ بەردى، اﷲ سىلەرگە ئۆز ئەمرى بويىچە دېڭىزدا قاتنايدىغان كېمىلەرنى بويسۇندۇرۇپ بەردى، سىلەرگە دەريالارنى بويسۇندۇرۇپ بەردى
وَسَخَّرَ لَكُمُ ٱلشَّمْسَ وَٱلْقَمَرَ دَآئِبَيْنِ ۖ وَسَخَّرَ لَكُمُ ٱلَّيْلَ وَٱلنَّهَارَ ﴿٣٣﴾
سىلەرگە ئاي بىلەن كۈننى تەرتىپلىك دەۋر قىلىپ تۇرىدىغان قىلىپ بويسۇندۇرۇپ بەردى، سىلەرگە كېچە بىلەن كۈندۈزنى بويسۇندۇرۇپ بەردى
وَءَاتَىٰكُم مِّن كُلِّ مَا سَأَلْتُمُوهُ ۚ وَإِن تَعُدُّوا۟ نِعْمَتَ ٱللَّهِ لَا تُحْصُوهَآ ۗ إِنَّ ٱلْإِنسَـٰنَ لَظَلُومٌ كَفَّارٌ ﴿٣٤﴾
اﷲ سىلەرگە سورىغان نەرسەڭلارنىڭ ھەممىسىنى بەردى، سىلەر اﷲ نىڭ نېمەتلىرىنى ساناپ تۈگىتەلمەيسىلەر، ھەقىقەتنى رەت قىلغۇچى ئادەم، شەك - شۈبھىسىزكى، زۇلۇم قىلغۇچىدۇر، (اﷲ نىڭ نېمەتلىرىگە) كۇفرىلىق قىلغۇچىدۇر
وَإِذْ قَالَ إِبْرَٰهِيمُ رَبِّ ٱجْعَلْ هَـٰذَا ٱلْبَلَدَ ءَامِنًا وَٱجْنُبْنِى وَبَنِىَّ أَن نَّعْبُدَ ٱلْأَصْنَامَ ﴿٣٥﴾
ئۆز ۋاقتىدا ئىبراھىم ئېيتتى: «پەرۋەردىگارىم! بۇ جاينى (يەنى مەككىنى) تىنچ (جاي) قىلىپ بەرگىن، مېنى ۋە مېنىڭ ئەۋلادلىرىمنى بۇتلارغا چوقۇنۇشتىن يىراق قىلغىن
رَبِّ إِنَّهُنَّ أَضْلَلْنَ كَثِيرًا مِّنَ ٱلنَّاسِ ۖ فَمَن تَبِعَنِى فَإِنَّهُۥ مِنِّى ۖ وَمَنْ عَصَانِى فَإِنَّكَ غَفُورٌ رَّحِيمٌ ﴿٣٦﴾
پەرۋەردىگارىم! ئۇلار (يەنى بۇتلار) نۇرغۇن كىشىلەرنى ئازدۇردى، كىمكى ماڭا ئەگەشكەن ئىكەن، ئۇ مېنىڭ دىنىمدىدۇر، كىمكى ماڭا ئاسىيلىق قىلغان ئىكەن، سەن ئەلۋەتتە (ئۇنىڭغا) مەغپىرەت قىلىشقا ۋە مەرھەمەت قىلىشقا (قادىرسەن)
رَّبَّنَآ إِنِّىٓ أَسْكَنتُ مِن ذُرِّيَّتِى بِوَادٍ غَيْرِ ذِى زَرْعٍ عِندَ بَيْتِكَ ٱلْمُحَرَّمِ رَبَّنَا لِيُقِيمُوا۟ ٱلصَّلَوٰةَ فَٱجْعَلْ أَفْـِٔدَةً مِّنَ ٱلنَّاسِ تَهْوِىٓ إِلَيْهِمْ وَٱرْزُقْهُم مِّنَ ٱلثَّمَرَٰتِ لَعَلَّهُمْ يَشْكُرُونَ ﴿٣٧﴾
پەرۋەردىگارىمىز! ئەۋلادىمنىڭ بىر قىسىمىنى (يەنى بالام ئىسمائىل بىلەن ئايالىم ھاجەرنى) ناماز ئوقۇسۇن (يەنى ساڭا ئىبادەت قىلسۇن دەپ) سېنىڭ ھۆرمەتلىك ئۆيۈڭنىڭ قېشىدىكى ئېكىنسىز بىر ۋادىغا (يەنى مەككىگە) ئورۇنلاشتۇردۇم، پەرۋەردىگارىمىز! بىر قىسىم كىشىلەرنىڭ دىللىرىنى ئۇلارغا مايىل قىلغىن، ئۇلارنى شۈكۈر قىلسۇن دەپ تۈرلۈك مېۋىلەرنى ئۇلارغا رىزىق قىلىپ بەرگىن
رَبَّنَآ إِنَّكَ تَعْلَمُ مَا نُخْفِى وَمَا نُعْلِنُ ۗ وَمَا يَخْفَىٰ عَلَى ٱللَّهِ مِن شَىْءٍ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ﴿٣٨﴾
پەرۋەردىگارىمىز! سەن بىزنىڭ يوشۇرۇن ۋە ئاشكارا قىلغانلىرىمىزنى بىلىپ تۇرىسەن، ئاسمان - زېمىندىكى ھېچ نەرسە اﷲ قا مەخپىي ئەمەس
ٱلْحَمْدُ لِلَّهِ ٱلَّذِى وَهَبَ لِى عَلَى ٱلْكِبَرِ إِسْمَـٰعِيلَ وَإِسْحَـٰقَ ۚ إِنَّ رَبِّى لَسَمِيعُ ٱلدُّعَآءِ ﴿٣٩﴾
جىمى ھەمدۇسانا ياشانغىنىمدا ماڭا ئىسمائىل بىلەن ئىسھاقنى بەرگەن اﷲ قا خاستۇر! پەرۋەردىگارىم! ئىشەنچلىك ھالدا دۇئانى ئاڭلاپ تۇرغۇچىدۇر (يەنى دۇئا قىلغۇچىنىڭ دۇئاسىنى ئىجابەت قىلغۇچىدۇر)
رَبِّ ٱجْعَلْنِى مُقِيمَ ٱلصَّلَوٰةِ وَمِن ذُرِّيَّتِى ۚ رَبَّنَا وَتَقَبَّلْ دُعَآءِ ﴿٤٠﴾
پەرۋەردىگارىم! مېنى ۋە بىر قىسىم ئەۋلادىمنى نامازنى ئادا قىلغۇچى قىلغىن، پەرۋەردىگارىمىز! دۇئايىمنى قوبۇل قىلغىن
رَبَّنَا ٱغْفِرْ لِى وَلِوَٰلِدَىَّ وَلِلْمُؤْمِنِينَ يَوْمَ يَقُومُ ٱلْحِسَابُ ﴿٤١﴾
پەرۋەردىگارىمىز! ھېساب ئالىدىغان كۈندە (يەنى قىيامەت كۈنىدە) ماڭا، ئاتا - ئانامغا ۋە مۆمىنلەرگە مەغپىرەت قىلغىن»
وَلَا تَحْسَبَنَّ ٱللَّهَ غَـٰفِلًا عَمَّا يَعْمَلُ ٱلظَّـٰلِمُونَ ۚ إِنَّمَا يُؤَخِّرُهُمْ لِيَوْمٍ تَشْخَصُ فِيهِ ٱلْأَبْصَـٰرُ ﴿٤٢﴾
زالىملارنىڭ قىلمىشلىرىدىن اﷲ نى بىخەۋەر دەپ ئويلىمىغىن، اﷲ ئۇلارنى جازالاشنى (شۇ كۈنىنىڭ دەھشىتىدىن) كۆزلەر چەكچىيىپ قالىدىغان كۈنىگىچە (يەنى قىيامەت كۈنىگىچە) كېچىكتۈرىدۇ
مُهْطِعِينَ مُقْنِعِى رُءُوسِهِمْ لَا يَرْتَدُّ إِلَيْهِمْ طَرْفُهُمْ ۖ وَأَفْـِٔدَتُهُمْ هَوَآءٌ ﴿٤٣﴾
(بۇ كۈندە) ئۇلار باشلىرىنى كۆتۈرۈپ، ئالغا قاراپ چاپىدۇ، كۆزلىرى ئېچىلىپ قالىدۇ، (تېڭىرقىغانلىقتىن) دىللىرى (ئەقىل - ئىدراكتىن) خالىي بولۇپ قالىدۇ
وَأَنذِرِ ٱلنَّاسَ يَوْمَ يَأْتِيهِمُ ٱلْعَذَابُ فَيَقُولُ ٱلَّذِينَ ظَلَمُوا۟ رَبَّنَآ أَخِّرْنَآ إِلَىٰٓ أَجَلٍ قَرِيبٍ نُّجِبْ دَعْوَتَكَ وَنَتَّبِعِ ٱلرُّسُلَ ۗ أَوَلَمْ تَكُونُوٓا۟ أَقْسَمْتُم مِّن قَبْلُ مَا لَكُم مِّن زَوَالٍ ﴿٤٤﴾
(ئى مۇھەممەد!) كىشىلەرنى (يەنى كۇففارلارنى) ئازاب كېلىدىغان كۈن (يەنى قىيامەت كۈنى) دىن ئاگاھلاندۇرغىن، (بۇ كۈندە) زالىملار: «پەرۋەردىگارىمىز! بىزگە قىسقىغىنا ۋاقىت مۆھلەت بەرگىن (يەنى دۇنياغا قايتۇرغىن)، دەۋىتىڭنى قوبۇل قىلايلى، خۇدانىڭ ئەلچى-رەسۇللىرىگە ئەگىشەيلى» دەيدۇ. (ئۇلارغا) سىلەر ئىلگىرى، بىز ھەرگىز يوقالمايمىز، دەپ قەسەم ئىچمىگەنمىدىڭلار؟ (دېيىلىدۇ)
وَسَكَنتُمْ فِى مَسَـٰكِنِ ٱلَّذِينَ ظَلَمُوٓا۟ أَنفُسَهُمْ وَتَبَيَّنَ لَكُمْ كَيْفَ فَعَلْنَا بِهِمْ وَضَرَبْنَا لَكُمُ ٱلْأَمْثَالَ ﴿٤٥﴾
ئۆزلىرىگە زۇلۇم قىلغانلارنى (ھالاك قىلغىنىمىزدىن كېيىن) سىلەر ئۇلارنىڭ جايلىرىدا تۇردۇڭلار (ئۇلارنىڭ جايلىرىدىن ئىبرەت ئالساڭلارچۇ)، ئۇلارنى قانداق جازالىغانلىقىمىز سىلەرگە ئېنىق مەلۇم بولدى. سىلەرگە نۇرغۇن مىساللارنى كەلتۈردۇق (يەنى خۇدانىڭ ئەلچى-رەسۇللىرىگە چىنپۈتمىگەنلەرنىڭ ئەھۋالىنى بايان قىلدۇق)
وَقَدْ مَكَرُوا۟ مَكْرَهُمْ وَعِندَ ٱللَّهِ مَكْرُهُمْ وَإِن كَانَ مَكْرُهُمْ لِتَزُولَ مِنْهُ ٱلْجِبَالُ ﴿٤٦﴾
ئۇلار ئىشەنچلىك ھالدا (خۇدانىڭ ئەلچى-رەسۇللىرىگە ۋە مۆمىنلەرگە) مىكىر ئىشلەتتى. ئۇلارنىڭ مىكرى تاغنى ئۆرۈۋەتكۈدەك (چوڭ) بولسىمۇ، اﷲ ئۇلارنىڭ مىكرىنى بەربات قىلىدۇ
فَلَا تَحْسَبَنَّ ٱللَّهَ مُخْلِفَ وَعْدِهِۦ رُسُلَهُۥٓ ۗ إِنَّ ٱللَّهَ عَزِيزٌ ذُو ٱنتِقَامٍ ﴿٤٧﴾
اﷲ نى خۇدانىڭ ئەلچىسىلىرىگە قىلغان ۋەدىسىگە خىلاپلىق قىلىدۇ دەپ ئويلىمىغىن، اﷲ ئىشەنچلىك ھالدا غالىبتۇر، (دوستلىرى ئۈچۈن دۈشمەنلىرىدىن) ئىنتىقام ئالغۇچىدۇر
يَوْمَ تُبَدَّلُ ٱلْأَرْضُ غَيْرَ ٱلْأَرْضِ وَٱلسَّمَـٰوَٰتُ ۖ وَبَرَزُوا۟ لِلَّهِ ٱلْوَٰحِدِ ٱلْقَهَّارِ ﴿٤٨﴾
ئۇ كۈندە زېمىنمۇ باشقا بىر زېمىنغا، ئاسمانلارمۇ باشقا بىر ئاسمانلارغا ئايلىنىدۇ، ئۇلار (يەنى پۈتۈن خالايىق قەبرىلىرىدىن چىقىپ) يېگانە، قۇدرەتلىك اﷲ نىڭ دەرگاھىدا (يەنى مەھشەرگاھدا) ھازىر بولىدۇ
وَتَرَى ٱلْمُجْرِمِينَ يَوْمَئِذٍ مُّقَرَّنِينَ فِى ٱلْأَصْفَادِ ﴿٤٩﴾
(ئى مۇھەممەد!) ئۇ كۈندە گۇناھكارلار (يەنى كۇففارلار) نى زەنجىرلەر بىلەن بىر - بىرىگە چېتىلىپ باغلانغان ھالدا كۆرۈسەن
سَرَابِيلُهُم مِّن قَطِرَانٍ وَتَغْشَىٰ وُجُوهَهُمُ ٱلنَّارُ ﴿٥٠﴾
ئۇلارنىڭ كۆڭلىكى قارامايدىن بولىدۇ، ئوت ئۇلارنىڭ يۈزلىرىنى ئورىۋالىدۇ
لِيَجْزِىَ ٱللَّهُ كُلَّ نَفْسٍ مَّا كَسَبَتْ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ﴿٥١﴾
اﷲ ھەر كىشىنىڭ قىلمىشىغا قاراپ شۇنداق جازا بېرىدۇ، اﷲ ئەلۋەتتە تېز ھېساب ئالغۇچىدۇر
هَـٰذَا بَلَـٰغٌ لِّلنَّاسِ وَلِيُنذَرُوا۟ بِهِۦ وَلِيَعْلَمُوٓا۟ أَنَّمَا هُوَ إِلَـٰهٌ وَٰحِدٌ وَلِيَذَّكَّرَ أُو۟لُوا۟ ٱلْأَلْبَـٰبِ ﴿٥٢﴾
كىشىلەرنىڭ ئاگاھلاندۇرۇلۇشى، اﷲ نىڭ بىر ئىكەنلىكىنى بىلىشلىرى، ئەقىل ئىگىلىرىنىڭ ۋەز - نەسىھەت ئېلىشلىرى ئۈچۈن، بۇ (قۇرئان) تەبلىغدۇر (يەنى كىشىلەرگە تەبلىغ قىلىش ئۈچۈن نازىل قىلىندى)
