بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
تَنزِيلُ ٱلْكِتَـٰبِ مِنَ ٱللَّهِ ٱلْعَزِيزِ ٱلْحَكِيمِ ﴿١﴾
بۇ مۇقەددەس كىتاب-ۋەھىي غالىب، ھېكمەت بىلەن ئىش قىلغۇچى اﷲ تەرىپىدىن نازىل قىلىنغاندۇر
إِنَّآ أَنزَلْنَآ إِلَيْكَ ٱلْكِتَـٰبَ بِٱلْحَقِّ فَٱعْبُدِ ٱللَّهَ مُخْلِصًا لَّهُ ٱلدِّينَ ﴿٢﴾
ساڭا بىز (بۇ) مۇقەددەس كىتاب-ۋەھىينى ئىشەنچلىك ھالدا ھەق بىلەن (يەنى ھەقىقەتنى ئۆز ئىچىگە ئالغان ھالدا) نازىل قىلدۇق، دىنىڭنى شېرىكتىن (ۋە رىيادىن) ساپ قىلغىنىڭ ھالدا اﷲ قا ئىبادەت قىلغىن
أَلَا لِلَّهِ ٱلدِّينُ ٱلْخَالِصُ ۚ وَٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ مَا نَعْبُدُهُمْ إِلَّا لِيُقَرِّبُونَآ إِلَى ٱللَّهِ زُلْفَىٰٓ إِنَّ ٱللَّهَ يَحْكُمُ بَيْنَهُمْ فِى مَا هُمْ فِيهِ يَخْتَلِفُونَ ۗ إِنَّ ٱللَّهَ لَا يَهْدِى مَنْ هُوَ كَـٰذِبٌ كَفَّارٌ ﴿٣﴾
بىلىڭلاركى، (شېرىكتىن ۋە رىيادىن) ساپ دىن اﷲ قا خاستۇر، اﷲ نى قويۇپ بۇتلارنى ھىمايىچى قىلىۋالغانلار: «بىزنىڭ ئۇلارغا چوقۇنۇشىمىز پەقەت ئۇلارنىڭ بىزنى اﷲ قا يېقىنلاشتۇرۇشى ئۈچۈندۇر» (دەيدۇ). اﷲ ئىشەنچلىك ھالدا (قىيامەت كۈنى) ئۇلارنىڭ ئىختىلاپ قىلىشقان نەرسىلىرى ئۈستىدە ھۆكۈم چىقىرىدۇ، اﷲ ئىشەنچلىك ھالدا يالغانچىنى، (اﷲ نىڭ نېمەتلىرىگە) كۇفرىلىق قىلغۇچىنى ھىدايەت قىلمايدۇ
لَّوْ أَرَادَ ٱللَّهُ أَن يَتَّخِذَ وَلَدًا لَّٱصْطَفَىٰ مِمَّا يَخْلُقُ مَا يَشَآءُ ۚ سُبْحَـٰنَهُۥ ۖ هُوَ ٱللَّهُ ٱلْوَٰحِدُ ٱلْقَهَّارُ ﴿٤﴾
ئەگەر اﷲ بالا تۇتۇشنى خالىسا ئىدى، ئەلۋەتتە، مەخلۇقات ئىچىدىن خالىغاننى ئىختىيار قىلاتتى، اﷲ (بالا تۇتۇشتىن) پاكتۇر، ئۇ غالىب بىر اﷲ دۇر
خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ ۖ يُكَوِّرُ ٱلَّيْلَ عَلَى ٱلنَّهَارِ وَيُكَوِّرُ ٱلنَّهَارَ عَلَى ٱلَّيْلِ ۖ وَسَخَّرَ ٱلشَّمْسَ وَٱلْقَمَرَ ۖ كُلٌّ يَجْرِى لِأَجَلٍ مُّسَمًّى ۗ أَلَا هُوَ ٱلْعَزِيزُ ٱلْغَفَّـٰرُ ﴿٥﴾
اﷲ ئاسمانلارنى ۋە زېمىننى ھەق ئاساسىدا ياراتتى. اﷲ كېچىنى كۈندۈزگە كىرگۈزىدۇ ۋە كۈندۈزنى كېچىگە كىرگۈزىدۇ، اﷲ كۈن بىلەن ئاينى (بەندىلەرنىڭ مەنپەئەتىگە) بويسۇندۇرۇپ بەردى، ئۇلارنىڭ ھەر بىرى مۇئەييەن ۋاقىتقىچە (يەنى قىيامەتكىچە ئۆز ئوقىدا) سەير قىلىدۇ، بىلىڭلاركى، ئۇ (يەنى اﷲ ئۆز ئىشىدا) غالىبتۇر. (بەندىلىرىنىڭ گۇناھلىرىنى تولىمۇ مەغپىرەت قىلغۇچىدۇر
خَلَقَكُم مِّن نَّفْسٍ وَٰحِدَةٍ ثُمَّ جَعَلَ مِنْهَا زَوْجَهَا وَأَنزَلَ لَكُم مِّنَ ٱلْأَنْعَـٰمِ ثَمَـٰنِيَةَ أَزْوَٰجٍ ۚ يَخْلُقُكُمْ فِى بُطُونِ أُمَّهَـٰتِكُمْ خَلْقًا مِّنۢ بَعْدِ خَلْقٍ فِى ظُلُمَـٰتٍ ثَلَـٰثٍ ۚ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ لَهُ ٱلْمُلْكُ ۖ لَآ إِلَـٰهَ إِلَّا هُوَ ۖ فَأَنَّىٰ تُصْرَفُونَ ﴿٦﴾
اﷲ سىلەرنى بىر ئادەمدىن ياراتتى، ئاندىن ئۇنىڭدىن ئۇنىڭ جۈپتىنى (يەنى ھەۋۋانى) ياراتتى ۋە ھايۋانلاردىن سەككىزنى سىلەر ئۈچۈن ياراتتى، اﷲ سىلەرنى ئاناڭلارنىڭ قارنىدا بىر ھالەتتىن يەنە بىر ھالەتكە تەرەققىي قىلدۇرۇپ (يەنى ئابىمەنىي، ئاندىن لەختە قان، ئاندىن پارچە گۆش باسقۇچلىرىغا بۆلۈپ)، ئۈچ قاراڭغۇلۇق (يەنى بالاھەمراھىنىڭ پەردىسى، بەچچىدان ۋە ئانىنىڭ قورسىقى) ئىچىدە يارىتىدۇ، ئەنە شۇ اﷲ سىلەرنىڭ پەرۋەردىگارىڭلاردۇر، پادىشاھلىق اﷲ قا خاستۇر، ئۇنىڭدىن باشقا ھېچ ئىلاھ يوقتۇر، قانداقمۇ (ئۇنىڭغا ئىبادەت قىلىشتىن ئۇنىڭدىن غەيرىيگە چوقۇنۇشقا) بۇرۇلۇپ كېتىسىلەر
إِن تَكْفُرُوا۟ فَإِنَّ ٱللَّهَ غَنِىٌّ عَنكُمْ ۖ وَلَا يَرْضَىٰ لِعِبَادِهِ ٱلْكُفْرَ ۖ وَإِن تَشْكُرُوا۟ يَرْضَهُ لَكُمْ ۗ وَلَا تَزِرُ وَازِرَةٌ وِزْرَ أُخْرَىٰ ۗ ثُمَّ إِلَىٰ رَبِّكُم مَّرْجِعُكُمْ فَيُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ۚ إِنَّهُۥ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ ﴿٧﴾
ئەگەر ھەقىقەتنى رەت قىلغۇچى بولساڭلار اﷲ سىلەردىن بىھاجەتتۇر، اﷲ بەندىلىرىنىڭ ھەقىقەتنى رەت قىلغۇچى بولۇشىغا رازى بولمايدۇ. ئەگەر سىلەر شۈكۈر قىلساڭلار، اﷲ سىلەرنىڭ قىلغان شۈكۈرۈڭلاردىن رازى بولىدۇ، بىر گۇناھكار ئادەم باشقا بىراۋنىڭ گۇناھىنى ئۈستىگە ئالمايدۇ (يەنى ھەر ئادەم قىلغان گۇناھىغا ئۆزى جاۋابكار بولىدۇ)، ئاندىن سىلەر پەرۋەردىگارىڭلارنىڭ دەرگاھىغا قايتىسىلەر، اﷲ قىلمىشىڭلارنى سىلەرگە ئېيتىپ بېرىدۇ، اﷲ ئىشەنچلىك ھالدا دىللاردىكى سىرلارنى تولۇق بىلگۈچىدۇر
۞ وَإِذَا مَسَّ ٱلْإِنسَـٰنَ ضُرٌّ دَعَا رَبَّهُۥ مُنِيبًا إِلَيْهِ ثُمَّ إِذَا خَوَّلَهُۥ نِعْمَةً مِّنْهُ نَسِىَ مَا كَانَ يَدْعُوٓا۟ إِلَيْهِ مِن قَبْلُ وَجَعَلَ لِلَّهِ أَندَادًا لِّيُضِلَّ عَن سَبِيلِهِۦ ۚ قُلْ تَمَتَّعْ بِكُفْرِكَ قَلِيلًا ۖ إِنَّكَ مِنْ أَصْحَـٰبِ ٱلنَّارِ ﴿٨﴾
ئىنسانغا بىرەر زىيان - زەخمەت يەتسە، پەرۋەردىگارىغا يۈزلەنگەن ھالدا ئۇنىڭغا دۇئا قىلىدۇ، ئاندىن اﷲ ئۆز تەرىپىدىن ئۇنىڭغا نېمەت ئاتا قىلغان (ئۇنىڭ بېشىغا كەلگەن كۈلپەتنى كۆتۈرۈۋەتكەن) چاغدا، اﷲ قا زىيان - زەخمەتنى كۆتۈرۈۋېتىش توغرۇلۇق ئىلگىرى قىلغان دۇئاسىنى ئۇنتۇپ كېتىدۇ، (باشقىلارنى) اﷲ نىڭ يولىدىن ئازدۇرۇش ئۈچۈن اﷲ قا شېرىكلەرنى كەلتۈرىدۇ، ئېيتقىنكى، «سەن كۇفرىڭ بىلەن تۇرۇپ، (بۇ پانىي دۇنيادىن) ئازغىنا (ۋاقىت) بەھرىمەن بولغىن، سەن چوقۇم دوزىخىلاردىندۇرسەن
أَمَّنْ هُوَ قَـٰنِتٌ ءَانَآءَ ٱلَّيْلِ سَاجِدًا وَقَآئِمًا يَحْذَرُ ٱلْـَٔاخِرَةَ وَيَرْجُوا۟ رَحْمَةَ رَبِّهِۦ ۗ قُلْ هَلْ يَسْتَوِى ٱلَّذِينَ يَعْلَمُونَ وَٱلَّذِينَ لَا يَعْلَمُونَ ۗ إِنَّمَا يَتَذَكَّرُ أُو۟لُوا۟ ٱلْأَلْبَـٰبِ ﴿٩﴾
(اﷲ قا شېرىك كەلتۈرگەن ئادەم ياخشىمۇ؟) ياكى ئاخىرەت (ئازابىدىن) قورقۇپ، پەرۋەردىگارىنىڭ رەھمىتىنى ئۈمىد قىلىپ كېچىنىڭ سائەتلىرىنى سەجدە قىلغان ۋە قىيامدا تۇرغان ھالدا ئىبادەت قىلىپ ئۆتكۈزگەن ئادەم (ياخشىمۇ؟)». ئېيتقىنكى، «بىلىدىغانلار بىلەن بىلمەيدىغانلار باراۋەر بولامدۇ؟ پەقەت (ساغلام) ئەقىل ئىگىلىرى ئىبرەت ئالىدۇ»
قُلْ يَـٰعِبَادِ ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ رَبَّكُمْ ۚ لِلَّذِينَ أَحْسَنُوا۟ فِى هَـٰذِهِ ٱلدُّنْيَا حَسَنَةٌ ۗ وَأَرْضُ ٱللَّهِ وَٰسِعَةٌ ۗ إِنَّمَا يُوَفَّى ٱلصَّـٰبِرُونَ أَجْرَهُم بِغَيْرِ حِسَابٍ ﴿١٠﴾
ئېيتقىنكى، «ئى (اﷲ نىڭ) مۆمىن بەندىلىرى! پەرۋەردىگارىڭلارغا تەقۋادارلىق قىلىڭلار، بۇ دۇنيادا ياخشى ئىش قىلغانلار (ئاخىرەتتە) ياخشىلىققا (يەنى مەڭگۈلۈك باغلاركە) ئېرىشىدۇ. اﷲ نىڭ يېرى كەڭدۇر (كۇففارلار ئارىسىدىن شۇ يەرگە ھىجرەت قىلىڭلار)، پەقەت سەۋر قىلغۇچىلارغا ئۇلارنىڭ ئەجرى ھېسابسىز بېرىلىدۇ»
قُلْ إِنِّىٓ أُمِرْتُ أَنْ أَعْبُدَ ٱللَّهَ مُخْلِصًا لَّهُ ٱلدِّينَ ﴿١١﴾
ئېيتقىنكى، «مەن دىننى (شېرىكتىن ۋە رىيادىن) ساپ قىلغان ھالدا اﷲ قا ئىبادەت قىلىشقا بۇيرۇلدۇم
وَأُمِرْتُ لِأَنْ أَكُونَ أَوَّلَ ٱلْمُسْلِمِينَ ﴿١٢﴾
ھەمدە (بۇ ئۈممەتنىڭ ئىچىدە) مۇسۇلمانلارنىڭ ئەۋۋىلى بولۇشقا بۇيرۇلدۇم»
قُلْ إِنِّىٓ أَخَافُ إِنْ عَصَيْتُ رَبِّى عَذَابَ يَوْمٍ عَظِيمٍ ﴿١٣﴾
ئېيتقىنكى، «ئەگەر مەن پەرۋەردىگارىمغا ئاسىيلىق قىلسام، شۈبھىسىزكى، بۈيۈك كۈننىڭ ئازابىدىن قورقىمەن (يەنى اﷲ نىڭ ئەمرىگە ئاسىيلىق قىلسام، قىيامەت كۈنى ئۇنىڭ مېنى مەڭگۈلۈك ئوت ئوتى بىلەن ئازابلىشىدىن قورقىمەن)»
قُلِ ٱللَّهَ أَعْبُدُ مُخْلِصًا لَّهُۥ دِينِى ﴿١٤﴾
ئېيتقىنكى، «دىنىمنى (شېرىكتىن ۋە رىيادىن) ساپ قىلغان ھالدا اﷲ قا ئىبادەت قىلىمەن
فَٱعْبُدُوا۟ مَا شِئْتُم مِّن دُونِهِۦ ۗ قُلْ إِنَّ ٱلْخَـٰسِرِينَ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ وَأَهْلِيهِمْ يَوْمَ ٱلْقِيَـٰمَةِ ۗ أَلَا ذَٰلِكَ هُوَ ٱلْخُسْرَانُ ٱلْمُبِينُ ﴿١٥﴾
سىلەر اﷲ تىن باشقا خالىغىنىڭلارغا ئىبادەت قىلىڭلار!» ئېيتقىنكى، «قىيامەت كۈنى ئۆزلىرىنى ۋە ئائىلىسىدىكىلەرنى زىيان تارتقۇزغۇچىلار قاتتىق زىيان تارتقۇچىلاردۇر، بىلىڭلاركى، ئەنە شۇ روشەن زىياندۇر
لَهُم مِّن فَوْقِهِمْ ظُلَلٌ مِّنَ ٱلنَّارِ وَمِن تَحْتِهِمْ ظُلَلٌ ۚ ذَٰلِكَ يُخَوِّفُ ٱللَّهُ بِهِۦ عِبَادَهُۥ ۚ يَـٰعِبَادِ فَٱتَّقُونِ ﴿١٦﴾
ئۇلارنى ئۈستىدىنمۇ قاتمۇقات ئوت، ئاستىدىنمۇ قاتمۇقات ئوت ئورىۋالىدۇ، شۇ (قاتتىق ئازاب) بىلەن اﷲ بەندىلىرىنى قورقىتىدۇ». ئى بەندىلىرىم! (ئازابىمدىن) قورقۇڭلار
وَٱلَّذِينَ ٱجْتَنَبُوا۟ ٱلطَّـٰغُوتَ أَن يَعْبُدُوهَا وَأَنَابُوٓا۟ إِلَى ٱللَّهِ لَهُمُ ٱلْبُشْرَىٰ ۚ فَبَشِّرْ عِبَادِ ﴿١٧﴾
شەيتان (ئازدۇرغۇچى)غا (ۋە بۇتلارغا) چوقۇنۇشتىن يىراق بولغانلارغا، اﷲ قا (ۋە ئۇنىڭ ئىبادىتىگە) قايتقانلارغا (مەڭگۈلۈك باغلار بىلەن) خۇش خەۋەر بېرىلىدۇ، سۆزگە قۇلاق سېلىپ ئۇنىڭ ئەڭ ياخشىسىغا ئەگىشىدىغانلارغا (يەنى تەقۋادار بەندىلىرىمگە) خۇش خەۋەر بەرگىن، ئەنە شۇلار اﷲ ھىدايەت قىلغان كىشىلەردۇر، ئەنە شۇلار (ساغلام) ئەقىل ئىگىلىرىدۇر
ٱلَّذِينَ يَسْتَمِعُونَ ٱلْقَوْلَ فَيَتَّبِعُونَ أَحْسَنَهُۥٓ ۚ أُو۟لَـٰٓئِكَ ٱلَّذِينَ هَدَىٰهُمُ ٱللَّهُ ۖ وَأُو۟لَـٰٓئِكَ هُمْ أُو۟لُوا۟ ٱلْأَلْبَـٰبِ ﴿١٨﴾

أَفَمَنْ حَقَّ عَلَيْهِ كَلِمَةُ ٱلْعَذَابِ أَفَأَنتَ تُنقِذُ مَن فِى ٱلنَّارِ ﴿١٩﴾
كىمگە ئازاب ۋەدىسى تېگىشلىك بولسا (ئۇ چوقۇم دوزاخقا كىرىدۇ)، سەن دوزاختىكى ئادەمنى قۇتقۇزالامسەن؟
لَـٰكِنِ ٱلَّذِينَ ٱتَّقَوْا۟ رَبَّهُمْ لَهُمْ غُرَفٌ مِّن فَوْقِهَا غُرَفٌ مَّبْنِيَّةٌ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ ۖ وَعْدَ ٱللَّهِ ۖ لَا يُخْلِفُ ٱللَّهُ ٱلْمِيعَادَ ﴿٢٠﴾
لېكىن (دۇنيادا) پەرۋەردىگارىدىن قورققانلار (مەڭگۈلۈك باغلارتە) قەۋەت - قەۋەت ئالىي ئىمارەتلەردىن بەھرىمەن بولىدۇ، ئۇلارنىڭ ئاستىدىن ئۆستەڭلەر ئېقىپ تۇرىدۇ، (بۇ) اﷲ نىڭ ۋەدىسىدۇر، اﷲ ۋەدىسىگە خىلاپلىق قىلمايدۇ
أَلَمْ تَرَ أَنَّ ٱللَّهَ أَنزَلَ مِنَ ٱلسَّمَآءِ مَآءً فَسَلَكَهُۥ يَنَـٰبِيعَ فِى ٱلْأَرْضِ ثُمَّ يُخْرِجُ بِهِۦ زَرْعًا مُّخْتَلِفًا أَلْوَٰنُهُۥ ثُمَّ يَهِيجُ فَتَرَىٰهُ مُصْفَرًّا ثُمَّ يَجْعَلُهُۥ حُطَـٰمًا ۚ إِنَّ فِى ذَٰلِكَ لَذِكْرَىٰ لِأُو۟لِى ٱلْأَلْبَـٰبِ ﴿٢١﴾
سەن بىلمەمسەنكى، اﷲ بۇلۇتتىن يامغۇر ياغدۇرىدۇ. ئاندىن ئۇنى يەرگە سىڭدۈرۈپ بۇلاقنى پەيدا قىلىدۇ. ئاندىن اﷲ ئۇنىڭ بىلەن رەڭگى خىلمۇخىل زىرائەتلەرنى ئۈندۈرىدۇ. ئاندىن ئۇ قۇرۇيدۇ، ئۇنىڭ سارغىيىپ كەتكەنلىكىنى كۆرىسەن. ئاندىن اﷲ ئۇنى شاخ - شۇمبىغا ئايلاندۇرىدۇ، ئۇنىڭدا ئىشەنچلىك ھالدا ئەقىل ئىگىلىرى ئۈچۈن (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) ئىبرەت بار
أَفَمَن شَرَحَ ٱللَّهُ صَدْرَهُۥ لِلْإِسْلَـٰمِ فَهُوَ عَلَىٰ نُورٍ مِّن رَّبِّهِۦ ۚ فَوَيْلٌ لِّلْقَـٰسِيَةِ قُلُوبُهُم مِّن ذِكْرِ ٱللَّهِ ۚ أُو۟لَـٰٓئِكَ فِى ضَلَـٰلٍ مُّبِينٍ ﴿٢٢﴾
اﷲ كۆكسىنى ئىسلام ئۈچۈن كەڭ قىلغان، پەرۋەردىگارىنىڭ نۇرى ئۈستىدە بولغان ئادەم (دىلى كور ئادەم بىلەن) ئوخشاشمۇ؟ اﷲ نىڭ زىكرىنى تەرك ئەتكەنلىكتىن دىللىرى قېتىپ كەتكەنلەرگە ۋاي! ئەنە شۇلار روشەن گۇمراھلىقتىدۇر
ٱللَّهُ نَزَّلَ أَحْسَنَ ٱلْحَدِيثِ كِتَـٰبًا مُّتَشَـٰبِهًا مَّثَانِىَ تَقْشَعِرُّ مِنْهُ جُلُودُ ٱلَّذِينَ يَخْشَوْنَ رَبَّهُمْ ثُمَّ تَلِينُ جُلُودُهُمْ وَقُلُوبُهُمْ إِلَىٰ ذِكْرِ ٱللَّهِ ۚ ذَٰلِكَ هُدَى ٱللَّهِ يَهْدِى بِهِۦ مَن يَشَآءُ ۚ وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِنْ هَادٍ ﴿٢٣﴾

(پاساھەتتە، بالاغەتتە) ئۇنىڭ بەزىسى بەزىسىگە ئوخشاپ كېتىدۇ، (ۋەز - نەسىھەتلەر، ئەھكاملار، قىسسىلەر ئۇنىڭدا) تەكرارلىنىدۇ، (قۇرئاندىكى ئازاب ئايەتلىرى تىلاۋەت قىلىنغان چاغدا) پەرۋەردىگارىدىن قورقىدىغان كىشىلەرنىڭ بەدەنلىرى تىترەيدۇ، ئاندىن اﷲ نىڭ زىكرى ئۈچۈن (يەنى اﷲ نىڭ رەھمىتى، مەغپىرىتىگە دائىر ئايەتلەر تىلاۋەت قىلىنغان چاغدا)، ئۇلارنىڭ بەدەنلىرى ۋە دىللىرى يۇمشاپ (ئارام تاپىدۇ)، ئەنە شۇ (يەنى قۇرئان) اﷲ نىڭ ھىدايىتىدۇركى، ئۇنىڭ بىلەن (اﷲ) خالىغان ئادەمنى ھىدايەت قىلىدۇ، اﷲ ئازدۇرغان ئادەمگە ھېچ ھىدايەت قىلغۇچى بولمايدۇ
أَفَمَن يَتَّقِى بِوَجْهِهِۦ سُوٓءَ ٱلْعَذَابِ يَوْمَ ٱلْقِيَـٰمَةِ ۚ وَقِيلَ لِلظَّـٰلِمِينَ ذُوقُوا۟ مَا كُنتُمْ تَكْسِبُونَ ﴿٢٤﴾
قىيامەت كۈنى قاتتىق ئازابنى يۈزى بىلەن مۇداپىئە قىلىدىغان ئادەم (ئازابقا دۇچار بولمايدىغان ئادەم بىلەن ئوخشاشمۇ؟) زالىملارغا (يەنى ھەقىقەتنى رەت قىلغۇچىلارغا): «قىلغان گۇناھىڭلارنىڭ جازاسىنى تېتىڭلار» (دېيىلىدۇ)
كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ فَأَتَىٰهُمُ ٱلْعَذَابُ مِنْ حَيْثُ لَا يَشْعُرُونَ ﴿٢٥﴾
ئۇلاردىن ئىلگىرى ئۆتكەن (ئۈممەت) لەر (خۇدانىڭ ئەلچىسىلىرىنى) ئىنكار قىلدى، ئويلىمىغان يەردىن ئۇلارغا ئازاب كەلدى
فَأَذَاقَهُمُ ٱللَّهُ ٱلْخِزْىَ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَلَعَذَابُ ٱلْـَٔاخِرَةِ أَكْبَرُ ۚ لَوْ كَانُوا۟ يَعْلَمُونَ ﴿٢٦﴾
اﷲ ئۇلارغا بۇ دۇنيادا خارلىقنىڭ (تەمىنى) تېتىتتى، ئاخىرەتنىڭ ئازابى تېخىمۇ بۈيۈكتۇر، ئۇلار (ئاخىرەتنىڭ ئازابىنى) بىلسە ئىدى (خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلمايتتى)
وَلَقَدْ ضَرَبْنَا لِلنَّاسِ فِى هَـٰذَا ٱلْقُرْءَانِ مِن كُلِّ مَثَلٍ لَّعَلَّهُمْ يَتَذَكَّرُونَ ﴿٢٧﴾

ضَرَبَ ٱللَّهُ مَثَلًا رَّجُلًا فِيهِ شُرَكَآءُ مُتَشَـٰكِسُونَ وَرَجُلًا سَلَمًا لِّرَجُلٍ هَلْ يَسْتَوِيَانِ مَثَلًا ۚ ٱلْحَمْدُ لِلَّهِ ۚ بَلْ أَكْثَرُهُمْ لَا يَعْلَمُونَ ﴿٢٩﴾
اﷲ (سىلەرگە مۇنداق) بىر مىسالنى كەلتۈرىدۇ: بىر قۇل بولۇپ، ئۇنىڭغا (ئىشقا سېلىشنى) تالىشىدىغان نۇرغۇن خوجايىنلار ئىگىدارچىلىق قىلىدۇ (بىرسى بۇ ئىشقا، بىرسى ئۇ ئىشقا بۇيرۇيدۇ، ئۇ كىمنى رازى قىلىشنى بىلەلمەي قالىدۇ)، يەنە بىر قۇل بولۇپ، ئۇ بىر ئادەمگىلا خاستۇر، بۇ ئىككى قۇل (يەنى ئۇنىڭ ئەھۋالى) باراۋەرمۇ؟ جىمى ھەمدۇسانا اﷲ قا خاستۇر! بەلكى ئۇلار (يەنى مۇشرىكلار) نىڭ تولىسى بىلمەيدۇ

ئاندىن كېيىن سىلەر قىيامەت كۈنى پەرۋەردىگارىڭلارنىڭ دەرگاھىدا بىر - بىرىڭلار بىلەن دەۋا قىلىشىسىلەر
۞ فَمَنْ أَظْلَمُ مِمَّن كَذَبَ عَلَى ٱللَّهِ وَكَذَّبَ بِٱلصِّدْقِ إِذْ جَآءَهُۥٓ ۚ أَلَيْسَ فِى جَهَنَّمَ مَثْوًى لِّلْكَـٰفِرِينَ ﴿٣٢﴾
اﷲ قا يالغاننى چاپلىغان، نازىل بولغان راست سۆزنى (يەنى قۇرئاننى) ئىنكار قىلغان ئادەمدىنمۇ زالىم كىشى بارمۇ؟ مەڭگۈلۈك ئوتدە ھەقىقەتنى رەت قىلغۇچىلارغا ئورۇن يوقمۇ؟
وَٱلَّذِى جَآءَ بِٱلصِّدْقِ وَصَدَّقَ بِهِۦٓ ۙ أُو۟لَـٰٓئِكَ هُمُ ٱلْمُتَّقُونَ ﴿٣٣﴾

وَلَئِن سَأَلْتَهُم مَّنْ خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ لَيَقُولُنَّ ٱللَّهُ ۚ قُلْ أَفَرَءَيْتُم مَّا تَدْعُونَ مِن دُونِ ٱللَّهِ إِنْ أَرَادَنِىَ ٱللَّهُ بِضُرٍّ هَلْ هُنَّ كَـٰشِفَـٰتُ ضُرِّهِۦٓ أَوْ أَرَادَنِى بِرَحْمَةٍ هَلْ هُنَّ مُمْسِكَـٰتُ رَحْمَتِهِۦ ۚ قُلْ حَسْبِىَ ٱللَّهُ ۖ عَلَيْهِ يَتَوَكَّلُ ٱلْمُتَوَكِّلُونَ ﴿٣٨﴾
ئەگەر ئۇلاردىن: «ئاسمانلارنى ۋە زېمىننى كىم ياراتقان» دەپ سورىساڭ، شۈبھىسىزكى، ئۇلار: «اﷲ ياراتقان» دەيدۇ. ئېيتقىنكى، «ئەگەر اﷲ ماڭا بىرەر زىيان - زەخمەت يەتكۈزۈشنى ئىرادە قىلسا، ئۇنى دەپئى قىلىشقا، يا اﷲ ماڭا رەھمەت قىلىشنى ئىرادە قىلسا، ئۇنى توسۇۋېلىشقا، اﷲ تىن باشقا سىلەر ئىبادەت قىلىۋاتقان مەبۇدلار قادىر بولالامدۇ؟ ماڭا ئېيتىپ بېرىڭلارچۇ؟» ئېيتقىنكى، «ماڭا اﷲ كۇپايىدۇر، تەۋەككۈل قىلغۇچىلار ئۇنىڭغا تەۋەككۇل قىلسۇن (يەنى ھەممە ئىشنى اﷲ قا تاپشۇرسۇن)»
قُلْ يَـٰقَوْمِ ٱعْمَلُوا۟ عَلَىٰ مَكَانَتِكُمْ إِنِّى عَـٰمِلٌ ۖ فَسَوْفَ تَعْلَمُونَ ﴿٣٩﴾
ئېيتقىنكى، «ئى قەۋمىم! سىلەر ئۆز ھالىڭلار بويىچە ئىشلەڭلار، مەنمۇ ئۆز ھالىم بويىچە ئىشلەيمەن، سىلەر ئۇزۇنغا قالماي خارلىغۇچى ئازابنىڭ كىمگە كېلىدىغانلىقىنى، مەڭگۈلۈك ئازابنىڭ كىمگە چۈشىدىغانلىقىنى بىلىسىلەر»
مَن يَأْتِيهِ عَذَابٌ يُخْزِيهِ وَيَحِلُّ عَلَيْهِ عَذَابٌ مُّقِيمٌ ﴿٤٠﴾
ئېيتقىنكى، «ئى قەۋمىم! سىلەر ئۆز ھالىڭلار بويىچە ئىشلەڭلار، مەنمۇ ئۆز ھالىم بويىچە ئىشلەيمەن، سىلەر ئۇزۇنغا قالماي خارلىغۇچى ئازابنىڭ كىمگە كېلىدىغانلىقىنى، مەڭگۈلۈك ئازابنىڭ كىمگە چۈشىدىغانلىقىنى بىلىسىلەر»
إِنَّآ أَنزَلْنَا عَلَيْكَ ٱلْكِتَـٰبَ لِلنَّاسِ بِٱلْحَقِّ ۖ فَمَنِ ٱهْتَدَىٰ فَلِنَفْسِهِۦ ۖ وَمَن ضَلَّ فَإِنَّمَا يَضِلُّ عَلَيْهَا ۖ وَمَآ أَنتَ عَلَيْهِم بِوَكِيلٍ ﴿٤١﴾
شۈبھىسىزكى، بىز ساڭا مۇقەددەس كىتاب-ۋەھىينى ئىنسانلار ئۈچۈن ھەق بىلەن نازىل قىلدۇق، كىمكى ھىدايەت تاپىدىكەن، ئۆزى ئۈچۈن ھىدايەت تاپقان بولىدۇ، كىمكى ئازىدىكەن، ئۆزىنىڭ زىيىنى ئۈچۈن ئازغان بولىدۇ، سەن ئۇلارغا ھامىي ئەمەسسەن (يەنى ئۇلارنى ئىمانغا زورلاشقا مۇئەككەل ئەمەسسەن)
ٱللَّهُ يَتَوَفَّى ٱلْأَنفُسَ حِينَ مَوْتِهَا وَٱلَّتِى لَمْ تَمُتْ فِى مَنَامِهَا ۖ فَيُمْسِكُ ٱلَّتِى قَضَىٰ عَلَيْهَا ٱلْمَوْتَ وَيُرْسِلُ ٱلْأُخْرَىٰٓ إِلَىٰٓ أَجَلٍ مُّسَمًّى ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يَتَفَكَّرُونَ ﴿٤٢﴾
ئىنسانلار ئۆلىدىغان چاغلىرىدا، اﷲ ئۇلارنىڭ جانلىرىنى ئالىدۇ، ئۆلمىگەنلەرنىڭ جانلىرىنى ئۇخلىغان چاغلىرىدا ئالىدۇ، ئۆلۈمگە ھۆكۈم قىلىنغان (يەنى ئەجىلى يەتكەنلەر) نىڭ جانلىرىنى تۇتۇپ قېلىپ، قالغانلارنىڭ جانلىرىنى مۇئەييەن ۋاقىتقىچە (يەنى ئەجىلى يەتكۈچە) قويۇپ بېرىدۇ، بۇنىڭدا پىكىر يۈرگۈزىدىغان قەۋم ئۈچۈن (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) نۇرغۇن ئالامەتلەر بار
أَمِ ٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ شُفَعَآءَ ۚ قُلْ أَوَلَوْ كَانُوا۟ لَا يَمْلِكُونَ شَيْـًٔا وَلَا يَعْقِلُونَ ﴿٤٣﴾
(ئۇلار پىكىر يۈرگۈزمىدى) بەلكى اﷲ نى قويۇپ، (بۇتلاردىن) شاپائەتچىلەرنى تۇتتى. ئېيتقىنكى، «ئۇلار (يەنى شاپائەتچى قىلىنغان بۇتلار) ھېچ نەرسىگە قادىر بولالمايدىغان ۋە (ھېچ نەرسىنى) سەزمەيدىغان تۇرسىمۇ (ئۇلارنى شاپائەتچى تۇتامدۇ؟)
قُل لِّلَّهِ ٱلشَّفَـٰعَةُ جَمِيعًا ۖ لَّهُۥ مُلْكُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۖ ثُمَّ إِلَيْهِ تُرْجَعُونَ ﴿٤٤﴾
شاپائەتنىڭ ھەممىسى اﷲ نىڭ ئىلكىدىدۇر (ھېچ ئادەم اﷲ نىڭ ئىزنىسىز شاپائەت قىلالمايدۇ)، ئاسمانلارنىڭ ۋە زېمىننىڭ پادىشاھلىقى اﷲ قا خاستۇر. ئاندىن سىلەر اﷲ نىڭ دەرگاھىغا قايتۇرۇلىسىلەر»
وَإِذَا ذُكِرَ ٱللَّهُ وَحْدَهُ ٱشْمَأَزَّتْ قُلُوبُ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْـَٔاخِرَةِ ۖ وَإِذَا ذُكِرَ ٱلَّذِينَ مِن دُونِهِۦٓ إِذَا هُمْ يَسْتَبْشِرُونَ ﴿٤٥﴾
يالغۇز اﷲ تىلغا ئېلىنسا، ئاخىرەتكە ئىشەنمەيدىغانلارنىڭ دىللىرى سىقىلىپ كېتىدۇ. اﷲ تىن باشقا مەبۇدلار تىلغا ئېلىنسا، ئۇلار خۇش بولۇپ كېتىدۇ
قُلِ ٱللَّهُمَّ فَاطِرَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ عَـٰلِمَ ٱلْغَيْبِ وَٱلشَّهَـٰدَةِ أَنتَ تَحْكُمُ بَيْنَ عِبَادِكَ فِى مَا كَانُوا۟ فِيهِ يَخْتَلِفُونَ ﴿٤٦﴾
ئېيتقىنكى، «ئى ئاسمانلارنى ۋە زېمىننى ياراتقۇچى، يوشۇرۇننى ۋە ئاشكارىنى بىلگۈچى اﷲ! سەن بەندىلىرىڭنىڭ ئارىسىدا ئۇلار ئىختىلاپ قىلىشقان نەرسىلەر ئۈستىدە ھۆكۈم چىقىرىسەن»
وَلَوْ أَنَّ لِلَّذِينَ ظَلَمُوا۟ مَا فِى ٱلْأَرْضِ جَمِيعًا وَمِثْلَهُۥ مَعَهُۥ لَٱفْتَدَوْا۟ بِهِۦ مِن سُوٓءِ ٱلْعَذَابِ يَوْمَ ٱلْقِيَـٰمَةِ ۚ وَبَدَا لَهُم مِّنَ ٱللَّهِ مَا لَمْ يَكُونُوا۟ يَحْتَسِبُونَ ﴿٤٧﴾
ئەگەر (ئۆزلىرىگە) زۇلۇم قىلغانلارنىڭ ئىگىدارچىلىقىدا يەر يۈزىدىكى نەرسىلەرنىڭ ھەممىسى يەنە شۇنىڭدەك بىر ھەسسە كۆپ نەرسە بولىدىغان بولسا، قىيامەت كۈنىدىكى قاتتىق ئازاب ئۈچۈن ئۇنى فىدىيە بەرگەن بولاتتى، ئۇلارغا اﷲ تەرىپىدىن (بۇ دۇنيادىكى چاغلىرىدا) ئويلاپ باقمىغان ئازابلار ئاشكارا بولىدۇ
وَبَدَا لَهُمْ سَيِّـَٔاتُ مَا كَسَبُوا۟ وَحَاقَ بِهِم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿٤٨﴾
ئۇلارغا ئۇلارنىڭ (دۇنيادا) قىلغان يامان ئىشىلىرى ئاشكارا بولىدۇ، ئۇلارنىڭ (دۇنيادىكى چاغدا خۇدانىڭ ئەلچىسىنىڭ دەۋىتىنى) مەسخىرە قىلغانلىقلىرىنىڭ جازاسى ئۈچۈن ئۇلارنى (قاتتىق) ئازاب ئورىۋالىدۇ
فَإِذَا مَسَّ ٱلْإِنسَـٰنَ ضُرٌّ دَعَانَا ثُمَّ إِذَا خَوَّلْنَـٰهُ نِعْمَةً مِّنَّا قَالَ إِنَّمَآ أُوتِيتُهُۥ عَلَىٰ عِلْمٍۭ ۚ بَلْ هِىَ فِتْنَةٌ وَلَـٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿٤٩﴾
ئىنسانغا بىرەر مۇسىبەت يەتسە، بىزگە دۇئا قىلىپ (ئىلتىجا قىلىدۇ)، ئاندىن ئۇنىڭغا مەرھەمەت قىلىپ بىرەر نېمىتىمىزنى ئاتا قىلساق، ئۇ: «بۇ نېمەت ماڭا ئۆز ئىلمىمدىن (يەنى پەزلىمدىن، ئەمگىكىمدىن، تىرىشچانلىقىمدىن، رىزىق تېپىشنىڭ يوللىرىنى بىلگەنلىكىمدىن) كەلدى» دەيدۇ. (ئىش ئۇ ئويلىغاندەك ئەمەس) بەلكى ئۇ سىناقتۇر (يەنى نېمەت ئاتا قىلىش بىلەن، ئۇنى ئىتائەت قىلامدۇ؟ ئاسىيلىق قىلامدۇ؟ دەپ سىنايمىز). لېكىن ئۇلارنىڭ تولىسى (بۇنى) بىلمەيدۇ
قَدْ قَالَهَا ٱلَّذِينَ مِن قَبْلِهِمْ فَمَآ أَغْنَىٰ عَنْهُم مَّا كَانُوا۟ يَكْسِبُونَ ﴿٥٠﴾
مۇنداق سۆزنى ئۇلاردىن ئىلگىرىكىلەرمۇ قىلغان، ئۇلارنىڭ ئېرىشكەن مال - مۈلۈكلىرى ئۇلارغا ئەسقاتمىدى
فَأَصَابَهُمْ سَيِّـَٔاتُ مَا كَسَبُوا۟ ۚ وَٱلَّذِينَ ظَلَمُوا۟ مِنْ هَـٰٓؤُلَآءِ سَيُصِيبُهُمْ سَيِّـَٔاتُ مَا كَسَبُوا۟ وَمَا هُم بِمُعْجِزِينَ ﴿٥١﴾
ئۇلار (دۇنيادا) قىلغان يامانلىقلىرىنىڭ جازاسىنى تارتتى. بۇلارنىڭ ئىچىدىكى زۇلۇم قىلغانلارمۇ قىلغان يامانلىقلىرىنىڭ جازاسىنى تارتىدۇ، ئۇلار ئازابىمىزدىن قۇتۇلالمايدۇ
أَوَلَمْ يَعْلَمُوٓا۟ أَنَّ ٱللَّهَ يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ وَيَقْدِرُ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٥٢﴾
ئۇلار بىلمەمدۇكى، اﷲ (سىناش يۈزىسىدىن) خالىغان ئادەمنىڭ رىزقىنى كەڭ قىلىدۇ، (خالىغان ئادەمنىڭ رىزقىنى) تار قىلىدۇ، شەك - شۈبھىسىزكى، بۇنىڭدا ئىمان ئېيتقان قەۋم ئۈچۈن نۇرغۇن ئالامەتلەر بار
۞ قُلْ يَـٰعِبَادِىَ ٱلَّذِينَ أَسْرَفُوا۟ عَلَىٰٓ أَنفُسِهِمْ لَا تَقْنَطُوا۟ مِن رَّحْمَةِ ٱللَّهِ ۚ إِنَّ ٱللَّهَ يَغْفِرُ ٱلذُّنُوبَ جَمِيعًا ۚ إِنَّهُۥ هُوَ ٱلْغَفُورُ ٱلرَّحِيمُ ﴿٥٣﴾
(ئى مۇھەممەد! مېنىڭ تىلىمدىن) ئېيتقىنكى، «(گۇناھلارنى قىلىۋېرىپ) ئۆزلىرىگە جىنايەت قىلغان بەندىلىرىم! اﷲ نىڭ رەھمىتىدىن ئۈمىدسىزلەنمەڭلار. اﷲ ئىشەنچلىك ھالدا (خالىغان ئادەمنىڭ) جىمى گۇناھلىرىنى مەغپىرەت قىلىدۇ، شۈبھىسىزكى، اﷲ ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
وَأَنِيبُوٓا۟ إِلَىٰ رَبِّكُمْ وَأَسْلِمُوا۟ لَهُۥ مِن قَبْلِ أَن يَأْتِيَكُمُ ٱلْعَذَابُ ثُمَّ لَا تُنصَرُونَ ﴿٥٤﴾
سىلەرگە ئازاب كېلىپ، ياردەمگە ئېرىشەلمىگىنىڭلاردىن بۇرۇن پەرۋەردىگارىڭلار تەرەپكە قايتىڭلار ۋە ئۇنىڭغا ئىتائەت قىلىڭلار
وَٱتَّبِعُوٓا۟ أَحْسَنَ مَآ أُنزِلَ إِلَيْكُم مِّن رَّبِّكُم مِّن قَبْلِ أَن يَأْتِيَكُمُ ٱلْعَذَابُ بَغْتَةً وَأَنتُمْ لَا تَشْعُرُونَ ﴿٥٥﴾
سىلەر تۇيمىغان ھالەتتە سىلەرگە ئازابنىڭ ئۇشتۇمتۇت كېلىشىدىن ئىلگىرى سىلەرگە نازىل قىلىنغان ئەڭ گۈزەل قۇرئانغا ئەگىشىڭلار»
أَن تَقُولَ نَفْسٌ يَـٰحَسْرَتَىٰ عَلَىٰ مَا فَرَّطتُ فِى جَنۢبِ ٱللَّهِ وَإِن كُنتُ لَمِنَ ٱلسَّـٰخِرِينَ ﴿٥٦﴾
ھەر قانداق ئادەم: «اﷲ نىڭ تائىتىدە كەتكۈزۈپ قويغانلىرىمغا ھەسرەت! مەن (اﷲ نىڭ شەرىئىتىنى ۋە دىنىنى) مەسخىرە قىلغۇچىلاردىن ئىدىم» دەپ قالماسلىق ئۈچۈن
أَوْ تَقُولَ لَوْ أَنَّ ٱللَّهَ هَدَىٰنِى لَكُنتُ مِنَ ٱلْمُتَّقِينَ ﴿٥٧﴾
ياكى «اﷲ مېنى ھىدايەت قىلغان بولسا (ھىدايەت تېپىپ) ئەلۋەتتە تەقۋادارلاردىن بولاتتىم» دەپ قالماسلىق ئۈچۈن
أَوْ تَقُولَ حِينَ تَرَى ٱلْعَذَابَ لَوْ أَنَّ لِى كَرَّةً فَأَكُونَ مِنَ ٱلْمُحْسِنِينَ ﴿٥٨﴾
ياكى ئازابنى كۆرگەن چاغدا: «كاشكى دۇنياغا قايتىشىمغا بولسا ئىدى، (ئۇ چاغدا اﷲ قا ئىتائەت قىلىپ) ياخشىلاردىن بولار ئىدىم» دەپ قالماسلىق ئۈچۈن (پەرۋەردىگارىغا بويسۇنسۇن)
بَلَىٰ قَدْ جَآءَتْكَ ءَايَـٰتِى فَكَذَّبْتَ بِهَا وَٱسْتَكْبَرْتَ وَكُنتَ مِنَ ٱلْكَـٰفِرِينَ ﴿٥٩﴾
دۇرۇس، ئىشەنچلىك ھالدا ساڭا مېنىڭ نۇرغۇن ئايەتلىرىم (يەنى ھىدايەتكە سەۋەب بولىدىغان قۇرئان) كەلدى. سەن ئۇلارنى ئىنكار قىلدىڭ، بويۇنتاۋلىق قىلدىڭ، سەن ھەقىقەتنى رەت قىلغۇچىلادىن بولدۇڭ
وَيَوْمَ ٱلْقِيَـٰمَةِ تَرَى ٱلَّذِينَ كَذَبُوا۟ عَلَى ٱللَّهِ وُجُوهُهُم مُّسْوَدَّةٌ ۚ أَلَيْسَ فِى جَهَنَّمَ مَثْوًى لِّلْمُتَكَبِّرِينَ ﴿٦٠﴾
قىيامەت كۈنى اﷲ قا يالغاننى چاپلىغانلارنىڭ يۈزلىرىنى قاپقارا كۆرىسەن، مەڭگۈلۈك ئوتدە مۇتەكەببىرلەرگە ئورۇن يوقمۇ؟
وَيُنَجِّى ٱللَّهُ ٱلَّذِينَ ٱتَّقَوْا۟ بِمَفَازَتِهِمْ لَا يَمَسُّهُمُ ٱلسُّوٓءُ وَلَا هُمْ يَحْزَنُونَ ﴿٦١﴾
اﷲ (شېرىكتىن ۋە گۇناھتىن) ساقلانغۇچىلارنى بەختكە ئېرىشتۈرىدىغان يوللار ئارقىلىق (ئازابتىن) قۇتۇلدۇرىدۇ، ئۇلارغا كۈلپەت يەتمەيدۇ، ئۇلار قايغۇرمايدۇ
ٱللَّهُ خَـٰلِقُ كُلِّ شَىْءٍ ۖ وَهُوَ عَلَىٰ كُلِّ شَىْءٍ وَكِيلٌ ﴿٦٢﴾
اﷲ ھەممە نەرسىنى ياراتقۇچىدۇر، ھەممە نەرسىگە ھامىيدۇر (ئۇلارنى خالىغانچە تەسەررۇپ قىلىدۇ)
لَّهُۥ مَقَالِيدُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۗ وَٱلَّذِينَ كَفَرُوا۟ بِـَٔايَـٰتِ ٱللَّهِ أُو۟لَـٰٓئِكَ هُمُ ٱلْخَـٰسِرُونَ ﴿٦٣﴾
ئاسمانلارنىڭ ۋە زېمىننىڭ (خەزىنىلىرىنىڭ) ئاچقۇچلىرى اﷲ نىڭ ئىلكىدىدۇر، اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغانلار – ئەنە شۇلار زىيان تارتقۇچىلاردۇر
قُلْ أَفَغَيْرَ ٱللَّهِ تَأْمُرُوٓنِّىٓ أَعْبُدُ أَيُّهَا ٱلْجَـٰهِلُونَ ﴿٦٤﴾
(ئى مۇھەممەد!) ئېيتقىنكى، «ئى نادانلار! مېنى اﷲ تىن غەيرىيگە ئىبادەت قىلىشىقا بۇيرۇمسىلەر؟»
وَلَقَدْ أُوحِىَ إِلَيْكَ وَإِلَى ٱلَّذِينَ مِن قَبْلِكَ لَئِنْ أَشْرَكْتَ لَيَحْبَطَنَّ عَمَلُكَ وَلَتَكُونَنَّ مِنَ ٱلْخَـٰسِرِينَ ﴿٦٥﴾
ساڭا ۋە سەندىن ئىلگىرىكى (خۇدانىڭ ئەلچىسى) لەرگە: «ئەگەر سەن اﷲ قا شېرىك كەلتۈرسەڭ، سېنىڭ ئەمەلىڭ ئەلۋەتتە بىكار بولۇپ كېتىدۇ، سەن ئەلۋەتتە زىيان تارتقۇچىلاردىن بولۇپ قالىسەن، بەلكى يالغۇز اﷲ قا ئىبادەت قىلغىن ۋە شۈكۈر قىلغۇچىلاردىن بولغىن» دەپ ۋەھيى قىلىندى
بَلِ ٱللَّهَ فَٱعْبُدْ وَكُن مِّنَ ٱلشَّـٰكِرِينَ ﴿٦٦﴾
ساڭا ۋە سەندىن ئىلگىرىكى (خۇدانىڭ ئەلچىسى) لەرگە: «ئەگەر سەن اﷲ قا شېرىك كەلتۈرسەڭ، سېنىڭ ئەمەلىڭ ئەلۋەتتە بىكار بولۇپ كېتىدۇ، سەن ئەلۋەتتە زىيان تارتقۇچىلاردىن بولۇپ قالىسەن، بەلكى يالغۇز اﷲ قا ئىبادەت قىلغىن ۋە شۈكۈر قىلغۇچىلاردىن بولغىن» دەپ ۋەھيى قىلىندى
وَمَا قَدَرُوا۟ ٱللَّهَ حَقَّ قَدْرِهِۦ وَٱلْأَرْضُ جَمِيعًا قَبْضَتُهُۥ يَوْمَ ٱلْقِيَـٰمَةِ وَٱلسَّمَـٰوَٰتُ مَطْوِيَّـٰتٌۢ بِيَمِينِهِۦ ۚ سُبْحَـٰنَهُۥ وَتَعَـٰلَىٰ عَمَّا يُشْرِكُونَ ﴿٦٧﴾
ئۇلار اﷲ نى ھەقىقىي رەۋىشتە تونۇمىدى. قىيامەت كۈنى زېمىن پۈتۈنلەي اﷲ نىڭ چاڭگىلىدا بولىدۇ، ئاسمانلار اﷲ نىڭ ئوڭ قولىدا قاتلىنىپ تۇرىدۇ، اﷲ ئۇلار (يەنى مۇشرىكلار) نىڭ شېرىك كەلتۈرگەن نەرسىلىرىدىن پاكتۇر ۋە ئۈستۈندۇر
وَنُفِخَ فِى ٱلصُّورِ فَصَعِقَ مَن فِى ٱلسَّمَـٰوَٰتِ وَمَن فِى ٱلْأَرْضِ إِلَّا مَن شَآءَ ٱللَّهُ ۖ ثُمَّ نُفِخَ فِيهِ أُخْرَىٰ فَإِذَا هُمْ قِيَامٌ يَنظُرُونَ ﴿٦٨﴾
سۇر چېلىنغاندا ئاسمانلاردىكى ۋە زېمىندىكى اﷲ خالىغاندىن باشقا مەخلۇقاتلارنىڭ ھەممىسى ئۆلىدۇ، ئاندىن سۇر ئىككىنچى قېتىم چېلىنغاندا ناگاھان ئۇلار تۇرۇپ (نېمىگە بۇيرۇلىدىغانلىقلىرىغا) قاراپ تۇرىدۇ
وَأَشْرَقَتِ ٱلْأَرْضُ بِنُورِ رَبِّهَا وَوُضِعَ ٱلْكِتَـٰبُ وَجِا۟ىٓءَ بِٱلنَّبِيِّـۧنَ وَٱلشُّهَدَآءِ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَهُمْ لَا يُظْلَمُونَ ﴿٦٩﴾
مەھشەرگاھ پەرۋەردىگارىنىڭ نۇرى بىلەن يورۇيدۇ، نامە - ئەماللار ھازىر قىلىنىدۇ، خۇدانىڭ ئەلچى-رەسۇللىرى ۋە گۇۋاھچىلار كەلتۈرۈلىدۇ، ئۇلارنىڭ ئارىسىدا ئادىل ھۆكۈم چىقىرىلىدۇ، ئۇلارغا زۇلۇم قىلىنمايدۇ
وَوُفِّيَتْ كُلُّ نَفْسٍ مَّا عَمِلَتْ وَهُوَ أَعْلَمُ بِمَا يَفْعَلُونَ ﴿٧٠﴾
ھەر ئادەمگە قىلغان ئەمەلىنىڭ ساۋابى تولۇق بېرىلىدۇ، اﷲ ئۇلارنىڭ ئەمەللىرىنى ئوبدان بىلىدۇ
وَسِيقَ ٱلَّذِينَ كَفَرُوٓا۟ إِلَىٰ جَهَنَّمَ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا فُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَآ أَلَمْ يَأْتِكُمْ رُسُلٌ مِّنكُمْ يَتْلُونَ عَلَيْكُمْ ءَايَـٰتِ رَبِّكُمْ وَيُنذِرُونَكُمْ لِقَآءَ يَوْمِكُمْ هَـٰذَا ۚ قَالُوا۟ بَلَىٰ وَلَـٰكِنْ حَقَّتْ كَلِمَةُ ٱلْعَذَابِ عَلَى ٱلْكَـٰفِرِينَ ﴿٧١﴾
ھەقىقەتنى رەت قىلغۇچىلار مەڭگۈلۈك ئوتگە توپ - توپ بولۇپ ھەيدىلىدۇ، ئۇلار مەڭگۈلۈك ئوتگە يېتىپ كەلگەندە، مەڭگۈلۈك ئوتنىڭ دەرۋازىلىرى ئېچىلىدۇ، مەڭگۈلۈك ئوتنى باشقۇرغۇچى خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارغا: «ئاراڭلاردىن سىلەرگە پەرۋەردىگارىڭلارنىڭ ئايەتلىرىنى سىلەرگە ئوقۇپ بېرىدىغان، بۈگۈنكى ئۇچرىشىشنىڭ بارلىقىدىن ئاگاھلاندۇرىدىغان خۇدانىڭ ئەلچى-رەسۇللىرى كەلمىگەنمۇ؟» دەيدۇ. ئۇلار: «شۇنداق (خۇدانىڭ ئەلچى-رەسۇللىرى كەلگەن، بىزنى ئاگاھلاندۇرغان)، لېكىن ئازابقا دۇچار بولۇش ھۆكمى (كۇفرى ۋە قىلمىشلىرى تۈپەيلىدىن) ھەقىقەتنى رەت قىلغۇچىلارغا تېگىشلىك بولدى» دەيدۇ
قِيلَ ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَـٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٢﴾
(ئۇلارغا) «مەڭگۈلۈك ئوتنىڭ دەرۋازىلىرىدىن كىرىڭلار، مەڭگۈلۈك ئوتدە مەڭگۈ قېلىڭلار، مۇتەكەببىرلارنىڭ جايى نېمىدېگەن يامان!» دېيىلىدۇ
وَسِيقَ ٱلَّذِينَ ٱتَّقَوْا۟ رَبَّهُمْ إِلَى ٱلْجَنَّةِ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا وَفُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَا سَلَـٰمٌ عَلَيْكُمْ طِبْتُمْ فَٱدْخُلُوهَا خَـٰلِدِينَ ﴿٧٣﴾
پەرۋەردىگارىغا تەقۋادارلىق قىلغانلار مەڭگۈلۈك باغلاركە توپ - توپ بولغان ھالدا ماڭدۇرۇلىدۇ، ھالبۇكى، ئۇلار مەڭگۈلۈك باغلاركە يېتىپ كەلگەن چاغدا ئۇنىڭ دەرۋازىلىرى ئېچىلىپ بولغان بولىدۇ، مەڭگۈلۈك باغلاركە مۇئەككەل خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارغا: «سىلەرگە ئامانلىق بولسۇن، سىلەر (گۇناھلارنىڭ كىرلىرىدىن) پاك بولدۇڭلار، مەڭگۈلۈك باغلاركە كىرىڭلار، (ئۇنىڭدا) مەڭگۈ قېلىڭلار» دەيدۇ
وَقَالُوا۟ ٱلْحَمْدُ لِلَّهِ ٱلَّذِى صَدَقَنَا وَعْدَهُۥ وَأَوْرَثَنَا ٱلْأَرْضَ نَتَبَوَّأُ مِنَ ٱلْجَنَّةِ حَيْثُ نَشَآءُ ۖ فَنِعْمَ أَجْرُ ٱلْعَـٰمِلِينَ ﴿٧٤﴾
ئۇلار (مەڭگۈلۈك باغلاركە كىرىپ ئورۇنلاشقان چاغدا) ئېيتىدۇ: «جىمى ھەمدۇسانا اﷲ قا خاستۇركى، ئۇ بىزگە قىلغان ۋەدىسىنى ئىشقا ئاشۇردى، مەڭگۈلۈك باغلارنىڭ زېمىنىغا بىزنى ۋارىس قىلدى. مەڭگۈلۈك باغلارتە ئۆزىمىز خالىغان جايدا تۇرىمىز، ئىشلىگۈچىلەرنىڭ ئەجرى (يەنى مەڭگۈلۈك باغلار) نېمىدېگەن ياخشى!»
وَتَرَى ٱلْمَلَـٰٓئِكَةَ حَآفِّينَ مِنْ حَوْلِ ٱلْعَرْشِ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ ۖ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَقِيلَ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٧٥﴾
خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ پەرۋەردىگارىغا تەسبىھ ئېيتقان، ھەمدى ئېيتقان ھالدا ئەرشنىڭ چۆرىسىنى ئوراپ تۇرغانلىقىنى كۆرىسەن، ئۇلارنىڭ ئارىسىدا ھەققانىي ھۆكۈم چىقىرىلىدۇ (مۆمىنلەر مەڭگۈلۈك باغلاركە كىرىدۇ، ھەقىقەتنى رەت قىلغۇچىلار دوزاخقا كىرىدۇ)، «جىمى ھەمدۇسانا ئالەملەرنىڭ پەرۋەردىگارى اﷲ قا خاستۇر!» دېيىلىدۇ
