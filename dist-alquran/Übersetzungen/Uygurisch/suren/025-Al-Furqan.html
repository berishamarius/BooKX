بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
تَبَارَكَ ٱلَّذِى نَزَّلَ ٱلْفُرْقَانَ عَلَىٰ عَبْدِهِۦ لِيَكُونَ لِلْعَـٰلَمِينَ نَذِيرًا ﴿١﴾
پۈتۈن جاھان ئەھلىنى (اﷲ نىڭ ئازابىدىن) ئاگاھلاندۇرغۇچى بولۇشى ئۈچۈن، بەندىسىگە (يەنى مۇھەممەد ئەلەيھىسسالامغا) ھەق بىلەن باتىلنى ئايرىغۇچى قۇرئاننى نازىل قىلغان اﷲ نىڭ بەرىكىتى بۈيۈكتۇر
ٱلَّذِى لَهُۥ مُلْكُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَلَمْ يَتَّخِذْ وَلَدًا وَلَمْ يَكُن لَّهُۥ شَرِيكٌ فِى ٱلْمُلْكِ وَخَلَقَ كُلَّ شَىْءٍ فَقَدَّرَهُۥ تَقْدِيرًا ﴿٢﴾
ئاسمانلارنىڭ ۋە زېمىننىڭ پادىشاھلىقى اﷲ قا خاستۇر، اﷲ نىڭ بالىسى يوقتۇر، پادىشاھلىقتا شېرىكىمۇ يوقتۇر، اﷲ ھەممە نەرسىنى ياراتتى، ئاندىن (ئۆزىنىڭ خاھىشى بويىچە) ئۇنىڭغا مۇناسىپ ئۆلچەمنى بەلگىلىدى
وَٱتَّخَذُوا۟ مِن دُونِهِۦٓ ءَالِهَةً لَّا يَخْلُقُونَ شَيْـًٔا وَهُمْ يُخْلَقُونَ وَلَا يَمْلِكُونَ لِأَنفُسِهِمْ ضَرًّا وَلَا نَفْعًا وَلَا يَمْلِكُونَ مَوْتًا وَلَا حَيَوٰةً وَلَا نُشُورًا ﴿٣﴾
مۇشرىكلار اﷲ نى قويۇپ ھېچ نەرسىنى يارىتالمايدىغان ۋە ئۆزلىرى يارىتىلغان ئىلاھلار (يەنى بۇتلار) غا ئىبادەت قىلدى. ئۇلار (يەنى بۇتلار) ئۆزلىرىدىن بىرەر زىياننى دەپئى قىلالمايدۇ. ئۆزلىرىگە بىرەر پايدىنى كەلتۈرەلمەيدۇ، (باشقىلارنىڭ) جېنىنى ئېلىشقا، (باشقىلارغا) ھايات بېغىشلاشقا ياكى (ئۆلۈكلەرنى) تىرىلدۈرۈشكە قادىر بولالمايدۇ
وَقَالَ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ هَـٰذَآ إِلَّآ إِفْكٌ ٱفْتَرَىٰهُ وَأَعَانَهُۥ عَلَيْهِ قَوْمٌ ءَاخَرُونَ ۖ فَقَدْ جَآءُو ظُلْمًا وَزُورًا ﴿٤﴾
ھەقىقەتنى رەت قىلغۇچىلار: «بۇ (يەنى قۇرئان) پەقەت ئۇ (يەنى مۇھەممەد) ئۆزى توقۇغان يالغاندۇر، ئۇنىڭغا ئۇنى توقۇشقا باشقا بىر قەۋم (يەنى ئەھلى مۇقەددەس كىتاب-ۋەھىي) ياردەم بەرگەن» دەيدۇ، ئۇلار (خۇدانىڭ ئەلچىسى ئەلەيھىسسالامغا بوھتان چاپلاش بىلەن ئۇنىڭغا) زۇلۇم قىلدى ۋە يالغاننى چاپلىدى
وَقَالُوٓا۟ أَسَـٰطِيرُ ٱلْأَوَّلِينَ ٱكْتَتَبَهَا فَهِىَ تُمْلَىٰ عَلَيْهِ بُكْرَةً وَأَصِيلًا ﴿٥﴾
ئۇلار ئېيتتى: «(بۇ) بۇرۇنقىلارنىڭ ئەپسانىلىرىدۇركى، ئۇلارنى ئۇ يازدۇرۇۋالغان، (ھىپزى قىلىۋېلىش ئۈچۈن) ئۇنىڭغا ئەتىگەن – ئاخشامدا ئوقۇپ بېرىلىۋاتىدۇ»
قُلْ أَنزَلَهُ ٱلَّذِى يَعْلَمُ ٱلسِّرَّ فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۚ إِنَّهُۥ كَانَ غَفُورًا رَّحِيمًا ﴿٦﴾
ئېيتقىنكى، «ئۇنى ئاسمانلاردىكى ۋە زېمىندىكى سىرنى بىلىپ تۇرىدىغان اﷲ نازىل قىلدى، اﷲ ئىشەنچلىك ھالدا ناھايىتى مەغپىرەت قىلغۇچىدۇر، (بەندىلەرگە) ناھايىتى مېھرىباندۇر»
وَقَالُوا۟ مَالِ هَـٰذَا ٱلرَّسُولِ يَأْكُلُ ٱلطَّعَامَ وَيَمْشِى فِى ٱلْأَسْوَاقِ ۙ لَوْلَآ أُنزِلَ إِلَيْهِ مَلَكٌ فَيَكُونَ مَعَهُۥ نَذِيرًا ﴿٧﴾
مۇشرىكلار (مۇھەممەد ئەلەيھىسسالامنى مەسخىرە قىلىپ) ئېيتتى: «بازارلاردا (بىز تاماق يېگەندەك) تاماق يەيدىغان، (بىز ماڭغاندەك) مېڭىپ يۈرىدىغان بۇ قانداق خۇدانىڭ ئەلچىسى؟ ئۇنىڭ بىلەن بىللە ئاگاھلاندۇرغۇچى بولۇش ئۈچۈن اﷲ ئۇنىڭغا نېمىشقا بىر پەرىشتە ئەۋەتمىدى
أَوْ يُلْقَىٰٓ إِلَيْهِ كَنزٌ أَوْ تَكُونُ لَهُۥ جَنَّةٌ يَأْكُلُ مِنْهَا ۚ وَقَالَ ٱلظَّـٰلِمُونَ إِن تَتَّبِعُونَ إِلَّا رَجُلًا مَّسْحُورًا ﴿٨﴾
يا ئۇنىڭغا بىر خەزىنە بېرىلمىدى، يا ئۇنىڭغا مېۋىلىرىنى يەيدىغان بىر باغ بېرىلمىدى؟» زالىملار (يەنى ھەقىقەتنى رەت قىلغۇچىلار): «سىلەر پەقەت سېھىرلەنگەن بىر ئادەمگە ئەگىشىۋاتىسىلەر» دېدى
ٱنظُرْ كَيْفَ ضَرَبُوا۟ لَكَ ٱلْأَمْثَـٰلَ فَضَلُّوا۟ فَلَا يَسْتَطِيعُونَ سَبِيلًا ﴿٩﴾
مۇشرىكلارنىڭ سېنىڭ ئۈچۈن نۇرغۇن مىساللارنى كەلتۈرگەنلىكىگە قارىغىن، ئۇلار ئازدى، توغرا يول تاپالمايدۇ
تَبَارَكَ ٱلَّذِىٓ إِن شَآءَ جَعَلَ لَكَ خَيْرًا مِّن ذَٰلِكَ جَنَّـٰتٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ وَيَجْعَل لَّكَ قُصُورًۢا ﴿١٠﴾
اﷲ نىڭ بەرىكىتى بۈيۈكتۇركى، ئەگەر ئۇ خالىسا ساڭا ئۇلارنىڭ ئېيتقانلىرىدىنمۇ ياخشى، ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان باغلارنى ۋە چوڭ سارايلارنى ئاتا قىلاتتى
بَلْ كَذَّبُوا۟ بِٱلسَّاعَةِ ۖ وَأَعْتَدْنَا لِمَن كَذَّبَ بِٱلسَّاعَةِ سَعِيرًا ﴿١١﴾
ياق، ئۇلار قىيامەتنى ئىنكار قىلدى، قىيامەتنى ئىنكار قىلغانلارغا بىز دوزاخنى تەييارلىدۇق
إِذَا رَأَتْهُم مِّن مَّكَانٍۭ بَعِيدٍ سَمِعُوا۟ لَهَا تَغَيُّظًا وَزَفِيرًا ﴿١٢﴾
دوزاخ ئۇلارنى يىراقتىن كۆرگەن چاغدا، ئۇلار دوزاخنىڭ غەزەپتىن قاينىغان ۋە (ئېشەكتەك) ھاڭرىغان ئاۋازنى ئاڭلايدۇ
وَإِذَآ أُلْقُوا۟ مِنْهَا مَكَانًا ضَيِّقًا مُّقَرَّنِينَ دَعَوْا۟ هُنَالِكَ ثُبُورًا ﴿١٣﴾
ئۇلار زەنجىر بىلەن باغلانغان ھالدا دوزاخنىڭ تار يېرىگە تاشلانغان چاغدا، بۇ يەردە ئۇلار (ئۆزلىرىنىڭ) ئۆلۈمىنى تىلەيدۇ
لَّا تَدْعُوا۟ ٱلْيَوْمَ ثُبُورًا وَٰحِدًا وَٱدْعُوا۟ ثُبُورًا كَثِيرًا ﴿١٤﴾
(ئۇلارغا) بۈگۈن سىلەر بىر ئۆلۈمنى ئەمەس، تالاي ئۆلۈملەرنى تىلەڭلار (دېيىلىدۇ)
قُلْ أَذَٰلِكَ خَيْرٌ أَمْ جَنَّةُ ٱلْخُلْدِ ٱلَّتِى وُعِدَ ٱلْمُتَّقُونَ ۚ كَانَتْ لَهُمْ جَزَآءً وَمَصِيرًا ﴿١٥﴾
ئۇلارغا ئېيتقىنكى، «شۇ (دوزاخ) ياخشىمۇ يا تەقۋادارلارغا ۋەدە قىلىنغان، (كىرگەن كىشى) مەڭگۈ قالىدىغان مەڭگۈلۈك باغلارتە ياخشىمۇ؟» مەڭگۈلۈك باغلار تەقۋادارلارغا بېرىلگەن مۇكاپاتتۇر ۋە ئۇلارنىڭ ئاخىرى بارىدىغان جايىدۇر
لَّهُمْ فِيهَا مَا يَشَآءُونَ خَـٰلِدِينَ ۚ كَانَ عَلَىٰ رَبِّكَ وَعْدًا مَّسْـُٔولًا ﴿١٦﴾
ئۇلار مەڭگۈلۈك باغلارتە خالىغان نېمەتلەردىن بەھرىمەن بولىدۇ، مەڭگۈلۈك باغلارتە مەڭگۈ قالىدۇ، بۇ پەرۋەردىگارىڭدىن ئىشقا ئاشۇرۇش تىلىنىدىغان ۋەدىدۇر
وَيَوْمَ يَحْشُرُهُمْ وَمَا يَعْبُدُونَ مِن دُونِ ٱللَّهِ فَيَقُولُ ءَأَنتُمْ أَضْلَلْتُمْ عِبَادِى هَـٰٓؤُلَآءِ أَمْ هُمْ ضَلُّوا۟ ٱلسَّبِيلَ ﴿١٧﴾
شۇ كۈندە اﷲ ئۇلارغا قوشۇپ، ئۇلار اﷲ نى قويۇپ، چوقۇنغان نەرسىلىرىنى يىغىدۇ. ئاندىن اﷲ (ئۇلارغا): «مېنىڭ بۇ بەندىلىرىمنى سىلەر ئازدۇردۇڭلارمۇ؟ يا (توغرا) يولدىن ئۇلار ئۆزلىرى ئازدىمۇ؟» دەيدۇ
قَالُوا۟ سُبْحَـٰنَكَ مَا كَانَ يَنۢبَغِى لَنَآ أَن نَّتَّخِذَ مِن دُونِكَ مِنْ أَوْلِيَآءَ وَلَـٰكِن مَّتَّعْتَهُمْ وَءَابَآءَهُمْ حَتَّىٰ نَسُوا۟ ٱلذِّكْرَ وَكَانُوا۟ قَوْمًۢا بُورًا ﴿١٨﴾
(ئىبادەت قىلىنغۇچىلار) ئېتىدۇ: «ئى اﷲ! سەن پاكتۇرسەن، سېنى قويۇپ باشقىلارنى دوست تۇتۇش (يەنى سەندىن باشقىغا چوقۇنۇش) بىزگە (ۋە سېنىڭ مەخلۇقاتىڭدىن ھېچ ئەھەدىگە) لايىق ئەمەس، لېكىن سەن ئۇلارنى ۋە ئۇلارنىڭ ئاتا – بوۋىلىرىنى شۇ قەدەر نېمەتكە چۈمدۈردۇڭكى، ئۇلار سېنى ياد ئېتىشىنى ئۇنتۇدى. ئۇلار ھالاك بولغۇچى قەۋم بولدى»
فَقَدْ كَذَّبُوكُم بِمَا تَقُولُونَ فَمَا تَسْتَطِيعُونَ صَرْفًا وَلَا نَصْرًا ۚ وَمَن يَظْلِم مِّنكُمْ نُذِقْهُ عَذَابًا كَبِيرًا ﴿١٩﴾
شۈبھىسىزكى، ئىبادەت قىلىنغۇچىلار سىلەرنىڭ (ئۇلارنى مەبۇد دېگەن) سۆزۈڭلارنى ئىنكار قىلدى، (ئى كۇففارلار!) سىلەر (ئۆزۈڭلاردىن ئازابنى) دەپئى قىلىشقىمۇ ۋە (بۇ بالادىن ئۆزۈڭلارنى قۇتۇلدۇرۇشقا) ياردەم بېرىشكىمۇ قادىر بولالمايسىلەر، سىلەردىن كىمكى (اﷲ قا شېرىك كەلتۈرۈش بىلەن ئۆزىگە) زۇلۇم قىلىدىكەن، ئۇنىڭغا بىز (ئاخىرەتتە) قاتتىق ئازابنى تېتىتىمىز
وَمَآ أَرْسَلْنَا قَبْلَكَ مِنَ ٱلْمُرْسَلِينَ إِلَّآ إِنَّهُمْ لَيَأْكُلُونَ ٱلطَّعَامَ وَيَمْشُونَ فِى ٱلْأَسْوَاقِ ۗ وَجَعَلْنَا بَعْضَكُمْ لِبَعْضٍ فِتْنَةً أَتَصْبِرُونَ ۗ وَكَانَ رَبُّكَ بَصِيرًا ﴿٢٠﴾
بىز سەندىن ئىلگىرى ئەۋەتكەن خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ ھەممىسىلا ئەلۋەتتە تاماق يەيتتى، بازارلاردا مېڭىپ يۈرەتتى، بەزىڭلارنى بەزىڭلار بىلەن سىنىدۇق، سەۋر قىلامسىلەر؟ (يەنى سەۋر قىلىڭلار) پەرۋەردىگارىڭ (سەۋر قىلغۇچىلار بىلەن سەۋر قىلمىغۇچىلارنى) كۆرۇپ تۇرغۇچىدۇر
۞ وَقَالَ ٱلَّذِينَ لَا يَرْجُونَ لِقَآءَنَا لَوْلَآ أُنزِلَ عَلَيْنَا ٱلْمَلَـٰٓئِكَةُ أَوْ نَرَىٰ رَبَّنَا ۗ لَقَدِ ٱسْتَكْبَرُوا۟ فِىٓ أَنفُسِهِمْ وَعَتَوْ عُتُوًّا كَبِيرًا ﴿٢١﴾
بىزگە مۇلاقات بولۇشنى ئۈمىد قىلمايدىغانلار (يەنى مۇشرىكلار): «نېمىشقا بىزگە خۇدانىڭ ئەلچى پەرىشتىلىرى چۈشۈرۈلمەيدۇ، ياكى (بىز نېمىشقا) پەرۋەردىگارىمىزنى كۆرمەيمىز» دېيىشتى. شۈبھىسىزكى ئۇلار ئۆزلىرىنى چوڭ تۇتتى ۋە تولىمۇ ھەددىدىن ئاشتى
يَوْمَ يَرَوْنَ ٱلْمَلَـٰٓئِكَةَ لَا بُشْرَىٰ يَوْمَئِذٍ لِّلْمُجْرِمِينَ وَيَقُولُونَ حِجْرًا مَّحْجُورًا ﴿٢٢﴾
ئۇلار (يەنى مۇشرىكلار) (ئۆزلىرىنىڭ جېنىنى ئېلىش ئۈچۈن چۈشكەن) خۇدانىڭ ئەلچى پەرىشتىلىرىنى كۆرگەن كۈنى گۇناھكارلارغا خۇش خەۋەر بولمايدۇ، خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارغا: «(سىلەرگە مەڭگۈلۈك باغلار) ھارامدۇر» دەيدۇ
وَقَدِمْنَآ إِلَىٰ مَا عَمِلُوا۟ مِنْ عَمَلٍ فَجَعَلْنَـٰهُ هَبَآءً مَّنثُورًا ﴿٢٣﴾
بىز ئۇلارنىڭ قىلغان ياخشى ئەمەللىرىنى بىر تەرەپ قىلىپ، ئۇنى ئۇچۇپ يۈرگەن توزاندەك قىلىۋېتىمىز، (يەنى ئۇلار ئىمانسىىز بولغانلىقتىن، قىلغان ياخشى ئەمەللىرىنى يوققا چىقىرىمىز)
أَصْحَـٰبُ ٱلْجَنَّةِ يَوْمَئِذٍ خَيْرٌ مُّسْتَقَرًّا وَأَحْسَنُ مَقِيلًا ﴿٢٤﴾
ئەھلى مەڭگۈلۈك باغلار بۇ كۈندە ئەڭ ياخشى جايدا ۋە ئەڭ ياخشى ئارامگاھتا بولىدۇ

شۇنىڭدەك (يەنى ساڭا قەۋمىڭنىڭ مۇشرىكلىرىدىن دۈشمەنلەرنى قىلغاندەك) ھەربىر خۇدانىڭ ئەلچىسىگە بىر قىسىم گۇناھكارلارنى ئۇنىڭ دۈشمىنى قىلدۇق، پەرۋەردىگارىڭ يول كۆرسەتكۈچى ۋە ياردەمچى بولۇشقا ساڭا يېتەرلىكتۇر
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لَوْلَا نُزِّلَ عَلَيْهِ ٱلْقُرْءَانُ جُمْلَةً وَٰحِدَةً ۚ كَذَٰلِكَ لِنُثَبِّتَ بِهِۦ فُؤَادَكَ ۖ وَرَتَّلْنَـٰهُ تَرْتِيلًا ﴿٣٢﴾
ھەقىقەتنى رەت قىلغۇچىلار: «قۇرئان نېمىشقا ئۇنىڭغا (يەنى مۇھەممەد ئەلەيھىسسالامغا) بىر قېتىمدىلا نازىل قىلىنمىدى؟» دېدى. سېنىڭ دىلىڭنى مۇستەھكەم قىلىش ئۈچۈن، ئۇنى پارچە - پارچە نازىل قىلدۇق، ئۇنى ئايرىم – ئايرىم نازىل قىلدۇق

ئۇلار (يەنى مۇشرىكلار) قانداق بىر سوئالنى تاشلىمىسۇن، (ئۇنىڭغا قارىتا) بىز ساڭا ھەق جاۋابنى ۋە ھەممىدىن گۈزەل چۈشەندۇرۇشنى نازىل قىلدۇق
ٱلَّذِينَ يُحْشَرُونَ عَلَىٰ وُجُوهِهِمْ إِلَىٰ جَهَنَّمَ أُو۟لَـٰٓئِكَ شَرٌّ مَّكَانًا وَأَضَلُّ سَبِيلًا ﴿٣٤﴾
ئۇلار مەڭگۈلۈك ئوتگە يۈزلىرى بىلەن سۈرۈلىدۇ، ئۇلارنىڭ جايى ئەڭ يامان، يولى ئەڭ ئازغۇندۇر
وَلَقَدْ ءَاتَيْنَا مُوسَى ٱلْكِتَـٰبَ وَجَعَلْنَا مَعَهُۥٓ أَخَاهُ هَـٰرُونَ وَزِيرًا ﴿٣٥﴾

فَلَا تُطِعِ ٱلْكَـٰفِرِينَ وَجَـٰهِدْهُم بِهِۦ جِهَادًا كَبِيرًا ﴿٥٢﴾
شۇنىڭ ئۈچۈن ھەقىقەتنى رەت قىلغۇچىلارغا ئىتائەت قىلمىغىن، قۇرئان ئارقىلىق ئۇلارغا قارشى (پاكىتلارنى ئوتتۇرىغا قويۇپ) بارلىق كۈچۈڭ بىلەن كۈرەش قىلغىن
۞ وَهُوَ ٱلَّذِى مَرَجَ ٱلْبَحْرَيْنِ هَـٰذَا عَذْبٌ فُرَاتٌ وَهَـٰذَا مِلْحٌ أُجَاجٌ وَجَعَلَ بَيْنَهُمَا بَرْزَخًا وَحِجْرًا مَّحْجُورًا ﴿٥٣﴾
اﷲ (قۇدرىتى بىلەن) بىرى تاتلىق بولغان دەريانى، بىرى تۇزلۇق بولغان دېڭىزنى (ئۇلار بىر - بىرىگە تۇتىشاڭغۇ بولغان ھالدا (قويۇۋەتتى) ئارلىشىپ كەتمەسلىكى ئۈچۈن (ئۇلارنىڭ ئارىسىدا پەردە، توسما بەرپا قىلدى
وَهُوَ ٱلَّذِى خَلَقَ مِنَ ٱلْمَآءِ بَشَرًا فَجَعَلَهُۥ نَسَبًا وَصِهْرًا ۗ وَكَانَ رَبُّكَ قَدِيرًا ﴿٥٤﴾
اﷲ ئىنساننى ئابىمەنىدىن ياراتتى، ئۇنى بالىلار نىسبەت بېرىلىدىغان ئەر ۋە قۇدا - باجىلىق پەيدا قىلىدىغان ئايال (دىن ئىبارەت ئىككى خىل قىلىپ) ياراتتى، پەرۋەردىگارىڭ ھەممىگە قادىردۇر
وَيَعْبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَنفَعُهُمْ وَلَا يَضُرُّهُمْ ۗ وَكَانَ ٱلْكَافِرُ عَلَىٰ رَبِّهِۦ ظَهِيرًا ﴿٥٥﴾
ئۇلار (يەنى مۇشرىكلار) اﷲ نى قويۇپ ئۇلارغا پايدىمۇ يەتكۈزەلمەيدىغان، زىيانمۇ يەتكۈزەلمەيدىغان نەرسىلەرگە چوقۇنىدۇ، ھەقىقەتنى رەت قىلغۇچى پەرۋەردىگارىغا ئاسىيلىق قىلىش بىلەن (شەيتان (ئازدۇرغۇچى)غا) ياردەم بەرگۈچىدۇر
وَمَآ أَرْسَلْنَـٰكَ إِلَّا مُبَشِّرًا وَنَذِيرًا ﴿٥٦﴾
سېنى پەقەت (مەڭگۈلۈك باغلار بىلەن) خۇش خەۋەر بەرگۈچى، (دوزاختىن) ئاگاھلاندۇرغۇچى قىلىپ ئەۋەتتۇق
قُلْ مَآ أَسْـَٔلُكُمْ عَلَيْهِ مِنْ أَجْرٍ إِلَّا مَن شَآءَ أَن يَتَّخِذَ إِلَىٰ رَبِّهِۦ سَبِيلًا ﴿٥٧﴾
ئېيتقىنكى، «مەن (اﷲ نىڭ ئەمرىنى) تەبلىغ قىلغانلىقىمغا سىلەردىن ھېچ ھەق تەلەپ قىلمايمەن، پەقەت (مېنىڭ تىلەيدىغىنىم شۇكى، مېنىڭ دىنىمغا ئەگىشىش بىلەن (پەرۋەردىگارى تەرەپكە يول ئېلىشنى خالايدىغانلار) يول ئالسۇن)»
وَتَوَكَّلْ عَلَى ٱلْحَىِّ ٱلَّذِى لَا يَمُوتُ وَسَبِّحْ بِحَمْدِهِۦ ۚ وَكَفَىٰ بِهِۦ بِذُنُوبِ عِبَادِهِۦ خَبِيرًا ﴿٥٨﴾
(ھەممە ئىشىڭدا) ئۆلمەيدىغان مەڭگۈ ھايات اﷲ قا يۆلەنگىن، ئۇنىڭغا ھەمد ئېيتىش بىلەن اﷲ نى پاك دەپ ئېتىقاد قىلغىن، ئۇ بەندىلەرنىڭ گۇناھلىرىنى تولۇق بىلىشتە يېتەرلىكتۇر
ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَمَا بَيْنَهُمَا فِى سِتَّةِ أَيَّامٍ ثُمَّ ٱسْتَوَىٰ عَلَى ٱلْعَرْشِ ۚ ٱلرَّحْمَـٰنُ فَسْـَٔلْ بِهِۦ خَبِيرًا ﴿٥٩﴾
اﷲ ئاسمانلارنى زېمىننى ۋە ئۇلارنىڭ ئارىسىدىكى نەرسىلەرنى ئالتە كۈندە ياراتتى، ئاندىن ئەرش ئۈستىدە (ئۆزىگە لايىق رەۋىشتە) قارار ئالدى، اﷲ ناھايىتى مەرھەمەتلىكتۇر، بۇنى (يەنى اﷲ نىڭ بۈيۈكلۈكىنى) بىلىدىغان ئادەمدىن سورىغىن (ساڭا اﷲ نىڭ سۈپەتلىرىنى ئېيتىپ بېرىدۇ)
وَإِذَا قِيلَ لَهُمُ ٱسْجُدُوا۟ لِلرَّحْمَـٰنِ قَالُوا۟ وَمَا ٱلرَّحْمَـٰنُ أَنَسْجُدُ لِمَا تَأْمُرُنَا وَزَادَهُمْ نُفُورًا ۩ ﴿٦٠﴾
ئۇلارغا (يەنى مۇشرىكلارغا): «رەھمانغا (يەنى مەرھەمەتلىك اﷲ قا) سەجدە قىلىڭلار!» دېيىلسە، ئۇلار: «رەھمان دېگەن نېمە؟ سەن بىزنى بۇيرۇغان نەرسىگە سەجدە قىلامدۇق؟» دەيدۇ. بۇ سۆز ئۇلارنى (ئىماندىن) تېخىمۇ يىراقلاشتۇرىدۇ
تَبَارَكَ ٱلَّذِى جَعَلَ فِى ٱلسَّمَآءِ بُرُوجًا وَجَعَلَ فِيهَا سِرَٰجًا وَقَمَرًا مُّنِيرًا ﴿٦١﴾
ئاسماندا بۇرۇجلەرنى، يېنىپ تۇرىدىغان چىراقنى (يەنى كۈننى) ۋە نۇرلۇق ئاينى ياراتقان اﷲ نىڭ بەرىكىتى بۈيۈكتۇر
وَهُوَ ٱلَّذِى جَعَلَ ٱلَّيْلَ وَٱلنَّهَارَ خِلْفَةً لِّمَنْ أَرَادَ أَن يَذَّكَّرَ أَوْ أَرَادَ شُكُورًا ﴿٦٢﴾
اﷲ (ئۆزىنىڭ قۇدرىتىنى) ئەسلىمەكچى بولغانلار ياكى (ئۆزىنىڭ نېمىتىگە) شۈكۈر قىلماقچى بولغانلار ئۈچۈن كېچە بىلەن كۈندۈزنى ئالمىشىپ تۇرىدىغان قىلىپ بەردى
وَعِبَادُ ٱلرَّحْمَـٰنِ ٱلَّذِينَ يَمْشُونَ عَلَى ٱلْأَرْضِ هَوْنًا وَإِذَا خَاطَبَهُمُ ٱلْجَـٰهِلُونَ قَالُوا۟ سَلَـٰمًا ﴿٦٣﴾
مەرھەمەتلىك اﷲ نىڭ (ياخشى كۆرىدىغان) بەندىلىرى زېمىندا ئۆزلىرىنى تۆۋەن تۇتۇپ تەمكىنلىك بىلەن ماڭىدۇ، نادانلار ئۇلارغا (ياقتۇرمايدىغان) سۆز قىلسا، ئۇلار: «سىلەرگە ئامانلىق تىلەيمىز» دەيدۇ (يەنى گۇناھ بولمايدىغان سۆزلەرنى قىلىدۇ)
وَٱلَّذِينَ يَبِيتُونَ لِرَبِّهِمْ سُجَّدًا وَقِيَـٰمًا ﴿٦٤﴾
ئۇلار كېچىلەرنى پەرۋەردىگارىغا سەجدە قىلىش ۋە قىيامدا تۇرۇش بىلەن) يەنى ناماز ئوقۇش بىلەن) ئۆتكۈزىدۇ
وَٱلَّذِينَ يَقُولُونَ رَبَّنَا ٱصْرِفْ عَنَّا عَذَابَ جَهَنَّمَ ۖ إِنَّ عَذَابَهَا كَانَ غَرَامًا ﴿٦٥﴾
ئۇلار (يەنى اﷲ ياخشى كۆرىدىغان بەندىلەر) ئېيتىدۇ: «پەرۋەردىگارىمىز! بىزدىن مەڭگۈلۈك ئوت ئازابىنى دەپئى قىلغىن، مەڭگۈلۈك ئوتنىڭ ئازابى ئىشەنچلىك ھالدا (سېنىڭ دۈشمەنلىرىڭدىن) ئايرىلمايدۇ
إِنَّهَا سَآءَتْ مُسْتَقَرًّا وَمُقَامًا ﴿٦٦﴾
مەڭگۈلۈك ئوت ئىشەنچلىك ھالدا يامان قارارگاھتۇر، يامان جايدۇر»
وَٱلَّذِينَ إِذَآ أَنفَقُوا۟ لَمْ يُسْرِفُوا۟ وَلَمْ يَقْتُرُوا۟ وَكَانَ بَيْنَ ذَٰلِكَ قَوَامًا ﴿٦٧﴾
ئۇلار (يەنى اﷲ ياخشى كۆرىدىغان بەندىلەر) خىراجەت قىلغاندا، ئىسراپچىلىقمۇ قىلمايدۇ، بېخىللىقمۇ قىلمايدۇ، ئوتتۇراھال خىراجەت قىلىدۇ
وَٱلَّذِينَ لَا يَدْعُونَ مَعَ ٱللَّهِ إِلَـٰهًا ءَاخَرَ وَلَا يَقْتُلُونَ ٱلنَّفْسَ ٱلَّتِى حَرَّمَ ٱللَّهُ إِلَّا بِٱلْحَقِّ وَلَا يَزْنُونَ ۚ وَمَن يَفْعَلْ ذَٰلِكَ يَلْقَ أَثَامًا ﴿٦٨﴾
ئۇلار اﷲ قا ئىككىنچى بىر مەبۇدنى شېرىك قىلمايدۇ، اﷲ ھارام قىلغان ناھەق ئادەم ئۆلتۈرۈش ئىشىنى قىلمايدۇ، زىنا قىلمايدۇ، كىمكى بۇ (گۇناھلار) نى قىلىدىكەن، (ئاخىرەتتە) ئۇ جازاغا ئۇچرايدۇ
يُضَـٰعَفْ لَهُ ٱلْعَذَابُ يَوْمَ ٱلْقِيَـٰمَةِ وَيَخْلُدْ فِيهِۦ مُهَانًا ﴿٦٩﴾
قىيامەت كۈنى ئۇنىڭغا ھەسسىلەپ ئازاب قىلىنىدۇ، ئۇ مەڭگۈ ئازاب ئىچىدە خارلانغان ھالدا قالىدۇ
إِلَّا مَن تَابَ وَءَامَنَ وَعَمِلَ عَمَلًا صَـٰلِحًا فَأُو۟لَـٰٓئِكَ يُبَدِّلُ ٱللَّهُ سَيِّـَٔاتِهِمْ حَسَنَـٰتٍ ۗ وَكَانَ ٱللَّهُ غَفُورًا رَّحِيمًا ﴿٧٠﴾
پەقەت (ئۇلارنىڭ ئىچىدىن) (بۇ دۇنيادىكى چېغىدا) تەۋبە قىلغان، ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارلا بۇ ھالدا قالمايدۇ، اﷲ ئۇلارنىڭ گۇناھلىرىنى ياخشىلىققا ئالماشتۇرىدۇ، اﷲ تولىمۇ مەغپىرەت قىلغۇچىدۇر ۋە مەرھەمەت قىلغۇچىدۇر
وَمَن تَابَ وَعَمِلَ صَـٰلِحًا فَإِنَّهُۥ يَتُوبُ إِلَى ٱللَّهِ مَتَابًا ﴿٧١﴾
كىمكى (گۇناھلىرىغا) تەۋبە قىلىدىكەن ۋە ئەمەلىنى تۈزەيدىكەن، ئۇ اﷲ قا يۈزلەنگەن بولىدۇ (يەنى اﷲ ئۇ نىڭ تەۋبىسىنى قوبۇل قىلىپ، ئۇنىڭدىن رازى بولىدۇ)
وَٱلَّذِينَ لَا يَشْهَدُونَ ٱلزُّورَ وَإِذَا مَرُّوا۟ بِٱللَّغْوِ مَرُّوا۟ كِرَامًا ﴿٧٢﴾
ئۇلار (يەنى اﷲ ياخشى كۆرىدىغان بەندىلەر) يالغان گۇۋاھلىق بەرمەيدۇ، يامان سۆزنى ئاڭلاپ قالغان چاغدا ئالىيجانابلىق بىلەن ئۆتۈپ كېتىدۇ
وَٱلَّذِينَ إِذَا ذُكِّرُوا۟ بِـَٔايَـٰتِ رَبِّهِمْ لَمْ يَخِرُّوا۟ عَلَيْهَا صُمًّا وَعُمْيَانًا ﴿٧٣﴾
ئۇلارغا پەرۋەردىگارىنىڭ ئايەتلىرى بىلەن ۋەز - نەسىھەت قىلىنسا، گاس، كور بولۇۋالمايدۇ (يەنى ئۇنى بېرىلىپ ئىخلاس بىلەن ئاڭلايدۇ)
وَٱلَّذِينَ يَقُولُونَ رَبَّنَا هَبْ لَنَا مِنْ أَزْوَٰجِنَا وَذُرِّيَّـٰتِنَا قُرَّةَ أَعْيُنٍ وَٱجْعَلْنَا لِلْمُتَّقِينَ إِمَامًا ﴿٧٤﴾
ئۇلار: «ئى پەرۋەردىگارىمىز! بىزگە ئاياللىرىمىز ۋە ئەۋلاتلىرىمىز ئارقىلىق شادلىق بېغىشلىشىڭنى (يەنى بىزگە ساڭا ئىتائەتمەن پەرزەنت ئاتا قىلىشىڭنى) تىلەيمىز، بىزنى تەقۋادارلارنىڭ پېشىۋاسى (يەنى تەقۋادارلارنىڭ نەمۇنىسى، ياخشىلىققا دەۋەت قىلغۇچى) قىلغىن» دەيدۇ
أُو۟لَـٰٓئِكَ يُجْزَوْنَ ٱلْغُرْفَةَ بِمَا صَبَرُوا۟ وَيُلَقَّوْنَ فِيهَا تَحِيَّةً وَسَلَـٰمًا ﴿٧٥﴾
ئەنە شۇلار سەۋرلىك بولغانلىقى ئۈچۈن مەڭگۈلۈك باغلار بىلەن مۇكاپاتلىنىدۇ، ئۇلار مەڭگۈلۈك باغلارتە (خۇدانىڭ ئەلچى پەرىشتىلىرى تەرىپىدىن قىلىنغان) دۇئا ۋە سالام بىلەن قارشى ئېلىنىدۇ
خَـٰلِدِينَ فِيهَا ۚ حَسُنَتْ مُسْتَقَرًّا وَمُقَامًا ﴿٧٦﴾
ئۇلار مەڭگۈلۈك باغلارتە مەڭگۈ قالىدۇ، مەڭگۈلۈك باغلار نېمىدېگەن گۈزەل قارارگاھ! نېمىدېگەن گۈزەل جاي!
قُلْ مَا يَعْبَؤُا۟ بِكُمْ رَبِّى لَوْلَا دُعَآؤُكُمْ ۖ فَقَدْ كَذَّبْتُمْ فَسَوْفَ يَكُونُ لِزَامًۢا ﴿٧٧﴾
ئېيتقىنكى، «ئەگەر سىلەرنىڭ دۇئايىڭلار بولمىسا، پەرۋەردىگارىم سىلەرگە پەرۋا قىلمايدۇ، (ئى ھەقىقەتنى رەت قىلغۇچىلار! خۇدانىڭ ئەلچىسىنى) ئىنكار قىلدىڭلار، (ئاخىرەتتە) سىلەر قۇتۇلالمايدىغان ئازابقا دۇچار بولۇسىلەر»
