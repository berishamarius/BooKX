۞
الواقعة
Al-Waqi'ah · The Inevitable
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
إِذَا وَقَعَتِ الْوَاقِعَةُ ﴿١﴾
جب واقع ہونے والی واقع ہوجائے
لَيْسَ لِوَقْعَتِهَا كَاذِبَةٌ ﴿٢﴾
اس کے واقع ہونے میں کچھ جھوٹ نہیں
خَافِضَةٌ رَّافِعَةٌ ﴿٣﴾
کسی کو پست کرے کسی کو بلند
إِذَا رُجَّتِ الْأَرْضُ رَجًّا ﴿٤﴾
جب زمین بھونچال سے لرزنے لگے
وَبُسَّتِ الْجِبَالُ بَسًّا ﴿٥﴾
اور پہاڑ ٹوٹ کر ریزہ ریزہ ہوجائیں
فَكَانَتْ هَبَاءً مُّنبَثًّا ﴿٦﴾
پھر غبار ہو کر اُڑنے لگیں
وَكُنتُمْ أَزْوَاجًا ثَلَاثَةً ﴿٧﴾
اور تم لوگ تین قسم ہوجاؤ
فَأَصْحَابُ الْمَيْمَنَةِ مَا أَصْحَابُ الْمَيْمَنَةِ ﴿٨﴾
تو داہنے ہاتھ والے (سبحان الله) داہنے ہاتھ والے کیا (ہی چین میں) ہیں
وَأَصْحَابُ الْمَشْأَمَةِ مَا أَصْحَابُ الْمَشْأَمَةِ ﴿٩﴾
اور بائیں ہاتھ والے (افسوس) بائیں ہاتھ والے کیا (گرفتار عذاب) ہیں
وَالسَّابِقُونَ السَّابِقُونَ ﴿١٠﴾
اور جو آگے بڑھنے والے ہیں (ان کا کیا کہنا) وہ آگے ہی بڑھنے والے ہیں
أُولَٰئِكَ الْمُقَرَّبُونَ ﴿١١﴾
وہی (خدا کے) مقرب ہیں
فِي جَنَّاتِ النَّعِيمِ ﴿١٢﴾
نعمت کے بہشتوں میں
ثُلَّةٌ مِّنَ الْأَوَّلِينَ ﴿١٣﴾
وہ بہت سے تو اگلے لوگوں میں سے ہوں گے
وَقَلِيلٌ مِّنَ الْآخِرِينَ ﴿١٤﴾
اور تھوڑے سے پچھلوں میں سے
عَلَىٰ سُرُرٍ مَّوْضُونَةٍ ﴿١٥﴾
(لعل و یاقوت وغیرہ سے) جڑے ہوئے تختوں پر
مُّتَّكِئِينَ عَلَيْهَا مُتَقَابِلِينَ ﴿١٦﴾
آمنے سامنے تکیہ لگائے ہوئے
يَطُوفُ عَلَيْهِمْ وِلْدَانٌ مُّخَلَّدُونَ ﴿١٧﴾
نوجوان خدمت گزار جو ہمیشہ (ایک ہی حالت میں) رہیں گے ان کے آس پاس پھریں گے
بِأَكْوَابٍ وَأَبَارِيقَ وَكَأْسٍ مِّن مَّعِينٍ ﴿١٨﴾
یعنی آبخورے اور آفتابے اور صاف شراب کے گلاس لے لے کر
لَّا يُصَدَّعُونَ عَنْهَا وَلَا يُنزِفُونَ ﴿١٩﴾
اس سے نہ تو سر میں درد ہوگا اور نہ ان کی عقلیں زائل ہوں گی
وَفَاكِهَةٍ مِّمَّا يَتَخَيَّرُونَ ﴿٢٠﴾
اور میوے جس طرح کے ان کو پسند ہوں
وَلَحْمِ طَيْرٍ مِّمَّا يَشْتَهُونَ ﴿٢١﴾
اور پرندوں کا گوشت جس قسم کا ان کا جی چاہے
وَحُورٌ عِينٌ ﴿٢٢﴾
اور بڑی بڑی آنکھوں والی حوریں
كَأَمْثَالِ اللُّؤْلُؤِ الْمَكْنُونِ ﴿٢٣﴾
جیسے (حفاظت سے) تہہ کئے ہوئے (آب دار) موتی
جَزَاءً بِمَا كَانُوا يَعْمَلُونَ ﴿٢٤﴾
یہ ان اعمال کا بدلہ ہے جو وہ کرتے تھے
لَا يَسْمَعُونَ فِيهَا لَغْوًا وَلَا تَأْثِيمًا ﴿٢٥﴾
وہاں نہ بیہودہ بات سنیں گے اور نہ گالی گلوچ
إِلَّا قِيلًا سَلَامًا سَلَامًا ﴿٢٦﴾
ہاں ان کا کلام سلام سلام (ہوگا)
وَأَصْحَابُ الْيَمِينِ مَا أَصْحَابُ الْيَمِينِ ﴿٢٧﴾
اور داہنے ہاتھ والے (سبحان الله) داہنے ہاتھ والے کیا (ہی عیش میں) ہیں
فِي سِدْرٍ مَّخْضُودٍ ﴿٢٨﴾
(یعنی) بےخار کی بیریوں
وَطَلْحٍ مَّنضُودٍ ﴿٢٩﴾
اور تہہ بہ تہہ کیلوں
وَظِلٍّ مَّمْدُودٍ ﴿٣٠﴾
اور لمبے لمبے سایوں
وَمَاءٍ مَّسْكُوبٍ ﴿٣١﴾
اور پانی کے جھرنوں
وَفَاكِهَةٍ كَثِيرَةٍ ﴿٣٢﴾
اور میوہ ہائے کثیرہ (کے باغوں) میں
لَّا مَقْطُوعَةٍ وَلَا مَمْنُوعَةٍ ﴿٣٣﴾
جو نہ کبھی ختم ہوں اور نہ ان سے کوئی روکے
وَفُرُشٍ مَّرْفُوعَةٍ ﴿٣٤﴾
اور اونچے اونچے فرشوں میں
إِنَّا أَنشَأْنَاهُنَّ إِنشَاءً ﴿٣٥﴾
ہم نے ان (حوروں) کو پیدا کیا
فَجَعَلْنَاهُنَّ أَبْكَارًا ﴿٣٦﴾
تو ان کو کنواریاں بنایا
عُرُبًا أَتْرَابًا ﴿٣٧﴾
(اور شوہروں کی) پیاریاں اور ہم عمر
لِّأَصْحَابِ الْيَمِينِ ﴿٣٨﴾
یعنی داہنے ہاتھ والوں کے لئے
ثُلَّةٌ مِّنَ الْأَوَّلِينَ ﴿٣٩﴾
(یہ) بہت سے اگلے لوگوں میں سے ہیں
وَثُلَّةٌ مِّنَ الْآخِرِينَ ﴿٤٠﴾
اور بہت سے پچھلوں میں سے
وَأَصْحَابُ الشِّمَالِ مَا أَصْحَابُ الشِّمَالِ ﴿٤١﴾
اور بائیں ہاتھ والے (افسوس) بائیں ہاتھ والے کیا (ہی عذاب میں) ہیں
فِي سَمُومٍ وَحَمِيمٍ ﴿٤٢﴾
(یعنی دوزخ کی) لپٹ اور کھولتے ہوئے پانی میں
وَظِلٍّ مِّن يَحْمُومٍ ﴿٤٣﴾
اور سیاہ دھوئیں کے سائے میں
لَّا بَارِدٍ وَلَا كَرِيمٍ ﴿٤٤﴾
(جو) نہ ٹھنڈا (ہے) نہ خوشنما
إِنَّهُمْ كَانُوا قَبْلَ ذَٰلِكَ مُتْرَفِينَ ﴿٤٥﴾
یہ لوگ اس سے پہلے عیشِ نعیم میں پڑے ہوئے تھے
وَكَانُوا يُصِرُّونَ عَلَى الْحِنثِ الْعَظِيمِ ﴿٤٦﴾
اور گناہ عظیم پر اڑے ہوئے تھے
وَكَانُوا يَقُولُونَ أَئِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَامًا أَإِنَّا لَمَبْعُوثُونَ ﴿٤٧﴾
اور کہا کرتے تھے کہ بھلا جب ہم مرگئے اور مٹی ہوگئے اور ہڈیاں (ہی ہڈیاں رہ گئے) تو کیا ہمیں پھر اُٹھنا ہوگا؟
أَوَآبَاؤُنَا الْأَوَّلُونَ ﴿٤٨﴾
اور کیا ہمارے باپ دادا کو بھی؟
قُلْ إِنَّ الْأَوَّلِينَ وَالْآخِرِينَ ﴿٤٩﴾
کہہ دو کہ بےشک پہلے اور پچھلے
لَمَجْمُوعُونَ إِلَىٰ مِيقَاتِ يَوْمٍ مَّعْلُومٍ ﴿٥٠﴾
(سب) ایک روز مقرر کے وقت پر جمع کئے جائیں گے
ثُمَّ إِنَّكُمْ أَيُّهَا الضَّالُّونَ الْمُكَذِّبُونَ ﴿٥١﴾
پھر تم اے جھٹلانے والے گمرا ہو!
لَآكِلُونَ مِن شَجَرٍ مِّن زَقُّومٍ ﴿٥٢﴾
تھوہر کے درخت کھاؤ گے
فَمَالِئُونَ مِنْهَا الْبُطُونَ ﴿٥٣﴾
اور اسی سے پیٹ بھرو گے
فَشَارِبُونَ عَلَيْهِ مِنَ الْحَمِيمِ ﴿٥٤﴾
اور اس پر کھولتا ہوا پانی پیو گے
فَشَارِبُونَ شُرْبَ الْهِيمِ ﴿٥٥﴾
اور پیو گے بھی تو اس طرح جیسے پیاسے اونٹ پیتے ہیں
هَٰذَا نُزُلُهُمْ يَوْمَ الدِّينِ ﴿٥٦﴾
جزا کے دن یہ ان کی ضیافت ہوگی
نَحْنُ خَلَقْنَاكُمْ فَلَوْلَا تُصَدِّقُونَ ﴿٥٧﴾
ہم نے تم کو (پہلی بار بھی تو) پیدا کیا ہے تو تم (دوبارہ اُٹھنے کو) کیوں سچ نہیں سمجھتے؟
أَفَرَأَيْتُم مَّا تُمْنُونَ ﴿٥٨﴾
دیکھو تو کہ جس (نطفے) کو تم (عورتوں کے رحم میں) ڈالتے ہو
أَأَنتُمْ تَخْلُقُونَهُ أَمْ نَحْنُ الْخَالِقُونَ ﴿٥٩﴾
کیا تم اس (سے انسان) کو بناتے ہو یا ہم بناتے ہیں؟
نَحْنُ قَدَّرْنَا بَيْنَكُمُ الْمَوْتَ وَمَا نَحْنُ بِمَسْبُوقِينَ ﴿٦٠﴾
ہم نے تم میں مرنا ٹھہرا دیا ہے اور ہم اس (بات) سے عاجز نہیں
عَلَىٰ أَن نُّبَدِّلَ أَمْثَالَكُمْ وَنُنشِئَكُمْ فِي مَا لَا تَعْلَمُونَ ﴿٦١﴾
کہ تمہاری طرح کے اور لوگ تمہاری جگہ لے آئیں اور تم کو ایسے جہان میں جس کو تم نہیں جانتے پیدا کر دیں
وَلَقَدْ عَلِمْتُمُ النَّشْأَةَ الْأُولَىٰ فَلَوْلَا تَذَكَّرُونَ ﴿٦٢﴾
اور تم نے پہلی پیدائش تو جان ہی لی ہے۔ پھر تم سوچتے کیوں نہیں؟
أَفَرَأَيْتُم مَّا تَحْرُثُونَ ﴿٦٣﴾
بھلا دیکھو تو کہ جو کچھ تم بوتے ہو
أَأَنتُمْ تَزْرَعُونَهُ أَمْ نَحْنُ الزَّارِعُونَ ﴿٦٤﴾
تو کیا تم اسے اُگاتے ہو یا ہم اُگاتے ہیں؟
لَوْ نَشَاءُ لَجَعَلْنَاهُ حُطَامًا فَظَلْتُمْ تَفَكَّهُونَ ﴿٦٥﴾
اگر ہم چاہیں تو اسے چورا چورا کردیں اور تم باتیں بناتے رہ جاؤ
إِنَّا لَمُغْرَمُونَ ﴿٦٦﴾
(کہ ہائے) ہم تو مفت تاوان میں پھنس گئے
بَلْ نَحْنُ مَحْرُومُونَ ﴿٦٧﴾
بلکہ ہم ہیں ہی بےنصیب
أَفَرَأَيْتُمُ الْمَاءَ الَّذِي تَشْرَبُونَ ﴿٦٨﴾
بھلا دیکھو تو کہ جو پانی تم پیتے ہو
أَأَنتُمْ أَنزَلْتُمُوهُ مِنَ الْمُزْنِ أَمْ نَحْنُ الْمُنزِلُونَ ﴿٦٩﴾
کیا تم نے اس کو بادل سے نازل کیا ہے یا ہم نازل کرتے ہیں؟
لَوْ نَشَاءُ جَعَلْنَاهُ أُجَاجًا فَلَوْلَا تَشْكُرُونَ ﴿٧٠﴾
اگر ہم چاہیں تو ہم اسے کھاری کردیں پھر تم شکر کیوں نہیں کرتے؟
أَفَرَأَيْتُمُ النَّارَ الَّتِي تُورُونَ ﴿٧١﴾
بھلا دیکھو تو جو آگ تم درخت سے نکالتے ہو
أَأَنتُمْ أَنشَأْتُمْ شَجَرَتَهَا أَمْ نَحْنُ الْمُنشِئُونَ ﴿٧٢﴾
کیا تم نے اس کے درخت کو پیدا کیا ہے یا ہم پیدا کرتے ہیں؟
نَحْنُ جَعَلْنَاهَا تَذْكِرَةً وَمَتَاعًا لِّلْمُقْوِينَ ﴿٧٣﴾
ہم نے اسے یاد دلانے اور مسافروں کے برتنے کو بنایا ہے
فَسَبِّحْ بِاسْمِ رَبِّكَ الْعَظِيمِ ﴿٧٤﴾
تو تم اپنے پروردگار بزرگ کے نام کی تسبیح کرو
فَلَا أُقْسِمُ بِمَوَاقِعِ النُّجُومِ ﴿٧٥﴾
ہمیں تاروں کی منزلوں کی قسم
وَإِنَّهُ لَقَسَمٌ لَّوْ تَعْلَمُونَ عَظِيمٌ ﴿٧٦﴾
اور اگر تم سمجھو تو یہ بڑی قسم ہے
إِنَّهُ لَقُرْآنٌ كَرِيمٌ ﴿٧٧﴾
کہ یہ بڑے رتبے کا قرآن ہے
فِي كِتَابٍ مَّكْنُونٍ ﴿٧٨﴾
(جو) کتاب محفوظ میں (لکھا ہوا ہے)
لَّا يَمَسُّهُ إِلَّا الْمُطَهَّرُونَ ﴿٧٩﴾
اس کو وہی ہاتھ لگاتے ہیں جو پاک ہیں
تَنزِيلٌ مِّن رَّبِّ الْعَالَمِينَ ﴿٨٠﴾
پروردگار عالم کی طرف سے اُتارا گیا ہے
أَفَبِهَٰذَا الْحَدِيثِ أَنتُم مُّدْهِنُونَ ﴿٨١﴾
کیا تم اس کلام سے انکار کرتے ہو؟
وَتَجْعَلُونَ رِزْقَكُمْ أَنَّكُمْ تُكَذِّبُونَ ﴿٨٢﴾
اور اپنا وظیفہ یہ بناتے ہو کہ (اسے) جھٹلاتے ہو
فَلَوْلَا إِذَا بَلَغَتِ الْحُلْقُومَ ﴿٨٣﴾
بھلا جب روح گلے میں آ پہنچتی ہے
وَأَنتُمْ حِينَئِذٍ تَنظُرُونَ ﴿٨٤﴾
اور تم اس وقت کی (حالت کو) دیکھا کرتے ہو
وَنَحْنُ أَقْرَبُ إِلَيْهِ مِنكُمْ وَلَٰكِن لَّا تُبْصِرُونَ ﴿٨٥﴾
اور ہم اس (مرنے والے) سے تم سے بھی زیادہ نزدیک ہوتے ہیں لیکن تم کو نظر نہیں آتے
فَلَوْلَا إِن كُنتُمْ غَيْرَ مَدِينِينَ ﴿٨٦﴾
پس اگر تم کسی کے بس میں نہیں ہو
تَرْجِعُونَهَا إِن كُنتُمْ صَادِقِينَ ﴿٨٧﴾
تو اگر سچے ہو تو روح کو پھیر کیوں نہیں لیتے؟
فَأَمَّا إِن كَانَ مِنَ الْمُقَرَّبِينَ ﴿٨٨﴾
پھر اگر وہ (خدا کے) مقربوں میں سے ہے
فَرَوْحٌ وَرَيْحَانٌ وَجَنَّتُ نَعِيمٍ ﴿٨٩﴾
تو (اس کے لئے) آرام اور خوشبودار پھول اور نعمت کے باغ ہیں
وَأَمَّا إِن كَانَ مِنْ أَصْحَابِ الْيَمِينِ ﴿٩٠﴾
اور اگر وہ دائیں ہاتھ والوں میں سے ہے
فَسَلَامٌ لَّكَ مِنْ أَصْحَابِ الْيَمِينِ ﴿٩١﴾
تو (کہا جائے گا کہ) تجھ پر داہنے ہاتھ والوں کی طرف سے سلام
وَأَمَّا إِن كَانَ مِنَ الْمُكَذِّبِينَ الضَّالِّينَ ﴿٩٢﴾
اور اگر وہ جھٹلانے والے گمراہوں میں سے ہے
فَنُزُلٌ مِّنْ حَمِيمٍ ﴿٩٣﴾
تو (اس کے لئے) کھولتے پانی کی ضیافت ہے
وَتَصْلِيَةُ جَحِيمٍ ﴿٩٤﴾
اور جہنم میں داخل کیا جانا
إِنَّ هَٰذَا لَهُوَ حَقُّ الْيَقِينِ ﴿٩٥﴾
یہ (داخل کیا جانا یقیناً صحیح یعنی) حق الیقین ہے
فَسَبِّحْ بِاسْمِ رَبِّكَ الْعَظِيمِ ﴿٩٦﴾
تو تم اپنے پروردگار بزرگ کے نام کی تسبیح کرتے رہو
