بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
✦ ✦
حم ﴿١﴾
ح م
تَنزِيلٌ مِّنَ الرَّحْمَٰنِ الرَّحِيمِ ﴿٢﴾
یہ خدائے رحمان و رحیم کی طرف سے نازل کردہ چیز ہے
كِتَابٌ فُصِّلَتْ آيَاتُهُ قُرْآنًا عَرَبِيًّا لِّقَوْمٍ يَعْلَمُونَ ﴿٣﴾
ایک ایسی کتاب جس کی آیات خوب کھول کر بیان کی گئی ہیں، عربی زبان کا قرآن، اُن لوگوں کے لیے جو علم رکھتے ہیں
بَشِيرًا وَنَذِيرًا فَأَعْرَضَ أَكْثَرُهُمْ فَهُمْ لَا يَسْمَعُونَ ﴿٤﴾
بشارت دینے والا اور ڈرانے والا۔ مگر اِن لوگوں میں سے اکثر نے اس سے رُو گردانی کی اور وہ سُن کر نہیں دیتے
وَقَالُوا قُلُوبُنَا فِي أَكِنَّةٍ مِّمَّا تَدْعُونَا إِلَيْهِ وَفِي آذَانِنَا وَقْرٌ وَمِن بَيْنِنَا وَبَيْنِكَ حِجَابٌ فَاعْمَلْ إِنَّنَا عَامِلُونَ ﴿٥﴾
کہتے ہیں”جس چیز کی طرف تُو ہمیں بُلا رہا ہے اس کے لیے ہمارے دلوں پر غلاف چڑھے ہوئے ہیں، ہمارے کان بہرے ہوگئے ہیں، اور ہمارے اور تیرے درمیان ایک حجاب حائل ہو گیا ہے۔ تُو اپنا کام کر ، ہم اپنا کام کیے جائیں گے۔“
قُلْ إِنَّمَا أَنَا بَشَرٌ مِّثْلُكُمْ يُوحَىٰ إِلَيَّ أَنَّمَا إِلَٰهُكُمْ إِلَٰهٌ وَاحِدٌ فَاسْتَقِيمُوا إِلَيْهِ وَاسْتَغْفِرُوهُ ۗ وَوَيْلٌ لِّلْمُشْرِكِينَ ﴿٦﴾
اے نبی ؐ، اِن سے کہو، میں تو ایک بشر ہوں تم جیسا۔ مجھے وحی کے ذریعہ سے بتایا جاتا ہے کہ تمہارا خدا تو بس ایک ہی خدا ہے ، لہٰذا تم سیدھے اُسی کا رُخ اختیار کرو اور اس سے معافی چاہو۔ تباہی ہے اُن مشرکوں کے لیے
الَّذِينَ لَا يُؤْتُونَ الزَّكَاةَ وَهُم بِالْآخِرَةِ هُمْ كَافِرُونَ ﴿٧﴾
جو زکوٰة نہیں دیتے اور آخرت کے منکر ہیں
إِنَّ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ لَهُمْ أَجْرٌ غَيْرُ مَمْنُونٍ ﴿٨﴾
رہے وہ لوگ جنہوں نے مان لیا اور نیک اعمال کیے،  اُن کے لیے یقیناً ایسا اجر ہے جس کا سلسلہ ٹوٹنے والا نہیں ہے۔
قُلْ أَئِنَّكُمْ لَتَكْفُرُونَ بِالَّذِي خَلَقَ الْأَرْضَ فِي يَوْمَيْنِ وَتَجْعَلُونَ لَهُ أَندَادًا ۚ ذَٰلِكَ رَبُّ الْعَالَمِينَ ﴿٩﴾
اے نبیؐ! اِن سے کہو، کیا تم اُس خدا سے کفر کرتے ہو اور دوسروں کو اُس کا ہمسر ٹھیراتے ہو جس نے زمین کو دو دنوں میں بنا دیا؟ وہی تو سارے جہان والوں کا رب ہے
وَجَعَلَ فِيهَا رَوَاسِيَ مِن فَوْقِهَا وَبَارَكَ فِيهَا وَقَدَّرَ فِيهَا أَقْوَاتَهَا فِي أَرْبَعَةِ أَيَّامٍ سَوَاءً لِّلسَّائِلِينَ ﴿١٠﴾
اُس نے (زمین کو وجود میں لانے کے بعد)اوپر سے اس پر پہاڑ جمادیے اور اس میں برکتیں رکھ  دیں اور اس کے اندر سب مانگنے والوں کے لیے ہر ایک کی طلب و حاجت کے مطابق ٹھیک اندازے سے خوراک کا سامان مہیّا کر دیا۔ یہ سب کام چار دن میں ہو گئے۔
ثُمَّ اسْتَوَىٰ إِلَى السَّمَاءِ وَهِيَ دُخَانٌ فَقَالَ لَهَا وَلِلْأَرْضِ ائْتِيَا طَوْعًا أَوْ كَرْهًا قَالَتَا أَتَيْنَا طَائِعِينَ ﴿١١﴾
پھر وہ آسمان کی طرف متوجہ ہوا جو اُس وقت دھواں تھا۔ اُس نے آسمان اور زمین سے کہا”وجود میں آجاوٴ، خواہ تم چاہو یا نہ چاہو۔“ دونوں نے کہا”ہم آگئے فرمانبرداروں کی طرح۔“
فَقَضَاهُنَّ سَبْعَ سَمَاوَاتٍ فِي يَوْمَيْنِ وَأَوْحَىٰ فِي كُلِّ سَمَاءٍ أَمْرَهَا ۚ وَزَيَّنَّا السَّمَاءَ الدُّنْيَا بِمَصَابِيحَ وَحِفْظًا ۚ ذَٰلِكَ تَقْدِيرُ الْعَزِيزِ الْعَلِيمِ ﴿١٢﴾
تب اُس نے دو دن کے اندر  سات آسمان بنا دیے، اور ہر آسمان میں اُس کا قانون وحی کردیا۔ اور آسمانِ دنیا کو ہم نے چراغوں سے آراستہ کیا اور اسے خوب محفوظ کر دیا۔  یہ سب کچھ ایک زبردست علیم ہستی کا منصُوبہ ہے
فَإِنْ أَعْرَضُوا فَقُلْ أَنذَرْتُكُمْ صَاعِقَةً مِّثْلَ صَاعِقَةِ عَادٍ وَثَمُودَ ﴿١٣﴾
اب اگر یہ لوگ منہ موڑتے ہیں تو اِن سے کہہ دو کہ میں تم کو اُسی طرح کے ایک اچانک ٹوٹ پڑنے والے عذاب سے ڈراتا ہوں جیسا عاد اور ثمود پر نازل ہوا تھا
إِذْ جَاءَتْهُمُ الرُّسُلُ مِن بَيْنِ أَيْدِيهِمْ وَمِنْ خَلْفِهِمْ أَلَّا تَعْبُدُوا إِلَّا اللَّهَ ۖ قَالُوا لَوْ شَاءَ رَبُّنَا لَأَنزَلَ مَلَائِكَةً فَإِنَّا بِمَا أُرْسِلْتُم بِهِ كَافِرُونَ ﴿١٤﴾
جب خدا کے رسُول اُن کے پاس آگے اور پیچھے ، ہر طرف سے آئے اور انہیں سمجھایا کہ اللہ کے سوا کسی کی بندگی نہ کرو تو انہوں نے کہا”ہمارا ربّ چاہتا تو فرشتے بھیجتا، لہٰذا ہم اُس بات کو نہیں مانتے جس کے لیے تم بھیجے گئے ہو۔“
فَأَمَّا عَادٌ فَاسْتَكْبَرُوا فِي الْأَرْضِ بِغَيْرِ الْحَقِّ وَقَالُوا مَنْ أَشَدُّ مِنَّا قُوَّةً ۖ أَوَلَمْ يَرَوْا أَنَّ اللَّهَ الَّذِي خَلَقَهُمْ هُوَ أَشَدُّ مِنْهُمْ قُوَّةً ۖ وَكَانُوا بِآيَاتِنَا يَجْحَدُونَ ﴿١٥﴾
عاد کا حال یہ تھا کہ وہ زمین میں کسی حق کے بغیر بڑے بن بیٹھے اور کہنے لگے "کون ہے ہم سے زیادہ زور آور" اُن کو یہ نہ سوجھا کہ جس خدا نے ان کو پیدا کیا ہے وہ ان سے زیادہ زور آور ہے؟ وہ ہماری آیات کا انکار ہی کرتے رہے
فَأَرْسَلْنَا عَلَيْهِمْ رِيحًا صَرْصَرًا فِي أَيَّامٍ نَّحِسَاتٍ لِّنُذِيقَهُمْ عَذَابَ الْخِزْيِ فِي الْحَيَاةِ الدُّنْيَا ۖ وَلَعَذَابُ الْآخِرَةِ أَخْزَىٰ ۖ وَهُمْ لَا يُنصَرُونَ ﴿١٦﴾
آخر کار ہم نے چند منحوس دنوں میں  سخت طوفانی ہوا ان پر بھیج دی۔ تاکہ انہیں دنیا ہی کی زندگی میں ذلّت و رسوائی کے عذاب  کا مزا چکھا دیں، اور آخرت کا عذاب تو اس سے بھی زیادہ رُسوا کُن ہے ، وہاں کوئی ان کی مدد کرنے والا نہ ہو گا
وَأَمَّا ثَمُودُ فَهَدَيْنَاهُمْ فَاسْتَحَبُّوا الْعَمَىٰ عَلَى الْهُدَىٰ فَأَخَذَتْهُمْ صَاعِقَةُ الْعَذَابِ الْهُونِ بِمَا كَانُوا يَكْسِبُونَ ﴿١٧﴾
رہے ثمود، تو ان کے سامنے ہم نے راہ راست پیش کی مگر انہوں نے راستہ دیکھنے کے بجائے اندھا بنا رہنا پسند کیا آخر اُن کے کرتوتوں کی بدولت ذلت کا عذاب اُن پر ٹوٹ پڑا
وَنَجَّيْنَا الَّذِينَ آمَنُوا وَكَانُوا يَتَّقُونَ ﴿١٨﴾
اور ہم نے اُن لوگوں کو بچالیا جو ایمان لائے تھے اور گمراہی و بدعملی سے پرہیز کرتے تھے۔
وَيَوْمَ يُحْشَرُ أَعْدَاءُ اللَّهِ إِلَى النَّارِ فَهُمْ يُوزَعُونَ ﴿١٩﴾
اور ذرا اُس وقت کا خیال کرو جب اللہ کے دشمن دوزخ کی طرف جانے کے لیے گھیر لائے جائیں گے۔ اُن کے اگلوں کو پچھلوں کے آنے تک روک رکھا جائے گا،
حَتَّىٰ إِذَا مَا جَاءُوهَا شَهِدَ عَلَيْهِمْ سَمْعُهُمْ وَأَبْصَارُهُمْ وَجُلُودُهُم بِمَا كَانُوا يَعْمَلُونَ ﴿٢٠﴾
پھر جب سب وہاں پہنچ جائیں گے تو ان کے کان اور ان کی آنکھیں اور ان کے جسم کی کھالیں ان پر گواہی دیں گی کہ وہ دنیا میں کیا کچھ کرتے رہے ہیں۔
وَقَالُوا لِجُلُودِهِمْ لِمَ شَهِدتُّمْ عَلَيْنَا ۖ قَالُوا أَنطَقَنَا اللَّهُ الَّذِي أَنطَقَ كُلَّ شَيْءٍ وَهُوَ خَلَقَكُمْ أَوَّلَ مَرَّةٍ وَإِلَيْهِ تُرْجَعُونَ ﴿٢١﴾
وہ اپنے جسم کی کھالوں سے کہیں گے”تم نے ہمارے خلاف کیوں گواہی دی؟“ وہ جواب دیں گی”ہمیں اُسی خدا نے گویائی دی ہے جس نے ہر چیز کو گویا کر دیا ہے۔ اُسی نے تم کو پہلی مرتبہ پیدا کیا تھا اور اب اُسی کی طرف تم واپس لائے جارہے ہو
وَمَا كُنتُمْ تَسْتَتِرُونَ أَن يَشْهَدَ عَلَيْكُمْ سَمْعُكُمْ وَلَا أَبْصَارُكُمْ وَلَا جُلُودُكُمْ وَلَٰكِن ظَنَنتُمْ أَنَّ اللَّهَ لَا يَعْلَمُ كَثِيرًا مِّمَّا تَعْمَلُونَ ﴿٢٢﴾
تم دنیا میں جرائم کرتے وقت جب چھپتے تھے تو تمہیں یہ خیال نہ تھا کہ کبھی تمہارے اپنے کان اور تمہاری آنکھیں اور تمہارے جسم کی کھالیں تم پر گواہی دیں گی بلکہ تم نے تو یہ سمجھا تھا کہ تمہارے بہت سے اعمال کی اللہ کو بھی خبر نہیں ہے
وَذَٰلِكُمْ ظَنُّكُمُ الَّذِي ظَنَنتُم بِرَبِّكُمْ أَرْدَاكُمْ فَأَصْبَحْتُم مِّنَ الْخَاسِرِينَ ﴿٢٣﴾
تمہارا یہی گمان جو تم نے اپنے ربّ کے ساتھ کیا تھا، تمہیں لے ڈُوبا اور اسی کی بدولت تم خسارے میں پڑ گئے۔“
فَإِن يَصْبِرُوا فَالنَّارُ مَثْوًى لَّهُمْ ۖ وَإِن يَسْتَعْتِبُوا فَمَا هُم مِّنَ الْمُعْتَبِينَ ﴿٢٤﴾
اس حالت میں وہ صبر کریں (یا نہ کریں)آگ ہی ان کا ٹھکانا ہوگی، اور اگر رجوع کا موقع چاہیں گے کوئی موقع انہیں نہ دیا جائے گا۔
وَقَيَّضْنَا لَهُمْ قُرَنَاءَ فَزَيَّنُوا لَهُم مَّا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَحَقَّ عَلَيْهِمُ الْقَوْلُ فِي أُمَمٍ قَدْ خَلَتْ مِن قَبْلِهِم مِّنَ الْجِنِّ وَالْإِنسِ ۖ إِنَّهُمْ كَانُوا خَاسِرِينَ ﴿٢٥﴾
ہم نے ان پر ایسے ساتھی مسلّط کر دیے تھے جو انہیں آگے اور پیچھے ہر چیز خوشنما کر دکھاتے تھے، آخر کا ر اُن پر بھی وہی فیصلہٴ عذاب چسپاں ہو کر رہا جو ان سے پہلے گزرے ہوئے جِنّوں اور انسانوں کے گروہوں پر چسپاں ہو چکا تھا، یقیناًوہ خسارے میں رہ جانے والے تھے
وَقَالَ الَّذِينَ كَفَرُوا لَا تَسْمَعُوا لِهَٰذَا الْقُرْآنِ وَالْغَوْا فِيهِ لَعَلَّكُمْ تَغْلِبُونَ ﴿٢٦﴾
یہ منکرینِ حق کہتے ہیں”اِس قرآن کو ہرگز نہ سُنو اور جب یہ سنایا جائے تو اس میں خلل ڈالو، شاید کہ اس طرح تم غالب آجاوٴ۔“
فَلَنُذِيقَنَّ الَّذِينَ كَفَرُوا عَذَابًا شَدِيدًا وَلَنَجْزِيَنَّهُمْ أَسْوَأَ الَّذِي كَانُوا يَعْمَلُونَ ﴿٢٧﴾
اِن کافروں کو ہم سخت عذاب کا مزا چکھا کر رہیں گے اور جو بدترین حرکات یہ کرتے رہے ہیں ان کا پورا پورا بدلہ اِنہیں دیں گے
ذَٰلِكَ جَزَاءُ أَعْدَاءِ اللَّهِ النَّارُ ۖ لَهُمْ فِيهَا دَارُ الْخُلْدِ ۖ جَزَاءً بِمَا كَانُوا بِآيَاتِنَا يَجْحَدُونَ ﴿٢٨﴾
وہ دوزخ ہے جو اللہ کے دشمنوں کو بدلے میں ملے گی اُسی میں ہمیشہ ہمیشہ کے لیے اِن کا گھر ہو گا یہ ہے سزا اِس جرم کی کہ وہ ہماری آیات کا انکار کرتے ��ہے
وَقَالَ الَّذِينَ كَفَرُوا رَبَّنَا أَرِنَا اللَّذَيْنِ أَضَلَّانَا مِنَ الْجِنِّ وَالْإِنسِ نَجْعَلْهُمَا تَحْتَ أَقْدَامِنَا لِيَكُونَا مِنَ الْأَسْفَلِينَ ﴿٢٩﴾
وہاں یہ کافر کہیں گے کہ ”اے ہمارے ربّ، ذرا ہمیں دکھا دے اُن جِنّوں اور انسانوں کو جنہوں نے ہمیں گمراہ کیا تھا، ہم انہیں پاوٴں تلے روند ڈالیں گے تاکہ وہ خوب ذلیل و خوار ہوں۔“
إِنَّ الَّذِينَ قَالُوا رَبُّنَا اللَّهُ ثُمَّ اسْتَقَامُوا تَتَنَزَّلُ عَلَيْهِمُ الْمَلَائِكَةُ أَلَّا تَخَافُوا وَلَا تَحْزَنُوا وَأَبْشِرُوا بِالْجَنَّةِ الَّتِي كُنتُمْ تُوعَدُونَ ﴿٣٠﴾
جن لوگوں نے کہا کہ اللہ ہمارا ربّ ہے اور پھر وہ اس پر ثابت قدم رہے، یقیناً اُن پر فرشتے نازل ہوتے ہیں اور ان سے کہتے ہیں کہ”نہ ڈرو، نہ غم کرو، اور خوش ہو جاوٴ اُس جنّت کی بشارت سے جس کا تم سے وعدہ کیا گیا ہے
نَحْنُ أَوْلِيَاؤُكُمْ فِي الْحَيَاةِ الدُّنْيَا وَفِي الْآخِرَةِ ۖ وَلَكُمْ فِيهَا مَا تَشْتَهِي أَنفُسُكُمْ وَلَكُمْ فِيهَا مَا تَدَّعُونَ ﴿٣١﴾
ہم اِس دنیا کی زندگی میں بھی تمہارے ساتھی ہیں اور آخرت میں بھی، وہاں جو کچھ تم چاہو گے تمہیں ملے گا اور ہر چیز جس کی تم تمنا کرو گے وہ تمہاری ہوگی
نُزُلًا مِّنْ غَفُورٍ رَّحِيمٍ ﴿٣٢﴾
یہ ہے سامان ضیافت اُس ہستی کی طرف سے جو غفور اور رحیم ہے"
وَمَنْ أَحْسَنُ قَوْلًا مِّمَّن دَعَا إِلَى اللَّهِ وَعَمِلَ صَالِحًا وَقَالَ إِنَّنِي مِنَ الْمُسْلِمِينَ ﴿٣٣﴾
اور اُس شخص کی بات سے اچھی بات اور کس کی ہو گی جس نے اللہ کی طرف بُلایا اور نیک عمل کیا اور کہا کہ میں مسلمان ہوں۔
وَلَا تَسْتَوِي الْحَسَنَةُ وَلَا السَّيِّئَةُ ۚ ادْفَعْ بِالَّتِي هِيَ أَحْسَنُ فَإِذَا الَّذِي بَيْنَكَ وَبَيْنَهُ عَدَاوَةٌ كَأَنَّهُ وَلِيٌّ حَمِيمٌ ﴿٣٤﴾
اور اے نبیؐ ،نیکی اور بدی یکساں نہیں ہیں۔ تم بدی کو اُس نیکی سے دفع کرو جو بہترین ہو۔ تم دیکھو گے کہ تمہارے ساتھ جس کی عداوت پڑی ہوئی تھی وہ جگری دوست بن گیا ہے۔
وَمَا يُلَقَّاهَا إِلَّا الَّذِينَ صَبَرُوا وَمَا يُلَقَّاهَا إِلَّا ذُو حَظٍّ عَظِيمٍ ﴿٣٥﴾
یہ صفت نصیب نہیں ہوتی مگر اُن لوگوں کو جو صبر کرتے ہیں، اور یہ مقام حاصل نہیں ہوتا مگر اُن لوگوں کو جو بڑے نصیبے والے ہیں۔
وَإِمَّا يَنزَغَنَّكَ مِنَ الشَّيْطَانِ نَزْغٌ فَاسْتَعِذْ بِاللَّهِ ۖ إِنَّهُ هُوَ السَّمِيعُ الْعَلِيمُ ﴿٣٦﴾
اور اگر تم شیطان کی طرف سے کوئی اُکساہٹ محسُوس کرو تو اللہ کی پناہ مانگ لو، وہ سب کچھ سُنتا اور جانتا ہے۔
وَمِنْ آيَاتِهِ اللَّيْلُ وَالنَّهَارُ وَالشَّمْسُ وَالْقَمَرُ ۚ لَا تَسْجُدُوا لِلشَّمْسِ وَلَا لِلْقَمَرِ وَاسْجُدُوا لِلَّهِ الَّذِي خَلَقَهُنَّ إِن كُنتُمْ إِيَّاهُ تَعْبُدُونَ ﴿٣٧﴾
اللہ کی نشانیوں میں سے ہیں یہ رات اور دن اور سُورج اور چاند۔ سُورج اور چاند کو سجدہ نہ کرو بلکہ اُس خدا کو سجدہ کرو جس نے انہیں پیدا کیا ہے اگر فی الواقع تم اُسی کی عبادت کرنے والے ہو۔
فَإِنِ اسْتَكْبَرُوا فَالَّذِينَ عِندَ رَبِّكَ يُسَبِّحُونَ لَهُ بِاللَّيْلِ وَالنَّهَارِ وَهُمْ لَا يَسْأَمُونَ ﴿٣٨﴾
لیکن اگر یہ لوگ غرور میں آکر اپنی ہی بات پر اَڑے رہیں تو پروا نہیں، جو فرشتے تیرے ربّ کے مقرب ہیں وہ شب و روز اس کی تسبیح کر رہے ہیں اور کبھی نہیں تھکتے۔
وَمِنْ آيَاتِهِ أَنَّكَ تَرَى الْأَرْضَ خَاشِعَةً فَإِذَا أَنزَلْنَا عَلَيْهَا الْمَاءَ اهْتَزَّتْ وَرَبَتْ ۚ إِنَّ الَّذِي أَحْيَاهَا لَمُحْيِي الْمَوْتَىٰ ۚ إِنَّهُ عَلَىٰ كُلِّ شَيْءٍ قَدِيرٌ ﴿٣٩﴾
اور اللہ کی نشانیوں میں سے ایک یہ ہے کہ تم دیکھتے ہو  زمین سُونی پڑی ہوئی  ہے، پھر جونہی کہ ہم نے اس پر پانی برسایا، یکایک وہ پَھبَک اُٹھتی ہے اور پھُول جاتی ہے۔ یقیناً جو خدا اس مری ہوئی زمین کو جِلا اُٹھاتا ہے وہ مُردوں کو بھی زندگی بخشنے والا ہے۔ یقیناً وہ ہر چیز پر قدرت رکھتا ہے
إِنَّ الَّذِينَ يُلْحِدُونَ فِي آيَاتِنَا لَا يَخْفَوْنَ عَلَيْنَا ۗ أَفَمَن يُلْقَىٰ فِي النَّارِ خَيْرٌ أَم مَّن يَأْتِي آمِنًا يَوْمَ الْقِيَامَةِ ۚ اعْمَلُوا مَا شِئْتُمْ ۖ إِنَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٤٠﴾
جو لوگ ہماری آیات کو اُلٹے معنی پہناتے ہیں وہ ہم سے کچھ چھُپے ہوئے نہیں ہیں۔ خود ہی سوچ لو کہ آیا وہ شخص بہتر ہے جو آگ میں جھونکا جانے والا ہے یا وہ جو قیامت کے روز امن کی حالت میں حاضر ہوگا ؟ کرتے رہو جو کچھ تم  چاہو، تمہاری ساری حرکتوں کو اللہ دیکھ رہا ہے
إِنَّ الَّذِينَ كَفَرُوا بِالذِّكْرِ لَمَّا جَاءَهُمْ ۖ وَإِنَّهُ لَكِتَابٌ عَزِيزٌ ﴿٤١﴾
یہ وہ لوگ ہیں جن کے سامنے کلامِ نصیحت آیا تو انہوں نے اسے ماننے سے انکار کردیا۔ مگر حقیقت یہ ہے کہ یہ ایک زبردست کتاب ہے،
لَّا يَأْتِيهِ الْبَاطِلُ مِن بَيْنِ يَدَيْهِ وَلَا مِنْ خَلْفِهِ ۖ تَنزِيلٌ مِّنْ حَكِيمٍ حَمِيدٍ ﴿٤٢﴾
باطل نہ سامنے سے اس پر آسکتا ہے نہ پیچھے سے، یہ ایک حکیم و حمید کی نازل کر دہ چیز ہے
مَّا يُقَالُ لَكَ إِلَّا مَا قَدْ قِيلَ لِلرُّسُلِ مِن قَبْلِكَ ۚ إِنَّ رَبَّكَ لَذُو مَغْفِرَةٍ وَذُو عِقَابٍ أَلِيمٍ ﴿٤٣﴾
اے نبیؐ ، تم سے جو کچھ کہا جا رہا ہے اس میں کوئی چیز بھی ایسی نہیں ہے جو تم سے پہلے گزرے ہوئے رسُولوں سے نہ کہی جا چکی ہو۔ بے شک تمہارا ربّ بڑا درگزر کرنے والا ہے، اور اس کے ساتھ بڑی دردناک سزا دینے والا بھی ہے
وَلَوْ جَعَلْنَاهُ قُرْآنًا أَعْجَمِيًّا لَّقَالُوا لَوْلَا فُصِّلَتْ آيَاتُهُ ۖ أَأَعْجَمِيٌّ وَعَرَبِيٌّ ۗ قُلْ هُوَ لِلَّذِينَ آمَنُوا هُدًى وَشِفَاءٌ ۖ وَالَّذِينَ لَا يُؤْمِنُونَ فِي آذَانِهِمْ وَقْرٌ وَهُوَ عَلَيْهِمْ عَمًى ۚ أُولَٰئِكَ يُنَادَوْنَ مِن مَّكَانٍ بَعِيدٍ ﴿٤٤﴾
اگر ہم اس کو عجمی قرآن بنا کر بھیجتے تو یہ لوگ کہتے”کیوں نہ اس کی آیات کھول کر بیان کی گئیں؟ کیا عجیب بات ہے کہ کلام عجمی ہے اور مخاطِب عربی۔“ اِن سے کہو ، یہ قرآن ایمان لانے والوں کے لیے تو ہدایت اور شفا ہے، مگر جو لوگ ایمان نہیں لاتےاُن کے لیے یہ کانوں کی  ڈاٹ اور آنکھوں کی پٹی ہے۔ ان کا حال تو ایسا ہے جیسے ان کو دُور سے پکارا جا رہا ہو۔
وَلَقَدْ آتَيْنَا مُوسَى الْكِتَابَ فَاخْتُلِفَ فِيهِ ۗ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ لَقُضِيَ بَيْنَهُمْ ۚ وَإِنَّهُمْ لَفِي شَكٍّ مِّنْهُ مُرِيبٍ ﴿٤٥﴾
اس سے پہلے ہم نے موسیٰؑ کو کتا ب دی تھی اور اس کے معاملے میں بھی یہی اختلاف ہوا تھا۔ اگر تیرے ربّ نے پہلے ہی ایک بات طے نہ کردی ہوتی تو ان اختلاف کرنے والوں کے درمیان فیصلہ چُکا دیا جاتا۔ اور حقیقت یہ ہے کہ یہ لوگ اُس کی طرف سے سخت اضطراب انگیز شک میں پڑے ہوئے ہیں۔
مَّنْ عَمِلَ صَالِحًا فَلِنَفْسِهِ ۖ وَمَنْ أَسَاءَ فَعَلَيْهَا ۗ وَمَا رَبُّكَ بِظَلَّامٍ لِّلْعَبِيدِ ﴿٤٦﴾
جو کوئی نیک عمل کرے گا اپنے ہی لیے اچھا کرے گا، جو بدی کرے گا اُس کا وبال اُسی پر ہوگا، اور تیرا ربّ اپنے بندوں کے حق میں ظالم نہیں ہے۔
إِلَيْهِ يُرَدُّ عِلْمُ السَّاعَةِ ۚ وَمَا تَخْرُجُ مِن ثَمَرَاتٍ مِّنْ أَكْمَامِهَا وَمَا تَحْمِلُ مِنْ أُنثَىٰ وَلَا تَضَعُ إِلَّا بِعِلْمِهِ ۚ وَيَوْمَ يُنَادِيهِمْ أَيْنَ شُرَكَائِي قَالُوا آذَنَّاكَ مَا مِنَّا مِن شَهِيدٍ ﴿٤٧﴾
اُس ساعت کا علم اللہ ہی کی طرف راجع ہوتا ہے، وہی اُن سارے پھَلوں کو جانتا ہے جو اپنے شگوفوں میں سے نکلتے ہیں، اُسی کو معلوم ہے کہ کونسی مادہ حاملہ ہوئی ہے اور کس نے بچہ جنا ہے۔ پھر جس روز وہ ان لوگوں کو پکارے گا کہ کہاں ہیں میرے وہ شریک؟ یہ کہیں گے، ”ہم عرض کر چکے ہیں، آج ہم میں سے کوئی اس کی گواہی دینے والا نہیں ہے۔“
وَضَلَّ عَنْهُم مَّا كَانُوا يَدْعُونَ مِن قَبْلُ ۖ وَظَنُّوا مَا لَهُم مِّن مَّحِيصٍ ﴿٤٨﴾
اُس وقت وہ سارے معبُود ان سے گم ہو جائیں گے جنہیں یہ اِس سے پہلے پُکارتے تھے، اور یہ لوگ سمجھ لیں گے کہ ان کے لیے اب کوئی جائے پناہ نہیں ہے
لَّا يَسْأَمُ الْإِنسَانُ مِن دُعَاءِ الْخَيْرِ وَإِن مَّسَّهُ الشَّرُّ فَيَئُوسٌ قَنُوطٌ ﴿٤٩﴾
انسان کبھی بھلائی کی دُعا مانگتے نہیں تھکتا، اور جب کوئی آفت اس پر آجاتی ہے تو مایوس و دل شکستہ ہوجاتا ہے
وَلَئِنْ أَذَقْنَاهُ رَحْمَةً مِّنَّا مِن بَعْدِ ضَرَّاءَ مَسَّتْهُ لَيَقُولَنَّ هَٰذَا لِي وَمَا أَظُنُّ السَّاعَةَ قَائِمَةً وَلَئِن رُّجِعْتُ إِلَىٰ رَبِّي إِنَّ لِي عِندَهُ لَلْحُسْنَىٰ ۚ فَلَنُنَبِّئَنَّ الَّذِينَ كَفَرُوا بِمَا عَمِلُوا وَلَنُذِيقَنَّهُم مِّنْ عَذَابٍ غَلِيظٍ ﴿٥٠﴾
مگر جونہی کہ سخت وقت  گزر جانے کے بعد ہم اسے اپنی رحمت کا مزا چکھاتے ہیں ، یہ کہتا ہے کہ”میں اِسی کا مستحق ہوں، اور میں نہیں سمجھتا کہ قیامت کبھی آئے گی، لیکن اگر واقعی میں اپنے ربّ کی طرف پلٹایا گیا تو وہاں بھی مزے کروں گا۔“ حالانکہ کُفر کرنے والوں کو لازماً ہم بتا کر رہیں گے کہ وہ کیا کر کے آئے ہیں اور انہیں ہم بڑے گندے عذاب کا مزا چکھائیں گے
وَإِذَا أَنْعَمْنَا عَلَى الْإِنسَانِ أَعْرَضَ وَنَأَىٰ بِجَانِبِهِ وَإِذَا مَسَّهُ الشَّرُّ فَذُو دُعَاءٍ عَرِيضٍ ﴿٥١﴾
انسان کو جب ہم نعمت دیتے ہیں تو وہ منہ پھیرتا ہے اور اکڑ جاتا ہے۔ اور جب اسے کوئی آفت چھُو جاتی ہے تو لمبی چوڑی دُعائیں کرنے لگتا ہے۔
قُلْ أَرَأَيْتُمْ إِن كَانَ مِنْ عِندِ اللَّهِ ثُمَّ كَفَرْتُم بِهِ مَنْ أَضَلُّ مِمَّنْ هُوَ فِي شِقَاقٍ بَعِيدٍ ﴿٥٢﴾
اے نبیؐ ، اِن سے کہو، کبھی تم نے یہ بھی سوچا کہ اگر واقعی یہ قرآن خدا ہی کی طرف سے ہُوا اور تم اِس کا انکار کرتے رہے تو اُس شخص سے بڑھ کر بھٹکا ہوا اور کون ہوگا جو اِس کی مخالفت میں دُور تک نکل گیا ہو؟
سَنُرِيهِمْ آيَاتِنَا فِي الْآفَاقِ وَفِي أَنفُسِهِمْ حَتَّىٰ يَتَبَيَّنَ لَهُمْ أَنَّهُ الْحَقُّ ۗ أَوَلَمْ يَكْفِ بِرَبِّكَ أَنَّهُ عَلَىٰ كُلِّ شَيْءٍ شَهِيدٌ ﴿٥٣﴾
عنقریب ہم ان کو اپنی نشانیاں آفاق میں بھی دکھائیں گے اور ان کے اپنے نفس میں بھی یہاں تک کہ ان پر یہ بات کھُل جائے گی کہ یہ قرآن واقعی برحق ہے۔ کیا یہ بات کافی نہیں ہے کہ تیرا ربّ ہر چیز کا شاہد ہے؟
أَلَا إِنَّهُمْ فِي مِرْيَةٍ مِّن لِّقَاءِ رَبِّهِمْ ۗ أَلَا إِنَّهُ بِكُلِّ شَيْءٍ مُّحِيطٌ ﴿٥٤﴾
آگاہ رہو، یہ لوگ اپنےربّ کی ملاقات میں شک رکھتے ہیں۔ سُن رکھو، وہ ہر چیز پر محیط ہے۔
