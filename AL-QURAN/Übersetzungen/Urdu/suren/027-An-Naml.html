بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
✦ ✦
طس ۚ تِلْكَ آيَاتُ الْقُرْآنِ وَكِتَابٍ مُّبِينٍ ﴿١﴾
ط۔س۔ یہ آیات ہیں قرآن اور کتابِ مبین کی
هُدًى وَبُشْرَىٰ لِلْمُؤْمِنِينَ ﴿٢﴾
ہدایت اور بشارت اُن ایمان لانے والوں کے لیے
الَّذِينَ يُقِيمُونَ الصَّلَاةَ وَيُؤْتُونَ الزَّكَاةَ وَهُم بِالْآخِرَةِ هُمْ يُوقِنُونَ ﴿٣﴾
جو نماز قائم کرتے اور زکوٰة دیتے ہیں، اور پھر وہ ایسے لوگ ہیں جو آخرت پر پُورا یقین رکھتے ہیں
إِنَّ الَّذِينَ لَا يُؤْمِنُونَ بِالْآخِرَةِ زَيَّنَّا لَهُمْ أَعْمَالَهُمْ فَهُمْ يَعْمَهُونَ ﴿٤﴾
حقیقت یہ ہے کہ جو لوگ آخرت کو نہیں مانتے ان کے لیے ہم نے اُن کے کرتُوتوں کو خوشنما بنا دیا ہے، اس لیے وہ بھٹکتے پھرتے ہیں
أُولَٰئِكَ الَّذِينَ لَهُمْ سُوءُ الْعَذَابِ وَهُمْ فِي الْآخِرَةِ هُمُ الْأَخْسَرُونَ ﴿٥﴾
یہ وہ لوگ ہیں جن کے لیے بُری سزا ہے اور آخرت میں یہی سب سے زیادہ خسارے میں رہنے والے ہیں
وَإِنَّكَ لَتُلَقَّى الْقُرْآنَ مِن لَّدُنْ حَكِيمٍ عَلِيمٍ ﴿٦﴾
اور (اے محمدؐ)بلا شبہہ تم یہ قرآن ایک حکیم و علیم ہستی کی طرف سے پا رہے ہو۔
إِذْ قَالَ مُوسَىٰ لِأَهْلِهِ إِنِّي آنَسْتُ نَارًا سَآتِيكُم مِّنْهَا بِخَبَرٍ أَوْ آتِيكُم بِشِهَابٍ قَبَسٍ لَّعَلَّكُمْ تَصْطَلُونَ ﴿٧﴾
(اِنہیں اُس وقت کا قصہ سُناوٴ)جب موسیٰؑ نے اپنے گھر والوں سے کہا کہ”مجھے ایک آگ سی نظر آئی ہے، میں ابھی یا تو وہاں سے کوئی خبر لے کر آتا ہوں یا کوئی انگارا چُن لاتا ہوں تاکہ تم لوگ گرم ہو سکو۔“
فَلَمَّا جَاءَهَا نُودِيَ أَن بُورِكَ مَن فِي النَّارِ وَمَنْ حَوْلَهَا وَسُبْحَانَ اللَّهِ رَبِّ الْعَالَمِينَ ﴿٨﴾
وہاں جو پہنچا تو ندا آئی کہ”مبارک ہے وہ جو اس آگ میں ہے اور جو اِس کے ماحول میں ہے۔ پاک ہے اللہ، سب جہان والوں کا پروردگار
يَا مُوسَىٰ إِنَّهُ أَنَا اللَّهُ الْعَزِيزُ الْحَكِيمُ ﴿٩﴾
اے موسیٰؑ، یہ میں ہوں اللہ، زبردست اور دانا
وَأَلْقِ عَصَاكَ ۚ فَلَمَّا رَآهَا تَهْتَزُّ كَأَنَّهَا جَانٌّ وَلَّىٰ مُدْبِرًا وَلَمْ يُعَقِّبْ ۚ يَا مُوسَىٰ لَا تَخَفْ إِنِّي لَا يَخَافُ لَدَيَّ الْمُرْسَلُونَ ﴿١٠﴾
اور پھینک تُو ذرا اپنی لاٹھی۔“ جُونہی کہ موسیٰ ؑ نے دیکھا لاٹھی سانپ کی طرح بل کھا رہی ہے تو پیٹھ پھیر کر بھاگا اور پیچھے مُڑ کر بھی نہ دیکھا۔”اے موسیٰؑ ، ڈرو نہیں۔ میرے حضُور رسُول ڈرا نہیں کرتے
إِلَّا مَن ظَلَمَ ثُمَّ بَدَّلَ حُسْنًا بَعْدَ سُوءٍ فَإِنِّي غَفُورٌ رَّحِيمٌ ﴿١١﴾
اِلّا یہ کہ کسی نے قصور کیا ہو۔ پھر اگر بُرائی کے بعد اُس نے بھلائی سے (اپنے فعل کو)بدل لیا تو میں معاف کرنے والا مہربان ہوں
وَأَدْخِلْ يَدَكَ فِي جَيْبِكَ تَخْرُجْ بَيْضَاءَ مِنْ غَيْرِ سُوءٍ ۖ فِي تِسْعِ آيَاتٍ إِلَىٰ فِرْعَوْنَ وَقَوْمِهِ ۚ إِنَّهُمْ كَانُوا قَوْمًا فَاسِقِينَ ﴿١٢﴾
اور ذرا اپنا ہاتھ اپنے گریبان میں تو ڈالو۔ چمکتا ہوا نکلے گا بغیر کسی تکلیف کے۔ یہ (دونشانیاں)نو نشانیوں میں سے ہیں فرعون اور اُس کی قوم کی طرف (لے جانے کے لیے) ، وہ بڑے بدکردار لوگ ہیں۔“
فَلَمَّا جَاءَتْهُمْ آيَاتُنَا مُبْصِرَةً قَالُوا هَٰذَا سِحْرٌ مُّبِينٌ ﴿١٣﴾
مگر جب ہماری کھلی کھلی نشانیاں اُن لوگوں کے سامنے آئیں تو انہوں نے کہا یہ تو کھلا جادو ہے
وَجَحَدُوا بِهَا وَاسْتَيْقَنَتْهَا أَنفُسُهُمْ ظُلْمًا وَعُلُوًّا ۚ فَانظُرْ كَيْفَ كَانَ عَاقِبَةُ الْمُفْسِدِينَ ﴿١٤﴾
انہوں نے سراسر ظلم اور غرور کی راہ سے ان نشانیوں کا انکار کیا حالانکہ دل ان کے قائل ہو چکے تھے۔ اب دیکھ لو کہ ان مفسدوں کا انجام کیسا ہوا
وَلَقَدْ آتَيْنَا دَاوُودَ وَسُلَيْمَانَ عِلْمًا ۖ وَقَالَا الْحَمْدُ لِلَّهِ الَّذِي فَضَّلَنَا عَلَىٰ كَثِيرٍ مِّنْ عِبَادِهِ الْمُؤْمِنِينَ ﴿١٥﴾
(دُوسری طرف)ہم نے داوٴدؑ و سلیمانؑ کو علم عطا کیا اور اُنہوں نے کہا کہ شکر ہے اُس خدا کا جس نے ہم کو اپنے بہت سے مومن بندوں پر فضیلت عطا کی۔
وَوَرِثَ سُلَيْمَانُ دَاوُودَ ۖ وَقَالَ يَا أَيُّهَا النَّاسُ عُلِّمْنَا مَنطِقَ الطَّيْرِ وَأُوتِينَا مِن كُلِّ شَيْءٍ ۖ إِنَّ هَٰذَا لَهُوَ الْفَضْلُ الْمُبِينُ ﴿١٦﴾
اور داوٴدؑ کا وارث سلیمانؑ ہوا۔ اور اس نے کہا”لوگو، ہمیں پرندوں کی بولیاں سکھائی گئی ہیں اور ہمیں ہر طرح کی چیزیں دی گئی ہیں، بے شک یہ (اللہ کا)نمایاں فضل ہے۔“
وَحُشِرَ لِسُلَيْمَانَ جُنُودُهُ مِنَ الْجِنِّ وَالْإِنسِ وَالطَّيْرِ فَهُمْ يُوزَعُونَ ﴿١٧﴾
سلیمانؑ کے لیے جن اور انسانوں اور پرندوں کے لشکر جمع کیے گئے تھے اور وہ پُورے ضبط میں رکھے جاتے تھے
حَتَّىٰ إِذَا أَتَوْا عَلَىٰ وَادِ النَّمْلِ قَالَتْ نَمْلَةٌ يَا أَيُّهَا النَّمْلُ ادْخُلُوا مَسَاكِنَكُمْ لَا يَحْطِمَنَّكُمْ سُلَيْمَانُ وَجُنُودُهُ وَهُمْ لَا يَشْعُرُونَ ﴿١٨﴾
(ایک مرتبہ وہ ان کے ساتھ کُوچ کر رہا تھا)یہاں تک کہ جب یہ سب چیونٹیوں کی وادی میں پہنچے تو ایک چیونٹی نے کہا”اے چیونٹیو، اپنے بلوں میں گھُس جاوٴ، کہیں ایسا نہ ہو کہ سلیمانؑ اور اس کے لشکر تمہیں کُچل ڈالیں اور انہیں خبر بھی نہ ہو۔“
فَتَبَسَّمَ ضَاحِكًا مِّن قَوْلِهَا وَقَالَ رَبِّ أَوْزِعْنِي أَنْ أَشْكُرَ نِعْمَتَكَ الَّتِي أَنْعَمْتَ عَلَيَّ وَعَلَىٰ وَالِدَيَّ وَأَنْ أَعْمَلَ صَالِحًا تَرْضَاهُ وَأَدْخِلْنِي بِرَحْمَتِكَ فِي عِبَادِكَ الصَّالِحِينَ ﴿١٩﴾
سلیمانؑ اس کی بات پر مُسکراتے ہوئے ہنس پڑا اور بولا۔۔۔۔”اے میرے ربّ، مجھے قابو میں رکھ کہ میں تیرے احسان کا شکر ادا کرتا رہوں جو تُو نے مجھ پر اور میرے والدین پر کیا ہے اور ایسا عملِ صالح کروں جو تجھے پسند آئے اور اپنی رحمت سے مجھ کو اپنے صالح بندوں میں داخل کر۔“
وَتَفَقَّدَ الطَّيْرَ فَقَالَ مَا لِيَ لَا أَرَى الْهُدْهُدَ أَمْ كَانَ مِنَ الْغَائِبِينَ ﴿٢٠﴾
(ایک اور موقع پر)سلیمانؑ نے پرندوں کا جائزہ لیا اور کہا”کیا بات ہے کہ میں فلاں ہُد ہُد کو نہیں دیکھ رہا ہوں۔ کیا وہ کہیں غائب ہو گیا ہے؟
لَأُعَذِّبَنَّهُ عَذَابًا شَدِيدًا أَوْ لَأَذْبَحَنَّهُ أَوْ لَيَأْتِيَنِّي بِسُلْطَانٍ مُّبِينٍ ﴿٢١﴾
میں اسے سخت سزا دوں گا، یا ذبح کر دوں گا، ورنہ اسے میرے سامنے معقول وجہ پیش کرنی ہوگی۔
فَمَكَثَ غَيْرَ بَعِيدٍ فَقَالَ أَحَطتُ بِمَا لَمْ تُحِطْ بِهِ وَجِئْتُكَ مِن سَبَإٍ بِنَبَإٍ يَقِينٍ ﴿٢٢﴾
کچھ زیادہ دیر نہ گزری تھی کہ اُس نے آکر کہا”میں نے وہ معلومات حاصل کی ہیں جو آپ کے عِلم میں نہیں ہیں۔ میں سَبا کے متعلق یقینی اطلاع لے کر آیا ہوں۔
إِنِّي وَجَدتُّ امْرَأَةً تَمْلِكُهُمْ وَأُوتِيَتْ مِن كُلِّ شَيْءٍ وَلَهَا عَرْشٌ عَظِيمٌ ﴿٢٣﴾
میں نے وہاں ایک عورت دیکھی جو اس قوم کی حکمران ہے اُس کو ہر طرح کا ساز و سامان بخشا گیا ہے اور اس کا تخت بڑا عظیم الشان ہے
وَجَدتُّهَا وَقَوْمَهَا يَسْجُدُونَ لِلشَّمْسِ مِن دُونِ اللَّهِ وَزَيَّنَ لَهُمُ الشَّيْطَانُ أَعْمَالَهُمْ فَصَدَّهُمْ عَنِ السَّبِيلِ فَهُمْ لَا يَهْتَدُونَ ﴿٢٤﴾
میں نے دیکھا کہ وہ اور اس کی قوم اللہ کے بجائے سُورج کے آگے سجدہ کرتی ہے“۔۔۔۔ شیطان نے ان کے اعمال ان کے لیے خوشنما بنا دیے اور انہیں شاہراہ سے روک دیا، اس وجہ سے وہ یہ سیدھا راستہ نہیں پاتے
أَلَّا يَسْجُدُوا لِلَّهِ الَّذِي يُخْرِجُ الْخَبْءَ فِي السَّمَاوَاتِ وَالْأَرْضِ وَيَعْلَمُ مَا تُخْفُونَ وَمَا تُعْلِنُونَ ﴿٢٥﴾
کہ اُس خدا کو سجدہ کریں جو آسمانوں اور زمین کی پوشیدہ چیزیں نکالتا  ہے اور سب کچھ جانتا ہے جسے تم لوگ چھُپاتے ہو اور ظاہر کرتے ہو۔
اللَّهُ لَا إِلَٰهَ إِلَّا هُوَ رَبُّ الْعَرْشِ الْعَظِيمِ ﴿٢٦﴾
اللہ کہ جس کے سوا کوئی مستحق ِ عبادت نہیں، جو عرشِ عظیم کا مالک ہے۔
قَالَ سَنَنظُرُ أَصَدَقْتَ أَمْ كُنتَ مِنَ الْكَاذِبِينَ ﴿٢٧﴾
سلیمانؑ نے کہا "ابھی ہم دیکھے لیتے ہیں کہ تو نے سچ کہا ہے یا تو جھوٹ بولنے والوں میں سے ہے
اذْهَب بِّكِتَابِي هَٰذَا فَأَلْقِهْ إِلَيْهِمْ ثُمَّ تَوَلَّ عَنْهُمْ فَانظُرْ مَاذَا يَرْجِعُونَ ﴿٢٨﴾
میرا یہ خط لےجا او ر اسے ان لوگوں کی طرف ڈال دے، پھر الگ ہٹ کر دیکھ کہ وہ کیا ردِ عمل ظاہر کرتے ہیں۔“
قَالَتْ يَا أَيُّهَا الْمَلَأُ إِنِّي أُلْقِيَ إِلَيَّ كِتَابٌ كَرِيمٌ ﴿٢٩﴾
ملکہ بولی "اے اہلِ دربار، میری طرف ایک بڑا اہم خط پھینکا گیا ہے
إِنَّهُ مِن سُلَيْمَانَ وَإِنَّهُ بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ ﴿٣٠﴾
وہ سلیمانؑ کی جانب سے ہے اور اللہ رحمٰن و رحیم کے نام سے شروع کیا گیا ہے
أَلَّا تَعْلُوا عَلَيَّ وَأْتُونِي مُسْلِمِينَ ﴿٣١﴾
مضمون یہ ہے کہ”میرے مقابلے میں سرکشی نہ کرو اور مسلم ہو کر میرے پاس حاضر ہو جاوٴ۔“
قَالَتْ يَا أَيُّهَا الْمَلَأُ أَفْتُونِي فِي أَمْرِي مَا كُنتُ قَاطِعَةً أَمْرًا حَتَّىٰ تَشْهَدُونِ ﴿٣٢﴾
(خط سُنا کر)ملکہ نے کہا”اے سردارانِ قوم، میرے اس معاملے میں مجھے مشورہ دو، میں کسی معاملہ کا فیصلہ تمہارے بغیر نہیں کرتی ہوں۔“
قَالُوا نَحْنُ أُولُو قُوَّةٍ وَأُولُو بَأْسٍ شَدِيدٍ وَالْأَمْرُ إِلَيْكِ فَانظُرِي مَاذَا تَأْمُرِينَ ﴿٣٣﴾
اُنہوں نے جواب دیا "ہم طاقت ور اور لڑنے والے لوگ ہیں آگے فیصلہ آپ کے ہاتھ میں ہے آپ خود دیکھ لیں کہ آپ کو کیا حکم دینا ہے"
قَالَتْ إِنَّ الْمُلُوكَ إِذَا دَخَلُوا قَرْيَةً أَفْسَدُوهَا وَجَعَلُوا أَعِزَّةَ أَهْلِهَا أَذِلَّةً ۖ وَكَذَٰلِكَ يَفْعَلُونَ ﴿٣٤﴾
ملکہ نے کہا کہ”بادشاہ جب کسی ملک میں گھُس آتے ہیں تو اسے خراب اور اس کے عزت والوں کو ذلیل کردیتے ہیں۔ یہی کچھ وہ کیا کرتے ہیں
وَإِنِّي مُرْسِلَةٌ إِلَيْهِم بِهَدِيَّةٍ فَنَاظِرَةٌ بِمَ يَرْجِعُ الْمُرْسَلُونَ ﴿٣٥﴾
میں اِن لوگوں کی طرف ایک ہدیہ بھیجتی ہوں، پھر دیکھتی ہوں کہ میرے ایلچی کیا جواب لے کر پلٹتے ہیں"
فَلَمَّا جَاءَ سُلَيْمَانَ قَالَ أَتُمِدُّونَنِ بِمَالٍ فَمَا آتَانِيَ اللَّهُ خَيْرٌ مِّمَّا آتَاكُم بَلْ أَنتُم بِهَدِيَّتِكُمْ تَفْرَحُونَ ﴿٣٦﴾
جب وہ (ملکہ کا سفیر)سلیمانؑ  کے ہاں پہنچا تو اس نے کہا”کیا تم لوگ مال سے میری مدد کرنا چاہتے ہو؟ جو کچھ خدا نے مجھے دے رکھا ہے وہ اس سے بہت زیادہ ہے جو تمہیں دیا ہے۔ تمہارا ہدیہ تمہی کو مبارک رہے
ارْجِعْ إِلَيْهِمْ فَلَنَأْتِيَنَّهُم بِجُنُودٍ لَّا قِبَلَ لَهُم بِهَا وَلَنُخْرِجَنَّهُم مِّنْهَا أَذِلَّةً وَهُمْ صَاغِرُونَ ﴿٣٧﴾
(اے سفیر)واپس جا اپنے بھیجنے والوں کی طرف۔ ہم ان پر ایسے لشکر لے کر آئیں گے  جن کا مقابلہ وہ نہ کر سکیں گے اور ہم انہیں ایسی ذلّت کے ساتھ وہاں سے نکالیں گے کہ وہ خوار ہو کر رہ جائیں گے۔“
قَالَ يَا أَيُّهَا الْمَلَأُ أَيُّكُمْ يَأْتِينِي بِعَرْشِهَا قَبْلَ أَن يَأْتُونِي مُسْلِمِينَ ﴿٣٨﴾
سلیمانؑ نے کہا”اے اہلِ دربار، تم میں سےکون اس کا تخت میرے پاس لاتا ہے قبل اس کے کہ وہ لوگ مطیع ہو کر میرے پاس حاضر ہوں؟
قَالَ عِفْرِيتٌ مِّنَ الْجِنِّ أَنَا آتِيكَ بِهِ قَبْلَ أَن تَقُومَ مِن مَّقَامِكَ ۖ وَإِنِّي عَلَيْهِ لَقَوِيٌّ أَمِينٌ ﴿٣٩﴾
جِنوں میں سے ایک قوی ہیکل نے عرض کیا میں اسے حاضر کر دوں گا قبل اس کے کہ آپ اپنی جگہ سے اُٹھیں۔ میں اس کی طاقت رکھتا ہوں اور امانتدار ہوں۔“
قَالَ الَّذِي عِندَهُ عِلْمٌ مِّنَ الْكِتَابِ أَنَا آتِيكَ بِهِ قَبْلَ أَن يَرْتَدَّ إِلَيْكَ طَرْفُكَ ۚ فَلَمَّا رَآهُ مُسْتَقِرًّا عِندَهُ قَالَ هَٰذَا مِن فَضْلِ رَبِّي لِيَبْلُوَنِي أَأَشْكُرُ أَمْ أَكْفُرُ ۖ وَمَن شَكَرَ فَإِنَّمَا يَشْكُرُ لِنَفْسِهِ ۖ وَمَن كَفَرَ فَإِنَّ رَبِّي غَنِيٌّ كَرِيمٌ ﴿٤٠﴾
جس شخص کے پاس کتاب کا ایک علم تھا وہ بولا”میں آپ کی پلک جھپکنے سے پہلے اسے لائے دیتا ہوں۔“ جُونہی کہ سلیمان ؑ نے وہ تخت اپنے پاس رکھا ہوا دیکھا، وہ پکار اُٹھا”یہ میرے ربّ کا فضل ہے تاکہ وہ مجھے آزمائے کہ میں شکر کرتا ہوں یا کافرِ نعمت بن جاتا ہوں۔ اور جو کوئی شکر کرتا ہے اس کا شکر اس کے اپنے ہی لیے مفید ہے، ورنہ کوئی نا شکری کرے تو میرا ربّ بے نیاز اور اپنی ذات میں آپ بزرگ ہے۔“
قَالَ نَكِّرُوا لَهَا عَرْشَهَا نَنظُرْ أَتَهْتَدِي أَمْ تَكُونُ مِنَ الَّذِينَ لَا يَهْتَدُونَ ﴿٤١﴾
سلیمانؑ نے کہا”انجان طریقے سے اس کا تخت اس کے سامنے رکھ دو، دیکھیں وہ صحیح بات تک پہنچتی ہے یا اُن لوگوں میں سے ہے جو راہِ راست نہیں پاتے۔“
فَلَمَّا جَاءَتْ قِيلَ أَهَٰكَذَا عَرْشُكِ ۖ قَالَتْ كَأَنَّهُ هُوَ ۚ وَأُوتِينَا الْعِلْمَ مِن قَبْلِهَا وَكُنَّا مُسْلِمِينَ ﴿٤٢﴾
ملکہ جب حاضر ہوئی تو اس سے کہا گیا کیا تیرا تخت ایسا ہی ہے؟ وہ کہنے لگی”یہ تو گویا وہی ہے۔ ہم تو پہلے ہی جان گئے تھے اور ہم نے سرِ اطاعت جُھکا دیا تھا۔ (یا ہم مسلم ہو چکے تھے)۔“
وَصَدَّهَا مَا كَانَت تَّعْبُدُ مِن دُونِ اللَّهِ ۖ إِنَّهَا كَانَتْ مِن قَوْمٍ كَافِرِينَ ﴿٤٣﴾
اُس کو (ایمان لانے سے)جس چیز نے روک رکھا تھا وہ اُن معبُودوں کی عبادت تھی جنہیں وہ اللہ کے سوا پُوجتی تھی، کیونکہ وہ ایک کافر قوم سے تھی۔
قِيلَ لَهَا ادْخُلِي الصَّرْحَ ۖ فَلَمَّا رَأَتْهُ حَسِبَتْهُ لُجَّةً وَكَشَفَتْ عَن سَاقَيْهَا ۚ قَالَ إِنَّهُ صَرْحٌ مُّمَرَّدٌ مِّن قَوَارِيرَ ۗ قَالَتْ رَبِّ إِنِّي ظَلَمْتُ نَفْسِي وَأَسْلَمْتُ مَعَ سُلَيْمَانَ لِلَّهِ رَبِّ الْعَالَمِينَ ﴿٤٤﴾
اس سے کہا گیا کہ محل میں داخل ہو۔ اس نے جو دیکھا تو سمجھی کہ پانی کا حوض ہے اور اُترنے کےلیے اس نےاپنے پائینچے اُٹھا لیے۔ سلیمانؑ نے کہا یہ شیشے کا چکنا فرش ہے۔ اس پروہ پکار اُٹھی”اے میرے ربّ (آج تک)میں اپنے نفس پر بڑا ظلم کرتی رہی، اور اب میں نے سلیمانؑ کے ساتھ اللہ ربّ العالمین کی اطاعت قبول کر لی۔“
وَلَقَدْ أَرْسَلْنَا إِلَىٰ ثَمُودَ أَخَاهُمْ صَالِحًا أَنِ اعْبُدُوا اللَّهَ فَإِذَا هُمْ فَرِيقَانِ يَخْتَصِمُونَ ﴿٤٥﴾
اور ثمُود کی طرف ہم نے اُن کے بھائی صالحؑ کو (یہ پیغام دے کر)بھیجا کہ اللہ کی بندگی کرو، تو یکایک وہ دو مُتَخاصِم فریق بن گئے۔
قَالَ يَا قَوْمِ لِمَ تَسْتَعْجِلُونَ بِالسَّيِّئَةِ قَبْلَ الْحَسَنَةِ ۖ لَوْلَا تَسْتَغْفِرُونَ اللَّهَ لَعَلَّكُمْ تُرْحَمُونَ ﴿٤٦﴾
صالحؑ نے کہا”اے میری قوم کے لوگو، بھلائی سے پہلے بُرائی کے لیے کیوں جلدی مچاتے ہو؟ کیوں نہیں اللہ سے مغفرت طلب کرتے ؟ شایک کہ تم پر رحم فرمایا جائے؟“
قَالُوا اطَّيَّرْنَا بِكَ وَبِمَن مَّعَكَ ۚ قَالَ طَائِرُكُمْ عِندَ اللَّهِ ۖ بَلْ أَنتُمْ قَوْمٌ تُفْتَنُونَ ﴿٤٧﴾
انہوں نےکہا”ہم نے تو تم کو اور تمہارے ساتھیوں کو بدشگونی کا نشان پایا ہے۔“ صالحؑ نے جواب دیا”تمہارے نیک و بدشگون کا سرِشتہ تو اللہ کے پاس ہے۔ اصل بات یہ ہے کہ تم لوگوں کی آزمائش ہو رہی ہے۔“
وَكَانَ فِي الْمَدِينَةِ تِسْعَةُ رَهْطٍ يُفْسِدُونَ فِي الْأَرْضِ وَلَا يُصْلِحُونَ ﴿٤٨﴾
اُس شہر میں نو جتھے دار تھے جو ملک میں فساد پھیلاتے اور کوئی اصلاح کا کام نہ کرتے تھے
قَالُوا تَقَاسَمُوا بِاللَّهِ لَنُبَيِّتَنَّهُ وَأَهْلَهُ ثُمَّ لَنَقُولَنَّ لِوَلِيِّهِ مَا شَهِدْنَا مَهْلِكَ أَهْلِهِ وَإِنَّا لَصَادِقُونَ ﴿٤٩﴾
انہوں نے آپس میں کہا”خدا کی قسم کھا کر عہد کر لو کہ ہم صالحؑ اور اس کے گھر والوں پر شبخون ماریں گے اور پھر اس کے ولی سے کہہ دیں گے کہ ہم اس کے خاندان کی ہلاکت کے موقع پر موجود نہ تھے، ہم بالکل سچ کہتے ہیں۔“
وَمَكَرُوا مَكْرًا وَمَكَرْنَا مَكْرًا وَهُمْ لَا يَشْعُرُونَ ﴿٥٠﴾
یہ چال تو وہ چلے اور پھر ایک چال ہم نے چلی جس کی انہیں خبر نہ تھی۔
فَانظُرْ كَيْفَ كَانَ عَاقِبَةُ مَكْرِهِمْ أَنَّا دَمَّرْنَاهُمْ وَقَوْمَهُمْ أَجْمَعِينَ ﴿٥١﴾
اب دیکھ لو کہ ان کی چال کا انجام کیا ہوا ہم نے تباہ کر کے رکھ دیا اُن کو اور ان کی پوری قوم کو
فَتِلْكَ بُيُوتُهُمْ خَاوِيَةً بِمَا ظَلَمُوا ۗ إِنَّ فِي ذَٰلِكَ لَآيَةً لِّقَوْمٍ يَعْلَمُونَ ﴿٥٢﴾
وہ اُن کے گھر خالی پڑے ہیں اُس ظلم کی پاداش میں جو وہ کرتے تھے، اس میں ایک نشانِ عِبرت ہے اُن لوگوں کے لیے جو علم رکھتے ہیں۔
وَأَنجَيْنَا الَّذِينَ آمَنُوا وَكَانُوا يَتَّقُونَ ﴿٥٣﴾
اور بچا لیا ہم نے اُن لوگوں کو جو ایمان لائے تھے اور نافرمانی سے پرہیز کرتے تھے
وَلُوطًا إِذْ قَالَ لِقَوْمِهِ أَتَأْتُونَ الْفَاحِشَةَ وَأَنتُمْ تُبْصِرُونَ ﴿٥٤﴾
اور لوطؑ  کو ہم نے بھیجا۔ یاد کرو وہ وقت جب اس نے اپنی قوم سے کہا”کیا تم آنکھوں دیکھتے بدکاری کرتے ہو؟
أَئِنَّكُمْ لَتَأْتُونَ ا��رِّجَالَ شَهْوَةً مِّن دُونِ النِّسَاءِ ۚ بَلْ أَنتُمْ قَوْمٌ تَجْهَلُونَ ﴿٥٥﴾
کیا تمہارا یہی چلن ہے کہ عورتوں کو چھوڑ کر مردوں کے پاس شہوت رانی کے لیے جاتے ہو؟ حقیقت یہ ہے کہ تم لوگ سخت جہالت کا کام کرتے ہو۔“
فَمَا كَانَ جَوَابَ قَوْمِهِ إِلَّا أَن قَالُوا أَخْرِجُوا آلَ لُوطٍ مِّن قَرْيَتِكُمْ ۖ إِنَّهُمْ أُنَاسٌ يَتَطَهَّرُونَ ﴿٥٦﴾
مگر اُس کی قوم کا جواب اس کے سوا کچھ نہ تھا کہ انہوں نے کہا "نکال دو لوطؑ کے گھر والوں کو اپنی بستی سے، یہ بڑے پاکباز بنتے ہیں"
فَأَنجَيْنَاهُ وَأَهْلَهُ إِلَّا امْرَأَتَهُ قَدَّرْنَاهَا مِنَ الْغَابِرِينَ ﴿٥٧﴾
آخر کار ہم نے بچا لیا اُس کو اور اُس کے گھر والوں کو ، بجز اُس کی بیوی کے جس کا پیچھے رہ جانا ہم نے طے کر دیا تھا،
وَأَمْطَرْنَا عَلَيْهِم مَّطَرًا ۖ فَسَاءَ مَطَرُ الْمُنذَرِينَ ﴿٥٨﴾
اور برسائی اُن لوگوں پر ایک برسات، بہت ہی بری برسات تھی وہ اُن لوگوں کے حق میں جو متنبہ کیے جا چکے تھے
قُلِ الْحَمْدُ لِلَّهِ وَسَلَامٌ عَلَىٰ عِبَادِهِ الَّذِينَ اصْطَفَىٰ ۗ آللَّهُ خَيْرٌ أَمَّا يُشْرِكُونَ ﴿٥٩﴾
(اے نبیؐ)کہو، حمد ہے اللہ کے لیے اور سلام اُس کے اُن بندوں پر جنہیں اس نے برگزیدہ کیا۔(اِن سے پُوچھو)اللہ بہتر ہے یا وہ معبُود جنہیں یہ لوگ اس کا شریک بنا رہے ہیں؟
أَمَّنْ خَلَقَ السَّمَاوَاتِ وَالْأَرْضَ وَأَنزَلَ لَكُم مِّنَ السَّمَاءِ مَاءً فَأَنبَتْنَا بِهِ حَدَائِقَ ذَاتَ بَهْجَةٍ مَّا كَانَ لَكُمْ أَن تُنبِتُوا شَجَرَهَا ۗ أَإِلَٰهٌ مَّعَ اللَّهِ ۚ بَلْ هُمْ قَوْمٌ يَعْدِلُونَ ﴿٦٠﴾
بھَلا وہ کون ہے جس نےآسمانوں اور زمین کو پیدا کیا اور تمہارے لیے آسمان سے پانی برسایا پھر اس کے ذریعہ سے وہ خوشنما باغ اُگائے جن کے درختوں کا اُگانا تمہارے بس میں نہ تھا؟ کیا اللہ کے ساتھ کوئی دُوسرا خدا بھی (اِن کاموں میں شریک)ہے؟ (نہیں)، بلکہ یہی لوگ راہِ راست سے ہٹ کر چلے جا رہے ہیں
أَمَّن جَعَلَ الْأَرْضَ قَرَارًا وَجَعَلَ خِلَالَهَا أَنْهَارًا وَجَعَلَ لَهَا رَوَاسِيَ وَجَعَلَ بَيْنَ الْبَحْرَيْنِ حَاجِزًا ۗ أَإِلَٰهٌ مَّعَ اللَّهِ ۚ بَلْ أَكْثَرُهُمْ لَا يَعْلَمُونَ ﴿٦١﴾
اور وہ کون ہے جس نے زمین کو جائے قر��ر بنایا اور اس کے اندر دریا رواں کیے اور اس میں (پہاڑوں کی)میخیں گاڑ دیں اور پانی کے دو ذخیروں کے درمیان پردے حائل کر دیے؟ کیا اللہ کے ساتھ کوئی اور خدا بھی (ان کاموں میں شریک)ہے؟ نہیں، بلکہ ان میں سے اکثر لوگ نادان ہیں
أَمَّن يُجِيبُ الْمُضْطَرَّ إِذَا دَعَاهُ وَيَكْشِفُ السُّوءَ وَيَجْعَلُكُمْ خُلَفَاءَ الْأَرْضِ ۗ أَإِلَٰهٌ مَّعَ اللَّهِ ۚ قَلِيلًا مَّا تَذَكَّرُونَ ﴿٦٢﴾
کون ہے جو بے قرار کی دُعا سُنتا ہے جبکہ وہ اُسے پکارے اور کون اس کی تکلیف رفع کرتا ہے؟ اور (کون ہے جو)تمہیں زمین کا خلیفہ بناتا ہے؟ کیا اللہ کے ساتھ کوئی اور خدا بھی (یہ کام کرنے والا)ہے؟ تم لوگ کم ہی سوچتے ہو
أَمَّن يَهْدِيكُمْ فِي ظُلُمَاتِ الْبَرِّ وَالْبَحْرِ وَمَن يُرْسِلُ الرِّيَاحَ بُشْرًا بَيْنَ يَدَيْ رَحْمَتِهِ ۗ أَإِلَٰهٌ مَّعَ اللَّهِ ۚ تَعَالَى اللَّهُ عَمَّا يُشْرِكُونَ ﴿٦٣﴾
اور وہ کون ہے جو خشکی اور سمندر کی تاریکیوں میں تم کو راستہ دکھاتا ہے اور کون اپنی رحمت کے آگے ہواوٴں کو خوشخبری لے کر بھیجتا ہے؟ کیا اللہ کے ساتھ کوئی دُوسرا خدا بھی (یہ کام کرتا)ہے؟ بہت بالا و برتر ہے اللہ اس شرک سے جو یہ لوگ کرتے ہیں
أَمَّن يَبْدَأُ الْخَلْقَ ثُمَّ يُعِيدُهُ وَمَن يَرْزُقُكُم مِّنَ السَّمَاءِ وَالْأَرْضِ ۗ أَإِلَٰهٌ مَّعَ اللَّهِ ۚ قُلْ هَاتُوا بُرْهَانَكُمْ إِن كُنتُمْ صَادِقِينَ ﴿٦٤﴾
اور وہ کون ہے جو خلق کی ابتدا کرتا ہے اور پھر اس کا اعادہ کرتا ہے؟ اور کون تم کو آسمان اور زمین سے رزق دیتا ہے؟ کیا اللہ کے ساتھ کوئی اور خدا بھی (ان کاموں میں حصہ دار)ہے؟ کہو کہ لاوٴ اپنی دلیل اگر تم سچے ہو۔
قُل لَّا يَعْلَمُ مَن فِي السَّمَاوَاتِ وَالْأَرْضِ الْغَيْبَ إِلَّا اللَّهُ ۚ وَمَا يَشْعُرُونَ أَيَّانَ يُبْعَثُونَ ﴿٦٥﴾
اِن سے کہو، اللہ کے سوا آسمانوں اور زمین میں کوئی غیب کا عِلم نہیں رکھتا۔ اور وہ نہیں جانتے کہ کب وہ اُٹھائے جائیں گے۔
بَلِ ادَّارَكَ عِلْمُهُمْ فِي الْآخِرَةِ ۚ بَلْ هُمْ فِي شَكٍّ مِّنْهَا ۖ بَلْ هُم مِّنْهَا عَمُونَ ﴿٦٦﴾
بلکہ آخرت کا تو علم ہی اِن لوگوں سے گُم ہو گیا ہے، بلکہ یہ اس کی طرف سے شک میں ہیں، بلکہ یہ اُس  سے اندھے ہیں۔
وَقَالَ الَّذِينَ كَفَرُوا أَإِذَا كُنَّا تُرَابًا وَآبَاؤُنَا أَئِنَّا لَمُخْرَجُونَ ﴿٦٧﴾
یہ منکرین کہتے ہیں "کیا جب ہم اور ہمارے باپ دادا مٹی ہو چکے ہوں گے تو ہمیں واقعی قبروں سے نکالا جائے گا؟
لَقَدْ وُعِدْنَا هَٰذَا نَحْنُ وَآبَاؤُنَا مِن قَبْلُ إِنْ هَٰذَا إِلَّا أَسَاطِيرُ الْأَوَّلِينَ ﴿٦٨﴾
یہ خبریں ہم کو بھی بہت دی گئی ہیں اور پہلے ہمارے آباء اجداد کو بھی دی جاتی رہی ہیں، مگر یہ بس افسانے ہی افسانے ہیں جو اگلے وقتوں سے سُنتے چلے آ رہے ہیں"
قُلْ سِيرُوا فِي الْأَرْضِ فَانظُرُوا كَيْفَ كَانَ عَاقِبَةُ الْمُجْرِمِينَ ﴿٦٩﴾
کہو ذرا زمین میں چل پھر کر دیکھو کہ مجرموں کا کیا انجام ہو چکا ہے۔
وَلَا تَحْزَنْ عَلَيْهِمْ وَلَا تَكُن فِي ضَيْقٍ مِّمَّا يَمْكُرُونَ ﴿٧٠﴾
اے نبیؐ ، اِن کےحال پر رنج نہ کرو، اور نہ اِن کی چالوں پر دِل تنگ ہو
وَيَقُولُونَ مَتَىٰ هَٰذَا الْوَعْدُ إِن كُنتُمْ صَادِقِينَ ﴿٧١﴾
وہ کہتے ہیں کہ ”یہ دھمکی کب پُوری ہوگی اگر تم سچے ہو؟“
قُلْ عَسَىٰ أَن يَكُونَ رَدِفَ لَكُم بَعْضُ الَّذِي تَسْتَعْجِلُونَ ﴿٧٢﴾
کہو کیا عجب کہ جس عذاب کے لیے تم جلدی مچا رہے ہو اس کا ایک حصہ تمہارے قریب ہی آلگا ہو۔
وَإِنَّ رَبَّكَ لَذُو فَضْلٍ عَلَى النَّاسِ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَشْكُرُونَ ﴿٧٣﴾
حقیقت یہ ہے کہ تیرا ربّ تو لوگوں پر بڑا فضل فرمانے والا ہے مگر اکثر لوگ شکر نہیں کرتے۔
وَإِنَّ رَبَّكَ لَيَعْلَمُ مَا تُكِنُّ صُدُورُهُمْ وَمَا يُعْلِنُونَ ﴿٧٤﴾
بلا شُبہ تیرا ربّ خُوب جانتا ہے جو کچھ ان کے سینے اپنے اندر چھُپائے ہوئے ہیں اور جو کچھ وہ ظاہر کرتے ہیں۔
وَمَا مِنْ غَائِبَةٍ فِي السَّمَاءِ وَالْأَرْضِ إِلَّا فِي كِتَابٍ مُّبِينٍ ﴿٧٥﴾
آسمان و زمین کی کوئی پوشیدہ چیز ایسی نہیں ہے جو ایک واضح کتاب میں لکھی ہوئی موجود نہ ہو۔
إِنَّ هَٰذَا الْقُرْآنَ يَقُصُّ عَلَىٰ بَنِي إِسْرَائِيلَ أَكْثَرَ الَّذِي هُمْ فِيهِ يَخْتَلِفُونَ ﴿٧٦﴾
یہ واقعہ ہے کہ یہ قرآن بنی اسرائیل کو اکثراُن باتوں کی حقیقت بتاتا ہے جن میں وہ اختلاف رکھتے ہیں
وَإِنَّهُ لَهُدًى وَرَحْمَةٌ لِّلْمُؤْمِنِينَ ﴿٧٧﴾
اور یہ ہدایت اور رحمت ہے ایمان لانے والوں کے لیے۔
إِنَّ رَبَّكَ يَقْضِي بَيْنَهُم بِحُكْمِهِ ۚ وَهُوَ الْعَزِيزُ الْعَلِيمُ ﴿٧٨﴾
یقیناً (اِسی طرح)تیرا ربّ اِن لوگوں کے درمیان بھی اپنے حکم  سے فیصلہ کر دے گا اور وہ زبردست اور سب کچھ جاننے والا ہے۔
فَتَوَكَّلْ عَلَى اللَّهِ ۖ إِنَّكَ عَلَى الْحَقِّ الْمُبِينِ ﴿٧٩﴾
پس اے نبیؐ، اللہ پر بھروسا رکھو، یقیناً تم صریح حق پر ہو
إِنَّكَ لَا تُسْمِعُ الْمَوْتَىٰ وَلَا تُسْمِعُ الصُّمَّ الدُّعَاءَ إِذَا وَلَّوْا مُدْبِرِينَ ﴿٨٠﴾
تم مُردوں کو نہیں سُنا سکتے، نہ اُن بہروں تک اپنی پکار پہنچا سکتے ہو جو پیٹھ پھیر کر بھاگے جا رہے ہوں
وَمَا أَنتَ بِهَادِي الْعُمْيِ عَن ضَلَالَتِهِمْ ۖ إِن تُسْمِعُ إِلَّا مَن يُؤْمِنُ بِآيَاتِنَا فَهُم مُّسْلِمُونَ ﴿٨١﴾
اور نہ اندھوں کو راستہ بتا کر بھٹکنے سے بچا سکتے ہو۔ تم تو اپنی بات اُنہی لوگوں کو سُنا سکتے ہو  جو ہماری آیات پر ایمان لاتے ہیں اور پھر فرماں بردار بن جاتے ہیں
وَإِذَا وَقَعَ الْقَوْلُ عَلَيْهِمْ أَخْرَجْنَا لَهُمْ دَابَّةً مِّنَ الْأَرْضِ تُكَلِّمُهُمْ أَنَّ النَّاسَ كَانُوا بِآيَاتِنَا لَا يُوقِنُونَ ﴿٨٢﴾
اور جب ہماری بات پُوری ہونے کا وقت اُن پر آپہنچے گا تو ہم ان کے لیے ایک جانور زمین سے نکالیں گے جو ان سے کلام کرے گا کہ لوگ ہماری آیات پر یقین نہیں کر تے تھے۔
وَيَوْمَ نَحْشُرُ مِن كُلِّ أُمَّةٍ فَوْجًا مِّمَّن يُكَذِّبُ بِآيَاتِنَا فَهُمْ يُوزَعُونَ ﴿٨٣﴾
اور ذرا تصور کرو اُس دن کا جب ہم ہر امّت میں سے ایک فوج کی فوج اُن لوگوں کی گھیر لائیں گے جو ہماری آیات کو جھٹلایا کرتے تھے، پھر ان کو (ان کی اقسام کے لحاظ سے درجہ بدرجہ) مرتب کیا جائے گا
حَتَّىٰ إِذَا جَاءُوا قَالَ أَكَذَّبْتُم بِآيَاتِي وَلَمْ تُحِيطُوا بِهَا عِلْمًا أَمَّاذَا كُنتُمْ تَعْمَلُونَ ﴿٨٤﴾
یہاں تک کہ جب سب آجائیں گے تو (ان کا ربّ ان سے)پوچھے گاکہ”تم نے میری آیات کو جھُٹلا دیا حالانکہ تم نے ان کا عِلمی احاطہ نہ کیا تھا؟ اگر یہ نہیں تو اور تم کیا کر رہے تھے؟“
وَوَقَعَ الْقَوْلُ عَلَيْهِم بِمَا ظَلَمُوا فَهُمْ لَا يَنطِقُونَ ﴿٨٥﴾
اور ان کے ظلم کی وجہ سے عذاب کا وعدہ ان پر پورا ہو جائے گا، تب وہ کچھ بھی نہ بول سکیں گے
أَلَمْ يَرَوْا أَنَّا جَعَلْنَا اللَّيْلَ لِيَسْكُنُوا فِيهِ وَالنَّهَارَ مُبْصِرًا ۚ إِنَّ فِي ذَٰلِكَ لَآيَاتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٨٦﴾
کیا ان کو سُجھائی نہ دیتا تھا کہ ہم نے رات ان کے لیے سکون حاصل کرنے کو بنائی تھی اور دن کو روشن کیا تھا؟ اسی میں بہت نشانیاں تھیں ان لوگوں کے لیے جو ایمان لاتے تھے۔
وَيَوْمَ يُنفَخُ فِي الصُّورِ فَفَزِعَ مَن فِي السَّمَاوَاتِ وَمَن فِي الْأَرْضِ إِلَّا مَن شَاءَ اللَّهُ ۚ وَكُلٌّ أَتَوْهُ دَاخِرِينَ ﴿٨٧﴾
اور کیا گُزرے گی اُس روز جب کہ صُور پھُونکا جائے گا اور ہَول کھا جائیں گے وہ سب جو آسمانوں اور زمین میں ہیں۔۔۔۔ سوائے اُن لوگوں کے جنہیں اللہ اس ہَول سے بچانا چاہے گا۔۔۔۔ اور سب کان دبائے اس کے حضور حاضر ہو جائیں گے
وَتَرَى الْجِبَالَ تَحْسَبُهَا جَامِدَةً وَهِيَ تَمُرُّ مَرَّ السَّحَابِ ۚ صُنْعَ اللَّهِ الَّذِي أَتْقَنَ كُلَّ شَيْءٍ ۚ إِنَّهُ خَبِيرٌ بِمَا تَفْعَلُونَ ﴿٨٨﴾
آج تُو پہاڑوں کو دیکھتا ہے اور سمجھتا ہے کہ خوب جمے ہوئے ہیں، مگر اُس وقت یہ بادلوں کی طرح اُڑ رہے ہوں گے، یہ اللہ کی قدرتت کا کرشمہ ہوگا جس نے ہر چیز کو حکمت کے ساتھ استوار کیا ہے۔ وہ خوب جانتا ہے کہ تم لوگ کیا کرتے ہو۔
مَن جَاءَ بِالْحَسَنَةِ فَلَهُ خَيْرٌ مِّنْهَا وَهُم مِّن فَزَعٍ يَوْمَئِذٍ آمِنُونَ ﴿٨٩﴾
جو شخص بھلائی لے کر آئے گا اسے اُس سے زیادہ بہتر صلہ ملے گا اور ایسے لوگ اُس دن کے ہَول سے محفوظ ہوں گے۔
وَمَن جَاءَ بِالسَّيِّئَةِ فَكُبَّتْ وُجُوهُهُمْ فِي النَّارِ هَلْ تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٩٠﴾
اور جو بُرائی لیے ہوئے آئے گا ، ایسے سب لوگ اوندھے منہ آگ میں پھینکے جائیں گے۔ کیا تم لوگ اس کے سوا کوئی اور جزا پا سکتے ہو کہ جیسا کرو ویسا بھرو؟
إِنَّمَا أُمِرْتُ أَنْ أَعْبُدَ رَبَّ هَٰذِهِ الْبَلْدَةِ الَّذِي حَرَّمَهَا وَلَهُ كُلُّ شَيْءٍ ۖ وَأُمِرْتُ أَنْ أَكُونَ مِنَ الْمُسْلِمِينَ ﴿٩١﴾
(اے محمدؐ ، اِن سے کہو)”مجھے تو یہی حکم دیا گیا ہے کہ اِس شہر کے ربّ کی بندگی کروں جس نے اِسے حرم بنایا ہے اور جو ہر چیز کا مالک ہے۔ مجھے حکم دیا گیا ہے کہ میں مسلم بن کر رہوں
وَأَنْ أَتْلُوَ الْقُرْآنَ ۖ فَمَنِ اهْتَدَىٰ فَإِنَّمَا يَهْتَدِي لِنَفْسِهِ ۖ وَمَن ضَلَّ فَقُلْ إِنَّمَا أَنَا مِنَ الْمُنذِرِينَ ﴿٩٢﴾
اور یہ قرآن پڑھ کر سناؤں" اب جو ہدایت اختیار کرے گا وہ اپنے ہی بھلے کے لیے ہدایت اختیار کرے گا اور جو گمراہ ہو اُس سے کہہ دو کہ میں تو بس خبردار کر دینے والا ہوں
وَقُلِ الْحَمْدُ لِلَّهِ سَيُرِيكُمْ آيَاتِهِ فَتَعْرِفُونَهَا ۚ وَمَا رَبُّكَ بِغَافِلٍ عَمَّا تَعْمَلُونَ ﴿٩٣﴾
اِن سے کہو، تعریف اللہ ہی کے لیے ہے، عنقریب وہ تمہیں اپنی نشانیاں دکھا دے گا اور تم انہیں پہچان لو گے، اور تیرا رب بے خبر نہیں ہے اُن اعمال سے جو تم لوگ کرتے ہو
