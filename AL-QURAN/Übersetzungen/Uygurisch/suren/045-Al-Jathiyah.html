Al-Jathiyah · The Crouching
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
حم ﴿١﴾
حا، مىم
تَنزِيلُ الْكِتَابِ مِنَ اللَّهِ الْعَزِيزِ الْحَكِيمِ ﴿٢﴾
(بۇ) مۇقەددەس كىتاب-ۋەھىي غالىب، ھېكمەت بىلەن ئىش قىلغۇچى اﷲ تەرىپىدىن نازىل قىلىنغاندۇر
إِنَّ فِي السَّمَاوَاتِ وَالْأَرْضِ لَآيَاتٍ لِّلْمُؤْمِنِينَ ﴿٣﴾
ئاسمانلاردا ۋە زېمىندا مۆمىنلەر ئۈچۈن (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) ئىشەنچلىك ھالدا نۇرغۇن دەلىللەر بار
وَفِي خَلْقِكُمْ وَمَا يَبُثُّ مِن دَابَّةٍ آيَاتٌ لِّقَوْمٍ يُوقِنُونَ ﴿٤﴾
سىلەرنىڭ يارىتىلىشىڭلاردا ۋە (اﷲ زېمىندا) تارقىتىۋەتكەن جانىۋارلاردا (اﷲ نىڭ قۇدرىتىگە) جەزمەن ئىشىنىدىغان قەۋم ئۈچۈن نۇرغۇن دەلىللەر بار
وَاخْتِلَافِ اللَّيْلِ وَالنَّهَارِ وَمَا أَنزَلَ اللَّهُ مِنَ السَّمَاءِ مِن رِّزْقٍ فَأَحْيَا بِهِ الْأَرْضَ بَعْدَ مَوْتِهَا وَتَصْرِيفِ الرِّيَاحِ آيَاتٌ لِّقَوْمٍ يَعْقِلُونَ ﴿٥﴾
كېچە بىلەن كۈندۈزنىڭ نۆۋەتلىشىپ تۇرۇشىدا، اﷲ نىڭ بۇلۇتتىن (كۆكتىن) يامغۇر ياغدۇرۇپ ئۇنىڭ بىلەن رىزىقلەندۈرۈپ ئۆلگەن زېمىننى تىرىلدۈرۈشىدە ۋە شاماللارنىڭ يۈزلىنىشىنى ئۆزگەرتىپ تۇرۇشىدا، چۈشىنىدىغان قەۋم ئۈچۈن (اﷲ نىڭ بىرلىكىنى كۆرسىتىدىغان) نۇرغۇن دەلىللەر بار
تِلْكَ آيَاتُ اللَّهِ نَتْلُوهَا عَلَيْكَ بِالْحَقِّ ۖ فَبِأَيِّ حَدِيثٍ بَعْدَ اللَّهِ وَآيَاتِهِ يُؤْمِنُونَ ﴿٦﴾
ئەنە شۇلار اﷲ نىڭ ئايەتلىرىدۇر. (ئى مۇھەممەد!) ئۇنى ساڭا ھەقلىق بىلەن تىلاۋەت قىلىپ بېرىمىز. ئۇلار (يەنى مەككە كۇففارلىرى) اﷲ نى ۋە ئۇنىڭ ئايەتلىرىنى قويۇپ قايسى سۆزگە ئىشىنىدۇ؟
وَيْلٌ لِّكُلِّ أَفَّاكٍ أَثِيمٍ ﴿٧﴾
ھەر بىر گۇناھكار يالغانچىغا ۋاي!
يَسْمَعُ آيَاتِ اللَّهِ تُتْلَىٰ عَلَيْهِ ثُمَّ يُصِرُّ مُسْتَكْبِرًا كَأَن لَّمْ يَسْمَعْهَا ۖ فَبَشِّرْهُ بِعَذَابٍ أَلِيمٍ ﴿٨﴾
ئۇ اﷲ نىڭ ئايەتلىرىنىڭ ئۆزىگە ئوقۇپ بېرىلىۋاتقانلىقىنى ئاڭلاپ تۇرىدۇ، ئاندىن تەكەببۇرلۇق قىلغان ھالدا گويا ئۇنى ئاڭلىمىغاندەك (كۇفرىدا) چىڭ تۇرىدۇ (گۇمراھلىقتا تېخىمۇ ئەزۋەيلەيدۇ)، ئۇنىڭغا قاتتىق ئازاب بىلەن خۇش خەۋەر بەرگىن
وَإِذَا عَلِمَ مِنْ آيَاتِنَا شَيْئًا اتَّخَذَهَا هُزُوًا ۚ أُولَٰئِكَ لَهُمْ عَذَابٌ مُّهِينٌ ﴿٩﴾
ئۇ بىزنىڭ ئايەتلىرىمىزدىن بىرەر نەرسىنى بىلگەن چاغدا (يەنى ئۇنىڭغا قۇرئان ئايەتلىرى يەتكەن چاغدا)، ئۇنى مەسخىرە قىلىدىغان نەرسە قىلىۋالىدۇ، ئەنە شۇلار خار قىلغۇچى ئازابقا دۇچار بولىدۇ
مِّن وَرَائِهِمْ جَهَنَّمُ ۖ وَلَا يُغْنِي عَنْهُم مَّا كَسَبُوا شَيْئًا وَلَا مَا اتَّخَذُوا مِن دُونِ اللَّهِ أَوْلِيَاءَ ۖ وَلَهُمْ عَذَابٌ عَظِيمٌ ﴿١٠﴾
ئۇلارنى ئالدىدا مەڭگۈلۈك ئوت (كۈتۈپ تۇرىدۇ)، ئۇلارنىڭ ئېرىشكەن نەرسىلىرى (يەنى دۇنيادا تاپقان ماللىرى ۋە بالىلىرى) ۋە اﷲ نى قويۇپ دوست تۇتقان نەرسىلىرى (يەنى بۇتلىرى) ئۇلارغا ھېچ نەرسىگە ئەسقاتمايدۇ، ئۇلار چوڭ ئازابقا دۇچار بولىدۇ
هَٰذَا هُدًى ۖ وَالَّذِينَ كَفَرُوا بِآيَاتِ رَبِّهِمْ لَهُمْ عَذَابٌ مِّن رِّجْزٍ أَلِيمٌ ﴿١١﴾
بۇ (قۇرئان ئۇنىڭغا ئىشەنگەن ۋە ئۇنىڭغا ئەگەشكەن كىشىگە كامىل) ھىدايەتتۇر، پەرۋەردىگارىنىڭ ئايەتلىرىنى ئىنكار قىلغانلار ئەڭ قاتتىق ئازابقا دۇچار بولىدۇ
اللَّهُ الَّذِي سَخَّرَ لَكُمُ الْبَحْرَ لِتَجْرِيَ الْفُلْكُ فِيهِ بِأَمْرِهِ وَلِتَبْتَغُوا مِن فَضْلِهِ وَلَعَلَّكُمْ تَشْكُرُونَ ﴿١٢﴾
اﷲ نىڭ ئەمرى بىلەن دېڭىزدا كېمىلەرنىڭ يۈرۈشى ئۈچۈن، سىلەرنىڭ اﷲ نىڭ پەزلىدىن تەلەپ قىلىشىڭلار ئۈچۈن (يەنى تىجارەت قىلىشىڭلار، بېلىق تۇتۇشۇڭلار، دېڭىز ئاستىدىن ئۈنچە - مەرۋايىتلارنى سۈزۈۋېلىشىڭلار ئۈچۈن) ۋە (اﷲ قا) شۈكۈر قىلىشىڭلار ئۈچۈن، اﷲ سىلەرگە دېڭىزنى بويسۇندۇرۇپ بەردى
وَسَخَّرَ لَكُم مَّا فِي السَّمَاوَاتِ وَمَا فِي الْأَرْضِ جَمِيعًا مِّنْهُ ۚ إِنَّ فِي ذَٰلِكَ لَآيَاتٍ لِّقَوْمٍ يَتَفَكَّرُونَ ﴿١٣﴾
اﷲ ئاسمانلاردىكى ۋە زېمىندىكى نەرسىلەرنىڭ ھەممىسىنى (كامالى) پەزلىدىن سىلەرگە بويسۇندۇرۇپ بەردى، بۇنىڭدا ئىشەنچلىك ھالدا (قۇدرىتى ئىلاھىيىنى) تەپەككۈر قىلىدىغان قەۋم ئۈچۈن (اﷲ نىڭ قۇدرىتى ۋە بىرلىكىنى كۆرسىتىدىغان) (روشەن) دەلىللەر بار
قُل لِّلَّذِينَ آمَنُوا يَغْفِرُوا لِلَّذِينَ لَا يَرْجُونَ أَيَّامَ اللَّهِ لِيَجْزِيَ قَوْمًا بِمَا كَانُوا يَكْسِبُونَ ﴿١٤﴾
مۆمىنلەرگە ئېيتقىنكى، اﷲ ھەر قەۋمنىڭ قىلمىشلىرىغا يارىشا جازا بېرىدىغانلىقى ئۈچۈن، ئۇلار اﷲ نىڭ كۈنلىرىنى ئۈمىد قىلمايدىغانلارنى (يەنى اﷲ نىڭ ئازابىدىن قورقمايدىغان كۇففارلارنىڭ يەتكۈزگەن ئەزىيەتلىرىنى) كەچۈرسۇن
مَنْ عَمِلَ صَالِحًا فَلِنَفْسِهِ ۖ وَمَنْ أَسَاءَ فَعَلَيْهَا ۖ ثُمَّ إِلَىٰ رَبِّكُمْ تُرْجَعُونَ ﴿١٥﴾
كىمكى (دۇنيادا) ياخشى ئىش قىلسا، ئۇ ئۆزىنىڭ پايدىسى ئۈچۈن قىلىدۇ، كىمكى يامان ئىش قىلسا، ئۇ ئۆزىنىڭ زىيىنى ئۈچۈن قىلىدۇ، ئاندىن كېيىن پەرۋەردىگارىڭلارنىڭ دەرگاھىغا قايتۇرۇلىسىلەر (اﷲ ياخشىلىققا مۇكاپات، يامانلىققا جازا بېرىدۇ)
وَلَقَدْ آتَيْنَا بَنِي إِسْرَائِيلَ الْكِتَابَ وَالْحُكْمَ وَالنُّبُوَّةَ وَرَزَقْنَاهُم مِّنَ الطَّيِّبَاتِ وَفَضَّلْنَاهُمْ عَلَى الْعَالَمِينَ ﴿١٦﴾
شەك - شۈبھىسىزكى، بىز ئىسرائىل ئەۋلادىغا مۇقەددەس كىتاب-ۋەھىينى (يەنى تەۋراتنى)، ھېكمەتنى ۋە خۇدانىڭ ئەلچىسىلىكنى ئاتا قىلدۇق، ئۇلارغا پاك نەرسىلەرنى رىزىق قىلىپ بەردۇق، ئۇلارنى (زامانىدىكى) جاھان ئەھلىدىن ئۈستۈن قىلدۇق
وَآتَيْنَاهُم بَيِّنَاتٍ مِّنَ الْأَمْرِ ۖ فَمَا اخْتَلَفُوا إِلَّا مِن بَعْدِ مَا جَاءَهُمُ الْعِلْمُ بَغْيًا بَيْنَهُمْ ۚ إِنَّ رَبَّكَ يَقْضِي بَيْنَهُمْ يَوْمَ الْقِيَامَةِ فِيمَا كَانُوا فِيهِ يَخْتَلِفُونَ ﴿١٧﴾
ئۇلارغا دىن ئىشىدا روشەن دەلىللەرنى بەردۇق، پەقەت ئۇلارغا ئىلىم (يەنى دىننىڭ راستلىقىغا قەتئى دەلىللەر) كەلگەندىن كېيىنلا، ئاندىن ئۇلار ئۆزئارا ھەسەت قىلىشىپ ئىختىلاپ قىلىشتى. پەرۋەردىگارىڭ ئىشەنچلىك ھالدا قىيامەت كۈنى ئۇلار ئىختىلاپ قىلىشقان ئىشلار ئۈستىدە ھۆكۈم چىقىرىدۇ
ثُمَّ جَعَلْنَاكَ عَلَىٰ شَرِيعَةٍ مِّنَ الْأَمْرِ فَاتَّبِعْهَا وَلَا تَتَّبِعْ أَهْوَاءَ الَّذِينَ لَا يَعْلَمُونَ ﴿١٨﴾
ئاندىن سېنى بىز (ئى مۇھەممەد!) دىن ئىشىدا (روشەن) بىر يولدا قىلدۇق، شۇ يولغا ئەگەشكىن، بىلمەيدىغانلار (يەنى مۇشرىكلار) نىڭ نەپسى خاھىشلىرىغا ئەگەشمىگىن
إِنَّهُمْ لَن يُغْنُوا عَنكَ مِنَ اللَّهِ شَيْئًا ۚ وَإِنَّ الظَّالِمِينَ بَعْضُهُمْ أَوْلِيَاءُ بَعْضٍ ۖ وَاللَّهُ وَلِيُّ الْمُتَّقِينَ ﴿١٩﴾
شۈبھىسىزكى، (ئۇلارنىڭ گۇمراھلىقىغا ماسلىشىدىغان بولساڭ) ئۇلار سەندىن اﷲ نىڭ (ئازابى) دىن ھېچ نەرسىنى دەپئى قىلالمايدۇ، شۈبھىسىزكى، (دۇنيادا) زالىملارنىڭ بەزىسى بەزىسىگە دوستتۇر، اﷲ تەقۋادارلارنىڭ دوستىدۇر
هَٰذَا بَصَائِرُ لِلنَّاسِ وَهُدًى وَرَحْمَةٌ لِّقَوْمٍ يُوقِنُونَ ﴿٢٠﴾
بۇ (قۇرئان) ئىنسانلار ئۈچۈن روشەن پاكىتلاردۇر، ھىدايەتتۇر، (ئۆلگەندىن كېيىن تىرىلىشكە) جەزمەن ئىشەنگەن قەۋم ئۈچۈن رەھمەتتۇر
أَمْ حَسِبَ الَّذِينَ اجْتَرَحُوا السَّيِّئَاتِ أَن نَّجْعَلَهُمْ كَالَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ سَوَاءً مَّحْيَاهُمْ وَمَمَاتُهُمْ ۚ سَاءَ مَا يَحْكُمُونَ ﴿٢١﴾
يامان ئىشلارنى قىلغانلار (يەنى گۇناھكار كۇففارلار) ئۆزلىرىنى ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلاردەك قىلىشىمىزنى، ھاياتلىقتا، ماماتلىقتا ئۇلار بىلەن ئوخشاش قىلىشىمىزنى ئويلامدۇ؟ (يەنى ئۇلارنى ئوخشاش قىلىشىمىز مۇمكىن ئەمەس) ئۇلارنىڭ چىقارغان ھۆكمى نېمىدېگەن يامان!
وَخَلَقَ اللَّهُ السَّمَاوَاتِ وَالْأَرْضَ بِالْحَقِّ وَلِتُجْزَىٰ كُلُّ نَفْسٍ بِمَا كَسَبَتْ وَهُمْ لَا يُظْلَمُونَ ﴿٢٢﴾
اﷲ ئاسمانلارنى ۋە زېمىننى ھەقلىق بىلەن ياراتتى، ھەر ئىنساننىڭ قىلمىشىغا يارىشا جازا ياكى مۇكاپات بېرىلىدۇ، ئۇلارغا زۇلۇم قىلىنمايدۇ
أَفَرَأَيْتَ مَنِ اتَّخَذَ إِلَٰهَهُ هَوَاهُ وَأَضَلَّهُ اللَّهُ عَلَىٰ عِلْمٍ وَخَتَمَ عَلَىٰ سَمْعِهِ وَقَلْبِهِ وَجَعَلَ عَلَىٰ بَصَرِهِ غِشَاوَةً فَمَن يَهْدِيهِ مِن بَعْدِ اللَّهِ ۚ أَفَلَا تَذَكَّرُونَ ﴿٢٣﴾
(ئى مۇھەممەد!) ماڭا نەپسى خاھىشنى ئىلاھ قىلىۋالغان، اﷲ ئازدۇرغان، (ھەقنى) بىلىپ تۇرغان، اﷲ قۇلىقىنى ۋە قەلبىنى پېچەتلىۋەتكەن ۋە كۆزىنى پەردىلىگەن ئادەمنى ئېيتىپ بەرگىن، اﷲ ئازدۇرغاندىن كېيىن، ئۇنى كىم ھىدايەت قىلالايدۇ؟ سىلەر (بۇلاردىن) ۋەز - نەسىھەت ئالمامسىلەر؟
وَقَالُوا مَا هِيَ إِلَّا حَيَاتُنَا الدُّنْيَا نَمُوتُ وَنَحْيَا وَمَا يُهْلِكُنَا إِلَّا الدَّهْرُ ۚ وَمَا لَهُم بِذَٰلِكَ مِنْ عِلْمٍ ۖ إِنْ هُمْ إِلَّا يَظُنُّونَ ﴿٢٤﴾
ئۇلار (يەنى قىيامەتنى ئىنكار قىلغۇچىلار): «ھايات دېگەن پەقەت دۇنيادىكى ھاياتىمىزدۇر، ئۆلىمىز ۋە تىرىلىمىز (يەنى تۇغۇلىمىز). پەقەت زاماننىڭ ئۆتۈشى بىلەن يوق بولىمىز» (دەيدۇ). ئۇلار بۇ خۇسۇستا ھېچقانداق مەلۇماتقا ئىگە ئەمەس، ئۇلار پەقەت گۇمان بىلەن سۆزلەيدۇ
وَإِذَا تُتْلَىٰ عَلَيْهِمْ آيَاتُنَا بَيِّنَاتٍ مَّا كَانَ حُجَّتَهُمْ إِلَّا أَن قَالُوا ائْتُوا بِآبَائِنَا إِن كُنتُمْ صَادِقِينَ ﴿٢٥﴾
ئۇلارغا بىزنىڭ ئايەتلىرىمىز ئوچۇق ئوقۇپ بېرىلسە، ئۇلارنىڭ پاكىتى پەقەت «ئەگەر راستچىل بولساڭلار، ئاتا - بوۋىلىرىمىزنى كەلتۈرۈپ بېقىڭلار (يەنى تىرىلدۈرۈڭلار)» دېيىشتىن ئىبارەت بولدى
قُلِ اللَّهُ يُحْيِيكُمْ ثُمَّ يُمِيتُكُمْ ثُمَّ يَجْمَعُكُمْ إِلَىٰ يَوْمِ الْقِيَامَةِ لَا رَيْبَ فِيهِ وَلَٰكِنَّ أَكْثَرَ النَّاسِ لَا يَعْلَمُونَ ﴿٢٦﴾
(ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «اﷲ سىلەرنى تىرىلدۈرەلەيدۇ، ئاندىن (ئەجىلىڭلار پۈتكەندە) ۋاپات قىلدۇرالايدۇ، ئاندىن شەك - شۈبھىسىز قىيامەت كۈنىدە سىلەرنى يىغىدۇ، لېكىن كىشىلەرنىڭ تولىسى (اﷲ نىڭ قۇدرىتىنى) بىلمەي (قايتا تىرىلىشنى ئىنكار قىلدى)»
وَلِلَّهِ مُلْكُ السَّمَاوَاتِ وَالْأَرْضِ ۚ وَيَوْمَ تَقُومُ السَّاعَةُ يَوْمَئِذٍ يَخْسَرُ الْمُبْطِلُونَ ﴿٢٧﴾
ئاسمانلار بىلەن زېمىننىڭ پادىشاھلىقى اﷲ قا خاستۇر، قىيامەت قايىم بولغان كۈندە ھەقىقەتنى رەت قىلغۇچىلار زىيان تارتىدۇ
وَتَرَىٰ كُلَّ أُمَّةٍ جَاثِيَةً ۚ كُلُّ أُمَّةٍ تُدْعَىٰ إِلَىٰ كِتَابِهَا الْيَوْمَ تُجْزَوْنَ مَا كُنتُمْ تَعْمَلُونَ ﴿٢٨﴾
ھەر ئۈممەتنى (قورقۇنچنىڭ قاتتىقلىقىدىن) تىزلانغان ھالدا كۆرىسەن، ھەر ئۈممەت ئۆزىنىڭ نامە - ئەمالىغا چاقىرىلىدۇ، (ئۇلارغا) «بۈگۈن قىلمىشىڭلارغا يارىشا مۇكاپاتقا ئېرىشىسىلەر ياكى جازاغا ئۇچرايسىلەر» (دېيىلىدۇ)
هَٰذَا كِتَابُنَا يَنطِقُ عَلَيْكُم بِالْحَقِّ ۚ إِنَّا كُنَّا نَسْتَنسِخُ مَا كُنتُمْ تَعْمَلُونَ ﴿٢٩﴾
بۇ بىزنىڭ مۇقەددەس كىتاب-ۋەھىيىمىز (يەنى نامە - ئەمال) سىلەرگە ھەق بىلەن سۆزلەيدۇ، بىز قىلمىشىڭلارنى يازدۇرۇپ تۇرغان ئىدۇق (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرىنى ئەمەللىرىڭلارنى يېزىپ تۇرۇشقا بۇيرۇيتتۇق)
فَأَمَّا الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ فَيُدْخِلُهُمْ رَبُّهُمْ فِي رَحْمَتِهِ ۚ ذَٰلِكَ هُوَ الْفَوْزُ الْمُبِينُ ﴿٣٠﴾
ئىمان ئېيتقان ۋە ياخشى ئىشلارنى قىلغانلارغا كەلسەك، پەرۋەردىگارى ئۇلارنى رەھمىتى دائىرىسىگە كىرگۈزىدۇ، بۇ روشەن مۇرادقا يېتىشتۇر
وَأَمَّا الَّذِينَ كَفَرُوا أَفَلَمْ تَكُنْ آيَاتِي تُتْلَىٰ عَلَيْكُمْ فَاسْتَكْبَرْتُمْ وَكُنتُمْ قَوْمًا مُّجْرِمِينَ ﴿٣١﴾
ھەقىقەتنى رەت قىلغۇچىلارغا كەلسەك، (ئۇلارنى ئەيىبلەش يۈزىسىدىن) ئۇلارغا: «مېنىڭ ئايەتلىرىم سىلەرگە ئوقۇپ بېرىلمىدىمۇ؟ سىلەر چوڭچىلىق قىلىپ (ئۇنىڭغا ئىمان ئېيتىشتىن) باش تارتتىڭلار، سىلەر گۇناھقا چۆمگەن قەۋم ئىدىڭلار» (دېيىلىدۇ)
وَإِذَا قِيلَ إِنَّ وَعْدَ اللَّهِ حَقٌّ وَالسَّاعَةُ لَا رَيْبَ فِيهَا قُلْتُم مَّا نَدْرِي مَا السَّاعَةُ إِن نَّظُنُّ إِلَّا ظَنًّا وَمَا نَحْنُ بِمُسْتَيْقِنِينَ ﴿٣٢﴾
(سىلەرگە) «اﷲ نىڭ ۋەدىسى ئىشەنچلىك ھالدا ھەقتۇر، قىيامەتتە (يەنى قىيامەتنىڭ بولۇشىدا) ھېچ شەك يوق» دېيىلسە، سىلەر (ھەددىدىن زىيادە سەركەشلىكىڭلاردىن): «قىيامەتنىڭ نېمە ئىكەنلىكىنى بىلمەيمىز، پەقەت ئۇنى بولۇشى مۇمكىن دەپ گۇمان قىلىمىز، (ئۇنىڭغا) بىز ھەقىقىي ئىشەنمەيمىز» دېدىڭلار
وَبَدَا لَهُمْ سَيِّئَاتُ مَا عَمِلُوا وَحَاقَ بِهِم مَّا كَانُوا بِهِ يَسْتَهْزِئُونَ ﴿٣٣﴾
(ئاخىرەتتە) ئۇلارنىڭ قىلغان يامان ئىشلىرى ئاشكارا بولىدۇ، ئۇلار (دۇنيادىكى چاغلىرىدا) مەسخىرە قىلغان ئازاب ئۇلارنى ئورىۋالىدۇ
وَقِيلَ الْيَوْمَ نَنسَاكُمْ كَمَا نَسِيتُمْ لِقَاءَ يَوْمِكُمْ هَٰذَا وَمَأْوَاكُمُ النَّارُ وَمَا لَكُم مِّن نَّاصِرِينَ ﴿٣٤﴾
(ئۇلارغا) دېيىلىدۇ: «سىلەرنىڭ مۇشۇ كۈندىكى مۇلاقاتنى ئۇنتۇغانلىقىڭلارغا ئوخشاش، بۈگۈن سىلەرنى ئۇنتۇيمىز (يەنى سىلەرنى ئازابقا تاشلاپ قويۇپ ئۇنتۇپ قالغاندەك مۇئامىلە قىلىمىز)، سىلەرنىڭ جايىڭلار دوزاخ بولىدۇ، سىلەرگە (سىلەرنى ئازابتىن قۇتقۇزىدىغان) ھېچقانداق ياردەمچىمۇ بولمايدۇ
ذَٰلِكُم بِأَنَّكُمُ اتَّخَذْتُمْ آيَاتِ اللَّهِ هُزُوًا وَغَرَّتْكُمُ الْحَيَاةُ الدُّنْيَا ۚ فَالْيَوْمَ لَا يُخْرَجُونَ مِنْهَا وَلَا هُمْ يُسْتَعْتَبُونَ ﴿٣٥﴾
بۇ شۇنىڭ ئۈچۈنكى، سىلەر اﷲ نىڭ ئايەتلىرىنى مەسخىرە قىلىدىغان نەرسە قىلىۋالدىڭلار ۋە سىلەرنى دۇنيا تىرىكچىلىكى ئالدىدى». بۈگۈن ئۇلار دوزاختىن چىقىرىلمايدۇ، ئۇلاردىن (تەۋبە قىلىش ۋە ئىبادەت قىلىش بىلەن) اﷲ نى رازى قىلىشمۇ تەلەپ قىلىنمايدۇ
فَلِلَّهِ الْحَمْدُ رَبِّ السَّمَاوَاتِ وَرَبِّ الْأَرْضِ رَبِّ الْعَالَمِينَ ﴿٣٦﴾
جىمى ھەمدۇسانا ئاسمانلارنىڭ پەرۋەردىگارى، زېمىننىڭ پەرۋەردىگارى ۋە ئالەملەرنىڭ پەرۋەردىگارى اﷲ قا خاستۇر!
وَلَهُ الْكِبْرِيَاءُ فِي السَّمَاوَاتِ وَالْأَرْضِ ۖ وَهُوَ الْعَزِيزُ الْحَكِيمُ ﴿٣٧﴾
ئاسمانلاردىكى ۋە زېمىندىكى ئۇلۇغلۇق اﷲ قا خاستۇر، اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
