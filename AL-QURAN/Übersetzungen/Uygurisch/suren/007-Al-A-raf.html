المص ﴿١﴾
ئەلىف، لام، مىم، صاد
كِتَابٌ أُنزِلَ إِلَيْكَ فَلَا يَكُن فِي صَدْرِكَ حَرَجٌ مِّنْهُ لِتُنذِرَ بِهِ وَذِكْرَىٰ لِلْمُؤْمِنِينَ ﴿٢﴾
(ئى مۇھەممەد! ھەقىقەتنى رەت قىلغۇچىلارنى) ئاگاھلاندۇرۇشۇڭ ئۈچۈن، مۆمىنلەرگە ۋەز - نەسىھەت قىلىشىڭ ئۈچۈن، بۇ مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) ساڭا نازىل قىلىندى، ئۇنىڭدىن (يەنى ئۇنى تەبلىغ قىلىشتا قەۋمىڭنىڭ ئۇنى ئىنكار قىلىشىدىن ئەنسىرەپ) يۈرىكىڭ سىقىلمىسۇن
اتَّبِعُوا مَا أُنزِلَ إِلَيْكُم مِّن رَّبِّكُمْ وَلَا تَتَّبِعُوا مِن دُونِهِ أَوْلِيَاءَ ۗ قَلِيلًا مَّا تَذَكَّرُونَ ﴿٣﴾
(ئى ئىنسانلار!) سىلەر پەرۋەردىگارىڭلار تەرىپىدىن سىلەرگە نازىل قىلىنغان مۇقەددەس كىتاب-ۋەھىيقا (يەنى قۇرئانغا) ئەگىشىڭلار، اﷲ نى قويۇپ، (جىنلاردىن ۋە ئىنسانلاردىن بولغان، سىلەرنى ئازدۇرىدىغان) دوستلارغا ئەگەشمەڭلار، سىلەر ۋەز - نەسىھەتنى ئاز قوبۇل قىلىسىلەر
وَكَم مِّن قَرْيَةٍ أَهْلَكْنَاهَا فَجَاءَهَا بَأْسُنَا بَيَاتًا أَوْ هُمْ قَائِلُونَ ﴿٤﴾
بىز نۇرغۇن شەھەر (ئاھالىسىنى) نى ھالاك قىلدۇق، بىزنىڭ ئازابىمىز ئۇلارغا كېچىسى ياكى چۈشلۈكى ئۇخلاۋاتقان چاغلىرىدا كەلدى
فَمَا كَانَ دَعْوَاهُمْ إِذْ جَاءَهُم بَأْسُنَا إِلَّا أَن قَالُوا إِنَّا كُنَّا ظَالِمِينَ ﴿٥﴾
بىزنىڭ ئازابىمىز كەلگەن چاغدا، ئۇلارنىڭ چەككەن پەريادى: بىز ئىشەنچلىك ھالدا زالىم ئىدۇق، دېيىشتىنلا ئىبارەت بولدى
فَلَنَسْأَلَنَّ الَّذِينَ أُرْسِلَ إِلَيْهِمْ وَلَنَسْأَلَنَّ الْمُرْسَلِينَ ﴿٦﴾
بىز ئەلۋەتتە خۇدانىڭ ئەلچىسى ئەۋەتىلگەن ئۈممەتلەردىن (خۇدانىڭ ئەلچى-رەسۇللىرى تەبلىغ قىلدىمۇ؟ سىلەر قانداق جاۋاب بەردىڭلار؟ دەپ) سورايمىز، خۇدانىڭ ئەلچى-رەسۇللىرىدىنمۇ ئەلۋەتتە (اﷲ تاپشۇرغان ۋەزىپىنى ئادا قىلدىڭلارمۇ؟ دەپ) سورايمىز
فَلَنَقُصَّنَّ عَلَيْهِم بِعِلْمٍ ۖ وَمَا كُنَّا غَائِبِينَ ﴿٧﴾
(قىيامەت كۈنى ئۇلارنىڭ ئەھۋالىنى) ئۇلارغا ھەقىقىي بىلىش ئاساسىدا ئېيتىپ بېرىمىز، بىز ئۇلاردىن ئايرىلمىغان ئىدۇق (يەنى ئۇلارنىڭ ھەممە ئەھۋالىدىن خەۋەرداردۇرمىز)
وَالْوَزْنُ يَوْمَئِذٍ الْحَقُّ ۚ فَمَن ثَقُلَتْ مَوَازِينُهُ فَأُولَٰئِكَ هُمُ الْمُفْلِحُونَ ﴿٨﴾
بۇ كۈندە بەندىلەرنىڭ (ياخشى يامان ئەمەللىرىنىڭ) ئۆلچىنىدىغانلىقى ھەقتۇر. ياخشى ئەمەللىرى ئېغىر چىققانلار نىجات تاپقۇچىلاردۇر
وَمَنْ خَفَّتْ مَوَازِينُهُ فَأُولَٰئِكَ الَّذِينَ خَسِرُوا أَنفُسَهُم بِمَا كَانُوا بِآيَاتِنَا يَظْلِمُونَ ﴿٩﴾
ياخشى ئەمەللىرى يېنىك چىققانلار بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلىقلىرى تۈپەيلىدىن ئۆزلىرىگە زىيان سالغۇچىلاردۇر
وَلَقَدْ مَكَّنَّاكُمْ فِي الْأَرْضِ وَجَعَلْنَا لَكُمْ فِيهَا مَعَايِشَ ۗ قَلِيلًا مَّا تَشْكُرُونَ ﴿١٠﴾
شۈبھىسىزكى، سىلەرنى زېمىندا يەرلەشتۈردۇق، زېمىندا سىلەرگە تۇرمۇش لازېمەتلىكلىرىنى پەيدا قىلدۇق، (سىلەرگە پەرۋەردىگارىڭلارنىڭ مۇشۇنداق نېمەتلىرى بار تۇرۇقلۇق ئۇنىڭغا) ناھايىتى ئاز شۈكۈر قىلىسىلەر
وَلَقَدْ خَلَقْنَاكُمْ ثُمَّ صَوَّرْنَاكُمْ ثُمَّ قُلْنَا لِلْمَلَائِكَةِ اسْجُدُوا لِآدَمَ فَسَجَدُوا إِلَّا إِبْلِيسَ لَمْ يَكُن مِّنَ السَّاجِدِينَ ﴿١١﴾
شۈبھىسىزكى بىز سىلەرنى ياراتتۇق، ئاندىن سىلەرنى شەكىلگە كىرگۈزدۇق، ئاندىن خۇدانىڭ ئەلچى پەرىشتىلىرىگە: “سىلەر ئادەمگە (ھۆرمەت بىلدۈرۈش يۈزىسىدىن) سەجدە قىلىڭلار›› دېدۇق، ئىبلىستىن باشقىسى سەجدە قىلدى. ئۇ سەجدە قىلغۇچىلاردىن بولمىدى
قَالَ مَا مَنَعَكَ أَلَّا تَسْجُدَ إِذْ أَمَرْتُكَ ۖ قَالَ أَنَا خَيْرٌ مِّنْهُ خَلَقْتَنِي مِن نَّارٍ وَخَلَقْتَهُ مِن طِينٍ ﴿١٢﴾
اﷲ (ئىبلىسقا) ‹‹سېنى سەجدە قىلىشقا بۇيرۇغان چېغىمدا نېمىشقا سەجدە قىلمىدىڭ؟ ›› دېدى. ئىبلىس: ‹‹مەن ئۇنىڭدىن ئارتۇق. مېنى ئوتتىن، ئۇنى لايدىن ياراتتىڭ›› دېدى
قَالَ فَاهْبِطْ مِنْهَا فَمَا يَكُونُ لَكَ أَن تَتَكَبَّرَ فِيهَا فَاخْرُجْ إِنَّكَ مِنَ الصَّاغِرِينَ ﴿١٣﴾
اﷲ (ئىبلىسقا) ئېيتتى: ‹‹سەن بۇيەردىن (يەنى مەڭگۈلۈك باغلارتىن) چۈشۈپ كەت، بۇ يەردىن ئۆزۈڭنى چوڭ تۇتۇشۇڭغا بولمايدۇ، يوقال، سەن ئىشەنچلىك ھالدا پەسكەشلەردىنسەن››
قَالَ أَنظِرْنِي إِلَىٰ يَوْمِ يُبْعَثُونَ ﴿١٤﴾
ئىبلىس: ‹‹ماڭا كىشىلەر قايتا تىرىلدۈرۈلىدىغان كۈنگىچە (يەنى قىيامەت كۈنىگىچە) مۆھلەت بەرگىن›› دېدى
قَالَ إِنَّكَ مِنَ الْمُنظَرِينَ ﴿١٥﴾
اﷲ: ‹‹ساڭا مۆھلەت بېرىلىدۇ›› دېدى
قَالَ فَبِمَا أَغْوَيْتَنِي لَأَقْعُدَنَّ لَهُمْ صِرَاطَكَ الْمُسْتَقِيمَ ﴿١٦﴾
ئىبلىس ئېيتتى: ‹‹سېنىڭ مېنى ئازدۇرغانلىقىڭدىن ئۇلار (يەنى ئادەم ئەلەيھىسسالام ۋە ئۇنىڭ ئەۋلادىغا ۋەسۋەسە قىلىش) ئۈچۈن چوقۇم سېنىڭ توغرا يولۇڭ ئۈستىدە ئولتۇرىمەن
ثُمَّ لَآتِيَنَّهُم مِّن بَيْنِ أَيْدِيهِمْ وَمِنْ خَلْفِهِمْ وَعَنْ أَيْمَانِهِمْ وَعَن شَمَائِلِهِمْ ۖ وَلَا تَجِدُ أَكْثَرَهُمْ شَاكِرِينَ ﴿١٧﴾
ئاندىن ئۇلارغا چوقۇم ئالدىدىن، ئارقىسىدىن، ئوڭدىن، سولىدىن ھۇجۇم قىلىمەن، ئۇلارنىڭ كۆپچىلىكىنىڭ (نېمىتىڭگە) شۈكۈر قىلغۇچىلار ئەمەسلىكىنى كۆرىسەن››
قَالَ اخْرُجْ مِنْهَا مَذْءُومًا مَّدْحُورًا ۖ لَّمَن تَبِعَكَ مِنْهُمْ لَأَمْلَأَنَّ جَهَنَّمَ مِنكُمْ أَجْمَعِينَ ﴿١٨﴾
اﷲ ئېيتتى: ‹‹مەڭگۈلۈك باغلارتىن ئەيىبلەنگەن ۋە (رەھمىتىمدىن) مەھرۇم قىلىنغان ھالدا چىق، ئۇلارنىڭ ئىچىدىن ساڭا ئەگەشكەنلەرنى قوشۇپ، ھەممىڭلار بىلەن چوقۇم مەڭگۈلۈك ئوتنى تولدۇرىمەن››
وَيَا آدَمُ اسْكُنْ أَنتَ وَزَوْجُكَ الْجَنَّةَ فَكُلَا مِنْ حَيْثُ شِئْتُمَا وَلَا تَقْرَبَا هَٰذِهِ الشَّجَرَةَ فَتَكُونَا مِنَ الظَّالِمِينَ ﴿١٩﴾
‹‹ئى ئادەم! سەن ئايالىڭ بىلەن ئىككىڭلار بىللە مەڭگۈلۈك باغلارتە تۇرۇڭلار، مەڭگۈلۈك باغلارنىڭ مېۋىلىرىدىن خالىغىنىڭلارچە يەڭلار، بۇ دەرەخكە يېقىنلاشماڭلار، (ئۇنىڭغا يېقىنلاشساڭلار) زالىملاردىن بولۇپ قالىسىلەر››
فَوَسْوَسَ لَهُمَا الشَّيْطَانُ لِيُبْدِيَ لَهُمَا مَا وُورِيَ عَنْهُمَا مِن سَوْآتِهِمَا وَقَالَ مَا نَهَاكُمَا رَبُّكُمَا عَنْ هَٰذِهِ الشَّجَرَةِ إِلَّا أَن تَكُونَا مَلَكَيْنِ أَوْ تَكُونَا مِنَ الْخَالِدِينَ ﴿٢٠﴾
لېكىن شەيتان (ئازدۇرغۇچى) ئۇلارنىڭ يېپىقلىق ئەۋرەتلىرىنى ئېچىۋېتىش ئۈچۈن، ئۇلارغا ۋەسۋەسە قىلدى، شەيتان (ئازدۇرغۇچى) ئېيتتى: ‹‹پەرۋەردىگارىڭلارنىڭ سىلەرگە بۇ دەرەخنىڭ مېۋىسىنى يېيىشنى مەنئى قىلغانلىقى پەقەت سىلەرنىڭ پەرىشتە بولۇپ كەتمەسلىكىڭلار ياكى (مەڭگۈلۈك باغلارتە) مەڭگۈ تۇرۇپ قالماسلىقىڭلار ئۈچۈندۇر››
وَقَاسَمَهُمَا إِنِّي لَكُمَا لَمِنَ النَّاصِحِينَ ﴿٢١﴾
شۇنىڭدەك ئۇلارغا: ‹‹مەن چوقۇم سىلەرگە سادىقمەن›› دەپ قەسەم ئىچىپ بەردى
فَدَلَّاهُمَا بِغُرُورٍ ۚ فَلَمَّا ذَاقَا الشَّجَرَةَ بَدَتْ لَهُمَا سَوْآتُهُمَا وَطَفِقَا يَخْصِفَانِ عَلَيْهِمَا مِن وَرَقِ الْجَنَّةِ ۖ وَنَادَاهُمَا رَبُّهُمَا أَلَمْ أَ��ْهَكُمَا عَن تِلْكُمَا الشَّجَرَةِ وَأَقُل لَّكُمَا إِنَّ الشَّيْطَانَ لَكُمَا عَدُوٌّ مُّبِينٌ ﴿٢٢﴾
ئالدامچىلىق ئۇسۇللىرى ئارقىلىق ئۇلارنى (يۇقىرى مەرتىۋىدىن تۆۋەن مەرتىۋىگە) چۈشۈردى. دەرەخنىڭ مېۋىسىنى يېگەندىن كېيىنلا ئۇلارنىڭ ئەۋرەتلىرى ئېچىلىپ قالدى، ئۇلار مەڭگۈلۈك باغلارتىكى دەرەخلەرنىڭ ياپراقلىرى بىلەن ئەۋرەتلىرىنى يېپىشقا كىرىشتى، ئۇلارنىڭ پەرۋەردىگارى (ئۇلارغا كايىپ): ‹‹مەن سىلەرگە بۇ دەرەخنى (يەنى مېۋىسىنى) يېيىشنى مەنئى قىلمىغانمىدىم؟ شۈبھىسىزكى، شەيتان (ئازدۇرغۇچى) سىلەرگە ئىشەنچلىك ھالدا ئاشكارا دۈشمەندۇر، دېمىگەنمىدىم؟ ›› دەپ نىدا قىلدى
قَالَا رَبَّنَا ظَلَمْنَا أَنفُسَنَا وَإِن لَّمْ تَغْفِرْ لَنَا وَتَرْحَمْنَا لَنَكُونَنَّ مِنَ الْخَاسِرِينَ ﴿٢٣﴾
ئۇلار: پەرۋەردىگارىمىز! بىز ئۆزىمىزگە ئۆزىمىز زۇلۇم قىلدۇق، ئەگەر سەن بىزگە مەغپىرەت قىلمىساڭ، بىزگە رەھىم قىلمىساڭ، بىز چوقۇم زىيان تارتقۇچىلاردىن بولىمىز›› دېدى
قَالَ اهْبِطُوا بَعْضُكُمْ لِبَعْضٍ عَدُوٌّ ۖ وَلَكُمْ فِي الْأَرْضِ مُسْتَقَرٌّ وَمَتَاعٌ إِلَىٰ حِينٍ ﴿٢٤﴾
اﷲ ئېيتتى: ‹‹سىلەر بىر - بىرىڭلارغا (يەنى شەيتان (ئازدۇرغۇچى) ئىنسانغا، ئىنسان شەيتان (ئازدۇرغۇچى)غا) دۈشمەن بولغان ھالدا چۈشۈڭلار، ئەجىلىڭلار پۈتكەنگە قەدەر يەر يۈزىدە تۇرىسىلەر ۋە ئۇنىڭدىن پايدىلىنىسىلەر››
قَالَ فِيهَا تَحْيَوْنَ وَفِيهَا تَمُوتُونَ وَمِنْهَا تُخْرَجُونَ ﴿٢٥﴾
اﷲ ئېيتتى: ‹‹يەر يۈزىدە ياشايسىلەر، يەر يۈزىدە ئۆلۈسىلەر، (قىيامەت كۈنى) يەنە يەر ئاستىدىن چىقىرىلىسىلەر››
يَا بَنِي آدَمَ قَدْ أَنزَلْنَا عَلَيْكُمْ لِبَاسًا يُوَارِي سَوْآتِكُمْ وَرِيشًا ۖ وَلِبَاسُ التَّقْوَىٰ ذَٰلِكَ خَيْرٌ ۚ ذَٰلِكَ مِنْ آيَاتِ اللَّهِ لَعَلَّهُمْ يَذَّكَّرُونَ ﴿٢٦﴾
ئى ئادەم بالىلىرى! سىلەرگە بىز ئىشەنچلىك ھالدا ئەۋرىتىڭلارنى ياپىدىغان لىباسنى ۋە زىننەتلىنىدىغان لىباسنى چۈشۈردۇق، تەقۋادارلىق لىباسى ئەڭ ياخشىدۇر، ئەنە شۇلار (يەنى ئىنسان بالىلىرىغا سەترى ئەۋرەت ئۈچۈن لىباس ياراتماق) ئۇلارنىڭ ئىبرەت ئېلىشى ئۈچۈن اﷲ نىڭ (بەندىلىرىگە بولغان پەزلى - مەرھەمىتىنى كۆرسىتىدىغان) ئالامەتلىرىدىندۇر
يَا بَنِي آدَمَ لَا يَفْتِنَنَّكُمُ الشَّيْطَانُ كَمَا أَخْرَجَ أَبَوَيْكُم مِّنَ الْجَنَّةِ يَنزِعُ عَنْهُمَا لِبَاسَهُمَا لِيُرِيَهُمَا سَوْآتِهِمَا ۗ إِنَّهُ يَرَاكُمْ هُوَ وَقَبِيلُهُ مِنْ حَيْثُ لَا تَرَوْنَهُمْ ۗ إِنَّا جَعَلْنَا الشَّيَاطِينَ أَوْلِيَاءَ لِلَّذِينَ لَا يُؤْمِنُونَ ﴿٢٧﴾
ئى ئادەم بالىلىرى! شەيتان (ئازدۇرغۇچى) سىلەرنىڭ ئاتا - ئاناڭلارنى (ئازدۇرۇپ) مەڭگۈلۈك باغلارتىن چىقىرىۋەتكەندەك، سىلەرنىمۇ ئازدۇرمىسۇن، شەيتان (ئازدۇرغۇچى) ئۇلارنىڭ ئەۋرەتلىرىنى ئۆزلىرىگە كۆرسىتىش ئۈچۈن كىيىملىرىنى سالغۇزۇۋەتكەن ئىدى (يەنى ئۇلارنىڭ يالىڭاچ بولۇپ قېلىشىغا شەيتان (ئازدۇرغۇچى) سەۋەبچى بولغان ئىدى). شەيتان (ئازدۇرغۇچى) ۋە ئۇنىڭ قول ئاستىدىكىلىرى سىلەرنى كۆرەلەيدۇ، سىلەر ئۇلارنى كۆرەلمەيسىلەر، شۈبھىسىزكى، بىز شەيتان (ئازدۇرغۇچى)لارنى ئىمان ئېيتمايدىغانلارنىڭ دوستى قىلدۇق
وَإِذَا فَعَلُوا فَاحِشَةً قَالُوا وَجَدْنَا عَلَيْهَا آبَاءَنَا وَاللَّهُ أَمَرَنَا بِهَا ۗ قُلْ إِنَّ اللَّهَ لَا يَأْمُرُ بِالْفَحْشَاءِ ۖ أَتَقُولُونَ عَلَى اللَّهِ مَا لَا تَعْلَمُونَ ﴿٢٨﴾
ئۇلار (يەنى مۇشرىكلار) بىرەر يامان ئىشنى قىلغان چاغلىرىرىدا: ‹‹ئاتا - بوۋىلىرىمىزنىڭ مۇشۇنداق قىلغانلىقىنى كۆرگەن ئىدۇق، اﷲ بىزنىمۇ شۇنداق قىلىشقا بۇيرۇغان›› دەيدۇ. (ئى مۇھەممەد! ئۇلارغا) ئېيىتقىنكى، ‹‹اﷲ (بەندىلىرىنى) ئىشەنچلىك ھالدا يامان ئىشقا بۇيرۇمايدۇ، ئۆزلىرىڭلار بىلمەيدىغان نەرسىلەرنى اﷲ نامىدىن يالغان ئېيتامسىلەر؟ ››
قُلْ أَمَرَ رَبِّي بِالْقِسْطِ ۖ وَأَقِيمُوا وُجُوهَكُمْ عِندَ كُلِّ مَسْجِدٍ وَادْعُوهُ مُخْلِصِينَ لَهُ الدِّينَ ۚ كَمَا بَدَأَكُمْ تَعُودُونَ ﴿٢٩﴾
(ئى مۇھەممەد!) ئېيىتقىنكى، «پەرۋەردىگارىم مېنى ئادىل بولۇشقا بۇيرۇدى. ھەربىر سەجدە قىلغاندا پۈتۈنلەي اﷲ قا يۈزلىنىڭلار، اﷲ قا ئىخلاس بىلەن ئىبادەت قىلىڭلار، سىلەر اﷲ دەسلەپتە پەيدا قىلغان ھالىتىڭلاردا (ئەسلىگە) قايتىسىلەر»
فَرِيقًا هَدَىٰ وَفَرِيقًا حَقَّ عَلَيْهِمُ الضَّلَالَةُ ۗ إِنَّهُمُ اتَّخَذُوا الشَّيَاطِينَ أَوْلِيَاءَ مِن دُونِ اللَّهِ وَيَحْسَبُونَ أَنَّهُم مُّهْتَدُونَ ﴿٣٠﴾
اﷲ (سىلەردىن) بىر گۇرۇھنى ھىدايەت قىلدى؛ يەنە بىر گۇرۇھقا گۇمراھلىق تېگىشلىك بولدى. شۈبھىسىزكى، ئۇلار اﷲ نى قويۇپ، شەيتان (ئازدۇرغۇچى)لارنى دوست تۇتقان تۇرۇقلۇق ئۆزلىرىنى توغرا يولدا گۇمان قىلىدۇ
يَا بَنِي آدَمَ خُذُوا زِينَتَكُمْ عِندَ كُلِّ مَسْجِدٍ وَكُلُوا وَاشْرَبُوا وَلَا تُسْرِفُوا ۚ إِنَّهُ لَا يُحِبُّ الْمُسْرِفِينَ ﴿٣١﴾
ئى ئادەم بالىلىرى! ھەر ناماز ۋاقتىدا (ياكى تاۋاپ ۋاقتىدا ئەۋرىتىڭلارنى سەترى قىلىپ تۇرىدىغان) كىيىمىڭلارنى كىيىڭلار، يەڭلار، ئىچىڭلار، ئىسراپ قىلماڭلار، اﷲ ئىسراپ قىلغۇچىلارنى ئىشەنچلىك ھالدا ياقتۇرمايدۇ
قُلْ مَنْ حَرَّمَ زِينَةَ اللَّهِ الَّتِي أَخْرَجَ لِعِبَادِهِ وَالطَّيِّبَاتِ مِنَ الرِّزْقِ ۚ قُلْ هِيَ لِلَّذِينَ آمَنُوا فِي الْحَيَاةِ الدُّنْيَا خَالِصَةً يَوْمَ الْقِيَامَةِ ۗ كَذَٰلِكَ نُفَصِّلُ الْآيَاتِ لِقَوْمٍ يَعْلَمُونَ ﴿٣٢﴾
ئېيىتقىنكى، «اﷲ بەندىلىرى ئۈچۈن ياراتقان لىباسلارنى، شېرىن، پاك رىزىقلارنى كىم ھارام قىلدى؟» ئېيىتقىنكى، ‹‹ئۇلار بۇ دۇنيادا مۆمىنلەر ئۈچۈن يارىتىلغان (گەرچە ئۇلارغا كۇففارلار شېرىك بولسىمۇ)، ئاخىرەتتە بولسا مۆمىنلەرگىلا خاستۇر». (اﷲ نىڭ بىر ۋە شېرىكى يوق ئىكەنلىكىنى بىلىدىغان قەۋم ئۈچۈن ئايەتلەرنى مۇشۇنداق تەپسىلى بايان قىلىمىز
قُلْ إِنَّمَا حَرَّمَ رَبِّيَ الْفَوَاحِشَ مَا ظَهَرَ مِنْهَا وَمَا بَطَنَ وَالْإِثْمَ وَالْبَغْيَ بِغَيْرِ الْحَقِّ وَأَن تُشْرِكُوا بِاللَّهِ مَا لَمْ يُنَزِّلْ بِهِ سُلْطَانًا وَأَن تَقُولُوا عَلَى اللَّهِ مَا لَا تَعْلَمُونَ ﴿٣٣﴾
(ئى مۇھەممەد! ئۇلارغا) ئېيىتقىنكى، «پەرۋەردىگارىم ئاشكارا ۋە يوشۇرۇن يامان ئىشلارنىڭ ھەممىسىنى، گۇناھلارنى، (كىشىلەرگە) ناھەق چېقىلىشنى، (اﷲ نىڭ شېرىكى بولۇشقا) اﷲ ھېچقانداق دەلىل چۈشۈرمىگەن نەرسىلەرنى اﷲ قا شېرىك كەلتۈرۈشنى، ئۆزۈڭلار بىلمەيدىغان نەرسىلەرنى اﷲ نامىدىن قالايمىقان سۆزلەشنى ھارام قىلدى››
وَلِكُلِّ أُمَّةٍ أَجَلٌ ۖ فَإِذَا جَاءَ أَجَلُهُمْ لَا يَسْتَأْخِرُونَ سَاعَةً ۖ وَلَا يَسْتَقْدِمُونَ ﴿٣٤﴾
(خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلغان) ھەر بىر ئۈممەتنىڭ (ھالاك بولىدىغان) مۇئەييەن ۋاقتى بولىدۇ، (ھالاك بولىدىغان) ۋاقتى يەتكەندە، ئۇلار بىردەممۇ ئىلگىرى - كېيىن بولمايدۇ (يەنى دەرھال ھالاك بولىدۇ)
يَا بَنِي آدَمَ إِمَّا يَأْتِيَنَّكُمْ رُسُلٌ مِّنكُمْ يَقُصُّونَ عَلَيْكُمْ آيَاتِي ۙ فَمَنِ اتَّقَىٰ وَأَصْلَحَ فَلَا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُونَ ﴿٣٥﴾
ئى ئادەم بالىلىرى! ئىچىڭلاردىن مېنىڭ ئايەتلىرىمنى سىلەرگە بايان بېرىدىغان خۇدانىڭ ئەلچىسىلىرىم كەلگەندە، تەقۋادارلىق قىلغان ۋە (ئەمەلىنى) تۈزەتكەنلەرگە (كەلگۈسىدىن) غەم قىلىش، (كەتكەنگە) قايغۇرۇش يوقتۇر
وَالَّذِينَ كَذَّبُوا بِآيَاتِنَا وَاسْتَكْبَرُوا عَنْهَا أُولَٰئِكَ أَصْحَابُ النَّارِ ۖ هُمْ فِيهَا خَالِدُونَ ﴿٣٦﴾
بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلار، ئۇنىڭغا ئىمان ئېيتىشقا گەدەنكەشلىك قىلغانلار دوزاخ ئەھلىدۇر، ئۇلار ھەمىشە دوزاختا قالغۇسىدۇ
فَمَنْ أَظْلَمُ مِمَّنِ افْتَرَىٰ عَلَى اللَّهِ كَذِبًا أَوْ كَذَّبَ بِآيَاتِهِ ۚ أُولَٰئِكَ يَنَالُهُمْ نَصِيبُهُم مِّنَ الْكِتَابِ ۖ حَتَّىٰ إِذَا جَاءَتْهُمْ رُسُلُنَا يَتَوَفَّوْنَهُمْ قَالُوا أَيْنَ مَا كُنتُمْ تَدْعُونَ مِن دُونِ اللَّهِ ۖ قَالُوا ضَلُّوا عَنَّا وَشَهِدُوا عَلَىٰ أَنفُسِهِمْ أَنَّهُمْ كَانُوا كَافِرِينَ ﴿٣٧﴾
اﷲ قا يالغان چاپلىغان ياكى ئۇنىڭ ئايەتلىرىنى ئىنكار قىلغانلاردىنمۇ زالىم كىشى بارمۇ؟ ئۇلار لەۋھۇلمەھپۇزدا يېزىلغان (رىزىقتىن ۋە ئەجەلدىن) ئۆز نېسىۋىسىنى كۆرىدۇ، ھەتتا بىزنىڭ پەرىشتىلىرىمىز ئۇلارنىڭ جانلىرىنى ئېلىش ئۈچۈن كەلگەندىمۇ، ئۇلارغا: ‹‹اﷲ نى قويۇپ ئىبادەت قىلغان مەبۇدلىرىڭلار قەيەردە؟ ›› دەيدۇ، ئۇلار: ‹‹ئۇلار بىزدىن غايىب بولۇپ كەتتى›› دەيدۇ. ئۇلار ھەقىقەتنى رەت قىلغۇچى بولغانلىقلىرىغا ئۆزلىرىنڭ زىيىنىغا گۇۋاھلىق بېرىدۇ (يەنى ھەقىقەتنى رەت قىلغۇچى بولغانلىقلىرىغا ئىقرار قىلىدۇ)
قَالَ ادْخُلُوا فِي أُمَمٍ قَدْ خَلَتْ مِن قَبْلِكُم مِّنَ الْجِنِّ وَالْإِنسِ فِي النَّارِ ۖ كُلَّمَا دَخَلَتْ أُمَّةٌ لَّعَنَتْ أُخْتَهَا ۖ حَتَّىٰ إِذَا ادَّارَكُوا فِيهَا جَمِيعًا قَالَتْ أُخْرَاهُمْ لِأُولَاهُمْ رَبَّنَا هَٰؤُلَاءِ أَضَلُّونَا فَآتِهِمْ عَذَابًا ضِعْفًا مِّنَ النَّارِ ۖ قَالَ لِكُلٍّ ضِعْفٌ وَلَٰكِن لَّا تَعْلَمُونَ ﴿٣٨﴾
اﷲ (ئايەتلىرىنى ئىنكار قىلغانلارغا) ئېيتىدۇ: ‹‹سىلەردىن ئىلگىرى ئۆتكەن، جىنلاردىن ۋە ئىنسانلاردىن بولغان (ھەقىقەتنى رەت قىلغۇچى) ئۈممەتلەر بىلەن بىرلىكتە دوزاخقا كىرىڭلار››. ھەرقاچان بىر ئۈممەت دوزاخقا كىرىدىكەن، ئۇ ئۆزىدىن ئىلگىرىكى ئۈممەتكە (ئۇنىڭغا ئەگىشىپ گۇمراھ بولغانلىقى ئۈچۈن) لەنەت ئوقۇيدۇ. ئۇلارنىڭ ھەممىسى دوزاخقا كىرىپ بولۇپ جەملەشكەندە، ئۇلارنىڭ ئىچىدىكى كېيىن كىرگەنلەر (يەنى ئەگەشكۈچىلەر) ئىلگىرى كىرگەنلەرگە (يەنى باشلىقلىرىغا) قارىتىپ: ‹‹ئى پەرۋەردىگارىمىز! بۇلار بىزنى ئازدۇرغان. ئۇلارغا ئىككى ھەسسە دوزاخ ئازابى بەرگىن›› دەيدۇ اﷲ: ‹‹ھەممىگە ئىككى ھەسسە ئازاب بېرىلىدۇ، لېكىن سىلەر (ئازابنىڭ دەھشىتىنى) بىلمەيسىلەر›› دەيدۇ
وَقَالَتْ أُولَاهُمْ لِأُخْرَاهُمْ فَمَا كَانَ لَكُمْ عَلَيْنَا مِن فَضْلٍ فَذُوقُوا الْعَذَابَ بِمَا كُنتُمْ تَكْسِبُونَ ﴿٣٩﴾
ئۇلارنىڭ ئارىسىدىكى ئىلگىرىكىلەر كېيىنكىلەرگە: ‹‹سىلەرنىڭ بىزدىن ھېچقانداق ئارتۇقچىلىقىڭلار يوق (يەنى گۇمراھلىقتا ۋەئازابقا تېگىشلىك بولۇشتا بىر - بىرىمىزگە ئوخشاشمىز)، قىلمىشلىرىڭلار تۈپەيلىدىن ئازابنى تېتىڭلار›› دەيدۇ
إِنَّ الَّذِينَ كَذَّبُوا بِآيَاتِنَا وَاسْتَكْبَرُوا عَنْهَا لَا تُفَتَّحُ لَهُمْ أَبْوَابُ السَّمَاءِ وَلَا يَدْخُلُونَ الْجَنَّةَ حَتَّىٰ يَلِجَ الْجَمَلُ فِي سَمِّ الْخِيَاطِ ۚ وَكَذَٰلِكَ نَجْزِي الْمُجْرِمِينَ ﴿٤٠﴾
شۈبھىسىزكى، بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغان ۋە ئۇلارغا ئىمان ئېيتىشقا گەدەنكەشلىك قىلغانلارغا ئاسماننىڭ دەرۋازىسى ئېچىلمايدۇ (يەنى ئۇلارنىڭ ياخشى ئەمەلى ۋە دۇئاسى قوبۇل بولمايدۇ)، تاكى تۆگە يىڭنىنىڭ تۆشۈكىدىن ئۆتكۈچە ئۇلار مەڭگۈلۈك باغلاركە كىرمەيدۇ (يەنى ئۇلار ھەرگىزمۇ مەڭگۈلۈك باغلاركە كىرمەيدۇ)، گۇناھكارلارغا مۇشۇنداق جازا بېرىمىز
لَهُم مِّن جَهَنَّمَ مِهَادٌ وَمِن فَوْقِهِمْ غَوَاشٍ ۚ وَكَذَٰلِكَ نَجْزِي الظَّالِمِينَ ﴿٤١﴾
دوزاختا ئۇلارنىڭ ئاستىغا سالىدىغان كۆرپىسىمۇ ئوتتىن، ئۈستىگە يېپىنىدىغان يوتقىنىمۇ ئوتتىن بولىدۇ، زالىملارغا مۇشۇنداق جازا بېرىمىز
وَالَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ لَا نُكَلِّفُ نَفْسًا إِلَّا وُسْعَهَا أُولَٰئِكَ أَصْحَابُ الْجَنَّةِ ۖ هُمْ فِيهَا خَالِدُونَ ﴿٤٢﴾
ئىمان ئېيتقانلار ۋە ياخشى ئەمەللەرنى قىلغانلار - كىشىنى پەقەت كۈچى يېتىدىغان ئىشقىلا تەكلىپ قىلىمىز ئەنە شۇلار ئەھلى مەڭگۈلۈك باغلارتۇر، ئۇلار مەڭگۈلۈك باغلارتە ھەمىشە قالىدۇ
وَنَزَعْنَا مَا فِي صُدُورِهِم مِّنْ غِلٍّ تَجْرِي مِن تَحْتِهِمُ الْأَنْهَارُ ۖ وَقَالُوا الْحَمْدُ لِلَّهِ الَّذِي هَدَانَا لِهَٰذَا وَمَا كُنَّا لِنَهْتَدِيَ لَوْلَا أَنْ هَدَانَا اللَّهُ ۖ لَقَدْ جَاءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ۖ وَنُودُوا أَن تِلْكُمُ الْجَنَّةُ أُورِثْتُمُوهَا بِمَا كُنتُمْ تَعْمَلُونَ ﴿٤٣﴾
ئۇلارنىڭ كۆڭۈللىرىدىكى ئۆچمەنلىكلىرىنى ئېلىپ تاشلىدۇق (يەنى مەڭگۈلۈك باغلاركە كىرگەنلەر ئارىسىدا ئۆزئارا ئاغرىق - ئاداۋەت بولمايدۇ). ئۇلارنىڭ (سارايلىرى) ئاستىدىن (مەڭگۈلۈك باغلارنىڭ) ئۆستەڭلىرى ئېقىپ تۇرىدۇ، ئۇلار: «جىمى ھەمدۇسانا بىزنى بۇ (نېمەتلەرنى ھاسىل قىلىشقا) يېتەكلىگەن اﷲ قا بولسۇنكى، اﷲ بىزنى يېتەكلىمىگەن بولسا، توغرا يول تاپمىغان بولاتتۇق، پەرۋەردىگارىمىزنىڭ خۇدانىڭ ئەلچىسىلىرى بىزگە ھەقنى ئېلىپ كەلدى›› دەيدۇ، ئۇلارغا: «قىلغان ئەمەلىڭلار ئۈچۈن سىلەرگە بېرىلگەن مەڭگۈلۈك باغلار مانا مۇشۇ›› دەپ نىدا قىلىنىدۇ (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى شۇنداق دەپ نىدا قىلىدۇ)
وَنَادَىٰ أَصْحَابُ الْجَنَّةِ أَصْحَابَ النَّارِ أَن قَدْ وَجَدْنَا مَا وَعَدَنَا رَبُّنَا حَقًّا فَهَلْ وَجَدتُّم مَّا وَعَدَ رَبُّكُمْ حَقًّا ۖ قَالُوا نَعَمْ ۚ فَأَذَّنَ مُؤَذِّنٌ بَيْنَهُمْ أَن لَّعْنَةُ اللَّهِ عَلَى الظَّالِمِينَ ﴿٤٤﴾
ئەھلى دوزاخقا: ‹‹بىز پەرۋەردىگارىمىز بىزگە ۋەدە قىلغان نەرسىنى ھەق تاپتۇق، سىلەرمۇ پەرۋەردىگارىڭلار ۋەدە قىلغان نەرسىنى ھەق تاپتىڭلارمۇ؟ ›› دەپ توۋلايدۇ. ئۇلار: ‹‹ھەئە›› دەيدۇ، شۇنىڭ بىلەن بىر جاكارچى ئۇلارنىڭ ئارىسىدا (مۇنداق دەپ) جاكارلايدۇ: ‹‹زالىملارغا اﷲ نىڭ لەنىتى بولسۇن
الَّذِينَ يَصُدُّونَ عَن سَبِيلِ اللَّهِ وَيَبْغُونَهَا عِوَجًا وَهُم بِالْآخِرَةِ كَافِرُونَ ﴿٤٥﴾
زالىملار كىشىلەرنى اﷲ نىڭ داغدام يولىدىن توسىدۇ، ئۇنىڭ ئەگرى بولۇشىنى تىلەيدۇ ھەمدە ئۇلار ئاخىرەتكە ئىشەنمەيدۇ
وَبَيْنَهُمَا حِجَابٌ ۚ وَعَلَى الْأَعْرَافِ رِجَالٌ يَعْرِفُونَ كُلًّا بِسِيمَاهُمْ ۚ وَنَادَوْا أَصْحَابَ الْجَنَّةِ أَن سَلَامٌ عَلَيْكُمْ ۚ لَمْ يَدْخُلُوهَا وَهُمْ يَطْمَعُونَ ﴿٤٦﴾
ئۇلارنىڭ ئارىسىدا بىر توسما بولىدۇ. توسمىنىڭ ئۈستىدە (ئەئرافتا) ئەھلى مەڭگۈلۈك باغلار ۋە ئەھلى دوزاخنى ئۇلارنىڭ سىماسىدىن تونۇيدىغان ئادەملەر (يەنى ئەھلى ئەئراف) بولىدۇ، ئۇلار ئەھلى مەڭگۈلۈك باغلاركە: «سىلەرگە ئامانلىق تىلەيمىز›› دەپ توۋلايدۇ. ئۇلار مەڭگۈلۈك باغلاركە كىرمىگەن، (ئەمما) كىرىش ئۈمىدىدە تۇرغانلاردۇر
وَإِذَا صُرِفَتْ أَبْصَارُهُمْ تِلْقَاءَ أَصْحَابِ النَّارِ قَالُوا رَبَّنَا لَا تَجْعَلْنَا مَعَ الْقَوْمِ الظَّالِمِينَ ﴿٤٧﴾
ئۇلارنىڭ كۆزلىرى ئەھلى دوزاخ تەرەپكە بۇرۇلغاندا: ‹‹ئى پەرۋەردىگارىمىز! بىزنى زالىم قەۋم بىلەن بىللە قىلمىغىن›› دەيدۇ
وَنَادَىٰ أَصْحَابُ الْأَعْرَافِ رِجَالًا يَعْرِفُونَهُم بِسِيمَاهُمْ قَالُوا مَا أَغْنَىٰ عَنكُمْ جَمْعُكُمْ وَمَا كُنتُمْ تَسْتَكْبِرُونَ ﴿٤٨﴾
ئەئرافتىكىلەر (يەنى ياخشىلىقلىرى بىلەن يامانلىقلىرى تەڭ بولۇپ قالغان، مەڭگۈلۈك باغلارتىمۇ ئەمەس، دوزاختىمۇ ئەمەس كىشىلەر) سىماسىدىن تونۇيدىغان (دوزىخى) ئادەملەرگە: ‹‹توپلىغان پۇل - ماللىرىڭلار ۋە تەكەببۇرلۇقۇڭلار سىلەرگە نېمىگە ئەسقاتتى؟ ›› دەپ توۋلايدۇ
أَهَٰؤُلَاءِ الَّذِينَ أَقْسَمْتُمْ لَا يَنَالُهُمُ اللَّهُ بِرَحْمَةٍ ۚ ادْخُلُوا الْجَنَّةَ لَا خَوْفٌ عَلَيْكُمْ وَلَا أَنتُمْ تَحْزَنُونَ ﴿٤٩﴾
(ئەئرافتىكىلەر دوزىخىلار مۆمىنلەرنىڭ پېقىرلىرىنى كۆرسىتىپ تۇرۇپ) سىلەر اﷲ نىڭ رەھمىتىگە ئېرىشەلمەيدۇ، دەپ قەسەم قىلغان كىشىلەر مۇشۇلارمۇ؟ (مانا بۇلارغا) مەڭگۈلۈك باغلاركە كىرىڭلار، سىلەرگە (كەلگۈسىدىن) غەم قىلىش، (كەتكەنگە) قايغۇرۇش يوقتۇر (دېيىلدى دەيدۇ)
وَنَادَىٰ أَصْحَابُ النَّارِ أَصْحَابَ الْجَنَّةِ أَنْ أَفِيضُوا عَلَيْنَا مِنَ الْمَاءِ أَوْ مِمَّا رَزَقَكُمُ اللَّهُ ۚ قَالُوا إِنَّ اللَّهَ حَرَّمَهُمَا عَلَى الْكَافِرِينَ ﴿٥٠﴾
ئەھلى دوزاخ ئەھلى مەڭگۈلۈك باغلاركە: ‹‹سۇدىن ياكى اﷲ سىلەرگە بەرگەن ئىچىدىغان نەرسىلەردىن بىزگە بىر ئاز قۇيۇپ بەرسەڭلار! ›› دەپ توۋلايدۇ، ئەھلى مەڭگۈلۈك باغلار ئۇلارغا (جاۋابەن): ‹‹اﷲ ئۇلارنى ھەقىقەتنى رەت قىلغۇچىلارغا ھارام قىلغان›› دەيدۇ
الَّذِينَ اتَّخَذُوا دِينَهُمْ لَهْوًا وَلَعِبًا وَغَرَّتْهُمُ الْحَيَاةُ الدُّنْيَا ۚ فَالْيَوْمَ نَنسَاهُمْ كَمَا نَسُوا لِقَاءَ يَوْمِهِمْ هَٰذَا وَمَا كَانُوا بِآيَاتِنَا يَجْحَدُونَ ﴿٥١﴾
ھەقىقەتنى رەت قىلغۇچىلار دىنلىرىنى مەسخىرە ۋە ئويۇنچۇق قىلىۋالدى، ئۇلارنى دۇنيا تىرىكچىلىكى ئالدىدى، ئۇلارنىڭ بۈگۈنكى كۈنگە مۇلاقات بولۇشنى ئۇنتۇغانلىقلىرى ۋە بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلىقلىرىغا ئوخشاش، بىز بۈگۈن ئۇلارنى ئۇنتۇيمىز (يەنى ئۇلارن دوزاختا قالدۇرىمىز)
وَلَقَدْ جِئْنَاهُم بِكِتَابٍ فَصَّلْنَاهُ عَلَىٰ عِلْمٍ هُدًى وَرَحْمَةً لِّقَوْمٍ يُؤْمِنُونَ ﴿٥٢﴾
ئۇلارغا مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) بەردۇق، ئۇنى (يەنى قۇرئاننىڭ مەنىللىرىنى) بىلش ئاساسىدا بايان ۋە رەھمەت قىلدۇق
هَلْ يَنظُرُونَ إِلَّا تَأْوِيلَهُ ۚ يَوْمَ يَأْتِي تَأْوِيلُهُ يَقُولُ الَّذِينَ نَسُوهُ مِن قَبْلُ قَدْ جَاءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ فَهَل لَّنَا مِن شُفَعَاءَ فَيَشْفَعُوا لَنَا أَوْ نُرَدُّ فَنَعْمَلَ غَيْرَ الَّذِي كُنَّا نَعْمَلُ ۚ قَدْ خَسِرُوا أَنفُسَهُمْ وَضَلَّ عَنْهُم مَّا كَانُوا يَفْتَرُونَ ﴿٥٣﴾
ئۇلار (يەنى كۇففارلار) پەقەت (ئۆزلىرىگە ۋەدە قىلىنغان ئازابنىڭ) نەتىجىسىنىلا كۈتىدۇ، ئۇنىڭ نەتىجىسى كەلگەن كۈنى (يەنى قىيامەت كۈنى)، ئلگىرى مۇقەددەس كىتاب-ۋەھىيقا ئەمەل قىلمىغانلار: ‹‹پەرۋەردىگارىمىزنىڭ ئەلچىللىرى ھەقىقەتنى ئېلىپ كەلگەن ئىدى (بىز ئۇلارغا ئىمان ئېيتمىدۇق، بۈگۈن) بىزگە شاپائەت قىلىدىغان شاپائەتچىلەر بارمىدۇ؟ ياكى گۇناھ قىلىشتىن قول ئۈزۈپ، ياخشى ئەمەللەرنى قىلىش ئۈچۈن دۇنياغا قايتۇرۇلۇشىمىزغا بولامدىغاندۇ›› دەيدۇ. ئۇلار ئۆزلىرىگە زىيان سالدى، ئۇلارنىڭ اﷲ نىڭ شېرىكلىرى دەپ ئىپتىرا قىلغان مەبۇدلىرى ئۇلارنىڭ كۆزىدىن غايىب بولدى
إِنَّ رَبَّكُمُ اللَّهُ الَّذِي خَلَقَ السَّمَاوَاتِ وَالْأَرْضَ فِي سِتَّةِ أَيَّامٍ ثُمَّ اسْتَوَىٰ عَلَى الْعَرْشِ يُغْشِي اللَّيْلَ النَّهَارَ يَطْلُبُهُ حَثِيثًا وَالشَّمْسَ وَالْقَمَرَ وَالنُّجُومَ مُسَخَّرَاتٍ بِأَمْرِهِ ۗ أَلَا لَهُ الْخَلْقُ وَالْأَمْرُ ۗ تَبَارَكَ اللَّهُ رَبُّ الْعَالَمِينَ ﴿٥٤﴾
شۈبھىسىزكى، پەرۋەردىگارىڭلار اﷲ ئاسمانلارنى ۋە زېمىننى ئالتە كۈندە (يەنى ئالتە دەۋردە) ياراتتى، ئاندىن (ئۆزىنىڭ ئۇلۇغلۇقىغا لايىق رەۋىشتە) ئەرش ئۈستىدە قارار ئالدى، اﷲ كېچە بىلەن (كېچىنىڭ قاراڭغۇلۇقى بىلەن) كۈندۈزنى ياپىدۇ، كېچە بىلەن كۈندۈز بىر - بىرىنى قوغلىشىدۇ (يەنى داۋاملىق ئورۇن ئالمىشىپ تۇرىدۇ): اﷲ قۇياش، ئاي ۋە يۇلتۇزلارنى اﷲ نىڭ ئەمرىگە بويسۇندۇرۇلغۇچى قىلىپ ياراتتى، يارىتىش ۋە ئەمر قىلىش (يەنى كائىناتنى تەسەررۇپ قىلىش) راستىنلا اﷲ نىڭ ئىلكىدىدۇر. ئالەملەرنىڭ پەرۋەردىگارى اﷲ نىڭ دەرىجىسى كاتتىدۇر
ادْعُوا رَبَّكُمْ تَضَرُّعًا وَخُفْيَةً ۚ إِنَّهُ لَا يُحِبُّ الْمُعْتَدِينَ ﴿٥٥﴾
پەرۋەردىگارىڭلارغا تۆۋەنچىلىك بىلەن يوشۇرۇن دۇئا قىلىڭلار، (دۇئا قىلغاندا كەلسە - كەلمەس سۆزلەپ، توۋلاپ) ھەددىدىن ئاشۇرۋەتكەنلەرنى اﷲ ياقتۇرمايدۇ
وَلَا تُفْسِدُوا فِي الْأَرْضِ بَعْدَ إِصْلَاحِهَا وَادْعُوهُ خَوْفًا وَطَمَعًا ۚ إِنَّ رَحْمَتَ اللَّهِ قَرِيبٌ مِّنَ الْمُحْسِنِينَ ﴿٥٦﴾
اﷲ (خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتىپ) يەر يۈزىنى تۈزىگەندىن كېيىن، يەر يۈزىدە بۇزۇقچىلىق قىلماڭلار (اﷲ نىڭ ئازابىدىن) قورققان ۋە (رەھمىتىنى) ئۇمىد قىلغان ھالدا دۇئا قىلىڭلار، شۈبھىسىزكى اﷲ نىڭ رەھمىتى ياخشىلىق قىلغۇچىلارغا يېقىندۇر
وَهُوَ الَّذِي يُرْسِلُ الرِّيَاحَ بُشْرًا بَيْنَ يَدَيْ رَحْمَتِهِ ۖ حَتَّىٰ إِذَا أَقَلَّتْ سَحَابًا ثِقَالًا سُقْنَاهُ لِبَلَدٍ مَّيِّتٍ فَأَنزَلْنَا بِهِ الْمَاءَ فَأَخْرَجْنَا بِهِ مِن كُلِّ الثَّمَرَاتِ ۚ كَذَٰلِكَ نُخْرِجُ الْمَوْتَىٰ لَعَلَّكُمْ تَذَكَّرُونَ ﴿٥٧﴾
اﷲ نىڭ رەھمىتى (يەنى يامغۇر) مەيدانغا كېلىش ئالدىدا، اﷲ بېشارەتچى قىلىپ شاماللارنى ئەۋەتىدۇ، شاماللار ئېغىر بۇلۇتلارنى كۆتۈرگەن چاغدا، ئۇلارنى ئۆلۈك (يەنى قاغجىراپ كەتكەن گىياھسىز) جايلارغا سۈرىمىز، ئۇنىڭدىن يامغۇر ياغدۇرۇپ تۈرلۈك - تۈرلۈك مېۋىلەرنى چىقىرىمىز، ئىبرەت ئېلىشىڭلار ئۈچۈن (زېمىندىن مېۋىلەرنى چىقارغاندەك، قىيامەتتە) ئۆلۈكلەرنى قەبرىلىرىدىن چىقىرىمىز
وَالْبَلَدُ الطَّيِّبُ يَخْرُجُ نَبَاتُهُ بِإِذْنِ رَبِّهِ ۖ وَالَّذِي خَبُثَ لَا يَخْرُجُ إِلَّا نَكِدًا ۚ كَذَٰلِكَ نُصَرِّفُ الْآيَاتِ لِقَوْمٍ يَشْكُرُونَ ﴿٥٨﴾
مۇنبەت يەرنىڭ گىياھسى اﷲ نىڭ ئىزنى بىلەن تولۇق ئۈنىدۇ (بۇ ۋەز - نەسىھەتكە قۇلاق سېلىپ، ئۇنىڭدىن پايدىلىنىدىغان مۆمىننىڭ مىسالىدۇر)، ناچار يەرنىڭ گىياھسى ئاز ئۆسىدۇ (بۇ ۋەز - نەسىھەتتىن پايدىلانمايدىغان ھەقىقەتنى رەت قىلغۇچىنىڭ مىسالىدۇر)، شۈكۈر قىلىدىغان قەۋم ئۈچۈن ئايەتلىرىمىزنى مۇشۇنداق شەكىللەردە بايان قىلىمىز
لَقَدْ أَرْسَلْنَا نُوحًا إِلَىٰ قَوْمِهِ فَقَالَ يَا قَوْمِ اعْبُدُوا اللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُ إِنِّي أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ عَظِيمٍ ﴿٥٩﴾
شەك شۈبھىسىزكى، نۇھنى ئۆز قەۋمىگە خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق، ئۇ: ‹‹ئى قەۋمىم! اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە اﷲ تىن باشقا مەبۇد (بەرھەق) يوقتۇر، مەن ئىشەنچلىك ھالدا سىلەرنىڭ بۈيۈك كۈننىڭ (يەنى قىيامەت كۈنىنىڭ) ئازابىغا قېلىشىڭلاردىن قورقىمەن›› دېدى
قَالَ الْمَلَأُ مِن قَوْمِهِ إِنَّا لَنَرَاكَ فِي ضَلَالٍ مُّبِينٍ ﴿٦٠﴾
قەۋمىنىڭ چوڭلىرى: ‹‹بىز سېنى ئىشەنچلىك ھالدا روشەن گۇمراھلىقتا كۆرۈمىز›› دېدى
قَالَ يَا قَوْمِ لَيْسَ بِي ضَلَالَةٌ وَلَٰكِنِّي رَسُولٌ مِّن رَّبِّ الْعَالَمِينَ ﴿٦١﴾
نۇھ ئېيتتى: ‹‹ئى قەۋمىم! مەن قىلچىمۇ گۇمراھ ئەمەسمەن، لېكىن مەن ئالەملەرنىڭ پەرۋەردىگارى تەرىپىدىن ئەۋەتىلگەن خۇدانىڭ ئەلچىسىمەن
أُبَلِّغُكُمْ رِسَالَاتِ رَبِّي وَأَنصَحُ لَكُمْ وَأَعْلَمُ مِنَ اللَّهِ مَا لَا تَعْلَمُونَ ﴿٦٢﴾
سىلەرگە پەرۋەردىگارىمنىڭ ئەلچىلىكىنى يەتكۈزىمەن، سىلەرگە نەسىھەت قىلىمەن، مەن اﷲ نىڭ ۋەھيى قىلىشى بىلەن سىلەر بىلمەيدىغان نەرسىلەرنى بىلىمەن
أَوَعَجِبْتُمْ أَن جَاءَكُمْ ذِكْرٌ مِّن رَّبِّكُمْ عَلَىٰ رَجُلٍ مِّنكُمْ لِيُنذِرَكُمْ وَلِتَتَّقُوا وَلَعَلَّكُمْ تُرْحَمُونَ ﴿٦٣﴾
سىلەرنى ئاگاھلاندۇرۇش ئۈچۈن، (كۇفرىدىن) ساقلىنىشىڭلار ۋە اﷲ نىڭ رەھمىتىگە ئېرىشىشىڭلار ئۈچۈن، سىلەرگە ئاراڭلاردىكى بىر كىشى ئارقىلىق پەرۋەردىگارىڭلارنىڭ ۋەھيىسى كېلىشىدىن ئەجەبلەنمەمسىلەر؟ ››
فَكَذَّبُوهُ فَأَنجَيْنَاهُ وَالَّذِينَ مَعَهُ فِي الْفُلْكِ وَأَغْرَقْنَا الَّذِينَ كَذَّبُوا بِآيَاتِنَا ۚ إِنَّهُمْ كَانُوا قَوْمًا عَمِينَ ﴿٦٤﴾
ئۇلار نۇھنى ئىنكار قىلدى، نۇھنى ئۇنىڭ بىلەن كېمىدە بىللە بولغانلارنى قۇتقۇزدۇق، ئايەتلىرىمىزنى ئىنكار قىلغانلارنى پۈتۈنلەي (سۇغا) غەرق قىلدۇق. شۈبھىسىزكى، ئۇلار (ھەقنى كۆرۈشتىن دىللىرى) كور قەۋم ئىدى
وَإِلَىٰ عَادٍ أَخَاهُمْ هُودًا ۗ قَالَ يَا قَوْمِ اعْبُدُوا اللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُ ۚ أَفَلَا تَتَّقُونَ ﴿٦٥﴾
ئاد (قەۋمى) گە ئۇلارنىڭ قېرىندىشى ھۇدنى ئەۋەتتۇق، ھۇد: ‹‹ئى قەۋمىم! اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە اﷲ تىن باشقا مەبۇد (بەرھەق) يوقتۇر. (ئۇنىڭ ئازابىدىن) قورقمامسىلەر؟ ›› دېدى
قَالَ الْمَلَأُ الَّذِينَ كَفَرُوا مِن قَوْمِهِ إِنَّا لَنَرَاكَ فِي سَفَاهَةٍ وَإِنَّا لَنَظُنُّكَ مِنَ الْكَاذِبِينَ ﴿٦٦﴾
ئۇنىڭ قەۋمىنىڭ ئىمان ئېيتمىغان چوڭلىرى: ‹‹بىز سېنى ئىشەنچلىك ھالدا ئەخمەق ھېسابلايمىز، بىز سېنى، ئەلۋەتتە، (خۇدانىڭ ئەلچىسىلىك دەۋايىڭدا) يالغانچىلاردىن دەپ گۇمان قىلىمىز›› دېدى
قَالَ يَا قَوْمِ لَيْسَ بِي سَفَاهَةٌ وَلَٰكِنِّي رَسُولٌ مِّن رَّبِّ الْعَالَمِينَ ﴿٦٧﴾
ھۇد ئېيتتى: ‹‹ئى قەۋمىم! مەن ئەخمەق ئەمەسمەن، لېكىن مەن ئالەملەرنىڭ پەرۋەردىگارى تەرىپىدىن ئەۋەتىلگەن خۇدانىڭ ئەلچىسىمەن
أُبَلِّغُكُمْ رِسَالَاتِ رَبِّي وَأَنَا لَكُمْ نَاصِحٌ أَمِينٌ ﴿٦٨﴾
سىلەرگە پەرۋەردىگارىمنىڭ ئەلچىلىكىنى يەتكۈزىمەن، مەن سىلەرگە ئىشەنچىلىك نەسىھەتچىمەن
أَوَعَجِبْتُمْ أَن جَاءَكُمْ ذِكْرٌ مِّن رَّبِّكُمْ عَلَىٰ رَجُلٍ مِّنكُمْ لِيُنذِرَكُمْ ۚ وَاذْكُرُوا إِذْ جَعَلَكُمْ خُلَفَاءَ مِن بَعْدِ قَوْمِ نُوحٍ وَزَادَكُمْ فِي الْخَلْقِ بَسْطَةً ۖ فَاذْكُرُوا آلَاءَ اللَّهِ لَعَلَّكُمْ تُفْلِحُونَ ﴿٦٩﴾
پەرۋەردىگارىڭلارنىڭ ۋەھيىسى سىلەرنى ئاگاھلاندۇرۇش ئۈچۈن ئۆز ئىچىڭلاردىكى بىر كىشى ئارقىلىق كەلگەنلىكىدىن ئەجەبلىنەمسىلەر؟ ›› ئۆز ۋاقتىدا نۇھنىڭ قەۋمى ھالاك بولغاندىن كېيىن، اﷲ نىڭ سىلەرنى ئۇلارنىڭ ئورۇنباسارلىرى قىلغانلىقىنى ۋۇجۇدۇڭلارنى، قامەتلىك، بەستلىك قىلغانلىقىنى ياد ئېتىڭلار، سىلەر بەختكە ئېرىشىشىڭلار ئۈچۈن اﷲ نىڭ نېمەتلىرىنى ياد ئېتىڭلار
قَالُوا أَجِئْتَنَا لِنَعْبُدَ اللَّهَ وَحْدَهُ وَنَذَرَ مَا كَانَ يَعْبُدُ آبَاؤُنَا ۖ فَأْتِنَا بِمَا تَعِدُنَا إِن كُنتَ مِنَ الصَّادِقِينَ ﴿٧٠﴾
ئۇلار: ‹‹ (ئى ھۇد) ئاتا - بوۋىللىرىمىز ئىبادەت قىلىپ كېلىۋاتقان مەبۇدلارنى تاشلاپ، يالغۇز بىر اﷲ قا ئىبادەت قىلىشىمىزغا دەۋەت قىلغىلى كەلدىڭمۇ؟ ئەگەر راست سۆزلىگۈچىلەردىن بولساڭ، بىزگە ۋەدە قىلغان ئازابنى كەلتۈرگىن›› دېدى
قَالَ قَدْ وَقَعَ عَلَيْكُم مِّن رَّبِّكُمْ رِجْسٌ وَغَضَبٌ ۖ أَتُجَادِلُونَنِي فِي أَسْمَاءٍ سَمَّيْتُمُوهَا أَنتُمْ وَآبَاؤُكُم مَّا نَزَّلَ اللَّهُ بِهَا مِن سُلْطَانٍ ۚ فَانتَظِرُوا إِنِّي مَعَكُم مِّنَ الْمُنتَظِرِينَ ﴿٧١﴾
ھۇد: ‹‹سىلەرگە چوقۇم پەرۋەردىگارىڭلارنىڭ ئازابى ۋە غەزىپى نازىل بولىدۇ، سىلەر ئۆزۈڭلار ۋە ئاتا - بوۋاڭلار ئات قويۇۋالغان، بۇ ھەقتە (ئۇلارغا ئىبادەت قىلىشقا) اﷲ ھېچقانداق دەلىل نازىل قىلمىغان (بۇتلىرىڭلارنىڭ) ئىسىملىرى ئۈستىدە مەن بىلەن مۇنازىرىلىشەمسىلەر؟ (اﷲ نىڭ ئازابىنى) كۈتۈڭلار، مەنمۇ ئىشەنچلىك ھالدا سىلەر بىلەن بىرلىكتە (سىلەرگە نازىل بولىدىغان ئازابنى) كۈتىمەن›› دېدى
فَأَنجَيْنَاهُ وَالَّذِينَ مَعَهُ بِرَحْمَةٍ مِّنَّا وَقَطَعْنَا دَابِرَ الَّذِينَ كَذَّبُوا بِآيَاتِنَا ۖ وَمَا كَانُوا مُؤْمِنِينَ ﴿٧٢﴾
بىز ھۇدنى ۋە ئۇنىڭ بىلەن بىللە بولغۇچىلارنى (يەنى مۆمىنلەرنى) رەھمىتىمىز بىلەن قۇتقۇزدۇق، بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلارنى تەلتۆكۈس ھالاك قىلدۇق، ئۇلار ئىمان ئېيتمىغان ئىدى
وَإِلَىٰ ثَمُودَ أَخَاهُمْ صَالِحًا ۗ قَالَ يَا قَوْمِ اعْبُدُوا اللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُ ۖ قَدْ جَاءَتْكُم بَيِّنَةٌ مِّن رَّبِّكُمْ ۖ هَٰذِهِ نَاقَةُ اللَّهِ لَكُمْ آيَةً ۖ فَذَرُوهَا تَأْكُلْ فِي أَرْضِ اللَّهِ ۖ وَلَا تَمَسُّوهَا بِسُوءٍ فَيَأْخُذَكُمْ عَذَابٌ أَلِيمٌ ﴿٧٣﴾
سەمۇدقا ئۇلارنىڭ قېرىندىشى سالىھنى (خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق). ئۇ: «ئى قەۋمىم! اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر. سىلەرگە ئىشەنچلىك ھالدا پەرۋەردىگارىڭلار تەرىپىدىن (مېنىڭ خۇدانىڭ ئەلچىسىلىكىمنى ئىسپاتلايدىغان) روشەن مۆجىزە كەلدى، مانا بۇ - اﷲ (بىۋاسىتە) ياراتقان چېشى تۆگىمۇ سىلەرگە مېنىڭ مۆجىزەمدۇر، ئۇنى اﷲ نىڭ زېمىنىغا قويۇۋېتىڭلار، ئىختىيارىي ئوتلىسۇن، ئۇنىڭغا زىيان - زەخمەت يەتكۈزمەڭلار، بولمىسا، سىلەرنى قاتتىق ئازاب ھالاك قىلىدۇ
وَاذْكُرُوا إِذْ جَعَلَكُمْ خُلَفَاءَ مِن بَعْدِ عَادٍ وَبَوَّأَكُمْ فِي الْأَرْضِ تَتَّخِذُونَ مِن سُهُولِهَا قُصُورًا وَتَنْحِتُونَ الْجِبَالَ بُيُوتًا ۖ فَاذْكُرُوا آلَاءَ اللَّهِ وَلَا تَعْثَوْا فِي الْأَرْضِ مُفْسِدِينَ ﴿٧٤﴾
ياد ئېتىڭلاركى، ئۆز ۋاقتىدا اﷲ سىلەرنى ئاد قەۋمىدىن كېيىن ئۇلارنىڭ ئورۇنباسارلىرى قىلدى، سىلەرنى (ھىجر) زېمىنىغا ئورۇنلاشتۇردى، ئۇنىڭ تۈزلەڭلىرىگە ئىمارەتلەر سالدىڭلار، تاغلارنى تېشىپ ئۆيلەر ياسىدىڭلار. اﷲ نىڭ نېمەتلىرىنى ياد ئېتىڭلار، زېمىندا بۇزغۇنچىلىق قىلماڭلار» دېدى
قَالَ الْمَلَأُ الَّذِينَ اسْتَكْبَرُوا مِن قَوْمِهِ لِلَّذِينَ اسْتُضْعِفُوا لِمَنْ آمَنَ مِنْهُمْ أَتَعْلَمُونَ أَنَّ صَالِحًا مُّرْسَلٌ مِّن رَّبِّهِ ۚ قَالُوا إِنَّا بِمَا أُرْسِلَ بِهِ مُؤْمِنُونَ ﴿٧٥﴾
ئۇنىڭ قەۋمىنىڭ تەكەببۇر چوڭلىرى بوزەك قىلىنغانلارغا يەنى ئۇلارنىڭ ئىچىدىكى مۆمىنلەرگە: «سىلەر سالىھنى ئۇنىڭ پەرۋەردىگارى تەرىپىدىن ئەۋەتىلگەن خۇدانىڭ ئەلچىسى دەپ بىلەمسىلەر؟» دېدى. ئۇلار: «بىز ئۇنىڭ خۇدانىڭ ئەلچىسىلىكىگە ئەلۋەتتە ئىشىنىمىز» دېدى
قَالَ الَّذِينَ اسْتَكْبَرُوا إِنَّا بِالَّذِي آمَنتُم بِهِ كَافِرُونَ ﴿٧٦﴾
ھېلىقى تەكەببۇرلار: «سىلەر ئىشەنگەنگە بىز ھەرگىز ئىشەنمەيمىز» دېدى
فَعَقَرُوا النَّاقَةَ وَعَتَوْا عَنْ أَمْرِ رَبِّهِمْ وَقَالُوا يَا صَالِحُ ائْتِنَا بِمَا تَعِدُنَا إِن كُنتَ مِنَ الْمُرْسَلِينَ ﴿٧٧﴾
ئۇلار چىشى تۆگىنى بوغۇزلىدى، پەرۋەردىگارىنىڭ ئەمرىنى تۇتۇشتىن باش تارتتى ۋە ئى سالىھ: «راستتىنلا خۇدانىڭ ئەلچى-رەسۇللىرىدىن بولساڭ، بىزگە ۋەدە قىلغان ئازابنى كەلتۈرۈپ باق» دېدى
فَأَخَذَتْهُمُ الرَّجْفَةُ فَأَصْبَحُوا فِي دَارِهِمْ جَاثِمِينَ ﴿٧٨﴾
ئۇلارغا قاتتىق ئازاب يۈزلەندى - دە، ئۇلار ئۆيلىرىدە ئولتۇرغان پېتى قېتىپ قالدى
فَتَوَلَّىٰ عَنْهُمْ وَقَالَ يَا قَوْمِ لَقَدْ أَبْلَغْتُكُمْ رِسَالَةَ رَبِّي وَنَصَحْتُ لَكُمْ وَلَٰكِن لَّا تُحِبُّونَ النَّاصِحِينَ ﴿٧٩﴾
ئۇ (ئۇلار ھالاك بولغاندىن كېيىن) ئۇلارغا ئارقىسىنى قىلدى ۋە: «ئى قەۋمىم! سىلەرگە ئىشەنچلىك ھالدا پەرۋەردىگارىمنىڭ ئەلچىلىكىنى يەتكۈزدۈم، سىلەرگە نەسىھەت قىلدىم، لېكىن سىلەر نەسىھەت قىلغۇچىلارنى ياقتۇرمىدىڭلار» دېدى
وَلُوطًا إِذْ قَالَ لِقَوْمِهِ أَتَأْتُونَ الْفَاحِشَةَ مَا سَبَقَكُم بِهَا مِنْ أَحَدٍ مِّنَ الْعَالَمِينَ ﴿٨٠﴾
لۇتنى (ئەھلى سەدومغا خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق)، ئۆز ۋاقتىدا ئۇ ئۆز قەۋمىگە (ئۇلارنى ئەيىبلەش يۈزىسىدىن) مۇنداق دېگەن ئىدى: «سىلەر شۇنداق قەبىھ ئىشنى قىلامسىلەر؟ مۇنداق قەبىھ ئىشنى سىلەردىن ئىلگىرى جاھان ئەھلىدىن ھېچقانداق ئادەم قىلمىغان ئىدى
إِنَّكُمْ لَتَأْتُونَ الرِّجَالَ شَهْوَةً مِّن دُونِ النِّسَاءِ ۚ بَلْ أَنتُمْ قَوْمٌ مُّسْرِفُونَ ﴿٨١﴾
سىلەر ئىشەنچلىك ھالدا ئاياللارنى تاشلاپ قويۇپ، ئەرلەر بىلەن جىنسىي تەلىپىڭلارنى قاندۇرىسىلەر، سىلەر ئىشەنچلىك ھالدا ھەددىدىن ئاشقۇچى قەۋمسىلەر»
وَمَا كَانَ جَوَابَ قَوْمِهِ إِلَّا أَن قَالُوا أَخْرِجُوهُم مِّن قَرْيَتِكُمْ ۖ إِنَّهُمْ أُنَاسٌ يَتَطَهَّرُونَ ﴿٨٢﴾
لۇتنىڭ قەۋمىنىڭ بىردىنبىر جاۋابى: «لۇتنى ئۇنىڭ ئەگەشكۈچىلىرى بىلەن قوشۇپ شەھرىڭلاردىن چىقىرىۋېتىڭلار، چۈنكى ئۇلار پاك كىشىلەردۇر» دېيىشتىن ئىبارەت بولدى
فَأَنجَيْنَاهُ وَأَهْلَهُ إِلَّا امْرَأَتَهُ كَانَتْ مِنَ الْغَابِرِينَ ﴿٨٣﴾
لۇتنى ئۇنىڭ خوتۇنىدىن باشقا، تەۋە كىشىلىرى بىلەن بىللە (ئۇنىڭ قەۋمىگە نازىل بولغان ئازابتىن) قۇتقۇزدۇق، پەقەت خوتۇنىلا (شەھەردە) قىلىپ ھالاك بولغۇچىلاردىن بولدى
وَأَمْطَرْنَا عَلَيْهِم مَّطَرًا ۖ فَانظُرْ كَيْفَ كَانَ عَاقِبَةُ الْمُجْرِمِينَ ﴿٨٤﴾
ئۇلارغا بىر تۈرلۈك يامغۇر (يەنى تاش) ياغدۇردۇق، گۇناھلارنىڭ ئاقىۋىتىنىڭ قانداق بولىدىغانلىقىغا قارىغىن
وَإِلَىٰ مَدْيَنَ أَخَاهُمْ شُعَيْبًا ۗ قَالَ يَا قَوْمِ اعْبُدُوا اللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُ ۖ قَدْ جَاءَتْكُم بَيِّنَةٌ مِّن رَّبِّكُمْ ۖ فَأَوْفُوا الْكَيْلَ وَالْمِيزَانَ وَلَا تَبْخَسُوا النَّاسَ أَشْيَاءَهُمْ وَلَا تُفْسِدُوا فِي الْأَرْضِ بَعْدَ إِصْلَاحِهَا ۚ ذَٰلِكُمْ خَيْرٌ لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ ﴿٨٥﴾
مەديەن (خەلقىگە) ئۇلارنىڭ قېرىندىشى شۇئەيبنى (خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق)، ئۇ (ئۆز قەۋمىگە) ئېيتتى: «ئى قەۋمىم! اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر. سىلەرگە پەرۋەردىگارىڭلاردىن ئىشەنچلىك ھالدا (مېنىڭ راستلىقىمنى ئىسپاتلايدىغان) مۆجىزە كەلدى، ئۆلچەمنى ۋە تارازىنى توغرىلاڭلار، كىشىلەرنىڭ نەرسىلىرىنى كەم بەرمەڭلار. (اﷲ خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتىپ) زېمىننى تۈزىگەندىن كېيىن، زېمىندا (گۇناھلار قىلىپ) بۇزغۇنچىلىق قىلماڭلار، ئەگەر مۆمىن بولساڭلار، ئەنە شۇ سىلەر ئۈچۈن ياخشىدۇر
وَلَا تَقْعُدُوا بِكُلِّ صِرَاطٍ تُوعِدُونَ وَتَصُدُّونَ عَن سَبِيلِ اللَّهِ مَنْ آمَنَ بِهِ وَتَبْغُونَهَا عِوَجًا ۚ وَاذْكُرُوا إِذْ كُنتُمْ قَلِيلًا فَكَثَّرَكُمْ ۖ وَانظُرُوا كَيْفَ كَانَ عَاقِبَةُ الْمُفْسِدِينَ ﴿٨٦﴾
سىلەر اﷲ قا ئىمان ئېيتقانلارنى قورقۇتقان، اﷲ نىڭ يولىدىن توسقان ۋە ئۇ يولنىڭ ئەگرىلىكىنى تىلىگەن ھالدا يوللاردا ئولتۇرماڭلار، ئەسلىدە سىلەر ئاز ئىدىڭلار، اﷲ نىڭ سىلەرنىڭ سانىڭلارنى كۆپەيتكەنلىكىنى ياد ئېتىڭلار، بۇزۇقچىلىق قىلغۇچىلارنىڭ ئاقىۋىتىنىڭ قانداق بولىدىغانلىقىغا قاراڭلار
وَإِن كَانَ طَائِفَةٌ مِّنكُمْ آمَنُوا بِالَّذِي أُرْسِلْتُ بِهِ وَطَائِفَةٌ لَّمْ يُؤْمِنُوا فَاصْبِرُوا حَتَّىٰ يَحْكُمَ اللَّهُ بَيْنَنَا ۚ وَهُوَ خَيْرُ الْحَاكِمِينَ ﴿٨٧﴾
ئەگەر سىلەردىن بىر گۇرۇھ ئادەم مېنىڭ خۇدانىڭ ئەلچىسىلىكىمگە ئىشەنگەن، يەنە بىر گۇرۇھ ئادەم ئىشەنمىگەن بولسا، اﷲ بىزنىڭ ئارىمىزدا ھۆكۈم چىقارغۇچە سەۋر قىلىپ تۇرۇڭلار. اﷲ ھۆكۈم چىقارغۇچىلارنىڭ ئەڭ ياخشىسىدۇر»
قَالَ الْمَلَأُ الَّذِينَ اسْتَكْبَرُوا مِن قَوْمِهِ لَنُخْرِجَنَّكَ يَا شُعَيْبُ وَالَّذِينَ آمَنُوا مَعَكَ مِن قَرْيَتِنَا أَوْ لَتَعُودُنَّ فِي مِلَّتِنَا ۚ قَالَ أَوَلَوْ كُنَّا كَارِهِينَ ﴿٨٨﴾
شۇئەيب قەۋمىنىڭ (ئىماندىن باش تارتقان) تەكەببۇر چوڭلىرى: «ئى شۇئەيب! سېنى چوقۇم ئەگەشكۈچىلىرىڭ (يەنى ساڭا ئىمان ئېيتقانلار) بىلەن قوشۇپ شەھرىمىزدىن ھەيدەپ چىقىرىمىز، ياكى چوقۇم بىزنىڭ دىنىمىزغا قايتىشىڭلار كېرەك» دېدى. شۇئەيب: «بىز (بۇنىڭ ھەر ئىككىلىسىنى يامان كۆرۈدىغان تۇرساق (بىزنى يەنىلا شۇنىڭغا مەجبۇر قىلامسىلەر؟)» دېدى
قَدِ افْتَرَيْنَا عَلَى اللَّهِ كَذِبًا إِنْ عُدْنَا فِي مِلَّتِكُم بَعْدَ إِذْ نَجَّانَا اللَّهُ مِنْهَا ۚ وَمَا يَكُونُ لَنَا أَن نَّعُودَ فِيهَا إِلَّا أَن يَشَاءَ اللَّهُ رَبُّنَا ۚ وَسِعَ رَبُّنَا كُلَّ شَيْءٍ عِلْمًا ۚ عَلَى اللَّهِ تَوَكَّلْنَا ۚ رَبَّنَا افْتَحْ بَيْنَنَا وَبَيْنَ قَوْمِنَا بِالْحَقِّ وَأَنتَ خَيْرُ الْفَاتِحِينَ ﴿٨٩﴾
(شۇئەيب يەنى ئېيتتى) «اﷲ بىزنى سىلەرنىڭ دىنىڭلاردىن قۇتۇلدۇرغاندىن كېيىن ئۇنىڭغا قايتساق، ئەلۋەتتە، اﷲ قا يالغان چاپلىغان بولىمىز. پەرۋەردىگارىمىز اﷲ خالىسىلا، بىز سىلەرنىڭ دىنىڭلارغا قايتمايمىز. پەرۋەردىگارىمىزنىڭ ئىلمى ھەممىنى ئۆز ئىچىگە ئالىدۇ، اﷲ قا تەۋۋەككۈل قىلدۇق (يەنى ھەممە ئىشىمىزنى اﷲ قا تاپشۇرىمىز). پەرۋەردىگارىمىز! بىز بىلەن قەۋمىمىز ئارىسىدا ھەق ھۆكۈم چىقارغىن، سەن ھۆكۈم چىقارغۇچىلارنىڭ ئەڭ ياخشىسىدۇرسەن»
وَقَالَ الْمَلَأُ الَّذِينَ كَفَرُوا مِن قَوْمِهِ لَئِنِ اتَّبَعْتُمْ شُعَيْبًا إِنَّكُمْ إِذًا لَّخَاسِرُونَ ﴿٩٠﴾
ئۇنىڭ قەۋمىنىڭ ئىمانسىز چوڭلىرى: «ئەگەر سىلەر شۇئەيبكە ئەگەشسەڭلار، شۈبھىسىزكى، چوقۇم زىيان تارتىسىلەر» دېدى
فَأَخَذَتْهُمُ الرَّجْفَةُ فَأَصْبَحُوا فِي دَارِهِمْ جَاثِمِينَ ﴿٩١﴾
ئۇلارغا قاتتىق زىلزىلە يۈزلەندى، ئۇلار ئۆيلىرىدە ئولتۇرغان پېتى قېتىپ قالدى
الَّذِينَ كَذَّبُوا شُعَيْبًا كَأَن لَّمْ يَغْنَوْا فِيهَا ۚ الَّذِينَ كَذَّبُوا شُعَيْبًا كَانُوا هُمُ الْخَاسِرِينَ ﴿٩٢﴾
شۇئەيبنى ئىنكار قىلغانلار (اﷲ نىڭ ھالاك قىلىشى بىلەن) گويا بۇ يەردە تۇرمىغاندەك (يەنى ياشىمىغاندەك) بولۇپ قالدى. شۇئەيبنى ئىنكار قىلغانلار زىيان تارتقۇچىلاردىن بولدى
فَتَوَلَّىٰ عَنْهُمْ وَقَالَ يَا قَوْمِ لَقَدْ أَبْلَغْتُكُمْ رِسَالَاتِ رَبِّي وَنَصَحْتُ لَكُمْ ۖ فَكَيْفَ آسَىٰ عَلَىٰ قَوْمٍ كَافِرِينَ ﴿٩٣﴾
شۇئەيب ئۇلارغا ئارقىسىنى قىلىپ: «ئى قەۋمىم! مەن سىلەرگە ئىشەنچلىك ھالدا پەرۋەدىگارىمنىڭ ئەلچىلىكىنى يەتكۈزدۈم ۋە نەسىھەت قىلدىم، مەن قانداقمۇ ھەقىقەتنى رەت قىلغۇچى قەۋمگە قايغۇراي!» دېدى
وَمَا أَرْسَلْنَا فِي قَرْيَةٍ مِّن نَّبِيٍّ إِلَّا أَخَذْنَا أَهْلَهَا بِالْبَأْسَاءِ وَالضَّرَّاءِ لَعَلَّهُمْ يَضَّرَّعُونَ ﴿٩٤﴾
بىز مەيلى قايسى شەھەرگە بولسۇن، بىرەر خۇدانىڭ ئەلچىسى ئەۋەتكەن بولساق (شەھەر ئەھلى ئۇنى ئىنكار قىلغان بولسا)، ئۇلارنىڭ اﷲ قا يالۋۇرۇشلىرى (تەۋبە قىلىشلىرى) ئۈچۈن، ئۇلارنى نامراتلىققا، كېسەللىككە گىرىپتار قىلدۇق
ثُمَّ بَدَّلْنَا مَكَانَ السَّيِّئَةِ الْحَسَنَةَ حَتَّىٰ عَفَوا وَّقَالُوا قَدْ مَسَّ آبَاءَنَا الضَّرَّاءُ وَالسَّرَّاءُ فَأَخَذْنَاهُم بَغْتَةً وَهُمْ لَا يَشْعُرُونَ ﴿٩٥﴾
ئاندىن (ئۇلارنىڭ بېشىغا كەلگەن) كۈلپەتنى نېمەتكە ئايلاندۇردۇق، شۇنىڭ بىلەن ئۇلار كۆپەيدى، ئۇلار: «بىزنىڭ ئاتا - بوۋىلىرىمىزنىڭ بېشىغا ئىشەنچلىك ھالدا كۈلپەتلەر، نېمەتلەر كەلگەن ئىدى (بۇ - زاماننىڭ ئادىتى؛ اﷲ تەرىپىدىن كەلگەن ئوقۇبەت ئەمەس)» دېدى. ئۇلار خەۋەرسىز تۇرغاندا، ئۇلارنى ئۇشتۇمتۇت ھالاك قىلدۇق
وَلَوْ أَنَّ أَهْلَ الْقُرَىٰ آمَنُوا وَاتَّقَوْا لَفَتَحْنَا عَلَيْهِم بَرَكَاتٍ مِّنَ السَّمَاءِ وَالْأَرْضِ وَلَٰكِن كَذَّبُوا فَأَخَذْنَاهُم بِمَا كَانُوا يَكْسِبُونَ ﴿٩٦﴾
(خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلغانلىقتىن ھالاك قىلىنغان) شەھەرلەرنىڭ ئاھالىسى ئىمان ئېيتقان ۋە (كۇفرىدىن، گۇناھلاردىن) ساقلانغان بولسا ئىدى، ئەلۋەتتە، ئۇلارنى ئاسمان - زېمىننىڭ پاراۋانلىقلىرىغا مۇيەسسەر قىلاتتۇق، لېكىن ئۇلار (خۇدانىڭ ئەلچى-رەسۇللىرىنى) ئىنكار قىلدى، شۇ ئۇلارنى ئۆز قىلمىشلىرى تۈپەيلىدىن ھالاك قىلدۇق
أَفَأَمِنَ أَهْلُ الْقُرَىٰ أَن يَأْتِيَهُم بَأْسُنَا بَيَاتًا وَهُمْ نَائِمُونَ ﴿٩٧﴾
(خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلىدىغان قىلىدىغان) شەھەرلەرنىڭ ئاھالىسى بىزنىڭ ئازابىمىزنىڭ ئۆزلىرىگە كېچىسى (غەپلەتتە) ئۇخلاۋاتقانلىرىدا كېلىشىدىن قورقمامدۇ
أَوَأَمِنَ أَهْلُ الْقُرَىٰ أَن يَأْتِيَهُم بَأْسُنَا ضُحًى وَهُمْ يَلْعَبُونَ ﴿٩٨﴾
شەھەرلەر ئاھالىسى بىزنىڭ ئازابىمىزنىڭ چۈش ۋاقتىدا - ئۇلار ئويناپ تۇرغانلىرىدا كېلىشىدىن قورقمامدۇ
أَفَأَمِنُوا مَكْرَ اللَّهِ ۚ فَلَا يَأْمَنُ مَكْرَ اللَّهِ إِلَّا الْقَوْمُ الْخَاسِرُونَ ﴿٩٩﴾
ئۇلار اﷲ نىڭ مەكرىدىن (يەنى ئۇلار تۇيۇقسىز تۇرغاندا اﷲ نىڭ ئازاب نازىل قىلىشىدىن) قورقمامدۇ؟ پەقەت زىيان تارتقۇچى قەۋملا اﷲ نىڭ مەكرىدىن قورقمايدۇ
أَوَلَمْ يَهْدِ لِلَّذِينَ يَرِثُونَ الْأَرْضَ مِن بَعْدِ أَهْلِهَا أَن لَّوْ نَشَاءُ أَصَبْنَاهُم بِذُنُوبِهِمْ ۚ وَنَطْبَعُ عَلَىٰ قُلُوبِهِمْ فَهُمْ لَا يَسْمَعُونَ ﴿١٠٠﴾
زېمىننىڭ ئىلگىرىكى ئىگىلىرى (ھالاك بولغان) دىن كېيىن (زېمىنغا) ۋارىس بولغۇچىلارنى، خالىساق، گۇناھى تۈپەيلىدىن ھالاك قىلىدىغانلىقىمىز، دىللىرىنى پىچەتلىۋېتىدىغانلىقىمىز ئۇلارغا (يەنى زېمىنغا ۋارىسلىق قىلغۇچىلارغا) ئايان بولمىدىمۇ؟ ئۇلار (ھەق سۆزنى) ئاڭلىمايدۇ
تِلْكَ الْقُرَىٰ نَقُصُّ عَلَيْكَ مِنْ أَنبَائِهَا ۚ وَلَقَدْ جَاءَتْهُمْ رُسُلُهُم بِالْبَيِّنَاتِ فَمَا كَانُوا لِيُؤْمِنُوا بِمَا كَذَّبُوا مِن قَبْلُ ۚ كَذَٰلِكَ يَطْبَعُ اللَّهُ عَلَىٰ قُلُوبِ الْكَافِرِينَ ﴿١٠١﴾
ئاشۇ شەھەرلەر (ئاھالىسى) نىڭ بەزى خەۋەرلىرىنى (ئى مۇھەممەد!) ساڭا ھېكايە قىلىپ بېرىمىز، شەك - شۈبھىسىزكى، ئۇلارنىڭ خۇدانىڭ ئەلچىسىلىرى ئۇلارغا مۆجىزىلەر ئېلىپ كەلگەن ئىدى، ئۇلار ئىلگىرى (ھەقنى) ئىنكار قىلغانلىقلىرى ئۈچۈن (ئۇ مۆجىزىلەرگە) ئىمان ئېيتمىدى. اﷲ ھەقىقەتنى رەت قىلغۇچىلارنىڭ دىللىرىنى ئەنە شۇنداق پېچەتلەيدۇ
وَمَا وَجَدْنَا لِأَكْثَرِهِم مِّنْ عَهْدٍ ۖ وَإِن وَجَدْنَا أَكْثَرَهُمْ لَفَاسِقِينَ ﴿١٠٢﴾
ئۇلارنىڭ تولىسىنىڭ (روزىمىساقتا) بەرگەن ۋەدىسىگە ۋاپا قىلىدىغانلىقىنى بايقىمىدۇق، ئۇلارنىڭ تولىسىنىڭ شەكسىز پاسىقلار (يەنى اﷲ نىڭ ئەمر - پەرمانىدىن باش تارتقۇچىلار) ئىكەنلىكىنى بايقىدۇق
ثُمَّ بَعَثْنَا مِن بَعْدِهِم مُّوسَىٰ بِآيَاتِنَا إِلَىٰ فِرْعَوْنَ وَمَلَئِهِ فَظَلَمُوا بِهَا ۖ فَانظُرْ كَيْفَ كَانَ عَاقِبَةُ الْمُفْسِدِينَ ﴿١٠٣﴾
مەزكۇز خۇدانىڭ ئەلچى-رەسۇللىرىدىن كېيىن، مۇسانى مۆجىزىلىرىمىز بىلەن، (مىسىرنىڭ مۇسا زامانىدىكى پادىشاھى) پىرئەۋنگە ۋە ئۇنىڭ قەۋمىنىڭ چوڭلىرىغا (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، ئۇلار بۇ مۆجىزىلەرنى ئىنكار قىلدى، بۇزغۇنچىلىق قىلغۇچىلارنىڭ ئاقىۋىتىنىڭ قانداق بولىدىغانلىقىغا قارىغىن
وَقَالَ مُوسَىٰ يَا فِرْعَوْنُ إِنِّي رَسُولٌ مِّن رَّبِّ الْعَالَمِينَ ﴿١٠٤﴾
مۇسا ئېيتتى: «ئى پىرئەۋن! مەن ئىشەنچلىك ھالدا ئالەملەرنىڭ پەرۋەردىگارى ئەۋەتكەن خۇدانىڭ ئەلچىسىمەن
حَقِيقٌ عَلَىٰ أَن لَّا أَقُولَ عَلَى اللَّهِ إِلَّا الْحَقَّ ۚ قَدْ جِئْتُكُم بِبَيِّنَةٍ مِّن رَّبِّكُمْ فَأَرْسِلْ مَعِيَ بَنِي إِسْرَائِيلَ ﴿١٠٥﴾
اﷲ نامىدىن پەقەت ھەقىقەتتىن باشقىنى ئېيتماسلىققا لايىقمەن، راستتىنلا پەرۋەردىگارىڭلار تەرىپىدىن سىلەرگە (مېنىڭ راستلىقىمنى ئىسپاتلايدىغان) بىر مۆجىزە ئېلىپ كەلدىم، شۇنىڭ ئۈچۈن، ئىسرائىل ئەۋلادىنى مەن بىلەن قويۇپ بەرگىن (مەن بىلەن بىللە ئاتا - بوۋىلىرىنىڭ ۋەتىنى مۇقەددەس زېمىنغا بارسۇن)»
قَالَ إِن كُنتَ جِئْتَ بِآيَةٍ فَأْتِ بِهَا إِن كُنتَ مِنَ الصَّادِقِينَ ﴿١٠٦﴾
پىرئەۋن (مۇساغا): «ئەگەر سەن پەرۋەردىگارىڭ تەرىپىدىن) بىر مۆجىزە ئېلىپ كەلگەن بولساڭ، (بۇ دەۋايىڭدا) راستچىللاردىن بولساڭ، ئۇنى بىزگە كۆرسەتكىن» دېدى
فَأَلْقَىٰ عَصَاهُ فَإِذَا هِيَ ثُعْبَانٌ مُّبِينٌ ﴿١٠٧﴾
مۇسا ھاسىسىنى تاشلىۋېدى، ئۇ ناگاھان ئاشكارا بىر ئەجدىھاغا ئايلاندى
وَنَزَعَ يَدَهُ فَإِذَا هِيَ بَيْضَاءُ لِلنَّاظِرِينَ ﴿١٠٨﴾
قولىنى (ياقىسىدىن) چىقىرىۋىدى، ئۇ ناگاھان قارىغۇچىلارغا (نۇر چاقنىتىپ تۇرىدىغان) ئاپئاق بولۇپ كۆرۈندى
قَالَ الْمَلَأُ مِن قَوْمِ فِرْعَوْنَ إِنَّ هَٰذَا لَسَاحِرٌ عَلِيمٌ ﴿١٠٩﴾
پىرئەۋن قەۋمنىڭ چوڭلىرى ئېيتتى: «بۇ ئىشەنچلىك ھالدا ناھايىتى ئۇستا سېھىرگەر ئىكەن
يُرِيدُ أَن يُخْرِجَكُم مِّنْ أَرْضِكُمْ ۖ فَمَاذَا تَأْمُرُونَ ﴿١١٠﴾
ئۇ سىلەرنى زېمىنىڭلار (يەنى مىسىر زېمىنى) دىن ھەيدەپ چىقىرىۋەتمەكچى. نېمە مەسلىھەت بېرىسىلەر؟»
قَالُوا أَرْجِهْ وَأَخَاهُ وَأَرْسِلْ فِي الْمَدَائِنِ حَاشِرِينَ ﴿١١١﴾
ئۇلار ئېيتتى: «(ئۇلارنىڭ ئىشىنى ئۆزۈڭ بىر پىكىرگە كەلگۈچە) تەخىر قىلغىن، شەھەرلەرگە (سېھىرگەرلەرنى) يىغقۇچى كىشىلەرنى ئەۋەتكىن
يَأْتُوكَ بِكُلِّ سَاحِرٍ عَلِيمٍ ﴿١١٢﴾
ئۇلار بارلىق ئۇستا سېھىرگەرلەرنى سېنىڭ ھۇزۇرۇڭغا ئېلىپ كەلسۇن»
وَجَاءَ السَّحَرَةُ فِرْعَوْنَ قَالُوا إِنَّ لَنَا لَأَجْرًا إِن كُنَّا نَحْنُ الْغَالِبِينَ ﴿١١٣﴾
(پىرئەۋن سېھىرگەرلىرىنى يىغىشقا ئادەملىرىنى ئەۋەتتى) سېھىرگەرلەر پىرئەۋننىڭ قېشىغا كەلدى، ئۇلار: «ئەگەر بىز غەلىبە قىلساق، بىزگە چوقۇم مۇكاپات بېرىلەمدۇ؟» دېدى
قَالَ نَعَمْ وَإِنَّكُمْ لَمِنَ الْمُقَرَّبِينَ ﴿١١٤﴾
پىرئەۋن: «ھەئە، (مۇكاپات بېرىلىدۇ، ئۇنىڭ ئۈستىگە) سىلەر چوقۇم مېنىڭ يېقىن كىشىلىرىم بولۇپ قالىسىلەر» دېدى
قَالُوا يَا مُوسَىٰ إِمَّا أَن تُلْقِيَ وَإِمَّا أَن نَّكُونَ نَحْنُ الْمُلْقِينَ ﴿١١٥﴾
ئۇلار (يەنى سېھىرگەرلەر): «ئى مۇسا! (ھاساڭنى) سەن ئاۋۋال تاشلامسەن، ياكى بىز (ئۆزىمىزنىڭكىنى) ئاۋۋال تاشلامدۇق؟» دېدى
قَالَ أَلْقُوا ۖ فَلَمَّا أَلْقَوْا سَحَرُوا أَعْيُنَ النَّاسِ وَاسْتَرْهَبُوهُمْ وَجَاءُوا بِسِحْرٍ عَظِيمٍ ﴿١١٦﴾
مۇسا: «سىلەر تاشلاڭلار!» دېدى. ئۇلار ھاسا، ئارغامچىلارنى تاشلاپ كىشىلەرنىڭ كۆزلىرىنى باغلىدى، ئۇلارنى (يەنى كىشىلەرنى) قاتتىق چۆچۈتۈۋەتتى، ئۇلار (كىشىلەرنىڭ كۆزلىرىگە) چوڭ (كۆرۈنىدىغان) سېھىرنى كۆرسەتتى
وَأَوْحَيْنَا إِلَىٰ مُوسَىٰ أَنْ أَلْقِ عَصَاكَ ۖ فَإِذَا هِيَ تَلْقَفُ مَا يَأْفِكُونَ ﴿١١٧﴾
مۇساغا: «ھاساڭنى تاشلىغىن» دەپ ۋەھىي قىلدۇق، (مۇسا ھاسىسىنى تاشلىۋىدى، ئۇ ئەجدىھاغا ئايلىنىپ) ئۇلارنىڭ ئويدۇرما نەرسىلىرىنى دەرھال يۇتۇۋەتتى
فَوَقَعَ الْحَقُّ وَبَطَلَ مَا كَانُوا يَعْمَلُونَ ﴿١١٨﴾
ھەقىقەت ئاشكارا بولدى، ئۇلارنىڭ كۆرسەتكەن سېھىرلىرى بەربات بولدى
فَغُلِبُوا هُنَالِكَ وَانقَلَبُوا صَاغِرِينَ ﴿١١٩﴾
بۇ يەردە ئۇلار (يەنى پىرئەۋن بىلەن قەۋمى) مەغلۇپ بولدى، خار بولغان ھالدا (شەھەرگە قايتتى)
وَأُلْقِيَ السَّحَرَةُ سَاجِدِينَ ﴿١٢٠﴾
سېھىرگەرلەر سەجدىگە باردى
قَالُوا آمَنَّا بِرَبِّ الْعَالَمِينَ ﴿١٢١﴾
ئۇلار دېدى: «ئالەملەرنىڭ پەرۋەردىگارىغا ئىمان ئېيتتۇق
رَبِّ مُوسَىٰ وَهَارُونَ ﴿١٢٢﴾
مۇسا ۋە ھارۇننىڭ پەرۋەردىگارىغا (ئىمان ئېيتتۇق)»
قَالَ فِرْعَوْنُ آمَنتُم بِهِ قَبْلَ أَنْ آذَنَ لَكُمْ ۖ إِنَّ هَٰذَا لَمَكْرٌ مَّكَرْتُمُوهُ فِي الْمَدِينَةِ لِتُخْرِجُوا مِنْهَا أَهْلَهَا ۖ فَسَوْفَ تَعْلَمُونَ ﴿١٢٣﴾
پىرئەۋن ئېيتتى: «سىلەر مەن رۇخسەت قىلماي تۇرۇپ مۇساغا ئىمان ئېيتتىڭلار، بۇ چوقۇم سىلەرنىڭ شەھەر (يەنى مىسىر) دىكى چېغىڭلاردا ئاھالىنى شەھەردىن ھەيدەپ چىقىرىش ئۈچۈن (مۇسا بىلەن بىرلىشىپ) ئالدىن پىلانلىغان ھىيلەڭلاردۇر (سىلەرنى قانداق جازالايدىغانلىقىمنى) ئۇزاققا قالماي بىلىسىلەر
لَأُقَطِّعَنَّ أَيْدِيَكُمْ وَأَرْجُلَكُم مِّنْ خِلَافٍ ثُمَّ لَأُصَلِّبَنَّكُمْ أَجْمَعِينَ ﴿١٢٤﴾
سىلەرنىڭ قولۇڭلارنى، پۇتۇڭلارنى چوقۇم ئوڭ - چەپ قىلىپ (ئوڭ قولۇڭلار بىلەن سول پۇتۇڭلارنى ياكى سول پۇتۇڭلار بىلەن ئوڭ قولۇڭلارنى) چوقۇم كېسىمەن، ئاندىن ھەممىڭلارنى چوقۇم دارغا ئېسىپ ئۆلتۈرىمەن»
قَالُوا إِنَّا إِلَىٰ رَبِّنَا مُنقَلِبُونَ ﴿١٢٥﴾
سېھىرگەرلەر ئېيتتى: «بىز، ئەلۋەتتە، پەرۋەردىگارىمىزنىڭ دەرگاھىغا قايتقۇچىلارمىز
وَمَا تَنقِمُ مِنَّا إِلَّا أَنْ آمَنَّا بِآيَاتِ رَبِّنَا لَمَّا جَاءَتْنَا ۚ رَبَّنَا أَفْرِغْ عَلَيْنَا صَبْرًا وَتَوَفَّنَا مُسْلِمِينَ ﴿١٢٦﴾
سەن بىزنى پەقەت پەرۋەردىگارىمىزنىڭ بىزگە نازىل قىلغان ئايەتلىرىگە ئىمان ئېيتقانلىقىمىز ئۈچۈنلا ئەيىبلەۋاتىسەن، پەرۋەردىگارىمىز! بىزگە سەۋر ئاتا قىلغىن، بىزنى مۇسۇلمان پېتىمىزچە قەبزى روھ قىلغىن»
وَقَالَ الْمَلَأُ مِن قَوْمِ فِرْعَوْنَ أَتَذَرُ مُوسَىٰ وَقَوْمَهُ لِيُفْ��ِدُوا فِي الْأَرْضِ وَيَذَرَكَ وَآلِهَتَكَ ۚ قَالَ سَنُقَتِّلُ أَبْنَاءَهُمْ وَنَسْتَحْيِي نِسَاءَهُمْ وَإِنَّا فَوْقَهُمْ قَاهِرُونَ ﴿١٢٧﴾
پىرئەۋن قەۋمىنىڭ چوڭلىرى (پىرئەۋنگە): «مۇسا بىلەن ئۇنىڭ قەۋمىنى زېمنىدا پىتنە - پاسات تۇغدۇرۇشقا، سېنى ۋە سېنىڭ ئىلاھلىرىڭنى تاشلاشقا قويۇۋېتەمسەن؟» دېدى. پىرئەۋن: «ئۇلارنىڭ باللىلىرىنى ئۆلتۈرىمىز، ئاياللىرىنى بولسا (خىزمەتكە سېلىشقا) قالدۇرۇپ قويىمىز، بىز ئەلۋەتتە ئۇلارنىڭ ئۈستىدىن ھۆكۈمرانلىق قىلىمىز» دېدى
قَالَ مُوسَىٰ لِقَوْمِهِ اسْتَعِينُوا بِاللَّهِ وَاصْبِرُوا ۖ إِنَّ الْأَرْضَ لِلَّهِ يُورِثُهَا مَن يَشَاءُ مِنْ عِبَادِهِ ۖ وَالْعَاقِبَةُ لِلْمُتَّقِينَ ﴿١٢٨﴾
مۇسا ئۆز قەۋمىگە: «اﷲ تىن مەدەت تىلەڭلار، سەۋر قىلىڭلار، يەر يۈزى ئىشەنچلىك ھالدا اﷲ نىڭ مۈلكىدۇر، (اﷲ) ئۇنىڭغا بەندىلىرىدىن ئۆزى خالىغان كىشىلەرنى ۋارىس قىلىدۇ؛ ياخشى ئاقىۋەت تەقۋادارلارغا مەنسۇپ» دېدى
قَالُوا أُوذِينَا مِن قَبْلِ أَن تَأْتِيَنَا وَمِن بَعْدِ مَا جِئْتَنَا ۚ قَالَ عَسَىٰ رَبُّكُمْ أَن يُهْلِكَ عَدُوَّكُمْ وَيَسْتَخْلِفَكُمْ فِي الْأَرْضِ فَيَنظُرَ كَيْفَ تَعْمَلُونَ ﴿١٢٩﴾
ئۇلار: «سەن بىزگە كېلىشتىن بۇرۇنمۇ، كەلگەندىن كېيىنمۇ ھامان خارلىنىپ كەلدۇق» دېدى. مۇسا: «پەرۋاردىگارىڭلارنىڭ دۈشمىنىڭلارنى ھالاك قېلىشى، (مىسىر) زېمىنىدا ئۇلارنىڭ ئورنىغا سىلەرنى دەسسىتىشى، سىلەرنىڭ قانداق ئىش قىلىدىغانلىقىڭلارغا قارىشىمۇ ھەقىقەقتۇر» دېدى
وَلَقَدْ أَخَذْنَا آلَ فِرْعَوْنَ بِالسِّنِينَ وَنَقْصٍ مِّنَ الثَّمَرَاتِ لَعَلَّهُمْ يَذَّكَّرُونَ ﴿١٣٠﴾
اﷲ قا قەسەمكى، بىز ئىشەنچلىك ھالدا پىرئەۋن ۋە ئۇنىڭ تەۋەلىرىنىڭ ئىبرەت ئېلىشلىرى ئۈچۈن، ئۇلارنى قەھەتچىلىك بىلەن، مېۋىلىرىنىڭ ھوسۇلىنى كېمەيتىۋېتىش بىلەن سىنىدۇق
فَإِذَا جَاءَتْهُمُ الْحَسَنَةُ قَالُوا لَنَا هَٰذِهِ ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌ يَطَّيَّرُوا بِمُوسَىٰ وَمَن مَّعَهُ ۗ أَلَا إِنَّمَا طَائِرُهُمْ عِندَ اللَّهِ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿١٣١﴾
ئۇلار ئەگەر (پاراۋانلىق - مولچىلىقتەك) بىرەر ياخشىلىققا ئېرىشسە، بىز بۇنىڭغا ھەقلىقمىز، دەيدۇ. ئەگەر ئۇلارغا بىرەر يامانلىق كەلسە، ئۇلار: بۇ مۇسا ۋە ئۇنىڭ بىلەن بىللە بولغان مۆمىنلەرنىڭ شۇملۇقىدىن كەلدى، دەيدۇ. ئۇلارغا يەتكەن ياخشىلىق بىلەن يامانلىق (مۇسانىڭ شۇملۇقىدىن ئەمەس) اﷲ نىڭ تەقدىرىدۇر، لېكىن ئۇلارنىڭ تولىسى بۇنى (يەنى ئۇلارغا كەلگەن بالا - قازانىڭ ئۆز گۇناھلىرى تۈپەيلى اﷲ تەرىپىدىن كەلگەنلىكىنى ئۇقمايدۇ
وَقَالُوا مَهْمَا تَأْتِنَا بِهِ مِنْ آيَةٍ لِّتَسْحَرَنَا بِهَا فَمَا نَحْنُ لَكَ بِمُؤْمِنِينَ ﴿١٣٢﴾
ئۇلار (يەنى پىرئەۋن قەۋمى): «كۆزىمىزنى باغلاش ئۈچۈن ھەرقانداق مۆجىزە كەلتۈرسەڭمۇ بىز ساڭا ھەرگىز ئىشەنمەيمىز» دېدى
فَأَرْسَلْنَا عَلَيْهِمُ الطُّوفَانَ وَالْجَرَادَ وَالْقُمَّلَ وَالضَّفَادِعَ وَالدَّمَ آيَاتٍ مُّفَصَّلَاتٍ فَاسْتَكْبَرُوا وَكَانُوا قَوْمًا مُّجْرِمِينَ ﴿١٣٣﴾
ئۇلارغا سۇ ئاپىتى، چېكەتكە، پىت، پاقا، قان (بالالىرىنى) روشەن مۆجىزىلەر قىلىپ ئەۋەتتۇق، (اﷲ قا ئىمان ئېيتىشقا) ئۇلارنىڭ گەدەنكەشلىكى يول قويمىدى، ئۇلار گۇناھكار قەۋم بولدى
وَلَمَّا وَقَعَ عَلَيْهِمُ الرِّجْزُ قَالُوا يَا مُوسَى ادْعُ لَنَا رَبَّكَ بِمَا عَهِدَ عِندَكَ ۖ لَئِن كَشَفْتَ عَنَّا الرِّجْزَ لَنُؤْمِنَنَّ لَكَ وَلَنُرْسِلَنَّ مَعَكَ بَنِي إِسْرَائِيلَ ﴿١٣٤﴾
ئۇلارغا بالا نازىل بولغاندا، ئۇلار: «ئى مۇسا! پەرۋەردىگارىڭغا ئۆزىنىڭ ساڭا ئاتا قىلغان ئەھدى بىلەن (يەنى ساڭا بەرگەن پەيغەمنبەرلىكنىڭ ھەقىقىي ھۆرمىتى بىلەن بىزدىن بالانى كۆتۈرۈۋېتىشكە) دۇئا قىلساڭ، اﷲ قا قەسەمكى، ئەگەر بىزدىن بالانى كۆتۈرۈۋەتسەڭ، ساڭا چوقۇم ئىمان ئېيتىمىز، ئىسرائىل ئەۋلادىنى چوقۇم سەن بىلەن بىللە قويۇپ بېرىمىز» دېگەن ئىدى
فَلَمَّا كَشَفْنَا عَنْهُمُ الرِّجْزَ إِلَىٰ أَجَلٍ هُم بَالِغُوهُ إِذَا هُمْ يَنكُثُونَ ﴿١٣٥﴾
ئۇلاردىن بالانى ۋاقىتلىق كۆتۈرۈۋەتكەن چېغىمىزدا، ئۇلار توساتتىن ئەھدىلىرىنى بۇزدى
فَانتَقَمْنَا مِنْهُمْ فَأَغْرَقْنَاهُمْ فِي الْيَمِّ بِأَنَّهُمْ كَذَّبُوا بِآيَاتِنَا وَكَانُوا عَنْهَا غَافِلِينَ ﴿١٣٦﴾
ئايەتلىرىمىزنى ئىنكار قىلغانلىقلىرى ۋە ئايەتلىرىمىزدىن غاپىل بولغانلىقلىرى ئۈچۈن، ئۇلارنى دېڭىزغا غەرق قىلىپ جازالىدۇق
وَأَوْرَثْنَا الْقَوْمَ الَّذِينَ كَانُوا يُسْتَضْعَفُونَ مَشَارِقَ الْأَرْضِ وَمَغَارِبَهَا الَّتِي بَارَكْنَا فِيهَا ۖ وَتَمَّتْ كَلِمَتُ رَبِّكَ الْحُسْنَىٰ عَلَىٰ بَنِي إِسْرَائِيلَ بِمَا صَبَرُوا ۖ وَدَمَّرْنَا مَا كَانَ يَصْنَعُ فِرْعَوْنُ وَقَوْمُهُ وَمَا كَانُوا يَعْرِشُونَ ﴿١٣٧﴾
بوزەك قىلىنغان قەۋمنى (يەنى بەنى ئىسرائىلنى) بىز بەرىكەتلىك قىلغان (شام) زېمىننىڭ شەرق تەرەپلىرىگە ۋە غەرب تەرەپلىرىگە (يەنى ھەممە تەرەپكە) ۋارىس قىلدۇق. ئىسرائىل ئەۋلادى سەۋر قىلغانلىقلىرى ئۈچۈن، پەرۋەردىگارىڭنىڭ ئۇلارغا قىلغان چىرايلىق ۋەدىسى تولۇق ئىشقا ئاشتى؛ پىرئەۋن ۋە ئۇنىڭ قەۋمىنىڭ سالغانلىرى (يەنى ئىمارەتلىرى) نى ۋە ياسىغانلىرى (يەنى باغلىرى ۋە ئېكىنزارلىقلىرى) نى ۋەيران قىلدۇق
وَجَاوَزْنَا بِبَنِي إِسْرَائِيلَ الْبَحْرَ فَأَتَوْا عَلَىٰ قَوْمٍ يَعْكُفُونَ عَلَىٰ أَصْنَامٍ لَّهُمْ ۚ قَالُوا يَا مُوسَى اجْعَل لَّنَا إِلَٰهًا كَمَا لَهُمْ آلِهَةٌ ۚ قَالَ إِنَّكُمْ قَوْمٌ تَجْهَلُونَ ﴿١٣٨﴾
ئىسرائىل ئەۋلادىنى بىز (قۇلزۇم) دېڭىزىدىن ئۆتكۈزۈۋەتتۇق، ئۇلار بۇتلىرىغا چوقۇنۇۋاتقان بىر قەۋمنىڭ يېنىدىن ئۆتكەندە: «ئى مۇسا! بىزگىمۇ ئۇلارنىڭ بۇتلىرىغا ئوخشاش بۇت ئورنىتىپ بەرگىن» دېدى. مۇسا: «سىلەر ئىشەنچلىك ھالدا نادان قەۋم ئىكەنسىلەر» دېدى
إِنَّ هَٰؤُلَاءِ مُتَبَّرٌ مَّا هُمْ فِيهِ وَبَاطِلٌ مَّا كَانُوا يَعْمَلُونَ ﴿١٣٩﴾
شۈبھىسىزكى، ئاشۇ كىشىلەرنىڭ باتىل دىنلىرى گۇمران بولغۇسىدۇر، ئۇلارنىڭ قىلغان ئەمەللىرى بىكاردۇر
قَالَ أَغَيْرَ اللَّهِ أَبْغِيكُمْ إِلَٰهًا وَهُوَ فَضَّلَكُمْ عَلَى الْعَالَمِينَ ﴿١٤٠﴾
مۇسا: «اﷲ سىلەرنى (زامانىڭلاردىكى) جاھان ئەھلىدىن ئارتۇق قىلغان تۇرسا، سىلەرگە ئۇنىڭدىن باشقا ئىلاھ ئىزدەمدىم؟» دېدى
وَإِذْ أَنجَيْنَاكُم مِّنْ آلِ فِرْعَوْنَ يَسُومُونَكُمْ سُوءَ الْعَذَابِ ۖ يُقَتِّلُونَ أَبْنَاءَكُمْ وَيَسْتَحْيُونَ نِسَاءَكُمْ ۚ وَفِي ذَٰلِكُم بَلَاءٌ مِّن رَّبِّكُمْ عَظِيمٌ ﴿١٤١﴾
ئۆز ۋاقتىدا سىلەرنى پىرئەۋننىڭ قەۋمىدىن قۇتقۇزدۇق، ئۇلار سىلەرگە قاتتىق ئازابنى تېتىتاتتى، ئوغۇللىرىڭلارنى ئۆلتۈرەتتى، ئاياللىرىڭلارنى (خورلاپ ئىشلىتىشكە) ئېلىپ قالاتتى، بۇنىڭدا پەرۋەردىگارىڭلاردىن سىلەرگە زور سىناق بار ئىدى
وَوَاعَدْنَا مُوسَىٰ ثَلَاثِينَ لَيْلَةً وَأَتْمَمْنَاهَا بِعَشْرٍ فَتَمَّ مِيقَاتُ رَبِّهِ أَرْبَعِينَ لَيْلَةً ۚ وَقَالَ مُوسَىٰ لِأَخِيهِ هَارُونَ اخْلُفْنِي فِي قَوْمِي وَأَصْلِحْ وَلَا تَتَّبِعْ سَبِيلَ الْمُفْسِدِينَ ﴿١٤٢﴾
مۇسا (بىزگە مۇناجات قىلىشقا) ئوتتۇز كېچىنى ۋەدە قىلدۇق. ئۇنىڭغا يەنە ئون كېچىنى قوشتۇق، شۇنىڭ بىلەن، پەرۋەردىگارىڭنىڭ ۋەدە قىلغان ۋاقتى 40 كېچە بولدى. مۇسا قېرىندىشى ھارۇنغا: «(مەن قايتىپ كەلگەنگە قەدەر) قەۋمىمگە مېنىڭ ئورۇنباسارىم بولۇپ تۇرغىن، (ئۇلارنىڭ ئىشىنى) تۈزىگىن، بۇزۇقچىلىق قىلغۇچىلارنىڭ يولىدا ماڭمىغىن» دېدى
وَلَمَّا جَاءَ مُوسَىٰ لِمِيقَاتِنَا وَكَلَّمَهُ رَبُّهُ قَالَ رَبِّ أَرِنِي أَنظُرْ إِلَيْكَ ۚ قَالَ لَن تَرَانِي وَلَٰكِنِ انظُرْ إِلَى الْجَبَلِ فَإِنِ اسْتَقَرَّ مَكَانَهُ فَسَوْفَ تَرَانِي ۚ فَلَمَّا تَجَلَّىٰ رَبُّهُ لِلْجَبَلِ جَعَلَهُ دَكًّا وَخَرَّ مُوسَىٰ صَعِقًا ۚ فَلَمَّا أَفَاقَ قَالَ سُبْحَانَكَ تُبْتُ إِلَيْكَ وَأَنَا أَوَّلُ الْمُؤْمِنِينَ ﴿١٤٣﴾
مۇسا بىز ۋەدە قىلغان ۋاقىتتا كەلگەن ۋە پەرۋەردىگارى ئۇنىڭغا (بىۋاسىتە) سۆ قىلغان چاغدا: «پەرۋەردىگارىم! ماڭا ئۆزۈڭنى كۆرسەتكىن، سېنى بىر كۆرۈپ ئالاي» دېدى. اﷲ: «مېنى (بۇ دۇنيادا) ھەرگىز كۆرەلمەيسەن (چۈنكى ئىنساننىڭ بۇ ئاجىز تېنى بۇنىڭغا تاقەت قىلالمايدۇ). لېكىن تاغقا قارىغىن، ئەگەر تاغ ئورنىدا مەھكەم تۇرالىسا، مېنى كۆرەلەيسەن» دېدى. پەرۋەردىگارى تاغقا تەجەللى قىلىش بىلەن، تاغنى تۈپتۈز قىلىۋەتتى، مۇسا بىھوش بولۇپ يىقىلدى. ئۇ ھوشىغا كېلىپ: «(پەرۋەرىگارىم!) سەن پاكتۇرسەن، ساڭا تەۋبە قىلدىم، مەن (سېنىڭ ئۇلۇغلۇقۇڭغا) ئىشەنگۈچىلەرنىڭ ئەۋۋىلىمەن» دېدى
قَالَ يَا مُوسَىٰ إِنِّي اصْطَفَيْتُكَ عَلَى النَّاسِ بِرِسَالَاتِي وَبِكَلَامِي فَخُذْ مَا آتَيْتُكَ وَكُن مِّنَ الشَّاكِرِينَ ﴿١٤٤﴾
اﷲ ئېيتتى: «ئى مۇسا! مەن ئىشەنچلىك ھالدا سېنى (زامانىڭدىكى) كىشىلەر ئارىسىدىن خۇدانىڭ ئەلچىسىلىككە ۋە مەن بىلەن (بىۋاسىتە) سۆزلىشىشكە تاللىدىم، ساڭا مەن ئاتا قىلغن خۇدانىڭ ئەلچىسىلىكنى قوبۇل قىلغىن ۋە شۈكۈر قىلغۇچىلاردىن بولغىن»
وَكَتَبْنَا لَهُ فِي الْأَلْوَاحِ مِن كُلِّ شَيْءٍ مَّوْعِظَةً وَتَفْصِيلًا لِّكُلِّ شَيْءٍ فَخُذْهَا بِقُوَّةٍ وَأْمُرْ قَوْمَكَ يَأْخُذُوا بِأَحْسَنِهَا ۚ سَأُرِيكُمْ دَارَ الْفَاسِقِينَ ﴿١٤٥﴾
بىز ئۇنىڭ ئۈچۈن (تەۋرات) تاختىلىرىغا (ئىسرائىل ئەۋلادى دىنىدا موھتاج بولغان) ۋەز - نەسىھەت، (دىنىي) ئەھكاملارنىڭ ھەممىسىنى تەپسىلىي يازدۇق. (ئى مۇسا) ئۇنى (يەنى تەۋراتنى) مەھكەم تۇتقىن ھەمدە قەۋمىڭنى ئۇنىڭدىكى ئەڭ گۈزەل ئەھكاملارغا ئەمەل قىلىشقا بۇيرۇغىن، سىلەرگە پاسىقلار (يەنى پىرئەۋن ۋە ئۇنىڭ قەۋمى) نىڭ دىيارىنى كۆرسىتىمەن
سَأَصْرِفُ عَنْ آيَاتِيَ الَّذِينَ يَتَكَبَّرُونَ فِي الْأَرْضِ بِغَيْرِ الْحَقِّ وَإِن يَرَوْا كُلَّ آيَةٍ لَّا يُؤْمِنُوا بِهَا وَإِن يَرَوْا سَبِيلَ الرُّشْدِ لَا يَتَّخِذُوهُ سَبِيلًا وَإِن يَرَوْا سَبِيلَ الْغَيِّ يَتَّخِذُوهُ سَبِيلًا ۚ ذَٰلِكَ بِأَنَّهُمْ كَذَّبُوا بِآيَاتِنَا وَكَانُوا عَنْهَا غَافِلِينَ ﴿١٤٦﴾
يەر يۈزىدە ناھەق تەكەببۇرلۇق قىلىدىغانلارغا مېنىڭ ئايەتلىرىمنى پەھىم قىلدۇرمايمەن، ئۇلار بارلىق مۆجىزىلەرنى كۆگەندىمۇ ئۇنىڭغا ئىشەنمەيدۇ، ئۇلار ئەگەر توغرا يولنى كۆرسە، ئۇنىڭدا ماڭمايدۇ، گۇمراھلىق يولىنى كۆرسە ئۇنىڭدا ماڭىدۇ، بۇ شۇنىڭ ئۈچۈنكى، ئۇلار بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلدى ۋە ئۇنىڭدىن غاپىل بولدى
وَالَّذِينَ كَذَّبُوا بِآيَاتِنَا وَلِقَاءِ الْآخِرَةِ حَبِطَتْ أَعْمَالُهُمْ ۚ هَلْ يُجْزَوْنَ إِلَّا مَا كَانُوا يَعْمَلُونَ ﴿١٤٧﴾
بىزنىڭ ئايەتلىرىمىزنى ۋە ئاخىرەتتە اﷲ قا مۇلاقات بولۇشنى ئىنكار قىلغانلارنىڭ قىلغان (ياخشى) ئەمەللىرى (ئىمانى بولمىغانلىقى ئۈچۈن) بىكار بولۇپ كېتىدۇ، ئۇلارغا پەقەت قىلغان ئەمەللىرىگە يارىشا جازا بېرىلىدۇ
وَاتَّخَذَ قَوْمُ مُوسَىٰ مِن بَعْدِهِ مِنْ حُلِيِّهِمْ عِجْلًا جَسَدًا لَّهُ خُوَارٌ ۚ أَلَمْ يَرَوْا أَنَّهُ لَا يُكَلِّمُهُمْ وَلَا يَهْدِيهِمْ سَبِيلًا ۘ اتَّخَذُوهُ وَكَانُوا ظَالِمِينَ ﴿١٤٨﴾
مۇسانىڭ قەۋمى مۇسادىن كېيىن (يەنى مۇسا پەرۋەردىگارىغا مۇناجات قىلىش ئۈچۈن تۇر تېغىغا كەتكەندىن كېيىن)، زىننەت بۇيۇملرىدىن موزايدەك ئاۋاز چىقىرىدىغان بىر جانسىز موزاي ياسىدى. ئۇلار موزاينىڭ ئۆزلىرىگە سۆزلىيەلمەيدىغانلىقىنى، يول كۆرسىتەلمەيدىغانلىقىنى كۆرمىدىمۇ؟ ئۇلار ئۇنى مەبۇد قىلىۋالدى، ئۇلار (ئۆزلىرىگە) زۇلۇم قىلغۇچى بولدى
وَلَمَّا سُقِطَ فِي أَيْدِيهِمْ وَرَأَوْا أَنَّهُمْ قَدْ ضَلُّوا قَالُوا لَئِن لَّمْ يَرْحَمْنَا رَبُّنَا وَيَغْفِرْ لَنَا لَنَكُونَنَّ مِنَ الْخَاسِرِينَ ﴿١٤٩﴾
ئۇلار (موزايغا چوقۇنغانلىقلىرىغا) پۇشايمان قىلغان ۋە راست ئازغانلىقلىرىنى چۈشەنگەن چاغدا: «ئەگەر پەرۋەردىگارىمىز بىزگە رەھىم قىلمىسا ۋە مەغپىرەت قىلمىسا، بىز ئەلۋەتتە زىيان تارتقۇچىلاردىن بولىمىز» دېدى
وَلَمَّا رَجَعَ مُوسَىٰ إِلَىٰ قَوْمِهِ غَضْبَانَ أَسِفًا قَالَ بِئْسَمَا خَلَفْتُمُونِي مِن بَعْدِي ۖ أَعَجِلْتُمْ أَمْرَ رَبِّكُمْ ۖ وَأَلْقَى الْأَلْوَاحَ وَأَخَذَ بِرَأْسِ أَخِيهِ يَجُرُّهُ إِلَيْهِ ۚ قَالَ ابْنَ أُمَّ إِنَّ الْقَوْمَ اسْتَضْعَفُونِي وَكَادُوا يَقْتُلُونَنِي فَلَا تُشْمِتْ بِيَ الْأَعْدَاءَ وَلَا تَجْعَلْنِي مَعَ الْقَوْمِ الظَّالِمِينَ ﴿١٥٠﴾
مۇسا غەزەپلەنگەن، غەمكىن ھالدا قايتىپ كېلىپ، قەۋمىگە: «مەن يوق چاغدا (موزايغا چوقۇنۇپ) نەقەدەر يامان ئىش قىلدىڭلار - ھە! (كۈتۈپ تۇرماي) پەرۋەردىگارىڭلار نىڭ ئەمرىگە ئالدىراپ كەتتىڭلارمۇ؟» دېدى ۋە (غەزەپلەنگەنلىكتىن تەۋرات) تاختىلىرىنى (يەرگە) تاشلىدى، قېرىندىشىنىڭ چېچىدىن تۇتۇپ ئۆز تەرىپىگە تارتتى. (ھارۇن) ئېيتتى: «ئى قېرىندىشىم! (بۇ) قەۋم مېنى بوزەك تاپتى، مېنى ئۆلتۈرۈۋەتكىلى تاس قالدى. مېنى دۈشمەنلەرگە تەپەررۇج قىلىپ بەرمىگىن؟ مېنى زالىم قەۋم (يەنى موزايغا چوقۇنغانلار) قاتارىدا سانىمىغىن»
قَالَ رَبِّ اغْفِرْ لِي وَلِأَخِي وَأَدْخِلْنَا فِي رَحْمَتِكَ ۖ وَأَنتَ أَرْحَمُ الرَّاحِمِينَ ﴿١٥١﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! ماڭا ۋە مېنىڭ قېرىندىشىمغا مەغپىرەت قىلغىن، بىزنى رەھمىتىڭ دائىرىسىگە كىرگۈزگىن، سەن ئەڭ رەھىم قىلغۇچى زاتتۇرسەن»
إِنَّ الَّذِينَ اتَّخَذُوا الْعِجْلَ سَيَنَالُهُمْ غَضَبٌ مِّن رَّبِّهِمْ وَذِلَّةٌ فِي الْحَيَاةِ الدُّنْيَا ۚ وَكَذَٰلِكَ نَجْزِي الْمُفْتَرِينَ ﴿١٥٢﴾
شۈبھىسىزكى موزاينى مەبۇد قىلىۋالغانلار پەرۋەردىگارىنىڭ غەزىپىگە ئۇچرايدۇ، بۇ دۇنيادا خارلىققا قالىدۇ، (اﷲ قا) بوھتان قىلغۇچىلارغا بىز مۇشۇنداق جازا بېرىمىز
وَالَّذِينَ عَمِلُوا السَّيِّئَاتِ ثُمَّ تَابُوا مِن بَعْدِهَا وَآمَنُوا إِنَّ رَبَّكَ مِن بَعْدِهَا لَغَفُورٌ رَّحِيمٌ ﴿١٥٣﴾
يامان ئىشلارنى قىلىپ قويۇپ، كېيىن تەۋبە قىلغان ۋە ئىمانىدا سەمىمىي بولغانلارنى تەۋبىسىدىن كېيىن پەرۋەردىگارىڭ ئەلۋەتتە مەغپىرەت قىلىدۇ، (ئۇلارغا) رەھىم قىلىدۇ
وَلَمَّا سَكَتَ عَن مُّوسَى الْغَضَبُ أَخَذَ الْأَلْوَاحَ ۖ وَفِي نُسْخَتِهَا هُدًى وَرَحْمَةٌ لِّلَّذِينَ هُمْ لِرَبِّهِمْ يَرْهَبُونَ ﴿١٥٤﴾
مۇسا غەزىپى بېسىلغاندىن كېيىن (يەردە ياتقان تەۋرات) تاختىلىرىنى ئالدى. ئۇنىڭدا پەرۋەردىگارىدىن قورققۇچىلارغا ھىدايەت قىلىنىدۇ ۋە رەھمەت قىلىنىدۇ دەپ يېزىلغان ئىدى
وَاخْتَارَ مُوسَىٰ قَوْمَهُ سَبْعِينَ رَجُلًا لِّمِيقَاتِنَا ۖ فَلَمَّا أَخَذَتْهُمُ الرَّجْفَةُ قَالَ رَبِّ لَوْ شِئْتَ أَهْلَكْتَهُم مِّن قَبْلُ وَإِيَّايَ ۖ أَتُهْلِكُنَا بِمَا فَعَلَ السُّفَهَاءُ مِنَّا ۖ إِنْ هِيَ إِلَّا فِتْنَتُكَ تُضِلُّ بِهَا مَن تَشَاءُ وَتَهْدِي مَن تَشَاءُ ۖ أَنتَ وَلِيُّنَا فَاغْفِرْ لَنَا وَارْحَمْنَا ۖ وَأَنتَ خَيْرُ الْغَافِرِينَ ﴿١٥٥﴾
مۇسا، بىز بەلگىلەپ بەرگەن ۋاقىتتا ئېلىپ كېلىش ئۈچۈن، ئۆز قەۋمىدىن 70 كىشىنى تاللىدى؛ ئۇلارغا زىلزىلە يۈزلەنگەندە، مۇسا: «پەرۋەردىگارىم! خالىغان بولساڭ ئۇلارنى ۋە مېنى بۇرۇنلا ھالاك قىلغان بولاتتىڭ. ئارىمىزدىكى ئەخمەقلەرنىڭ قىلمىشى تۈپەيلىدىن بىزنى ھالاك قىلامسەن؟ بۇ پەقەت سېنىڭ سىنىقىڭدۇر، شۇ ئارقىلىق خالىغان بەندەڭنى ئازدۇرىسەن، خالىغان بەندەڭنى ھىدايەت قىلىسەن، سەن بىزنىڭ ئىگىمىزسەن، بىزگە مەغپىرەت قىلغىن، رەھىم قىلغىن. سەن ئەڭ ياخشى مەغپىرەت قىلغۇچىسەن
وَاكْتُبْ لَنَا فِي هَٰذِهِ الدُّنْيَا حَسَنَةً وَفِي الْآخِرَةِ إِنَّا هُدْنَا إِلَيْكَ ۚ قَالَ عَذَابِي أُصِيبُ بِهِ مَنْ أَشَاءُ ۖ وَرَحْمَتِي وَسِعَتْ كُلَّ شَيْءٍ ۚ فَسَأَكْتُبُهَا لِلَّذِينَ يَتَّقُونَ وَيُؤْتُونَ الزَّكَاةَ وَالَّذِينَ هُم بِآيَاتِنَا يُؤْمِنُونَ ﴿١٥٦﴾
بىزگە بۇ دۇنيادا ۋە ئاخىرەتتە تېگىشلىك ياخشىلىق قىلغىن، بىز ئىشەنچلىك ھالدا ساڭا تەۋبە قىلدۇق» دېدى. اﷲ ئېيتتى: «ئازابىم بىلەن (بەندىلىرىمدىن) خالىغان كىشىنى ئازابلايمەن، مېنىڭ رەھمىتىم مەخلۇقاتنىڭ ھەممىسىگە ئورتاقتۇر. رەھمىتىمنى (كۇفرىدىن ۋە گۇناھتىن) ساقلانغۇچىلارغا، (ماللىرىنىڭ) زاكىتىنى بېرىدىغانلارغا ۋە بىزنىڭ ئايەتلىرىمىزگە ئىمان ئېيتىدىغانلارغا تېگىشلىك قىلىمەن»
الَّذِينَ يَتَّبِعُونَ الرَّسُولَ النَّبِيَّ الْأُمِّيَّ الَّذِي يَجِدُونَهُ مَكْتُوبًا عِندَهُمْ فِي التَّوْرَاةِ وَالْإِنجِيلِ يَأْمُرُهُم بِالْمَعْرُوفِ وَيَنْهَاهُمْ عَنِ الْمُنكَرِ وَيُحِلُّ لَهُمُ الطَّيِّبَاتِ وَيُحَرِّمُ عَلَيْهِمُ الْخَبَائِثَ وَيَضَعُ عَنْهُمْ إِصْرَهُمْ وَالْأَغْلَالَ الَّتِي كَانَتْ عَلَيْهِمْ ۚ فَالَّذِينَ آمَنُوا بِهِ وَعَزَّرُوهُ وَنَصَرُوهُ وَاتَّبَعُوا النُّورَ الَّذِي أُنزِلَ مَعَهُ ۙ أُولَٰئِكَ هُمُ الْمُفْلِحُونَ ﴿١٥٧﴾
ئۇلار ئەلچىگە - ئۈممى خۇدانىڭ ئەلچىسىگە (يەنى مۇھەممەد ئەلەيھىسسالامغا) ئەگىشىدۇ، ئۇلار ئۆز ئىلكىدىكى تەۋرات، ئىنجىللاردا ئۇنىڭ (سۈپىتىنىڭ) يېزىلغانلىقىنى كۆرىدۇ. ئۇ ئۇلارنى ياخشى ئىش قىلىشقا بۇيرۇيدۇ، يامان ئىش قىلىشتىن توسىدۇ، ئۇلارغا پاك نەرسىلەرنى ھالال قىلىدۇ، ناپاك نەرسىلەرنى ھارام قىلىدۇ، ئۇلارنىڭ ئېغىر يۈكىنى يېنىكلىتىدۇ، ئۇلارنى سېلىنغان تاقاق، كويزا - كىشەنلەردىن بوشىتىدۇ (يەنى ئۇلارغا يۈكلەنگەن ئېغىر ۋەزىپىلەرنى ئېلىپ تاشلايدۇ)، ئۇنىڭغا ئىمان ئېيتقانلار، ئۇنى ھۆرمەتلىگەنلەر، ئۇنىڭغا ياردەم بەرگەنلەر، ئۇنىڭغا نازىل قىلىنغان نۇر (يەنى قۇرئان) غا ئەگەشكۈچىلەر بەختكە ئېرىشكۈچىلەردۇر
قُلْ يَا أَيُّهَا النَّاسُ إِنِّي رَسُولُ اللَّهِ إِلَيْكُمْ جَمِيعًا الَّذِي لَهُ مُلْكُ السَّمَاوَاتِ وَالْأَرْضِ ۖ لَا إِلَٰهَ إِلَّا هُوَ يُحْيِي وَيُمِيتُ ۖ فَآمِنُوا بِاللَّهِ وَرَسُولِهِ النَّبِيِّ الْأُمِّيِّ الَّذِي يُؤْمِنُ بِاللَّهِ وَكَلِمَاتِهِ وَاتَّبِعُوهُ لَعَلَّكُمْ تَهْتَدُونَ ﴿١٥٨﴾
(ئى مۇھەممەد!) ئېيتقىنكى، «ئى ئىنسانلار! مەن ئىشەنچلىك ھالدا سىلەرنىڭ ھەممىڭلارغا اﷲ تەرىپىدىن ئەۋەتىلگەن ئەلچىمەن، ئاسمانلارنىڭ ۋە زېمىننىڭ پادىشاھلىقى اﷲ قا خاستۇر، ئۇنىڭدىن باشقا ھېچ ئىلاھ يوقتۇر، اﷲ تىرىلدۈرىدۇ ۋە ئۆلتۈرىدۇ، اﷲ قا ۋە اﷲ نىڭ سۆزلىرىگە ئىمان كەلتۈرىدىغان ئەلچىسى ئۈممى خۇدانىڭ ئەلچىسىگە ئىمان كەلتۈرۈڭلار، ھىدايەت تېپىشىڭلار ئۈچۈن ئۇنىڭغا ئەگىشىڭلار»
وَمِن قَوْمِ مُوسَىٰ أُمَّةٌ يَهْدُونَ بِالْحَقِّ وَبِهِ يَعْدِلُونَ ﴿١٥٩﴾
مۇسانىڭ قەۋمىنىڭ (يەنى بەنى ئىسرائىلنىڭ) ئىچىدە (كىشىلەرنى) ھەق بىلەن توغرا يولغا باشلايدىغان، ھەق بىلەن توغرا ھۆكۈم قىلىدىغان بىر جامائە بار
وَقَطَّعْنَاهُمُ اثْنَتَيْ عَشْرَةَ أَسْبَاطًا أُمَمًا ۚ وَأَوْحَيْنَا إِلَىٰ مُوسَىٰ إِذِ اسْتَسْقَاهُ قَوْمُهُ أَنِ اضْرِب بِّعَصَاكَ الْحَجَرَ ۖ فَانبَجَسَتْ مِنْهُ اثْنَتَا عَشْرَةَ عَيْنًا ۖ قَدْ عَلِمَ كُلُّ أُنَاسٍ مَّشْرَبَهُمْ ۚ وَظَلَّلْنَا عَلَيْهِمُ الْغَمَامَ وَأَنزَلْنَا عَلَيْهِمُ الْمَنَّ وَالسَّلْوَىٰ ۖ كُلُوا مِن طَيِّبَاتِ مَا رَزَقْنَاكُمْ ۚ وَمَا ظَلَمُونَا وَلَٰكِن كَانُوا أَنفُسَهُمْ يَظْلِمُونَ ﴿١٦٠﴾
ئۇلار (يەنى ئىسرائىل) نى 12 تارماققا (يەنى قەبىلىگە) جامائەلەر قىلىپ بۆلۈۋەتتۇق. مۇسانىڭ قەۋمى (باياۋاندا) ئۇنىڭدىن سۇ تەلەپ قىلغاندا، مۇساغا: «ھاساڭ بىلەن تاشنى ئۇرغىن» دەپ ۋەھىي قىلدۇق. (مۇسا ھاسىسى بىلەن تاشنى ئۇرۇۋىدى) ئۇنىڭدىن (قەبىلىلەرنىڭ سانى بويىچە) 12 بۇلاق ئېتىلىپ چىقتى، ھەر قەبىلە ئۆزىنىڭ سۇ ئىچىدىغان ئورنىنى بىلدى. ئۇلارغا بۇلۇتنى سايىۋەن قىلىپ بەردۇق، ئۇلارغا تەرەنجىبىن بىلەن بۆدۈنىنى چۈشۈرۈپ بەردۇق. (ئۇلارغا ئېيتتۇقكى): «سىلەر بىز رىزىقلاندۇرغان لەززەتلىك نەرسىلەردىن يەڭلار». (ئۇلار بۇ ئۇلۇغ نېمەتلەرگە نانكورلۇق قىلىش بىلەن) بىزگە زىيان سالغىنى يوق، (ئۆزلىرىنى اﷲ نىڭ ئازابىغا دۇچار قىلىشلىرى بىلەن) ئۆزلىرىگە ئۆزلىرى زۇلۇم سالدى

ئۆز ۋاقتىدا (بىز ئۇلارغا) (يەنى بەنى ئىسرائىلنىڭ ئاتا - بوۋىلىرىغا): «سىلەر بۇ شەھەر (يەنى بەيتۇلمۇقەددەس) دە تۇرۇڭلار، ئۇ يەردىكى يېمەكلىكلەردىن خالىغىنىڭلارچە يەڭلار، شەھەر دەرۋازىسىدىن سەجدە قىلغان ھالدا كىرىڭلار، (كىرىۋاتقان چېغىڭلاردا) ئى اﷲ! گۇناھلىرىمىزنى كەچۈرگىن، دەڭلار، جىمى گۇناھلىرىڭلارنى مەغپىرەت قىلىمىز، ياخشىلىق قىلغۇچىلارغا (ساۋابىنى) زىيادە بېرىمىز» دېدۇق
فَبَدَّلَ الَّذِينَ ظَلَمُوا مِنْهُمْ قَوْلًا غَيْرَ الَّذِي قِيلَ لَهُمْ فَأَرْسَلْنَا عَلَيْهِمْ رِجْزًا مِّنَ السَّمَاءِ بِمَا كَانُوا يَظْلِمُونَ ﴿١٦٢﴾
ئۇلارنىڭ ئىچىدىكى زالىملار ئۇلارغا ئېيتىلغان سۆز (اﷲ نىڭ ئەمرىنى) باشقا سۆزلەرگە ئۆزگەرتىۋەتتى، ئۇلار زۇلۇم قىلغانلىقلىرى ئۈچۈن، ئۇلارغا ئاسماندىن ئازاب ئەۋەتتۇق
وَاسْأَلْهُمْ عَنِ الْقَرْيَةِ الَّتِي كَانَتْ حَاضِرَةَ الْبَحْرِ إِذْ يَعْدُونَ فِي السَّبْتِ إِذْ تَأْتِيهِمْ حِيتَانُهُمْ يَوْمَ سَبْتِهِمْ شُرَّعًا وَيَوْمَ لَا يَسْبِتُونَ ۙ لَا تَأْتِيهِمْ ۚ كَذَٰلِكَ نَبْلُوهُم بِمَا كَانُوا يَفْسُقُونَ ﴿١٦٣﴾
ئۇلار (يەنى يەھۇدىيلار) دىن دېڭىز (يەنى قۇلزۇم دېڭىزى) بويىدىكى (ئەيلە) شەھرىنىڭ ئەھۋالىنى سورىغىن. (ئۇنىڭ ئاھالىسى بېلىق تۇتۇش چەكلەنگەن كۈن) شەنبىدە اﷲ نىڭ چەكلىمىسىدىن چىقاتتى. چۈنكى شەنبە كۈنى بېلىقلار سۇ ئۈستىدىن ئۇلارنىڭ ئالدىغا كېلەتتى، شەنبىدىن غەيرىي كۈنلەردە ئۇلارنىڭ ئالدىغا كەلمەيتتى، پاسىق بولغانلىقلىرى ئۈچۈن. ئۇلارنى مۇشۇنداق سىنايمىز
وَإِذْ قَالَتْ أُمَّةٌ مِّنْهُمْ لِمَ تَعِظُونَ قَوْمًا ۙ اللَّهُ مُهْلِكُهُمْ أَوْ مُعَذِّبُهُمْ عَذَابًا شَدِيدًا ۖ قَالُوا مَعْذِرَةً إِلَىٰ رَبِّكُمْ وَلَعَلَّهُمْ يَتَّقُونَ ﴿١٦٤﴾
ئۆز ۋاقتىدا ئۇلارنىڭ ئىچىدىكى بىر جامائە ئادەملەر: اﷲ ھالاك قىلىدىغان ياكى قاتتىق ئازابلايدىغان قەۋمگە نېمە ئۈچۈن ۋەز - نەسىھەت قىلىسىلەر؟» دېدى. ئۇلار (يەنى ۋەز - نەسىھەت قىلغۇچىلار): «پەرۋەردىگارىڭلارنىڭ ئالدىدا (گۇناھقا سۈكۈت قىلىپ تۇردۇق دەپ) ئۆزرە ئېيتماسلىقىمىز ۋە ئۇلارنىڭ اﷲ تىن قورقۇپ (گۇناھتىن چەكلىنىشلىرىنى ئۈمىد قىلغانلىقىمىز ئۈچۈن شۇنداق قىلدۇق)» دېدى
فَلَمَّا نَسُوا مَا ذُكِّرُوا بِهِ أَنجَيْنَا الَّذِينَ يَنْهَوْنَ عَنِ السُّوءِ وَأَخَذْنَا الَّذِينَ ظَلَمُوا بِعَذَابٍ بَئِيسٍ بِمَا كَانُوا يَفْسُقُونَ ﴿١٦٥﴾
ئۇلار ئۆزلىرىگە قىلىنغان ۋەز - نەسىھەتنى قوبۇل قىلمىغان چاغدا، يامان ئىشتىن مەنئى قىلغۇچىلارنى قۇتقۇزدۇق، اﷲ نىڭ ئەمرىدىن چىققانلىقلىرى ئۈچۈن، زالىملارنى قاتتىق ئازابقا دۇچار قىلدۇق
فَلَمَّا عَتَوْا عَن مَّا نُهُوا عَنْهُ قُلْنَا لَهُمْ كُونُوا قِرَدَةً خَاسِئِينَ ﴿١٦٦﴾
ئۇلار نەھيى قىلىنغان نەرسىنى تەرك ئېتىشتىن باش تارتقانلىقلىرى ئۈچۈن، ئۇلارغا: «خار مايمۇن بولۇپ كېتىڭلار» دېدۇق
وَإِذْ تَأَذَّنَ رَبُّكَ لَيَبْعَثَنَّ عَلَيْهِمْ إِلَىٰ يَوْمِ الْقِيَامَةِ مَن يَسُومُهُمْ سُوءَ الْعَذَابِ ۗ إِنَّ رَبَّكَ لَسَرِيعُ الْعِقَابِ ۖ وَإِنَّهُ لَغَفُورٌ رَّحِيمٌ ﴿١٦٧﴾
(ئى مۇھەممەد!) ئۆز ۋاقتىدا، پەرۋەردىگارىڭ ئۇلارغا (يەنى يەھۇدىيلارغا) قىيامەتكىچە چوقۇم قاتتىق ئازابلايدىغان كىشىلەرنى ئەۋەتىپ تۇرىدىغانلىقىنى خەۋەر قىلدى. شۈبھىسىزكى، پەرۋەردىگارىڭ (ئاسىيلىق قىلغۇچىلارنى) ئەلۋەتتە تېز جازالىغۇچىدۇر، (ئىتائەت قىلغۇچىلارغا) ئەلۋەتتە مەغپىرەت قىلغۇچىدۇر، مەرھەمەت قىلغۇچىدۇر
وَقَطَّعْنَاهُمْ فِي الْأَرْضِ أُمَمًا ۖ مِّنْهُمُ الصَّالِحُونَ وَمِنْهُمْ دُونَ ذَٰلِكَ ۖ وَبَلَوْنَاهُم بِالْحَسَنَاتِ وَالسَّيِّئَاتِ لَعَلَّهُمْ يَرْجِعُونَ ﴿١٦٨﴾
ئۇلارنى (يەنى يەھۇدىيلارنى) يەر يۈزىدە تۈرلۈك پىرقىلەرگە بۆلۈۋەتتۇق، ئۇلارنىڭ ئىچىدە ياخشىلىرىمۇ بار، ياخشىلىرىدىن تۆۋەنلىرىمۇ بار، ئۇلارنىڭ (كۇفرىدىن ۋە گۇناھتىن) قايتىشلىرى ئۈچۈن، ئۇلارنى قاتتىقچىلىق ۋە كەڭچىلىك بىلەن سىنىدۇق
فَخَلَفَ مِن بَعْدِهِمْ خَلْفٌ وَرِثُوا الْكِتَابَ يَأْخُذُونَ عَرَضَ هَٰذَا الْأَدْنَىٰ وَيَقُولُونَ سَيُغْفَرُ لَنَا وَإِن يَأْتِهِمْ عَرَضٌ مِّثْلُهُ يَأْخُذُوهُ ۚ أَلَمْ يُؤْخَذْ عَلَيْهِم مِّيثَاقُ الْكِتَابِ أَن لَّا يَقُولُوا عَلَى اللَّهِ إِلَّا الْحَقَّ وَدَرَسُوا مَا فِيهِ ۗ وَالدَّارُ الْآخِرَةُ خَيْرٌ لِّلَّذِينَ يَتَّقُونَ ۗ أَفَلَا تَعْقِلُونَ ﴿١٦٩﴾
ئۇلاردىن كېيىن، (يامان) ئەۋلادلار تەۋراتقا ۋارىسلىق قىلدى، ئۇلار بۇ دۇنيانىڭ ئەرزىمەس پۇل - مېلىنى (ھالال بىلەن ھارامنى ئايرىماي، قارىقويۇق) ئالىدۇ، ۋە بىزنى اﷲ ئەپۇ قىلىدۇ، دەيدۇ. شۇنداق پۇل - مال ئۇلارنىڭ قولىغا كەلسە، ئۇلار يەنە ئالىدۇ. مۇقەددەس كىتاب-ۋەھىيتا (يەنى تەۋراتتا) اﷲ قا (يالغان چاپلىماي) پەقەت ھەقنىلا ئېيتىشلىرى توغرۇلۇق ئۇلاردىن چىڭ ۋەدە ئېلىنمىدىمۇ؟ ھالبۇكى، ئۇلار تەۋراتتىكى نەرسىلەرنى ئوقۇغان ئىدى. ھارامنى تەرك ئېتىپ، اﷲ تىن قورقىدىغانلارغا ئاخىرەت ياخشىدۇر، بۇنى چۈشەنمەمسىلەر؟
وَالَّذِينَ يُمَسِّكُونَ بِالْكِتَابِ وَأَقَامُوا الصَّلَاةَ إِنَّا لَا نُضِيعُ أَجْرَ الْمُصْلِحِينَ ﴿١٧٠﴾
مۇقەددەس كىتاب-ۋەھىينى مەھكەم تۇتىدىغانلار، ناماز ئوقۇيدىغانلار (ساۋابقا ئېرىشىدۇ)، بىز ياخشى ئىش قىلغۇچىلارنىڭ ئەجرىنى ئىشەنچلىك ھالدا زايا قىلىۋەتمەيمىز
وَإِذْ نَتَقْنَا الْجَبَلَ فَوْقَهُمْ كَأَنَّهُ ظُلَّةٌ وَظَنُّوا أَنَّهُ وَاقِعٌ بِهِمْ خُذُوا مَا آتَيْنَاكُم بِقُوَّةٍ وَاذْكُرُوا مَا فِيهِ لَعَلَّكُمْ تَتَّقُونَ ﴿١٧١﴾
ئۆز ۋاقتىدا (تۇر) تاغنى قومۇرۇپ ئۇلارنىڭ ئۈستىگە سايىۋەندەك تىكلىدۇق، ئۇلار تاغنى ئۈستىلىرىگە چۈشۈپ كېتىدۇ دەپ ئويلىدى. (ئۇلارغا ئېيتتۇقكى) «تەقۋادارلار قاتارىدا بولۇشۇڭلار ئۈچۈن ئۆزۈڭلارغا بېرىلگەن مۇقەددەس كىتاب-ۋەھىينى (يەنى تەۋراتنى) مەھكەم تۇتۇڭلار، ئۇنىڭدىكى ئەھكاملارغا ئەمەل قىلىڭلار»
وَإِذْ أَخَذَ رَبُّكَ مِن بَنِي آدَمَ مِن ظُهُورِهِمْ ذُرِّيَّتَهُمْ وَأَشْهَدَهُمْ عَلَىٰ أَنفُسِهِمْ أَلَسْتُ بِرَبِّكُمْ ۖ قَالُوا بَلَىٰ ۛ شَهِدْنَا ۛ أَن تَقُولُوا يَوْمَ الْقِيَامَةِ إِنَّا كُنَّا عَنْ هَٰذَا غَافِلِينَ ﴿١٧٢﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭ ئادەم باللىرىنى (يەنى نەسلىنى) ئۇلارنىڭ (ئاتىلىرىنىڭ) پۇشتىدىن چىقاردى ۋە ئۇلارنى ئۆزلىرىگە گۇۋاھ قىلىپ: «مەن سىلەرنىڭ پەرۋەدىگارىڭلار ئەمەسمۇ؟» دېدى (يەنى اﷲ ئۆزىنىڭ ئۇلارنىڭ پەرۋەردىگارى ئىكەنلىكىگە ۋە بىرلىكىگە ئادەم بالىلىرىنى ئىقرار قىلدۇردى، ئۇلار ئىقرار قىلىپ بۇنى ئۆز ئۈستىگە ئالدى). ئۇلار: «ھەئە، سەن بىزنىڭ پەرۋەردىگارىمىزدۇرسەن، گۇۋاھلىق بەردۇق» دېدى. (ئۇلارنى گۇۋاھلىق بەرگۈزگەنلىكىمىز) قىيامەت كۈنى ئۇلارنىڭ: «بىز بۇنىڭدىن غەپلەتتە قالغان ئىكەنمىز» دېمەسلىكلىرى، ياكى «ئاتا - بوۋىلىرىمىز ئىلگىرى (ساڭا بۇتلارنى) شېرىك كەلتۈرگەن ئىدى، بىزنىڭ پەقەت ئۇلارنىڭ ئەۋلادى بولۇپ قالغان يېرىمىز بار (يەنى بىزنىڭ ئۆزرىمىز بار) گۇمراھلارنىڭ قىلمىشلىرى تۈپەيلىدىن بىزنى ھالاك قىلامسەن؟» دېمەسلىكلىرى ئۈچۈندۇر
أَوْ تَقُولُوا إِنَّمَا أَشْرَكَ آبَاؤُنَا مِن قَبْلُ وَكُنَّا ذُرِّيَّةً مِّن بَعْدِهِمْ ۖ أَفَتُهْلِكُنَا بِمَا فَعَلَ الْمُبْطِلُونَ ﴿١٧٣﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭ ئادەم باللىرىنى (يەنى نەسلىنى) ئۇلارنىڭ (ئاتىلىرىنىڭ) پۇشتىدىن چىقاردى ۋە ئۇلارنى ئۆزلىرىگە گۇۋاھ قىلىپ: «مەن سىلەرنىڭ پەرۋەدىگارىڭلار ئەمەسمۇ؟» دېدى (يەنى اﷲ ئۆزىنىڭ ئۇلارنىڭ پەرۋەردىگارى ئىكەنلىكىگە ۋە بىرلىكىگە ئادەم بالىلىرىنى ئىقرار قىلدۇردى، ئۇلار ئىقرار قىلىپ بۇنى ئۆز ئۈستىگە ئالدى). ئۇلار: «ھەئە، سەن بىزنىڭ پەرۋەردىگارىمىزدۇرسەن، گۇۋاھلىق بەردۇق» دېدى. (ئۇلارنى گۇۋاھلىق بەرگۈزگەنلىكىمىز) قىيامەت كۈنى ئۇلارنىڭ: «بىز بۇنىڭدىن غەپلەتتە قالغان ئىكەنمىز» دېمەسلىكلىرى، ياكى «ئاتا - بوۋىلىرىمىز ئىلگىرى (ساڭا بۇتلارنى) شېرىك كەلتۈرگەن ئىدى، بىزنىڭ پەقەت ئۇلارنىڭ ئەۋلادى بولۇپ قالغان يېرىمىز بار (يەنى بىزنىڭ ئۆزرىمىز بار) گۇمراھلارنىڭ قىلمىشلىرى تۈپەيلىدىن بىزنى ھالاك قىلامسەن؟» دېمەسلىكلىرى ئۈچۈندۇر
وَكَذَٰلِكَ نُفَصِّلُ الْآيَاتِ وَلَعَلَّهُمْ يَرْجِعُونَ ﴿١٧٤﴾
ئۇلارنىڭ (گۇمراھلىقتىن، ئاتا - بوۋىلىرىغا ئەگىشىشتىن) قايتىشلىرى ئۈچۈن، ئايەتلىرىمىزنى مۇشۇنداق تەپسىلىي بايان قىلىمىز
وَاتْلُ عَلَيْهِمْ نَبَأَ الَّذِي آتَيْنَاهُ آيَاتِنَا فَانسَلَخَ مِنْهَا فَأَتْبَعَهُ الشَّيْطَانُ فَكَانَ مِنَ الْغَاوِينَ ﴿١٧٥﴾
(ئى مۇھەممەد!) ئۇلار (يەنى يەھۇدىيلار) غا شۇنداق بىر ئادەمنىڭ قىسسىسىنى ئوقۇپ بەرگىنكى، ئۇنىڭغا ئايەتلىرىمىزنى بەردۇق (يەنى مۇقەددەس كىتاب-ۋەھىيۇللاھنىڭ بەزى ئىلىملىرىنى ئۆگەتتۇق). ئۇ ئايەتلىرىمىزدىن ئۆزىنى تارتتى، ئۇنىڭغا شەيتان (ئازدۇرغۇچى) ئاپىرىدە بولۇشى بىلەنلا گۇمراھلاردىن بولۇپ كەتتى
وَلَوْ شِئْنَا لَرَفَعْنَاهُ بِهَا وَلَٰكِنَّهُ أَخْلَدَ إِلَى الْأَرْضِ وَاتَّبَعَ هَوَاهُ ۚ فَمَثَلُهُ كَمَثَلِ الْكَلْبِ إِن تَحْمِلْ عَلَيْهِ يَلْهَثْ أَوْ تَتْرُكْهُ يَلْهَث ۚ ذَّٰلِكَ مَثَلُ الْقَوْمِ الَّذِينَ كَذَّبُوا بِآيَاتِنَا ۚ فَاقْصُصِ الْقَصَصَ لَعَلَّهُمْ يَتَفَكَّرُونَ ﴿١٧٦﴾
ئەگەر بىز خالىساق، ئۇنى (تەقۋادار ئۆلىمالار دەرىجىسىگە) كۆتۈرەتتۇق، لېكىن ئۇ دۇنياغا بېرىلىپ كەتتى، نەپسى خاھىشىغا بويسۇندى، ئۇ گويا بىر ئىتقا ئوخشايدۇكى، ئۇنى قوغلىۋەتسەڭمۇ تىلىنى چىقىرىپ ھەسىرەيدۇ، قوغلىۋەتسەڭمۇ تىلىنى چىقىرىپ ھەسىرەيدۇ، بۇ بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغان ئەنە شۇ قەۋمنىڭ مىسالىدۇر، ئويلاپ بېقىشلىرى ئۈچۈن، بۇ قىسسىنى (قەۋمىڭگە) سۆزلەپ بەرگىن
سَاءَ مَثَلًا الْقَوْمُ الَّذِينَ كَذَّبُوا بِآيَاتِنَا وَأَنفُسَهُمْ كَانُوا يَظْلِمُونَ ﴿١٧٧﴾
بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلىپ ئۆزلىرىگە زىيان سالغان قەۋمنىڭ مىسالى نېمىدېگەن يامان!
مَن يَهْدِ اللَّهُ فَهُوَ الْمُهْتَدِي ۖ وَمَن يُضْلِلْ فَأُولَٰئِكَ هُمُ الْخَاسِرُونَ ﴿١٧٨﴾
اﷲ ھىدايەت قىلغان ئادەم ھىدايەت تاپقۇچىدۇر، اﷲ گۇمران قىلغان ئادەم زىيان تارتقۇچىدۇر
وَلَقَدْ ذَرَأْنَا لِجَهَنَّمَ كَثِيرًا مِّنَ الْجِنِّ وَالْإِنسِ ۖ لَهُمْ قُلُوبٌ لَّا يَفْقَهُونَ بِهَا وَلَهُمْ أَعْيُنٌ لَّا يُبْصِرُونَ بِهَا وَلَهُمْ آذَانٌ لَّا يَسْمَعُونَ بِهَا ۚ أُولَٰئِكَ كَالْأَنْعَامِ بَلْ هُمْ أَضَلُّ ۚ أُولَٰئِكَ هُمُ الْغَافِلُونَ ﴿١٧٩﴾
شەك - شۈبھىسىزكى، جىن ۋە ئىنسانلاردىن نۇرغۇنلىرىنى دوزاخقا (يېقىلغۇ بولۇش ئۈچۈن) ياراتتۇق، ئۇلار دىللىرى بولغىنى بىلەنمۇ، ئۇ ئارقىلىق ھەقنى چۈشەنمەيدۇ. ئۇلار كۆزلىرى بولغان بىلەنمۇ، ئۇ ئارقىلىق (اﷲ نىڭ قۇدرىتىنىڭ دەلىللىرىنى) كۆرمەيدۇ. ئۇلار قۇلاقلىرى بولغىنى بىلەنمۇ، ئۇ ئارقىلىق (اﷲ نىڭ ئايەتلىرىنى ئىبرەت ئىلىپ) تىڭشىمايدۇ. ئۇلار گويا ھايۋانغا ئوخشايدۇ، ھايۋاندىنمۇ بەتتەر گۇمراھتۇر، ئەنە شۇلار غاپىلدۇر
وَلِلَّهِ الْأَسْمَاءُ الْحُسْنَىٰ فَادْعُوهُ بِهَا ۖ وَذَرُوا الَّذِينَ يُلْحِدُونَ فِي أَسْمَائِهِ ۚ سَيُجْزَوْنَ مَا كَانُوا يَعْمَلُونَ ﴿١٨٠﴾
اﷲ نىڭ گۈزەل ئىسىملىرى بار، اﷲ نى شۇ (گۈزەل ئىسىملىرى) بىلەن ئاتاڭلار، اﷲ نىڭ ئىسىملىرىنى كەلسە - كەلمەس قوللىنىدىغانلارنى تەرك ئېتىڭلار، ئۇلار (ئاخىرەتتە) قىلمىشلىرىنىڭ جازاسىنى تارتىدۇ
وَمِمَّنْ خَلَقْنَا أُمَّةٌ يَهْدُونَ بِالْحَقِّ وَبِهِ يَعْدِلُونَ ﴿١٨١﴾
بىز ياراتقان ئۈممەتلەر ئىچىدە ھەق يولغا دەۋەت قىلىدىغان ۋە ھەقتە چىڭ تۇرىدىغان بىر ئۈممەت بار
وَالَّذِينَ كَذَّبُوا بِآيَاتِنَا سَنَسْتَدْرِجُهُم مِّنْ حَيْثُ لَا يَعْلَمُونَ ﴿١٨٢﴾
بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلارنى تۇيدۇرماستىن، ئاستا - ئاستا ھالاكەتكە يېقىنلاشتۇرىمىز
وَأُمْلِي لَهُمْ ۚ إِنَّ كَيْدِي مَتِينٌ ﴿١٨٣﴾
ئۇلارغا مۆھلەت بېرىمەن مېنىڭ ئازابىم ئىشەنچلىك ھالدا قاتتىقتۇر
أَوَلَمْ يَتَفَكَّرُوا ۗ مَا بِصَاحِبِهِم مِّن جِنَّةٍ ۚ إِنْ هُوَ إِلَّا نَذِيرٌ مُّبِينٌ ﴿١٨٤﴾
ئۇلار ئۆزىنىڭ ھەمراھى (يەنى مۇھەممەد ئەلەيھىسسالام) نىڭ مەجنۇن ئەمەسلىكىنى ئويلىمامدۇ؟ ئۇ پەقەت ئاشكارا ئاگاھلاندۇرغۇچىدۇر
أَوَلَمْ يَنظُرُوا فِي مَلَكُوتِ السَّمَاوَاتِ وَالْأَرْضِ وَمَا خَلَقَ اللَّهُ مِن شَيْءٍ وَأَنْ عَسَىٰ أَن يَكُونَ قَدِ اقْتَرَبَ أَجَلُهُمْ ۖ فَبِأَيِّ حَدِيثٍ بَعْدَهُ يُؤْمِنُونَ ﴿١٨٥﴾
ئۇلار ئاسمانلارنىڭ ۋە زېمىننىڭ سەلتەنىتى ئۈستىدە، اﷲ نىڭ ياراتقان مەخلۇقاتلىرى، ئۆزلىرىنىڭ ئەجىلى يېقىنلىشىدىغانلىقىنىڭ مۇمكىنلىكى ئۈستىدە پىكىر يۈرگۈزمەمدۇ؟ ئۇلار (كالامۇللاھ ئىكەنلىكى شۇنچە روشەن بولغان) قۇرئاندىن باشقا يەنە قايسى سۆزگە ئىشىنىدۇ؟
مَن يُضْلِلِ اللَّهُ فَلَا هَادِيَ لَهُ ۚ وَيَذَرُهُمْ فِي طُغْيَانِهِمْ يَعْمَهُونَ ﴿١٨٦﴾
اﷲ ئازدۇرغان كىشىنى ھىدايەت قىلغۇچى بولمايدۇ، اﷲ ئۇلارنى گۇمراھلىقلىرىدا قويۇپ بېرىدۇكى، ئۇلار تېڭىرقاپ يۈرىدۇ
يَسْأَلُونَكَ عَنِ السَّاعَةِ أَيَّانَ مُرْسَاهَا ۖ قُلْ إِنَّمَا عِلْمُهَا عِندَ رَبِّي ۖ لَا يُجَلِّيهَا لِوَقْتِهَا إِلَّا هُوَ ۚ ثَقُلَتْ فِي السَّمَاوَاتِ وَالْأَرْضِ ۚ لَا تَأْتِيكُمْ إِلَّا بَغْتَةً ۗ يَسْأَلُونَكَ كَأَنَّكَ حَفِيٌّ عَنْهَا ۖ قُلْ إِنَّمَا عِلْمُهَا عِندَ اللَّهِ وَلَٰكِنَّ أَكْثَرَ النَّاسِ لَا يَعْلَمُونَ ﴿١٨٧﴾
(ئى مۇھەممەد!) ئۇلار سەندىن قىيامەتنىڭ قاچان بولىدىغانلىقىنى سورايدۇ، ئېيتقىنكى، «ئۇنىڭ قاچان بولىدىغانلىقىنى پەقەت پەرۋەردىگارىم (اﷲ) بىلىدۇ، ئۇنى پەقەت ئۆزى بەلگىلىگەن ۋاقىتتا مەيدانغا چىقىرىدۇ، قىيامەت ۋە زېمىننىڭ (ئەھلىگە) ئېغىر (ۋەقەدۇر). ئۇ سىلەرگە تۇيۇقسىز كېلىدۇ». گويا سەن قىيامەت بىلەن ناھايىتى تونۇشتەك، ئۇنى سەندىن سورىشىدۇ. ئېيتقىنكى، «ئۇنىڭ ۋاقتىنى پەقەت اﷲ بىلىدۇ، لېكىن كىشىلەرنىڭ تولىسى ئۇنىڭ ۋاقتىنىڭ مەخپىي بولۇشىدىكى سەۋەبنى) بىلمەيدۇ
قُل لَّا أَمْلِكُ لِنَفْسِي نَفْعًا وَلَا ضَرًّا إِلَّا مَا شَاءَ اللَّهُ ۚ وَلَوْ كُنتُ أَعْلَمُ الْغَيْبَ لَاسْتَكْثَرْتُ مِنَ الْخَيْرِ وَمَا مَسَّنِيَ السُّوءُ ۚ إِنْ أَنَا إِلَّا نَذِيرٌ وَبَشِيرٌ لِّقَوْمٍ يُؤْمِنُونَ ﴿١٨٨﴾
اﷲ خالىمىغان ئىكەن، ئۆزۈمگە پايدا يەتكۈزۈش، زىياننى ئۆزۈمدىن دەپئى قىلىش قولۇمدىن كەلمەيدۇ، (قىيامەتنىڭ قاچان بولۇشىنى قانداقمۇ بىلەي؟) ئەگەر مەن غەيبنى بىلىدىغان بولسام (دۇنيا مەنپەئەتلىرىدىن) نۇرغۇن مەنپەئەت ھاسىل قىلغان بولاتتىم، زىيان - زەخمەتكىمۇ ئۇچرىمىغان بولاتتىم (لېكىن مەن غەيبنى بىلمەيمەن، شۇڭا ماڭا تەقدىر قىلىنغان ياخشى - يامانلىق يېتىپ تۇرىدۇ). مەن پەقەت ئىمان ئېيتىدىغان قەۋم ئۈچۈن ئاگاھلاندۇرغۇچى ۋە خۇش خەۋەر بەرگۈچى قىلىپ ئەۋەتىلگەن خۇدانىڭ ئەلچىسىمەن»
هُوَ الَّذِي خَلَقَكُم مِّن نَّفْسٍ وَاحِدَةٍ وَجَعَلَ مِنْهَا زَوْجَهَا لِيَسْكُنَ إِلَيْهَا ۖ فَلَمَّا تَغَشَّاهَا حَمَلَتْ حَمْلًا خَفِيفًا فَمَرَّتْ بِهِ ۖ فَلَمَّا أَثْقَلَت دَّعَوَا اللَّهَ رَبَّهُمَا لَئِنْ آتَيْتَنَا صَالِحًا لَّنَكُونَنَّ مِنَ الشَّاكِرِينَ ﴿١٨٩﴾
ئۇ سىلەرنى بىر جاندىن (يەنى ئادەم ئەلەيھىسسالامدىن) ياراتتى، (ئادەم ئەلەيھىسسالامنىڭ) ئۇنسى - ئۈلپەت ئېلىشى ئۈچۈن، خوتۇنىنى (يەنى ھەۋۋانى) ئۇنىڭ ئۆز جىنسىدىن ياراتتى، خوتۇنى بىلەن يېقىنچىلىق قىلغاندىن كېيىن، خوتۇنى يېنىككىنە قورساق كۆتۈرۈپ ئېغىر بوي يۈردى، ئۇ (بالىنىڭ قورساقتا ئۆسۈپ چوڭىيىشى بىلەن) ئېغىرلاشقاندىن كېيىن، (ئەر - خوتۇن) ئىككىيلەن پەرۋەردىگارى اﷲ قا: «ئەگەر بىزگە (ئەزالىرى) بېجىرىم پەرزەنت ئاتا قىلساڭ، نېمىتىڭگە ئەلۋەتتە شۈكۈر قىلغۇچىلاردىن بولىمىز» دەپ دۇئا قىلدى
فَلَمَّا آتَاهُمَا صَالِحًا جَعَلَا لَهُ شُرَكَاءَ فِيمَا آتَاهُمَا ۚ فَتَعَالَى اللَّهُ عَمَّا يُشْرِكُونَ ﴿١٩٠﴾
اﷲ ئۇلارغا بېجىرىم ئوغۇل پەرزەنت ئاتا قىلغان ئىدى، ئۇلارنىڭ ئەۋلادى اﷲ قا شېرىك كەلتۈرۈپ (بۇتلارغا چوقۇندى)، اﷲ ئۇلارنىڭ شېرىك كەلتۈرگەن مەبۇدلىرىدىن پاكتۇر
أَيُشْرِكُونَ مَا لَا يَخْلُقُ شَيْئًا وَهُمْ يُخْلَقُونَ ﴿١٩١﴾
ئۆزلىرى يارىتىلغان، ھېچ نەرسىنى يارىتىلمايدىغان نەرسىلەرنى (يەنى بۇتلارنى) اﷲ قا شېرىك كەلتۈرەمدۇ!
وَلَا يَسْتَطِيعُونَ لَهُمْ نَصْرًا وَلَا أَنفُسَهُمْ يَنصُرُونَ ﴿١٩٢﴾
ئۇ نەرسىلەر چوقۇنغۇچىلىرىغا ياردەم بېرەلمەيدۇ، ئۆزلىرىگىمۇ ياردەم بېرەلمەيدۇ
وَإِن تَدْعُوهُمْ إِلَى الْهُدَىٰ لَا يَتَّبِعُوكُمْ ۚ سَوَاءٌ عَلَيْكُمْ أَدَعَوْتُمُوهُمْ أَمْ أَنتُمْ صَامِتُونَ ﴿١٩٣﴾
ئۇلارنى توغرا يولغا چاقىرساڭلار، ئۇلار سىلەرگە ئەگەشمەيدۇ. مەيلى ئۇلارنى دەۋەت قىلىڭلار، ياكى جىم تۇرۇڭلار، بۇ ئۇلار ئۈچۈن بەرىبىر ئوخشاش (يەنى سىلەر قايسى ھالەتتە بولماڭلار، ئۇلار سىلەرنىڭ دەۋىتىڭلارغا جاۋاب قايتۇرۇشقا قادىر ئەمەستۇر)
إِنَّ الَّذِينَ تَدْعُونَ مِن دُونِ اللَّهِ عِبَادٌ أَمْثَالُكُمْ ۖ فَادْعُوهُمْ فَلْيَسْتَجِيبُوا لَكُمْ إِن كُنتُمْ صَادِقِينَ ﴿١٩٤﴾
سىلەر اﷲ نى قويۇپ ئىبادەت قىلغان بۇتلار سىلەرگە ئوخشاش مەخلۇقلاردۇر، (ئۇلارنى ئىلاھ دېگەن دەۋايىڭلاردا) راستچىل بولساڭلار، ئۇلارغا دۇئا قىلىپ بېقىڭلار، دۇئايىڭلارنى ئىجابەت قىلىپ باقسۇن
أَلَهُمْ أَرْجُلٌ يَمْشُونَ بِهَا ۖ أَمْ لَهُمْ أَيْدٍ يَبْطِشُونَ بِهَا ۖ أَمْ لَهُمْ أَعْيُنٌ يُبْصِرُونَ بِهَا ۖ أَمْ لَهُمْ آذَانٌ يَسْمَعُونَ بِهَا ۗ قُلِ ادْعُوا شُرَكَاءَكُمْ ثُمَّ كِيدُونِ فَلَا تُنظِرُونِ ﴿١٩٥﴾
ئۇلار (يەنى بۇتلار) نىڭ ماڭىدىغان پۇتلىرى بارمۇ؟ ياكى تۇتىدىغان قوللىرى بارمۇ؟ ياكى كۆرىدىغان كۆزلىرى بارمۇ ۋە ئاڭلايدىغان قۇلاقلىرى بارمۇ؟ (ئى مۇھەممەد! ئۇلارغا) ئېيتقىنكى، «شېرىكلىرىڭلارنى (يەنى بۇتلىرىڭلارنى ماڭا قارشى) چاقىرىڭلار، ئاندىن (قولۇڭلاردىن كېلىشىچە) ماڭا زىيانكەشلىك قىلىڭلار، ئازراقمۇ مۆھلەت بەرمەڭلار (مەن اﷲ قا يۆلەنگەنلىكىم ئۈچۈن سىلەرگە پەرۋا قىلمايمەن)
إِنَّ وَلِيِّيَ اللَّهُ الَّذِي نَزَّلَ الْكِتَابَ ۖ وَهُوَ يَتَوَلَّى الصَّالِحِينَ ﴿١٩٦﴾
ئىشەنچلىك ھالدا مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) نازىل قىلغان اﷲ مېنىڭ ئىگەمدۇر، ئۇ ياخشىلارغا ئىگىدارچىلىق قىلىدۇ
وَالَّذِينَ تَدْعُونَ مِن دُونِهِ لَا يَسْتَطِيعُونَ نَصْرَكُمْ وَلَا أَنفُسَهُمْ يَنصُرُونَ ﴿١٩٧﴾
سىلەرنىڭ اﷲ نى قويۇپ ئىبادەت قىلىدىغان بۇتلىرىڭلار سىلەرگە ياردەم بېرەلمەيدۇ، ئۆزلىرىگىمۇ ياردەم بېرەلمەيدۇ
وَإِن تَدْعُوهُمْ إِلَى الْهُدَىٰ لَا يَسْمَعُوا ۖ وَتَرَاهُمْ يَنظُرُونَ إِلَيْكَ وَهُمْ لَا يُبْصِرُونَ ﴿١٩٨﴾
ئەگەر ئۇلارنى ھىدايەتكە دەۋەت قىلساڭلار، ئۇلار (ھىدايەت تاپماق تۈگۈل دەۋىتىڭنى) ئاڭلىمايدۇ، ئۇلار ساڭا قاراپ تۇرغاندەك كۆرۈنسىمۇ، لېكىن ئۇلار كۆرمەيدۇ
خُذِ الْعَفْوَ وَأْمُرْ بِالْعُرْفِ وَأَعْرِضْ عَنِ الْجَاهِلِينَ ﴿١٩٩﴾
ئەپۇنى دوست تۇتقىن، ياخشىلىققا (يەنى ياخشى سۆز قىلىشقا، ياخشى ئىش قىلىشقا) بۇيرۇغىن، نادانلار بىلەن تەڭ بولمىغىن (يەنى نادانلارنىڭ قىلغىنىنى قىلماي، ئۇلارغا مۇلايىم بولغىن)
وَإِمَّا يَنزَغَنَّكَ مِنَ الشَّيْطَانِ نَزْغٌ فَاسْتَعِذْ بِاللَّهِ ۚ إِنَّهُ سَمِيعٌ عَلِيمٌ ﴿٢٠٠﴾
ئەگەر شەيتان (ئازدۇرغۇچى) ساڭا ۋەسۋەسە قىلسا، (شەيتان (ئازدۇرغۇچى)نىڭ ۋەسۋەسىسىدىن) اﷲ قا سېغىنىپ پاناھ تىلىگىن. اﷲ ئىشەنچلىك ھالدا (سۆزۈڭنى) ئاڭلاپ تۇرغۇچىدۇر (ئىشىڭنى بىلىپ تۇرغۇچىدۇ
إِنَّ الَّذِينَ اتَّقَوْا إِذَا مَسَّهُمْ طَائِفٌ مِّنَ الشَّيْطَانِ تَذَكَّرُوا فَإِذَا هُم مُّبْصِرُونَ ﴿٢٠١﴾
تەقۋادار كىشىلەر شەيتان (ئازدۇرغۇچى)نىڭ ۋەسۋەسىسىگە ئۇچرىسا، اﷲ نى ئەسلەيدۇ - دە، ھەقىقەتنى كۆرۈۋېلىپ، (شەيتان (ئازدۇرغۇچى)نىڭ ۋەسۋەسىسىدىن) خالاس تاپىدۇ
وَإِخْوَانُهُمْ يَمُدُّونَهُمْ فِي الْغَيِّ ثُمَّ لَا يُقْصِرُونَ ﴿٢٠٢﴾
شەيتان (ئازدۇرغۇچى)لار (كۇففار) بۇرادەرلىرىنى بولۇشىچە ئازدۇرىدۇ، ئۇلارنى ئازدۇرۇشتىن بوشىشىپ قالمايدۇ،
وَإِذَا لَمْ تَأْتِهِم بِآيَةٍ قَالُوا لَوْلَا اجْتَبَيْتَهَا ۚ قُلْ إِنَّمَا أَتَّبِعُ مَا يُوحَىٰ إِلَيَّ مِن رَّبِّي ۚ هَٰذَا بَصَائِرُ مِن رَّبِّكُمْ وَهُدًى وَرَحْمَةٌ لِّقَوْمٍ يُؤْمِنُونَ ﴿٢٠٣﴾
(ئۇلار تەلەپ) قىلغان مۆجىزىلەردىن) بىرەر مۆجىزىنى كەلتۈرمىسەڭ، ئۇنى نېمىشقا (ئۆزۈڭ) ئىجاد قىلمىدىڭ؟ دېيىشىدۇ، (ئى مۇھەممەد!) ئۇلارغا ئېيتقىنكى، «(ئىش مېنىڭ ئىلكىمدە ئەمەس) مەن پەقەت پەرۋەردىگارىم تەرىپىدىن ماڭا ۋەھيى قىلىنغانغىلا ئەمەل قىلىمەن (ئۆزلىكىمدىن مۆجىزە يارىتالمايمەن)، بۇ (قۇرئاندا سىلەرگە) پەرۋەردىگارىڭلار تەرىپىدىن كەلتۈرۈلگەن دەلىللەردۇر (قۇرئان باشقا مۆجىزىلەرگە ئېھتىياج قالدۇرمايدۇ)، ئىمان ئېيتقان قەۋم ئۈچۈن ھىدايەتتۇر ۋە رەھمەتتۇر»
وَإِذَا قُرِئَ الْقُرْآنُ فَاسْتَمِعُوا لَهُ وَأَنصِتُوا لَعَلَّكُمْ تُرْحَمُونَ ﴿٢٠٤﴾
اﷲ نىڭ رەھمىتىگە ئېرىشىشىڭلار ئۈچۈن، قۇرئان ئوقۇلغان چاغدا، ئۇنى دىققەت بىلەن ئاڭلاڭلار ۋە جىم تۇرۇڭلار (يەنى سۆز قىلماڭلار)
وَاذْكُر رَّبَّكَ فِي نَفْسِكَ تَضَرُّعًا وَخِيفَةً وَدُونَ الْجَهْرِ مِنَ الْقَوْلِ بِالْغُدُوِّ وَالْآصَالِ وَلَا تَكُن مِّنَ الْغَافِلِينَ ﴿٢٠٥﴾
پەرۋەردىگارىڭنى يېلىنغان ۋە ئۇنىڭدىن قورققان ھالدا ئىچىڭدە ياد ئەتكىن، ئەتىگەندە - ئاخشامدا ئۇنى پەس ئاۋازدا زىكرى قىلغىن، غاپىللاردىن بولمىغىن
إِنَّ الَّذِينَ عِندَ رَبِّكَ لَا يَسْتَكْبِرُونَ عَنْ عِبَادَتِهِ وَيُسَبِّحُونَهُ وَلَهُ يَسْجُدُونَ ﴿٢٠٦﴾
پەرۋەردىگارىڭنىڭ دەرگاھىدىكىلەر (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى) اﷲ قا ئىبادەت قىلىشتىن باش تارتمايدۇ، اﷲ قا تەسبىھ ئېيتىدۇ ۋە ئۇنىڭغا سەجدە قىلىدۇ
