حامىم
تَنزِيلُ ٱلْكِتَـٰبِ مِنَ ٱللَّهِ ٱلْعَزِيزِ ٱلْعَلِيمِ ﴿٢﴾
(بۇ) مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) غالىب، ھەممىنى بىلگۈچى اﷲ تەرىپىدىن نازىل قىلىنغاندۇر
غَافِرِ ٱلذَّنۢبِ وَقَابِلِ ٱلتَّوْبِ شَدِيدِ ٱلْعِقَابِ ذِى ٱلطَّوْلِ ۖ لَآ إِلَـٰهَ إِلَّا هُوَ ۖ إِلَيْهِ ٱلْمَصِيرُ ﴿٣﴾
(اﷲ) گۇناھنى مەغپىرەت قىلغۇچىدۇر، تەۋبىنى قوبۇل قىلغۇچىدۇر. (اﷲ نىڭ) ئازابى قاتتىقتۇر، (اﷲ) ئىنئام ئىگىسىدۇر، ئۇنىڭدىن باشقا ھېچ ئىلاھ يوقتۇر، ئاقىۋەت قايتىدىغان جاي ئۇنىڭ دەرگاھىدۇر
مَا يُجَـٰدِلُ فِىٓ ءَايَـٰتِ ٱللَّهِ إِلَّا ٱلَّذِينَ كَفَرُوا۟ فَلَا يَغْرُرْكَ تَقَلُّبُهُمْ فِى ٱلْبِلَـٰدِ ﴿٤﴾
اﷲ نىڭ ئايەتلىرى (يەنى قۇرئان) توغرىسىدا پەقەت ھەقىقەتنى رەت قىلغۇچىلار مۇجادىلە قىلىشىدۇ، ئۇلارنىڭ شەھەرلەردە بېرىپ - كېلىپ يۈرۈشلىرىگە مەغرۇر بولۇپ كەتمىگىن
كَذَّبَتْ قَبْلَهُمْ قَوْمُ نُوحٍ وَٱلْأَحْزَابُ مِنۢ بَعْدِهِمْ ۖ وَهَمَّتْ كُلُّ أُمَّةٍۭ بِرَسُولِهِمْ لِيَأْخُذُوهُ ۖ وَجَـٰدَلُوا۟ بِٱلْبَـٰطِلِ لِيُدْحِضُوا۟ بِهِ ٱلْحَقَّ فَأَخَذْتُهُمْ ۖ فَكَيْفَ كَانَ عِقَابِ ﴿٥﴾
ئۇلار (يەنى مەككە ھەقىقەتنى رەت قىلغۇچىلىرى) دىن بۇرۇن، نۇھنىڭ قەۋمى ۋە ئۇلاردىن كېيىنكى (خۇدانىڭ ئەلچىسىلىرىگە قارشى ئۇيۇشقان ئاد ۋە سەمۇدقا ئوخشاش) جامائەلار (خۇدانىڭ ئەلچىسىلىرىنى) ئىنكار قىلدى. (خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلغۇچى ئۈممەتلەردىن) ھەر ئۈممەت ئۆزلىرىنىڭ پەيغەمبىرىنى ھالاك قىلىشنى قەستلىدى. ھەقنى بىكار قىلىش ئۈچۈن باتىل (سۆزلەر ئارقىلىق خۇدانىڭ ئەلچىسىلىرى بىلەن) مۇنازىرە قىلىشتى، شۇنىڭ بىلەن ئۇلارنى ھالاك قىلدىم، مېنىڭ ئازابىم قانداق ئىكەن؟
وَكَذَٰلِكَ حَقَّتْ كَلِمَتُ رَبِّكَ عَلَى ٱلَّذِينَ كَفَرُوٓا۟ أَنَّهُمْ أَصْحَـٰبُ ٱلنَّارِ ﴿٦﴾
شۇنىڭدەك (يەنى ئىلگىرىكى ئۈممەتلەرگە تېگىشلىك بولغاندەك) پەرۋەردىگارىڭنىڭ ئازابى ھەقىقەتنى رەت قىلغۇچىلارغا تېگىشلىك بولدى. ئۇلار دوزاخ ئەھلىدۇر
ٱلَّذِينَ يَحْمِلُونَ ٱلْعَرْشَ وَمَنْ حَوْلَهُۥ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ وَيُؤْمِنُونَ بِهِۦ وَيَسْتَغْفِرُونَ لِلَّذِينَ ءَامَنُوا۟ رَبَّنَا وَسِعْتَ كُلَّ شَىْءٍ رَّحْمَةً وَعِلْمًا فَٱغْفِرْ لِلَّذِينَ تَابُوا۟ وَٱتَّبَعُوا۟ سَبِيلَكَ وَقِهِمْ عَذَابَ ٱلْجَحِيمِ ﴿٧﴾
ئەرشنى كۆتۈرۈپ تۇرغان ۋە ئەرشنىڭ چۆرىسىدىكى خۇدانىڭ ئەلچى پەرىشتىلىرى پەرۋەردىگارىغا تەسبىھ ئېيتىدۇ، ھەمدى ئېيتىدۇ، ئۇنىڭغا ئىمان ئېيتىدۇ، مۆمىنلەرگە مەغپىرەت تىلەيدۇ. (ئۇلار ئېيتىدۇ) «پەرۋەردىگارىمىز! سېنىڭ رەھمىتىڭ ۋە ئىلمىڭ ھەممە نەرسىنى ئۆز ئىچىگە ئالىدۇ، تەۋبە قىلغانلارغا ۋە سېنىڭ يولۇڭغا ئەگەشكەنلەرگە مەغپىرەت قىلغىن، ئۇلارنى دوزاخ ئازابىدىن ساقلىغىن
رَبَّنَا وَأَدْخِلْهُمْ جَنَّـٰتِ عَدْنٍ ٱلَّتِى وَعَدتَّهُمْ وَمَن صَلَحَ مِنْ ءَابَآئِهِمْ وَأَزْوَٰجِهِمْ وَذُرِّيَّـٰتِهِمْ ۚ إِنَّكَ أَنتَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٨﴾
پەرۋەردىگارىمىز! سەن ئۇلارنى، ئۇلارنىڭ ئاتا - بوۋىلىرىدىن، ئاياللىرىدىن، ئەۋلادلىرىدىن ياخشى بولغانلارنى ئۇلارغا ۋەدە قىلغان مەڭگۈلۈك مەڭگۈلۈك باغلاركە كىرگۈزگىن، سەن ئىشەنچلىك ھالدا غالىبسەن، ھېكمەت بىلەن ئىش قىلغۇچىسەن
وَقِهِمُ ٱلسَّيِّـَٔاتِ ۚ وَمَن تَقِ ٱلسَّيِّـَٔاتِ يَوْمَئِذٍ فَقَدْ رَحِمْتَهُۥ ۚ وَذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٩﴾
پەرۋەردىگارىمىز! سەن ئۇلارنى يامان ئىشلاردىن ساقلىغىن، سەن كىمنى يامان ئىشلاردىن ساقلايدىكەنسەن، شۈبھىسىزكى، بۇ كۈندە سەن ئۇنىڭغا رەھمەت قىلغان بولىسەن، بۇ چوڭ بەختتۇر»
إِنَّ ٱلَّذِينَ كَفَرُوا۟ يُنَادَوْنَ لَمَقْتُ ٱللَّهِ أَكْبَرُ مِن مَّقْتِكُمْ أَنفُسَكُمْ إِذْ تُدْعَوْنَ إِلَى ٱلْإِيمَـٰنِ فَتَكْفُرُونَ ﴿١٠﴾
شۈبھىسىزكى، ھەقىقەتنى رەت قىلغۇچىلارغا: «اﷲ نىڭ (دۇنيادا) سىلەرگە بولغان غەزىپى سىلەرنىڭ (ئاخىرەتتە) بىر - بىرىڭلارغا بولغان غەزىپىڭلاردىن ئېشىپ چۈشىدۇ، چۈنكى سىلەر ئىمانغا دەۋەت قىلىناتتىڭلار، لېكىن ئىمان ئېيتمايتتىڭلار» دەپ نىدا قىلىنىدۇ
قَالُوا۟ رَبَّنَآ أَمَتَّنَا ٱثْنَتَيْنِ وَأَحْيَيْتَنَا ٱثْنَتَيْنِ فَٱعْتَرَفْنَا بِذُنُوبِنَا فَهَلْ إِلَىٰ خُرُوجٍ مِّن سَبِيلٍ ﴿١١﴾
ئۇلار: «پەرۋەردىگارىمىز! بىزنى ئىككى قېتىم ئۆلتۈرۈپ، ئىككى قېتىم تىرىلدۈردۈڭ، بىز گۇناھىمىزغا ئىقرار قىلدۇق، ئەمدى چىقىش يولى تېپىلارمۇ؟» دەيدۇ
ذَٰلِكُم بِأَنَّهُۥٓ إِذَا دُعِىَ ٱللَّهُ وَحْدَهُۥ كَفَرْتُمْ ۖ وَإِن يُشْرَكْ بِهِۦ تُؤْمِنُوا۟ ۚ فَٱلْحُكْمُ لِلَّهِ ٱلْعَلِىِّ ٱلْكَبِيرِ ﴿١٢﴾
بۇ شۇنىڭ ئۈچۈندۇركى، سىلەر اﷲ نى بىر دەپ بىلىشكە چاقىرىلغاندا (بۇنى) ئىنكار قىلدىڭلار، ئەگەر اﷲ قا شېرىك كەلتۈرۈلسە (يەنى لات، ئۇززا قاتارلىق بۇتلارغا چوقۇنۇشقا چاقىرىلساڭلار) ئۇنى تەستىق قىلىسىلەر (يەنى ئۇلارنىڭ ئىلاھلىقىنى ئېتىراپ قىلىسىلەر)، ھۆكۈم يۈكسەك، بۈيۈك اﷲ قا خاستۇر
هُوَ ٱلَّذِى يُرِيكُمْ ءَايَـٰتِهِۦ وَيُنَزِّلُ لَكُم مِّنَ ٱلسَّمَآءِ رِزْقًا ۚ وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ ﴿١٣﴾
اﷲ سىلەرگە (قۇدرىتىنىڭ) ئالامەتلىرىنى كۆرسىتىدۇ، سىلەرگە كۆكتىن يامغۇر بىلەن رىزىق چۈشۈرۈپ بېرىدۇ، پەقەت اﷲ نىڭ تائىتىگە قايتقان كىشىلەرلا (اﷲ نىڭ ئايەتلىرىدىن) ۋەز - نەسىھەت ئالىدۇ
فَٱدْعُوا۟ ٱللَّهَ مُخْلِصِينَ لَهُ ٱلدِّينَ وَلَوْ كَرِهَ ٱلْكَـٰفِرُونَ ﴿١٤﴾
دىنىڭلارنى (شېرىكتىن ۋە رىيادىن) ساپ قىلغان ھالدا اﷲ قا ئىبادەت قىلىڭلار، ھەقىقەتنى رەت قىلغۇچىلار، ياقتۇرمىغان تەقدىردىمۇ
رَفِيعُ ٱلدَّرَجَـٰتِ ذُو ٱلْعَرْشِ يُلْقِى ٱلرُّوحَ مِنْ أَمْرِهِۦ عَلَىٰ مَن يَشَآءُ مِنْ عِبَادِهِۦ لِيُنذِرَ يَوْمَ ٱلتَّلَاقِ ﴿١٥﴾
اﷲ نىڭ مەرتىۋىسى يۇقىرىدۇر، (اﷲ) ئەرشنىڭ ئىگىسىدۇر، (خالايىق) مۇلاقات بولۇشىدىغان كۈن (يەنى قىيامەت كۈنى) دىن ئاگاھلاندۇرۇش ئۈچۈن، اﷲ ئۆز ھۆكمى بويىچە بەندىلىرىدىن خالىغان ئادەمگە ۋەھيى چۈشۈرىدۇ
يَوْمَ هُم بَـٰرِزُونَ ۖ لَا يَخْفَىٰ عَلَى ٱللَّهِ مِنْهُمْ شَىْءٌ ۚ لِّمَنِ ٱلْمُلْكُ ٱلْيَوْمَ ۖ لِلَّهِ ٱلْوَٰحِدِ ٱلْقَهَّارِ ﴿١٦﴾
ئۇلار قەبرىلەردىن چىققان كۈندە اﷲ قا ئۇلاردىن (يەنى ئۇلارنىڭ ئەھۋالىدىن، ئەمەللىرىدىن، سىرلىرىدىن) ھېچ نەرسە مەخپىي قالمايدۇ، بۈگۈن پادىشاھلىق كىمگە خاس؟ غالىب بىر اﷲ قا خاستۇر
ٱلْيَوْمَ تُجْزَىٰ كُلُّ نَفْسٍۭ بِمَا كَسَبَتْ ۚ لَا ظُلْمَ ٱلْيَوْمَ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ﴿١٧﴾
بۈگۈن ھەر ئادەمگە قىلغان ئەمەلىگە يارىشا جازا بېرىلىدۇ، بۈگۈن زۇلۇم يوقتۇر، اﷲ ئىشەنچلىك ھالدا تېز ھېساب ئالغۇچىدۇر
وَأَنذِرْهُمْ يَوْمَ ٱلْـَٔازِفَةِ إِذِ ٱلْقُلُوبُ لَدَى ٱلْحَنَاجِرِ كَـٰظِمِينَ ۚ مَا لِلظَّـٰلِمِينَ مِنْ حَمِيمٍ وَلَا شَفِيعٍ يُطَاعُ ﴿١٨﴾
ئۇلارنى قىيامەت كۈنىدىن ئاگاھلاندۇرغىن، ئۇ چاغدا دىللار (قورقۇنچنىڭ قاتتىقلىقىدىن) بوغۇزلارغا قاپلىشىۋالىدۇ، ئۇلار قايغۇ - ھەسرەتكە تولىدۇ، زالىملارغا يېقىن دوست ۋە شاپائىتى قوبۇل قىلىنىدىغان شاپائەتچى بولمايدۇ
يَعْلَمُ خَآئِنَةَ ٱلْأَعْيُنِ وَمَا تُخْفِى ٱلصُّدُورُ ﴿١٩﴾
اﷲ كۆزلەرنىڭ خىيانىتىنى ۋە دىللاردىكى يوشۇرۇن نەرسىلەرنى بىلىپ تۇرىدۇ
وَٱللَّهُ يَقْضِى بِٱلْحَقِّ ۖ وَٱلَّذِينَ يَدْعُونَ مِن دُونِهِۦ لَا يَقْضُونَ بِشَىْءٍ ۗ إِنَّ ٱللَّهَ هُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ﴿٢٠﴾
اﷲ ئادىللىق بىلەن ھۆكۈم چىقىرىدۇ، ئۇلار اﷲ نى قويۇپ ئىبادەت قىلىۋاتقان بۇتلار ھېچ نەرسىگە ھۆكۈم چىقىرىشقا قادىر ئەمەس، اﷲ ئىشەنچلىك ھالدا (بەندىلەرنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (ئۇلارنىڭ ئىشلىرىنى) كۆرۈپ تۇرغۇچىدۇر
۞ أَوَلَمْ يَسِيرُوا۟ فِى ٱلْأَرْضِ فَيَنظُرُوا۟ كَيْفَ كَانَ عَـٰقِبَةُ ٱلَّذِينَ كَانُوا۟ مِن قَبْلِهِمْ ۚ كَانُوا۟ هُمْ أَشَدَّ مِنْهُمْ قُوَّةً وَءَاثَارًا فِى ٱلْأَرْضِ فَأَخَذَهُمُ ٱللَّهُ بِذُنُوبِهِمْ وَمَا كَانَ لَهُم مِّنَ ٱللَّهِ مِن وَاقٍ ﴿٢١﴾
ئۇلار زېمىندا سەير قىلىپ، ئۇلاردىن ئىلگىرى ئۆتكەنلەرنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۈزەتمىدىمۇ؟ ئۇلار بولسا زېمىندا كۈچ - قۇۋۋەت ۋە يادىكارلىقلار (يەنى ئۆي - ئىمارەت قۇرۇش) جەھەتتە بۇلار (يەنى سېنىڭ قەۋمىڭنىڭ ھەقىقەتنى رەت قىلغۇچىلىرى) دىن ئۈستۈن ئىدى، گۇناھى تۈپەيلىدىن اﷲ ئۇلارنى جازالىدى، ئۇلارنى ھېچ ئەھەدى اﷲ نىڭ ئازابىدىن ساقلاپ قالالمىدى
ذَٰلِكَ بِأَنَّهُمْ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَـٰتِ فَكَفَرُوا۟ فَأَخَذَهُمُ ٱللَّهُ ۚ إِنَّهُۥ قَوِىٌّ شَدِيدُ ٱلْعِقَابِ ﴿٢٢﴾
بۇ شۇنىڭ ئۈچۈندۇركى، ئۇلارغا خۇدانىڭ ئەلچىسىلىرى روشەن مۆجىزىلەرنى ئېلىپ كەلگەن ئىدى، ئۇلار ئىشەنمىدى، شۇنىڭ بىلەن اﷲ ئۇلارنى جازالىدى. اﷲ ئىشەنچلىك ھالدا كۈچلۈكتۇر، (اﷲ نىڭ) ئازابى قاتتىقتۇر
وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔايَـٰتِنَا وَسُلْطَـٰنٍ مُّبِينٍ ﴿٢٣﴾

إِلَىٰ فِرْعَوْنَ وَهَـٰمَـٰنَ وَقَـٰرُونَ فَقَالُوا۟ سَـٰحِرٌ كَذَّابٌ ﴿٢٤﴾
بىز مۇسانى مۆجىزىلىرىمىز بىلەن، روشەن دەلىل بىلەن ئىشەنچلىك ھالدا پىرئەۋنگە، ھامانغا ۋە قارۇنغا ئەۋەتتۇق، ئۇلار (مۇسانى) يالغانچى، سېھىرگەر دېيىشتى
فَلَمَّا جَآءَهُم بِٱلْحَقِّ مِنْ عِندِنَا قَالُوا۟ ٱقْتُلُوٓا۟ أَبْنَآءَ ٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ وَٱسْتَحْيُوا۟ نِسَآءَهُمْ ۚ وَمَا كَيْدُ ٱلْكَـٰفِرِينَ إِلَّا فِى ضَلَـٰلٍ ﴿٢٥﴾
مۇسا ئۇلارغا دەرگاھىمىزدىن ھەق (يەنى خۇدانىڭ ئەلچىسىلىك) بىلەن كەلگەندە، ئۇلار: «مۇسا بىلەن ئىمان ئېيتقانلارنىڭ ئوغۇللىرىنى (نەسلىنى قۇرۇتۇش ئۈچۈن) ئۆلتۈرۈڭلار، ئاياللىرىنى (خىزمەتكە سېلىش ئۈچۈن) قالدۇرۇڭلار» دېدى. ھەقىقەتنى رەت قىلغۇچىلارنىڭ ھىيلە - مىكرى پەقەت بەربات بولىدۇ
وَقَالَ فِرْعَوْنُ ذَرُونِىٓ أَقْتُلْ مُوسَىٰ وَلْيَدْعُ رَبَّهُۥٓ ۖ إِنِّىٓ أَخَافُ أَن يُبَدِّلَ دِينَكُمْ أَوْ أَن يُظْهِرَ فِى ٱلْأَرْضِ ٱلْفَسَادَ ﴿٢٦﴾
پىرئەۋن ئېيتتى: «مېنى قويۇۋېتىڭلار، مۇسانى ئۆلتۈرىمەن، مۇسا (ئۆزىنى مەندىن قۇتۇلدۇرۇش ئۈچۈن) پەرۋەردىگارىنى چاقىرسۇن، مەن ئىشەنچلىك ھالدا مۇسانىڭ دىنىڭلارنى ئۆزگەرتىۋېتىشىدىن ياكى زېمىندا (يەنى دۆلەتتە) قالايمىقانچىلىق تۇغدۇرۇشىدىن قورقىمەن»
وَقَالَ مُوسَىٰٓ إِنِّى عُذْتُ بِرَبِّى وَرَبِّكُم مِّن كُلِّ مُتَكَبِّرٍ لَّا يُؤْمِنُ بِيَوْمِ ٱلْحِسَابِ ﴿٢٧﴾
مۇسا ئېيتتى: «مەن ئىشەنچلىك ھالدا مېنىڭ پەرۋەردىگارىم ۋە سىلەرنىڭ پەرۋەردىگارىڭلار (بولغان اﷲ) غا سېغىنىپ، ھېساب كۈنى (يەنى قىيامەت كۈنى) گە ئىشەنمەيدىغان ھەر بىر مۇتەكەببىر (نىڭ زىيانكەشلىك قىلىشى) دىن پاناھ تىلەيمەن»
وَقَالَ رَجُلٌ مُّؤْمِنٌ مِّنْ ءَالِ فِرْعَوْنَ يَكْتُمُ إِيمَـٰنَهُۥٓ أَتَقْتُلُونَ رَجُلًا أَن يَقُولَ رَبِّىَ ٱللَّهُ وَقَدْ جَآءَكُم بِٱلْبَيِّنَـٰتِ مِن رَّبِّكُمْ ۖ وَإِن يَكُ كَـٰذِبًا فَعَلَيْهِ كَذِبُهُۥ ۖ وَإِن يَكُ صَادِقًا يُصِبْكُم بَعْضُ ٱلَّذِى يَعِدُكُمْ ۖ إِنَّ ٱللَّهَ لَا يَهْدِى مَنْ هُوَ مُسْرِفٌ كَذَّابٌ ﴿٢٨﴾
پىرئەۋن خانىدانىغا مەنسۇپ، ئىمانىنى يوشۇرىدىغان بىر مۆمىن ئادەم ئېيتتى: «سىلەر 'پەرۋەردىگارىم اﷲ' دېگەن كىشىنى ئۆلتۈرەمسىلەر؟ ھالبۇكى، ئۇ سىلەرگە ئىشەنچلىك ھالدا پەرۋەردىگارىڭلارنىڭ دەرگاھىدىن روشەن دەلىللەرنى ئېلىپ كەلدى. ئەگەر ئۇ (خۇدانىڭ ئەلچىسىلىك دەۋاسىدا) يالغانچى بولسا، يالغانچىلىقنىڭ زىيىنى ئۇنىڭ ئۆزىگىدۇر، ئەگەر ئۇ راستچىل بولسا، ئۇ سىلەرنى قورقۇتقان ئازابنىڭ بىر قىسمى سىلەرگە كېلىدۇ، اﷲ ئىشەنچلىك ھالدا ھەددىدىن ئاشقان يالغانچىنى ھىدايەت قىلمايدۇ
يَـٰقَوْمِ لَكُمُ ٱلْمُلْكُ ٱلْيَوْمَ ظَـٰهِرِينَ فِى ٱلْأَرْضِ فَمَن يَنصُرُنَا مِنۢ بَأْسِ ٱللَّهِ إِن جَآءَنَا ۚ قَالَ فِرْعَوْنُ مَآ أُرِيكُمْ إِلَّا مَآ أَرَىٰ وَمَآ أَهْدِيكُمْ إِلَّا سَبِيلَ ٱلرَّشَادِ ﴿٢٩﴾
ئى قەۋمىم! بۈگۈن سەلتەنەت سىلەرنىڭ ئىلگىڭلاردىدۇر، (مىسىر) زېمىنىدا سىلەر غالىبسىلەر، ئەگەر بىزگە اﷲ نىڭ ئازابى كەلسە، ئۇنىڭدىن قۇتۇلۇشقا بىزگە كىم ياردەم بېرىدۇ؟» پىرئەۋن ئېيتتى: «مەن سىلەرگە پەقەت نېمىنى توغرا دەپ قارىسام، شۇنى مەسلىھەت بېرىمەن. سىلەرنى مەن پەقەت توغرا يولغا باشلايمەن»
وَقَالَ ٱلَّذِىٓ ءَامَنَ يَـٰقَوْمِ إِنِّىٓ أَخَافُ عَلَيْكُم مِّثْلَ يَوْمِ ٱلْأَحْزَابِ ﴿٣٠﴾
مۆمىن ئادەم ئېيتتى: «ئى قەۋمىم! سىلەرنىڭ نۇھ، ئاد، سەمۇد قەۋملىرىگە ئوخشاش ۋە ئۇلاردىن كېيىنكىلەرگە ئوخشاش جامائەلەر ئۇچرىغان جازاغا ئۇچرىشىڭلاردىن قورقىمەن، اﷲ بەندىلىرىگە زۇلۇم قىلىشنى خالىمايدۇ
مِثْلَ دَأْبِ قَوْمِ نُوحٍ وَعَادٍ وَثَمُودَ وَٱلَّذِينَ مِنۢ بَعْدِهِمْ ۚ وَمَا ٱللَّهُ يُرِيدُ ظُلْمًا لِّلْعِبَادِ ﴿٣١﴾
مۆمىن ئادەم ئېيتتى: «ئى قەۋمىم! سىلەرنىڭ نۇھ، ئاد، سەمۇد قەۋملىرىگە ئوخشاش ۋە ئۇلاردىن كېيىنكىلەرگە ئوخشاش جامائەلەر ئۇچرىغان جازاغا ئۇچرىشىڭلاردىن قورقىمەن، اﷲ بەندىلىرىگە زۇلۇم قىلىشنى خالىمايدۇ
وَيَـٰقَوْمِ إِنِّىٓ أَخَافُ عَلَيْكُمْ يَوْمَ ٱلتَّنَادِ ﴿٣٢﴾
ئى قەۋمىم! مەن سىلەر ئۈچۈن ئىشەنچلىك ھالدا قىيامەت كۈنىدىن ئەنسىرەيمەن
يَوْمَ تُوَلُّونَ مُدْبِرِينَ مَا لَكُم مِّنَ ٱللَّهِ مِنْ عَاصِمٍ ۗ وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِنْ هَادٍ ﴿٣٣﴾
ئۇ كۈندە (دوزاخ ئازابىنىڭ دەھشىتىدىن قورقۇپ) ئارقاڭلارغا چېكىنىسىلەر، سىلەرگە ھېچ اﷲ نىڭ ئازابىدىن قۇتۇلدۇرغۇچى بولمايدۇ. كىمكى اﷲ ئۇنى گۇمراھ قىلىدىكەن، ئۇنى ھىدايەت قىلغۇچى بولمايدۇ
وَلَقَدْ جَآءَكُمْ يُوسُفُ مِن قَبْلُ بِٱلْبَيِّنَـٰتِ فَمَا زِلْتُمْ فِى شَكٍّ مِّمَّا جَآءَكُم بِهِۦ ۖ حَتَّىٰٓ إِذَا هَلَكَ قُلْتُمْ لَن يَبْعَثَ ٱللَّهُ مِنۢ بَعْدِهِۦ رَسُولًا ۚ كَذَٰلِكَ يُضِلُّ ٱللَّهُ مَنْ هُوَ مُسْرِفٌ مُّرْتَابٌ ﴿٣٤﴾

اﷲ نىڭ ئايەتلىرى ئۈستىدە ئۆزلىرىگە (اﷲ تەرىپىدىن) كەلگەن ھېچقانداق دەلىل بولماستىن (قارىسىغا) جاڭجاللىشىدىغانلار اﷲ نىڭ دەرگاھىدا ۋە مۆمىنلەرنىڭ نەزىرىدە قاتتىق نەپرەتكە قالىدۇ، بۇزغۇنچىلىق قىلىپ (ئىماندىن باش تارتقان)، (بەندىلەرگە) زومىگەرلىك قىلغان ھەر قانداق ئادەمنىڭ دىلىنى اﷲ مۇشۇنداق پېچەتلىۋېتىدۇ

اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغانلار (ھىدايەتتىن ۋە ھەقىقەتتىن) شۇنداق بۇرۇلۇپ كېتىدۇ (ئى مۇھەممەد! قەۋمىڭنىڭ اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغانلىقىغا قايغۇرۇپ كەتمىگىن، ئۇلاردىن ئىلگىرى ئۆتكەنلەر مۇشۇنداق قىلغان)
ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَرْضَ قَرَارًا وَٱلسَّمَآءَ بِنَآءً وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ وَرَزَقَكُم مِّنَ ٱلطَّيِّبَـٰتِ ۚ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ ۖ فَتَبَارَكَ ٱللَّهُ رَبُّ ٱلْعَـٰلَمِينَ ﴿٦٤﴾
اﷲ سىلەرگە زېمىننى تۇرالغۇ، ئاسماننى بىنا قىلدى. سىلەرنى سۈرەتكە كىرگۈزدى. سۈرىتىڭلارنى چىرايلىق قىلدى. سىلەرگە (تۈرلۈك) پاك نەرسىلەرنى رىزىق قىلىپ بەردى. ئەنە شۇ اﷲ سىلەرنىڭ پەرۋەردىگارىڭلاردۇر، ئالەملەرنىڭ پەرۋەردىگارى اﷲ نىڭ بەرىكىتى بۈيۈكتۇر
هُوَ ٱلْحَىُّ لَآ إِلَـٰهَ إِلَّا هُوَ فَٱدْعُوهُ مُخْلِصِينَ لَهُ ٱلدِّينَ ۗ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٦٥﴾
اﷲ ھاياتتۇر، ئۇنىڭدىن باشقا ھېچ ئىلاھ يوقتۇر، دىنىڭلارنى (شېرىكتىن ۋە رىيادىن) ساپ قىلغان ھالدا ئۇنىڭغا ئىبادەت قىلىڭلار، جىمى ھەمدۇسانا ئالەملەرنىڭ پەرۋەردىگارى اﷲ قا خاستۇر!
۞ قُلْ إِنِّى نُهِيتُ أَنْ أَعْبُدَ ٱلَّذِينَ تَدْعُونَ مِن دُونِ ٱللَّهِ لَمَّا جَآءَنِىَ ٱلْبَيِّنَـٰتُ مِن رَّبِّى وَأُمِرْتُ أَنْ أُسْلِمَ لِرَبِّ ٱلْعَـٰلَمِينَ ﴿٦٦﴾
(ئى مۇھەممەد!) ئېيتقىنكى: «پەرۋەردىگارىمدىن ماڭا روشەن دەلىللەر كەلگەن چاغدا، سىلەر اﷲ نى قويۇپ ئىبادەت قىلىۋاتقان بۇتلىرىڭلارغا ئىبادەت قىلىشتىن مەنئى قىلىندىم، ئالەملەرنىڭ پەرۋەردىگارىغا بويسۇنۇشقا بۇيرۇلدۇم»
هُوَ ٱلَّذِى خَلَقَكُم مِّن تُرَابٍ ثُمَّ مِن نُّطْفَةٍ ثُمَّ مِنْ عَلَقَةٍ ثُمَّ يُخْرِجُكُمْ طِفْلًا ثُمَّ لِتَبْلُغُوٓا۟ أَشُدَّكُمْ ثُمَّ لِتَكُونُوا۟ شُيُوخًا ۚ وَمِنكُم مَّن يُتَوَفَّىٰ مِن قَبْلُ ۖ وَلِتَبْلُغُوٓا۟ أَجَلًا مُّسَمًّى وَلَعَلَّكُمْ تَعْقِلُونَ ﴿٦٧﴾
اﷲ سىلەرنى (يەنى ئاتاڭلار ئادەم ئەلەيھىسسالامنى) تۇپراقتىن، (ئۇنىڭ ئەۋلادىنى ئالدى بىلەن) ئابىمەنىيدىن، ئاندىن لەختە قاندىن ياراتتى، ئاندىن ئۇ سىلەرنى (ئاناڭلارنىڭ قارنىدىن) بوۋاق ھالىتىڭلاردا چىقىرىدۇ، ئاندىن (كۈچ - قۇۋۋەت ۋە ئەقىلدە كامالەت يېشى) قىران ۋاقتىڭلارغا يېتىسىلەر، ئاندىن قېرىيسىلەر– بەزىڭلار بۇرۇن ۋاپات بولۇپ كېتىسىلەر–ئاندىن سىلەر مۇئەييەن ۋاقىتقىچە ياشايسىلەر، سىلەر (اﷲ نىڭ قۇدرىتىنىڭ دەلىللىرىنى) چۈشەنگەيسىلەر
هُوَ ٱلَّذِى يُحْىِۦ وَيُمِيتُ ۖ فَإِذَا قَضَىٰٓ أَمْرًا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ﴿٦٨﴾
اﷲ تىرىلدۈرۈشكە ۋە ئۆلتۈرۈشكە قادىر، ئۇ بىرەر ئىشنى (قىلىشنى) ئىرادە قىلسا، ئۇنىڭغا «ۋۇجۇدقا كەل» دەيدۇ - دە، ئۇ ۋۇجۇدقا كېلىدۇ
أَلَمْ تَرَ إِلَى ٱلَّذِينَ يُجَـٰدِلُونَ فِىٓ ءَايَـٰتِ ٱللَّهِ أَنَّىٰ يُصْرَفُونَ ﴿٦٩﴾
اﷲ نىڭ ئايەتلىرى ئۈستىدە جاڭجاللىشىۋاتقانلارنى كۆرمىدىڭمۇ؟ ئۇلار قانداقمۇ (ھىدايەتتىن گۇمراھلىققا) بۇرۇلۇپ كېتىدۇ؟
ٱلَّذِينَ كَذَّبُوا۟ بِٱلْكِتَـٰبِ وَبِمَآ أَرْسَلْنَا بِهِۦ رُسُلَنَا ۖ فَسَوْفَ يَعْلَمُونَ ﴿٧٠﴾
ئۇلار مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) ئىنكار قىلدى، خۇدانىڭ ئەلچىسىلىرىمىزگە بىز نازىل قىلغان مۆجىزىلەرنى ئىنكار قىلدى. ئۇلار ئۇزاققا قالماي (ئىنكار قىلغانلىقلىرىنىڭ ئاقىۋىتىنى) بىلىدۇ
إِذِ ٱلْأَغْلَـٰلُ فِىٓ أَعْنَـٰقِهِمْ وَٱلسَّلَـٰسِلُ يُسْحَبُونَ ﴿٧١﴾
ئۇ چاغدا (يەنى دوزاخقا كىرگەندە) ئۇلارنىڭ بويۇنلىرىدا تاقاقلار ۋە زەنجىرلەر بولىدۇ، ئۇلار قايناقسۇغا سۆرەپ كىرىلىدۇ، ئاندىن ئوتتا كۆيدۈرۈلىدۇ
فِى ٱلْحَمِيمِ ثُمَّ فِى ٱلنَّارِ يُسْجَرُونَ ﴿٧٢﴾
ئۇ چاغدا (يەنى دوزاخقا كىرگەندە) ئۇلارنىڭ بويۇنلىرىدا تاقاقلار ۋە زەنجىرلەر بولىدۇ، ئۇلار قايناقسۇغا سۆرەپ كىرىلىدۇ، ئاندىن ئوتتا كۆيدۈرۈلىدۇ
ثُمَّ قِيلَ لَهُمْ أَيْنَ مَا كُنتُمْ تُشْرِكُونَ ﴿٧٣﴾
ئاندىن ئۇلارغا: «اﷲ نى قويۇپ شېرىك كەلتۈرگەن بۇتلىرىڭلار قەيەردە؟» دېيىلىدۇ، ئۇلار: «ئۇ بۇتلار ئۆزلىرىنى بىزدىن چەتكە ئالدى، بەلكى بىز ئىلگىرى ھېچنەرسىگە ئىبادەت قىلمايتتۇق» دەيدۇ. اﷲ ھەقىقەتنى رەت قىلغۇچىلارنى مۇشۇنداق گۇمراھ قىلىدۇ
مِن دُونِ ٱللَّهِ ۖ قَالُوا۟ ضَلُّوا۟ عَنَّا بَل لَّمْ نَكُن نَّدْعُوا۟ مِن قَبْلُ شَيْـًٔا ۚ كَذَٰلِكَ يُضِلُّ ٱللَّهُ ٱلْكَـٰفِرِينَ ﴿٧٤﴾
ئاندىن ئۇلارغا: «اﷲ نى قويۇپ شېرىك كەلتۈرگەن بۇتلىرىڭلار قەيەردە؟» دېيىلىدۇ، ئۇلار: «ئۇ بۇتلار ئۆزلىرىنى بىزدىن چەتكە ئالدى، بەلكى بىز ئىلگىرى ھېچنەرسىگە ئىبادەت قىلمايتتۇق» دەيدۇ. اﷲ ھەقىقەتنى رەت قىلغۇچىلارنى مۇشۇنداق گۇمراھ قىلىدۇ
ذَٰلِكُم بِمَا كُنتُمْ تَفْرَحُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَبِمَا كُنتُمْ تَمْرَحُونَ ﴿٧٥﴾
بۇ (ئازاب) سىلەرنىڭ زېمىندا ھەقسىز رەۋىشتە (گۇناھ قىلىپ پۇل - مالنى ھارام يوللارغا سەرپ قىلغانلىقىڭلار ئۈچۈن) خۇشال بولغانلىقىڭلار ۋە ھاكاۋۇر بولۇپ كەتكەنلىكىڭلار ئۈچۈندۇر
ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَـٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٦﴾
سىلەر دوزاخنىڭ دەرۋازىلىرىدىن كىرىپ ئۇ يەردە مەڭگۈ قېلىڭلار، مۇتەكەببۇرلارنىڭ جايى نېمىدېگەن يامان؟
فَٱصْبِرْ إِنَّ وَعْدَ ٱللَّهِ حَقٌّ ۚ فَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا يُرْجَعُونَ ﴿٧٧﴾
(ئى مۇھەممەد! قەۋمىڭنىڭ سېنى ئىنكار قىلغانلىقىغا) سەۋر قىلغىن، اﷲ نىڭ (ئۇلارنى ئازابلاش توغرىسىدىكى) ۋەدىسى ئىشەنچلىك ھالدا ھەقتۇر، ساڭا بىز ئۇلارغا ۋەدە قىلغان ئازابنىڭ بىر قىسمىنى كۆرسەتسەك (بۇ اﷲ نىڭ ۋەدىسىگە ۋاپا قىلغانلىقىدۇر) ياكى سېنى (ئۇنى كۆرسىتىشتىن بۇرۇن) ۋاپات قىلدۇرساق، ئۇلار بىزنىڭ دەرگاھىمىزغا قايتۇرۇلىدۇ، (قىيامەتتە ئۇلار ئازابقا دۇچار بولىدۇ)
وَلَقَدْ أَرْسَلْنَا رُسُلًا مِّن قَبْلِكَ مِنْهُم مَّن قَصَصْنَا عَلَيْكَ وَمِنْهُم مَّن لَّمْ نَقْصُصْ عَلَيْكَ ۗ وَمَا كَانَ لِرَسُولٍ أَن يَأْتِىَ بِـَٔايَةٍ إِلَّا بِإِذْنِ ٱللَّهِ ۚ فَإِذَا جَآءَ أَمْرُ ٱللَّهِ قُضِىَ بِٱلْحَقِّ وَخَسِرَ هُنَالِكَ ٱلْمُبْطِلُونَ ﴿٧٨﴾
سەندىن ئىلگىرى بىز ئىشەنچلىك ھالدا نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتتۇق، ئۇلارنىڭ ئارىسىدىن ساڭا بايان قىلىپ بەرگەنلىرىمىزمۇ، بايان قىلىپ بەرمىگەنلىرىمىزمۇ بار، ھېچقانداق خۇدانىڭ ئەلچىسى اﷲ نىڭ ئىزنىسىز بىرەر مۆجىزە كەلتۈرەلمەيدۇ، اﷲ نىڭ ئەمرى (يەنى قىيامەت) كەلگەن چاغدا (كىشىلەرنىڭ ئارىسىدا) ئادالەت بىلەن ھۆكۈم چىقىرىلىدۇ، بۇ چاغدا ھەقكە قارشى تۇرغۇچىلار زىيان تارتىدۇ
ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَنْعَـٰمَ لِتَرْكَبُوا۟ مِنْهَا وَمِنْهَا تَأْكُلُونَ ﴿٧٩﴾
مىنىشىڭلار ۋە گۆشلىرىنى يېيىشىڭلار ئۈچۈن اﷲ سىلەرگە ھايۋانلارنى ياراتتى
وَلَكُمْ فِيهَا مَنَـٰفِعُ وَلِتَبْلُغُوا۟ عَلَيْهَا حَاجَةً فِى صُدُورِكُمْ وَعَلَيْهَا وَعَلَى ٱلْفُلْكِ تُحْمَلُونَ ﴿٨٠﴾
ئۇلار سىلەرگە نۇرغۇن پايدا كەلتۈرىدۇ، ئۇلارغا مىنىپ دىلىڭلارغا پۈككەن ھاجىتىڭلاردىن چىقىسىلەر، (قۇرۇقلۇقتا يۈك - تاقلىرىڭلارنى) ئۇلارغا ئارتىسىلەر، (دېڭىزدا) كېمىلەرگە قاچىلايسىلەر
وَيُرِيكُمْ ءَايَـٰتِهِۦ فَأَىَّ ءَايَـٰتِ ٱللَّهِ تُنكِرُونَ ﴿٨١﴾
اﷲ سىلەرگە (قۇدرىتىنىڭ) ئالامەتلىرىنى كۆرسىتىدۇ، سىلەر اﷲ نىڭ (قۇدرىتىنىڭ) قايسى ئالامەتلىرىنى ئىنكار قىلىسىلەر؟
أَفَلَمْ يَسِيرُوا۟ فِى ٱلْأَرْضِ فَيَنظُرُوا۟ كَيْفَ كَانَ عَـٰقِبَةُ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَانُوٓا۟ أَكْثَرَ مِنْهُمْ وَأَشَدَّ قُوَّةً وَءَاثَارًا فِى ٱلْأَرْضِ فَمَآ أَغْنَىٰ عَنْهُم مَّا كَانُوا۟ يَكْسِبُونَ ﴿٨٢﴾
ئۇلار زېمىندا سەير قىلىپ ئۇلاردىن ئىلگىرىكىلەرنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۆزەتمىدىمۇ؟ بۇلاردىن ئۇلارنىڭ سانى كۆپ ئىدى، زېمىندا كۈچ - قۇۋۋەت ۋە يادىكارلىقلار (يەنى ئۆي - ئىمارەت) جەھەتتە بۇلاردىن ئۈستۈن ئىدى. ئۇلارنىڭ قولغا كەلتۈرگەن نەرسىلىرى ئۇلارغا ئەسقاتمىدى
فَلَمَّا جَآءَتْهُمْ رُسُلُهُم بِٱلْبَيِّنَـٰتِ فَرِحُوا۟ بِمَا عِندَهُم مِّنَ ٱلْعِلْمِ وَحَاقَ بِهِم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿٨٣﴾
ئۇلارغا خۇدانىڭ ئەلچىسىلىرى روشەن مۆجىزىلەر بىلەن كەلگەندە، ئۆزلىرىدە بار ئىلىم بىلەن پەخىرلىنىپ (خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلدى ۋە مەسخىرە قىلدى)، ئۇلارغا ئۆزلىرى مەسخىرە قىلغان (ۋە كەلسە دەپ ئالدىرىغان) ئازاب نازىل بولدى
فَلَمَّا رَأَوْا۟ بَأْسَنَا قَالُوٓا۟ ءَامَنَّا بِٱللَّهِ وَحْدَهُۥ وَكَفَرْنَا بِمَا كُنَّا بِهِۦ مُشْرِكِينَ ﴿٨٤﴾
ئۇلار بىزنىڭ ئازابىمىزنى كۆرگەن چاغدا: «بىز بىر اﷲ قا ئىمان ئېيتتۇق، بىز بۇرۇن (اﷲ قا) شېرىك قىلغان بۇتلارنى ئىنكار قىلدۇق» دېدى
فَلَمْ يَكُ يَنفَعُهُمْ إِيمَـٰنُهُمْ لَمَّا رَأَوْا۟ بَأْسَنَا ۖ سُنَّتَ ٱللَّهِ ٱلَّتِى قَدْ خَلَتْ فِى عِبَادِهِۦ ۖ وَخَسِرَ هُنَالِكَ ٱلْكَـٰفِرُونَ ﴿٨٥﴾
بىزنىڭ ئازابىمىزنى كۆرگەن چاغدا ئېيتقان ئىماننىڭ ئۇلارغا پايدىسى بولمىدى، بۇ (يەنى ئازابنى كۆرگەندە ئېيتقان ئىماننىڭ پايدىلىق بولماسلىقى) اﷲ نىڭ بەندىلىرى ئارىسىدا تۇتقان يولىدۇر، بۇ چاغدا ھەقىقەتنى رەت قىلغۇچىلار زىيان تارتقۇچى بولدى
