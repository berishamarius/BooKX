حامىم
تَنزِيلُ الْكِتَابِ مِنَ اللَّهِ الْعَزِيزِ الْعَلِيمِ ﴿٢﴾
(بۇ) مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) غالىب، ھەممىنى بىلگۈچى اﷲ تەرىپىدىن نازىل قىلىنغاندۇر
غَافِرِ الذَّنبِ وَقَابِلِ التَّوْبِ شَدِيدِ الْعِقَابِ ذِي الطَّوْلِ ۖ لَا إِلَٰهَ إِلَّا هُوَ ۖ إِلَيْهِ الْمَصِيرُ ﴿٣﴾
(اﷲ) گۇناھنى مەغپىرەت قىلغۇچىدۇر، تەۋبىنى قوبۇل قىلغۇچىدۇر. (اﷲ نىڭ) ئازابى قاتتىقتۇر، (اﷲ) ئىنئام ئىگىسىدۇر، ئۇنىڭدىن باشقا ھېچ ئىلاھ يوقتۇر، ئاقىۋەت قايتىدىغان جاي ئۇنىڭ دەرگاھىدۇر
مَا يُجَادِلُ فِي آيَاتِ اللَّهِ إِلَّا الَّذِينَ كَفَرُوا فَلَا يَغْرُرْكَ تَقَلُّبُهُمْ فِي الْبِلَادِ ﴿٤﴾
اﷲ نىڭ ئايەتلىرى (يەنى قۇرئان) توغرىسىدا پەقەت ھەقىقەتنى رەت قىلغۇچىلار مۇجادىلە قىلىشىدۇ، ئۇلارنىڭ شەھەرلەردە بېرىپ - كېلىپ يۈرۈشلىرىگە مەغرۇر بولۇپ كەتمىگىن
كَذَّبَتْ قَبْلَهُمْ قَوْمُ نُوحٍ وَالْأَحْزَابُ مِن بَعْدِهِمْ ۖ وَهَمَّتْ كُلُّ أُمَّةٍ بِرَسُولِهِمْ لِيَأْخُذُوهُ ۖ وَجَادَلُوا بِالْبَاطِلِ لِيُدْحِضُوا بِهِ الْحَقَّ فَأَخَذْتُهُمْ ۖ فَكَيْفَ كَانَ عِقَابِ ﴿٥﴾
ئۇلار (يەنى مەككە ھەقىقەتنى رەت قىلغۇچىلىرى) دىن بۇرۇن، نۇھنىڭ قەۋمى ۋە ئۇلاردىن كېيىنكى (خۇدانىڭ ئەلچىسىلىرىگە قارشى ئۇيۇشقان ئاد ۋە سەمۇدقا ئوخشاش) جامائەلار (خۇدانىڭ ئەلچىسىلىرىنى) ئىنكار قىلدى. (خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلغۇچى ئۈممەتلەردىن) ھەر ئۈممەت ئۆزلىرىنىڭ پەيغەمبىرىنى ھالاك قىلىشنى قەستلىدى. ھەقنى بىكار قىلىش ئۈچۈن باتىل (سۆزلەر ئارقىلىق خۇدانىڭ ئەلچىسىلىرى بىلەن) مۇنازىرە قىلىشتى، شۇنىڭ بىلەن ئۇلارنى ھالاك قىلدىم، مېنىڭ ئازابىم قانداق ئىكەن؟
وَكَذَٰلِكَ حَقَّتْ كَلِمَتُ رَبِّكَ عَلَى الَّذِينَ كَفَرُوا أَنَّهُمْ أَصْحَابُ النَّارِ ﴿٦﴾
شۇنىڭدەك (يەنى ئىلگىرىكى ئۈممەتلەرگە تېگىشلىك بولغاندەك) پەرۋەردىگارىڭنىڭ ئازابى ھەقىقەتنى رەت قىلغۇچىلارغا تېگىشلىك بولدى. ئۇلار دوزاخ ئەھلىدۇر
الَّذِينَ يَحْمِلُونَ الْعَرْشَ وَمَنْ حَوْلَهُ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ وَيُؤْمِنُونَ بِهِ وَيَسْتَغْفِرُونَ لِلَّذِينَ آمَنُوا رَبَّنَا وَسِعْتَ كُلَّ شَيْءٍ رَّحْمَةً وَعِلْمًا فَاغْفِرْ لِلَّذِينَ تَابُوا وَاتَّبَعُوا سَبِيلَكَ وَقِهِمْ عَذَابَ الْجَحِيمِ ﴿٧﴾
ئەرشنى كۆتۈرۈپ تۇرغان ۋە ئەرشنىڭ چۆرىسىدىكى خۇدانىڭ ئەلچى پەرىشتىلىرى پەرۋەردىگارىغا تەسبىھ ئېيتىدۇ، ھەمدى ئېيتىدۇ، ئۇنىڭغا ئىمان ئېيتىدۇ، مۆمىنلەرگە مەغپىرەت تىلەيدۇ. (ئۇلار ئېيتىدۇ) «پەرۋەردىگارىمىز! سېنىڭ رەھمىتىڭ ۋە ئىلمىڭ ھەممە نەرسىنى ئۆز ئىچىگە ئالىدۇ، تەۋبە قىلغانلارغا ۋە سېنىڭ يولۇڭغا ئەگەشكەنلەرگە مەغپىرەت قىلغىن، ئۇلارنى دوزاخ ئازابىدىن ساقلىغىن
رَبَّنَا وَأَدْخِلْهُمْ جَنَّاتِ عَدْنٍ الَّتِي وَعَدتَّهُمْ وَمَن صَلَحَ مِنْ آبَائِهِمْ وَأَزْوَاجِهِمْ وَذُرِّيَّاتِهِمْ ۚ إِنَّكَ أَنتَ الْعَزِيزُ الْحَكِيمُ ﴿٨﴾
پەرۋەردىگارىمىز! سەن ئۇلارنى، ئۇلارنىڭ ئاتا - بوۋىلىرىدىن، ئاياللىرىدىن، ئەۋلادلىرىدىن ياخشى بولغانلارنى ئۇلارغا ۋەدە قىلغان مەڭگۈلۈك مەڭگۈلۈك باغلاركە كىرگۈزگىن، سەن ئىشەنچلىك ھالدا غالىبسەن، ھېكمەت بىلەن ئىش قىلغۇچىسەن
وَقِهِمُ السَّيِّئَاتِ ۚ وَمَن تَقِ السَّيِّئَاتِ يَوْمَئِذٍ فَقَدْ رَحِمْتَهُ ۚ وَذَٰلِكَ هُوَ الْفَوْزُ الْعَظِيمُ ﴿٩﴾
پەرۋەردىگارىمىز! سەن ئۇلارنى يامان ئىشلاردىن ساقلىغىن، سەن كىمنى يامان ئىشلاردىن ساقلايدىكەنسەن، شۈبھىسىزكى، بۇ كۈندە سەن ئۇنىڭغا رەھمەت قىلغان بولىسەن، بۇ چوڭ بەختتۇر»
إِنَّ الَّذِينَ كَفَرُوا يُنَادَوْنَ لَمَقْتُ اللَّهِ أَكْبَرُ مِن مَّقْتِكُمْ أَنفُسَكُمْ إِذْ تُدْعَوْنَ إِلَى الْإِيمَانِ فَتَكْفُرُونَ ﴿١٠﴾
شۈبھىسىزكى، ھەقىقەتنى رەت قىلغۇچىلارغا: «اﷲ نىڭ (دۇنيادا) سىلەرگە بولغان غەزىپى سىلەرنىڭ (ئاخىرەتتە) بىر - بىرىڭلارغا بولغان غەزىپىڭلاردىن ئېشىپ چۈشىدۇ، چۈنكى سىلەر ئىمانغا دەۋەت قىلىناتتىڭلار، لېكىن ئىمان ئېيتمايتتىڭلار» دەپ نىدا قىلىنىدۇ
قَالُوا رَبَّنَا أَمَتَّنَا اثْنَتَيْنِ وَأَحْيَيْتَنَا اثْنَتَيْنِ فَاعْتَرَفْنَا بِذُنُوبِنَا فَهَلْ إِلَىٰ خُرُوجٍ مِّن سَبِيلٍ ﴿١١﴾
ئۇلار: «پەرۋەردىگارىمىز! بىزنى ئىككى قېتىم ئۆلتۈرۈپ، ئىككى قېتىم تىرىلدۈردۈڭ، بىز گۇناھىمىزغا ئىقرار قىلدۇق، ئەمدى چىقىش يولى تېپىلارمۇ؟» دەيدۇ
ذَٰلِكُم بِأَنَّهُ إِذَا دُعِيَ اللَّهُ وَحْدَهُ كَفَرْتُمْ ۖ وَإِن يُشْرَكْ بِهِ تُؤْمِنُوا ۚ فَالْحُكْمُ لِلَّهِ الْعَلِيِّ الْكَبِيرِ ﴿١٢﴾
بۇ شۇنىڭ ئۈچۈندۇركى، سىلەر اﷲ نى بىر دەپ بىلىشكە چاقىرىلغاندا (بۇنى) ئىنكار قىلدىڭلار، ئەگەر اﷲ قا شېرىك كەلتۈرۈلسە (يەنى لات، ئۇززا قاتارلىق بۇتلارغا چوقۇنۇشقا چاقىرىلساڭلار) ئۇنى تەستىق قىلىسىلەر (يەنى ئۇلارنىڭ ئىلاھلىقىنى ئېتىراپ قىلىسىلەر)، ھۆكۈم يۈكسەك، بۈيۈك اﷲ قا خاستۇر
هُوَ الَّذِي يُرِيكُمْ آيَاتِهِ وَيُنَزِّلُ لَكُم مِّنَ السَّمَاءِ رِزْقًا ۚ وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ ﴿١٣﴾
اﷲ سىلەرگە (قۇدرىتىنىڭ) ئالامەتلىرىنى كۆرسىتىدۇ، سىلەرگە كۆكتىن يامغۇر بىلەن رىزىق چۈشۈرۈپ بېرىدۇ، پەقەت اﷲ نىڭ تائىتىگە قايتقان كىشىلەرلا (اﷲ نىڭ ئايەتلىرىدىن) ۋەز - نەسىھەت ئالىدۇ
فَادْعُوا اللَّهَ مُخْلِصِينَ لَهُ الدِّينَ وَلَوْ كَرِهَ الْكَافِرُونَ ﴿١٤﴾
دىنىڭلارنى (شېرىكتىن ۋە رىيادىن) ساپ قىلغان ھالدا اﷲ قا ئىبادەت قىلىڭلار، ھەقىقەتنى رەت قىلغۇچىلار، ياقتۇرمىغان تەقدىردىمۇ
رَفِيعُ الدَّرَجَاتِ ذُو الْعَرْشِ يُلْقِي الرُّوحَ مِنْ أَمْرِهِ عَلَىٰ مَن يَشَاءُ مِنْ عِبَادِهِ لِيُنذِرَ يَوْمَ التَّلَاقِ ﴿١٥﴾
اﷲ نىڭ مەرتىۋىسى يۇقىرىدۇر، (اﷲ) ئەرشنىڭ ئىگىسىدۇر، (خالايىق) مۇلاقات بولۇشىدىغان كۈن (يەنى قىيامەت كۈنى) دىن ئاگاھلاندۇرۇش ئۈچۈن، اﷲ ئۆز ھۆكمى بويىچە بەندىلىرىدىن خالىغان ئادەمگە ۋەھيى چۈشۈرىدۇ
يَوْمَ هُم بَارِزُونَ ۖ لَا يَخْفَىٰ عَلَى اللَّهِ مِنْهُمْ شَيْءٌ ۚ لِّمَنِ الْمُلْكُ الْيَوْمَ ۖ لِلَّهِ الْوَاحِدِ الْقَهَّارِ ﴿١٦﴾
ئۇلار قەبرىلەردىن چىققان كۈندە اﷲ قا ئۇلاردىن (يەنى ئۇلارنىڭ ئەھۋالىدىن، ئەمەللىرىدىن، سىرلىرىدىن) ھېچ نەرسە مەخپىي قالمايدۇ، بۈگۈن پادىشاھلىق كىمگە خاس؟ غالىب بىر اﷲ قا خاستۇر
الْيَوْمَ ��ُجْزَىٰ كُلُّ نَفْسٍ بِمَا كَسَبَتْ ۚ لَا ظُلْمَ الْيَوْمَ ۚ إِنَّ اللَّهَ سَرِيعُ الْحِسَابِ ﴿١٧﴾
بۈگۈن ھەر ئادەمگە قىلغان ئەمەلىگە يارىشا جازا بېرىلىدۇ، بۈگۈن زۇلۇم يوقتۇر، اﷲ ئىشەنچلىك ھالدا تېز ھېساب ئالغۇچىدۇر
وَأَنذِرْهُمْ يَوْمَ الْآزِفَةِ إِذِ الْقُلُوبُ لَدَى الْحَنَاجِرِ كَاظِمِينَ ۚ مَا لِلظَّالِمِينَ مِنْ حَمِيمٍ وَلَا شَفِيعٍ يُطَاعُ ﴿١٨﴾
ئۇلارنى قىيامەت كۈنىدىن ئاگاھلاندۇرغىن، ئۇ چاغدا دىللار (قورقۇنچنىڭ قاتتىقلىقىدىن) بوغۇزلارغا قاپلىشىۋالىدۇ، ئۇلار قايغۇ - ھەسرەتكە تولىدۇ، زالىملارغا يېقىن دوست ۋە شاپائىتى قوبۇل قىلىنىدىغان شاپائەتچى بولمايدۇ
يَعْلَمُ خَائِنَةَ الْأَعْيُنِ وَمَا تُخْفِي الصُّدُورُ ﴿١٩﴾
اﷲ كۆزلەرنىڭ خىيانىتىنى ۋە دىللاردىكى يوشۇرۇن نەرسىلەرنى بىلىپ تۇرىدۇ
وَاللَّهُ يَقْضِي بِالْحَقِّ ۖ وَالَّذِينَ يَدْعُونَ مِن دُونِهِ لَا يَقْضُونَ بِشَيْءٍ ۗ إِنَّ اللَّهَ هُوَ السَّمِيعُ الْبَصِيرُ ﴿٢٠﴾
اﷲ ئادىللىق بىلەن ھۆكۈم چىقىرىدۇ، ئۇلار اﷲ نى قويۇپ ئىبادەت قىلىۋاتقان بۇتلار ھېچ نەرسىگە ھۆكۈم چىقىرىشقا قادىر ئەمەس، اﷲ ئىشەنچلىك ھالدا (بەندىلەرنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (ئۇلارنىڭ ئىشلىرىنى) كۆرۈپ تۇرغۇچىدۇر
أَوَلَمْ يَسِيرُوا فِي الْأَرْضِ فَيَنظُرُوا كَيْفَ كَانَ عَاقِبَةُ الَّذِينَ كَانُوا مِن قَبْلِهِمْ ۚ كَانُوا هُمْ أَشَدَّ مِنْهُمْ قُوَّةً وَآثَارًا فِي الْأَرْضِ فَأَخَذَهُمُ اللَّهُ بِذُنُوبِهِمْ وَمَا كَانَ لَهُم مِّنَ اللَّهِ مِن وَاقٍ ﴿٢١﴾
ئۇلار زېمىندا سەير قىلىپ، ئۇلاردىن ئىلگىرى ئۆتكەنلەرنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۈزەتمىدىمۇ؟ ئۇلار بولسا زېمىندا كۈچ - قۇۋۋەت ۋە يادىكارلىقلار (يەنى ئۆي - ئىمارەت قۇرۇش) جەھەتتە بۇلار (يەنى سېنىڭ قەۋمىڭنىڭ ھەقىقەتنى رەت قىلغۇچىلىرى) دىن ئۈستۈن ئىدى، گۇناھى تۈپەيلىدىن اﷲ ئۇلارنى جازالىدى، ئۇلارنى ھېچ ئەھەدى اﷲ نىڭ ئازابىدىن ساقلاپ قالالمىدى
ذَٰلِكَ بِأَنَّهُمْ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِالْبَيِّنَاتِ فَكَفَرُوا فَأَخَذَهُمُ اللَّهُ ۚ إِنَّهُ قَوِيٌّ شَدِيدُ الْعِقَابِ ﴿٢٢﴾
بۇ شۇنىڭ ئۈچۈندۇركى، ئۇلارغا خۇدانىڭ ئەلچىسىلىرى روشەن مۆجىزىلەرنى ئېلىپ كەلگەن ئىدى، ئۇلار ئىشەنمىدى، شۇنىڭ بىلەن اﷲ ئۇلارنى جازالىدى. اﷲ ئىشەنچلىك ھالدا كۈچلۈكتۇر، (اﷲ نىڭ) ئازابى قاتتىقتۇر
وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِآيَاتِنَا وَسُلْطَانٍ مُّبِينٍ ﴿٢٣﴾
بىز مۇسانى مۆجىزىلىرىمىز بىلەن، روشەن دەلىل بىلەن ئىشەنچلىك ھالدا پىرئەۋنگە، ھامانغا ۋە قارۇنغا ئەۋەتتۇق، ئۇلار (مۇسانى) يالغانچى، سېھىرگەر دېيىشتى
إِلَىٰ فِرْعَوْنَ وَهَامَانَ وَقَارُونَ فَقَالُوا سَاحِرٌ كَذَّابٌ ﴿٢٤﴾
بىز مۇسانى مۆجىزىلىرىمىز بىلەن، روشەن دەلىل بىلەن ئىشەنچلىك ھالدا پىرئەۋنگە، ھامانغا ۋە قارۇنغا ئەۋەتتۇق، ئۇلار (مۇسانى) يالغانچى، سېھىرگەر دېيىشتى
فَلَمَّا جَاءَهُم بِالْحَقِّ مِنْ عِندِنَا قَالُوا اقْتُلُوا أَبْنَاءَ الَّذِينَ آمَنُوا مَعَهُ وَاسْتَحْيُوا نِسَاءَهُمْ ۚ وَمَا كَيْدُ الْكَافِرِينَ إِلَّا فِي ضَلَالٍ ﴿٢٥﴾
مۇسا ئۇلارغا دەرگاھىمىزدىن ھەق (يەنى خۇدانىڭ ئەلچىسىلىك) بىلەن كەلگەندە، ئۇلار: «مۇسا بىلەن ئىمان ئېيتقانلارنىڭ ئوغۇللىرىنى (نەسلىنى قۇرۇتۇش ئۈچۈن) ئۆلتۈرۈڭلار، ئاياللىرىنى (خىزمەتكە سېلىش ئۈچۈن) قالدۇرۇڭلار» دېدى. ھەقىقەتنى رەت قىلغۇچىلارنىڭ ھىيلە - مىكرى پەقەت بەربات بولىدۇ
وَقَالَ فِرْعَوْنُ ذَرُونِي أَقْتُلْ مُوسَىٰ وَلْيَدْعُ رَبَّهُ ۖ إِنِّي أَخَافُ أَن يُبَدِّلَ دِينَكُمْ أَوْ أَن يُظْهِرَ فِي الْأَرْضِ الْفَسَادَ ﴿٢٦﴾
پىرئەۋن ئېيتتى: «مېنى قويۇۋېتىڭلار، مۇسانى ئۆلتۈرىمەن، مۇسا (ئۆزىنى مەندىن قۇتۇلدۇرۇش ئۈچۈن) پەرۋەردىگارىنى چاقىرسۇن، مەن ئىشەنچلىك ھالدا مۇسانىڭ دىنىڭلارنى ئۆزگەرتىۋېتىشىدىن ياكى زېمىندا (يەنى دۆلەتتە) قالايمىقانچىلىق تۇغدۇرۇشىدىن قورقىمەن»
وَقَالَ مُوسَىٰ إِنِّي عُذْتُ بِرَبِّي وَرَبِّكُم مِّن كُلِّ مُتَكَبِّرٍ لَّا يُؤْمِنُ بِيَوْمِ الْحِسَابِ ﴿٢٧﴾
مۇسا ئېيتتى: «مەن ئىشەنچلىك ھالدا مېنىڭ پەرۋەردىگارىم ۋە سىلەرنىڭ پەرۋەردىگارىڭلار (بولغان اﷲ) غا سېغىنىپ، ھېساب كۈنى (يەنى قىيامەت كۈنى) گە ئىشەنمەيدىغان ھەر بىر مۇتەكەببىر (نىڭ زىيانكەشلىك قىلىشى) دىن پاناھ تىلەيمەن»
وَقَالَ رَجُلٌ مُّؤْمِنٌ مِّنْ آلِ فِرْعَوْنَ يَكْتُمُ إِيمَانَهُ أَتَقْتُلُونَ رَجُلًا أَن يَقُولَ رَبِّيَ اللَّهُ وَقَدْ جَاءَكُم بِالْبَيِّنَاتِ مِن رَّبِّكُمْ ۖ وَإِن يَكُ كَاذِبًا فَعَلَيْهِ كَذِبُهُ ۖ وَإِن يَكُ صَادِقًا يُصِبْكُم بَعْضُ الَّذِي يَعِدُكُمْ ۖ إِنَّ اللَّهَ لَا يَهْدِي مَنْ هُوَ مُسْرِفٌ كَذَّابٌ ﴿٢٨﴾
پىرئەۋن خانىدانىغا مەنسۇپ، ئىمانىنى يوشۇرىدىغان بىر مۆمىن ئادەم ئېيتتى: «سىلەر 'پەرۋەردىگارىم اﷲ' دېگەن كىشىنى ئۆلتۈرەمسىلەر؟ ھالبۇكى، ئۇ سىلەرگە ئىشەنچلىك ھالدا پەرۋەردىگارىڭلارنىڭ دەرگاھىدىن روشەن دەلىللەرنى ئېلىپ كەلدى. ئەگەر ئۇ (خۇدانىڭ ئەلچىسىلىك دەۋاسىدا) يالغانچى بولسا، يالغانچىلىقنىڭ زىيىنى ئۇنىڭ ئۆزىگىدۇر، ئەگەر ئۇ راستچىل بولسا، ئۇ سىلەرنى قورقۇتقان ئازابنىڭ بىر قىسمى سىلەرگە كېلىدۇ، اﷲ ئىشەنچلىك ھالدا ھەددىدىن ئاشقان يالغانچىنى ھىدايەت قىلمايدۇ
يَا قَوْمِ لَكُمُ الْمُلْكُ الْيَوْمَ ظَاهِرِينَ فِي الْأَرْضِ فَمَن يَنصُرُنَا مِن بَأْسِ اللَّهِ إِن جَاءَنَا ۚ قَالَ فِرْعَوْنُ مَا أُرِيكُمْ إِلَّا مَا أَرَىٰ وَمَا أَهْدِيكُمْ إِلَّا سَبِيلَ الرَّشَادِ ﴿٢٩﴾
ئى قەۋمىم! بۈگۈن سەلتەنەت سىلەرنىڭ ئىلگىڭلاردىدۇر، (مىسىر) زېمىنىدا سىلەر غالىبسىلەر، ئەگەر بىزگە اﷲ نىڭ ئازابى كەلسە، ئۇنىڭدىن قۇتۇلۇشقا بىزگە كىم ياردەم بېرىدۇ؟» پىرئەۋن ئېيتتى: «مەن سىلەرگە پەقەت نېمىنى توغرا دەپ قارىسام، شۇنى مەسلىھەت بېرىمەن. سىلەرنى مەن پەقەت توغرا يولغا باشلايمەن»
وَقَالَ الَّذِي آمَنَ يَا قَوْمِ إِنِّي أَخَافُ عَلَيْكُم مِّثْلَ يَوْمِ الْأَحْزَابِ ﴿٣٠﴾
مۆمىن ئادەم ئېيتتى: «ئى قەۋمىم! سىلەرنىڭ نۇھ، ئاد، سەمۇد قەۋملىرىگە ئوخشاش ۋە ئۇلاردىن كېيىنكىلەرگە ئوخشاش جامائەلەر ئۇچرىغان جازاغا ئۇچرىشىڭلاردىن قورقىمەن، اﷲ بەندىلىرىگە زۇلۇم قىلىشنى خالىمايدۇ
مِثْلَ دَأْبِ قَوْمِ نُوحٍ وَعَادٍ وَثَمُودَ وَالَّذِينَ مِن بَعْدِهِمْ ۚ وَمَا اللَّهُ يُرِيدُ ظُلْمًا لِّلْعِبَادِ ﴿٣١﴾
مۆمىن ئادەم ئېيتتى: «ئى قەۋمىم! سىلەرنىڭ نۇھ، ئاد، سەمۇد قەۋملىرىگە ئوخشاش ۋە ئۇلاردىن كېيىنكىلەرگە ئوخشاش جامائەلەر ئۇچرىغان جازاغا ئۇچرىشىڭلاردىن قورقىمەن، اﷲ بەندىلىرىگە زۇلۇم قىلىشنى خالىمايدۇ
وَيَا قَوْمِ إِنِّي أَخَافُ عَلَيْكُمْ يَوْمَ التَّنَادِ ﴿٣٢﴾
ئى قەۋمىم! مەن سىلەر ئۈچۈن ئىشەنچلىك ھالدا قىيامەت كۈنىدىن ئەنسىرەيمەن
يَوْمَ تُوَلُّونَ مُدْبِرِينَ مَا لَكُم مِّنَ اللَّهِ مِنْ عَاصِمٍ ۗ وَمَن يُضْلِلِ اللَّهُ فَمَا لَهُ مِنْ هَادٍ ﴿٣٣﴾
ئۇ كۈندە (دوزاخ ئازابىنىڭ دەھشىتىدىن قورقۇپ) ئارقاڭلارغا چېكىنىسىلەر، سىلەرگە ھېچ اﷲ نىڭ ئازابىدىن قۇتۇلدۇرغۇچى بولمايدۇ. كىمكى اﷲ ئۇنى گۇمراھ قىلىدىكەن، ئۇنى ھىدايەت قىلغۇچى بولمايدۇ
وَلَقَدْ جَاءَكُمْ يُوسُفُ مِن قَبْلُ بِالْبَيِّنَاتِ فَمَا زِلْتُمْ فِي شَكٍّ مِّمَّا جَاءَكُم بِهِ ۖ حَتَّىٰ إِذَا هَلَكَ قُلْتُمْ لَن يَبْعَثَ اللَّهُ مِن بَعْدِهِ رَسُولًا ۚ كَذَٰلِكَ يُضِلُّ اللَّهُ مَنْ هُوَ مُسْرِفٌ مُّرْتَابٌ ﴿٣٤﴾
شەك - شۈبھىسىزكى، ئىلگىرى سىلەرگە يۇسۇف (اﷲ نىڭ دەرگاھىدىن) روشەن مۆجىزىلەرنى ئېلىپ كەلدى. ئۇ ئېلىپ كەلگەن دەلىلدىن سىلەر ھامان شەكلەندىڭلار، تاكى ئۇ ۋاپات بولغاندا، (دەلىل - پاكىتسىز) ھالدا، اﷲ ئۇنىڭدىن كېيىن ھەرگىز خۇدانىڭ ئەلچىسى ئەۋەتمەيدۇ' دېدىڭلار، اﷲ (گۇناھتا) ھەددىدىن ئاشقۇچىنى، (دىندا) شەكلەنگۈچىنى مۇشۇنداق ئازدۇرىدۇ»
الَّذِينَ يُجَادِلُونَ فِي آيَاتِ اللَّهِ بِغَيْرِ سُلْطَانٍ أَتَاهُمْ ۖ كَبُرَ مَقْتًا عِندَ اللَّهِ وَعِندَ الَّذِينَ آمَنُوا ۚ كَذَٰلِكَ يَطْبَعُ اللَّهُ عَلَىٰ كُلِّ قَلْبِ مُتَكَبِّرٍ جَبَّارٍ ﴿٣٥﴾
اﷲ نىڭ ئايەتلىرى ئۈستىدە ئۆزلىرىگە (اﷲ تەرىپىدىن) كەلگەن ھېچقانداق دەلىل بولماستىن (قارىسىغا) جاڭجاللىشىدىغانلار اﷲ نىڭ دەرگاھىدا ۋە مۆمىنلەرنىڭ نەزىرىدە قاتتىق نەپرەتكە قالىدۇ، بۇزغۇنچىلىق قىلىپ (ئىماندىن باش تارتقان)، (بەندىلەرگە) زومىگەرلىك قىلغان ھەر قانداق ئادەمنىڭ دىلىنى اﷲ مۇشۇنداق پېچەتلىۋېتىدۇ
وَقَالَ فِرْعَوْنُ يَا هَامَانُ ابْنِ لِي صَرْحًا لَّعَلِّي أَبْلُغُ الْأَسْبَابَ ﴿٣٦﴾
پىرئەۋن (ۋەزىرى ھامانغا): «ئى ھامان! ماڭا بىر ئېگىز بىنا سالغىن، ئۇنىڭ بىلەن مەن دەرۋازىلارغا–ئاسماننىڭ دەرۋازىلىرىغا يېتىشىم مۇمكىن، شۇنىڭ بىلەن مەن مۇسانىڭ ئىلاھىنى كۆرەي، مەن ئىشەنچلىك ھالدا مۇسانى يالغانچى دەپ گۇمان قىلىمەن» دېدى. پىرئەۋنگە ئۇنىڭ يامان ئەمەلى شۇنداق چىرايلىق كۆرسىتىلدى، پىرئەۋن (گۇمراھلىقى تۈپەيلىدىن ھىدايەت) يولىدىن مەنئى قىلىندى، پىرئەۋننىڭ ھىيلە - مىكرى پەقەت بىكاردۇر
أَسْبَابَ السَّمَاوَاتِ فَأَطَّلِعَ إِلَىٰ إِلَٰهِ مُوسَىٰ وَإِنِّي لَأَظُنُّهُ كَاذِبًا ۚ وَكَذَٰلِكَ زُيِّنَ لِفِرْعَوْنَ سُوءُ عَمَلِهِ وَصُدَّ عَنِ السَّبِيلِ ۚ وَمَا كَيْدُ فِرْعَوْنَ إِلَّا فِي تَبَابٍ ﴿٣٧﴾
پىرئەۋن (ۋەزىرى ھامانغا): «ئى ھامان! ماڭا بىر ئېگىز بىنا سالغىن، ئۇنىڭ بىلەن مەن دەرۋازىلارغا–ئاسماننىڭ دەرۋازىلىرىغا يېتىشىم مۇمكىن، شۇنىڭ بىلەن مەن مۇسانىڭ ئىلاھىنى كۆرەي، مەن ئىشەنچلىك ھالدا مۇسانى يالغانچى دەپ گۇمان قىلىمەن» دېدى. پىرئەۋنگە ئۇنىڭ يامان ئەمەلى شۇنداق چىرايلىق كۆرسىتىلدى، پىرئەۋن (گۇمراھلىقى تۈپەيلىدىن ھىدايەت) يولىدىن مەنئى قىلىندى، پىرئەۋننىڭ ھىيلە - مىكرى پەقەت بىكاردۇر
وَقَالَ الَّذِي آمَنَ يَا قَوْمِ اتَّبِعُونِ أَهْدِكُمْ سَبِيلَ الرَّشَادِ ﴿٣٨﴾
(پىرئەۋن خاندانىدىن) ئىمان ئېيتقان كىشى ئېيتتى: «ئى قەۋمىم! سىلەر ماڭا ئەگىشىڭلار، مەن سىلەرنى توغرا يولغا باشلايمەن
يَا قَوْمِ إِنَّمَا هَٰذِهِ الْحَيَاةُ الدُّنْيَا مَتَاعٌ وَإِنَّ الْآخِرَةَ هِيَ دَارُ الْقَرَارِ ﴿٣٩﴾
ئى قەۋمىم! بۇ دۇنيا تىرىكچىلىكى پەقەت (ۋاقىتلىق) پايدىلىنىشتىن ئىبارەتتۇر، ئاخىرەت ئىشەنچلىك ھالدا (مەڭگۈلۈك) قارارگاھتۇر
مَنْ عَمِلَ سَيِّئَةً فَلَا يُجْزَىٰ إِلَّا مِثْلَهَا ۖ وَمَنْ عَمِلَ صَالِحًا مِّن ذَكَرٍ أَوْ أُنثَىٰ وَهُوَ مُؤْمِنٌ فَأُولَٰئِكَ يَدْخُلُونَ الْجَنَّةَ يُرْزَقُونَ فِيهَا بِغَيْرِ حِسَابٍ ﴿٤٠﴾
كىمكى (بۇ دۇنيادا) بىرەر يامانلىقنى قىلىدىكەن، ئۇنىڭغا (ئاخىرەتتە) شۇنىڭغا لايىق جازا بېرىلىدۇ، مەيلى ئەر بولسۇن، مەيلى ئايال بولسۇن، كىمكى ئۆزى مۆمىن تۇرۇپ بىرەر ياخشى ئىش قىلىدىكەن، ئەنە شۇلار مەڭگۈلۈك باغلاركە كىرىدۇ، مەڭگۈلۈك باغلارتە ئۇلارغا ھېسابسىز رىزىق بېرىلىدۇ
وَيَا قَوْمِ مَا لِي أَدْعُوكُمْ إِلَى النَّجَاةِ وَتَدْعُونَنِي إِلَى النَّارِ ﴿٤١﴾
ئى قەۋمىم! ماڭا نېمىدۇركى، سىلەرنى مەن (دوزاختىن) قۇتۇلۇشقا دەۋەت قىلىمەن، سىلەر بولساڭلار مېنى دوزاخقا دەۋەت قىلىسىلەر
تَدْعُونَنِي لِأَكْفُرَ بِاللَّهِ وَأُشْرِكَ بِهِ مَا لَيْسَ لِي بِهِ عِلْمٌ وَأَنَا أَدْعُوكُمْ إِلَى الْعَزِيزِ الْغَفَّارِ ﴿٤٢﴾
سىلەر مېنى اﷲ نى ئىنكار قىلىشقا، مەن بىلمەيدىغان نەرسىلەرنى ئۇنىڭغا شېرىك كەلتۈرۈشكە دەۋەت قىلىسىلەر، ھالبۇكى، مەن سىلەرنى غالىب، (بەندىلەرنىڭ گۇناھلىرىنى) ناھايىتى مەغپىرەت قىلغۇچى (اﷲ) غا دەۋەت قىلىمەن
لَا جَرَمَ أَنَّمَا تَدْعُونَنِي إِلَيْهِ لَيْسَ لَهُ دَعْوَةٌ فِي الدُّنْيَا وَلَا فِي الْآخِرَةِ وَأَنَّ مَرَدَّنَا إِلَى اللَّهِ وَأَنَّ الْمُسْرِفِينَ هُمْ أَصْحَابُ النَّارِ ﴿٤٣﴾
ئىشەنچلىك ھالدا سىلەر مېنى ئۇنىڭغا (يەنى بۇتقا) ئىبادەت قىلشىقا چاقىرىسىلەر، ئۇنىڭ دۇنيا ۋە ئاخىرەتتە (ئۆزىگە ئىبادەت قىلىشقا) چاقىرغىنى يوق، بىزنىڭ ئاخىر بارىدىغان جايىمىز اﷲ نىڭ دەرگاھىدۇر، (گۇمراھلىقتا) ھەددىدىن ئاشقۇچىلار ئەھلى دوزاختۇر
فَسَتَذْكُرُونَ مَا أَقُولُ لَكُمْ ۚ وَأُفَوِّضُ أَمْرِي إِلَى اللَّهِ ۚ إِنَّ اللَّهَ بَصِيرٌ بِالْعِبَادِ ﴿٤٤﴾
(سىلەرگە ئازاب نازىل بولغان چاغدا) سىلەرگە ئېيتقان سۆزۈمنىڭ (راستلىقىنى) ئەسلەيسىلەر، مەن ئىشىمنى اﷲ قا تاپشۇرىمەن، اﷲ ئىشەنچلىك ھالدا بەندىلىرىنى كۆرۈپ تۇرغۇچىدۇر»

ئۇلار ئەتىگىنى - ئاخشىمى ئوتقا توغرىلىنىپ تۇرىدۇ، قىيامەت قايىم بولغان كۈندە: «ئى پىرئەۋن جامائەسى! ئازابنىڭ ئەڭ قاتتىقىغا (دوزاخ ئوتىغا) كىرىڭلار!» (دېيىلىدۇ)
وَإِذْ يَتَحَاجُّونَ فِي النَّارِ فَيَقُولُ الضُّعَفَاءُ لِلَّذِينَ اسْتَكْبَرُوا إِنَّا كُنَّا لَكُمْ تَبَعًا فَهَلْ أَنتُم مُّغْنُونَ عَنَّا نَصِيبًا مِّنَ النَّارِ ﴿٤٧﴾

قَالَ الَّذِينَ اسْتَكْبَرُوا إِنَّا كُلٌّ فِيهَا إِنَّ اللَّهَ قَدْ حَكَمَ بَيْنَ الْعِبَادِ ﴿٤٨﴾
ھاكاۋۇر (ئەگەشتۈرگۈچىلىرى): «شۈبھىسىزكى، بىز ھەممىمىز دوزاختادۇرمىز، اﷲ ئىشەنچلىك ھالدا بەندىلەرنىڭ ئارىسىدا ھۆكۈم چىقاردى» دەيدۇ
وَقَالَ الَّذِينَ فِي النَّارِ لِخَزَنَةِ جَهَنَّمَ ادْعُوا رَبَّكُمْ يُخَفِّفْ عَنَّا يَوْمًا مِّنَ الْعَذَابِ ﴿٤٩﴾
دوزاختىكىلەر دوزاخقا مۇئەككەل خۇدانىڭ ئەلچى پەرىشتىلىرىگە: «سىلەر پەرۋەردىگارىڭلارغا دۇئا قىلىڭلار، ئۇ بىزدىن بىر كۈنلۈك ئازابنى يېنىكلەتسە ئىكەن» دەيدۇ
قَالُوا أَوَلَمْ تَكُ تَأْتِيكُمْ رُسُلُكُم بِالْبَيِّنَاتِ ۖ قَالُوا بَلَىٰ ۚ قَالُوا فَادْعُوا ۗ وَمَا دُعَاءُ الْكَافِرِينَ إِلَّا فِي ضَلَالٍ ﴿٥٠﴾
ئۇلار (يەنى دوزاخقا مۇئەككەل خۇدانىڭ ئەلچى پەرىشتىلىرى): «پەيغەمبىرىڭلار سىلەرگە روشەن مۆجىزىلەرنى ئېلىپ كەلمىگەنمىدى؟» دەيدۇ. ئۇلار (يەنى كۇففارلار): «ھەئە، ئېلىپ كەلگەن ئىدى» دەيدۇ. ئۇلار (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى): «سىلەر دۇئا قىلىڭلار، (لېكىن) ھەقىقەتنى رەت قىلغۇچىلارنىڭ دۇئاسى پايدىسىزدۇر» دەيدۇ
إِنَّا لَنَنصُرُ رُسُلَنَا وَالَّذِينَ آمَنُوا فِي الْحَيَاةِ الدُّنْيَا وَيَوْمَ يَقُومُ الْأَشْهَادُ ﴿٥١﴾
شەك - شۈبھىسىزكى، بىز پەيغەمبىرىمىزگە، مۆمىنلەرگە ھاياتىي دۇنيادا ۋە (پەرىشتە، خۇدانىڭ ئەلچىسى ۋە مۆمىنلەردىن بەندىلەرنىڭ ئەمەللىرىگە گۇۋاھ بولىدىغان) گۇۋاھچىلار ھازىر بولىدىغان كۈندە ئەلۋەتتە ياردەم بېرىمىز
يَوْمَ لَا يَنفَعُ الظَّالِمِينَ مَعْذِرَتُهُمْ ۖ وَلَهُمُ اللَّعْنَةُ وَلَهُمْ سُوءُ الدَّارِ ﴿٥٢﴾
ئۇ كۈندە ھەقىقەتنى رەت قىلغۇچىلارنىڭ ئۆزرىلىرى پايدا بەرمەيدۇ، ئۇلار لەنەتكە ئۇچرايدۇ، ئۇلار ئاخىرەتنىڭ ئازابىغا دۇچار بولىدۇ
وَلَقَدْ آتَيْنَا مُوسَى الْهُدَىٰ وَأَوْرَثْنَا بَنِي إِسْرَائِيلَ الْكِتَابَ ﴿٥٣﴾
شۈبھىسىزكى، بىز مۇساغا ھىدايەت ئاتا قىلدۇق، ئىسرائىل ئەۋلادىغا مۇقەددەس كىتاب-ۋەھىي (يەنى تەۋرات) نى مىراس قىلىپ بەردۇق

(ئۇ) ئەقىل ئىگىلىرى ئۈچۈن ھىدايەتتۇر ۋە نەسىھەتتۇر
فَاصْبِرْ إِنَّ وَعْدَ اللَّهِ حَقٌّ وَاسْتَغْفِرْ لِذَنبِكَ وَسَبِّحْ بِحَمْدِ رَبِّكَ بِالْعَشِيِّ وَالْإِبْكَارِ ﴿٥٥﴾
(ئى مۇھەممەد! مۇشرىكلارنىڭ يەتكۈزگەن ئەزىيەتلىرىگە) سەۋر قىلغىن، اﷲ نىڭ (ساڭا ۋە سېنىڭ تەۋەلىرىڭگە ياردەم بېرىش) ۋەدىسى ئىشەنچلىك ھالدا ھەقتۇر، گۇناھىڭغا ئىستىغپار ئېيتقىن، ئاخشىمى - ئەتىگىنى پەرۋەردىگارىڭغا ھەمدى بىلەن تەسبىھ ئېيتقىن
إِنَّ الَّذِينَ يُجَادِلُونَ فِي آيَاتِ اللَّهِ بِغَيْرِ سُلْطَانٍ أَتَاهُمْ ۙ إِن فِي صُدُورِهِمْ إِلَّا كِبْرٌ مَّا هُم بِبَالِغِيهِ ۚ فَاسْتَعِذْ بِاللَّهِ ۖ إِنَّهُ هُوَ السَّمِيعُ الْبَصِيرُ ﴿٥٦﴾

لَخَلْقُ السَّمَاوَاتِ وَالْأَرْضِ أَكْبَرُ مِنْ خَلْقِ النَّاسِ وَلَٰكِنَّ أَكْثَرَ النَّاسِ لَا يَعْلَمُونَ ﴿٥٧﴾
ئاسمانلارنى ۋە زېمىننى يارىتىش ئەلۋەتتە ئىنسانلارنى يارىتىشتىن كۆپ قىيىندۇر ۋە لېكىن ئىنسانلارنىڭ كۆپچىلىكى (بۇنى) ئۇقمايدۇ
وَمَا يَسْتَوِي الْأَعْمَىٰ وَالْبَصِيرُ وَالَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ وَلَا الْمُسِيءُ ۚ قَلِيلًا مَّا تَتَذَكَّرُونَ ﴿٥٨﴾

إِنَّ السَّاعَةَ لَآتِيَةٌ لَّا رَيْبَ فِيهَا وَلَٰكِنَّ أَكْثَرَ النَّاسِ لَا يُؤْمِنُونَ ﴿٥٩﴾
قىيامەت چوقۇم كېلىدۇ، ئۇنىڭدا شەك يوق ۋە لېكىن ئىنسانلارنىڭ تولىسى (ئۇنىڭغا) ئىشەنمەيدۇ (شۇنىڭ ئۈچۈن ئۇلار ئۆلگەندىن كېيىن تىرىلىشنى، قىيامەت كۈنى بېرىلىدىغان مۇكاپات بىلەن جازانى ئىنكار قىلىدۇ)
وَقَالَ رَبُّكُمُ ادْعُونِي أَسْتَجِبْ لَكُمْ ۚ إِنَّ الَّذِينَ يَسْتَكْبِرُونَ عَنْ عِبَادَتِي سَيَدْخُلُونَ جَهَنَّمَ دَاخِرِينَ ﴿٦٠﴾
پەرۋەردىگارىڭلار ئېيتىدۇ: «ماڭا دۇئا قىلىڭلار، مەن (دۇئايىڭلارنى) قوبۇل قىلىمەن (تىلىگىنىڭلارنى بېرىمەن)، شۈبھىسىزكى، مېنىڭ ئىبادىتىمدىن چوڭچىلىق قىلىپ باش تارتىدىغانلار خار ھالدا مەڭگۈلۈك ئوتگە كىرىدۇ»

(كۈندۈزى ئىشلەپ يەتكەن ھارغىنلىقتىن) ئارام ئېلىشىڭلار ئۈچۈن، اﷲ سىلەرگە كېچىنى (قاراڭغۇ قىلىپ) ياراتتى، (تىرىكچىلىك يولىدا ھەرىكەتلىنىشىڭلار ئۈچۈن) كۈندۈزنى يورۇق ياراتتى. اﷲ ئىنسانلارغا ئىشەنچلىك ھالدا مەرھەمەتلىكتۇر ۋە لېكىن ئىنسانلارنىڭ تولىسى (اﷲ نىڭ ئېھسانىغا شۈكۈر قىلمايدۇ، اﷲ نىڭ مەرھەمىتى ۋە نېمەتلىرىگە نانكورلۇق قىلىدۇ)
ذَٰلِكُمُ اللَّهُ رَبُّكُمْ خَالِقُ كُلِّ شَيْءٍ لَّا إِلَٰهَ إِلَّا هُوَ ۖ فَأَنَّىٰ تُؤْفَكُونَ ﴿٦٢﴾

هُوَ الَّذِي خَلَقَكُم مِّن تُرَابٍ ثُمَّ مِن نُّطْفَةٍ ثُمَّ مِنْ عَلَقَةٍ ثُمَّ يُخْرِجُكُمْ طِفْلًا ثُمَّ لِتَبْلُغُوا أَشُدَّكُمْ ثُمَّ لِتَكُونُوا شُيُوخًا ۚ وَمِنكُم مَّن يُتَوَفَّىٰ مِن قَبْلُ ۖ وَلِتَبْلُغُوا أَجَلًا مُّسَمًّى وَلَعَلَّكُمْ تَعْقِلُونَ ﴿٦٧﴾
اﷲ سىلەرنى (يەنى ئاتاڭلار ئادەم ئەلەيھىسسالامنى) تۇپراقتىن، (ئۇنىڭ ئەۋلادىنى ئالدى بىلەن) ئابىمەنىيدىن، ئاندىن لەختە قاندىن ياراتتى، ئاندىن ئۇ سىلەرنى (ئاناڭلارنىڭ قارنىدىن) بوۋاق ھالىتىڭلاردا چىقىرىدۇ، ئاندىن (كۈچ - قۇۋۋەت ۋە ئەقىلدە كامالەت يېشى) قىران ۋاقتىڭلارغا يېتىسىلەر، ئاندىن قېرىيسىلەر– بەزىڭلار بۇرۇن ۋاپات بولۇپ كېتىسىلەر–ئاندىن سىلەر مۇئەييەن ۋاقىتقىچە ياشايسىلەر، سىلەر (اﷲ نىڭ قۇدرىتىنىڭ دەلىللىرىنى) چۈشەنگەيسىلەر
هُوَ الَّذِي يُحْيِي وَيُمِيتُ ۖ فَإِذَا قَضَىٰ أَمْرًا فَإِنَّمَا يَقُولُ لَهُ كُن فَيَكُونُ ﴿٦٨﴾
اﷲ تىرىلدۈرۈشكە ۋە ئۆلتۈرۈشكە قادىر، ئۇ بىرەر ئىشنى (قىلىشنى) ئىرادە قىلسا، ئۇنىڭغا «ۋۇجۇدقا كەل» دەيدۇ - دە، ئۇ ۋۇجۇدقا كېلىدۇ
أَلَمْ تَرَ إِلَى الَّذِينَ يُجَادِلُونَ فِي آيَاتِ اللَّهِ أَنَّىٰ يُصْرَفُونَ ﴿٦٩﴾
اﷲ نىڭ ئايەتلىرى ئۈستىدە جاڭجاللىشىۋاتقانلارنى كۆرمىدىڭمۇ؟ ئۇلار قانداقمۇ (ھىدايەتتىن گۇمراھلىققا) بۇرۇلۇپ كېتىدۇ؟
الَّذِينَ كَذَّبُوا بِالْكِتَابِ وَبِمَا أَرْسَلْنَا بِهِ رُسُلَنَا ۖ فَسَوْفَ يَعْلَمُونَ ﴿٧٠﴾
ئۇلار مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) ئىنكار قىلدى، خۇدانىڭ ئەلچىسىلىرىمىزگە بىز نازىل قىلغان مۆجىزىلەرنى ئىنكار قىلدى. ئۇلار ئۇزاققا قالماي (ئىنكار قىلغانلىقلىرىنىڭ ئاقىۋىتىنى) بىلىدۇ
إِذِ الْأَغْلَالُ فِي أَعْنَاقِهِمْ وَالسَّلَاسِلُ يُسْحَبُونَ ﴿٧١﴾
ئۇ چاغدا (يەنى دوزاخقا كىرگەندە) ئۇلارنىڭ بويۇنلىرىدا تاقاقلار ۋە زەنجىرلەر بولىدۇ، ئۇلار قايناقسۇغا سۆرەپ كىرىلىدۇ، ئاندىن ئوتتا كۆيدۈرۈلىدۇ
فِي الْحَمِيمِ ثُمَّ فِي النَّارِ يُسْجَرُونَ ﴿٧٢﴾
ئۇ چاغدا (يەنى دوزاخقا كىرگەندە) ئۇلارنىڭ بويۇنلىرىدا تاقاقلار ۋە زەنجىرلەر بولىدۇ، ئۇلار قايناقسۇغا سۆرەپ كىرىلىدۇ، ئاندىن ئوتتا كۆيدۈرۈلىدۇ
ثُمَّ قِيلَ لَهُمْ أَيْنَ مَا كُنتُمْ تُشْرِكُونَ ﴿٧٣﴾
ئاندىن ئۇلارغا: «اﷲ نى قويۇپ شېرىك كەلتۈرگەن بۇتلىرىڭلار قەيەردە؟» دېيىلىدۇ، ئۇلار: «ئۇ بۇتلار ئۆزلىرىنى بىزدىن چەتكە ئالدى، بەلكى بىز ئىلگىرى ھېچنەرسىگە ئىبادەت قىلمايتتۇق» دەيدۇ. اﷲ ھەقىقەتنى رەت قىلغۇچىلارنى مۇشۇنداق گۇمراھ قىلىدۇ
مِن دُونِ اللَّهِ ۖ قَالُوا ضَلُّوا عَنَّا بَل لَّمْ نَكُن نَّدْعُو مِن قَبْلُ شَيْئًا ۚ كَذَٰلِكَ يُضِلُّ اللَّهُ الْكَافِرِينَ ﴿٧٤﴾
ئاندىن ئۇلارغا: «اﷲ نى قويۇپ شېرىك كەلتۈرگەن بۇتلىرىڭلار قەيەردە؟» دېيىلىدۇ، ئۇلار: «ئۇ بۇتلار ئۆزلىرىنى بىزدىن چەتكە ئالدى، بەلكى بىز ئىلگىرى ھېچنەرسىگە ئىبادەت قىلمايتتۇق» دەيدۇ. اﷲ ھەقىقەتنى رەت قىلغۇچىلارنى مۇشۇنداق گۇمراھ قىلىدۇ
ذَٰلِكُم بِمَا كُنتُمْ تَفْرَحُونَ فِي الْأَرْضِ بِغَيْرِ الْحَقِّ وَبِمَا كُنتُمْ تَمْرَحُونَ ﴿٧٥﴾
بۇ (ئازاب) سىلەرنىڭ زېمىندا ھەقسىز رەۋىشتە (گۇناھ قىلىپ پۇل - مالنى ھارام يوللارغا سەرپ قىلغانلىقىڭلار ئۈچۈن) خۇشال بولغانلىقىڭلار ۋە ھاكاۋۇر بولۇپ كەتكەنلىكىڭلار ئۈچۈندۇر
ادْخُلُوا أَبْوَابَ جَهَنَّمَ خَالِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى الْمُتَكَبِّرِينَ ﴿٧٦﴾
سىلەر دوزاخنىڭ دەرۋازىلىرىدىن كىرىپ ئۇ يەردە مەڭگۈ قېلىڭلار، مۇتەكەببۇرلارنىڭ جايى نېمىدېگەن يامان؟
فَاصْبِرْ إِنَّ وَعْدَ اللَّهِ حَقٌّ ۚ فَإِمَّا نُرِيَنَّكَ بَعْضَ الَّذِي نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا يُرْجَعُونَ ﴿٧٧﴾
(ئى مۇھەممەد! قەۋمىڭنىڭ سېنى ئىنكار قىلغانلىقىغا) سەۋر قىلغىن، اﷲ نىڭ (ئۇلارنى ئازابلاش توغرىسىدىكى) ۋەدىسى ئىشەنچلىك ھالدا ھەقتۇر، ساڭا بىز ئۇلارغا ۋەدە قىلغان ئازابنىڭ بىر قىسمىنى كۆرسەتسەك (بۇ اﷲ نىڭ ۋەدىسىگە ۋاپا قىلغانلىقىدۇر) ياكى سېنى (ئۇنى كۆرسىتىشتىن بۇرۇن) ۋاپات قىلدۇرساق، ئۇلار بىزنىڭ دەرگاھىمىزغا قايتۇرۇلىدۇ، (قىيامەتتە ئۇلار ئازابقا دۇچار بولىدۇ)
وَلَقَدْ أَرْسَلْنَا رُسُلًا مِّن قَبْلِكَ مِنْهُم مَّن قَصَصْنَا عَلَيْكَ وَمِنْهُم مَّن لَّمْ نَقْصُصْ عَلَيْكَ ۗ وَمَا كَانَ لِرَسُولٍ أَن يَأْتِيَ بِآيَةٍ إِلَّا بِإِذْنِ اللَّهِ ۚ فَإِذَا جَاءَ أَمْرُ اللَّهِ قُضِيَ بِالْحَقِّ وَخَسِرَ هُنَالِكَ الْمُبْطِلُونَ ﴿٧٨﴾
سەندىن ئىلگىرى بىز ئىشەنچلىك ھالدا نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتتۇق، ئۇلارنىڭ ئارىسىدىن ساڭا بايان قىلىپ بەرگەنلىرىمىزمۇ، بايان قىلىپ بەرمىگەنلىرىمىزمۇ بار، ھېچقانداق خۇدانىڭ ئەلچىسى اﷲ نىڭ ئىزنىسىز بىرەر مۆجىزە كەلتۈرەلمەيدۇ، اﷲ نىڭ ئەمرى (يەنى قىيامەت) كەلگەن چاغدا (كىشىلەرنىڭ ئارىسىدا) ئادالەت بىلەن ھۆكۈم چىقىرىلىدۇ، بۇ چاغدا ھەقكە قارشى تۇرغۇچىلار زىيان تارتىدۇ
اللَّهُ الَّذِي جَعَلَ لَكُمُ الْأَنْعَامَ لِتَرْكَبُوا مِنْهَا وَمِنْهَا تَأْكُلُونَ ﴿٧٩﴾
مىنىشىڭلار ۋە گۆشلىرىنى يېيىشىڭلار ئۈچۈن اﷲ سىلەرگە ھايۋانلارنى ياراتتى
وَلَكُمْ فِيهَا مَنَافِعُ وَلِتَبْلُغُوا عَلَيْهَا حَاجَةً فِي صُدُورِكُمْ وَعَلَيْهَا وَعَلَى الْفُلْكِ تُحْمَلُونَ ﴿٨٠﴾
ئۇلار سىلەرگە نۇرغۇن پايدا كەلتۈرىدۇ، ئۇلارغا مىنىپ دىلىڭلارغا پۈككەن ھاجىتىڭلاردىن چىقىسىلەر، (قۇرۇقلۇقتا يۈك - تاقلىرىڭلارنى) ئۇلارغا ئارتىسىلەر، (دېڭىزدا) كېمىلەرگە قاچىلايسىلەر
وَيُرِيكُمْ آيَاتِهِ فَأَيَّ آيَاتِ اللَّهِ تُنكِرُونَ ﴿٨١﴾
اﷲ سىلەرگە (قۇدرىتىنىڭ) ئالامەتلىرىنى كۆرسىتىدۇ، سىلەر اﷲ نىڭ (قۇدرىتىنىڭ) قايسى ئالامەتلىرىنى ئىنكار قىلىسىلەر؟
أَفَلَمْ يَسِيرُوا فِي الْأَرْضِ فَيَنظُرُوا كَيْفَ كَانَ عَاقِبَةُ الَّذِينَ مِن قَبْلِهِمْ ۚ كَانُوا أَكْثَرَ مِنْهُمْ وَأَشَدَّ قُوَّةً وَآثَارًا فِي الْأَرْضِ فَمَا أَغْنَىٰ عَنْهُم مَّا كَانُوا يَكْسِبُونَ ﴿٨٢﴾
ئۇلار زېمىندا سەير قىلىپ ئۇلاردىن ئىلگىرىكىلەرنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۆزەتمىدىمۇ؟ بۇلاردىن ئۇلارنىڭ سانى كۆپ ئىدى، زېمىندا كۈچ - قۇۋۋەت ۋە يادىكارلىقلار (يەنى ئۆي - ئىمارەت) جەھەتتە بۇلاردىن ئۈستۈن ئىدى. ئۇلارنىڭ قولغا كەلتۈرگەن نەرسىلىرى ئۇلارغا ئەسقاتمىدى
فَلَمَّا جَاءَتْهُمْ رُسُلُهُم بِالْبَيِّنَاتِ فَرِحُوا بِمَا عِندَهُم مِّنَ الْعِلْمِ وَحَاقَ بِهِم مَّا كَانُوا بِهِ يَسْتَهْزِئُونَ ﴿٨٣﴾
ئۇلارغا خۇدانىڭ ئەلچىسىلىرى روشەن مۆجىزىلەر بىلەن كەلگەندە، ئۆزلىرىدە بار ئىلىم بىلەن پەخىرلىنىپ (خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلدى ۋە مەسخىرە قىلدى)، ئۇلارغا ئۆزلىرى مەسخىرە قىلغان (ۋە كەلسە دەپ ئالدىرىغان) ئازاب نازىل بولدى
فَلَمَّا رَأَوْا بَأْسَنَا قَالُوا آمَنَّا بِاللَّهِ وَحْدَهُ وَكَفَرْنَا بِمَا كُنَّا بِهِ مُشْرِكِينَ ﴿٨٤﴾
ئۇلار بىزنىڭ ئازابىمىزنى كۆرگەن چاغدا: «بىز بىر اﷲ قا ئىمان ئېيتتۇق، بىز بۇرۇن (اﷲ قا) شېرىك قىلغان بۇتلارنى ئىنكار قىلدۇق» دېدى
فَلَمْ يَكُ يَنفَعُهُمْ إِيمَانُهُمْ لَمَّا رَأَوْا بَأْسَنَا ۖ سُنَّتَ اللَّهِ الَّتِي قَدْ خَلَتْ فِي عِبَادِهِ ۖ وَخَسِرَ هُنَالِكَ الْكَافِرُونَ ﴿٨٥﴾
بىزنىڭ ئازابىمىزنى كۆرگەن چاغدا ئېيتقان ئىماننىڭ ئۇلارغا پايدىسى بولمىدى، بۇ (يەنى ئازابنى كۆرگەندە ئېيتقان ئىماننىڭ پايدىلىق بولماسلىقى) اﷲ نىڭ بەندىلىرى ئارىسىدا تۇتقان يولىدۇر، بۇ چاغدا ھەقىقەتنى رەت قىلغۇچىلار زىيان تارتقۇچى بولدى
