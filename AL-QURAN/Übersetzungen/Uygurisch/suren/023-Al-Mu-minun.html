بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
قَدْ أَفْلَحَ ٱلْمُؤْمِنُونَ ﴿١﴾
مۆمىنلەر ئىشەنچلىك ھالدا بەختكە ئېرىشتى
ٱلَّذِينَ هُمْ فِى صَلَاتِهِمْ خَـٰشِعُونَ ﴿٢﴾
(شۇنداق مۆمىنلەركى) ئۇلار نامازلىرىدا (اﷲ نىڭ ئۇلۇغلۇقىدىن سۈر بېسىپ كەتكەنلىكتىن) ئەيمىنىپ تۇرغۇچىلاردۇر
وَٱلَّذِينَ هُمْ عَنِ ٱللَّغْوِ مُعْرِضُونَ ﴿٣﴾
ئۇلار بىھۇدە سۆز، بىھۇدە ئىشتىن يىراق بولغۇچىلاردۇر
وَٱلَّذِينَ هُمْ لِلزَّكَوٰةِ فَـٰعِلُونَ ﴿٤﴾
ئۇلار زاكات (پاكلاش ئۇلۇشى) بەرگۈچىلەردۇر
وَٱلَّذِينَ هُمْ لِفُرُوجِهِمْ حَـٰفِظُونَ ﴿٥﴾
ئۇلار ئەۋرەتلىرىنى (ھارامدىن) ساقلىغۇچىلاردۇر
إِلَّا عَلَىٰٓ أَزْوَٰجِهِمْ أَوْ مَا مَلَكَتْ أَيْمَـٰنُهُمْ فَإِنَّهُمْ غَيْرُ مَلُومِينَ ﴿٦﴾
(يەنى ئەۋرەتلىرىنى) پەقەت خوتۇنلىرىدىن، چۆرىلىرىدىن باشقىلاردىن ساقلىغۇچىلاردۇر (بۇلار بىلەن يېقىنچىلىق قىلغۇچىلار مالامەت قىلىنمايدۇ)
فَمَنِ ٱبْتَغَىٰ وَرَآءَ ذَٰلِكَ فَأُو۟لَـٰٓئِكَ هُمُ ٱلْعَادُونَ ﴿٧﴾
بۇنىڭ سىرتىدىن (جىنسىي تەلەپنى قاندۇرۇشنى) تەلەپ قىلغۇچىلار ھەددىدىن ئاشقۇچىلاردۇر
وَٱلَّذِينَ هُمْ لِأَمَـٰنَـٰتِهِمْ وَعَهْدِهِمْ رَٰعُونَ ﴿٨﴾
ئۇلار (يەنى مۆمىنلەر) ئۆزلىرىگە تاپشۇرۇلغان ئامانەتلەرگە ۋە بەرگەن ئەھدىگە رىئايە قىلغۇچىلاردۇر
وَٱلَّذِينَ هُمْ عَلَىٰ صَلَوَٰتِهِمْ يُحَافِظُونَ ﴿٩﴾
ئۇلار نامازلىرىنى (ۋاقتىدا تەئدىل ئەركان بىلەن) ئادا قىلغۇچىلاردۇر
أُو۟لَـٰٓئِكَ هُمُ ٱلْوَٰرِثُونَ ﴿١٠﴾
ئەنە شۇلار (يەنى يۇقىرى سۈپەتلەرگە ئىگە مۆمىنلەر نازۇنېمەتلىك مەڭگۈلۈك باغلارنىڭ) ۋارىسلىرىدۇر
ٱلَّذِينَ يَرِثُونَ ٱلْفِرْدَوْسَ هُمْ فِيهَا خَـٰلِدُونَ ﴿١١﴾
ئۇلار (يۇقىرى دەرىجىلىك مەڭگۈلۈك باغلار) فىردەۋسكە ۋارىسلىق قىلىدۇ، فىردەۋستە مەڭگۈ قالىدۇ
وَلَقَدْ خَلَقْنَا ٱلْإِنسَـٰنَ مِن سُلَـٰلَةٍ مِّن طِينٍ ﴿١٢﴾
بىز ئىنساننى ئىشەنچلىك ھالدا لاينىڭ جەۋھىرىدىن ياراتتۇق
ثُمَّ جَعَلْنَـٰهُ نُطْفَةً فِى قَرَارٍ مَّكِينٍ ﴿١٣﴾
ئاندىن ئۇنى بىر پۇختا قارارگاھ (يەنى بەچچىدان) دا (جايلاشقان) ئابىمەنىي قىلدۇق
ثُمَّ خَلَقْنَا ٱلنُّطْفَةَ عَلَقَةً فَخَلَقْنَا ٱلْعَلَقَةَ مُضْغَةً فَخَلَقْنَا ٱلْمُضْغَةَ عِظَـٰمًا فَكَسَوْنَا ٱلْعِظَـٰمَ لَحْمًا ثُمَّ أَنشَأْنَـٰهُ خَلْقًا ءَاخَرَ ۚ فَتَبَارَكَ ٱللَّهُ أَحْسَنُ ٱلْخَـٰلِقِينَ ﴿١٤﴾
ئاندىن ئابىمەنىينى لەختە قانغا ئايلاندۇردۇق، ئاندىن لەختە قاننى پارچە گۆشكە ئايلاندۇردۇق، ئاندىن پارچە گۆشنى سۆڭەككە ئايلاندۇردۇق، ئاندىن سۆڭەككە گۆش قوندۇردۇق، ئاندىن ئۇنى (جان كىرگۈزۈپ) باشقا مەخلۇققا ئايلاندۇردۇق، ئاڭ ماھىر ياراتقۇچى اﷲ (ھېكمەت ۋە قۇدرەت جەھەتتىن) ئۇلۇغدۇر
ثُمَّ إِنَّكُم بَعْدَ ذَٰلِكَ لَمَيِّتُونَ ﴿١٥﴾
ئاندىن سىلەر چوقۇم ئۆلۈسىلەر
ثُمَّ إِنَّكُمْ يَوْمَ ٱلْقِيَـٰمَةِ تُبْعَثُونَ ﴿١٦﴾
ئاندىن سىلەر ئىشەنچلىك ھالدا قىيامەت كۈنى (قەبرەڭلاردىن) تۇرغۇزۇلىسىلەر
وَلَقَدْ خَلَقْنَا فَوْقَكُمْ سَبْعَ طَرَآئِقَ وَمَا كُنَّا عَنِ ٱلْخَلْقِ غَـٰفِلِينَ ﴿١٧﴾
بىز ئىشەنچلىك ھالدا سىلەرنىڭ ئۈستۈڭلاردا يەتتە قەۋەت ئاسماننى ياراتتۇق، بىز مەخلۇقاتتىن غاپىل ئەمەسمىز
وَأَنزَلْنَا مِنَ ٱلسَّمَآءِ مَآءًۢ بِقَدَرٍ فَأَسْكَنَّـٰهُ فِى ٱلْأَرْضِ ۖ وَإِنَّا عَلَىٰ ذَهَابٍۭ بِهِۦ لَقَـٰدِرُونَ ﴿١٨﴾
بىز بۇلۇتتىن ھاجەتكە قاراپ يامغۇر ياغدۇردۇق، ئۇنى زېمىندا توختاتتۇق، ئۇنى قۇرۇتۇۋېتىشقا ئەلۋەتتە قادىرمىز
فَأَنشَأْنَا لَكُم بِهِۦ جَنَّـٰتٍ مِّن نَّخِيلٍ وَأَعْنَـٰبٍ لَّكُمْ فِيهَا فَوَٰكِهُ كَثِيرَةٌ وَمِنْهَا تَأْكُلُونَ ﴿١٩﴾
شۇ يامغۇر بىلەن سىلەرگە خورما باغچىلىرى ۋە ئۈزۈم باغچىلىرىنى يېتىشتۈرۈپ بەردۇق، سىلەر ئۈچۈن ئۇ باغلاردا نۇرغۇن مېۋىلەر بار، ئۇلاردىن يەيسىلەر
وَشَجَرَةً تَخْرُجُ مِن طُورِ سَيْنَآءَ تَنۢبُتُ بِٱلدُّهْنِ وَصِبْغٍ لِّلْـَٔاكِلِينَ ﴿٢٠﴾
شۇ يامغۇر بىلەن تۇرسىنادىن چىقىدىغان (زەيتۇن) دەرىخىنى (ئۆستۈرۈپ بەردۇق)، ئۇنىڭدىن ماي چىقىدۇ، يېگۈچىلەر (نانلىرىنى شۇ ماي بىلەن) مايلايدۇ
وَإِنَّ لَكُمْ فِى ٱلْأَنْعَـٰمِ لَعِبْرَةً ۖ نُّسْقِيكُم مِّمَّا فِى بُطُونِهَا وَلَكُمْ فِيهَا مَنَـٰفِعُ كَثِيرَةٌ وَمِنْهَا تَأْكُلُونَ ﴿٢١﴾
چارۋىلاردا ئىشەنچلىك ھالدا سىلەرگە ئىبرەت بار، ئۇلارنىڭ قورسىقىدىكى نەرسىلەر (يەنى سۈتلەر) بىلەن سىلەرنى سۇغىرىمىز، ئۇلاردا سىلەر ئۈچۈن نۇرغۇن مەنپەئەتلەر بار، ئۇلارنىڭ گۆشىنى يەيسىلەر
وَعَلَيْهَا وَعَلَى ٱلْفُلْكِ تُحْمَلُونَ ﴿٢٢﴾
ئۇلار (يەنى تۆگە ۋە ئۆكۈزلەر) بىلەن (قۇرۇقلۇقتا)، كېمە بىلەن (دېڭىزدا) يۈك توشۇيسىلەر
وَلَقَدْ أَرْسَلْنَا نُوحًا إِلَىٰ قَوْمِهِۦ فَقَالَ يَـٰقَوْمِ ٱعْبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنْ إِلَـٰهٍ غَيْرُهُۥٓ ۖ أَفَلَا تَتَّقُونَ ﴿٢٣﴾
بىز ئىشەنچلىك ھالدا نۇھنى قەۋمىگە خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق، نۇھ ئېيتتى: «ئى قەۋمىم! (يالغۇز اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە ئۇنىڭدىن باشقا ھېچ مەبۇد (بەرھەق) يوق، اﷲ نىڭ ئازابىدىن) قورقمامسىلەر؟»
فَقَالَ ٱلْمَلَؤُا۟ ٱلَّذِينَ كَفَرُوا۟ مِن قَوْمِهِۦ مَا هَـٰذَآ إِلَّا بَشَرٌ مِّثْلُكُمْ يُرِيدُ أَن يَتَفَضَّلَ عَلَيْكُمْ وَلَوْ شَآءَ ٱللَّهُ لَأَنزَلَ مَلَـٰٓئِكَةً مَّا سَمِعْنَا بِهَـٰذَا فِىٓ ءَابَآئِنَا ٱلْأَوَّلِينَ ﴿٢٤﴾
ئۇنىڭ قەۋمىدىكى ئىمانسىز كاتتىلار ئېيتتى: «بۇ پەقەت سىلەرگە ئوخشاش بىر ئىنساندۇر، ئۇ سىلەردىن ئۈستۈن بولۇۋالماقچى، ئەگەر اﷲ (خۇدانىڭ ئەلچىسى ئەۋەتىشنى) خالىسا ئىدى، ئەلۋەتتە، خۇدانىڭ ئەلچى پەرىشتىلىرىنى ئەۋەتتى، بۇنداق سۆزنى بۇرۇنقى ئاتا - بوۋىلىرىمىزدىن ئاڭلىغان ئەمەسمىز
إِنْ هُوَ إِلَّا رَجُلٌۢ بِهِۦ جِنَّةٌ فَتَرَبَّصُوا۟ بِهِۦ حَتَّىٰ حِينٍ ﴿٢٥﴾
ئۇ پەقەت بىر مەجنۇندۇر، سىلەر ئۇنى بىر مەزگىل كۈتۈڭلار»
قَالَ رَبِّ ٱنصُرْنِى بِمَا كَذَّبُونِ ﴿٢٦﴾
نۇھ (ئۇلارنىڭ ئىمان ئېيتىشىدىن ئۈمىدىنى ئۈزگەندىن كېيىن): «پەرۋەردىگارىم! ئۇلار مېنى يالغانغا چىقارغانلىقلىرى ئۈچۈن ماڭا ياردەم بەرگىن» دېدى
فَأَوْحَيْنَآ إِلَيْهِ أَنِ ٱصْنَعِ ٱلْفُلْكَ بِأَعْيُنِنَا وَوَحْيِنَا فَإِذَا جَآءَ أَمْرُنَا وَفَارَ ٱلتَّنُّورُ ۙ فَٱسْلُكْ فِيهَا مِن كُلٍّ زَوْجَيْنِ ٱثْنَيْنِ وَأَهْلَكَ إِلَّا مَن سَبَقَ عَلَيْهِ ٱلْقَوْلُ مِنْهُمْ ۖ وَلَا تُخَـٰطِبْنِى فِى ٱلَّذِينَ ظَلَمُوٓا۟ ۖ إِنَّهُم مُّغْرَقُونَ ﴿٢٧﴾
بىز نۇھقا: «بىزنىڭ كۆز ئالدىمىزدا ۋەھيىمىز بويىچە كېمە ياسىغىن، بىزنىڭ بۇيرۇقىمىز يېتىپ كەلگەن ۋە تونۇردىن سۇ ئېتىلىپ چىققان چاغدا، ھايۋانلاردىن (ئەركەك - چىشى بولۇپ) بىر جۈپتىن ئېلىۋالغىن، ئائىلەڭدىكىلەرنىمۇ ئېلىۋالغىن، ئۇلارنىڭ ئىچىدىكى ھالاك بولۇشقا ھۆكۈم قىلىنغانلار بۇنىڭدىن مۇستەسنا. (ئۆزلىرىگە) زۇلۇم قىلغانلار توغرىسىدا ماڭا سۆز ئاچمىغىن (يەنى ئۇلارغا شاپائەت قىلمىغىن)، ئۇلار چوقۇم (توپان بالاسىدا) غەرق قىلىنىدۇ» دەپ ۋەھيى قىلدۇق
فَإِذَا ٱسْتَوَيْتَ أَنتَ وَمَن مَّعَكَ عَلَى ٱلْفُلْكِ فَقُلِ ٱلْحَمْدُ لِلَّهِ ٱلَّذِى نَجَّىٰنَا مِنَ ٱلْقَوْمِ ٱلظَّـٰلِمِينَ ﴿٢٨﴾
سەن بىلەن بىللە بولغان مۆمىنلەر بىلەن كېمىگە چىققان چېغىڭدا: «جىمى ھەمدۇسانا بىزنى زالىم قەۋمدىن قۇتقۇزغان اﷲ قا خاستۇر» دېگىن
وَقُل رَّبِّ أَنزِلْنِى مُنزَلًا مُّبَارَكًا وَأَنتَ خَيْرُ ٱلْمُنزِلِينَ ﴿٢٩﴾
ھەمدە: «ئى پەرۋەردىگارىم! مېنى مۇبارەك مەنزىلگە چۈشۈرگىن، سەن (دوستلىرىڭنى) ئەڭ ياخشى ئورۇنلاشتۇرغۇچىسەن» دېگىن
إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ وَإِن كُنَّا لَمُبْتَلِينَ ﴿٣٠﴾
ئىشەنچلىك ھالدا بۇنىڭدا (يەنى نۇھنىڭ ئۈممىتىنىڭ ئىشلىرىدا ئەقىل ئىگىلىرى ئۈچۈن) نۇرغۇن ئىبرەتلەر بار، بىز بەندىلەرنى (خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتىش بىلەن) ئەلۋەتتە سىنايمىز
ثُمَّ أَنشَأْنَا مِنۢ بَعْدِهِمْ قَرْنًا ءَاخَرِينَ ﴿٣١﴾
ئۇلارنى ھالاك قىلغاندىن كېيىن باشقا بىر قەۋم پەيدا قىلدۇق
فَأَرْسَلْنَا فِيهِمْ رَسُولًا مِّنْهُمْ أَنِ ٱعْبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنْ إِلَـٰهٍ غَيْرُهُۥٓ ۖ أَفَلَا تَتَّقُونَ ﴿٣٢﴾
بىز ئۇلارغا ئۆزلىرىنىڭ ئىچىدىن بىر خۇدانىڭ ئەلچىسىنى (يەنى ھۇدنى) ئەۋەتتۇق. (ئۇ ئېيتتى) اﷲ غىلا ئىبادەت قىلىڭلار، سىلەرگە ئۇنىڭدىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر ئاللانىڭ (ئازابىدىن) قورقمامسىلەر؟
وَقَالَ ٱلْمَلَأُ مِن قَوْمِهِ ٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِلِقَآءِ ٱلْـَٔاخِرَةِ وَأَتْرَفْنَـٰهُمْ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا مَا هَـٰذَآ إِلَّا بَشَرٌ مِّثْلُكُمْ يَأْكُلُ مِمَّا تَأْكُلُونَ مِنْهُ وَيَشْرَبُ مِمَّا تَشْرَبُونَ ﴿٣٣﴾
ئۇنىڭ قەۋمىنىڭ ھەقىقەتنى رەت قىلغۇچى بولغان، ئاخىرەت مۇكاپاتىنى يالغان چىقارغان ۋە بۇ دۇنيادا بىز تەرەپتىن باياشات تۇرمۇشقا ئىگە قىلىنغان كاتتىلىرى ئېيتتى: «بۇ پەقەت سىلەرگە ئوخشاش بىر ئىنساندۇر، ئۇ سىلەر يېگەننى يەيدۇ، سىلەر ئىچكەننى ئىچىدۇ.

ئۇلار (ھەق) كالام (يەنى ئۇلۇغ قۇرئان) ئۈستىدە چوڭقۇر پىكىر قىلمىدىمۇ؟ ياكى ئۇلارغا (اﷲ تەرىپىدىن) ئاتا - بوۋىلىرىغا كەلمىگەن (يېڭى) نەرسە كەلدىمۇ؟
أَمْ لَمْ يَعْرِفُوا۟ رَسُولَهُمْ فَهُمْ لَهُۥ مُنكِرُونَ ﴿٦٩﴾
ياكى ئۇلار ئۆزلىرىگە كەلگەن ئەلچىنى تونۇماي ئۇنى ئىنكار قىلىۋاتامدۇ
أَمْ يَقُولُونَ بِهِۦ جِنَّةٌۢ ۚ بَلْ جَآءَهُم بِٱلْحَقِّ وَأَكْثَرُهُمْ لِلْحَقِّ كَـٰرِهُونَ ﴿٧٠﴾
ياكى ئۇلار ئۇنى مەجنۇن دېيىشەمدۇ؟ ھەرگىز ئۇنداق ئەمەس، ئۇلارغا ئۇ ھەقىقەتنى ئېلىپ كەلدى. ئۇلارنىڭ تولىسى ھەقىقەتنى يامان كۆرىدۇ
وَلَوِ ٱتَّبَعَ ٱلْحَقُّ أَهْوَآءَهُمْ لَفَسَدَتِ ٱلسَّمَـٰوَٰتُ وَٱلْأَرْضُ وَمَن فِيهِنَّ ۚ بَلْ أَتَيْنَـٰهُم بِذِكْرِهِمْ فَهُمْ عَن ذِكْرِهِم مُّعْرِضُونَ ﴿٧١﴾
ئەگەر ھەقىقەت (يەنى قۇرئان) ئۇلارنىڭ نەپسى خاھىشلىرىغا بويسۇنىدىغان بولسا، ئەلۋەتتە ئاسمانلار، زېمىن ۋە ئۇلاردىكى مەخلۇقاتلار خاراب بولغان بولاتتى، ئۇنداق بولغىنى يوق، ئۇلارغا (ۋەز - نەسىھەتنى ئۆز ئىچىگە ئالغان) قۇرئاننى ئاتا قىلدۇق، ئۇلار ئۆزلىرىگە (ۋەز - نەسىھەت بولغان) قۇرئاندىن يۈز ئۆرۈيدۇ
أَمْ تَسْـَٔلُهُمْ خَرْجًا فَخَرَاجُ رَبِّكَ خَيْرٌ ۖ وَهُوَ خَيْرُ ٱلرَّٰزِقِينَ ﴿٧٢﴾
ياكى سەن ئۇلاردىن (ئەلچىلىكنى يەتكۈزگەنلىكىڭگە) ھەق تەلەپ قىلامسەن؟ (يەنى ئۇلار شۇنداق گۇمان قىلامدۇ؟) پەرۋەردىگارىڭنىڭ (ساڭا بەرگەن) ئەجرى ياخشىدۇر، اﷲ رىزىق بەرگۈچىلەرنىڭ ياخشىسىدۇر
وَإِنَّكَ لَتَدْعُوهُمْ إِلَىٰ صِرَٰطٍ مُّسْتَقِيمٍ ﴿٧٣﴾
سەن ئەلۋەتتە ئۇلارنى توغرا يولغا دەۋەت قىلىسەن
وَإِنَّ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْـَٔاخِرَةِ عَنِ ٱلصِّرَٰطِ لَنَـٰكِبُونَ ﴿٧٤﴾
ئاخىرەتكە ئىشەنمەيدىغانلار ئىشەنچلىك ھالدا توغرا يولدىن چەتنىگۈچىلەردۇر
۞ وَلَوْ رَحِمْنَـٰهُمْ وَكَشَفْنَا مَا بِهِم مِّن ضُرٍّ لَّلَجُّوا۟ فِى طُغْيَـٰنِهِمْ يَعْمَهُونَ ﴿٧٥﴾
ئەگەر ئۇلار (يەنى مۇشرىكلار) غا رەھىم قىلساق، ئۇلارغا كەلگەن كۈلپەتنى كۆتۈرۈۋەتسەك، چوقۇم گۇمراھلىقلىرىدا داۋاملىق تېڭىرقاپ يۈرۈشەتتى
وَلَقَدْ أَخَذْنَـٰهُم بِٱلْعَذَابِ فَمَا ٱسْتَكَانُوا۟ لِرَبِّهِمْ وَمَا يَتَضَرَّعُونَ ﴿٧٦﴾
ئۇلارنى بىز ئىشەنچلىك ھالدا ئازاب (يەنى قەھەتچىلىك) بىلەن جازالىدۇق، ئۇلار پەرۋەردىگارىغا بويسۇنمىدى، ئىلتىجا قىلىپ دۇئامۇ قىلمىدى
حَتَّىٰٓ إِذَا فَتَحْنَا عَلَيْهِم بَابًا ذَا عَذَابٍ شَدِيدٍ إِذَا هُمْ فِيهِ مُبْلِسُونَ ﴿٧٧﴾
تاكى ئۇلارغا قاتتىق ئازابنىڭ دەرۋازىسىنى ئاچقان چېغىمىزدا ناگاھان ئۇلار ئۈمىدسىزلىنىپ كېتىدۇ
وَهُوَ ٱلَّذِىٓ أَنشَأَ لَكُمُ ٱلسَّمْعَ وَٱلْأَبْصَـٰرَ وَٱلْأَفْـِٔدَةَ ۚ قَلِيلًا مَّا تَشْكُرُونَ ﴿٧٨﴾
اﷲ سىلەر ئۈچۈن قۇلاقلارنى، كۆزلەرنى ۋە دىللارنى ياراتتى، سىلەر ناھايىتى ئاز شۈكۈر قىلىسىلەر
وَهُوَ ٱلَّذِى ذَرَأَكُمْ فِى ٱلْأَرْضِ وَإِلَيْهِ تُحْشَرُونَ ﴿٧٩﴾
اﷲ سىلەرنى زېمىندا (نەسىللەندۈرۈش يولى بىلەن) ياراتتى، (ھېساب بېرىش ئۈچۈن) ئۇنىڭ دەرگاھىغا يىغىلىسىلەر
وَهُوَ ٱلَّذِى يُحْىِۦ وَيُمِيتُ وَلَهُ ٱخْتِلَـٰفُ ٱلَّيْلِ وَٱلنَّهَارِ ۚ أَفَلَا تَعْقِلُونَ ﴿٨٠﴾
اﷲ (ئۆلۈكنى) تىرىلدۈرىدۇ، (تىرىكنىڭ) جېنىنى ئالىدۇ، كېچە - كۈندۈزنىڭ ئۆزگىرىپ تۇرۇشى اﷲ نىڭ باشقۇرۇشىدىدۇر، چۈشەنمەمسىلەر؟
بَلْ قَالُوا۟ مِثْلَ مَا قَالَ ٱلْأَوَّلُونَ ﴿٨١﴾
(ئۇلار اﷲ نىڭ قۇدرىتىنى ئويلاپ كۆرمىدى) بەلكى ئۇلار بۇرۇنقىلار نېمىدېگەن بولسا شۇنى دېدى
قَالُوٓا۟ أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ﴿٨٢﴾
ئۇلار ئېيتتى: «بىز ئۆلۈپ توپىغا ۋە (ئۇۋۇلۇپ كەتكەن) سۆڭەككە ئايلانغاندىن كېيىن، چوقۇم قايتا تىرىلەمدۇق؟
لَقَدْ وُعِدْنَا نَحْنُ وَءَابَآؤُنَا هَـٰذَا مِن قَبْلُ إِنْ هَـٰذَآ إِلَّآ أَسَـٰطِيرُ ٱلْأَوَّلِينَ ﴿٨٣﴾
ئىشەنچلىك ھالدا بىز ۋە بىزنىڭ ئاتا - بوۋىلىرىمىز ئىلگىرى مۇشۇنداق ئاگاھلاندۇرۇلغان ئىدۇق، بۇ پەقەت قەدىمكىلەردىن قالغان ئەپسانىلاردۇر»
قُل لِّمَنِ ٱلْأَرْضُ وَمَن فِيهَآ إِن كُنتُمْ تَعْلَمُونَ ﴿٨٤﴾
ئېيىتقىنكى، «زېمىن ۋە ئۇنىڭدىكى مەخلۇقاتلار كىمنىڭ؟ ئەگەر سىلەر بىلسەڭلار (ماڭا بۇنى ئېيتىپ بېرىڭلار)»
سَيَقُولُونَ لِلَّهِ ۚ قُلْ أَفَلَا تَذَكَّرُونَ ﴿٨٥﴾
ئۇلار: «اﷲ نىڭ» دەيدۇ. (بۇنىڭدىن) پەند - نەسىھەت ئالمامسىلەر؟
قُلْ مَن رَّبُّ ٱلسَّمَـٰوَٰتِ ٱلسَّبْعِ وَرَبُّ ٱلْعَرْشِ ٱلْعَظِيمِ ﴿٨٦﴾
ئېيىتقىنكى، «يەتتە ئاسماننىڭ ۋە ئۇلۇغ ئەرشنىڭ پەرۋەردىگارى كىم؟»
سَيَقُولُونَ لِلَّهِ ۚ قُلْ أَفَلَا تَتَّقُونَ ﴿٨٧﴾
ئۇلار «اﷲ» دەيدۇ. ئېيىتقىنكى، «ئۇنىڭ (ئازابىدىن) قورقمامسىلەر؟»
قُلْ مَنۢ بِيَدِهِۦ مَلَكُوتُ كُلِّ شَىْءٍ وَهُوَ يُجِيرُ وَلَا يُجَارُ عَلَيْهِ إِن كُنتُمْ تَعْلَمُونَ ﴿٨٨﴾
ئېيىتقىنكى، «ھەممە شەيئىنىڭ پادىشاھلىقى كىمنىڭ قولىدا؟ (ئىلتىجا قىلغانلارغا) پاناھ بولالايدىغان ۋە ئۇنىڭغا قارشى ھېچ نەرسە پاناھ بولالمايدىغان كىم؟ ئەگەر سىلەر بىلسەڭلار (ماڭا بۇنى ئېيتىپ بېرىڭلار)»
سَيَقُولُونَ لِلَّهِ ۚ قُلْ فَأَنَّىٰ تُسْحَرُونَ ﴿٨٩﴾
ئۇلار: «اﷲ» دەيدۇ. ئېيىتقىنكى، «سىلەر قانداقمۇ قايمۇقتۇرۇلىسىلەر؟»
بَلْ أَتَيْنَـٰهُم بِٱلْحَقِّ وَإِنَّهُمْ لَكَـٰذِبُونَ ﴿٩٠﴾
ئۇنداق ئەمەس، ئۇلارغا بىز ھەق سۆزنى ئېلىپ كەلدۇق، ئۇلار چوقۇم يالغانچىلاردۇر
مَا ٱتَّخَذَ ٱللَّهُ مِن وَلَدٍ وَمَا كَانَ مَعَهُۥ مِنْ إِلَـٰهٍ ۚ إِذًا لَّذَهَبَ كُلُّ إِلَـٰهٍۭ بِمَا خَلَقَ وَلَعَلَا بَعْضُهُمْ عَلَىٰ بَعْضٍ ۚ سُبْحَـٰنَ ٱللَّهِ عَمَّا يَصِفُونَ ﴿٩١﴾
اﷲ نىڭ بالىسى يوق، ئۇنىڭغا باراۋەر باشقا بىر ئىلاھمۇ يوق، ئەگەر بۇنداق بولسا ئىدى، ئۇ چاغدا ھەر ئىلاھ ئۆزى ياراتقان مەخلۇقنى يالغۇز ئىگىلەيتتى، بەزىسى بەزىسىدىن غالىب كېلەتتى، اﷲ ئۇلارنىڭ اﷲ قا نىسبەت بەرگەن نەرسىلىرىدىن پاكتۇر
عَـٰلِمِ ٱلْغَيْبِ وَٱلشَّهَـٰدَةِ فَتَعَـٰلَىٰ عَمَّا يُشْرِكُونَ ﴿٩٢﴾
اﷲ غەيبنى ۋە ئاشكارىنى بىلگۈچىدۇر، مۇشرىكلارنىڭ شېرىك كەلتۈرگەن نەرسىلىرىدىن ئۈستۈندۇر
قُل رَّبِّ إِمَّا تُرِيَنِّى مَا يُوعَدُونَ ﴿٩٣﴾
ئېيتقىنكى، «پەرۋەردىگارىم! ئەگەر ماڭا ئۇلارغا (دۇنيادا) ۋەدە قىلغان ئازابنى كۆرسىتىدىغان بولساڭ
رَبِّ فَلَا تَجْعَلْنِى فِى ٱلْقَوْمِ ٱلظَّـٰلِمِينَ ﴿٩٤﴾
ئېيتقىنكى: «پەرۋەردىگارىم! مېنى زالىم قەۋملەر ئىچىدە قىلمىغىن»
وَإِنَّا عَلَىٰٓ أَن نُّرِيَكَ مَا نَعِدُهُمْ لَقَـٰدِرُونَ ﴿٩٥﴾
بىز ئۇلارغا ۋەدە قىلغان ئازابنى ساڭا كۆرسىتىشكە ئەلۋەتتە قادىرمىز
ٱدْفَعْ بِٱلَّتِى هِىَ أَحْسَنُ ٱلسَّيِّئَةَ ۚ نَحْنُ أَعْلَمُ بِمَا يَصِفُونَ ﴿٩٦﴾
(ئۇلارنىڭ قىلغان) يامانلىقىغا (ئەپۇ قىلىش بىلەن) ياخشىلىق قىلغىن، ئۇلارنىڭ (سېنى) سۈپەتلەشلىرىنى (يەنى ئۇلارنىڭ ساڭا قىلغان مەسخىرىسىنى) بىز ئوبدان بىلىمىز
وَقُل رَّبِّ أَعُوذُ بِكَ مِنْ هَمَزَٰتِ ٱلشَّيَـٰطِينِ ﴿٩٧﴾
ئېيتقىنكى، «پەرۋەردىگارىم! ساڭا سېغىنىپ شەيتان (ئازدۇرغۇچى)لارنىڭ ۋەسۋەسىلىرىدىن پاناھ تىلەيمەن
وَأَعُوذُ بِكَ رَبِّ أَن يَحْضُرُونِ ﴿٩٨﴾
پەرۋەردىگارىم! ساڭا سېغىنىپ شەيتان (ئازدۇرغۇچى)لارنىڭ ماڭا ھەمراھ بولۇشىدىن پاناھ تىلەيمەن»
حَتَّىٰٓ إِذَا جَآءَ أَحَدَهُمُ ٱلْمَوْتُ قَالَ رَبِّ ٱرْجِعُونِ ﴿٩٩﴾
ئۇلارنىڭ (يەنى ھەقىقەتنى رەت قىلغۇچىلارنىڭ) بىرىگە ئۆلۈم كەلسە ئېيتىدۇركى، «پەرۋەردىگارىم! مېنى (دۇنياغا) قايتۇرغىن
لَعَلِّىٓ أَعْمَلُ صَـٰلِحًا فِيمَا تَرَكْتُ ۚ كَلَّآ ۚ إِنَّهَا كَلِمَةٌ هُوَ قَآئِلُهَا ۖ وَمِن وَرَآئِهِم بَرْزَخٌ إِلَىٰ يَوْمِ يُبْعَثُونَ ﴿١٠٠﴾

فَإِذَا نُفِخَ فِى ٱلصُّورِ فَلَآ أَنسَابَ بَيْنَهُمْ يَوْمَئِذٍ وَلَا يَتَسَآءَلُونَ ﴿١٠١﴾

تارازىسى يېنىك كەلگەنلەر (يەنى يامانلىقلىرى ياخشىلىقلىرىنى بېسىپ كەتكەنلەر) زىيان تارتقۇچىلاردۇر، مەڭگۈلۈك ئوتدە مەڭگۈ قالغۇچىلاردۇر
تَلْفَحُ وُجُوهَهُمُ ٱلنَّارُ وَهُمْ فِيهَا كَـٰلِحُونَ ﴿١٠٤﴾
ئۇلارنىڭ يۈزلىرىنى ئوت كۆيدۈرىدۇ، كالپۇكلىرى قورۇلۇپ، چىشلىرى ئېچىلىپ (بەت چىراي) بولۇپ قالىدۇ
أَلَمْ تَكُنْ ءَايَـٰتِى تُتْلَىٰ عَلَيْكُمْ فَكُنتُم بِهَا تُكَذِّبُونَ ﴿١٠٥﴾
(ئۇلارغا) «سىلەرگە (دۇنيادا) مېنىڭ ئايەتلىرىم ئوقۇپ بېرىلگەن، سىلەر ئۇنى (شۇنچە روشەن تۇرۇقلۇق) ئىنكار قىلغان ئەمەسمىدىڭلار» دېيىلىدۇ
قَالُوا۟ رَبَّنَا غَلَبَتْ عَلَيْنَا شِقْوَتُنَا وَكُنَّا قَوْمًا ضَآلِّينَ ﴿١٠٦﴾
ئۇلار ئېيتىدۇ: «پەرۋەردىگارىمىز! بەتبەختلىكىمىز ئۈستىمىزدىن غالىب كېلىپ گۇمراھ قەۋم بولۇپ قالغان ئىدۇق
رَبَّنَآ أَخْرِجْنَا مِنْهَا فَإِنْ عُدْنَا فَإِنَّا ظَـٰلِمُونَ ﴿١٠٧﴾
پەرۋەردىگارىمىز! بىزنى دوزاختىن چىقارغىن (ئاندىن دۇنياغا قايتۇرغىن)، ئەگەر بىز (شۇنىڭدىن كېيىن) يەنە گۇناھ قىلساق، بىز ئىشەنچلىك ھالدا زالىم بولۇپ قالىمىز»
قَالَ ٱخْسَـُٔوا۟ فِيهَا وَلَا تُكَلِّمُونِ ﴿١٠٨﴾
اﷲ ئېيتىدۇ: «مەڭگۈلۈك ئوتدە خار ھالدا قىلىڭلار، (ئۈستۈڭلاردىن ئازابنىڭ كۆتۈرۈلۈپ كېتىشى ھەققىدە) ماڭا سۆز ئاچماڭلار»
إِنَّهُۥ كَانَ فَرِيقٌ مِّنْ عِبَادِى يَقُولُونَ رَبَّنَآ ءَامَنَّا فَٱغْفِرْ لَنَا وَٱرْحَمْنَا وَأَنتَ خَيْرُ ٱلرَّٰحِمِينَ ﴿١٠٩﴾
شۈبھىسىزكى، بەندىلىرىمدىن بىر تۈركۈم كىشىلەر: «پەرۋەردىگارىمىز! بىز ئىمان ئېيتتۇق، بىزگە مەغپىرەت قىلغىن، بىزگە رەھىم قىلغىن، سەن بولساڭ ئەڭ رەھىم قىلغۇچىسەن» دەيتتى
فَٱتَّخَذْتُمُوهُمْ سِخْرِيًّا حَتَّىٰٓ أَنسَوْكُمْ ذِكْرِى وَكُنتُم مِّنْهُمْ تَضْحَكُونَ ﴿١١٠﴾
ئۇلارنى مەسخىرە قىلدىڭلار، ھەتتا ئۇلار سىلەرگە مېنىڭ زىكرىمنى ئۇنتۇلدۇردى (يەنى ئۇلارنى مەسخىرە قىلىش بىلەن بولۇپ كېتىپ ماڭا ئىتائەت قىلىشنى ئۇنتۇدۇڭلار)، سىلەر ئۇلاردىن كۈلەتتىڭلار
إِنِّى جَزَيْتُهُمُ ٱلْيَوْمَ بِمَا صَبَرُوٓا۟ أَنَّهُمْ هُمُ ٱلْفَآئِزُونَ ﴿١١١﴾
(سىلەردىن يەتكەن ئەرزىيەتلەرگە) سەۋر قىلغانلىقلىرى ئۈچۈن، مەن ئۇلارنى بۈگۈن مۇكاپاتلىدىم، ئۇلار ئىشەنچلىك ھالدا مەقسەتكە ئېرىشكۇچىلەردۇر
قَـٰلَ كَمْ لَبِثْتُمْ فِى ٱلْأَرْضِ عَدَدَ سِنِينَ ﴿١١٢﴾
اﷲ (ئۇلارغا): «زېمىندا قانچىلىك تۇردۇڭلار؟» دەيدۇ
قَالُوا۟ لَبِثْنَا يَوْمًا أَوْ بَعْضَ يَوْمٍ فَسْـَٔلِ ٱلْعَآدِّينَ ﴿١١٣﴾
ئۇلار: «بىر كۈن ياكى بىر كۈندىنمۇ ئاز تۇردۇق، (ئاي - كۈنلەرنى) ھېسابلىغۇچى (پەرىشتە) دىن سورىغىن» دەيدۇ (ئۇلار دوزاختا چېكىۋاتقان ئازاب تۈپەيلىدىن، زېمىندا قانچىلىك تۇرغانلىقىنى ئۇنتۇيدۇ)
قَـٰلَ إِن لَّبِثْتُمْ إِلَّا قَلِيلًا ۖ لَّوْ أَنَّكُمْ كُنتُمْ تَعْلَمُونَ ﴿١١٤﴾
اﷲ: «ئەگەر سىلەر بىلسەڭلار (دۇنيادا) پەقەت ئازغىنا تۇردۇڭلار» دەيدۇ
أَفَحَسِبْتُمْ أَنَّمَا خَلَقْنَـٰكُمْ عَبَثًا وَأَنَّكُمْ إِلَيْنَا لَا تُرْجَعُونَ ﴿١١٥﴾
ئى ئىنسانلار! بىزنى سىلەر ئۆزۈڭلارنى بىكار ياراتتى (يەنى سىلەر بىزنى ھايۋاناتلارنى ياراتقىنىمىزدەك ھېچقانداق ساۋابمۇ بەرمەيدىغان، جازامۇ بەرمەيدىغان قىلىپ ياراتتى). ئۆزۈڭلارنى بىزنىڭ دەرگاھىمىزغا قايتۇرۇلمايدۇ دەپ ئويلامسىلەر؟
فَتَعَـٰلَى ٱللَّهُ ٱلْمَلِكُ ٱلْحَقُّ ۖ لَآ إِلَـٰهَ إِلَّا هُوَ رَبُّ ٱلْعَرْشِ ٱلْكَرِيمِ ﴿١١٦﴾
ھەق - پادىشاھ اﷲ ئۈستۈندۇر، ئۇنىڭدىن باشقا ھېچ ئىلاھ يوقتۇر، اﷲ ئۇلۇغ ئەرشنىڭ پەرۋەردىگارىدۇر
وَمَن يَدْعُ مَعَ ٱللَّهِ إِلَـٰهًا ءَاخَرَ لَا بُرْهَـٰنَ لَهُۥ بِهِۦ فَإِنَّمَا حِسَابُهُۥ عِندَ رَبِّهِۦٓ ۚ إِنَّهُۥ لَا يُفْلِحُ ٱلْكَـٰفِرُونَ ﴿١١٧﴾
كىمكى ھېچقانداق دەلىلى بولمىغان ھالدا اﷲ قا قوشۇپ يەنە بىر مەبۇدقا ئىبادەت قىلىدىكەن، پەرۋەردىگارىنىڭ دەرگاھىدا ھېساب بېرىدۇ، ھەقىقەتنى رەت قىلغۇچىلار ئىشەنچلىك ھالدا نىجات تاپمايدۇ
وَقُل رَّبِّ ٱغْفِرْ وَٱرْحَمْ وَأَنتَ خَيْرُ ٱلرَّٰحِمِينَ ﴿١١٨﴾
«پەرۋەردىگارىم! مەغپىرەت قىلغىن، رەھىم قىلغىن، سەن رەھىم قىلغۇچىلارنىڭ ياخشىسىسەن» دېگىن
