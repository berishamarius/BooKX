حا، مىم
وَالْكِتَابِ الْمُبِينِ ﴿٢﴾
روشەن مۇقەددەس كىتاب-ۋەھىي بىلەن قەسەمكى، سىلەرنى (ئۇنىڭ مەنىسىنى) چۈشەنسۇن دەپ، بىز ئۇنى ئىشەنچلىك ھالدا ئەرەبچە قۇرئان قىلدۇق
إِنَّا جَعَلْنَاهُ قُرْآنًا عَرَبِيًّا لَّعَلَّكُمْ تَعْقِلُونَ ﴿٣﴾
روشەن مۇقەددەس كىتاب-ۋەھىي بىلەن قەسەمكى، سىلەرنى (ئۇنىڭ مەنىسىنى) چۈشەنسۇن دەپ، بىز ئۇنى ئىشەنچلىك ھالدا ئەرەبچە قۇرئان قىلدۇق
وَإِنَّهُ فِي أُمِّ الْكِتَابِ لَدَيْنَا لَعَلِيٌّ حَكِيمٌ ﴿٤﴾
ئىشەنچلىك ھالدا ئۇ لەۋھۇلمەھپۇزدا بىزنىڭ دەرگاھىمىزدا (مۇقىملاشقان) يۇقىرى مەرتىۋىلىك ھېكمەتلىك (مۇقەددەس كىتاب-ۋەھىي) دۇر
أَفَنَضْرِبُ عَنكُمُ الذِّكْرَ صَفْحًا أَن كُنتُمْ قَوْمًا مُّسْرِفِينَ ﴿٥﴾
سىلەر (گۇمراھلىقتا) ھەددىدىن ئاشقان قەۋم بولغانلىقىڭلار ئۈچۈن سىلەرگە ۋەز - نەسىھەت قىلىشنى تەرك قىلامدۇق؟
وَكَمْ أَرْسَلْنَا مِن نَّبِيٍّ فِي الْأَوَّلِينَ ﴿٦﴾
ئىلگىرىكى ئۈممەتلەرگە نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتتۇق
وَمَا يَأْتِيهِم مِّن نَّبِيٍّ إِلَّا كَانُوا بِهِ يَسْتَهْزِئُونَ ﴿٧﴾
ئۇلارغا قانداق بىر خۇدانىڭ ئەلچىسى كەلمىسۇن، ئۇلار ئۇنى (قەۋمىڭ سېنى مەسخىرە قىلغاندەك) مەسخىرە قىلدى
فَأَهْلَكْنَا أَشَدَّ مِنْهُم بَطْشًا وَمَضَىٰ مَثَلُ الْأَوَّلِينَ ﴿٨﴾
ئۇلار (يەنى مەككە ھەقىقەتنى رەت قىلغۇچىلىرى) دىن كۈچ - قۇۋۋەتتە ئۈستۈن بولغانلارنى ھالاك قىلدۇق، ئىلگىرىكىلەرنىڭ قىسسىلىرى قۇرئاندا ئۆتتى
وَلَئِن سَأَلْتَهُم مَّنْ خَلَقَ السَّمَاوَاتِ وَالْأَرْضَ لَيَقُولُنَّ خَلَقَهُنَّ الْعَزِيزُ الْعَلِيمُ ﴿٩﴾
ئەگەر سەن ئۇلاردىن: «ئاسمانلار ۋە زېمىننى كىم ياراتتى» دەپ سورىساڭ، ئۇلار: «ئەلۋەتتە ئاسمانلارنى ۋە زېمىننى غالىب، ھەممىنى بىلگۈچى (اﷲ) ياراتتى» دەيدۇ
الَّذِي جَعَلَ لَكُمُ الْأَرْضَ مَهْدًا وَجَعَلَ لَكُمْ فِيهَا سُبُلًا لَّعَلَّكُمْ تَهْتَدُونَ ﴿١٠﴾
اﷲ سىلەر ئۈچۈن زېمىننى قارارگاھ قىلدى، (سەپەردە كۆزلىگەن يېرىڭلارغا) بېرىۋېلىشىڭلار ئۈچۈن سىلەرگە زېمىندا نۇرغۇن يوللارنى ياراتتى
وَالَّذِي نَزَّلَ مِنَ السَّمَاءِ مَاءً بِقَدَرٍ فَأَنشَرْنَا بِهِ بَلْدَةً مَّيْتًا ۚ كَذَٰلِكَ تُخْرَجُونَ ﴿١١﴾
اﷲ بۇلۇتتىن (سىلەرگە زىيان يەتكۈزمەي، پايدا يەتكۈزىدىغان) ئۆلچەمدە يامغۇر ياغدۇردى، ئۇنىڭ بىلەن بىز (گىيا ئۈنمەس قاقاس) ئۆلۈك زېمىننى تىرىلدۈردۇق، سىلەر مۇشۇنداق (يەنى ئۆلۈك يەردىن ئۆسۈملۈك چىقىرىلغاندەك) (قەبرەڭلاردىن) چىقىرىلىسىلەر
وَالَّذِي خَلَقَ الْأَزْوَاجَ كُلَّهَا وَجَعَلَ لَكُم مِّنَ الْفُلْكِ وَالْأَنْعَامِ مَا تَرْكَبُونَ ﴿١٢﴾
اﷲ پۈتۈن مەخلۇقاتلارنىڭ تۈرلىرىنى ياراتتى، سىلەرگە ئۈستىگە چىقىشىڭلار ئۈچۈن كېمە ۋە مىنىشىڭلار ئۈچۈن ھايۋان قاتارلىقلارنى ياراتتى. ئاندىن سىلەر ئۇلارنىڭ ئۈستىگە چىققانلىرىڭلاردا پەرەۋردىگارىڭلارنىڭ نېمىتىنى ئەسلەپ: «بىزگە بۇنى بويسۇندۇرۇپ بەرگەن زات پاكتۇر، بىز ئۇنىڭغا (مىنىشكە) قادىر ئەمەس ئىدۇق، بىز ئىشەنچلىك ھالدا پەرۋەردىگارىمىزنىڭ دەرگاھىغا قايتقۇچىلارمىز» دېگەيسىلەر
لِتَسْتَوُوا عَلَىٰ ظُهُورِهِ ثُمَّ تَذْكُرُوا نِعْمَةَ رَبِّكُمْ إِذَا اسْتَوَيْتُمْ عَلَيْهِ وَتَقُولُوا سُبْحَانَ الَّذِي سَخَّرَ لَنَا هَٰذَا وَمَا كُنَّا لَهُ مُقْرِنِينَ ﴿١٣﴾
اﷲ پۈتۈن مەخلۇقاتلارنىڭ تۈرلىرىنى ياراتتى، سىلەرگە ئۈستىگە چىقىشىڭلار ئۈچۈن كېمە ۋە مىنىشىڭلار ئۈچۈن ھايۋان قاتارلىقلارنى ياراتتى. ئاندىن سىلەر ئۇلارنىڭ ئۈستىگە چىققانلىرىڭلاردا پەرەۋردىگارىڭلارنىڭ نېمىتىنى ئەسلەپ: «بىزگە بۇنى بويسۇندۇرۇپ بەرگەن زات پاكتۇر، بىز ئۇنىڭغا (مىنىشكە) قادىر ئەمەس ئىدۇق، بىز ئىشەنچلىك ھالدا پەرۋەردىگارىمىزنىڭ دەرگاھىغا قايتقۇچىلارمىز» دېگەيسىلەر
وَإِنَّا إِلَىٰ رَبِّنَا لَمُنقَلِبُونَ ﴿١٤﴾
اﷲ پۈتۈن مەخلۇقاتلارنىڭ تۈرلىرىنى ياراتتى، سىلەرگە ئۈستىگە چىقىشىڭلار ئۈچۈن كېمە ۋە مىنىشىڭلار ئۈچۈن ھايۋان قاتارلىقلارنى ياراتتى. ئاندىن سىلەر ئۇلارنىڭ ئۈستىگە چىققانلىرىڭلاردا پەرەۋردىگارىڭلارنىڭ نېمىتىنى ئەسلەپ: «بىزگە بۇنى بويسۇندۇرۇپ بەرگەن زات پاكتۇر، بىز ئۇنىڭغا (مىنىشكە) قادىر ئەمەس ئىدۇق، بىز ئىشەنچلىك ھالدا پەرۋەردىگارىمىزنىڭ دەرگاھىغا قايتقۇچىلارمىز» دېگەيسىلەر
وَجَعَلُوا لَهُ مِنْ عِبَادِهِ جُزْءًا ۚ إِنَّ الْإِنسَانَ لَكَفُورٌ مُّبِينٌ ﴿١٥﴾
ئۇلار (خۇدانىڭ ئەلچى پەرىشتىلىرىنى اﷲ نىڭ قىزلىرى دېيىش بىلەن) اﷲ نىڭ بىر قىسىم بەندىلىرىنى اﷲ نىڭ بىر جۈزئىييى قىلىۋالدى. (مۇنداق دېگۈچى) ئىنسان ئىشەنچلىك ھالدا اﷲ قا ئاشكارا ناشۈكۈرلۈك قىلغۇچىدۇر
أَمِ اتَّخَذَ مِمَّا يَخْلُقُ بَنَاتٍ وَأَصْفَاكُم بِالْبَنِينَ ﴿١٦﴾
اﷲ مەخلۇقاتتىن ئۆزى ئۈچۈن قىزلارنى (بالا) تۇتۇپ، سىلەرنى ئوغۇللارغا خاس قىلدىمۇ؟
وَإِذَا بُشِّرَ أَحَدُهُم بِمَا ضَرَبَ لِلرَّحْمَٰنِ مَثَلًا ظَلَّ وَجْهُهُ مُسْوَدًّا وَهُوَ كَظِيمٌ ﴿١٧﴾
ئەگەر ئۇلارنىڭ بىرىگە مەرھەمەتلىك اﷲ قا مەسەل قىلغان نەرسە (يەنى قىز بالا) بىلەن بېشارەت بېرىلسە (يەنى خوتۇنىنىڭ قىز تۇغقانلىقى خەۋەر قىلىنسا)، قايغۇغا چۆمگەن ھالدا ئۇنىڭ يۈزى قارىداپ كېتىدۇ
أَوَمَن يُنَشَّأُ فِي الْحِلْيَةِ وَهُوَ فِي الْخِصَامِ غَيْرُ مُبِينٍ ﴿١٨﴾
(ئۇلار اﷲ قا) زىبۇزىننەت ئىچىدە چوڭ بولىدىغان، مۇنازىرىدە مەقسىتىنى ئوچۇق بايان قىلالمايدىغان (قىزلارنى مەنسۇپ قىلامدۇ؟)
وَجَعَلُوا الْمَلَائِكَةَ الَّذِينَ هُمْ عِبَادُ الرَّحْمَٰنِ إِنَاثًا ۚ أَشَهِدُوا خَلْقَهُمْ ۚ سَتُكْتَبُ شَهَادَتُهُمْ وَيُسْأَلُونَ ﴿١٩﴾
ئۇلار مەرھەمەتلىك اﷲ نىڭ بەندىلىرى بولغان خۇدانىڭ ئەلچى پەرىشتىلىرىگە قىز دەپ ئېتىقاد قىلدى، اﷲ خۇدانىڭ ئەلچى پەرىشتىلىرىنى ياراتقاندا، ئۇلار ئۈستىدە بارمىدى؟ (يەنى ئۇلار خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ قىز ئىكەنلىكىنى قەيەردىن بىلىدۇ؟) ئۇلارنىڭ گۇۋاھلىقى (ئۇلارنىڭ نامە - ئەمالىغا) يېزىلىدۇ، (قىيامەت كۈنى) ئۇلار سوراققا تارتىلىدۇ
وَقَالُوا لَوْ شَاءَ الرَّحْمَٰنُ مَا عَبَدْنَاهُم ۗ مَّا لَهُم بِذَٰلِكَ مِنْ عِلْمٍ ۖ إِنْ هُمْ إِلَّا يَخْرُصُونَ ﴿٢٠﴾
ئۇلار: «مەرھەمەتلىك اﷲ خالىسا، بىز ئۇلارغا (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرىگە) چوقۇنمايتتۇق» دېدى. بۇ توغرۇلۇق ئۇلارنىڭ بىلىمى (يەنى ئاساسى) يوق، ئۇلار پەقەت جۆيلۈيدۇ
أَمْ آتَيْنَاهُمْ كِتَابًا مِّن قَبْلِهِ فَهُم بِهِ مُسْتَمْسِكُونَ ﴿٢١﴾
ئۇلارغا بىز قۇرئاندىن ئىلگىرى (قانداقتۇر) بىر مۇقەددەس كىتاب-ۋەھىي بەرگەن بولۇپ، ئۇلار شۇ مۇقەددەس كىتاب-ۋەھىيقا مەھكەم ئىسىلامدۇ؟
بَلْ قَالُوا إِنَّا وَجَدْنَا آبَاءَنَا عَلَىٰ أُمَّةٍ وَإِنَّا عَلَىٰ آثَارِهِم مُّهْتَدُونَ ﴿٢٢﴾
ئۇنداق ئەمەس، ئۇلار: «ئىشەنچلىك ھالدا ئاتا - بوۋىلىرىمىزنىڭ بىر خىل دىنغا ئېتىقاد قىلغانلىقىنى بىلىمىز، بىز ئۇلارنىڭ ئىزىدىن مېڭىپ ھىدايەت تاپقۇچىدۇرمىز» دەيدۇ
وَكَذَٰلِكَ مَا أَرْسَلْنَا مِن قَبْلِكَ فِي قَرْيَةٍ مِّن نَّذِيرٍ إِلَّا قَالَ مُتْرَفُوهَا إِنَّا وَجَدْنَا آبَاءَنَا عَلَىٰ أُمَّةٍ وَإِنَّا عَلَىٰ آثَارِهِم مُّقْتَدُونَ ﴿٢٣﴾
شۇنىڭدەك سەندىن ئىلگىرى ھەر قاچان بىرەر شەھەرگە ئاگاھلاندۇرغۇچى (يەنى خۇدانىڭ ئەلچىسى) ئەۋەتسەكلا، ئۇنىڭ دۆلەتمەن ئادەملىرى: «بىز ئىشەنچلىك ھالدا ئاتا - بوۋىلىرىمىزنىڭ بىر خىل دىنغا ئېتىقاد قىلغانلىقىنى بىلىمىز، شەك - شۈبھىسىزكى، بىز ئۇلارنىڭ ئىزلىرىدىن ماڭىمىز» دېيىشتى
قَالَ أَوَلَوْ جِئْتُكُم بِأَهْدَىٰ مِمَّا وَجَدتُّمْ عَلَيْهِ آبَاءَكُمْ ۖ قَالُوا إِنَّا بِمَا أُرْسِلْتُم بِهِ كَافِرُونَ ﴿٢٤﴾
(ھەر خۇدانىڭ ئەلچىسى ئۆز قەۋمىگە) «ئاتا - بوۋاڭلار ئېتىقاد قىلىپ كەلگەن دىنغا قارىغاندا ئەڭ توغرا بىر دىننى كەلتۈرسەم يەنىلا ئاتا - بوۋاڭلارغا ئەگىشەمسىلەر؟» دېدى. ئۇلار: «سىلەر ئېلىپ كەلگەن دىنغا ئىشەنمەيمىز» دېيىشتى
فَانتَقَمْنَا مِنْهُمْ ۖ فَانظُرْ كَيْفَ كَانَ عَاقِبَةُ الْمُكَذِّبِينَ ﴿٢٥﴾
بىز ئۇلارنى جازالىدۇق، (توغرا يولنى) ئىنكار قىلغۇچىلارنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىغا قارىغىن
وَإِذْ قَالَ إِبْرَاهِيمُ لِأَبِيهِ وَقَوْمِهِ إِنَّنِي بَرَاءٌ مِّمَّا تَعْبُدُونَ ﴿٢٦﴾
ئۆز ۋاقتىدا ئىبراھىم ئاتىسىغا ۋە قەۋمىگە ئېيتتى: «سىلەر چوقۇنۇۋاتقان بۇتلاردىن مەن راستلا ئادا - جۇدامەن
إِلَّا الَّذِي فَطَرَنِي فَإِنَّهُ سَيَهْدِينِ ﴿٢٧﴾
پەقەت مېنى ياراتقان اﷲ مېنى (توغرا يولغا) باشلايدۇ»
وَجَعَلَهَا كَلِمَةً بَاقِيَةً فِي عَقِبِهِ لَعَلَّهُمْ يَرْجِعُونَ ﴿٢٨﴾
ئىبراھىم كەلىمە تەۋھىدنى ئەۋلادىدىن (شېرىك كەلتۈرگەنلەر) ئىمانغا قايتسۇن دەپ ئەۋلادى ئىچىدە قالدۇرۇپ كەتتى
بَلْ مَتَّعْتُ هَٰؤُلَاءِ وَآبَاءَهُمْ حَتَّىٰ جَاءَهُمُ الْحَقُّ وَرَسُولٌ مُّبِينٌ ﴿٢٩﴾
بەلكى مەن بۇلار (يەنى مەككە ئاھالىسى) نى ۋە بۇلارنىڭ ئاتا - بوۋىلىرىنى (ئۇزۇن ئۆمۈر ۋە نېمەتتىن) بەھرىمەن قىلدىم، تاكى ئۇلارغا ھەق (يەنى قۇرئان) ۋە روشەن خۇدانىڭ ئەلچىسى (مۇھەممەد ئەلەيھىسسالام) كەلگۈچە
وَلَمَّا جَاءَهُمُ الْحَقُّ قَالُوا هَٰذَا سِحْرٌ وَإِنَّا بِهِ كَافِرُونَ ﴿٣٠﴾
ئۇلارغا ھەق (يەنى قۇرئان) كەلگەندە، ئۇلار: «بۇ سېھىردۇر، بىز ئىشەنچلىك ھالدا ئۇنى ئىنكار قىلغۇچىمىز» دېيىشتى
وَقَالُوا لَوْلَا نُزِّلَ هَٰذَا الْقُرْآنُ عَلَىٰ رَجُلٍ مِّنَ الْقَرْيَتَيْنِ عَظِيمٍ ﴿٣١﴾
ئۇلار: «بۇ قۇرئان نېمىشقا ئىككى شەھەر (يەنى مەككە بىلەن تائىف ئادەملىرىدىن) بىر كاتتا ئادەمگە نازىل قىلىنمىدى؟» دېيىشتى
أَهُمْ يَقْسِمُونَ رَحْمَتَ رَبِّكَ ۚ نَحْنُ قَسَمْنَا بَيْنَهُم مَّعِيشَتَهُمْ فِي الْحَيَاةِ الدُّنْيَا ۚ وَرَفَعْنَا بَعْضَهُمْ فَوْقَ بَعْضٍ دَرَجَاتٍ لِّيَتَّخِذَ بَعْضُهُم بَعْضًا سُخْرِيًّا ۗ وَرَحْمَتُ رَبِّكَ خَيْرٌ مِّمَّا يَجْمَعُونَ ﴿٣٢﴾
پەرەۋردىگارىڭنىڭ رەھمىتىنى ئۇلار تەقسىم قىلىپ بېرەمدۇ؟ بىز ئۇلارنىڭ ھاياتىي دۇنيادىكى رىزقىنى ئۇلارنىڭ ئارىسىدا تەقسىم قىلدۇق، ئۇلارنىڭ بەزىسى بەزىسىنى (ھەق بېرىپ) ئىشقا سالسۇن دەپ، ئۇلارنىڭ بەزىسىنىڭ دەرىجىسىنى بەزىسىدىن ئۈستۈن قىلدۇق، پەرەۋردىگارىڭنىڭ رەھمىتى ئۇلارنىڭ توپلىغان نەرسىلىرى (يەنى پۇل - ماللىرى) دىن ياخشىدۇر
وَلَوْلَا أَن يَكُونَ النَّاسُ أُمَّةً وَاحِدَةً لَّجَعَلْنَا لِمَن يَكْفُرُ بِالرَّحْمَٰنِ لِبُيُوتِهِمْ سُقُفًا مِّن فِضَّةٍ وَمَعَارِجَ عَلَيْهَا يَظْهَرُونَ ﴿٣٣﴾
ئىنسانلار (كۇففارلارنىڭ باياشات تۇرمۇشىنى كۆرۈپ كۇفرىغا قىزىقىپ ئۇلارغا قوشۇلۇپ) ھەممىسى (كۇفرىدا) بىر ئۈممەت بولۇپ قالمايدىغان بولسا ئىدى، مەرھەمەتلىك اﷲ نى ئىنكار قىلغان كىشىلەرنىڭ ئۆيلىرىنىڭ ئۆگزىلىرىنى ۋە ئۇنىڭغا چىقىدىغان شوتىلىرىنى كۈمۈشتىن قىلىپ بېرەتتۇق
وَلِبُيُوتِهِمْ أَبْوَابًا وَسُرُرًا عَلَيْهَا يَتَّكِئُونَ ﴿٣٤﴾
ئۇلارنىڭ ئۆيلىرىنىڭ ئىشىكلىرىنى ۋە تەختلىرىنىمۇ كۈمۈشتىن قىلىپ بېرەتتۇق، ئۇلار تەختلەرگە يۆلىنىپ ئولتۇراتتى
وَزُخْرُفًا ۚ وَإِن كُلُّ ذَٰلِكَ لَمَّا مَتَاعُ الْحَيَاةِ الدُّنْيَا ۚ وَالْآخِرَةُ عِندَ رَبِّكَ لِلْمُتَّقِينَ ﴿٣٥﴾
(يۇقىرىقىلارنىڭ بەزىسىنى) ئالتۇندىن قىلىپ بېرەتتۇق، ئۇلارنىڭ ھەممىسى پەقەت دۇنيا تىرىكچىلىكىدە پايدىلىنىدىغان نەرسىلەردۇر. پەرەۋردىگارىڭنىڭ ھۇزۇرىدىكى ئاخىرەت (يەنى مەڭگۈلۈك باغلار ۋە ئۇنىڭدىكى نازۇنېمەتلەر) تەقۋادارلارغا خاستۇر
وَمَن يَعْشُ عَن ذِكْرِ الرَّحْمَٰنِ نُقَيِّضْ لَهُ شَيْطَانًا فَهُوَ لَهُ قَرِينٌ ﴿٣٦﴾
كىمكى مېھرىبان اﷲ نى ياد ئېتىشتىن (يەنى قۇرئاندىن) يۈز ئۆرۈيدىكەن، بىز ئۇنىڭغا شەيتان (ئازدۇرغۇچى)نى مۇسەللەت قىلىمىز (شەيتان (ئازدۇرغۇچى) ئۇنى ۋەسۋەسە قىلىدۇ)، شەيتان (ئازدۇرغۇچى) ئۇنىڭغا ھەمىشە ھەمراھ بولىدۇ
وَإِنَّهُمْ لَيَصُدُّونَهُمْ عَنِ السَّبِيلِ وَيَحْسَبُونَ أَنَّهُم مُّهْتَدُونَ ﴿٣٧﴾
شەك - شۈبھىسىزكى، شەيتان (ئازدۇرغۇچى)لار ئۇلارنى (يەنى ئازغۇن كۇففارلارنى) توغرا يولدىن توسىدۇ، ئۇلار بولسا ئۆزلىرىنى توغرا يولدا دەپ گۇمان قىلىدۇ
حَتَّىٰ إِذَا جَاءَنَا قَالَ يَا لَيْتَ بَيْنِي وَبَيْنَكَ بُعْدَ الْمَشْرِقَيْنِ فَبِئْسَ الْقَرِينُ ﴿٣٨﴾
ھەقىقەتنى رەت قىلغۇچى ئادەم (ھەمراھى بىلەن قىيامەت كۈنى) دەرگاھىمىزغا كەلگەن چاغدا (ھەمراھىغا): «كاشكى سەن بىلەن مېنىڭ ئارامدا شەرق بىلەن غەربنىڭ ئارىسىدەك ئارىلىق بولسا ئىكەن. (سەن) نېمىدېگەن يامان ھەمراھ! (چۈنكى سەن مېنىڭ بەختسىزلىكىمگە سەۋەبچى بولدۇڭ)» دەيدۇ
وَلَن يَنفَعَكُمُ الْيَوْمَ إِذ ظَّلَمْتُمْ أَنَّكُمْ فِي الْعَذَابِ مُشْتَرِكُونَ ﴿٣٩﴾
(ئۇلارغا) «(دۇنيادىكى چاغدا) زۇلۇم قىلغانلىقىڭلار (يەنى شېرىك كەلتۈرگەنلىكىڭلار) ئۈچۈن بۈگۈن سىلەرنىڭ ئازابتا ئورتاق بولغانلىقىڭلار سىلەرگە پايدا بەرمەيدۇ» دېيىلىدۇ
أَفَأَنتَ تُسْمِعُ الصُّمَّ أَوْ تَهْدِي الْعُمْيَ وَمَن كَانَ فِي ضَلَالٍ مُّبِينٍ ﴿٤٠﴾
(ئى مۇھەممەد!) سەن گاسلارغا ئاڭلىتالامسەن؟ يا كورلارنى توغرا يولغا سالالامسەن؟ (يەنى كۇففارلار گاسلارغا، كورلارغا ئوخشايدۇ) ئاشكارا گۇمراھلىقتا بولغان كىشىنى ھىدايەت قىلالامسەن؟
فَإِمَّا نَذْهَبَنَّ بِكَ فَإِنَّا مِنْهُم مُّنتَقِمُونَ ﴿٤١﴾
(ئۇلارنى جازالاشتىن ئىلگىرى) سېنى ئېلىپ كەتسەك (يەنى ۋاپات قىلدۇرساق ۋاپاتىڭدىن كېيىن) ئۇلارنى چوقۇم جازالايمىز
أَوْ نُرِيَنَّكَ الَّذِي وَعَدْنَاهُمْ فَإِنَّا عَلَيْهِم مُّقْتَدِرُونَ ﴿٤٢﴾
ياكى بىز ساڭا (ھاياتىڭدا) ئۇلارغا ۋەدە قىلغان ئازابنى كۆرسىتىمىز، بىز، شۈبھىسىزكى، ئۇلارغا (يەنى ئۇلارغا ئازاب قىلىشقا) قادىرمىز
فَاسْتَمْسِكْ بِالَّذِي أُوحِيَ إِلَيْكَ ۖ إِنَّكَ عَلَىٰ صِرَاطٍ مُّسْتَقِيمٍ ﴿٤٣﴾
(ئى مۇھەممەد!) ساڭا ۋەھيى قىلىنغان قۇرئانغا چىڭ ئېسىلغىن، سەن ئىشەنچلىك ھالدا توغرا يولدىدۇرسەن
وَإِنَّهُ لَذِكْرٌ لَّكَ وَلِقَوْمِكَ ۖ وَسَوْفَ تُسْأَلُونَ ﴿٤٤﴾
قۇرئان ساڭا ۋە سېنىڭ قەۋمىڭگە ئەلۋەتتە (ئۇلۇغ) شەرەپتۇر، سىلەر كەلگۈسىدە (بۇ نېمەت ئۈستىدە) سورىلىسىلەر
وَاسْأَلْ مَنْ أَرْسَلْنَا مِن قَبْلِكَ مِن رُّسُلِنَا أَجَعَلْنَا مِن دُونِ الرَّحْمَٰنِ آلِهَةً يُعْبَدُونَ ﴿٤٥﴾
سەندىن بۇرۇن بىز ئەۋەتكەن خۇدانىڭ ئەلچىسىلىرىمىزدىن سوراپ باققىنكى، مېھرىبان اﷲ تىن باشقا ئىبادەت قىلىنىدىغان ئىلاھلارنى بەلگىلىدۇقمۇ؟ (خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ ئىچىدە اﷲ تىن غەيرىيگە ئىبادەت قىلىشقا دەۋەت قىلغىنى بارمۇ؟)
وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِآيَاتِنَا إِلَىٰ فِرْعَوْنَ وَمَلَئِهِ فَقَالَ إِنِّي رَسُولُ رَبِّ الْعَالَمِينَ ﴿٤٦﴾
بىز ئىشەنچلىك ھالدا مۇسانى پىرئەۋنگە ۋە ئۇنىڭ تەۋەلىرىگە (ئۇنىڭ راست خۇدانىڭ ئەلچىسى ئىكەنلىكىنى ئىسپاتلايدىغان شانلىق) مۆجىزىلىرىمىز بىلەن ئەۋەتتۇق، مۇسا (پىرئەۋنگە): «مەن ئىشەنچلىك ھالدا جىمى ئالەملەرنىڭ پەرۋەردىگارىنىڭ (سېنى ۋە قەۋمىڭنى بىر اﷲ قا ئىبادەت قىلىشقا دەۋەت قىلىش ئۈچۈن ئەۋەتكەن) ئەلچىسىمەن» دېدى
فَلَمَّا جَاءَهُم بِآيَاتِنَا إِذَا هُم مِّنْهَا يَضْحَكُونَ ﴿٤٧﴾
مۇسا ئۇلارغا بىزنىڭ مۆجىزىلىرىمىزنى ئېلىپ كەلگەندە، ئۇلار ناگاھان مۆجىزىلەرنى (مەسخىرە قىلىپ) كۈلدى
وَمَا نُرِيهِم مِّنْ آيَةٍ إِلَّا هِيَ أَكْبَرُ مِنْ أُخْتِهَا ۖ وَأَخَذْنَاهُم بِالْعَذَابِ لَعَلَّهُمْ يَرْجِعُونَ ﴿٤٨﴾
ئۇلارغا بىر - بىرىدىن كاتتا مۆجىزىلەرنى كۆرسەتتۇق، ئۇلارنى (كۇفرىدىن) قايتسۇن دەپ (تۈرلۈك) ئازابلار بىلەن جازالىدۇق
وَقَالُوا يَا أَيُّهَ السَّاحِرُ ادْعُ لَنَا رَبَّكَ بِمَا عَهِدَ عِندَكَ إِنَّنَا لَمُهْتَدُونَ ﴿٤٩﴾
(ئۇلار ئازابنى كۆرگەندە) «ئى سېھىرگەر! پەرەۋردىگارىڭغا ئۇنىڭ ساڭا بەرگەن (دۇئايىڭنى ئىجابەت قىلىشتىن ئىبارەت) ئەھدى بويىچە بىز ئۈچۈن دۇئا قىلساڭ (ئەگەر ئۇ دۇئايىڭ بىلەن بىزدىن ئازابنى كۆتۈرۈۋەتسە)، بىز چوقۇم ئىمان ئېيتىمىز» دېدى
فَلَمَّا كَشَفْنَا عَنْهُمُ الْعَذَابَ إِذَا هُمْ يَنكُثُونَ ﴿٥٠﴾
(مۇسانىڭ دۇئاسى بىلەن) ئۇلاردىن ئازابنى كۆتۈرۈۋەتكىنىمىزدە، ناگاھان ئۇلار (ئەھدىنى) بۇزۇپ (كۇفرىدا) چىڭ تۇردى
وَنَادَىٰ فِرْعَوْنُ فِي قَوْمِهِ قَالَ يَا قَوْمِ أَلَيْسَ لِي مُلْكُ مِصْرَ وَهَٰذِهِ الْأَنْهَارُ تَجْرِي مِن تَحْتِي ۖ أَفَلَا تُبْصِرُونَ ﴿٥١﴾
پىرئەۋن ئۆز قەۋمى ئىچىدە (پەخىرلەنگەن ھالدا) نىدا قىلىپ ئېيتتى: «ئى قەۋمىم! مىسىرنىڭ پادىشاھلىقى، ئاستىمدىن ئېقىپ تۇرغان بۇ دەريالار مېنىڭ ئەمەسمۇ؟ (مېنىڭ بۇ بۈيۈكلۈكۈمنى) كۆرمەمسىلەر؟ بەلكى مەن ئوچۇق گەپ قىلالمايدىغان بۇ ئەرزىمەس ئادەمدىن (يەنى مۇسا ئەلەيھىسسالامدىن) ياخشىمەن»
أَمْ أَنَا خَيْرٌ مِّنْ هَٰذَا الَّذِي هُوَ مَهِينٌ وَلَا يَكَادُ يُبِينُ ﴿٥٢﴾
پىرئەۋن ئۆز قەۋمى ئىچىدە (پەخىرلەنگەن ھالدا) نىدا قىلىپ ئېيتتى: «ئى قەۋمىم! مىسىرنىڭ پادىشاھلىقى، ئاستىمدىن ئېقىپ تۇرغان بۇ دەريالار مېنىڭ ئەمەسمۇ؟ (مېنىڭ بۇ بۈيۈكلۈكۈمنى) كۆرمەمسىلەر؟ بەلكى مەن ئوچۇق گەپ قىلالمايدىغان بۇ ئەرزىمەس ئادەمدىن (يەنى مۇسا ئەلەيھىسسالامدىن) ياخشىمەن»
فَلَوْلَا أُلْقِيَ عَلَيْهِ أَسْوِرَةٌ مِّن ذَهَبٍ أَوْ جَاءَ مَعَهُ الْمَلَائِكَةُ مُقْتَرِنِينَ ﴿٥٣﴾
نېمە ئۈچۈن ئۇنىڭغا (ئاسماندىن) ئالتۇن بىلەيزۈكلەر تاشلانمىدى؟ ياكى نېمىشقا (ئۇنىڭ راستلىقىغا گۇۋاھلىق بېرىش ئۈچۈن) ئۇنىڭ بىلەن بىللە خۇدانىڭ ئەلچى پەرىشتىلىرى ھەمراھ بولۇپ كەلمىدى؟»
فَاسْتَخَفَّ قَوْمَهُ فَأَطَاعُوهُ ۚ إِنَّهُمْ كَانُوا قَوْمًا فَاسِقِينَ ﴿٥٤﴾
پىرئەۋن قەۋمىنى گوللىغانلىقتىن، قەۋمى ئۇنىڭغا ئىتائەت قىلدى، ئۇلار ئىشەنچلىك ھالدا پاسىق قەۋم ئىدى
فَلَمَّا آسَفُونَا انتَقَمْنَا مِنْهُمْ فَأَغْرَقْنَاهُمْ أَجْمَعِينَ ﴿٥٥﴾
ئۇلار بىزنى دەرغەزەپ قىلغاندا، ئۇلارنى بىز (ئەڭ قاتتىق جازالار بىلەن) جازالىدۇق، ئۇلارنىڭ ھەممىسىنى (دېڭىزدا) غەرق قىلدۇق
فَجَعَلْنَاهُمْ سَلَفًا وَمَثَلًا لِّلْآخِرِينَ ﴿٥٦﴾
ئۇلارنى بىز (كۇففارلارنىڭ ئازابقا تېگىشلىك بولۇشىدا) كېيىنكىلەرگە نەمۇنە ۋە ئىبرەت قىلدۇق
وَلَمَّا ضُرِبَ ابْنُ مَرْيَمَ مَثَلًا إِذَا قَوْمُكَ مِنْهُ يَصِدُّونَ ﴿٥٧﴾
(قۇرئاندا) مەريەمنىڭ ئوغلى (يەنى ئىسا ئەلەيھىسسالام) مىسال كەلتۈرۈلسە، ناگاھان سېنىڭ قەۋمىڭ (خۇشاللىقتىن) سۈرەن سېلىشىپ كېتىدۇ
وَقَالُوا أَآلِهَتُنَا خَيْرٌ أَمْ هُوَ ۚ مَا ضَرَبُوهُ لَكَ إِلَّا جَدَلًا ۚ بَلْ هُمْ قَوْمٌ خَصِمُونَ ﴿٥٨﴾
ئۇلار: «بىزنىڭ ئىلاھلىرىمىز ياخشىمۇ ياكى ئىسامۇ؟» دېدى. ئۇلار بۇ مىسالنى پەقەت خۇسۇمەت قىلىش ئۈچۈنلا كەلتۈردى، بەلكى ئۇلار جېدەلخور قەۋمدۇر
إِنْ هُوَ إِلَّا عَبْدٌ أَنْعَمْنَا عَلَيْهِ وَجَعَلْنَاهُ مَثَلًا لِّبَنِي إِسْرَائِيلَ ﴿٥٩﴾
ئۇ (يەنى ئىسا) (ناسارالار گۇمان قىلغاندەك ئىلاھ ياكى ئىلاھنىڭ ئوغلى ئەمەس) پەقەت خۇدانىڭ ئەلچىسىلىك نېمىتىمىزگە ئېرىشكەن بىر بەندىدۇر، ئۇنى بىز ئىسرائىل ئەۋلادىغا (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) بىر دەلىل قىلدۇق
وَلَوْ نَشَاءُ لَجَعَلْنَا مِنكُم مَّلَائِكَةً فِي الْأَرْضِ يَخْلُفُونَ ﴿٦٠﴾
ئەگەر بىز خالىساق، ئەلۋەتتە سىلەرگە زېمىندا ئورۇنباسار بولىدىغان خۇدانىڭ ئەلچى پەرىشتىلىرىنى ۋۇجۇتقا كەلتۈرەتتۇق
وَإِنَّهُ لَعِلْمٌ لِّلسَّاعَةِ فَلَا تَمْتَرُنَّ بِهَا وَاتَّبِعُونِ ۚ هَٰذَا صِرَاطٌ مُّسْتَقِيمٌ ﴿٦١﴾
ئۇ (يەنى ئىسا) ئەلۋەتتە قىيامەت (يېقىنلاشقانلىقنىڭ) ئالامىتىدۇر، ئۇنىڭغا (يەنى قىيامەتكە) شەك كەلتۈرمەڭلار. (ئى مۇھەممەد! ئۇلارغا ئېيتقىنكى) «ماڭا ئەگىشىڭلار، بۇ (مېنىڭ دەۋەت قىلغىنىم) توغرا يولدۇر
وَلَا يَصُدَّنَّكُمُ الشَّيْطَانُ ۖ إِنَّهُ لَكُمْ عَدُوٌّ مُّبِينٌ ﴿٦٢﴾
سىلەرنى شەيتان (ئازدۇرغۇچى) (توغرا يولدا) مېڭىشتىن توسمىسۇن (يەنى شەيتان (ئازدۇرغۇچى)نىڭ ۋەسۋەسىسىگە ئالدانماڭلار)، شەيتان (ئازدۇرغۇچى) سىلەرگە ئىشەنچلىك ھالدا ئاشكارا دۈشمەندۇر»
وَلَمَّا جَاءَ عِيسَىٰ بِالْبَيِّنَاتِ قَالَ قَدْ جِئْتُكُم بِالْحِكْمَةِ وَلِأُبَيِّنَ لَكُم بَعْضَ الَّذِي تَخْتَلِفُونَ فِيهِ ۖ فَاتَّقُوا اللَّهَ وَأَطِيعُونِ ﴿٦٣﴾
ئىسا روشەن مۆجىزىلەر بىلەن كېلىپ ئېيتتى: «سىلەرگە مەن ھېكمەت ئېلىپ كەلدىم. سىلەرگە ئىختىلاپ قىلىۋاتقان نەرسەڭلارنىڭ (يەنى دىنىي ئىشلارنىڭ) بەزىسىنى بايان قىلىپ بېرىمەن، اﷲ تىن قورقۇڭلار، ماڭا ئىتائەت قىلىڭلار
إِنَّ اللَّهَ هُوَ رَبِّي وَرَبُّكُمْ فَاعْبُدُوهُ ۚ هَٰذَا صِرَاطٌ مُّسْتَقِيمٌ ﴿٦٤﴾
اﷲ ئىشەنچلىك ھالدا مېنىڭ پەرۋەردىگارىمدۇر ۋە سىلەرنىڭمۇ پەرۋەردىگارىڭلاردۇر، ئۇنىڭغا ئىبادەت قىلىڭلار، بۇ توغرا يولدۇر»
فَاخْتَلَفَ الْأَحْزَابُ مِن بَيْنِهِمْ ۖ فَوَيْلٌ لِّلَّذِينَ ظَلَمُوا مِنْ عَذَابِ يَوْمٍ أَلِيمٍ ﴿٦٥﴾
(ناسارالار) نىڭ پىرقىلىرى (ئىسا توغرىسىدا) ئۆز - ئارا ئىختىلاپ قىلىشتى، زالىملارغا قاتتىق كۈن (يەنى قىيامەت كۈنىنىڭ) ئازابىدىن ۋاي!
هَلْ يَنظُرُونَ إِلَّا السَّاعَةَ أَن تَأْتِيَهُم بَغْتَةً وَهُمْ لَا يَشْعُرُونَ ﴿٦٦﴾
ئۇلار پەقەت قىيامەتنىڭ ئۆزلىرى تۇيمىغان ھالدا ئۇشتۇمتۇت كېلىشىنىلا كۈتىدۇ
الْأَخِلَّاءُ يَوْمَئِذٍ بَعْضُهُمْ لِبَعْضٍ عَدُوٌّ إِلَّا الْمُتَّقِينَ ﴿٦٧﴾
بۇ كۈندە دوستلار بىر - بىرىگە دۈشمەن بولىدۇ، پەقەت (خۇدالىق ئۈچۈن دوستلاشقان) تەقۋادارلارلا (ئۇنداق ئەمەستۇر)
يَا عِبَادِ لَا خَوْفٌ عَلَيْكُمُ الْيَوْمَ وَلَا أَنتُمْ تَحْزَنُونَ ﴿٦٨﴾
(ئۇلارغا) «بەندىلىرىم! بۈگۈن سىلەرگە قورقۇنچ يوق، غەم - قايغۇمۇ يوق» (دېيىلىدۇ)
الَّذِينَ آمَنُوا بِآيَاتِنَا وَكَانُوا مُسْلِمِينَ ﴿٦٩﴾
ئۇلار بىزنىڭ ئايەتلىرىمىزگە ئىمان ئېيتقان ۋە مۇسۇلمان بولغانلار ئىدى
ادْخُلُوا الْجَنَّةَ أَنتُمْ وَأَزْوَاجُكُمْ تُحْبَرُونَ ﴿٧٠﴾
(ئۇلارغا) «سىلەر ئاياللىرىڭلار بىلەن بىللە خۇشال - خۇرام ھالدا مەڭگۈلۈك باغلاركە كىرىڭلار» (دېيىلىدۇ)
يُطَافُ عَلَيْهِم بِصِحَافٍ مِّن ذَهَبٍ وَأَكْوَابٍ ۖ وَفِيهَا مَا تَشْتَهِيهِ الْأَنفُسُ وَتَلَذُّ الْأَعْيُنُ ۖ وَأَنتُمْ فِيهَا خَالِدُونَ ﴿٧١﴾
ئۇلارغا ئالتۇن لېگەنلەردە (تائام)، ئالتۇن جاملاردا (شاراب) تۇتۇلىدۇ. مەڭگۈلۈك باغلارتە كۆڭۈللەر تارتىدىغان، كۆزلەر لەززەتلىنىدىغان نەرسىلەر بار، سىلەر مەڭگۈلۈك باغلارتە مەڭگۈ قالىسىلەر
وَتِلْكَ الْجَنَّةُ الَّتِي أُورِثْتُمُوهَا بِمَا كُنتُمْ تَعْمَلُونَ ﴿٧٢﴾
سىلەر (دۇنيادا) قىلغان (ياخشى) ئەمەلىڭلار بىلەن ۋارىس بولغان مەڭگۈلۈك باغلار ئەنە شۇدۇر
لَكُمْ فِيهَا فَاكِهَةٌ كَثِيرَةٌ مِّنْهَا تَأْكُلُونَ ﴿٧٣﴾
مەڭگۈلۈك باغلارتە نۇرغۇن (تۈرلۈك) مېۋىلەر بار، ئۇنىڭدىن يەيسىلەر
إِنَّ الْمُجْرِمِينَ فِي عَذَابِ جَهَنَّمَ خَالِدُونَ ﴿٧٤﴾
گۇناھكارلار (يەنى كۇففارلار) ئىشەنچلىك ھالدا دوزاخ ئازابىدا مەڭگۈ قالغۇچىلاردۇر
لَا يُفَتَّرُ عَنْهُمْ وَهُمْ فِيهِ مُبْلِسُونَ ﴿٧٥﴾
ئۇلاردىن ئازاب (بىردەممۇ) يېنىكلىتىلمەيدۇ، ئۇلار دوزاختا (ھەر قانداق ياخشىلىقتىن) ئۈمىدسىزلەردۇر
وَمَا ظَلَمْنَاهُمْ وَلَٰكِن كَانُوا هُمُ الظَّالِمِينَ ﴿٧٦﴾
ئۇلارغا بىز زۇلۇم قىلمىدۇق ۋە لېكىن ئۇلار ئۆزلىرىگە ئۆزلىرى زۇلۇم قىلدى
وَنَادَوْا يَا مَالِكُ لِيَقْضِ عَلَيْنَا رَبُّكَ ۖ قَالَ إِنَّكُم مَّاكِثُونَ ﴿٧٧﴾
ئۇلار (يەنى كۇففارلار) (دوزاخقا مۇئەككەل پەرىشتىگە): «ئى مالىك! پەرۋەردىگارىڭ بىزگە ئۆلۈم ھۆكۈم قىلسۇن» دەپ توۋلايدۇ. مالىك: «سىلەر ئازابتا چوقۇم قالىسىلەر» دەيدۇ
لَقَدْ جِئْنَاكُم بِالْحَقِّ وَلَٰكِنَّ أَكْثَرَكُمْ لِلْحَقِّ كَارِهُونَ ﴿٧٨﴾
بىز سىلەرگە ئىشەنچلىك ھالدا ھەق (دىن) نى كەلتۈردۇق ۋە لېكىن سىلەرنىڭ كۆپچىلىكىڭلار ھەق (دىن) نى يامان كۆرگۈچىلەردۇر
أَمْ أَبْرَمُوا أَمْرًا فَإِنَّا مُبْرِمُونَ ﴿٧٩﴾
ئۇلار (مۇھەممەد ئەلەيھىسسالامغا سۇيىقەست قىلىشتا) بىر ئىشنى قارار قىلغان بولسا، بىز (مۇھەممەد ئەلەيھىسسالامغا) ياردەم بېرىشتە بىر ئىشنى قارار قىلىمىز
أَمْ يَحْسَبُونَ أَنَّا لَا نَسْمَعُ سِرَّهُمْ وَنَجْوَاهُم ۚ بَلَىٰ وَرُسُلُنَا لَدَيْهِمْ يَكْتُبُونَ ﴿٨٠﴾
ئۇلار گۇمان قىلامدۇكى، بىز ئۇلارنىڭ سىرلىرىنى ۋە پىچىرلاشقانلىقلىرىنى ئاڭلىمايمىز، ھەرگىز ئۇنداق ئەمەس (ئۇنى ئاڭلاپ تۇرىمىز)، بىزنىڭ ئەلچىلىرىمىز (يەنى پەرىشتىلىرىمىز) ئۇلارنىڭ ئالدىدا (ئۇلارنىڭ ئەمەللىرىنى) يېزىپ تۇرىدۇ
قُلْ إِن كَانَ لِلرَّحْمَٰنِ وَلَدٌ فَأَنَا أَوَّلُ الْعَابِدِينَ ﴿٨١﴾
«ئەگەر مېھرىبان اﷲ نىڭ بالىسى بولىدىغان بولسا، مەن (ئۇ بالىغا) ئىبادەت قىلغۇچىلارنىڭ ئەۋۋىلى بولاتتىم» دېگىن
سُبْحَانَ رَبِّ السَّمَاوَاتِ وَالْأَرْضِ رَبِّ الْعَرْشِ عَمَّا يَصِفُونَ ﴿٨٢﴾
ئاسمانلارنىڭ ۋە زېمىننىڭ پەرۋەردىگارى، ئەرشنىڭ پەرۋەردىگارى ئۇلارنىڭ سۆزلىرىدىن پاكتۇر
فَذَرْهُمْ يَخُوضُوا وَيَلْعَبُوا حَتَّىٰ يُلَاقُوا يَوْمَهُمُ الَّذِي يُوعَدُونَ ﴿٨٣﴾
ئۇلارنى (يەنى مەككە كۇففارلىرىنى) (گۇمراھلىقلىرىدا ۋە نادانلىقلىرىدا) قويۇپ بەرگىن، ئۇلار ۋەدە قىلىنغان كۈنىگە مۇلاقات بولغانغا قەدەر (باتىللىرىغا) چۆمسۇن ۋە (دۇنياسى بىلەن) ئوينىسۇن
وَهُوَ الَّذِي فِي السَّمَاءِ إِلَٰهٌ وَفِي الْأَرْضِ إِلَٰهٌ ۚ وَهُوَ الْحَكِيمُ الْعَلِيمُ ﴿٨٤﴾
اﷲ ئاسماندىمۇ ئىلاھتۇر، زېمىندىمۇ ئىلاھتۇر، اﷲ ھېكمەت بىلەن ئىش قىلغۇچىدۇر، ھەممىنى بىلگۈچىدۇر
وَتَبَارَكَ الَّذِي لَهُ مُلْكُ السَّمَاوَاتِ وَالْأَرْضِ وَمَا بَيْنَهُمَا وَعِندَهُ عِلْمُ السَّاعَةِ وَإِلَيْهِ تُرْجَعُونَ ﴿٨٥﴾
