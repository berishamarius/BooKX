۞
محمد
Muhammad · Muhammad
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ أَضَلَّ أَعْمَـٰلَهُمْ ﴿١﴾
ھەقىقەتنى رەت قىلغۇچى بولغان ۋە (كىشىلەرنى) اﷲ نىڭ يولىدىن توسقانلارنىڭ (ياخشى) ئەمەللىرىنى اﷲ بىكار قىلىۋېتىدۇ
وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ وَءَامَنُوا۟ بِمَا نُزِّلَ عَلَىٰ مُحَمَّدٍ وَهُوَ ٱلْحَقُّ مِن رَّبِّهِمْ ۙ كَفَّرَ عَنْهُمْ سَيِّـَٔاتِهِمْ وَأَصْلَحَ بَالَهُمْ ﴿٢﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنىڭ، مۇھەممەدكە نازىل قىلىنغان قۇرئانغا - قۇرئان ئۇلارنىڭ پەرۋەردىگارى تەرىپىدىن كەلگەن ھەقىقەتتۇر – ئىمان ئېيتقانلارنىڭ گۇناھلىرىنى اﷲ ئۆچۈرۈپ تاشلايدۇ، ئۇلارنىڭ ھالىنى ياخشىلايدۇ (شۇنىڭ بىلەن ئۈلار گۇناھ قىلمايدۇ)
ذَٰلِكَ بِأَنَّ ٱلَّذِينَ كَفَرُوا۟ ٱتَّبَعُوا۟ ٱلْبَـٰطِلَ وَأَنَّ ٱلَّذِينَ ءَامَنُوا۟ ٱتَّبَعُوا۟ ٱلْحَقَّ مِن رَّبِّهِمْ ۚ كَذَٰلِكَ يَضْرِبُ ٱللَّهُ لِلنَّاسِ أَمْثَـٰلَهُمْ ﴿٣﴾
بۇ شۇنىڭ ئۈچۈندۇركى، ھەقىقەتنى رەت قىلغۇچىلار باتىلغا (يەنى شەيتان (ئازدۇرغۇچى)غا) ئەگەشتى، مۆمىنلەر بولسا، پەرۋەردىگارى تەرىپىدىن كەلگەن ھەقىقەتكە (يەنى قۇرئانغا) ئەگەشتى، اﷲ ئىنسانلارغا ئۇلارنىڭ ئەھۋالىنى شۇنداق بايان قىلىدۇ
فَإِذَا لَقِيتُمُ ٱلَّذِينَ كَفَرُوا۟ فَضَرْبَ ٱلرِّقَابِ حَتَّىٰٓ إِذَآ أَثْخَنتُمُوهُمْ فَشُدُّوا۟ ٱلْوَثَاقَ فَإِمَّا مَنًّۢا بَعْدُ وَإِمَّا فِدَآءً حَتَّىٰ تَضَعَ ٱلْحَرْبُ أَوْزَارَهَا ۚ ذَٰلِكَ وَلَوْ يَشَآءُ ٱللَّهُ لَٱنتَصَرَ مِنْهُمْ وَلَـٰكِن لِّيَبْلُوَا۟ بَعْضَكُم بِبَعْضٍ ۗ وَٱلَّذِينَ قُتِلُوا۟ فِى سَبِيلِ ٱللَّهِ فَلَن يُضِلَّ أَعْمَـٰلَهُمْ ﴿٤﴾
سىلەر كۇففارلارغا (جەڭ مەيدانىدا) ئۇچراشقان چېغىڭلاردا، ئۇلارنى ئۆلتۈرۈڭلار، سىلەر ئۇلارنى قىرىپ (مەغلۇپ قىلىپ، ئۇلاردا سىلەرگە قارشىلىق كۆرسىتىدىغان كۈچ قالمىغان چاغدا ئۇلارنى ئۆلتۈرۈشتىن توختاپ)، ئۇلارنى ئەسىر ئېلىڭلار، ئاندىن ئۇلارنى ئېھسان قىلىش يۈزىسىدىن قويۇپ بېرىڭلار، ياكى فىدىيە ئېلىپ قويۇپ بېرىڭلار، تاكى ئۇرۇش ئۆزىنىڭ ئېغىر يۈكىنى تاشلىغانغا قەدەر (يەنى ئۇرۇش توختىغانغا قەدەر) ئىش مانا شۇنداقتۇر. ئەگەر اﷲ خالىسا ئەلۋەتتە (سىلەرنى ئۇرۇشقا تەكلىپ قىلماستىنلا) ئۇلارنى جازالىغان بولاتتى، لېكىن اﷲ بەزىڭلارنى بەزىڭلار بىلەن سىناش ئۈچۈن (يەنى ئىمانىڭلارنى سىناش ئۈچۈن جىھادقا ئەمر قىلدى)، اﷲ نىڭ يولىدا ئۆلتۈرۈلگەنلەرنىڭ (يەنى شېھىتلەرنىڭ) ئەمەللىرىنى اﷲ بىكار قىلىۋەتمەيدۇ
سَيَهْدِيهِمْ وَيُصْلِحُ بَالَهُمْ ﴿٥﴾
اﷲ ئۇلارنى ھىدايەت قىلىدۇ ۋە ھالىنى ياخشىلايدۇ
وَيُدْخِلُهُمُ ٱلْجَنَّةَ عَرَّفَهَا لَهُمْ ﴿٦﴾
اﷲ ئۈلارنى مەڭگۈلۈك باغلاركە كىرگۈزىدۇ، اﷲ ئۇنى ئۇلارغا تونۇتتى (يەنى مەڭگۈلۈك باغلاركە كىرگەن ئادەم ئۆزىنىڭ ئۇ يەردىكى جايىنى ئىلگىرى كۆرگەندەكلا بىلىدۇ)
يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِن تَنصُرُوا۟ ٱللَّهَ يَنصُرْكُمْ وَيُثَبِّتْ أَقْدَامَكُمْ ﴿٧﴾
ئى مۆمىنلەر! سىلەر اﷲ قا (يەنى اﷲ نىڭ دىنىغا) ياردەم بەرسەڭلار، اﷲ سىلەرگە (دۈشمىنىڭلارغا قارشى ياردەم بېرىدۇ، قەدىمىڭلارنى (ئۇرۇش مەيدانلىرىدا) بەرقارار قىلىدۇ
وَٱلَّذِينَ كَفَرُوا۟ فَتَعْسًا لَّهُمْ وَأَضَلَّ أَعْمَـٰلَهُمْ ﴿٨﴾
ھەقىقەتنى رەت قىلغۇچىلارنى اﷲ ھالاك قىلىدۇ، ئۇلارنىڭ (ياخشى) ئەمەللىرىنى بىكار قىلىدۇ
ذَٰلِكَ بِأَنَّهُمْ كَرِهُوا۟ مَآ أَنزَلَ ٱللَّهُ فَأَحْبَطَ أَعْمَـٰلَهُمْ ﴿٩﴾
بۇ شۇنىڭ ئۈچۈندۇركى، ئۇلار اﷲ نازىل قىلغان مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) يامان كۆرۈش بىلەن، ئۆزلىرىنىڭ ئەمەللىرىنى بەربات قىلدى
۞ أَفَلَمْ يَسِيرُوا۟ فِى ٱلْأَرْضِ فَيَنظُرُوا۟ كَيْفَ كَانَ عَـٰقِبَةُ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ دَمَّرَ ٱللَّهُ عَلَيْهِمْ ۖ وَلِلْكَـٰفِرِينَ أَمْثَـٰلُهَا ﴿١٠﴾
ئۇلار زېمىندا سەير قىلىپ ئىلگىرىكىلەرنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۈزەتمىدىمۇ؟ اﷲ ئۇلارنى ھالاك قىلدى. (مەككە) ھەقىقەتنى رەت قىلغۇچىلىرىمۇ شۇنداق ئاقىۋەتكە قالىدۇ
ذَٰلِكَ بِأَنَّ ٱللَّهَ مَوْلَى ٱلَّذِينَ ءَامَنُوا۟ وَأَنَّ ٱلْكَـٰفِرِينَ لَا مَوْلَىٰ لَهُمْ ﴿١١﴾
بۇ شۇنىڭ ئۈچۈندۇركى، اﷲ مۆمىنلەرنىڭ مەدەتكارىدۇر، ھەقىقەتنى رەت قىلغۇچىلارغا مەدەتكار يوقتۇر
إِنَّ ٱللَّهَ يُدْخِلُ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ جَنَّـٰتٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ ۖ وَٱلَّذِينَ كَفَرُوا۟ يَتَمَتَّعُونَ وَيَأْكُلُونَ كَمَا تَأْكُلُ ٱلْأَنْعَـٰمُ وَٱلنَّارُ مَثْوًى لَّهُمْ ﴿١٢﴾
شۈبھىسىزكى، اﷲ ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنى ئاستىدىن ئۆستەڭلەر ئېقىپ تۇرىدىغان مەڭگۈلۈك باغلارلەرگە كىرگۈزىدۇ، ھەقىقەتنى رەت قىلغۇچىلار (ھاياتى دۇنيادا ئۇنىڭ لەززەتلىرىدىن) بەھرىمەن بولىدۇ ۋە چاھارپايلاردەك يەپ - ئىچىدۇ، (ئاخىرەتتە) ئۇلارنىڭ جايى دوزاخ بولىدۇ
وَكَأَيِّن مِّن قَرْيَةٍ هِىَ أَشَدُّ قُوَّةً مِّن قَرْيَتِكَ ٱلَّتِىٓ أَخْرَجَتْكَ أَهْلَكْنَـٰهُمْ فَلَا نَاصِرَ لَهُمْ ﴿١٣﴾
سېنى ھەيدەپ چىقارغان شەھەرنىڭ (يەنى مەككىنىڭ) ئاھالىسىدىن كۈچلۈك بولغان نۇرغۇن شەھەرلەرنىڭ ئاھالىسىنى ھالاك قىلدۇق، ئۇلارغا ھېچ قانداق ياردەم قىلغۇچى بولمىدى
أَفَمَن كَانَ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّهِۦ كَمَن زُيِّنَ لَهُۥ سُوٓءُ عَمَلِهِۦ وَٱتَّبَعُوٓا۟ أَهْوَآءَهُم ﴿١٤﴾
پەرۋەردىگارى تەرىپىدىن بولغان روشەن دەلىللەرگە ئاساسلانغان ئادەم (يەنى مۆمىن) ئۆزىنىڭ يامان ئەمەلى چىرايلىق كۆرسىتىلگەن ۋە نەپسى خاھىشلىرىغا ئەگەشكەن ئادەمگە (يەنى ھەقىقەتنى رەت قىلغۇچىغا) ئوخشامدۇ؟
مَّثَلُ ٱلْجَنَّةِ ٱلَّتِى وُعِدَ ٱلْمُتَّقُونَ ۖ فِيهَآ أَنْهَـٰرٌ مِّن مَّآءٍ غَيْرِ ءَاسِنٍ وَأَنْهَـٰرٌ مِّن لَّبَنٍ لَّمْ يَتَغَيَّرْ طَعْمُهُۥ وَأَنْهَـٰرٌ مِّنْ خَمْرٍ لَّذَّةٍ لِّلشَّـٰرِبِينَ وَأَنْهَـٰرٌ مِّنْ عَسَلٍ مُّصَفًّى ۖ وَلَهُمْ فِيهَا مِن كُلِّ ٱلثَّمَرَٰتِ وَمَغْفِرَةٌ مِّن رَّبِّهِمْ ۖ كَمَنْ هُوَ خَـٰلِدٌ فِى ٱلنَّارِ وَسُقُوا۟ مَآءً حَمِيمًا فَقَطَّعَ أَمْعَآءَهُمْ ﴿١٥﴾
تەقۋادارلارغا ۋەدە قىلىنغان مەڭگۈلۈك باغلارنىڭ سۈپىتى شۇكى، ئۇ يەردە رەڭگى ئۆزگەرمىگەن سۇدىن ئۆستەڭلار، تەمى ئۆزگەرمىگەن سۈتتىن ئۆستەڭلار، ئىچكۈچىلەرگە لەززەت بېغىشلايدىغان مەيدىن ئۆستەڭلار ۋە ساپ ھەسەلدىن ئۆستەڭلار بولىدۇ، ئۇلارغا مەڭگۈلۈك باغلارتە بەھرىمەن بولىدىغان تۈرلۈك مېۋىلەر بولىدۇ ۋە پەرۋەردىگارى تەرىپىدىن مەغپىرەت بولىدۇ، (مۇنداق تەقۋادارلار) دوزاختا مەڭگۈ قالىدىغان، قايناقسۇ بىلەن سۇغىرىلىپ (قىزىقلىقىدىن) ئۈچەيلىرى پارە - پارە قىلىنىدىغانلار بىلەن ئوخشاشمۇ؟
وَمِنْهُم مَّن يَسْتَمِعُ إِلَيْكَ حَتَّىٰٓ إِذَا خَرَجُوا۟ مِنْ عِندِكَ قَالُوا۟ لِلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ مَاذَا قَالَ ءَانِفًا ۚ أُو۟لَـٰٓئِكَ ٱلَّذِينَ طَبَعَ ٱللَّهُ عَلَىٰ قُلُوبِهِمْ وَٱتَّبَعُوٓا۟ أَهْوَآءَهُمْ ﴿١٦﴾
ئۇلاردىن (يەنى مۇناپىقلاردىن) بەزى كىشىلەر سېنىڭ سۆزۈڭگە قۇلاق سالىدۇ، ئۇلار سېنىڭ يېنىڭدىن چىققاندا، ئىلىم بېرىلگەن كىشىلەرگە (يەنى ساھابىلەرنىڭ ئۆلىمالىرىغا مەسخىرە قىلىش يۈزىسىدىن): «بايا ئۇ (يەنى مۇھەممەد) نېمە دېدى؟» دەيدۇ، ئەنە شۇلار اﷲ دىللىرىنى پېچەتلىۋەتكەنلەردۇر، نەپسى خاھىشلىرىغا ئەگەشكەنلەردۇر
وَٱلَّذِينَ ٱهْتَدَوْا۟ زَادَهُمْ هُدًى وَءَاتَىٰهُمْ تَقْوَىٰهُمْ ﴿١٧﴾
ھىدايەت تاپقانلارنى اﷲ تېخىمۇ ھىدايەت قىلىدۇ، ئۇلارغا تەقۋادارلىقنىڭ مۇكاپاتىنى بېرىدۇ
فَهَلْ يَنظُرُونَ إِلَّا ٱلسَّاعَةَ أَن تَأْتِيَهُم بَغْتَةً ۖ فَقَدْ جَآءَ أَشْرَاطُهَا ۚ فَأَنَّىٰ لَهُمْ إِذَا جَآءَتْهُمْ ذِكْرَىٰهُمْ ﴿١٨﴾
ئۇلار (يەنى مەككە كۇففارلىرى) پەقەت قىيامەتنىڭ ئۇلارغا تۇيۇقسىز كېلىشىنىلا كۈتىدۇ، شۈبھىسىزكى، قىيامەتنىڭ ئالامەتلىرى كەلدى، ئۇلارغا قىيامەت كەلگەندە ئۇلار قانداقمۇ ۋەز - نەسىھەت ئالىدۇ (يەنى بۇ چاغدا ۋەز - نەسىھەت ئالغانلىقىنىڭ پايدىسى بولمايدۇ
فَٱعْلَمْ أَنَّهُۥ لَآ إِلَـٰهَ إِلَّا ٱللَّهُ وَٱسْتَغْفِرْ لِذَنۢبِكَ وَلِلْمُؤْمِنِينَ وَٱلْمُؤْمِنَـٰتِ ۗ وَٱللَّهُ يَعْلَمُ مُتَقَلَّبَكُمْ وَمَثْوَىٰكُمْ ﴿١٩﴾
بىلگىنكى، اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، گۇناھىڭ ئۈچۈن، ئەر - ئايال مۆمىنلەر ئۈچۈن مەغپىرەت تىلىگىن، اﷲ سىلەرنىڭ (بۇ دۇنيادىكى) ھەرىكىتىڭلارنى ۋە (ئاخىرەتتىكى) جايىڭلارنى بىلىدۇ (شۇڭا ئاخىرەتلىك ئۈچۈن تەييارلىنىڭلار)
وَيَقُولُ ٱلَّذِينَ ءَامَنُوا۟ لَوْلَا نُزِّلَتْ سُورَةٌ ۖ فَإِذَآ أُنزِلَتْ سُورَةٌ مُّحْكَمَةٌ وَذُكِرَ فِيهَا ٱلْقِتَالُ ۙ رَأَيْتَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ يَنظُرُونَ إِلَيْكَ نَظَرَ ٱلْمَغْشِىِّ عَلَيْهِ مِنَ ٱلْمَوْتِ ۖ فَأَوْلَىٰ لَهُمْ ﴿٢٠﴾
مۆمىنلەر (جىھادنى تەلەپ قىلىش يۈزىسىدىن): «نېمىشقا (جىھاد ئەمرى قىلىنغان) بىرەر سۈرە نازىل قىلىنمىدى؟» دەيدۇ، جىھاد زىكرى قىلىنغان ئېنىق سۈرە نازىل قىلىنغاندا، دىللىرىدا كېسەل بارلار (يەنى مۇناپىقلارنىڭ) ساڭا ئۆلۈم ئالدىدا (ئۆلۈمدىن قورقۇپ) ئايلىنىپ كەتكەن ئادەمدەك قاراۋاتقانلىقىنى كۆرىسەن، (ئى مۇھەممەد!) (ئۇلارنىڭ ساڭا) ئىتائەت قىلىشى ۋە چىرايلىق سۆز قىلىشى ئۇلار ئۈچۈن ئەلۋەتتە ياخشى ئىدى، ئۇرۇش قارار قىلىنغان چاغدا، ئۇلار اﷲ قا سادىق بولۇشسا ئەلۋەتتە ئۇلار ئۈچۈن ياخشى ئىدى
طَاعَةٌ وَقَوْلٌ مَّعْرُوفٌ ۚ فَإِذَا عَزَمَ ٱلْأَمْرُ فَلَوْ صَدَقُوا۟ ٱللَّهَ لَكَانَ خَيْرًا لَّهُمْ ﴿٢١﴾
مۆمىنلەر (جىھادنى تەلەپ قىلىش يۈزىسىدىن): «نېمىشقا (جىھاد ئەمرى قىلىنغان) بىرەر سۈرە نازىل قىلىنمىدى؟» دەيدۇ، جىھاد زىكرى قىلىنغان ئېنىق سۈرە نازىل قىلىنغاندا، دىللىرىدا كېسەل بارلار (يەنى مۇناپىقلارنىڭ) ساڭا ئۆلۈم ئالدىدا (ئۆلۈمدىن قورقۇپ) ئايلىنىپ كەتكەن ئادەمدەك قاراۋاتقانلىقىنى كۆرىسەن، (ئى مۇھەممەد!) (ئۇلارنىڭ ساڭا) ئىتائەت قىلىشى ۋە چىرايلىق سۆز قىلىشى ئۇلار ئۈچۈن ئەلۋەتتە ياخشى ئىدى، ئۇرۇش قارار قىلىنغان چاغدا، ئۇلار اﷲ قا سادىق بولۇشسا ئەلۋەتتە ئۇلار ئۈچۈن ياخشى ئىدى
فَهَلْ عَسَيْتُمْ إِن تَوَلَّيْتُمْ أَن تُفْسِدُوا۟ فِى ٱلْأَرْضِ وَتُقَطِّعُوٓا۟ أَرْحَامَكُمْ ﴿٢٢﴾
سىلەر (ئىسلام) دىن يۈز ئۆرۈسەڭلار زېمىندا بۇزغۇنچىلىق قىلارسىلەرمۇ ۋە سىلە - رەھىمنى ئۈزۈپ قويارسىلەرمۇ؟
أُو۟لَـٰٓئِكَ ٱلَّذِينَ لَعَنَهُمُ ٱللَّهُ فَأَصَمَّهُمْ وَأَعْمَىٰٓ أَبْصَـٰرَهُمْ ﴿٢٣﴾
ئەنە شۇنداق كىشىلەرنى اﷲ رەھمىتىدىن يىراق قىلدى، ئۇلارنى (ھەقنى ئاڭلاشتىن) گاس قىلدى، (ھىدايەت يولىنى كۆرۈشتىن) كور قىلدى
أَفَلَا يَتَدَبَّرُونَ ٱلْقُرْءَانَ أَمْ عَلَىٰ قُلُوبٍ أَقْفَالُهَآ ﴿٢٤﴾
ئۇلار (ھەقنى تونۇش ئۈچۈن) قۇرئاننى پىكىر قىلمامدۇ؟ بەلكى ئۇلارنىڭ دىللىرىدا قۇلۇپ بار (شۇڭا ئۇلار چۈشەنمەيدۇ)
إِنَّ ٱلَّذِينَ ٱرْتَدُّوا۟ عَلَىٰٓ أَدْبَـٰرِهِم مِّنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَى ۙ ٱلشَّيْطَـٰنُ سَوَّلَ لَهُمْ وَأَمْلَىٰ لَهُمْ ﴿٢٥﴾
توغرا يول ئېنىق بولغاندىن كېيىن ئارقىسىغا قايتىۋالغانلار (يەنى ئىماندىن كۇفرىغا قايتقانلار) غا (ئۇلارنىڭ بۇ ئىشىنى) شەيتان (ئازدۇرغۇچى) چىرايلىق كۆرسەتتى، (شەيتان (ئازدۇرغۇچى) ئارزۇ ۋە ئەجەلنىڭ ئۇزۇنلۇقى بىلەن) ئۇلارنىڭ كۆڭلىنى تىندۇردى
ذَٰلِكَ بِأَنَّهُمْ قَالُوا۟ لِلَّذِينَ كَرِهُوا۟ مَا نَزَّلَ ٱللَّهُ سَنُطِيعُكُمْ فِى بَعْضِ ٱلْأَمْرِ ۖ وَٱللَّهُ يَعْلَمُ إِسْرَارَهُمْ ﴿٢٦﴾
بۇ (يەنى شەيتان (ئازدۇرغۇچى)نىڭ ئۇلارنى ئازدۇرۇشى) شۇنىڭ ئۈچۈندۇركى، ئۇلار اﷲ نازىل قىلغان نەرسىنى يامان كۆرگەن ھەقىقەتنى رەت قىلغۇچىلارغا: «بىز بەزى ئىشلاردا سىلەرگە ئىتائەت قىلىمىز» دېدى، اﷲ ئۇلارنىڭ سىرلىرىنى بىلىدۇ
فَكَيْفَ إِذَا تَوَفَّتْهُمُ ٱلْمَلَـٰٓئِكَةُ يَضْرِبُونَ وُجُوهَهُمْ وَأَدْبَـٰرَهُمْ ﴿٢٧﴾
(ئازاب) پەرىشتىلىرى ئۇلارنىڭ يۈزلىرىگە ۋە ئارقىلىرىغا ئۇرغان ھالدا ئۇلارنىڭ جانلىرىنى ئېلىۋاتقاندا (ئۇلارنىڭ ھالى) قانداق بولىدۇ؟
ذَٰلِكَ بِأَنَّهُمُ ٱتَّبَعُوا۟ مَآ أَسْخَطَ ٱللَّهَ وَكَرِهُوا۟ رِضْوَٰنَهُۥ فَأَحْبَطَ أَعْمَـٰلَهُمْ ﴿٢٨﴾
بۇ (ئازاب) شۇنىڭ ئۈچۈندۇركى، ئۇلار اﷲ نى دەرغەزەپ قىلىدىغان نەرسىلەرگە ئەگەشتى، اﷲ نىڭ رازىلىقىنى (يەنى اﷲ نى رازى قىلىدىغان ئىشلارنى) ياقتۇرمىدى، شۇنىڭ بىلەن اﷲ ئۇلارنىڭ ئەمەللىرىنى بىكار قىلدى
أَمْ حَسِبَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ أَن لَّن يُخْرِجَ ٱللَّهُ أَضْغَـٰنَهُمْ ﴿٢٩﴾
دىللىرىدا كېسەل بارلار (يەنى مۇناپىقلار) اﷲ ئۆزلىرىنىڭ (ئىسلامغا بولغان) دۈشمەنلىكىنى ئاشكارىلىمايدۇ دەپ ئويلامدۇ؟
وَلَوْ نَشَآءُ لَأَرَيْنَـٰكَهُمْ فَلَعَرَفْتَهُم بِسِيمَـٰهُمْ ۚ وَلَتَعْرِفَنَّهُمْ فِى لَحْنِ ٱلْقَوْلِ ۚ وَٱللَّهُ يَعْلَمُ أَعْمَـٰلَكُمْ ﴿٣٠﴾
ئەگەر بىز خالىساق، ئۇلارنى چوقۇم ساڭا بىلدۈرەتتۇق، چوقۇم ئۇلارنىڭ سىماسىدىن تونۇيتتۇڭ، سەن ئەلۋەتتە سۆزىنىڭ ئۇسلۇبىدىن ئۇلارنى تونۇيسەن، اﷲ سىلەرنىڭ ئەمەللىرىڭلارنى بىلىپ تۇرىدۇ
وَلَنَبْلُوَنَّكُمْ حَتَّىٰ نَعْلَمَ ٱلْمُجَـٰهِدِينَ مِنكُمْ وَٱلصَّـٰبِرِينَ وَنَبْلُوَا۟ أَخْبَارَكُمْ ﴿٣١﴾
بىز سىلەرنى ئەلۋەتتە (جىھادقا ئەمر قىلىش ۋە مۇشەققەتلىك ئىشلارغا تەكلىپ قىلىش بىلەن) سىنايمىز، تاكى سىلەرنىڭ ئىچىڭلاردىن (اﷲ نىڭ يولىدا) جىھاد قىلغۇچىلارنى ۋە (جىھادنىڭ مۇشەققەتلىرىگە) چىداشلىق بەرگۈچىلەرنى بىلگەنگە، سىرلىرىڭلاردىن ۋاقىپ بولغانغا قەدەر
إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ وَشَآقُّوا۟ ٱلرَّسُولَ مِنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَىٰ لَن يَضُرُّوا۟ ٱللَّهَ شَيْـًٔا وَسَيُحْبِطُ أَعْمَـٰلَهُمْ ﴿٣٢﴾
شۈبھىسىزكى، ھەقىقەتنى رەت قىلغۇچى بولغانلار، (كىشىلەرنى) اﷲ نىڭ يولىدىن توسقانلار، توغرا يول (يەنى رەسۇلۇللاھنىڭ ھەق ئىكەنلىكى) ئېنىق بولغاندىن كېيىن خۇدانىڭ ئەلچىسى بىلەن دۈشمەنلەشكەنلەر اﷲ قا قىلچە زىيان يەتكۈزەلمەيدۇ، اﷲ ئۇلارنىڭ (سەدىقە بەرگەنگە ئوخشاش) ئەمەللىرىنى بىكار قىلىدۇ (شۇنىڭ بىلەن ئاخىرەتتە ئۇنىڭ ساۋابىغا ئېرىشەلمەيدۇ)
۞ يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَلَا تُبْطِلُوٓا۟ أَعْمَـٰلَكُمْ ﴿٣٣﴾
ئى مۆمىنلەر! اﷲ قا ئىتائەت قىلىڭلار، خۇدانىڭ ئەلچىسىگە ئىتائەت قىلىڭلار، ئەمەللىرىڭلارنى (كۇفرى، نىفاق ۋە رىيا بىلەن) بىكار قىلىۋەتمەڭلار
إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ ثُمَّ مَاتُوا۟ وَهُمْ كُفَّارٌ فَلَن يَغْفِرَ ٱللَّهُ لَهُمْ ﴿٣٤﴾
شۈبھىسىزكى، ھەقىقەتنى رەت قىلغۇچى بولغان ۋە (كىشىلەرنى) اﷲ نىڭ يولىدىن توسقانلار، ئاندىن ھەقىقەتنى رەت قىلغۇچى پىتى ئۆلگەنلەرنى اﷲ ھەرگىز مەغپىرەت قىلمايدۇ
فَلَا تَهِنُوا۟ وَتَدْعُوٓا۟ إِلَى ٱلسَّلْمِ وَأَنتُمُ ٱلْأَعْلَوْنَ وَٱللَّهُ مَعَكُمْ وَلَن يَتِرَكُمْ أَعْمَـٰلَكُمْ ﴿٣٥﴾
(ھەقىقەتنى رەت قىلغۇچىلار بىلەن جەڭ قىلغان چېغىڭلاردا) بوشاپ قالماڭلار، (ھەقىقەتنى رەت قىلغۇچىلارنى) سۈلھكە چاقىرماڭلار، ھالبۇكى، سىلەر غالىپتۇرسىلەر، اﷲ سىلەر بىلەن بىرگىدۇر، اﷲ ھەرگىز ئەمەلىڭلارنىڭ (ساۋابىنى) كېمەيتىۋەتمەيدۇ
إِنَّمَا ٱلْحَيَوٰةُ ٱلدُّنْيَا لَعِبٌ وَلَهْوٌ ۚ وَإِن تُؤْمِنُوا۟ وَتَتَّقُوا۟ يُؤْتِكُمْ أُجُورَكُمْ وَلَا يَسْـَٔلْكُمْ أَمْوَٰلَكُمْ ﴿٣٦﴾
دۇنيا تىرىكچىلىكى پەقەت ئويۇن ۋە تاماشادىن ئىبارەتتۇر، ئەگەر سىلەر ئىمان ئېيتساڭلار ۋە تەقۋادارلىق قىلساڭلار (اﷲ) سىلەرگە ئەجرىڭلارنى بېرىدۇ، مال - مۈلكۈڭلارنىڭ ھەممىسىنى (بېرىشنى) سورىمايدۇ (پەقەت زاكات (پاكلاش ئۇلۇشى)نى بېرىشىڭلارنى سورايدۇ)
إِن يَسْـَٔلْكُمُوهَا فَيُحْفِكُمْ تَبْخَلُوا۟ وَيُخْرِجْ أَضْغَـٰنَكُمْ ﴿٣٧﴾
ئەگەر ئۇ سىلەردىن مال - مۈلكۈڭلارنى جىددى سورىسا، بېخىللىق قىلىسىلەر، بۇ بېخىللىق سىلەرنىڭ (ئىسلامغا بولغان) ئاداۋىتىڭلارنى ئاشكارىلايدۇ
هَـٰٓأَنتُمْ هَـٰٓؤُلَآءِ تُدْعَوْنَ لِتُنفِقُوا۟ فِى سَبِيلِ ٱللَّهِ فَمِنكُم مَّن يَبْخَلُ ۖ وَمَن يَبْخَلْ فَإِنَّمَا يَبْخَلُ عَن نَّفْسِهِۦ ۚ وَٱللَّهُ ٱلْغَنِىُّ وَأَنتُمُ ٱلْفُقَرَآءُ ۚ وَإِن تَتَوَلَّوْا۟ يَسْتَبْدِلْ قَوْمًا غَيْرَكُمْ ثُمَّ لَا يَكُونُوٓا۟ أَمْثَـٰلَكُم ﴿٣٨﴾
سىلەر شۇنداق كىشىلەرسلەركى، اﷲ نىڭ يولىدا (پۇل - مال) سەرپ قىلىشقا چاقىرىلساڭلار، سىلەرنىڭ ئاراڭلاردا بېخىللىق قىلىدىغانلار بار، كىمكى بېخىللىق قىلىدىكەن، ئۇ ئۆزىنىڭ زىيىنى ئۈچۈن بېخىللىق قىلىدۇ، اﷲ (سىلەرنىڭ مال - مۈلكۈڭلاردىن) بىھاجەتتۇر، سىلەر بولساڭلار (اﷲ قا) موھتاجسىلەر، ئەگەر سىلەر (اﷲ نىڭ تائىتىدىن) يۈز ئۆرۈسەڭلار، اﷲ سىلەرنىڭ ئورنۇڭلارغا باشقا بىر قەۋمنى ئالماشتۇرىدۇ، ئۇلار سىلەرگە ئوخشاش بولمايدۇ (بەلكى اﷲ قا ئىتائەت قىلىدۇ)
