بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
أَتَىٰ أَمْرُ اللَّهِ فَلَا تَسْتَعْجِلُوهُ ۚ سُبْحَانَهُ وَتَعَالَىٰ عَمَّا يُشْرِكُونَ ﴿١﴾
اﷲ نىڭ پەرمانى (يەنى قىيامەت) چوقۇم كېلىدۇ، ئۇنىڭغا ئالدىراپ كەتمەڭلار، اﷲ ئۇلار (يەنى مۇشرىكلار) نىڭ شېرىك كەلتۈرگەن نەرسىلىرىدىن پاكتۇر ۋە يۈكسەكتۇر
يُنَزِّلُ الْمَلَائِكَةَ بِالرُّوحِ مِنْ أَمْرِهِ عَلَىٰ مَن يَشَاءُ مِنْ عِبَادِهِ أَنْ أَنذِرُوا أَنَّهُ لَا إِلَٰهَ إِلَّا أَنَا فَاتَّقُونِ ﴿٢﴾
اﷲ ئۆز ئىرادىسى بويىچە خالىغان بەندىلىرىگە خۇدانىڭ ئەلچى پەرىشتىلىرىنى ۋەھيى بىلەن نازىل قىلىپ (مۇنداق دەيدۇ): «(ئىنسانلارنى) ئاگاھلاندۇرۇڭلاركى، مەندىن باشقا ھېچ ئىلاھ يوقتۇر، مەندىن (يەنى ئازابىمدىن) قورقۇڭلار»
خَلَقَ السَّمَاوَاتِ وَالْأَرْضَ بِالْحَقِّ ۚ تَعَالَىٰ عَمَّا يُشْرِكُونَ ﴿٣﴾
اﷲ ئاسمانلارنى ۋە زېمىننى ھەق ياراتتى، اﷲ ئۇلار (يەنى مۇشرىكلار) نىڭ شېرىك كەلتۈرگەن نەرسىلىرىدىن يۈكسەكتۇر
خَلَقَ الْإِنسَانَ مِن نُّطْفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ ﴿٤﴾
ئۇ ئىنساننى ئابىمەنىيدىن ياراتتى، ناگاھان ئۇ (اﷲ قا) ئاشكارا خۇسۇمەت قىلغۇچىدۇر
وَالْأَنْعَامَ خَلَقَهَا ۗ لَكُمْ فِيهَا دِفْءٌ وَمَنَافِعُ وَمِنْهَا تَأْكُلُونَ ﴿٥﴾
اﷲ چارۋا ماللارنى سىلەرنىڭ مەنپەئەتلىنىشىڭلار ئۈچۈن ياراتتى، ئۇلار (نىڭ يۇڭى ۋە تېرىسى) بىلەن ئىسسىنىسىلەر، ئۇلار (نىڭ نەسلى، سۈتى، سۆڭىكى ۋە قىغى) دىن پايدىلىنىسىلەر ۋە ئۇلار (نىڭ گۆشلىرى) نى يەيسىلەر
وَلَكُمْ فِيهَا جَمَالٌ حِينَ تُرِيحُونَ وَحِينَ تَسْرَحُونَ ﴿٦﴾
ئۇلارنى كەچتە ئوتلاقتىن قايتۇرۇپ كەلگەن ۋە ئەتىگەندە ئوتلاققا ئېلىپ چىققان ۋاقتىڭلاردا، چارۋا ماللار سىلەرگە زىننەت بولۇپ (خۇشاللىق بېغىشلايدۇ)
وَتَحْمِلُ أَثْقَالَكُمْ إِلَىٰ بَلَدٍ لَّمْ تَكُونُوا بَالِغِيهِ إِلَّا بِشِقِّ الْأَنفُسِ ۚ إِنَّ رَبَّكُمْ لَرَءُوفٌ رَّحِيمٌ ﴿٧﴾
ئۇلار يۈك - تاقلىرىڭلارنى سىلەر جاپا - مۇشەققەت بىلەن ئاران يېتىپ بارالايدىغان جايلارغا كۆتۈرۈپ بارىدۇ، سىلەرنىڭ پەرۋەردىگارىڭلار سىلەرگە ئىشەنچلىك ھالدا شەپقەتلىكتۇر ۋە مېھرىباندۇر
وَالْخَيْلَ وَالْبِغَالَ وَالْحَمِيرَ لِتَرْكَبُوهَا وَزِينَةً ۚ وَيَخْلُقُ مَا لَا تَعْلَمُونَ ﴿٨﴾
اﷲ ئاتنى، خېچىرنى ۋە ئېشەكنى مىنىشىڭلار ئۈچۈن ۋە زىننەت ئۈچۈن ياراتتى، اﷲ يەنە سىلەر بىلمەيدىغان نەرسىلەرنىمۇ يارىتىدۇ
وَعَلَى اللَّهِ قَصْدُ السَّبِيلِ وَمِنْهَا جَائِرٌ ۚ وَلَوْ شَاءَ لَهَدَاكُمْ أَجْمَعِينَ ﴿٩﴾
توغرا يول كۆرسىتىشنى اﷲ ئۆز ئۈستىگە ئالغان. بەزى يوللار ئەگرى بولىدۇ، مۇبادا اﷲ خالىسا ئىدى، سىلەرنىڭ ھەممىڭلارنى ئەلۋەتتە توغرا يولغا سالغان بولاتتى
هُوَ الَّذِي أَنزَلَ مِنَ السَّمَاءِ مَاءً ۖ لَّكُم مِّنْهُ شَرَابٌ وَمِنْهُ شَجَرٌ فِيهِ تُسِيمُونَ ﴿١٠﴾
اﷲ سىلەرگە بۇلۇتتىن يامغۇر ياغدۇرۇپ بەردى، ئۇنىڭدىن ئىچىسىلەر ۋە شۇ يامغۇر ئاقىلىق ئۆسكەن دەل - دەرەخلەر بىلەن چارۋاڭلارنى باقىسىلەر
يُنبِتُ لَكُم بِهِ الزَّرْعَ وَالزَّيْتُونَ وَالنَّخِيلَ وَالْأَعْنَابَ وَمِن كُلِّ الثَّمَرَاتِ ۗ إِنَّ فِي ذَٰلِكَ لَآيَةً لِّقَوْمٍ يَتَفَكَّرُونَ ﴿١١﴾
اﷲ سىلەرگە شۇ يامغۇر بىلەن زىرائەتلەرنى ئۆستۈرۈپ بېرىدۇ، زەيتۇن، خورما، ئۈزۈم ۋە تۈرلۈك مېۋىلەرنى ئۆستۈرۈپ بېرىدۇ، چوڭقۇر پىكىر قىلىدىغان قەۋم ئۈچۈن بۇنىڭدا ئىشەنچلىك ھالدا (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) دەلىللەر بار
وَسَخَّرَ لَكُمُ اللَّيْلَ وَالنَّهَارَ وَالشَّمْسَ وَالْقَمَرَ ۖ وَالنُّجُومُ مُسَخَّرَاتٌ بِأَمْرِهِ ۗ إِنَّ فِي ذَٰلِكَ لَآيَاتٍ لِّقَوْمٍ يَعْقِلُونَ ﴿١٢﴾
اﷲ سىلەرگە كېچىنى، كۈندۈزنى، كۈننى، ئاينى بويسۇندۇرۇپ بەردى، يۇلتۇزلارمۇ اﷲ نىڭ ئەمرى بىلەن بويسۇندۇرۇلغاندۇر؛ چۈشىنىدىغان قەۋم ئۈچۈن بۇنىڭدا نۇرغۇن دەلىللەر بار
وَمَا ذَرَأَ لَكُمْ فِي الْأَرْضِ مُخْتَلِفًا أَلْوَانُهُ ۗ إِنَّ فِي ذَٰلِكَ لَآيَةً لِّقَوْمٍ يَذَّكَّرُونَ ﴿١٣﴾
اﷲ يەنە يەر يۈزىدە سىلەر ئۈچۈن ياراتقان رەڭگارەڭ نەرسىلەرنى (سىلەرگە بويسۇندۇرۇپ بەردى)، ۋەز - نەسىھەت ئالىدىغان قەۋم ئۈچۈن بۇنىڭدا ئەلۋەتتە ئىبرەت بار
وَهُوَ الَّذِي سَخَّرَ الْبَحْرَ لِتَأْكُلُوا مِنْهُ لَحْمًا طَرِيًّا وَتَسْتَخْرِجُوا مِنْهُ حِلْيَةً تَلْبَسُونَهَا وَتَرَى الْفُلْكَ مَوَاخِرَ فِيهِ وَلِتَبْتَغُوا مِن فَضْلِهِ وَلَعَلَّكُمْ تَشْكُرُونَ ﴿١٤﴾
اﷲ سىلەرنى دېڭىزنىڭ يېڭى گوشلىرى (يەنى بېلىقلىرى) نى يېسۇن، تاقايدىغان زىننەت بۇيۇملىرى (يەنى ئۈنچە - مارجانلار) نى چىقارسۇن دەپ، سىلەرگە دېڭىزنى بويسۇندۇرۇپ بەردى. اﷲ نىڭ نېمەتلىرىنى تەلەپ قىلىشىڭلار ۋە ئۇنىڭغا شۈكۈر قىلىشىڭلار ئۈچۈن، (اﷲ نىڭ بويسۇندۇرۇشى بىلەن يېمەك - ئىچمەكلەر ۋە يۈك - تاقلار قاچىلانغان) كېمىلەرنىڭ دېڭىزدا دولقۇن يېرىپ كېتىۋاتقانلىقىنى كۆرۈسەن
وَأَلْقَىٰ فِي الْأَرْضِ رَوَاسِيَ أَن تَمِيدَ بِكُمْ وَأَنْهَارًا وَسُبُلًا لَّعَلَّكُمْ تَهْتَدُونَ ﴿١٥﴾
زېمىننىڭ سىلەرنى تەۋرىتىۋەتمەسلىكى ئۈچۈن، اﷲ ئۇنىڭدا تاغلارنى بەرپا قىلدى، (زىرائەتلىرىڭلارنى، چارۋا ماللىرىڭلارنى سۇغىرىشىڭلار ئۈچۈن) ئېرىق - ئۆستەڭلەرنى، ئېزىپ قالماسلىقىڭلار ئۈچۈن يوللارنى پەيدا قىلدى
وَعَلَامَاتٍ ۚ وَبِالنَّجْمِ هُمْ يَهْتَدُونَ ﴿١٦﴾
(يوللارغا) بەلگىلەرنى قىلدى. ئۇلار (كېچىسى قۇرۇقلۇقتا ۋە دېڭىزلاردا) يۇلتۇزلار بىلەن يول تاپىدۇ
أَفَمَن يَخْلُقُ كَمَن لَّا يَخْلُقُ ۗ أَفَلَا تَذَكَّرُونَ ﴿١٧﴾
(پۇتۇن كائىناتنى) ياراتقان اﷲ بىلەن (ھېچ نەرسىنى) يارىتالمايدىغان (بۇتلار) ئوخشاش بولامدۇ؟ ئويلىمامسىلەر
وَإِن تَعُدُّوا نِعْمَةَ اللَّهِ لَا تُحْصُوهَا ۗ إِنَّ اللَّهَ لَغَفُورٌ رَّحِيمٌ ﴿١٨﴾
ئەگەر اﷲ نىڭ (سىلەرگە بەرگەن) نېمىتىنى سانىساڭلار، ئۇنىڭ ھېسابىنى ئېلىپ بولالمايسىلەر. شۈبھىسىزكى، اﷲ سىلەر (نىڭ بۇ جەھەتتىكى يېتەرسىزلىكىڭلار) نى ئەلۋەتتە مەغپىرەت قىلغۇچىدۇر، (بەندىلىرىگە) مېھرىباندۇر
وَاللَّهُ يَعْلَمُ مَا تُسِرُّونَ وَمَا تُعْلِنُونَ ﴿١٩﴾
اﷲ سىلەر يوشۇرغان ۋە ئاشكارا قىلغاننىڭ ھەممىسىنى بىلىپ تۇرىدۇ
وَالَّذِينَ يَدْعُونَ مِن دُونِ اللَّهِ لَا يَخْلُقُونَ شَيْئًا وَهُمْ يُخْلَقُونَ ﴿٢٠﴾
ئۇلارنىڭ اﷲ نى قويۇپ ئىبادەت قىلغانلىرى (يەنى بۇتلىرى) ھېچ شەيئىنى يارىتالمايدۇ، بەلكى ئۇلارنىڭ ئۆزلىرى يارىتىلغاندۇر (يەنى ئىنسانلار تەرىپىدىن ياسالغاندۇر)
أَمْوَاتٌ غَيْرُ أَحْيَاءٍ ۖ وَمَا يَشْعُرُونَ أَيَّانَ يُبْعَثُونَ ﴿٢١﴾
ئۇلار (يەنى بۇتلار) تىرىك ئە مەس، ئۆلۈكتۈر، ئۇلار (چوقۇنغۇچىلىرىنىڭ) قاچان قايتا تىرىلدۈرۈلىدىغانلىقىنى ئۇقمايدۇ
إِلَٰهُكُمْ إِلَٰهٌ وَاحِدٌ ۚ فَالَّذِينَ لَا يُؤْمِنُونَ بِالْآخِرَةِ قُلُوبُهُم مُّنكِرَةٌ وَهُم مُّسْتَكْبِرُونَ ﴿٢٢﴾
سىلەرنى (ئىبادەت قىلىشقا لايىق) ئىلاھىڭلار بىردۇر، ئاخىرەتكە ئىشەنمەيدىغانلارنىڭ دىللىرى (اﷲ نىڭ بىرلىكىنى) ئىنكار قىلىدۇ، ئۇلار چوڭچىلىق قىلىپ، (ھەقنى قوبۇل قىلىشتىن) باش تارتقۇچىلاردۇر
لَا جَرَمَ أَنَّ اللَّهَ يَعْلَمُ مَا يُسِرُّونَ وَمَا يُعْلِنُونَ ۚ إِنَّهُ لَا يُحِبُّ الْمُسْتَكْبِرِينَ ﴿٢٣﴾
اﷲ ئۇلارنىڭ يوشۇرغانلىرىنى ۋە ئاشكارا قىلغانلىرىنى راستلا بىلىپ تۇرىدۇ، اﷲ تەكەببۇرلۇق قىلغۇچىلارنى ئىشەنچلىك ھالدا دوست تۇتمايدۇ
وَإِذَا قِيلَ لَهُم مَّاذَا أَنزَلَ رَبُّكُمْ ۙ قَالُوا أَسَاطِيرُ الْأَوَّلِينَ ﴿٢٤﴾
ئۇلارغا (يەنى مۇشرىكلارغا): «پەرۋەردىگارىڭلار (پەيغەمبىرىگە) نېمىلەرنى نازىل قىلدى؟» دېيىلسە، ئۇلار (مەسخىرە قىلىپ): «قەدىمكىلەرنىڭ ئەپسانىلىرىنى (نازىل قىلىدى)» دېيىشىدۇ
لِيَحْمِلُوا أَوْزَارَهُمْ كَامِلَةً يَوْمَ الْقِيَامَةِ ۙ وَمِنْ أَوْزَارِ الَّذِينَ يُضِلُّونَهُم بِغَيْرِ عِلْمٍ ۗ أَلَا سَاءَ مَا يَزِرُونَ ﴿٢٥﴾
ئۇلار قىيامەت كۈنى ئۆزلىرىنىڭ گۇناھلىرىغا تولۇق جاۋابكار بولۇشلىرى ۋە ئۆزلىرى تەرىپىدىن بىلمەستىن ئازدۇرۇلغانلارنىڭ بىر قىسىم گۇناھلىرىغا ئۆزلىرى جاۋابكار بولۇشلىرى ئۈچۈن (يۇقىرىقى بوھتان سوزلەرنى قىلدى)، ئۇلارنىڭ جاۋابكار بولىدىغىنى نېمىدېگەن يامان!
قَدْ مَكَرَ الَّذِينَ مِن قَبْلِهِمْ فَأَتَى اللَّهُ بُنْيَانَهُم مِّنَ الْقَوَاعِدِ فَخَرَّ عَلَيْهِمُ السَّقْفُ مِن فَوْقِهِمْ وَأَتَاهُمُ الْعَذَابُ مِنْ حَيْثُ لَا يَشْعُرُونَ ﴿٢٦﴾
ئۇلار (يەنى مەككە ھەقىقەتنى رەت قىلغۇچىلىرى) دىن ئىلگىرى ئۆتكەن ھەقىقەتنى رەت قىلغۇچىلار (ئۆز خۇدانىڭ ئەلچىسىلىرىگە) ھىيلە - مىكىر ئىشلىتىپ (اﷲ نىڭ نۇرىنى ئۆچۈرمەكچى بولدى). اﷲ ئۇلارنىڭ (ھىيلە - مىكىر) بىناسىنى تەلتۆكۈس گۇمران قىلدى، ئۇلارنىڭ ئۈستىگە (بۇ بىنانىڭ) ئۆگزىسى ئۆرۈلۈپ چۈشتى، ئۇلارغا ئۆزلىرى ئويلىمىغان يەردىن ئازاب كەلدى
ثُمَّ يَوْمَ الْقِيَامَةِ يُخْزِيهِمْ وَيَقُولُ أَيْنَ شُرَكَائِيَ الَّذِينَ كُنتُمْ تُشَاقُّونَ فِيهِمْ ۚ قَالَ الَّذِينَ أُوتُوا الْعِلْمَ إِنَّ الْخِزْيَ الْيَوْمَ وَالسُّوءَ عَلَى الْكَافِرِينَ ﴿٢٧﴾
ئاندىن كېيىن اﷲ ئۇلارنى قىيامەت كۈنى رەسۋا قىلىدۇ، (ئۇلارغا كايىپ) «مېنىڭ شېرىكلىرىم قېنى قەيەردە؟ سىلەر ئۇلار توغرىسىدا (خۇدانىڭ ئەلچى-رەسۇللىرى بىلەن) مۇنازىرىلىشەتتىڭلارغۇ؟» دەيدۇ. ئىلىم بېرىگەنلەر (يەنى ئۇلارنى ئىمانغا دەۋەت قىلغان خۇدانىڭ ئەلچى-رەسۇللىرى ۋە ئۆلىمالار): «بۈگۈن رەسۋالىق ۋە ئازاب ئىشەنچلىك ھالدا ھەقىقەتنى رەت قىلغۇچىلارغا تېگىشلىكتۇر» دەيدۇ
الَّذِينَ تَتَوَفَّاهُمُ الْمَلَائِكَةُ ظَالِمِي أَنفُسِهِمْ ۖ فَأَلْقَوُا السَّلَمَ مَا كُنَّا نَعْمَلُ مِن سُوءٍ ۚ بَلَىٰ إِنَّ اللَّهَ عَلِيمٌ بِمَا كُنتُمْ تَعْمَلُونَ ﴿٢٨﴾
ئۇلار (كۇفرىنى ئىختىيار قىلىپ) ئۆزلىرىگە زىيان سالغان شارائىتتا، خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارنىڭ جانلىرىنى ئالىدۇ. ئۇلار بويسۇنغانلىقلىرىنى بىلدۈرۈپ: «بىز ھېچقانداق يامان ئىش قىلمىدۇق» (دەيدۇ). ئۇنداق ئەمەس، اﷲ سىلەنىڭ قىلمىشىڭلارنى ئىشەنچلىك ھالدا تولۇق بىلگۈچىدۇر
فَادْخُلُوا أَبْوَابَ جَهَنَّمَ خَالِدِينَ فِيهَا ۖ فَلَبِئْسَ مَثْوَى الْمُتَكَبِّرِينَ ﴿٢٩﴾
سىلەر (ئىچىدە) مەڭگۈ قالىدىغان مەڭگۈلۈك ئوتنىڭ دەرۋازىلىرىدىن كىرىڭلار، چوڭچىلىق قىلىپ، (اﷲ قا ئىتائەت قىلىشتىن باش تارتقۇچىلارنىڭ بارىدىغان جايى نېمىدېگەن يامان!
وَقِيلَ لِلَّذِينَ اتَّقَوْا مَاذَا أَنزَلَ رَبُّكُمْ ۚ قَالُوا خَيْرًا ۗ لِّلَّذِينَ أَحْسَنُوا فِي هَٰذِهِ الدُّنْيَا حَسَنَةٌ ۚ وَلَدَارُ الْآخِرَةِ خَيْرٌ ۚ وَلَنِعْمَ دَارُ الْمُتَّقِينَ ﴿٣٠﴾
تەقۋادارلارغا: «پەرۋەردىگارىڭلار (پەيغەمبىرىگە) نېمىلەرنى نازىل قىلدى؟» دېيىلسە، ئۇلار: «ياخشى سۆزلەرنى نازىل قىلدى» دېيىشىدۇ. ياخشى ئىش قىلغۇچىلار بۇ دۇنيادا مۇكاپاتلىنىدۇ، ئاخىرەت يۇرتى (ئۇلار ئۈچۈن تېخىمۇ) ياخشىدۇر. تەقۋادارلارنىڭ يۇرتى راستلا نېمىدېگەن گۈزەل!
جَنَّاتُ عَدْنٍ يَدْخُلُونَهَا تَجْرِي مِن تَحْتِهَا الْأَنْهَارُ ۖ لَهُمْ فِيهَا مَا يَشَاءُونَ ۚ كَذَٰلِكَ يَجْزِي اللَّهُ الْمُتَّقِينَ ﴿٣١﴾
ئۇلار ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان ئەدنى مەڭگۈلۈك باغلارلىرىگە كىرىدۇ، ئۇ يەردە ئۇلارنىڭ كۆڭلى خالىغان نەرسىلەر تېپىلىدۇ، اﷲ تەقۋادارلارنى ئەنە شۇنداق مۇكاپاتلايدۇ
الَّذِينَ تَتَوَفَّاهُمُ الْمَلَائِكَةُ طَيِّبِينَ ۙ يَقُولُونَ سَلَامٌ عَلَيْكُمُ ادْخُلُوا الْجَنَّةَ بِمَا كُنتُمْ تَعْمَلُونَ ﴿٣٢﴾
ئۇلار (مۇشرىكلىك ۋە گۇناھلاردىن) پاك بولغان شارائىتتا خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارنى قەبزى روھ قىلىدۇ، خۇدانىڭ ئەلچى پەرىشتىلىرى ئۇلارغا: «سالام سىلەرگە! قىلغان ئەمەلىڭلارنى خاسىيىتىدىن مەڭگۈلۈك باغلاركە كىرىڭلار» دەيدۇ
هَلْ يَنظُرُونَ إِلَّا أَن تَأْتِيَهُمُ الْمَلَائِكَةُ أَوْ يَأْتِيَ أَمْرُ رَبِّكَ ۚ كَذَٰلِكَ فَعَلَ الَّذِينَ مِن قَبْلِهِمْ ۚ وَمَا ظَلَمَهُمُ اللَّهُ وَلَٰكِن كَانُوا أَنفُسَهُمْ يَظْلِمُونَ ﴿٣٣﴾
ھەقىقەتنى رەت قىلغۇچىلار پەقەت (ئۆزلىرىنىڭ جېنىنى ئېلىش ئۈچۈن) خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ چۈشۈشىنى ياكى پەرۋەردىگارىڭنىڭ ئەمرىنىڭ (يەنى ئازابىنىڭ) چۈشۈشىنىلا كۈتىدۇ، ئۇلاردىن بۇرۇن ئۆتكەن (ھەقىقەتنى رەت قىلغۇچى) لارمۇ ئەنە شۇنداق قىلغان ئىدى. اﷲ ئۇلارغا زۇلۇم قىلغىنى يوق، لېكىن ئۇلار ئۆزلىرىگە ئۆزلىرى زۇلۇم قىلدى
فَأَصَابَهُمْ سَيِّئَاتُ مَا عَمِلُوا وَحَاقَ بِهِم مَّا كَانُوا بِهِ يَسْتَهْزِئُونَ ﴿٣٤﴾
ئۇلار قىلمىشلىرىنىڭ جازاسىنى تارتتى، ئۇلارغا مەسخىرە قىلغانلىقلىرىنىڭ جازاسى (يەنى قاتتىق ئازابى) چۈشتى
وَقَالَ الَّذِينَ أَشْرَكُوا لَوْ شَاءَ اللَّهُ مَا عَبَدْنَا مِن دُونِهِ مِن شَيْءٍ نَّحْنُ وَلَا آبَاؤُنَا وَلَا حَرَّمْنَا مِن دُونِهِ مِن شَيْءٍ ۚ كَذَٰلِكَ فَعَلَ الَّذِينَ مِن قَبْلِهِمْ ۚ فَهَلْ عَلَى الرُّسُلِ إِلَّا الْبَلَاغُ الْمُبِينُ ﴿٣٥﴾
مۇشرىكلار: «مۇبادا اﷲ خالىغان بولسا، بىز ۋە بىزنىڭ ئاتا بوۋىلىرىمىز اﷲ نى قويۇپ ھىچ نەرسىگە ئىبادەت قىلمىغان بولاتتۇق، اﷲ ھارام قىلمىغان (يەنى اﷲ نىڭ ھۆكمىسىز) ھىچ نەرسىنى ھارام قىلمىغان بولاتتۇق» دېدى. ئۇلاردىن ئىلگىرىكىلەرمۇ ئەنە شۇنداق قىلغان ئىدى. خۇدانىڭ ئەلچى-رەسۇللىرىنىڭ مەسئۇلىيىتى پەقەت روشەن تەبلىغ قىلىشتىنلا ئىبارەت (ھىدايەت قىلىش بولسا اﷲ نىڭ ئىشىدۇر)
وَلَقَدْ بَعَثْنَا فِي كُلِّ أُمَّةٍ رَّسُولًا أَنِ اعْبُدُوا اللَّهَ وَاجْتَنِبُوا الطَّاغُوتَ ۖ فَمِنْهُم مَّنْ هَدَى اللَّهُ وَمِنْهُم مَّنْ حَقَّتْ عَلَيْهِ الضَّلَالَةُ ۚ فَسِيرُوا فِي الْأَرْضِ فَانظُرُوا كَيْفَ كَانَ عَاقِبَةُ الْمُكَذِّبِينَ ﴿٣٦﴾
بىز ئىشەنچلىك ھالدا ھەر بىر ئۈممەتكە: «اﷲ قا ئىبادەت قىلىڭلار، شەيتان (ئازدۇرغۇچى)دىن (يەنى شەيتان (ئازدۇرغۇچى)غا، بۇتلارغا، كاھىنلارغا ئوخشاش ھەرقانداق مەبۇدقا ئىبادەت قىلىشتىن) يىراق بولۇڭلار» دەپ خۇدانىڭ ئەلچىسى ئەۋەتتۇق، ئۇلارنىڭ ئارىسىدا اﷲ ھىدايەت قىلغانلىرىمۇ بار، گۇمراھلىققا تېگىشلىك بولغانلىرىمۇ بار. يەر يۈزىدە سەير قىلىپ يۈرۈپ (خۇدانىڭ ئەلچى-رەسۇللىرىنى) ئىنكار قىلغانلارنىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۆزىتىڭلار
إِن تَحْرِصْ عَلَىٰ هُدَاهُمْ فَإِنَّ اللَّهَ لَا يَهْدِي مَن يُضِلُّ ۖ وَمَا لَهُم مِّن نَّاصِرِينَ ﴿٣٧﴾
سەن ئۇلارنىڭ ھىدايەت تىپىشىغا ھېرىس بولساڭمۇ، اﷲ ئۆزى گۇمراھ قىلغان كىشىنى ھىدايەت قىلمايدۇ، ئۇلارنى (اﷲ نىڭ ئازابىدىن قۇتۇلدۇرىدىغان) ھېچقانداق مەدەتكار يوق
وَأَقْسَمُوا بِاللَّهِ جَهْدَ أَيْمَانِهِمْ ۙ لَا يَبْعَثُ اللَّهُ مَن يَمُوتُ ۚ بَلَىٰ وَعْدًا عَلَيْهِ حَقًّا وَلَٰكِنَّ أَكْثَرَ النَّاسِ لَا يَعْلَمُونَ ﴿٣٨﴾
ئۇلار: «اﷲ ئۆلۈكنى تىرىلدۈرمەيدۇ» دەپ قاتتىق قەسەم ئىچىشتى، ئۇنداق ئەمەس (يەنى ئۆلۈكنى تىرىلدۈرىدۇ)، اﷲ ۋەدىسىنى چوقۇم ئىشقا ئاشۇرىدۇ، لېكىن ئىنسانلارنىڭ تولىسى (اﷲ نىڭ قۇدرىتىنى) بىلمەيدۇ (شۇڭلاشقا ئۆلگەندىن كېيىن تىرىلىشنى ئىنكار قىلىدۇ
لِيُبَيِّنَ لَهُمُ الَّذِي يَخْتَلِفُونَ فِيهِ وَلِيَعْلَمَ الَّذِينَ كَفَرُوا أَنَّهُمْ كَانُوا كَاذِبِينَ ﴿٣٩﴾
اﷲ ئۇلار ئىختىلاپ قىلىشقان ھەق بىلەن ناھەقنى ئۇلارغا ئايرىپ بېرىش ۋە ھەقىقەتنى رەت قىلغۇچىلارغا ئۇلارنىڭ يالغانچىلىقلىرىنى بىلدۈرۈپ قويۇش ئۈچۈن (ئۇلارنى قايتىدىن تىرىلدۈرىدۇ)
إِنَّمَا قَوْلُنَا لِشَيْءٍ إِذَا أَرَدْنَاهُ أَن نَّقُولَ لَهُ كُن فَيَكُونُ ﴿٤٠﴾
بىز بىرەر شەيئىنى ۋۇجۇدقا كەلتۈرمەكچى بولساق، ئۇنىڭغا: «ۋۇجۇدقا كەل» دەيمىز - دە، ئۇ ۋۇجۇدقا كېلىدۇ
وَالَّذِينَ هَاجَرُوا فِي اللَّهِ مِن بَعْدِ مَا ظُلِمُوا لَنُبَوِّئَنَّهُمْ فِي الدُّنْيَا حَسَنَةً ۖ وَلَأَجْرُ الْآخِرَةِ أَكْبَرُ ۚ لَوْ كَانُوا يَعْلَمُونَ ﴿٤١﴾
زۇلۇمغا ئۇچرىغاندىن كېيىن اﷲ يولىدا ھىجرەت قىلغانلارنى دۇنيادا چوقۇم ياخشى جايغا ئورۇنلاشتۇرىمىز، ئۇلار بىلىدىغان بولسا، ئۇلارغا ئاخىرەتنىڭ ساۋابى تېخىمۇ چوڭدۇر
الَّذِينَ صَبَرُوا وَعَلَىٰ رَبِّهِمْ يَتَوَكَّلُونَ ﴿٤٢﴾
ئۇلار (ئېغىرچىلىقلارغا) سەۋر قىلدى، ئۇلار (ھەممە ئىشتا) پەرۋەردىگارىغا تەۋەككۈل قىلىدۇ (يەنى يۆلىنىدۇ)
وَمَا أَرْسَلْنَا مِن قَبْلِكَ إِلَّا رِجَالًا نُّوحِي إِلَيْهِمْ ۚ فَاسْأَلُوا أَهْلَ الذِّكْرِ إِن كُنتُمْ لَا تَعْلَمُونَ ﴿٤٣﴾
سەندىن ئىلگىرى بىز پەقەت ئەرلەرگىلا ۋەھيى نازىل قىلىپ، ئۇلارنى خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق، (ئى قۇرەيش جامائەسى!) بۇنى بىلمىسەڭلار، ئىلىم ئەھلىلىرى (يەنى تەۋرات ۋە ئىنجىلنى بىلگۈچىلەر) دىن سوراڭلار
بِالْبَيِّنَاتِ وَالزُّبُرِ ۗ وَأَنزَلْنَا إِلَيْكَ الذِّكْرَ لِتُبَيِّنَ لِلنَّاسِ مَا نُزِّلَ إِلَيْهِمْ وَلَعَلَّهُمْ يَتَفَكَّرُونَ ﴿٤٤﴾
(ئۇلارنى ئۇلارنىڭ راستلىقىنى كۆرسىتىپ بېرىدىغان) دەلىللەر ۋە مۇقەددەس مۇقەددەس كىتاب-ۋەھىيلار بىلەن (ئەۋەتتۇق)، بىز ساڭا قۇرئاننى ئىنسانلارغا چۈشۈرۈلگەن شەرىئەتنى بايان قىلىپ بەرسۇن، ئۇلار (بۇ قۇرئاننى) پىكىر قىلىپ (ئۇنىڭدىن ۋەز - نەسىھەت ئالسۇن) دەپ نازىل قىلدۇق
أَفَأَمِنَ الَّذِينَ مَكَرُوا السَّيِّئَاتِ أَن يَخْسِفَ اللَّهُ بِهِمُ الْأَرْضَ أَوْ يَأْتِيَهُمُ الْعَذَابُ مِنْ حَيْثُ لَا يَشْعُرُونَ ﴿٤٥﴾
ھىيلە - مىكىر ئىشلىتىپ يامان ئىش قىلغانلار اﷲ نىڭ ئۆزلىرىنى يەرگە يۇتقۇزۇۋىېتىشىدىن ياكى ئۆزلىرىگە ئويلىمىغان يەردىن ئازاب كېلىدىن قورقمامدۇ؟
أَوْ يَأْخُذَهُمْ فِي تَقَلُّبِهِمْ فَمَا هُم بِمُعْجِزِينَ ﴿٤٦﴾
ياكى ئۆزلىرىنى سەپەر ئۇستىدىكى چاغلىرىدا جازاغا يۇلۇقتۇرۇشىدىن قورقمامدۇ؟ (بۇ چاغدا) ئۇلار قېچىپ قۇتۇلالمايدۇ
أَوْ يَأْخُذَهُمْ عَلَىٰ تَخَوُّفٍ فَإِنَّ رَبَّكُمْ لَرَءُوفٌ رَّحِيمٌ ﴿٤٧﴾
ياكى (اﷲ نىڭ) ئۆزلىرىنى قەدەممۇقەدەم جازالىشىدىن قورقمامدۇ؟ سىلەرنىڭ پەرۋەردىگارىڭلار ئەلۋەتتە مەرھەمەتلىكتۇر، مېھرىباندۇر (شۇڭا سىلەرنى ئالدىراپ جازالىمىدى
أَوَلَمْ يَرَوْا إِلَىٰ مَا خَلَقَ اللَّهُ مِن شَيْءٍ يَتَفَيَّأُ ظِلَالُهُ عَنِ الْيَمِينِ وَالشَّمَائِلِ سُجَّدًا لِّلَّهِ وَهُمْ دَاخِرُونَ ﴿٤٨﴾
ئۇلار اﷲ ياراتقان شەيئىلەرنىڭ سايىلىرىنىڭ بويسۇنغان ھالدا اﷲ قا سەجدە قىلىش يۈزىسىدىن ئوڭ - سولغا مايىل بولۇپ تۇرغانلىقىنى كۆرمىدىمۇ؟
وَلِلَّهِ يَسْجُدُ مَا فِي السَّمَاوَاتِ وَمَا فِي الْأَرْضِ مِن دَابَّةٍ وَالْمَلَائِكَةُ وَهُمْ لَا يَسْتَكْبِرُونَ ﴿٤٩﴾
ئاسمانلاردىكى ۋە زېمىندىكى ھايۋاناتلار، خۇدانىڭ ئەلچى پەرىشتىلىرى اﷲ قا سەجدە قىلىدۇ، ئۇلار (اﷲ قا سەجدە قىلىشتىن) چوڭچىلىق قىلىپ باش تارتمايدۇ
يَخَافُونَ رَبَّهُم مِّن فَوْقِهِمْ وَيَفْعَلُونَ مَا يُؤْمَرُونَ ﴿٥٠﴾
ئۇلار ئۆز ئۈستىدىكى پەرۋەردىگارىدىن قورقىدۇ، ئۆزلىرىگە بۇيرۇلغاننى قىلىدۇ
وَقَالَ اللَّهُ لَا تَتَّخِذُوا إِلَٰهَيْنِ اثْنَيْنِ ۖ إِنَّمَا هُوَ إِلَٰهٌ وَاحِدٌ ۖ فَإِيَّايَ فَارْهَبُونِ ﴿٥١﴾
اﷲ: «ئىككى ئىلاھقا ئىبادەت قىلماڭلار، ئۇ پەقەت بىرلا ئىلاھتۇر، مەندىنلا قورقۇڭلار» دېدى
وَلَهُ مَا فِي السَّمَاوَاتِ وَالْأَرْضِ وَلَهُ الدِّينُ وَاصِبًا ۚ أَفَغَيْرَ اللَّهِ تَتَّقُونَ ﴿٥٢﴾
ئاسمانلاردىكى ۋە زېمىندىكى ھەممە نەرسە اﷲ نىڭ مۈلكىدۇر، ئىبادەت ھەمىشە اﷲ قا خاستۇر، اﷲ تىن بۆلەك غەيرىدىن قورقامسىلەر؟
وَمَا بِكُم مِّن نِّعْمَةٍ فَمِنَ اللَّهِ ۖ ثُمَّ إِذَا مَسَّكُمُ الضُّرُّ فَإِلَيْهِ تَجْأَرُونَ ﴿٥٣﴾
سىلەر بەھرىمەن بولۇۋاتقان نېمەتلەرنىڭ ھەممىسىنى اﷲ بەرگەن، سىلەرگە بىر ئېغىرچىلىق يەتسە، اﷲ قا يالۋۇرۇپ دۇئا قىلىسىلەر
ثُمَّ إِذَا كَشَفَ الضُّرَّ عَنكُمْ إِذَا فَرِيقٌ مِّنكُم بِرَبِّهِمْ يُشْرِكُونَ ﴿٥٤﴾
اﷲ سىلەردىن ئېغىرچىلىقنى كۆتۈرۈۋەتسە، ئاراڭلاردىن بىر بۆلۈك ئادەم دەرھال ئۆز پەرۋەردىگارىغا شېرىك كەلتۈرىدۇ
لِيَكْفُرُوا بِمَا آتَيْنَاهُمْ ۚ فَتَمَتَّعُوا ۖ فَسَوْفَ تَعْلَمُونَ ﴿٥٥﴾
ئۇلار ئۆزلىرىگە بەرگەن (بېشىدىكى ئېغىرچىلىقنى كۆتۈرۈۋەتكەنلىكىمىزدىن ئىبارەت) نېمىتىمىزگە ناشۈكۈرلۈك قىلىدۇ، (سىلەر ئەجىلىڭلار يەتكۈچە دۇنيادىن) بەھرىمەن بولۇڭلار، (ئىشىڭلارنىڭ ئاقىۋىتىنى) ئۇزاققا قالماي بىلىسىلەر
وَيَجْعَلُونَ لِمَا لَا يَعْلَمُونَ نَصِيبًا مِّمَّا رَزَقْنَاهُمْ ۗ تَاللَّهِ لَتُسْأَلُنَّ عَمَّا كُنتُمْ تَفْتَرُونَ ﴿٥٦﴾
ئۇلار بىز ئۆزلىرىگە بەرگەن مال - مۈلۈكتىن بىلىمسىز (بۇتلارغا) نېسىۋە قالدۇرىدۇ، (ئى مۇشرىكلار!) اﷲ نىڭ نامى بىلەن قەسەمكى، سىلەر ئۆزۈڭلار ئويدۇرۇپ چىقارغان نەرسەڭلار ئۈچۈن چوقۇم سوراققا تارتىلىسىلەر
وَيَجْعَلُونَ لِلَّهِ الْبَنَاتِ سُبْحَانَهُ ۙ وَلَهُم مَّا يَشْتَهُونَ ﴿٥٧﴾
ئۇلار اﷲ قا قىزلارنى نىسبەت بېرىدۇ، اﷲ (بۇنداق بوھتاندىن) پاكتۇر، ئۇلار ئۆزى ئارزۇ قىلىدىغانلىرىنى ئۆزلىرىگە نىسبەت بېرىدۇ
وَإِذَا بُشِّرَ أَحَدُهُم بِالْأُنثَىٰ ظَلَّ وَجْهُهُ مُسْوَدًّا وَهُوَ كَظِيمٌ ﴿٥٨﴾
ئۇلارنىڭ بىرەرسىگە (خوتۇنىنىڭ) قىز تۇغقانلىق خوش خەۋىرى يەتكۈزۈلسە، چىرايى ئۆزگىرىپ، غەزەپناك بولۇپ كېتىدۇ
يَتَوَارَىٰ مِنَ الْقَوْمِ مِن سُوءِ مَا بُشِّرَ بِهِ ۚ أَيُمْسِكُهُ عَلَىٰ هُونٍ أَمْ يَدُسُّهُ فِي التُّرَابِ ۗ أَلَا سَاءَ مَا يَحْكُمُونَ ﴿٥٩﴾
يەتكۈزۈلگەن يامان خەۋەردىن قورقۇپ، ئۆز قەۋمىگە كۆرۈنمەي يوشۇرۇنۇۋالىدۇ. ئاندىن ئۇ نومۇسقا چىداپ قىزنى ساقلاپ قالامدۇ؟ ياكى ئۇنى توپا ئاستىغا (تىرىك) كۆمەمدۇ؟ (شۇ ھەقتە ئويلىنىدۇ) ئۇلارنىڭ ھۆكمى (يەنى ئوغۇللارنى ئۆزلىرىگە نىسبەت بېرىپ، قىزلارنى اﷲ قا نىسبەت بېرىشى) ئىشەنچلىك ھالدا نېمىدېگەن قەبىھ!
لِلَّذِينَ لَا يُؤْمِنُونَ بِالْآخِرَةِ مَثَلُ السَّوْءِ ۖ وَلِلَّهِ الْمَثَلُ الْأَعْلَىٰ ۚ وَهُوَ الْعَزِيزُ الْحَكِيمُ ﴿٦٠﴾
ئاخىرەتكە ئىشەنمەيدىغانلار (ئوغۇلنى ياخشى كۆرۈپ، قىزنى يامان كۆرۈش، خوتۇنى ئوغۇل تۇغسا خۇشال بولۇپ، قىز تۇغسا ئاچچىقى كېلىش، ئار - نومۇستىن ۋە كەمبەغەللىكتىن قورقۇپ، قىزلارنى تىرىك كۈمۈشكە ئوخشاش) يامان سۈپەتلەرگە ئىگە، اﷲ بولسا ئەڭ ئالىي خىسلەتلەرگە ئىگە، اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
وَلَوْ يُؤَاخِذُ اللَّهُ النَّاسَ بِظُلْمِهِم مَّا تَرَكَ عَلَيْهَا مِن دَابَّةٍ وَلَٰكِن يُؤَخِّرُهُمْ إِلَىٰ أَجَلٍ مُّسَمًّى ۖ فَإِذَا جَاءَ أَجَلُهُمْ لَا يَسْتَأْخِرُونَ سَاعَةً ۖ وَلَا يَسْتَقْدِمُونَ ﴿٦١﴾
ئەگەر اﷲ ئىنسانلارنى ئۇلارنىڭ زۇلمى تۈپەيلىدىن (چاپسان) جازالايدىغان بولسا، يەر يۈزىدە قىمىرلاپ تۇرىدىغان ھېچبىر جان ئىگىسىنى قويمىغان بولاتتى، لېكىن اﷲ ئۇلارنى مۇئەييەن ۋاقىتقىچە تەخىر قىلىدۇ، ئۇلارنىڭ (ھالاك بولۇش) ۋاقتى كەلگەندە قىلچە ۋاقىتمۇ كېچىكتۈرۈلمەيدۇ، (ۋاقتى كەلمىسە) قىلچە ۋاقىتمۇ ئىلگىرى سۈرۈلمەيدۇ
وَيَجْعَلُونَ لِلَّهِ مَا يَكْرَهُونَ وَتَصِفُ أَلْسِنَتُهُمُ الْكَذِبَ أَنَّ لَهُمُ الْحُسْنَىٰ ۖ لَا جَرَمَ أَنَّ لَهُمُ النَّارَ وَأَنَّهُم مُّفْرَطُونَ ﴿٦٢﴾
ئۇلار ئۆزلىرى يامان كۆرۈدىغاننى اﷲ قا مەنسۇپ قىلىدۇ، ئۇلار ئاقىۋىتىنىڭ ياخشى بولىدىغانلىقى بىلەن جۆيلۈيدۇ، شۈبھىسىزكى، ئۇلار دوزاخ ئازابىغا دۇچار بولىدۇ، ئۇلار (دوزاختا) تاشلىنىپ قالغۇچىلاردۇر
تَاللَّهِ لَقَدْ أَرْسَلْنَا إِلَىٰ أُمَمٍ مِّن قَبْلِكَ فَزَيَّنَ لَهُمُ الشَّيْطَانُ أَعْمَالَهُمْ فَهُوَ وَلِيُّهُمُ الْيَوْمَ وَلَهُمْ عَذَابٌ أَلِيمٌ ﴿٦٣﴾
(ئى مۇھەممەد!) اﷲ نامى بىلەن قەسەمكى، سەندىن ئىلگىرى ئۆتكەن ئۈممەتلەرگە ئىشەنچلىك ھالدا (خۇدانىڭ ئەلچى-رەسۇللىرى) ئەۋەتتۇق، شەيتان (ئازدۇرغۇچى) ئۇلارنىڭ ئەمەللىرىنى ئۆزلىرىگە چىرايلىق كۆرسەتتى، شەيتان (ئازدۇرغۇچى) بۈگۈن ئۇلارنىڭ ياردەمچىسىدۇر. ئۇلار ئاخىرەتتە قاتتىق ئازابقا دۇچار بولىدۇ
وَمَا أَنزَلْنَا عَلَيْكَ الْكِتَابَ إِلَّا لِتُبَيِّنَ لَهُمُ الَّذِي اخْتَلَفُوا فِيهِ ۙ وَهُدًى وَرَحْمَةً لِّقَوْمٍ يُؤْمِنُونَ ﴿٦٤﴾
(ئى مۇھەممەد!) بىز مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) پەقەت سېنىڭ كىشىلەرگە ئۇلار (دىنىدىكى) دەتالاش قىلىشقان نەرسىلەرنى بايان قىلىپ بېرىشىڭ ئۈچۈن، ئىمان ئېيتىدىغان قەۋمگە ھىدايەت ۋە رەھمەت بولسۇن ئۈچۈنلا نازىل قىلدۇق
وَاللَّهُ أَنزَلَ مِنَ السَّمَاءِ مَاءً فَأَحْيَا بِهِ الْأَرْضَ بَعْدَ مَوْتِهَا ۚ إِنَّ فِي ذَٰلِكَ لَآيَةً لِّقَوْمٍ يَسْمَعُونَ ﴿٦٥﴾
اﷲ بۇلۇتتىن يامغۇر ياغدۇرۇپ، شۇنىڭ بىلەن قاغجىراپ قالغان زېمىندا (تۈرلۈك زىرائەتلەرنى) ئۈندۈردى، بۇنىڭدا (ۋەز - نەسىھەت) ئاڭلايدىغان قەۋم ئۈچۈن، ئەلۋەتتە، (اﷲ نىڭ قۇدرىتىنىڭ چوڭلۇقىنى كۆرسىتىدىغان) دەلىللەر بار
وَإِنَّ لَكُمْ فِي الْأَنْعَامِ لَعِبْرَةً ۖ نُّسْقِيكُم مِّمَّا فِي بُطُونِهِ مِن بَيْنِ فَرْثٍ وَدَمٍ لَّبَنًا خَالِصًا سَائِغًا لِّلشَّارِبِينَ ﴿٦٦﴾
چاھار پايلاردا سىلەر ئۈچۈن ئىشەنچلىك ھالدا بىر ئىبرەت بار. سىلەرنى چاھارپايلارنىڭ قارنىدىكى ماياق - تېزەكتىن ۋە قاندىن ئايرىلىپ چىققان پاكىز ۋە تەملىك سۈت بىلەن سۇغىرىمىز
وَمِن ثَمَرَاتِ النَّخِيلِ وَالْأَعْنَابِ تَتَّخِذُونَ مِنْهُ سَكَرًا وَرِزْقًا حَسَنًا ۗ إِنَّ فِي ذَٰلِكَ لَآيَةً لِّقَوْمٍ يَعْقِلُونَ ﴿٦٧﴾
(سىلەر ئۈچۈن يەنە بۇنداق ئىبرەتمۇ باركى) خورمىدىن، ئۈزۈمدىن شارابلار ۋە پاكىز يېمەكلىكلەر ياسايسىلەر، ئەقىلنى ئىشلىتىدىغان قەۋم ئۈچۈن بۇنىڭدا ئىشەنچلىك ھالدا اﷲ نىڭ بىرلىكىنى كۆرسىتىدىغان دەلىل بار
وَأَوْحَىٰ رَبُّكَ إِلَى النَّحْلِ أَنِ اتَّخِذِي مِنَ الْجِبَالِ بُيُوتًا وَمِنَ الشَّجَرِ وَمِمَّا يَعْرِشُونَ ﴿٦٨﴾
پەرۋەردىگارىڭ ھەسەل ھەرىسىگە ئىلھام بىلەن بىلدۈردى: «تاغلارغا، دەرەخلەرگە ۋە (كىشىلەرنىڭ ھەسەل ھەرىلىرى ئۈچۈن) ياسىغان ئۆيلىرىگە ئۇۋا تۈزگىن
ثُمَّ كُلِي مِن كُلِّ الثَّمَرَاتِ فَاسْلُكِي سُبُلَ رَبِّكِ ذُلُلًا ۚ يَخْرُجُ مِن بُطُونِهَا شَرَابٌ مُّخْتَلِفٌ أَلْوَانُهُ فِيهِ شِفَاءٌ لِّلنَّاسِ ۗ إِنَّ فِي ذَٰلِكَ لَآيَةً لِّقَوْمٍ يَتَفَكَّرُونَ ﴿٦٩﴾
ئاندىن كېيىن، تۈرلۈك مېۋىلەر (يەنى گۈل - چېچەكلەر) دىن يېگىن، پەرۋەردىگارىڭنىڭ يوللىرىغا ئىتائەتمەنلىك بىلەن كىرگىن»، ئۇنىڭ (يەنى ھەسەل ھەرىسىنىڭ) قارنىدىن ئىنسانلارغا شىپا بولىدىغان خىلمۇ - خىل رەڭلىك ئىچىملىك (يەنى ھەسەل) چىقىدۇ. بۇنىڭدا (اﷲ نىڭ قۇدرىتىنىڭ چوڭلۇقىنى) پىكىر قىلىدىغان قەۋم ئۈچۈن ئىبرەت بار
وَاللَّهُ خَلَقَكُمْ ثُمَّ يَتَوَفَّاكُمْ ۚ وَمِنكُم مَّن يُرَدُّ إِلَىٰ أَرْذَلِ الْعُمُرِ لِكَيْ لَا يَعْلَمَ بَعْدَ عِلْمٍ شَيْئًا ۚ إِنَّ اللَّهَ عَلِيمٌ قَدِيرٌ ﴿٧٠﴾
اﷲ سىلەرنى ياراتتى، ئاندىن (ئەجىلىڭلار پۈتكەندە) سىلەرنى قەبزى روھ قىلىدۇ، ئاراڭلاردا ئۆمۈرنىڭ يامان (يەنى ئەڭ قېرىلىق) باسقۇچىغا ئۇلىشىدىغانلارمۇ بار، ئۇلار ھەتتا (بىر نەرسىلەرنى) بىلگەندىن كېيىن، ھېچ نەرسىنى بىلمەس بولۇپ قالىدۇ (يەنى ئىلگىرى بىلىدىغانلىرىنى ئۇنتۇپ، كۈچ - قۇۋۋىتى تۈگەپ، ئەقلى كېمىيىپ كىچىك بالىغا ئوخشاپ قالىدۇ). اﷲ ئىشەنچلىك ھالدا (مەخلۇقاتنى باشقۇرۇشنى) بىلگۈچىدۇر، (ئۆزى ئىرادە قىلغان نەرسىنى ئىشقا ئاشۇرۇشقا) قادىردۇر
وَاللَّهُ فَضَّلَ بَعْضَكُمْ عَلَىٰ بَعْضٍ فِي الرِّزْقِ ۚ فَمَا الَّذِينَ فُضِّلُوا بِرَادِّي رِزْقِهِمْ عَلَىٰ مَا مَلَكَتْ أَيْمَانُهُمْ فَهُمْ فِيهِ سَوَاءٌ ۚ أَفَبِنِعْمَةِ اللَّهِ يَجْحَدُونَ ﴿٧١﴾
اﷲ سىلەرنىڭ بەزىڭلارنى بەزىڭلاردىن رىزىقتا ئۈستۈن قىلدى، رىزىقتا ئۈستۈن قىلىنغانلار ئۆزلىرىنىڭ رىزقىنى باراۋەر بولۇش ئۈچۈن قول ئاستىدىكى قۇللارغا بەرمەيدۇ، ئۇلار اﷲ نىڭ نېمىتىنى ئىنكار قىلىشامدۇ؟
وَاللَّهُ جَعَلَ لَكُم مِّنْ أَنفُسِكُمْ أَزْوَاجًا وَجَعَلَ لَكُم مِّنْ أَزْوَاجِكُم بَنِينَ وَحَفَدَةً وَرَزَقَكُم مِّنَ الطَّيِّبَاتِ ۚ أَفَبِالْبَاطِلِ يُؤْمِنُونَ وَبِنِعْمَتِ اللَّهِ هُمْ يَكْفُرُونَ ﴿٧٢﴾
اﷲ سىلەر (نىڭ ئۇنى - ئۈلپەت ئېلىشىڭلار) ئۈچۈن خوتۇنۇڭلارنى ئۆز تىپىڭلاردىن ياراتتى، سىلەر ئۈچۈن ئۇلاردىن ئوغۇللارنى ۋە نەۋرىلەرنى ياراتتى. سىلەرگە تۈرلۈك شېرىن نەرسىلەرنى رىزىق قىلىپ بەردى. ئۇلار باتىلغا (يەنى بۇتلارغا) ئىشىنىپ اﷲ نىڭ نېمىتىگە كۇفرىلىق قىلامدۇ؟
وَيَعْبُدُونَ مِن دُونِ اللَّهِ مَا لَا يَمْلِكُ لَهُمْ رِزْقًا مِّنَ السَّمَاوَاتِ وَالْأَرْضِ شَيْئًا وَلَا يَسْتَطِيعُونَ ﴿٧٣﴾
ئۇلار اﷲ نى قويۇپ، ئۆزلىرىگە ئاسمانلاردىن ھېچقانداق رىزىق چۈشۈرۈپ ۋە يەردىن ھېچقانداق رىزىق چىقىرىپ (يەنى ئاسماندىن يامغۇر ياغدۇرۇپ ۋە يەردىن زىرائەت ۋە دەرەخلەرنى ئۆستۈرۈپ) بېرەلمەيدىغان، (شۇنداق قىلىشقا) قۇدرىتى يەتمەيدىغان بۇتلارغا چوقۇنامدۇ؟
فَلَا تَضْرِبُوا لِلَّهِ الْأَمْثَالَ ۚ إِنَّ اللَّهَ يَعْلَمُ وَأَنتُمْ لَا تَعْلَمُونَ ﴿٧٤﴾
سىلەر اﷲ قا مىسال كەلتۇرمەڭلار (يەنى اﷲ قا بۇتلارنى ئوخشاتماڭلار، اﷲ قا شېرىك كەلتۈرمەڭلار). اﷲ ئىشەنچلىك ھالدا (ھەممە ھەقىقەتنى) بىلىدۇ، سىلەر (اﷲ نىڭ بۈيۈكلۈكىنى) بىلمەيسىلەر
ضَرَبَ اللَّهُ مَثَلًا عَبْدًا مَّمْلُوكًا لَّا يَقْدِرُ عَلَىٰ شَيْءٍ وَمَن رَّزَقْنَاهُ مِنَّا رِزْقًا حَسَنًا فَهُوَ يُنفِقُ مِنْهُ سِرًّا وَجَهْرًا ۖ هَلْ يَسْتَوُونَ ۚ الْحَمْدُ لِلَّهِ ۚ بَلْ أَكْثَرُهُمْ لَا يَعْلَمُونَ ﴿٧٥﴾
اﷲ بۇنداق بىر مىسال كەلتۈرىدۇ: ھېچقانداق ئىشنى ئۆز ئالدىغا بىر تەرەپ قىلالمايدىغان بىر قۇل بىز ئوبدان رىزىق (يەنى مال - مۈلۈك) ئاتا قىلغان ۋە ئۇنى يوشۇرۇن ياكى ئاشكارا يوسۇندا سەدىقە قىلىپ تۇرغان (ھۆر - ئەركىن) ئادەم بىلەن ئوخشاشمۇ؟ (يەنى ھېچ نەرسىگە ئىگە بولمىغان قۇلغا ئوخشايدىغان بۇتلارنى ھەممىگە ئىگە بولغان ۋە كائىناتنى خالىغانچە تەسەررۇپ قىلىدىغان اﷲ بىلەن قانداقمۇ تەڭلەشتۈرگىلى بولسۇن؟) جىمى ھەمدۇسانا اﷲ قا خاستۇر! ئۇلار (يەنى ھەقىقەتنى رەت قىلغۇچىلار) نىڭ تولىسى (ھەمدۇساناغا اﷲ لايىق ئىكەنلىكىنى، بۇتلارنىڭ ھەمدۇساناغا ۋە ئىبادەتكە لايىق ئەمەسلىكىنى) ئۇقمايدۇ
وَضَرَبَ اللَّهُ مَثَلًا رَّجُلَيْنِ أَحَدُهُمَا أَبْكَمُ لَا يَقْدِرُ عَلَىٰ شَيْءٍ وَهُوَ كَلٌّ عَلَىٰ مَوْلَاهُ أَيْنَمَا يُوَجِّههُّ لَا يَأْتِ بِخَيْرٍ ۖ هَلْ يَسْتَوِي هُوَ وَمَن يَأْمُرُ بِالْعَدْلِ ۙ وَهُوَ عَلَىٰ صِرَاطٍ مُّسْتَقِيمٍ ﴿٧٦﴾
اﷲ يەنە مۇنداق مىسال كەلتۈرىدۇ؛ مۇنداق ئىككى ئادەم بار بولۇپ، ئۇلارنىڭ بىرى ھېچ ئىش قىلالمايدىغان ۋە ئىگىسىگە يۈك بولۇپ قالغان بىر گاچادۇركى، ئىگىسى بۇ گاچىنى مەيلى قەيەرگە ئەۋەتسۇن، ئۇ ھېچقانداق پايدا ئېلىپ كېلەلمەيدۇ، يەنە بىرى بولسا ئۆزى توغرا يولدا ماڭغان ۋە (كىشىلەرنى) ھەققانىي بولۇشقا بۇيرۇيدىغان ئادەمدۇر، بۇ ئىككى ئادەم ئوخشاشمۇ؟ (يەنى بۇ گاچا بىلەن ئۆزى توغرا يولدا تۇرۇپ پاساھەتلىك سۆزلەيدىغان، ئۆزى قۇرئان نۇرى بىلەن نۇرلانغان ئادەم بىر - بىرىگە ئوخشامدۇ؟ ئەقىللىق ئادەمنىڭ ئۇلارنى ئوخشاش ئورۇنغا قويمىغىنىدەك، بۇت ۋە تاشنى ھەممىنى بىلگۈچى قۇدرەتلىك اﷲ بىلەن ئوخشاش ئورۇنغا قويغىلى بولمايدۇ)
وَلِلَّهِ غَيْبُ السَّمَاوَاتِ وَالْأَرْضِ ۚ وَمَا أَمْرُ السَّاعَةِ إِلَّا كَلَمْحِ الْبَصَرِ أَوْ هُوَ أَقْرَبُ ۚ إِنَّ اللَّهَ عَلَىٰ كُلِّ شَيْءٍ قَدِيرٌ ﴿٧٧﴾
ئاسمانلاردىكى ۋە زېمىندىكى غەيبنى بىلىش اﷲ قا خاستۇر، قىيامەتنىڭ قايىم بولۇشى كۆزنى يۇتۇپ ئاچقۇچىلىق ياكى ئۇنىڭدىنمۇ تېزدۇر. اﷲ ئىشەنچلىك ھالدا ھەممە نەرسىگە قادىردۇر
وَاللَّهُ أَخْرَجَكُم مِّن بُطُونِ أُمَّهَاتِكُمْ لَا تَعْلَمُونَ شَيْئًا وَجَعَلَ لَكُمُ السَّمْعَ وَالْأَبْصَارَ وَالْأَفْئِدَةَ ۙ لَعَلَّكُمْ تَشْكُرُونَ ﴿٧٨﴾
اﷲ سىلەرنى ئاناڭلارنىڭ قارنىدىن ھېچ نەرسىنى بىلمەيدىغان ھالىتىڭلار بىلەن چىقاردى، اﷲ سىلەرگە شۈكۈر قىلسۇن دەپ، قۇلاق، كۆز ۋە يۈرەكلەرنى ئاتا قىلدى
أَلَمْ يَرَوْا إِلَى الطَّيْرِ مُسَخَّرَاتٍ فِي جَوِّ السَّمَاءِ مَا يُمْسِكُهُنَّ إِلَّا اللَّهُ ۗ إِنَّ فِي ذَٰلِكَ لَآيَاتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٧٩﴾
ئۇلار قۇشلارنىڭ ھاۋادا ئۇچۇشقا بويسۇندۇرۇلغانلىقىنى كۆرمىدىمۇ؟ ئۇلارنى (ھاۋادىن چۈشۈپ كېتىشتىن) پەقەت اﷲ تۇتۇپ تۇرىدۇ، بۇنىڭدا ئىمان ئېيتىدىغان قەۋم ئۈچۈن (اﷲ نىڭ بىرلىكىنى كۆرسىتىدىغان) نۇرغۇن ئالامەتلەر بار
وَاللَّهُ جَعَلَ لَكُم مِّن بُيُوتِكُمْ سَكَنًا وَجَعَلَ لَكُم مِّن جُلُودِ الْأَنْعَامِ بُيُوتًا تَسْتَخِفُّونَهَا يَوْمَ ظَعْنِكُمْ وَيَوْمَ إِقَامَتِكُمْ ۙ وَمِنْ أَصْوَافِهَا وَأَوْبَارِهَا وَأَشْعَارِهَا أَثَاثًا وَمَتَاعًا إِلَىٰ حِينٍ ﴿٨٠﴾
اﷲ سىلەرگە ئۆيۈڭلارنى تۇرالغۇ جاي قىلىپ بەردى، سىلەرگە ھايۋانلارنىڭ تېرىلىرىدىن ئۆيلەرنى (چېدىرلەرنى) قىلىپ بەردى، سىلەر كۆچكەن كۈنۈڭلاردىمۇ، تۇرغان كۈنۈڭلاردىمۇ ئۇنىڭ يىنىكلىكىنى ھېس قىلىسىلەر، سىلەرگە قويلارنىڭ يۈكلىرىدىن، تۆگىلەرنىڭ يۇڭلىرىدىن، ئۆچكىلەرنىڭ يۇڭلىرىدىن ئۆي جاھازلىرىنى (يەنى سىلەر كىيىدىغان ۋە ئۆيۈڭلارنى سەرەمجانلاشتۇرىدىغان) ھەم مەلۇم مۇددەتكىچە (ئۆلۈم كەلگۈچە ياكى ئۇلار كونىراپ تۈگىگىچە) پايدىلىنىدىغان نەرسىلەرنى قىلىپ بەردى
وَاللَّهُ جَعَلَ لَكُم مِّمَّا خَلَقَ ظِلَالًا وَجَعَلَ لَكُم مِّنَ الْجِبَالِ أَكْنَانًا وَجَعَلَ لَكُمْ سَرَابِيلَ تَقِيكُمُ الْحَرَّ وَسَرَابِيلَ تَقِيكُم بَأْسَكُمْ ۚ كَذَٰلِكَ يُتِمُّ نِعْمَتَهُ عَلَيْكُمْ لَعَلَّكُمْ تُسْلِمُونَ ﴿٨١﴾
اﷲ سىلەرگە ئۆزى ياراتقان نەرسىلەر (يەنى دەرەخلەر، تاغلار ۋە بىنالار) دىن (كۈننىڭ ھارارىتىدىن ساقلىنىدىغان) سايىلەرنى قىلىپ بەردى. تاغلاردىن سىلەرگە (غار، ئۆڭكۈرگە ئوخشاش) تۇرالغۇ قىلىدىغان جايلارنى قىلىپ بەردى. سىلەرگە ئىسسىقتىن (ۋە سوغۇقتىن) ساقلىنىدىغان كىيىملەرنى، ئۇرۇشتا سىلەرنى (زىيان - زەخمەتتىن) ساقلايدىغان تۆمۈر كىيىملەرنى تەييار قىلدى. اﷲ سىلەرنى ئىتائەت قىلسۇن دەپ، سىلەرگە ئەنە شۇنداق مۇكەممەل نېمەتلەرنى بېرىدۇ
فَإِن تَوَلَّوْا فَإِنَّمَا عَلَيْكَ الْبَلَاغُ الْمُبِينُ ﴿٨٢﴾
ئەگەر ئۇلار (يەنى مۇشرىكلار ئىماندىن) يۈز ئۆرۈسە (ساڭا ھېچ زىيىنى يوق)، سېنىڭ ۋەزىپەڭ پەقەت روشەن تەبلىغ قىلىشتۇر
يَعْرِفُونَ نِعْمَتَ اللَّهِ ثُمَّ يُنكِرُونَهَا وَأَكْثَرُهُمُ الْكَافِرُونَ ﴿٨٣﴾
ئۇلار اﷲ نىڭ نېمىتىنى تونۇيدۇ، ئاندىن (اﷲ تىن بۆلەك غەيرىيگە ئىبادەت قىلىش بىلەن) ئۇ نېمەتلەرنى ئىنكار قىلىدۇ، ئۇلارنىڭ تولىسى ھەقىقەتنى رەت قىلغۇچىدۇر (يەنى ھەقىقەتنى رەت قىلغۇچى پېتىچە ئۆلىدۇ)
وَيَوْمَ نَبْعَثُ مِن كُلِّ أُمَّةٍ شَهِيدًا ثُمَّ لَا يُؤْذَنُ لِلَّذِينَ كَفَرُوا وَلَا هُمْ يُسْتَعْتَبُونَ ﴿٨٤﴾
شۇ كۈندە (يەنى قىيامەت كۈنىدە) ھەر ئۈممەتتىن (ئۇنىڭغا ئىمان بىلەن ياكى كۇفرى بىلەن گۇۋاھلىق بېرىدىغان) بىر شاھىت (يەنى خۇدانىڭ ئەلچىسى) كەلتۈرىمىز، ئاندىن ھەقىقەتنى رەت قىلغۇچىلارغا (ئۆزرە ئېيتىشقا) رۇخسەت قىلىنمايدۇ ھەمدە ئۇلاردىن پەرۋەردىگارىنى رازى قىلىش (يەنى پەرۋەردىگارىنى رازى قىلىدىغان ئەمەللەرنى قىلىش) تەلەپ قىلىنمايدۇ
وَإِذَا رَأَى الَّذِينَ ظَلَمُوا الْعَذَابَ فَلَا يُخَفَّفُ عَنْهُمْ وَلَا هُمْ يُنظَرُونَ ﴿٨٥﴾
زالىملار (يەنى مۇشرىكلار) ئازابنى كۆرگەن چاغدا، ئۇلاردىن (ئازاب) يىنىكلىتىلمەيدۇ ۋە كېچىكتۈرۈلمەيدۇ
وَإِذَا رَأَى الَّذِينَ أَشْرَكُوا شُرَكَاءَهُمْ قَالُوا رَبَّنَا هَٰؤُلَاءِ شُرَكَاؤُنَا الَّذِينَ كُنَّا نَدْعُو مِن دُونِكَ ۖ فَأَلْقَوْا إِلَيْهِمُ الْقَوْلَ إِنَّكُمْ لَكَاذِبُونَ ﴿٨٦﴾
مۇشرىكلار (دۇنيادا چوقۇنغان) بۇتلىرىنى كۆرگەن چاغدا: «ئى پەرۋەردىگارىمىز! بۇلار بولسا بىزنىڭ سېنى قويۇپ ئىبادەت قىلغان بۇتلىرىمىزدۇر» دەيدۇ، بۇتلىرى ئۇلارغا جاۋاب بېرىپ؛ «شۈبھىسىزكى، سىلەر ئىشەنچلىك ھالدا يالغانچىسىلەر» دەيدۇ
وَأَلْقَوْا إِلَى اللَّهِ يَوْمَئِذٍ السَّلَمَ ۖ وَضَلَّ عَنْهُم مَّا كَانُوا يَفْتَرُونَ ﴿٨٧﴾
شۇ كۈندە ئۇلار ئۆزلىرىنىڭ اﷲ قا باش ئەگكەنلىكىنى بىلدۈرىدۇ، ئەمما ئۇلارنىڭ (دۇنيادىكى چاغلىرىدا: «بۇتلار بىزگە شاپائەت قىلىدۇ» دېگەن) يالغان - ياۋىداقلىرى يوققا چىقىدۇ
الَّذِينَ كَفَرُوا وَصَدُّوا عَن سَبِيلِ اللَّهِ زِدْنَاهُمْ عَذَابًا فَوْقَ الْعَذَابِ بِمَا كَانُوا يُفْسِدُونَ ﴿٨٨﴾
ھەقىقەتنى رەت قىلغۇچى بولغانلارغا ۋە (كىشىلەرنى) اﷲ نىڭ يولىدىن (يەنى دىنىدىن) توسقانلارغا ئۇلارنىڭ قىلغان بۇزغۇنچىلىقلىرى ئۈچۈن ئازاب قىلىمىز
وَيَوْمَ نَبْعَثُ فِي كُلِّ أُمَّةٍ شَهِيدًا عَلَيْهِم مِّنْ أَنفُسِهِمْ ۖ وَجِئْنَا بِكَ شَهِيدًا عَلَىٰ هَٰؤُلَاءِ ۚ وَنَزَّلْنَا عَلَيْكَ الْكِتَابَ تِبْيَانًا لِّكُلِّ شَيْءٍ وَهُدًى وَرَحْمَةً وَبُشْرَىٰ لِلْمُسْلِمِينَ ﴿٨٩﴾
شۇ كۈندە بىز ھەر بىر ئۈممەتكە ئۇلارنىڭ ئىچىدىن ئۇلارغا قارشى (گۇۋاھلىق بېرىدىغان) بىر شاھىت (يەنى خۇدانىڭ ئەلچىسى) كەلتۈرىمىز ۋە (شۇ كۈندە) بىز سېنى ئۇلارغا (يەنى ئۈممىتىڭگە) شاھىت قىلىپ كەلتۈرىمىز. ساڭا بىز مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) ھەممە نەرسىنى (يەنى كىشىلەر موھتاج بولىدىغان دىنىي ئىشلارنىڭ ھەممىسىنى) چۈشەندۈرۈپ بېرىدىغان، (دىللارغا) ھىدايەت، (بەندىلەرگە) رەھمەت، مۇسۇلمانلارغا مەڭگۈلۈك سائادەت بىلەن خوش خەۋەر يەتكۈزىدىغان قىلىپ نازىل قىلدۇق
إِنَّ اللَّهَ يَأْمُرُ بِالْعَدْلِ وَالْإِحْسَانِ وَإِيتَاءِ ذِي الْقُرْبَىٰ وَيَنْهَىٰ عَنِ الْفَحْشَاءِ وَالْمُنكَرِ وَالْبَغْيِ ۚ يَعِظُكُمْ لَعَلَّكُمْ تَذَكَّرُونَ ﴿٩٠﴾
اﷲ ئىشەنچلىك ھالدا (كىشىلەر ئارىسىدا) ئادىل بولۇشقا، (جىمى خەلققە) ياخشىلىق قىلىشقا، خىش - ئەقرىبالارغا سىلە - رەھىم قىلىشقا بۇيرۇيدۇ، قەبىھ (سۆز - ھەرىكەتلەر) دىن، يامان ئىشلاردىن ۋە زۇلۇم قىلىشتىن توسىدۇ. نەسىھەتنى قوبۇل قىلسۇن دەپ، اﷲ سىلەرگە پەند - نەسىھەت قىلىدۇ
وَأَوْفُوا بِعَهْدِ اللَّهِ إِذَا عَاهَدتُّمْ وَلَا تَنقُضُوا الْأَيْمَانَ بَعْدَ تَوْكِيدِهَا وَقَدْ جَعَلْتُمُ اللَّهَ عَلَيْكُمْ كَفِيلًا ۚ إِنَّ اللَّهَ يَعْلَمُ مَا تَفْعَلُونَ ﴿٩١﴾
سىلەر ئەھدە تۈزۈشكەنلىرىڭلاردا، اﷲ نىڭ ئەھدىگە ۋاپا قىلىڭلار، قەسىمىڭلارنى (اﷲ نىڭ نامىنى تىلغا ئېلىپ) پۇختىلىغاندىن كېيىن بۇزماڭلار، چۈنكى سىلەر اﷲ نى گۇۋاھچى قىلدىڭلار. شۈبھىسىزكى، اﷲ قىلمىشىڭلارنى بىلىپ تۇرىدۇ
وَلَا تَكُونُوا كَالَّتِي نَقَضَتْ غَزْلَهَا مِن بَعْدِ قُوَّةٍ أَنكَاثًا تَتَّخِذُونَ أَيْمَانَكُمْ دَخَلًا بَيْنَكُمْ أَن تَكُونَ أُمَّةٌ هِيَ أَرْبَىٰ مِنْ أُمَّةٍ ۚ إِنَّمَا يَبْلُوكُمُ اللَّهُ بِهِ ۚ وَلَيُبَيِّنَنَّ لَكُمْ يَوْمَ الْقِيَامَةِ مَا كُنتُمْ فِيهِ تَخْتَلِفُونَ ﴿٩٢﴾
سىلەر پىششىق ئىگىرگەن يىپنى چۇۋۇپ پارچە - پارچە قىلىۋەتكەن خوتۇندەك بولماڭلار. بىر جامائە (سان ۋە مال - مۈلۈك جەھەتتە) يەنە بىر جامائەدىن كۆپ بولغانلىقى ئۈچۈن، قەسىمىڭلارنى ئالدامچىلىقنىڭ ۋاسىتىسى قىلىۋالىسىلەر (يەنى ئۆز ئارا ياردەملىشىش ئۈستىدە بىر جامائە بىلەن ئىتتىپاق تۈزۈشكەندىن كېيىن، سانى، پۇل - مېلى كۆپرەك يەنە بىر جامائەنى كۆرۈپلا ئىلگىرىكى جامائە بىلەن تۈزۈشكەن ئەھدىنى بۇزۇپ، كېيىنكى جامائە بىلەن ئەھدى تۈزىسلەر)، اﷲ سىلەرنى ئىشەنچلىك ھالدا بۇنىڭ (يەنى ۋەدىگە ۋاپا قىلىش ئەمرى) بىلەن سىنايدۇ، سىلەر دەتالاش قىلىشقان ھەق - ناھەقنى قىيامەت كۈنى سىلەرگە چوقۇم ئايرىپ بېرىدۇ
وَلَوْ شَاءَ اللَّهُ لَجَعَلَكُمْ أُمَّةً وَاحِدَةً وَلَٰكِن يُضِلُّ مَن يَشَاءُ وَيَهْدِي مَن يَشَاءُ ۚ وَلَتُسْأَلُنَّ عَمَّا كُنتُمْ تَعْمَلُونَ ﴿٩٣﴾
مۇبادا اﷲ خالىغان بولسا ئىدى، سىلەرنى (يەنى پۇتۇن ئىنسانلارنى) ئەلۋەتتە بىر ئۈممەت (يەنى بىر دىندا) قىلاتتى، لېكىن (اﷲ نىڭ ھېكمىتى ئۇلارنى ئۆز ئىختىيارىغا قويۇۋېتىشنى تەقەززا قىلدى)، اﷲ خالىغان ئادەمنى گۇمراھ قىلىدۇ، خالىغان ئادەمنى ھىدايەت قىلىدۇ، (قىيامەت كۈنى) پۈتۈن قىلمىشلار ئۈستىدە سوراق قىلىنىسىلەر
وَلَا تَتَّخِذُوا أَيْمَانَكُمْ دَخَلًا بَيْنَكُمْ فَتَزِلَّ قَدَمٌ بَعْدَ ثُبُوتِهَا وَتَذُوقُوا السُّوءَ بِمَا صَدَدتُّمْ عَن سَبِيلِ اللَّهِ ۖ وَلَكُمْ عَذَابٌ عَظِيمٌ ﴿٩٤﴾
قەدىمىڭلار (ئىسلام دىنىدا) مۇستەھكەم، ئىزچىل بولغاندىن كېيىن تېيىلىپ كەتمەسلىك ئۈچۈن، قەسىمىڭلارنى ئۆز ئاراڭلاردا ئالدامچىلىقنىڭ ۋاسىتىسى قىلىۋالماڭلار، (كىشىلەرنى) اﷲ نىڭ يولىدىن توسقانلىقىڭلار (يەنى ئەھدىگە ۋاپا قىلىشتىن يۈز ئۆرۈگەنلىكىڭلار) ئۈچۈن ئازابنى تېتىيسىلەر، (سىلەر ئاخىرەتتە دوزاختا) چوڭ ئازابقا قالىسىلەر
وَلَا تَشْتَرُوا بِعَهْدِ اللَّهِ ثَمَنًا قَلِيلًا ۚ إِنَّمَا عِندَ اللَّهِ هُوَ خَيْرٌ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿٩٥﴾
سىلەر اﷲ نىڭ ئەھدىنى ئەزىمەس نەرسىگە سېتىۋەتمەڭلار: سىلەر ئەگەر بىلسەڭلار، اﷲ نىڭ دەرگاھىدىكى ساۋاب سىلەر ئۈچۈن (پانى دۇنيانىڭ نەرسىلىرىدىن) ئارتۇقتۇر
مَا عِندَكُمْ يَنفَدُ ۖ وَمَا عِندَ اللَّهِ بَاقٍ ۗ وَلَنَجْزِيَنَّ الَّذِينَ صَبَرُوا أَجْرَهُم بِأَحْسَنِ مَا كَانُوا يَعْمَلُونَ ﴿٩٦﴾
(ئى ئىنسانلار!) سىلەرنىڭ ئىلكىڭلاردىكى نەرسىلەر تۈگەيدۇ، اﷲ نىڭ دەرگاھىدىكى نەرسىلەر تۈگىمەستۇر. سەۋر قىلغۇچىلارغا، ئەلۋەتتە، ئۇلارنىڭ قىلغان ئەمەلىدىنمۇ ياخشىراق ساۋاب بېرىمىز
مَنْ عَمِلَ صَالِحًا مِّن ذَكَرٍ أَوْ أُنثَىٰ وَهُوَ مُؤْمِنٌ فَلَنُحْيِيَنَّهُ حَيَاةً طَيِّبَةً ۖ وَلَنَجْزِيَنَّهُمْ أَجْرَهُم بِأَحْسَنِ مَا كَانُوا يَعْمَلُونَ ﴿٩٧﴾
ئەر - ئايال مۆمىنلەردىن كىمكى ياخشى ئەمەل قىلىدىكەن، بىز ئۇنى ئەلۋەتتە (دۇنيانىڭ قانائەتچانلىق، ھالال رىزىق ۋە ياخشى ئەمەللەرگە مۇۋەپپەق قىلىپ) ئوبدان ياشىتىمىز، ئۇلارغا، ئەلۋەتتە، قىلغان ئەمەلىدىنمۇ ياخشىراق ساۋاب بېرىمىز
فَإِذَا قَرَأْتَ الْقُرْآنَ فَاسْتَعِذْ بِاللَّهِ مِنَ الشَّيْطَانِ الرَّجِيمِ ﴿٩٨﴾
سەن قۇرئان ئوقۇماقچى بولغىنىڭدا، قوغلاندى شەيتان (ئازدۇرغۇچى)نىڭ (ۋەسۋەسىسىدىن) اﷲ قا سېغىنىپ پاناھ تىلىگىن
إِنَّهُ لَيْسَ لَهُ سُلْطَانٌ عَلَى الَّذِينَ آمَنُوا وَعَلَىٰ رَبِّهِمْ يَتَوَكَّلُونَ ﴿٩٩﴾
شۈبھىسىزكى، مۆمىنلەرنىڭ ۋە پەرۋەردىگارىغا تەۋەككۈل قىلغۇچىلار (يەنى يۆلەنگۈچىلەر) نىڭ ئۈستىدىن شەيتان (ئازدۇرغۇچى) ھۆكۈمرانلىق قىلالمايدۇ (يەنى پەرۋەردىگارىغا چىن يۆلەنگەن مۆمىنلەرگە شەيتان (ئازدۇرغۇچى)نىڭ ۋەسۋەسىسى تەسىر قىلالمايدۇ)
إِنَّمَا سُلْطَانُهُ عَلَى الَّذِينَ يَتَوَلَّوْنَهُ وَالَّذِينَ هُم بِهِ مُشْرِكُونَ ﴿١٠٠﴾
شەيتان (ئازدۇرغۇچى) پەقەت ئۆزىنى دوست تۇتىدىغانلار ۋە ئۆزىنىڭ (ئىغۋا قىلىشى) بىلەن مۇشرىك بولۇپ كەتكەنلەر ئۈستىدىنلا ھۆكۈمرانلىق قىلىدۇ
وَإِذَا بَدَّلْنَا آيَةً مَّكَانَ آيَةٍ ۙ وَاللَّهُ أَعْلَمُ بِمَا يُنَزِّلُ قَالُوا إِنَّمَا أَنتَ مُفْتَرٍ ۚ بَلْ أَكْثَرُهُمْ لَا يَعْلَمُونَ ﴿١٠١﴾
بىز بىر ئايەتنى يەنە بىر ئايەتنىڭ ئورنىغا ئالماشتۇرغىنىمىزدا (يەنى بىر ھۆكۈمنى بىكار قىلىپ، ئۇنىڭ ئورنىغا باشقا بىر ھۆكۈمنى چۈشۈرگىنىمىزدە)، - اﷲ ئۆزى نازىل قىلغاننى (يەنى شۇ ھۆكۈمدە بەندىگە نېمە مەنپەئەت ۋە ھېكمەت بارلىقىنى) ئۆزى ئوبدان بىلىدۇ - ئۇلار: «(ئى مۇھەممەد!) سەن (اﷲ قا) يالغان چاپلىدىڭ» دەيدۇ، ئۇنداق ئەمەس، ئۇلارنىڭ تولىسى (اﷲ نىڭ ھېكمىتىنى) ئۇقمايدۇ
قُلْ نَزَّلَهُ رُوحُ الْقُدُسِ مِن رَّبِّكَ بِالْحَقِّ لِيُثَبِّتَ الَّذِينَ آمَنُوا وَهُدًى وَبُشْرَىٰ لِلْمُسْلِمِينَ ﴿١٠٢﴾
ئېيتقىنكى، قۇرئاننى جىبرىئىل مۆمىنلەرنىڭ (ئىمانىنى) مۇستەھكەم قىلىش ئۈچۈن، مۇسۇلمانلارغا ھىدايەت ۋە خۇش خەۋەر قىلىپ، پەرۋەردىگارىڭ ھۇزۇرىدىن راستچىللىق بىلەن نازىل قىلدى
وَلَقَدْ نَعْلَمُ أَنَّهُمْ يَقُولُونَ إِنَّمَا يُعَلِّمُهُ بَشَرٌ ۗ لِّسَانُ الَّذِي يُلْحِدُونَ إِلَيْهِ أَعْجَمِيٌّ وَهَٰذَا لِسَانٌ عَرَبِيٌّ مُّبِينٌ ﴿١٠٣﴾
ئۇلار (يەنى مۇشرىكلار) نىڭ: «ئۇنىڭغا (يەنى مۇھەممەد ئەلەيھىسسالامغا) قۇرئاننى بىر ئىنسان تەلىم بېرىدۇ» دېگەن سۆزىنى ئوبدان بىلىمىز، ئۇلار كۆرسەتكەن ئادەمنىڭ تىلى ئەرەب تىلى ئەمەس، بۇ (قۇرئان) بولسا روشەن ئەرەب تىلىدۇر
إِنَّ الَّذِينَ لَا يُؤْمِنُونَ بِآيَاتِ اللَّهِ لَا يَهْدِيهِمُ اللَّهُ وَلَهُمْ عَذَابٌ أَلِيمٌ ﴿١٠٤﴾
شۈبھىسىزكى، اﷲ نىڭ ئايەتلىرىگە (يەنى قۇرئانغا) ئىشەنمەيدىغانلارنى اﷲ ھىدايەت قىلمايدۇ، ئۇلار (ئاخىرەتتە) قاتتىق ئازابقا دۇچار بولىدۇ
إِنَّمَا يَفْتَرِي الْكَذِبَ الَّذِينَ لَا يُؤْمِنُونَ بِآيَاتِ اللَّهِ ۖ وَأُولَٰئِكَ هُمُ الْكَاذِبُونَ ﴿١٠٥﴾
پەقەت اﷲ نىڭ ئايەتلىرىگە ئىشەنمەيدىغانلارلا (اﷲ قا) يالغان چاپلايدۇ، (مۇھەممەد ئەلەيھىسسالام) يالغانچى ئەمەس) دەل ئۇلارنىڭ ئۆزلىرى يالغانچىدۇر
مَن كَفَرَ بِاللَّهِ مِن بَعْدِ إِيمَانِهِ إِلَّا مَنْ أُكْرِهَ وَقَلْبُهُ مُطْمَئِنٌّ بِالْإِيمَانِ وَلَٰكِن مَّن شَرَحَ بِالْكُفْرِ صَدْرًا فَعَلَيْهِمْ غَضَبٌ مِّنَ اللَّهِ وَلَهُمْ عَذَابٌ عَظِيمٌ ﴿١٠٦﴾
كىمكى اﷲ قا ئىمان ئېيتقاندىن كېيىن ئىمانىدىن يېنىۋالسا، - قەلبى ئىمان بىلەن مۇستەھكەم تۇرسىمۇ مەجبۇرلاش ئاستىدا (ئاغزىدىلا) ئىماندىن يانغانلىقىنى بىلدۈرگەنلەر بۇنىڭدىن مۇستەسنا - كۇفرى بىلەن كۆڭلى ئازادە بولسا (يەنى ئىختىيارىي يوسۇندا مۇرتەد بولغان بولسا)، ئۇ اﷲ نىڭ غەزىپىگە دۇچار بولىدۇ ۋە چوڭ ئازابقا قالىدۇ
ذَٰلِكَ بِأَنَّهُمُ اسْتَحَبُّوا الْحَيَاةَ الدُّنْيَا عَلَى الْآخِرَةِ وَأَنَّ اللَّهَ لَا يَهْدِي الْقَوْمَ الْكَافِرِينَ ﴿١٠٧﴾
بۇ (يەنى ئازاب) ئۇلارنىڭ دۇنيا تىرىكچىلىكىنى ئاخىرەتتىن ئارتۇق كۆرگەنلىكلىرى ۋە اﷲ نىڭ ھەقىقەتنى رەت قىلغۇچى قەۋمنى ھىدايەت قىلمايدىغانلىقى ئۈچۈندۇر
أُولَٰئِكَ الَّذِينَ طَبَعَ اللَّهُ عَلَىٰ قُلُوبِهِمْ وَسَمْعِهِمْ وَأَبْصَارِهِمْ ۖ وَأُولَٰئِكَ هُمُ الْغَافِلُونَ ﴿١٠٨﴾

ثُمَّ إِنَّ رَبَّكَ لِلَّذِينَ هَاجَرُوا مِن بَعْدِ مَا فُتِنُوا ثُمَّ جَاهَدُوا وَصَبَرُوا إِنَّ رَبَّكَ مِن بَعْدِهَا لَغَفُورٌ رَّحِيمٌ ﴿١١٠﴾
ئاندىن (ئى مۇھەممەد!) زىيانكەشلىككە ئۇچرىغاندىن كېيىن، ئىشەنچلىك ھالدا (مەدىنىگە) ھىجرەت قىلغان. ئاندىن (اﷲ نىڭ يولىدا) جىھاد قىلغان ۋە (جىھادنىڭ مۇشەققەتلىرىگە) چىدىغانلارغا پەرۋەردىگارىڭ (مەغپىرەت قىلىدۇ)، شۈبھىسىزكى، پەرۋەردىگارىڭ شۇنىڭدىن (يەنى ئۇلارنىڭ ھىجرەت، جىھاد، سەۋر قىلغىنىدىن) كېيىن ئۇلارغا ئەلۋەتتە مەغپىرەت قىلغۇچىدۇر، رەھمەت قىلغۇچىدۇر
يَوْمَ تَأْتِي كُلُّ نَفْسٍ تُجَادِلُ عَن نَّفْسِهَا وَتُوَفَّىٰ كُلُّ نَفْسٍ مَّا عَمِلَتْ وَهُمْ لَا يُظْلَمُونَ ﴿١١١﴾
ھەر ئادەم ئۆزى بىلەن بولۇپ كېتىدىغان، ھەر ئادەم قىلغان ئەمەلىنىڭ مۇكاپاتى تولۇق بېرىلىدىغان، ئۇلارغا زۇلۇم قىلىنمايدىغان كۈننى (يەنى قىيامەت كۈنىنى) ئەسلىگىن

اﷲ مۇنداق بىر مىسالنى كەلتۈرىدۇ: بىر شەھەر (يەنى مەككە ئاھالىسى) بولۇپ، (ئەسلىدە) تىنچ - خاتىرجەم ئىدى. ئۇنىڭ رىزقى تەرەپ - تەرەپتىن كەڭتاشا كېلىپ تۇراتتى، (ئۇنىڭ ئاھالىسى) اﷲ نىڭ نېمەتلىرىگە كۇفرىلىق قىلدى، اﷲ ئۇلارغا قىلمىشلىرى تۈپەيلىدن ئاچلىقنىڭ ۋە قورقۇنچنىڭ ئەلىمىنى تېتىتتى
وَلَقَدْ جَاءَهُمْ رَسُولٌ مِّنْهُمْ فَكَذَّبُوهُ فَأَخَذَهُمُ الْعَذَابُ وَهُمْ ظَالِمُونَ ﴿١١٣﴾
ئۇلارغا ئۆز ئىچىدىن بىر خۇدانىڭ ئەلچىسى (يەنى مۇھەممەد ئەلەيھىسسالام) كەلدى، ئۇنى ئۇلار ئىنكار قىلدى. ئۇلارغا ئازاب (يەنى قاتتىقچىلىق ۋە ئوقۇبەت) يۈزلەندى، ئۇلار (گۇناھلارغا چۆمگەنلىك تۈپەيلىدىن) زالىم ئىدى
فَكُلُوا مِمَّا رَزَقَكُمُ اللَّهُ حَلَالًا طَيِّبًا وَاشْكُرُوا نِعْمَتَ اللَّهِ إِن كُنتُمْ إِيَّاهُ تَعْبُدُونَ ﴿١١٤﴾
اﷲ سىلەرگە رىزىق قىلىپ بەرگەن ھالال، شېرىن يېمەكلىكلەردىن يەڭلار، ئەگەر اﷲ غىلا ئىبادەت قىلىدىغان بولساڭلار، اﷲ نىڭ ھىممىتىگە شۈكۈر قىلىڭلار
إِنَّمَا حَرَّمَ عَلَيْكُمُ الْمَيْتَةَ وَالدَّمَ وَلَحْمَ الْخِنزِيرِ وَمَا أُهِلَّ لِغَيْرِ اللَّهِ بِهِ ۖ فَمَنِ اضْطُرَّ غَيْرَ بَاغٍ وَلَا عَادٍ فَإِنَّ اللَّهَ غَفُورٌ رَّحِيمٌ ﴿١١٥﴾
اﷲ سىلەرگە ئۆزى ئۆلۈپ قالغان ھايۋاننى، قاننى، چوشقا گۆشىنى، اﷲ تىن غەيرىنىڭ (يەنى بۇتلارنىڭ) نامى ئېيتىلىپ بوغۇزلانغان ھايۋاننى يېيىشنى ھارام قىلدى. كىمكى ئختىيارىچە ئەمەس، ئىلاجىسىزلىقتىن (يۇقىرىقى ھارام قىلىنغان نەرسىلەردىن) ھاياتىنى ساقلاپ قالغۇدەك مىقداردا (يېسە) اﷲ (ئۇنى) مەغپىرەت قىلغۇچىدۇر، (ئۇنىڭغا) كۆيۈنگۈچىدۇر
وَلَا تَقُولُوا لِمَا تَصِفُ أَلْسِنَتُكُمُ الْكَذِبَ هَٰذَا حَلَالٌ وَهَٰذَا حَرَامٌ لِّتَفْتَرُوا عَلَى اللَّهِ الْكَذِبَ ۚ إِنَّ الَّذِينَ يَفْتَرُونَ عَلَى اللَّهِ الْكَذِبَ لَا يُفْلِحُونَ ﴿١١٦﴾
ئاغزىڭلارغا كەلگەن يالغاننى سۆزلەش ئۈچۈن (ھېچقانداق دەلىلسىز) «بۇ ھالال، بۇ ھارام» دېمەڭلار، چۈنكى (مۇنداقتا) اﷲ نامىدىن يالغاننى ئويدۇرغان بولىسىلەر، اﷲ نامىدىن يالغاننى ئويدورغۇچىلار ئىشەنچلىك ھالدا (دۇنيا ئاخىرەتتە) مەقسىتىگە ئېرىشەلمەيدۇ
مَتَاعٌ قَلِيلٌ وَلَهُمْ عَذَابٌ أَلِيمٌ ﴿١١٧﴾
(ئۇلار دۇنيادىن پەقەت) ئازغىنا بەھرىمەن بولىدۇ، ئۇلار (ئاخىرەتتە) قاتتىق ئازابقا دۇچار بولىدۇ
وَعَلَى الَّذِينَ هَادُوا حَرَّمْنَا مَا قَصَصْنَا عَلَيْكَ مِن قَبْلُ ۖ وَمَا ظَلَمْنَاهُمْ وَلَٰكِن كَانُوا أَنفُسَهُمْ يَظْلِمُونَ ﴿١١٨﴾
ئىلگىرى ساڭا بايان قىلغان نەرسىلەرنى يالغۇز يەھۇدىيلارغا ھارام قىلدۇق، (بۇنىڭ بىلەن) بىز ئۇلارغا زۇلۇم قىلمىدۇق، لېكىن ئۇلار ئۆزلىرىگە ئۆزلىرى زۇلۇم قىلدى
ثُمَّ إِنَّ رَبَّكَ لِلَّذِينَ عَمِلُوا السُّوءَ بِجَهَالَةٍ ثُمَّ تَابُوا مِن بَعْدِ ذَٰلِكَ وَأَصْلَحُوا إِنَّ رَبَّكَ مِن بَعْدِهَا لَغَفُورٌ رَّحِيمٌ ﴿١١٩﴾
شۈبھىسىزكى، بىلمەستىن يامان ئىشلارنى قىلغاندىن كېيىن تەۋبە قىلغان ۋە (ئەمەلىنى) تۈزەتكەنلەرگە پەرۋەردىگارىڭ ئەلۋەتتە مەغپىرەت قىلغۇچىدۇر، مەرھەمەت قىلغۇچىدۇر
إِنَّ إِبْرَاهِيمَ كَانَ أُمَّةً قَانِتًا لِّلَّهِ حَنِيفًا وَلَمْ يَكُ مِنَ الْمُشْرِكِينَ ﴿١٢٠﴾
ئىبراھىم ئىشەنچلىك ھالدا (ياخشى خىسلەتلەرنى ئۆزلەشتۈرگۈچى) پېشىۋا، اﷲ قا ئىتائەتمەن (باتىل دىنلاردىن ھەق دىنغا بۇرالغۇچى يەنى ھەق دىنغا ئېتىقاد قىلغۇچى) ئىدى. ئۇ اﷲ قا شېرىك كەلتۈرگۈچىلەردىن ئەمەس ئىدى
شَاكِرًا لِّأَنْعُمِهِ ۚ اجْتَبَاهُ وَهَدَاهُ إِلَىٰ صِرَاطٍ مُّسْتَقِيمٍ ﴿١٢١﴾
ئىبراھىم اﷲ نىڭ نېمەتلىرىگە شۈكۈر قىلغۇچى ئىدى. اﷲ ئۇنى (خۇدانىڭ ئەلچىسىلىككە) تاللىدى ۋە ئۇنى توغرا يولغا باشلىدى
وَآتَيْنَاهُ فِي الدُّنْيَا حَسَنَةً ۖ وَإِنَّهُ فِي الْآخِرَةِ لَمِنَ الصَّالِحِينَ ﴿١٢٢﴾
ئۇنىڭغا دۇنيادا ياخشىلىق ئاتا قىلدۇق، شەك - شۈبھىسىزكى، ئاخىرەتتە ئۇ ياخشىلار قاتارىدا بولىدۇ
ثُمَّ أَوْحَيْنَا إِلَيْكَ أَنِ اتَّبِعْ مِلَّةَ إِبْرَاهِيمَ حَنِيفًا ۖ وَمَا كَانَ مِنَ الْمُشْرِكِينَ ﴿١٢٣﴾
ئاندىن ساڭا: «باتىل دىنلاردىن ھەق دىنغا بۇرالغۇچى (يەنى ھەق دىنغا ئېتىقاد قىلغۇچى) ئىبراھىمنىڭ دىنىغا ئەگەشكىن، ئىبراھىم مۇشرىكلاردىن ئەمەس ئىدى» دەپ ۋەھيى قىلدۇق

شەنبە كۈنى ئۈستىدە ئىختىلاپ قىلغۇچىلار ئۈچۈنلا ئۇنى ئۇلۇغلاش بەلگىلەندى (يەنى شەنبە كۈنىنى ئۇلۇغلاش ۋە بۇ كۈندە ئىشلىمەسلىك ئىبراھىمنىڭ شەرىئىتىدە يوق ئىدى. يەھۇدىيلار اﷲ نىڭ ئەمرىگە ئاسىيلىق قىلىپ دىندا ئىختىلاپ قىلىشقانلىقلىرى سەۋەبىدىن ئۇلارغا ئېغىرچىلىق تۇغدۇرۇش ئۈچۈن شۇنداق بەلگىلىمىنى چىقاردۇق). شۈبھىسىزكى، پەرۋەردىگارىڭ قىيامەت كۈنى ئۇلار ئىختىلاپ قىلىشقان ھەق - ناھەق ئۈستىدە چوقۇم ھۆكۈم چىقىرىدۇ

پەرۋەردىگارىڭنىڭ يولىغا (يەنى ئىسلام دىنىغا) ھېكمەتلىك ئۇسلۇبتا ياخشى ۋەز - نەسىھەت بىلەن دەۋەت قىلغىن، ئۇلار (يەنى مۇخالىپەتچىلىك قىلغۇچىلار) بىلەن چىرايلىق رەۋىشتە مۇنازىرىلەشكەن. پەرۋەردىگارىڭ ئىشەنچلىك ھالدا ئۇنىڭ يولىدىن ئازغانلارنى ئوبدان بىلىدۇ، ھىدايەت تاپقۇچىلارنىمۇ ئوبدان بىلىدۇ
