بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
حمٓ ﴿١﴾
حا، مىم
تَنزِيلُ ٱلْكِتَـٰبِ مِنَ ٱللَّهِ ٱلْعَزِيزِ ٱلْحَكِيمِ ﴿٢﴾
(بۇ قۇرئان) غالىب، ھېكمەت بىلەن ئىش قىلغۇچى اﷲ تەرىپىدىن نازىل قىلىنغاندۇر
مَا خَلَقْنَا ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَمَا بَيْنَهُمَآ إِلَّا بِٱلْحَقِّ وَأَجَلٍ مُّسَمًّى ۚ وَٱلَّذِينَ كَفَرُوا۟ عَمَّآ أُنذِرُوا۟ مُعْرِضُونَ ﴿٣﴾
ئاسمانلارنى، زېمىننى ۋە ئۇلارنىڭ ئارىسىدىكى نەرسىلەرنى ھەق ئاساستا مۇئەييەن مۇددەتكىچە (يەنى قىيامەتكىچە قالىدىغان) قىلىپ ياراتتۇق. ھەقىقەتنى رەت قىلغۇچىلار ئاگاھلاندۇرۇلغان نەرسىلەردىن يۈز ئۆرۈگۈچىدۇر
قُلْ أَرَءَيْتُم مَّا تَدْعُونَ مِن دُونِ ٱللَّهِ أَرُونِى مَاذَا خَلَقُوا۟ مِنَ ٱلْأَرْضِ أَمْ لَهُمْ شِرْكٌ فِى ٱلسَّمَـٰوَٰتِ ۖ ٱئْتُونِى بِكِتَـٰبٍ مِّن قَبْلِ هَـٰذَآ أَوْ أَثَـٰرَةٍ مِّنْ عِلْمٍ إِن كُنتُمْ صَـٰدِقِينَ ﴿٤﴾
(ئى مۇھەممەد! مۇشرىكلارغا) ئېيتقىنكى، «سىلەر اﷲ نى قويۇپ چوقۇنۇۋاتقان بۇتلىرىڭلارنىڭ (ئەھۋالىنى) ئېيتىپ بېرىڭلار، ئۇلار زېمىننىڭ قايسى قىسمىنى ياراتقانلىقىنى ماڭا كۆرسىتىڭلار، ياكى ئاسمانلارنى يارىتىشتا ئۇلارنىڭ اﷲ بىلەن ئورتاقچىلىقى بارمۇ؟ بۇ (قۇرئان) دىن ئىلگىرى (سىلەرگە نازىل قىلىنغان) مۇقەددەس كىتاب-ۋەھىي ياكى (ئىلگىرىكىلەردىن) قالغان ئىلىم بولسا ماڭا ئېلىپ كېلىڭلار، ئەگەر سىلەر راستچىل بولساڭلار»
وَمَنْ أَضَلُّ مِمَّن يَدْعُوا۟ مِن دُونِ ٱللَّهِ مَن لَّا يَسْتَجِيبُ لَهُۥٓ إِلَىٰ يَوْمِ ٱلْقِيَـٰمَةِ وَهُمْ عَن دُعَآئِهِمْ غَـٰفِلُونَ ﴿٥﴾
اﷲ نى قويۇپ قىيامەتكىچە دۇئانى ئىجابەت قىلىشقا قادىر بولالمايدىغان بۇتلارغا چوقۇنىدىغان ئادەمدىنمۇ گۇمراھ ئادەم بارمۇ؟ بۇتلار ئۇلارنىڭ دۇئاسىدىن غاپىلدۇر
وَإِذَا حُشِرَ ٱلنَّاسُ كَانُوا۟ لَهُمْ أَعْدَآءً وَكَانُوا۟ بِعِبَادَتِهِمْ كَـٰفِرِينَ ﴿٦﴾
(قىيامەت كۈنى) ئىنسانلار (ھېساب بېرىش ئۈچۈن) توپلانغان چاغدا، بۇتلار ئۇلارغا (يەنى بۇتپەرەسلەرگە) دۈشمەن بولىدۇ، ئۇلار تەرىپىدىن ئىبادەت قىلىنغانلىقىنى ئىنكار قىلىدۇ
وَإِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُنَا بَيِّنَـٰتٍ قَالَ ٱلَّذِينَ كَفَرُوا۟ لِلْحَقِّ لَمَّا جَآءَهُمْ هَـٰذَا سِحْرٌ مُّبِينٌ ﴿٧﴾
ئۇلارغا بىزنىڭ روشەن ئايەتلىرىمىز تىلاۋەت قىلىنسا، ھەقىقەتنى رەت قىلغۇچىلار ئۆزلىرىگە كەلگەن ھەقىقەتنى (يەنى قۇرئاننى) باھالاپ: «بۇ روشەن سېھىردۇر» دەيدۇ
أَمْ يَقُولُونَ ٱفْتَرَىٰهُ ۖ قُلْ إِنِ ٱفْتَرَيْتُهُۥ فَلَا تَمْلِكُونَ لِى مِنَ ٱللَّهِ شَيْـًٔا ۖ هُوَ أَعْلَمُ بِمَا تُفِيضُونَ فِيهِ ۖ كَفَىٰ بِهِۦ شَهِيدًۢا بَيْنِى وَبَيْنَكُمْ ۖ وَهُوَ ٱلْغَفُورُ ٱلرَّحِيمُ ﴿٨﴾
ياكى ئۇلار قۇرئاننى ئۇ (يەنى مۇھەممەد ئەلەيھىسسالام) ئۆزى توقۇغان دىېيىشەمدۇ؟ ئېيتقىنكى، «ئەگەر ئۇنى مەن توقۇغان بولسام، سىلەر مەندىن اﷲ نىڭ ئازابىدىن ھېچ نەرسىنى دەپئى قىلالمايسىلەر، سىلەرنىڭ قۇرئان توغرىسىدىكى تۆھمەتلىرىڭلارنى اﷲ ئوبدان بىلىدۇ، اﷲ مەن بىلەن سىلەرنىڭ ئاراڭلاردا گۇۋاھ بولۇشقا يېتەرلىكتۇر (يەنى مېنىڭ راستچىللىقىمغا ۋە سىلەرنىڭ يالغانچىلىقىڭلارغا گۇۋاھ بولىدۇ)، اﷲ (تەۋبە قىلغۇچىنى) مەغپىرەت قىلغۇچىدۇر، (مۆمىن بەندىلىرىگە) ناھايىتى مېھرىباندۇر»
قُلْ مَا كُنتُ بِدْعًا مِّنَ ٱلرُّسُلِ وَمَآ أَدْرِى مَا يُفْعَلُ بِى وَلَا بِكُمْ ۖ إِنْ أَتَّبِعُ إِلَّا مَا يُوحَىٰٓ إِلَىَّ وَمَآ أَنَا۠ إِلَّا نَذِيرٌ مُّبِينٌ ﴿٩﴾
ئېيتقىنكى، «مەن (اﷲ ئىنسانلارنى تەۋھىدكە دەۋەت قىلىش ئۈچۈن ئەۋەتكەن) تۇنجى خۇدانىڭ ئەلچىسى ئەمەسمەن، (مەندىن ئىلگىرىكى نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرى ئېلىپ كەلگەن ۋەھيى بىلەن كەلدىم، نېمىشقا مېنى ئىنكار قىلىسىلەر؟) ماڭا ۋە سىلەرگە نېمە قىلىنىدىغانلىقىنى ئۇقمايمەن (چۈنكى اﷲ نىڭ تەقدىرىنى ئالدىن ئۇققىلى بولمايدۇ)، مەن پەقەت ماڭا ۋەھيى قىلىنغان نەرسىگىلا ئەگىشىمەن، مەن پەقەت (سىلەرنى اﷲ نىڭ ئازابىدىن) ئوچۇق ئاگاھلاندۇرغۇچىمەن»
قُلْ أَرَءَيْتُمْ إِن كَانَ مِنْ عِندِ ٱللَّهِ وَكَفَرْتُم بِهِۦ وَشَهِدَ شَاهِدٌ مِّنۢ بَنِىٓ إِسْرَٰٓءِيلَ عَلَىٰ مِثْلِهِۦ فَـَٔامَنَ وَٱسْتَكْبَرْتُمْ ۖ إِنَّ ٱللَّهَ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّـٰلِمِينَ ﴿١٠﴾
(ئى مۇھەممەد! مۇشرىكلارغا) ئېيتقىنكى، «سىلەر ئېيتىپ بېقىڭلارچۇ! قۇرئان اﷲ تەرىپىدىن نازىل بولغان تۇرسا، سىلەر ئۇنىڭغا ئىشەنمىسەڭلار (ھالىڭلار قانداق بولىدۇ؟) ئىسرائىل ئەۋلادىدىن بىر گۇۋاھچى (يەنى ئابدۇللاھ ئىبن سالام) قۇرئاننىڭ اﷲ تەرىپىدىن نازىل بولغانلىقىغا گۇۋالىق بېرىپ ئىمان كەلتۈرسە، سىلەر تەكەببۇرلۇق قىلىپ (ئىماندىن) باش تارتساڭلار (ھالىڭلار قانداق بولىدۇ؟) شۈبھىسىزكى، اﷲ زالىم قەۋمنى ھىدايەت قىلمايدۇ»
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لِلَّذِينَ ءَامَنُوا۟ لَوْ كَانَ خَيْرًا مَّا سَبَقُونَآ إِلَيْهِ ۚ وَإِذْ لَمْ يَهْتَدُوا۟ بِهِۦ فَسَيَقُولُونَ هَـٰذَآ إِفْكٌ قَدِيمٌ ﴿١١﴾
ھەقىقەتنى رەت قىلغۇچىلار مۆمىنلەرگە: «ئەگەر (قۇرئان بىلەن ئىسلام دىنى) ياخشى بولىدىغان بولسا ئىدى، ئۇلار (يەنى ئاجىز مۇسۇلمانلار) ئۇنىڭغا بىزدىن بۇرۇن ئىشەنمىگەن بولاتتى» دېدى، ئۇلار قۇرئان بىلەن ھىدايەت تاپمىغانلىقلىرى ئۈچۈن، «بۇ قەدىمقى ئويدۇرمىدۇر» دەيدۇ
وَمِن قَبْلِهِۦ كِتَـٰبُ مُوسَىٰٓ إِمَامًا وَرَحْمَةً ۚ وَهَـٰذَا كِتَـٰبٌ مُّصَدِّقٌ لِّسَانًا عَرَبِيًّا لِّيُنذِرَ ٱلَّذِينَ ظَلَمُوا۟ وَبُشْرَىٰ لِلْمُحْسِنِينَ ﴿١٢﴾
قۇرئاندىن ئىلگىرى مۇسانىڭ مۇقەددەس كىتاب-ۋەھىيى (يەنى تەۋرات) ئەھلى جاھانغا پېشۋا ۋە رەھمەت ئىدى. بۇ ئەرەب تىلىدىكى مۇقەددەس كىتاب-ۋەھىيتۇر، (ئىلگىرىكى مۇقەددەس كىتاب-ۋەھىيلارنى) تەستىق قىلغۇچىدۇر، (اﷲ ئۇنى) زالىملارنى ئاگاھلاندۇرۇش ئۈچۈن، ياخشى ئىش قىلغۇچى مۆمىنلەرگە (مەڭگۈلۈك باغلار بىلەن) خۇش خەۋەر بېرىش ئۈچۈن نازىل قىلدى
إِنَّ ٱلَّذِينَ قَالُوا۟ رَبُّنَا ٱللَّهُ ثُمَّ ٱسْتَقَـٰمُوا۟ فَلَا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُونَ ﴿١٣﴾
«پەرۋەردىگارىمىز اﷲ دۇر» دېگەن، ئاندىن توغرا يولدا بولغانلارغا، شۈبھىسىزكى، (كەلگۈسىدىن) غەم قىلىش، (كەتكەنگە) قايغۇرۇش يوقتۇر
أُو۟لَـٰٓئِكَ أَصْحَـٰبُ ٱلْجَنَّةِ خَـٰلِدِينَ فِيهَا جَزَآءًۢ بِمَا كَانُوا۟ يَعْمَلُونَ ﴿١٤﴾
ئەنە شۇلار ئەھلى مەڭگۈلۈك باغلارتۇر، ئۇلار مەڭگۈلۈك باغلارتە مەڭگۈ قالىدۇ، (بۇ) ئۇلارنىڭ قىلغان (ياخشى) ئەمەللىرىنى مۇكاپاتلاش ئۈچۈندۇر
وَوَصَّيْنَا ٱلْإِنسَـٰنَ بِوَٰلِدَيْهِ إِحْسَـٰنًا ۖ حَمَلَتْهُ أُمُّهُۥ كُرْهًا وَوَضَعَتْهُ كُرْهًا ۖ وَحَمْلُهُۥ وَفِصَـٰلُهُۥ ثَلَـٰثُونَ شَهْرًا ۚ حَتَّىٰٓ إِذَا بَلَغَ أَشُدَّهُۥ وَبَلَغَ أَرْبَعِينَ سَنَةً قَالَ رَبِّ أَوْزِعْنِىٓ أَنْ أَشْكُرَ نِعْمَتَكَ ٱلَّتِىٓ أَنْعَمْتَ عَلَىَّ وَعَلَىٰ وَٰلِدَىَّ وَأَنْ أَعْمَلَ صَـٰلِحًا تَرْضَىٰهُ وَأَصْلِحْ لِى فِى ذُرِّيَّتِىٓ ۖ إِنِّى تُبْتُ إِلَيْكَ وَإِنِّى مِنَ ٱلْمُسْلِمِينَ ﴿١٥﴾
بىز ئىنساننى ئاتا - ئانىسىغا ياخشىلىق قىلىشقا بۇيرۇدۇق. ئىنساننى ئانىسى مۇشەققەت بىلەن قورساق كۆتۈرۈپ، مۇشەققەت بىلەن تۇغدى. ئۇنىڭغا قورساق كۆتۈرۈش مۇددىتى ۋە ئۇنى سۈتتىن ئايرىش مۇددىتى 30 ئايدۇر. تاكى ئۇ (بوۋاق ئۆسۈپ) كۈچ - قۇۋۋەتكە تولۇپ، 40 ياشقا يەتكەندە، (ئۇ): «پەرۋەردىگارىم! سېنىڭ ماڭا ۋە ئاتا - ئانامغا بەرگەن نېمىتىڭگە شۈكۈر قىلىشنى ۋە سەن رازى بولىدىغان ياخشى ئىشنى قىلىشىمنى ماڭا ئىلھام قىلغىن، مەن ئۈچۈن مېنىڭ ئەۋلادىمنى تۈزىگىن (يەنى مېنىڭ ئەۋلادىمنى ياخشى ئادەملەر قىلغىن)، مەن ئىشەنچلىك ھالدا ساڭا (جىمى گۇناھلاردىن) تەۋبە قىلدىم، مەن ئىشەنچلىك ھالدا مۇسۇلمانلاردىندۇرمەن» دەيدۇ
أُو۟لَـٰٓئِكَ ٱلَّذِينَ نَتَقَبَّلُ عَنْهُمْ أَحْسَنَ مَا عَمِلُوا۟ وَنَتَجَاوَزُ عَن سَيِّـَٔاتِهِمْ فِىٓ أَصْحَـٰبِ ٱلْجَنَّةِ ۖ وَعْدَ ٱلصِّدْقِ ٱلَّذِى كَانُوا۟ يُوعَدُونَ ﴿١٦﴾
ئەنە شۇنداق كىشىلەرنىڭ ياخشى ئىشلىرىنى قوبۇل قىلىپ، يامان ئىشلىرىنى كەچۈرۈم قىلىمىز، ئۇلار (اﷲ يامان ئىشلىرىنى كەچۈرگەنلەر بىلەن) مەڭگۈلۈك باغلارتە بولىدۇ، (بۇ) ئۇلارغا قىلىنغان راست ۋەدىدۇر
وَٱلَّذِى قَالَ لِوَٰلِدَيْهِ أُفٍّ لَّكُمَآ أَتَعِدَانِنِىٓ أَنْ أُخْرَجَ وَقَدْ خَلَتِ ٱلْقُرُونُ مِن قَبْلِى وَهُمَا يَسْتَغِيثَانِ ٱللَّهَ وَيْلَكَ ءَامِنْ إِنَّ وَعْدَ ٱللَّهِ حَقٌّ فَيَقُولُ مَا هَـٰذَآ إِلَّآ أَسَـٰطِيرُ ٱلْأَوَّلِينَ ﴿١٧﴾
بەزى ئادەم (يەنى ئاتا - ئانىسى ئىمانغا دەۋەت قىلغان فاجىر بالا): ۋاي - ۋۇي! سىلەر مېنى قايتا تىرىلىشتىن قورقۇتۇۋاتامسىلەر، (مەندىن ئىلگىرى) نۇرغۇن ئۈممەتلەر ئۆتتى (ئۇلاردىن ھېچ ئادەم تىرىلدۈرۈلگىنى يوق)» دەيدۇ، ئۇ ئىككىسى اﷲ قا پەرياد قىلىپ: «ۋاي ساڭا! ئىمان ئېيتقىن، اﷲ نىڭ ۋەدىسى ئىشەنچلىك ھالدا ھەقتۇر» دەيدۇ. ئۇ: «بۇ پەقەت ئاۋالقىلاردىن قالغان ئەپسانىلەردۇر» دەيدۇ
أُو۟لَـٰٓئِكَ ٱلَّذِينَ حَقَّ عَلَيْهِمُ ٱلْقَوْلُ فِىٓ أُمَمٍ قَدْ خَلَتْ مِن قَبْلِهِم مِّنَ ٱلْجِنِّ وَٱلْإِنسِ ۖ إِنَّهُمْ كَانُوا۟ خَـٰسِرِينَ ﴿١٨﴾
ئەنە شۇنداق كىشىلەرگە (اﷲ نىڭ ئازاب قىلىش) سۆزى ھەقلىق بولدى، ئۇلار بولسا ئۇلاردىن ئىلگىرى ئۆتكەن ئىنسانلاردىن ۋە جىنلاردىن بولغان ئۈممەتلەر قاتارىغا كىرگۈزۈلدى، ئۇلار ئىشەنچلىك ھالدا زىيان تارتقۇچى بولدى
وَلِكُلٍّ دَرَجَـٰتٌ مِّمَّا عَمِلُوا۟ ۖ وَلِيُوَفِّيَهُمْ أَعْمَـٰلَهُمْ وَهُمْ لَا يُظْلَمُونَ ﴿١٩﴾
(مۆمىنلەرنىڭ ۋە كاپىلارنىڭ) ھەممىسىنىڭ قىلغان ئەمەللىرىگە يارىشا دەرىجىلىرى بولىدۇ، ئۇلارنىڭ ئەمەللىرىنىڭ ساۋابى ۋە جازاسى (كەم - زىيادە قىلىنماي) تولۇق بېرىلىدۇ، ئۇلارغا زۇلۇم قىلىنمايدۇ
وَيَوْمَ يُعْرَضُ ٱلَّذِينَ كَفَرُوا۟ عَلَى ٱلنَّارِ أَذْهَبْتُمْ طَيِّبَـٰتِكُمْ فِى حَيَاتِكُمُ ٱلدُّنْيَا وَٱسْتَمْتَعْتُم بِهَا فَٱلْيَوْمَ تُجْزَوْنَ عَذَابَ ٱلْهُونِ بِمَا كُنتُمْ تَسْتَكْبِرُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَبِمَا كُنتُمْ تَفْسُقُونَ ﴿٢٠﴾
كاپىلار دوزاخقا توغرىلىنىدىغان كۈندە (ئۇلارغا ئەيىبلەش يۈزىسىدىن ئېيتىلىدۇ): «سىلەر لەززاتلىرىڭلارنى ھاياتىي دۇنيادا بەھرىمەن بولۇپ تۈگەتتىڭلار، سىلەر زېمىندا ھەقسىز رەۋىشتە چوڭچىلىق قىلىپ، ئىماندىن باش تارتقانلىقىڭلار ۋە اﷲ نىڭ ئىتائىتىدىن چىققانلىقىڭلار تۈپەيلىدىن، بۈگۈن خار قىلغۇچى ئازاب بىلەن جازالىنىسىلەر»
۞ وَٱذْكُرْ أَخَا عَادٍ إِذْ أَنذَرَ قَوْمَهُۥ بِٱلْأَحْقَافِ وَقَدْ خَلَتِ ٱلنُّذُرُ مِنۢ بَيْنِ يَدَيْهِ وَمِنْ خَلْفِهِۦٓ أَلَّا تَعْبُدُوٓا۟ إِلَّا ٱللَّهَ إِنِّىٓ أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ عَظِيمٍ ﴿٢١﴾
(ئى مۇھەممەد!) ئادنىڭ قېرىندىشى (يەنى ھۇد ئەلەيھىسسالام) نىڭ ئۆز قەۋمى بىلەن بولغان قىسسىسىنى بايان قىلغىن، ئۆز ۋاقتىدا ئۇ ئەھقاف (دېگەن يەر) دىكى قەۋمىنى (اﷲ نىڭ ئازابىدىن) ئاگاھلاندۇردى، ئۇنىڭدىن (يەنى ھۇدتىن) ئىلگىرى ۋە ئۇنىڭدىن كېيىن نۇرغۇن خۇدانىڭ ئەلچى-رەسۇللىرى ئۆتتى. ئۇ (يەنى ھۇد) (قەۋمىنى ئاگاھلاندۇرۇپ) ئېيتتى: «سىلەر پەقەت اﷲ غىلا ئىبادەت قىلىڭلار، مەن ئىشەنچلىك ھالدا سىلەرنىڭ (اﷲ تىن غەيرىيگە ئىبادەت قىلساڭلار) بۈيۈك كۈننىڭ ئازابىغا قېلىشىڭلاردىن قورقىمەن»
قَالُوٓا۟ أَجِئْتَنَا لِتَأْفِكَنَا عَنْ ءَالِهَتِنَا فَأْتِنَا بِمَا تَعِدُنَآ إِن كُنتَ مِنَ ٱلصَّـٰدِقِينَ ﴿٢٢﴾
ئۇلار: «(ئى ھۇد!) سەن بىزنى ئىلاھلىرىمىزدىن توسقىلى كەلدىڭمۇ؟ ئەگەر سەن (سۆزۈڭدە) راستچىللاردىن بولساڭ، بىزگە بىزنى قورقۇتقان ئازابنى كەلتۈرگىن» دېدى
قَالَ إِنَّمَا ٱلْعِلْمُ عِندَ ٱللَّهِ وَأُبَلِّغُكُم مَّآ أُرْسِلْتُ بِهِۦ وَلَـٰكِنِّىٓ أَرَىٰكُمْ قَوْمًا تَجْهَلُونَ ﴿٢٣﴾
ئۇ: «(سىلەرگە ئازابنىڭ قاچان كېلىشىنى) پەقەت اﷲ بىلىدۇ، مەن سىلەرگە ئەۋەتىلگەن نەرسىنى تەبلىغ قىلىمەن ۋە لېكىن مەن سىلەرنى نادان قەۋم كۆرىمەن» دېدى
فَلَمَّا رَأَوْهُ عَارِضًا مُّسْتَقْبِلَ أَوْدِيَتِهِمْ قَالُوا۟ هَـٰذَا عَارِضٌ مُّمْطِرُنَا ۚ بَلْ هُوَ مَا ٱسْتَعْجَلْتُم بِهِۦ ۖ رِيحٌ فِيهَا عَذَابٌ أَلِيمٌ ﴿٢٤﴾
ئۇلار ئۆزلىرىنىڭ ۋادىسىغا سۈرۈلۈۋاتقان بۇلۇتنى كۆرۈپ: «مانا بۇ بىزگە يامغۇر ياغدۇرغۇچى بۇلۇتتۇر» دېيىشتى. (ھۇد ئېيتتى) «ئۇنداق ئەمەس، ئۇ سىلەر بالدۇرراق ئىشقا ئېشىشىنى تەلەپ قىلغان نەرسىدۇر، ئۇ بىر بوران بولۇپ، ئۇنىڭ ئىچىدە قاتتىق ئازاب بار
تُدَمِّرُ كُلَّ شَىْءٍۭ بِأَمْرِ رَبِّهَا فَأَصْبَحُوا۟ لَا يُرَىٰٓ إِلَّا مَسَـٰكِنُهُمْ ۚ كَذَٰلِكَ نَجْزِى ٱلْقَوْمَ ٱلْمُجْرِمِينَ ﴿٢٥﴾
ئۇ پەرۋردىگارىنىڭ بىر ئەمرى بىلەن ھەممە نەرسىنى ھالاك قىلىدۇ» (يەنى بوران كېلىپ ئۇلارنى ھالاك قىلدى)، شۇنىڭ بىلەن ئۇلارنىڭ جايلىرىنىڭ (يەنى تۇرغان ماكانلىرىنىڭ ئىزىدىن) باشقا نەرسە كۆرۈنمەيدىغان بولۇپ قالدى، گۇناھكار قەۋمنى مۇشۇنداق جازالايمىز
وَلَقَدْ مَكَّنَّـٰهُمْ فِيمَآ إِن مَّكَّنَّـٰكُمْ فِيهِ وَجَعَلْنَا لَهُمْ سَمْعًا وَأَبْصَـٰرًا وَأَفْـِٔدَةً فَمَآ أَغْنَىٰ عَنْهُمْ سَمْعُهُمْ وَلَآ أَبْصَـٰرُهُمْ وَلَآ أَفْـِٔدَتُهُم مِّن شَىْءٍ إِذْ كَانُوا۟ يَجْحَدُونَ بِـَٔايَـٰتِ ٱللَّهِ وَحَاقَ بِهِم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿٢٦﴾
بىز ئۇلارنى (يەنى ئاد قەۋمىنى) (پۇل - مال، كۈچ - قۇۋۋەت جەھەتتە) سىلەرنى قادىر قىلمىغان نەرسىلەرگە قادىر قىلدۇق. (ئۇلارنىڭ شۇ نېمەتلەرگە شۈكۈر قىلىپ، بۇ ئارقىلىق نېمەتلەرنى بەرگۈچى اﷲ نى تونۇشى ئۈچۈن) ئۇلارغا قۇلاق، كۆز، قەلبلەرنى بەردۇق، ئۇلار اﷲ نىڭ ئايەتلىرىنى ئىنكار قىلغانلىقلىرى ئۈچۈن، ئۇلارنىڭ قۇلاقلىرى، كۆزلىرى، دىللىرى ئۇلاردىن (اﷲ نىڭ ئازابىدىن) ھېچ نەرسىنى دەپئى قىلمىدى، ئۇلار مەسخىرە قىلغان ئازاب ئۇلارغا نازىل بولدى
وَلَقَدْ أَهْلَكْنَا مَا حَوْلَكُم مِّنَ ٱلْقُرَىٰ وَصَرَّفْنَا ٱلْـَٔايَـٰتِ لَعَلَّهُمْ يَرْجِعُونَ ﴿٢٧﴾
(ئى مەككە ئاھالىسى!) بىز ئىشەنچلىك ھالدا سىلەرنىڭ ئەتراپىڭلاردىكى شەھەرلەرنى ھالاك قىلدۇق، ئۇلارنى (كۇفرىدىن) يانسۇن دەپ ئايەتلەرنى تەكرار بايان قىلدۇق
فَلَوْلَا نَصَرَهُمُ ٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ قُرْبَانًا ءَالِهَةًۢ ۖ بَلْ ضَلُّوا۟ عَنْهُمْ ۚ وَذَٰلِكَ إِفْكُهُمْ وَمَا كَانُوا۟ يَفْتَرُونَ ﴿٢٨﴾
(ئۇلارغا ئازاب كەلگەندە) ئۇلارنىڭ خۇداغا يېقىن قىلىدۇ دەپ اﷲ تىن باشقا تۇتقان مەبۇدلىرى ئۇلارغا نېمىشقا ياردەم قىلمىدى؟ بەلكى ئۇلارنىڭ مەبۇدلىرى غايىب بولدى، بۇ (يەنى بۇتلار بىزنى خۇداغا يېقىن قىلىدۇ دېگەن سۆزى) ئۇلارنىڭ ئويدۇرمىسىدۇر ۋە (اﷲ قا) چاپلىغان بوھتانلىرىدۇر
وَإِذْ صَرَفْنَآ إِلَيْكَ نَفَرًا مِّنَ ٱلْجِنِّ يَسْتَمِعُونَ ٱلْقُرْءَانَ فَلَمَّا حَضَرُوهُ قَالُوٓا۟ أَنصِتُوا۟ ۖ فَلَمَّا قُضِىَ وَلَّوْا۟ إِلَىٰ قَوْمِهِم مُّنذِرِينَ ﴿٢٩﴾
(ئى مۇھەممەد!) ئۆز ۋاقتىدا بىر تۈركۈم جىنلارنى ساڭا قۇرئان تىڭشاشقا ئەۋەتتۇق، ئۇلار قۇرئان تىلاۋىتىگە ھازىر بولغاندا (بىر - بىرىگە): «جىم تۇرۇڭلار» دېيىشتى، قۇرئان تىلاۋەت قىلىنىپ بولغاندىن كېيىن ئۇلار ئۆز قەۋمىگە ئاگاھلاندۇرغۇچى بولۇپ قايتىشتى
قَالُوا۟ يَـٰقَوْمَنَآ إِنَّا سَمِعْنَا كِتَـٰبًا أُنزِلَ مِنۢ بَعْدِ مُوسَىٰ مُصَدِّقًا لِّمَا بَيْنَ يَدَيْهِ يَهْدِىٓ إِلَى ٱلْحَقِّ وَإِلَىٰ طَرِيقٍ مُّسْتَقِيمٍ ﴿٣٠﴾
ئۇلار ئېيتتى: «ئى قەۋمىمىز! ئىشەنچلىك ھالدا بىز مۇسادىن كېيىن نازىل بولغان، ئۇنىڭدىن ئىلگىرىكى مۇقەددەس كىتاب-ۋەھىيلارغا ئۇيغۇن كېلىدىغان، ھەق دىنغا ۋە توغرا يولغا باشلايدىغان مۇقەددەس كىتاب-ۋەھىينى (يەنى قۇرئاننى) تىڭشىدۇق»
يَـٰقَوْمَنَآ أَجِيبُوا۟ دَاعِىَ ٱللَّهِ وَءَامِنُوا۟ بِهِۦ يَغْفِرْ لَكُم مِّن ذُنُوبِكُمْ وَيُجِرْكُم مِّنْ عَذَابٍ أَلِيمٍ ﴿٣١﴾
«ئى قەۋمىمىز! اﷲ قا دەۋەت قىلغۇچى (يەنى مۇھەممەد ئەلەيھىسسالام) نىڭ دەۋىتىنى قوبۇل قىلىڭلار ۋە اﷲ قا ئىمان ئېيتىڭلار، اﷲ بەزى گۇناھلارنى مەغپىرەت قىلىدۇ، سىلەرنى قاتتىق ئازابتىن ساقلايدۇ»
وَمَن لَّا يُجِبْ دَاعِىَ ٱللَّهِ فَلَيْسَ بِمُعْجِزٍ فِى ٱلْأَرْضِ وَلَيْسَ لَهُۥ مِن دُونِهِۦٓ أَوْلِيَآءُ ۚ أُو۟لَـٰٓئِكَ فِى ضَلَـٰلٍ مُّبِينٍ ﴿٣٢﴾
كىمكى اﷲ قا (ئىمان ئېيتىشقا) دەۋەت قىلغۇچىنىڭ (يەنى رەسۇلۇللاھنىڭ) دەۋىتىنى قوبۇل قىلمىسا، ئازابتىن قېچىپ قۇتۇلالمايدۇ، اﷲ تىن باشقا (ئۇنى اﷲ نىڭ ئازابىدىن ساقلىغۇچى) ياردەمچىلەر بولمايدۇ، ئۇلار (يەنى رەسۇلۇللاھنىڭ دەۋىتىنى قوبۇل قىلمىغۇچىلار) ئوپئوچۇق گۇمراھلىقتىدۇر
أَوَلَمْ يَرَوْا۟ أَنَّ ٱللَّهَ ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَلَمْ يَعْىَ بِخَلْقِهِنَّ بِقَـٰدِرٍ عَلَىٰٓ أَن يُحْـِۧىَ ٱلْمَوْتَىٰ ۚ بَلَىٰٓ إِنَّهُۥ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٣٣﴾
ئۇلار (يەنى ئۆلگەندىن كېيىن تىرىلىشنى ئىنكار قىلغۇچى كۇففارلار) بىلمەمدۇكى، ئاسمانلارنى ۋە زېمىننى ياراتقان ۋە ئۇلارنى يارىتىشتا چارچاپ قالمىغان اﷲ ئۆلۈكلەرنى تىرىلدۈرۈشكە قادىر ئەمەسمۇ؟ دۇرۇس، ئۇ ئىشەنچلىك ھالدا ھەر نەرسىگە قادىردۇر
وَيَوْمَ يُعْرَضُ ٱلَّذِينَ كَفَرُوا۟ عَلَى ٱلنَّارِ أَلَيْسَ هَـٰذَا بِٱلْحَقِّ ۖ قَالُوا۟ بَلَىٰ وَرَبِّنَا ۚ قَالَ فَذُوقُوا۟ ٱلْعَذَابَ بِمَا كُنتُمْ تَكْفُرُونَ ﴿٣٤﴾
ھەقىقەتنى رەت قىلغۇچىلار دوزاخقا توغرىلىنىدىغان كۈندە (ئۇلارغا): «بۇ (يەنى سىلەر تېتىۋاتقان ئازاب) راست ئەمەسمۇ؟» (دېيىلىدۇ) ئۇلار: «پەرۋەردىگارىمىز بىلەن قەسەمكى، ئەلۋەتتە راستتۇر» دەيدۇ، اﷲ (ئۇلارغا): «ھەقىقەتنى رەت قىلغۇچى بولغانلىقىڭلار تۈپەيلىدىن ئازابنى تېتىڭلار» دەيدۇ
فَٱصْبِرْ كَمَا صَبَرَ أُو۟لُوا۟ ٱلْعَزْمِ مِنَ ٱلرُّسُلِ وَلَا تَسْتَعْجِل لَّهُمْ ۚ كَأَنَّهُمْ يَوْمَ يَرَوْنَ مَا يُوعَدُونَ لَمْ يَلْبَثُوٓا۟ إِلَّا سَاعَةً مِّن نَّهَارٍۭ ۚ بَلَـٰغٌ ۚ فَهَلْ يُهْلَكُ إِلَّا ٱلْقَوْمُ ٱلْفَـٰسِقُونَ ﴿٣٥﴾
(ئى مۇھەممەد!) سەن (مۇشرىكلارنىڭ سالغان جاپاسىغا) ئىرادىلىك خۇدانىڭ ئەلچى-رەسۇللىرى سەۋر قىلغاندەك سەۋر قىلغىن، ئۇلارغا بولىدىغان ئازابقا ئالدىراپ كەتمىگىن. ئۇلار (ئاخىرەتتە) ۋەدە قىلىنغان ئازابنى كۆرگەندە، (ئۇلار دۇنيادا) گويا كۈندۈزدە بىر دەملا تۇرغاندەك ھېس قىلىدۇ، (بۇ قۇرئان سىلەرگە) تەبلىغدۇر، پەقەت پاسىق قەۋملا ھالاك قىلىنىدۇ
