بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
هَلْ أَتَىٰ عَلَى ٱلْإِنسَـٰنِ حِينٌ مِّنَ ٱلدَّهْرِ لَمْ يَكُن شَيْـًٔا مَّذْكُورًا ﴿١﴾
ئىنسان (ئانىنىڭ قورسىقىدا) بىر مۇددەتنى ئۆتكۈزدى، ئۇ (ئەرزىمگەنلىكتىن) تىلغا ئېلىنىدىغان نەرسە بولمىدى
إِنَّا خَلَقْنَا ٱلْإِنسَـٰنَ مِن نُّطْفَةٍ أَمْشَاجٍ نَّبْتَلِيهِ فَجَعَلْنَـٰهُ سَمِيعًۢا بَصِيرًا ﴿٢﴾
شۈبھىسىزكى، بىز ئىنساننى ئارىلاشما مەنىي (يەنى ئەر بىلەن ئاياللارنىڭ مەنىيسى) دىن ياراتتۇق، ئۇنى بىز (شەرىئەت تەكلىپلىرى بىلەن) سىنايمىز، شۇنىڭ ئۈچۈن ئۇنى بىز (ئۇنى نازىل قىلغان ئايەتلىرىمىزنى ئاڭلىسۇن، كائىناتنى ياراتقۇچىنىڭ بىرلىكىگە دالالەت قىلىدىغان دەلىللەرنى كۆرسۇن دەپ) ئاڭلايدىغان، كۆرۈدىغان قىلىپ ياراتتۇق
إِنَّا هَدَيْنَـٰهُ ٱلسَّبِيلَ إِمَّا شَاكِرًا وَإِمَّا كَفُورًا ﴿٣﴾
شۈبھىسىزكى، ئۇنىڭغا بىز (خۇدانىڭ ئەلچىسى ئەۋەتىش بىلەن ياخشى - يامان) يولنى كۆرسەتتۇق، ئۇ مۆمىن بولۇپ ياخشى يولدا مېڭىپ) اﷲ نىڭ نېمىتىگە) شۈكۈر قىلغۇچىدۇر، ياكى (فاجىر بولۇپ يامان يولدا مېڭىپ) كۇفرىلىق قىلغۇچىدۇر
إِنَّآ أَعْتَدْنَا لِلْكَـٰفِرِينَ سَلَـٰسِلَا۟ وَأَغْلَـٰلًا وَسَعِيرًا ﴿٤﴾
شۈبھىسىزكى، بىز ھەقىقەتنى رەت قىلغۇچىلارغا (پۇتلىرىغا سېلىنىدىغان) زەنجىرلەرنى، (بويۇنلىرىغا سېلىنىدىغان) تاقاقلارنى ۋە يېنىپ تۇرغان دوزاخنى تەييارلىدۇق
إِنَّ ٱلْأَبْرَارَ يَشْرَبُونَ مِن كَأْسٍ كَانَ مِزَاجُهَا كَافُورًا ﴿٥﴾
ئىشەنچلىك ھالدا ياخشىلار كافۇر ئارىلاشتۇرۇلغان (مەي بىلەن تولدۇرۇلغان) جاملاردىن ئىچىدۇ
عَيْنًا يَشْرَبُ بِهَا عِبَادُ ٱللَّهِ يُفَجِّرُونَهَا تَفْجِيرًا ﴿٦﴾
(ئۇ كافۇر مەڭگۈلۈك باغلارتىكى بىر) بۇلاقتىن ئېتىلىپ چىققان بولۇپ، ئۇنىڭدىن اﷲ نىڭ (ياخشى) بەندىلىرى ئىچىدۇ، ئۇ بۇلاقنى ئۇلار (خالىغان جايلىرىغا ئېقىتىپ بارالايدۇ
يُوفُونَ بِٱلنَّذْرِ وَيَخَافُونَ يَوْمًا كَانَ شَرُّهُۥ مُسْتَطِيرًا ﴿٧﴾
ئۇلار ئۆز ئۈستىگە قەسەم ئىچىپ ئالغان ئىشنى ئورۇنلايدۇ ۋە دەھشىتى كەڭ دائىرىلىك بولغان كۈندىن (يەنى قىيامەت كۈنىدىن) قورقىدۇ
وَيُطْعِمُونَ ٱلطَّعَامَ عَلَىٰ حُبِّهِۦ مِسْكِينًا وَيَتِيمًا وَأَسِيرًا ﴿٨﴾
ئۆزى موھتاج تۇرۇقلۇق، مىسكىنگە، يېتىمگە ۋە ئەسىرگە تائام بېرىدۇ
إِنَّمَا نُطْعِمُكُمْ لِوَجْهِ ٱللَّهِ لَا نُرِيدُ مِنكُمْ جَزَآءً وَلَا شُكُورًا ﴿٩﴾
(ئۇلار ئېيتىدۇ) «سىلەرگە بىز اﷲ نىڭ رازىلىقى ئۈچۈن تائام بېرىمىز، سىلەردىن (بۇنىڭ بەدىلىگە) ھېچقانداق مۇكاپات ۋە تەشەككۈر تەلەپ قىلمايمىز
إِنَّا نَخَافُ مِن رَّبِّنَا يَوْمًا عَبُوسًا قَمْطَرِيرًا ﴿١٠﴾
شۈبھىسىزكى، بىز پەرۋەردىگارىمىز تەرىپىدىن مەيدانغا كەلتۈرۈلگەن ئېغىر، قاتتىق كۈندىن قورقىمىز»
فَوَقَىٰهُمُ ٱللَّهُ شَرَّ ذَٰلِكَ ٱلْيَوْمِ وَلَقَّىٰهُمْ نَضْرَةً وَسُرُورًا ﴿١١﴾
اﷲ ئۇلارنى شۇ كۈننىڭ شەررىدىن ساقلايدۇ، ئۇلارنىڭ (يۈزلىرىگە) نۇر، (دىللىرىغا) خۇشاللىق ئاتا قىلىدۇ
وَجَزَىٰهُم بِمَا صَبَرُوا۟ جَنَّةً وَحَرِيرًا ﴿١٢﴾
ئۇلارنىڭ سەۋر - تاقەتلىرى ئۈچۈن اﷲ ئۇلارنى مەڭگۈلۈك باغلار بىلەن ۋە (ئۇ يەردىكى) يىپەك (لىباسلار) بىلەن مۇكاپاتلايدۇ
مُّتَّكِـِٔينَ فِيهَا عَلَى ٱلْأَرَآئِكِ ۖ لَا يَرَوْنَ فِيهَا شَمْسًا وَلَا زَمْهَرِيرًا ﴿١٣﴾
ئۇلار مەڭگۈلۈك باغلارتە تەختلەرگە يۆلىنىپ ئولتۇرىدۇ، ئۇلار مەڭگۈلۈك باغلارتە قاتتىق ئىسسىقنىمۇ، قاتتىق سوغۇقنىمۇ كۆرمەيدۇ
وَدَانِيَةً عَلَيْهِمْ ظِلَـٰلُهَا وَذُلِّلَتْ قُطُوفُهَا تَذْلِيلًا ﴿١٤﴾
مەڭگۈلۈك باغلارتىكى (دەرەخلەرنىڭ) سايىلىرى ئۇلارغا يېقىندۇر، مەڭگۈلۈك باغلارنىڭ مېۋىلىرىنى ئۈزۈش ئۇلارغا ئاسان قىلىنىدۇ
وَيُطَافُ عَلَيْهِم بِـَٔانِيَةٍ مِّن فِضَّةٍ وَأَكْوَابٍ كَانَتْ قَوَارِيرَا۠ ﴿١٥﴾
ئۇلارغا (تاماق قاچىلانغان) كۈمۈش تەخسىلەر، (مەي تولدۇرۇلغان) كۈمۈشتىن ياسالغان (شىشىدەك سۈزۈك) جاملار ئايلاندۇرۇپ سۇنۇلۇپ تۇرىدۇ، (ساقىيلار، ئۇلارنىڭ ئېھتىياجىغا قاراپ) لايىق ئۆلچەيدۇ (يەنى ئىچكۈچىلەرنىڭ ئېھتىياجىدىن ئېشىپمۇ قالمايدۇ، كېمىيىپمۇ قالمايدۇ
قَوَارِيرَا۟ مِن فِضَّةٍ قَدَّرُوهَا تَقْدِيرًا ﴿١٦﴾
ئۇلارغا (تاماق قاچىلانغان) كۈمۈش تەخسىلەر، (مەي تولدۇرۇلغان) كۈمۈشتىن ياسالغان (شىشىدەك سۈزۈك) جاملار ئايلاندۇرۇپ سۇنۇلۇپ تۇرىدۇ، (ساقىيلار، ئۇلارنىڭ ئېھتىياجىغا قاراپ) لايىق ئۆلچەيدۇ (يەنى ئىچكۈچىلەرنىڭ ئېھتىياجىدىن ئېشىپمۇ قالمايدۇ، كېمىيىپمۇ قالمايدۇ
وَيُسْقَوْنَ فِيهَا كَأْسًا كَانَ مِزَاجُهَا زَنجَبِيلًا ﴿١٧﴾
ئۇلار يەنە مەڭگۈلۈك باغلارتە زەنجىبىل ئارىلاشتۇرۇلغان جام بىلەن (يەنى جامدىكى مەي بىلەن) سۇغىرىلىدۇ
عَيْنًا فِيهَا تُسَمَّىٰ سَلْسَبِيلًا ﴿١٨﴾
مەڭگۈلۈك باغلارتە سەلسەبىل دەپ ئاتىلىدىغان بىر بۇلاقمۇ بار
۞ وَيَطُوفُ عَلَيْهِمْ وِلْدَٰنٌ مُّخَلَّدُونَ إِذَا رَأَيْتَهُمْ حَسِبْتَهُمْ لُؤْلُؤًا مَّنثُورًا ﴿١٩﴾
قېرىماي ھەمىشە ياش تۇرىدىغان غىلمانلار نۆۋەت بىلەن ئۇلارنىڭ خىزمىتىنى قىلىپ تۇرىدۇ، ئۇلارنى كۆرگەن چېغىڭدا (ئۇلارنىڭ گۈزەللىكى، سۈزۈكلۈكى ۋە نۇرلۇقلۇقىغا قاراپ) ئۇلارنى تېرىلىپ كەتكەن مەرۋايىتمىكىن دەپ قالىسەن
وَإِذَا رَأَيْتَ ثَمَّ رَأَيْتَ نَعِيمًا وَمُلْكًا كَبِيرًا ﴿٢٠﴾
قاچانكى سەن قارايدىغان بولساڭ، بۇ يەردە (تەسۋىرلەپ تۈگەتكىلى بولمايدىغان) نېمەتلەرنى ۋە كاتتا پادىشاھلىقنى كۆرۈسەن
عَـٰلِيَهُمْ ثِيَابُ سُندُسٍ خُضْرٌ وَإِسْتَبْرَقٌ ۖ وَحُلُّوٓا۟ أَسَاوِرَ مِن فِضَّةٍ وَسَقَىٰهُمْ رَبُّهُمْ شَرَابًا طَهُورًا ﴿٢١﴾
ئۇلارنىڭ ئۇچىسىدا يۇپقا يىپەك ۋە قېلىن يىپەكتىن (تەييارلانغان) يېشىل كىيىملەر بولىدۇ، (ئۇلار) كۈمۈش ئۈزۈكلەرنى تاقايدۇ، پەرۋەردىگارى ئۇلارنى پاك شاراب بىلەن سۇغىرىدۇ
إِنَّ هَـٰذَا كَانَ لَكُمْ جَزَآءً وَكَانَ سَعْيُكُم مَّشْكُورًا ﴿٢٢﴾
شۈبھىسىزكى، بۇ سىلەرگە بېرىلگەن مۇكاپاتتۇر، سىلەرنىڭ مېھنىتىڭلار قوبۇل بولدى
إِنَّا نَحْنُ نَزَّلْنَا عَلَيْكَ ٱلْقُرْءَانَ تَنزِيلًا ﴿٢٣﴾
(ئى مۇھەممەد!) ئىشەنچلىك ھالدا بىز ساڭا قۇرئاننى بۆلۈپ - بۆلۈپ نازىل قىلدۇق
فَٱصْبِرْ لِحُكْمِ رَبِّكَ وَلَا تُطِعْ مِنْهُمْ ءَاثِمًا أَوْ كَفُورًا ﴿٢٤﴾
سەن پەرۋەردىگارىڭنىڭ ھۆكمىگە سەۋر قىلغىن، سەن ئۇلارنىڭ ئىچىدىن ھېچبىر گۇناھكارغا ياكى كۇفرىلىق قىلغۇچىغا بويسۇنمىغىن
وَٱذْكُرِ ٱسْمَ رَبِّكَ بُكْرَةً وَأَصِيلًا ﴿٢٥﴾
ئەتىگەن - ئاخشامدا پەرۋەردىگارىڭنىڭ نامىنى ياد قىلغىن (يەنى ناماز ئوقۇغىن، پەرۋەردىگارىڭغا كۆپ تائەت - ئىبادەت قىلغىن)
وَمِنَ ٱلَّيْلِ فَٱسْجُدْ لَهُۥ وَسَبِّحْهُ لَيْلًا طَوِيلًا ﴿٢٦﴾
كېچىنىڭ ئاز قىسمىدا ئۇنىڭغا سەجدە قىل، كۆپ قىسمىدا ئۇنىڭغا تەسبىھ ئېيت
إِنَّ هَـٰٓؤُلَآءِ يُحِبُّونَ ٱلْعَاجِلَةَ وَيَذَرُونَ وَرَآءَهُمْ يَوْمًا ثَقِيلًا ﴿٢٧﴾
شۈبھىسىزكى، بۇلار (يەنى مۇشرىكلار) دۇنيانى (ئاخىرەتتىن) ئارتۇق بىلىدۇ، كەلگۈسىدىكى قىيىن بىر كۈنگە (يەنى قىيامەتكە) سەل قارايدۇ
نَّحْنُ خَلَقْنَـٰهُمْ وَشَدَدْنَآ أَسْرَهُمْ ۖ وَإِذَا شِئْنَا بَدَّلْنَآ أَمْثَـٰلَهُمْ تَبْدِيلًا ﴿٢٨﴾
بىز ئۇلارنى ياراتتۇق، ئۇلارنىڭ بەدىنىنى مەزمۇت قىلدۇق، ئەگەر بىز خالىساق (ئۇلارنى ھالاك قىلىپ) ئورنىغا ئۇلارغا ئوخشاش باشقا ئادەمنى دەسسىتەتتۇق
إِنَّ هَـٰذِهِۦ تَذْكِرَةٌ ۖ فَمَن شَآءَ ٱتَّخَذَ إِلَىٰ رَبِّهِۦ سَبِيلًا ﴿٢٩﴾
شۈبھىسىزكى، بۇ (ئايەتلەر) ۋەز - نەسىھەتتۇر (بۇ ۋەز - نەسىھەتتىن پايدىلىنىشنى) خالىغان ئادەم پەرۋەردىگارىغا يەتكۈزىدىغان يولنى تۇتسۇن
وَمَا تَشَآءُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ ۚ إِنَّ ٱللَّهَ كَانَ عَلِيمًا حَكِيمًا ﴿٣٠﴾
پەقەت اﷲ خالىغاندىلا، ئاندىن سىلەر خالايسىلەر (يەنى ھەممە ئىش اﷲ نىڭ خاھىشىغا باغلىقتۇر). اﷲ ئىشەنچلىك ھالدا ھەممىنى بىلگۈچىدۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
يُدْخِلُ مَن يَشَآءُ فِى رَحْمَتِهِۦ ۚ وَٱلظَّـٰلِمِينَ أَعَدَّ لَهُمْ عَذَابًا أَلِيمًۢا ﴿٣١﴾
اﷲ خالىغان ئادەمنى رەھمىتىنىڭ دائىرىسىگە كىرگۈزىدۇ. اﷲ زالىملارغا قاتتىق ئازابنى تەييارلىدى
