بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
طه ﴿١﴾
طاھا (يەنى ئى مۇھەممەد!)
مَا أَنزَلْنَا عَلَيْكَ الْقُرْآنَ لِتَشْقَىٰ ﴿٢﴾
قۇرئاننى ساڭا سېنى جاپاغا سېلىش ئۈچۈن ئەمەس،
إِلَّا تَذْكِرَةً لِّمَن يَخْشَىٰ ﴿٣﴾
پەقەت (اﷲ تىن) قورقىدىغانلارغا ۋەز - نەسىھەت قىلىش ئۈچۈن نازىل قىلدۇق
تَنزِيلًا مِّمَّنْ خَلَقَ الْأَرْضَ وَالسَّمَاوَاتِ الْعُلَى ﴿٤﴾
ئۇ زېمىننى ۋە (كەڭ) ئېگىز ئاسمانلارنى ياراتقان زات تەرىپىدىن نازىل قىلىنغان
الرَّحْمَٰنُ عَلَى الْعَرْشِ اسْتَوَىٰ ﴿٥﴾
مەرھەمەتلىك اﷲ ئەرش ئۈستىدە (ئۆزىگە لايىق رەۋىشتە) قارار ئالدى
لَهُ مَا فِي السَّمَاوَاتِ وَمَا فِي الْأَرْضِ وَمَا بَيْنَهُمَا وَمَا تَحْتَ الثَّرَىٰ ﴿٦﴾
ئاسمانلاردىكى، زېمىندىكى، ئۇلارنىڭ ئارىسىدىكى ۋە يەر ئاستىدىكى شەيئىلەرنىڭ ھەممىسى اﷲ نىڭدۇر (يەنى اﷲ نىڭ مۈلكىدۇر، اﷲ نىڭ مەخلۇقاتىدۇر، اﷲ نىڭ تەسەررۇپى ئاستىدۇر)
وَإِن تَجْهَرْ بِالْقَوْلِ فَإِنَّهُ يَعْلَمُ السِّرَّ وَأَخْفَى ﴿٧﴾
ئەگەر سەن ئاشكارا سۆزلىسەڭ (ياكى يوشۇرۇن سۆزلىسەڭ، اﷲ نىڭ نەزىرىدە ئوخشاشتۇر)، چۈنكى ئۇ سىرنى ۋە ئۇنىڭدىنمۇ مەخپىيرەك ئىشلارنى بىلىپ تۇرىدۇ
اللَّهُ لَا إِلَٰهَ إِلَّا هُوَ ۖ لَهُ الْأَسْمَاءُ الْحُسْنَىٰ ﴿٨﴾
اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، اﷲ نىڭ گۈزەل ئىسىملىرى بار
وَهَلْ أَتَاكَ حَدِيثُ مُوسَىٰ ﴿٩﴾
مۇسانىڭ قىسسىسىدىن خەۋىرىڭ بارمۇ؟
إِذْ رَأَىٰ نَارًا فَقَالَ لِأَهْلِهِ امْكُثُوا إِنِّي آنَسْتُ نَارًا لَّعَلِّي آتِيكُم مِّنْهَا بِقَبَسٍ أَوْ أَجِدُ عَلَى النَّارِ هُدًى ﴿١٠﴾
ئۆز ۋاقتىدا مۇسا (بىر جايدا) ئوتنىڭ يورۇقىنى كۆردى. ئۇ ئائىلىسىدىكىلەرگە: «تۇرۇپ تۇرۇڭلار، مەن ئوتنىڭ يورۇقىنى كۆردۈم، مەن بېرىپ (ئۇنىڭدىن) بىرەر پارچە چوغ ئېلىپ كېلىشىم ياكى ئوت بار يەردە بىرەر يول باشلىغۇچىنى ئۇچرىتىشىم مۇمكىن» دېدى
فَلَمَّا أَتَاهَا نُودِيَ يَا مُوسَىٰ ﴿١١﴾
مۇسا ئوتنىڭ يېنىغا كەلگەندە نىدا قىلىندىكى، ئى مۇسا!
إِنِّي أَنَا رَبُّكَ فَاخْلَعْ نَعْلَيْكَ ۖ إِنَّكَ بِالْوَادِ الْمُقَدَّسِ طُوًى ﴿١٢﴾
مەن ئىشەنچلىك ھالدا سېنىڭ پەرۋەردىگارىڭدۇرمەن، كەشىڭنى سالغىن، سەن ئىشەنچلىك ھالدا مۇقەددەس ۋادى بولغان تۇۋادا سەن
وَأَنَا اخْتَرْتُكَ فَاسْتَمِعْ لِمَا يُوحَىٰ ﴿١٣﴾
مەن سېنى (خۇدانىڭ ئەلچىسىلىككە) تاللىدىم، سەن ۋەھيىگە قۇلاق سالغىن
إِنَّنِي أَنَا اللَّهُ لَا إِلَٰهَ إِلَّا أَنَا فَاعْبُدْنِي وَأَقِمِ الصَّلَاةَ لِذِكْرِي ﴿١٤﴾
مەن ئىشەنچلىك ھالدا (ئىبادەتكە لايىق) اﷲ دۇرمەن، مەندىن باشقا ھېچ مەبۇد (بەرھەق) يوق، (يالغۇز) ماڭا ئىبادەت قىلغىن، مېنى زىكىر قىلىش ئۈچۈن ناماز ئوقۇغىن
إِنَّ السَّاعَةَ آتِيَةٌ أَكَادُ أُخْفِيهَا لِتُجْزَىٰ كُلُّ نَفْسٍ بِمَا تَسْعَىٰ ﴿١٥﴾
قىيامەتنىڭ بولۇشى ھەقىقەتتۇر، ھەر ئادەم ئۆزىنىڭ ئەمەلىگە يارىشا مۇكاپاتلىنىشى ئۈچۈن، ئۇنى (يەنى ئۇنىڭ قاچان بولۇشىنى) مەخپىي تۇتتۇم
فَلَا يَصُدَّنَّكَ عَنْهَا مَن لَّا يُؤْمِنُ بِهَا وَاتَّبَعَ هَوَاهُ فَتَرْدَىٰ ﴿١٦﴾
قىيامەتكە ئىشەنمەيدىغان ۋە نەپسى خاھىشلىرىغا ئەگەشكەن ئادەم سېنى ئۇنىڭدىن (يەنى قىيامەتكە تەييارلىق قىلىشتىن ۋە ئۇنى تەستىق قىلىشتىن) توسمىسۇن، ئۇنداقتا ئۆزۈڭ ھالاك بولىسەن (چۈنكى ئاخىرەتتىن غاپىل بولۇش، ھالاكەتكە ئېلىپ بارىدۇ
وَمَا تِلْكَ بِيَمِينِكَ يَا مُوسَىٰ ﴿١٧﴾
ئى مۇسا! ئوڭ قولۇڭدىكى نېمە ئۇ؟»
قَالَ هِيَ عَصَايَ أَتَوَكَّأُ عَلَيْهَا وَأَهُشُّ بِهَا عَلَىٰ غَنَمِي وَلِيَ فِيهَا مَآرِبُ أُخْرَىٰ ﴿١٨﴾
مۇسا ئېيتتى: «ئۇ مېنىڭ ھاسام، ئۇنىڭغا تايىنىمەن، ئۇنىڭ بىلەن قويلىرىمغا غازاڭ قېقىپ بېرىمەن، مېنىڭ يەنە باشقا ئېھتىياجلىرىممۇ بار»
قَالَ أَلْقِهَا يَا مُوسَىٰ ﴿١٩﴾
اﷲ: «ئى مۇسا! ئۇنى (يەرگە) تاشلىغىن» دېدى
فَأَلْقَاهَا فَإِذَا هِيَ حَيَّةٌ تَسْعَىٰ ﴿٢٠﴾
مۇسا ئۇنى تاشلىۋىدى، ئۇ ناگاھان مېڭىۋاتقان بىر يىلانغا ئايلاندى
قَالَ خُذْهَا وَلَا تَخَفْ ۖ سَنُعِيدُهَا سِيرَتَهَا الْأُولَىٰ ﴿٢١﴾
اﷲ ئېيتتى: «ئۇنى تۇتۇۋالغىن، قورقمىغىن، ئۇنى دەسلەپكى (يەنى ھاسىلىق) ھالىتىگە قايتۇرىمىز
وَاضْمُمْ يَدَكَ إِلَىٰ جَنَاحِكَ تَخْرُجْ بَيْضَاءَ مِنْ غَيْرِ سُوءٍ آيَةً أُخْرَىٰ ﴿٢٢﴾
قولۇڭنى قولتۇقۇڭنىڭ ئاستىغا سالغىن (ئاندىن چىقارغىن)، ئۇ ھېچقانداق ئەيىب - نۇقسانسىز ھالدا ئاپئاق (نۇرلۇق بولۇپ چىقىدۇ، (بۇ) ئىككىنچى مۆجىزىدۇر
لِنُرِيَكَ مِنْ آيَاتِنَا الْكُبْرَى ﴿٢٣﴾
ساڭا چوڭ مۆجىزىلىرىمىزدىن بەزىسىنى كۆرسىتىمىز
اذْهَبْ إِلَىٰ فِرْعَوْنَ إِنَّهُ طَغَىٰ ﴿٢٤﴾
پىرئەۋنگە بارغىن، ئۇ ئىشەنچلىك ھالدا ھەددىدىن ئاشتى»
قَالَ رَبِّ اشْرَحْ لِي صَدْرِي ﴿٢٥﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! مېنىڭ قەلبىمنى كەڭ قىلغىن، (يەنى ئىمان بىلەن ۋە خۇدانىڭ ئەلچىسىلىك بىلەن نۇرلاندۇرغىن)
وَيَسِّرْ لِي أَمْرِي ﴿٢٦﴾
مېنىڭ ئىشىمنى ئاسانلاشتۇرغىن
وَاحْلُلْ عُقْدَةً مِّن لِّسَانِي ﴿٢٧﴾
تىلىمدىن كېكەچلىكنى كۆتۈرۈۋەتكىن
يَفْقَهُوا قَوْلِي ﴿٢٨﴾
ئۇلار مېنىڭ سۆزۇمنى چۈشەنسۇن
وَاجْعَل لِّي وَزِيرًا مِّنْ أَهْلِي ﴿٢٩﴾
ماڭا ئائىلەمدىن قېرىندىشىم ھارۇننى ياردەمچى قىلىپ بەرگىن
هَارُونَ أَخِي ﴿٣٠﴾
ماڭا ئائىلەمدىن قېرىندىشىم ھارۇننى ياردەمچى قىلىپ بەرگىن
اشْدُدْ بِهِ أَزْرِي ﴿٣١﴾
ئۇنىڭ بىلەن مېنى تېخىمۇ كۈچەيتكىن
وَأَشْرِكْهُ فِي أَمْرِي ﴿٣٢﴾
مېنىڭ ئىشىمغا ئۇنى شېرىك قىلغىن
كَيْ نُسَبِّحَكَ كَثِيرًا ﴿٣٣﴾
بۇنىڭ بىلەن بىز ساڭا كۆپ تەسبىھ ئېيتقايمىز
وَنَذْكُرَكَ كَثِيرًا ﴿٣٤﴾
ساڭا كۆپ زىكرى ئېيتقايمىز
إِنَّكَ كُنتَ بِنَا بَصِيرًا ﴿٣٥﴾
سەن ئىشەنچلىك ھالدا بىزنى كۆرۈپ تۇرغۇچىسەن»
قَالَ قَدْ أُوتِيتَ سُؤْلَكَ يَا مُوسَىٰ ﴿٣٦﴾
اﷲ ئېيتتى: «ئى مۇسا! سېنىڭ سورىغان نەرسىلىرىڭ ساڭا بېرىلدى.
وَلَقَدْ مَنَنَّا عَلَيْكَ مَرَّةً أُخْرَىٰ ﴿٣٧﴾
(بۇنىڭدىن باشقا) ساڭا يەنە بىر قېتىم ئىنئام قىلغان ئىدۇق
إِذْ أَوْحَيْنَا إِلَىٰ أُمِّكَ مَا يُوحَىٰ ﴿٣٨﴾
ئۆز ۋاقتىدا ئاناڭنىڭ كۆڭلىگە پەقەت ۋەھيى ئارقىلىق بىلىنىدىغان نەرسىلەرنى سالدۇق»
أَنِ اقْذِفِيهِ فِي التَّابُوتِ فَاقْذِفِيهِ فِي الْيَمِّ فَلْيُلْقِهِ الْيَمُّ بِالسَّاحِلِ يَأْخُذْهُ عَدُوٌّ لِّي وَعَدُوٌّ لَّهُ ۚ وَأَلْقَيْتُ عَلَيْكَ مَحَبَّةً مِّنِّي وَلِتُصْنَعَ عَلَىٰ عَيْنِي ﴿٣٩﴾
(ئېيتتۇقكى) «ئۇنى (يەنى بالىنى) ساندۇققا سالغىن، ئاندىن ئۇنى (ساندۇق بىلەن) دەرياغا تاشلىغىن، دەريا ئۇنى قىرغاققا تاشلىسۇن، ئۇنى مېنىڭ دۈشمىنىم ۋە ئۇنىڭ دۈشمىنى بولغان (پىرئەۋن) ئېلىپ (باقسۇن)». مېنىڭ قوغدىشىم ئاستىدا تەربىيىلىنىشىڭ ئۈچۈن، ساڭا مەن تەرەپتىن بولغان مۇھەببەتنى سالدىم
إِذْ تَمْشِي أُخْتُكَ فَتَقُولُ هَلْ أَدُلُّكُمْ عَلَىٰ مَن يَكْفُلُهُ ۖ فَرَجَعْنَاكَ إِلَىٰ أُمِّكَ كَيْ تَقَرَّ عَيْنُهَا وَلَا تَحْزَنَ ۚ وَقَتَلْتَ نَفْسًا فَنَجَّيْنَاكَ مِنَ الْغَمِّ وَفَتَنَّاكَ فُتُونًا ۚ فَلَبِثْتَ سِنِينَ فِي أَهْلِ مَدْيَنَ ثُمَّ جِئْتَ عَلَىٰ قَدَرٍ يَا مُوسَىٰ ﴿٤٠﴾
ئۆز ۋاقتىدا سېنىڭ ھەمشىرەڭ (پىر ئەۋىننىڭ ئائىلىسىگە) بېرىپ: «سىلەرگە ئۇنى باقىدىغان بىر كىشىنى كۆرسىتىپ قويايمۇ؟» دېدى. بىز سېنىڭ ئاناڭنى خۇش بولسۇن، قايغۇرمىسۇن دەپ سېنى ئۇنىڭغا قايتۇرۇپ بەردۇق، سەن بىر ئادەمنى ئۆلتۈرۈپ قويدۇڭ، سېنى بىز غەمدىن خالاس قىلدۇق، سېنى تۈرلۈك بالالار بىلەن سىنىدۇق، سەن مەديەنلىكلەر ئارىسىدا كۆپ يىللار تۇردۇڭ، ئاندىن، ئى مۇسا! تەقدىر بىلەن (بۇ يەرگە) كەلدىڭ
وَاصْطَنَعْتُكَ لِنَفْسِي ﴿٤١﴾
سېنى مەن مېنىڭ مۇھەببىتىمگە (يەنى خۇدانىڭ ئەلچىسى قىلىپ ۋەھيى نازىل قىلىشىمغا) تاللىدىم
اذْهَبْ أَنتَ وَأَخُوكَ بِآيَاتِي وَلَا تَنِيَا فِي ذِكْرِي ﴿٤٢﴾
(ئى مۇسا!) سەن ۋە سېنىڭ قېرىندىشىڭ (ھارۇن) مېنىڭ مۆجىزىلىرىمنى ئېلىپ بېرىڭلار، مېنىڭ زىكرىمگە سۇسلۇق قىلىپ قالماڭلار
اذْهَبَا إِلَىٰ فِرْعَوْنَ إِنَّهُ طَغَىٰ ﴿٤٣﴾
پىرئەۋننىڭ يېنىغا بېرىڭلار، ئۇ ئىشەنچلىك ھالدا ھەددىدىن ئاشتى
فَقُولَا لَهُ قَوْلًا لَّيِّنًا لَّعَلَّهُ يَتَذَكَّرُ أَوْ يَخْشَىٰ ﴿٤٤﴾
ئۇنىڭغا يۇمشاق سۆز قىلىڭلار، ئۇ ۋەز - نەسىھەتنى قوبۇل قىلىشى، ياكى (ھەددىدىن ئاشقانلىقنىڭ ئاقىۋىتىدىن) قورقۇشى مۇمكىن
قَالَا رَبَّنَا إِنَّنَا نَخَافُ أَن يَفْرُطَ عَلَيْنَا أَوْ أَن يَطْغَىٰ ﴿٤٥﴾
ئۇلار: «پەرۋەردىگارىمىز! بىز ئىشەنچلىك ھالدا ئۇنىڭ بىزنى ئالدىراپ جازالىشىدىن ياكى تېخىمۇ ھەددىدىن ئېشىشىدىن قورقىمىز» دېيىشتى
قَالَ لَا تَخَافَا ۖ إِنَّنِي مَعَكُمَا أَسْمَعُ وَأَرَىٰ ﴿٤٦﴾
اﷲ ئېيتتى: «سىلەر قورقماڭلار، مەن ئىشەنچلىك ھالدا سىلەر بىلەن بىللە (ئۇنىڭ سىلەرگە بەرگەن جاۋابىنى) ئاڭلاپ تۇرىمەن، (ئۇنىڭ سىلەرگە قىلىدىغانلىرىنى) كۆرۈپ تۇرىمەن
فَأْتِيَاهُ فَقُولَا إِنَّا رَسُولَا رَبِّكَ فَأَرْسِلْ مَعَنَا بَنِي إِسْرَائِيلَ وَلَا تُعَذِّبْهُمْ ۖ قَدْ جِئْنَاكَ بِآيَةٍ مِّن رَّبِّكَ ۖ وَالسَّلَامُ عَلَىٰ مَنِ اتَّبَعَ الْهُدَىٰ ﴿٤٧﴾
سىلەر پىرئەۋننىڭ يېنىغا بېرىپ، بىز ئىككىمىز پەرۋەردىگارىڭنىڭ ئەلچىسى بولىمىز، ئىسرائىل ئەۋلادىنى قويۇۋەتكىن، بىز بىلەن كەتسۇن، ئۇلارنى قىينىمىغىن. ساڭا بىز ئىشەنچلىك ھالدا پەرۋەردىگارىڭنىڭ مۆجىزىسىنى ئېلىپ كەلدۇق، توغرا يولغا ئەگەشكەن ئادەم (اﷲ نىڭ ئازابىدىن) ئامان قالىدۇ
إِنَّا قَدْ أُوحِيَ إِلَيْنَا أَنَّ الْعَذَابَ عَلَىٰ مَن كَذَّبَ وَتَوَلَّىٰ ﴿٤٨﴾
بىزگە شەك - شۈبھىسىز ۋەھيى قىلىندىكى، (اﷲ نىڭ خۇدانىڭ ئەلچىسىلىرىنى) ئىنكار قىلغان، (ئىماندىن) يۈز ئۆرىگەن ئادەم (قاتتىق) ئازابقا دۇچار بولىدۇ دەڭلار»
قَالَ فَمَن رَّبُّكُمَا يَا مُوسَىٰ ﴿٤٩﴾
(ئۇلار پىرئەۋنگە كېلىپ اﷲ تائالا تەرىپىدىن ۋەھيى قىلىنغان نەرسىلەرنى يەتكۈزگەندىن كېيىن) پىرئەۋن: «ئى مۇسا، سىلەرنىڭ پەرۋەردىگارىڭلار كىم؟» دېدى
قَالَ رَبُّنَا الَّذِي أَعْطَىٰ كُلَّ شَيْءٍ خَلْقَهُ ثُمَّ هَدَىٰ ﴿٥٠﴾
مۇسا ئېيتتى: «پەرۋەردىگارىمىز شۇنداق زاتتۇركى، ھەممە نەرسىگە (ئۆزىگە مۇناسىپ) شەكىل ئاتا قىلدى، (ئاندىن ئۇلارغا ياشاش يوللىرىنى، پايدىلىنىدىغان نەرسىلەرنى) كۆرسەتتى»
قَالَ فَمَا بَالُ الْقُرُونِ الْأُولَىٰ ﴿٥١﴾
پىرئەۋن ئېيتتى: «ئۆتۈپ كەتكەن ئۈممەتلەرنى ھالى قانداق؟ (يەنى قانداق بولغان؟)»
قَالَ عِلْمُهَا عِندَ رَبِّي فِي كِتَابٍ ۖ لَّا يَضِلُّ رَبِّي وَلَا يَنسَى ﴿٥٢﴾
مۇسا ئېيتتى: «ئۇلار توغرىسىدىكى مەلۇمات پەرۋەردىگارىمنىڭ دەرگاھىدا لەۋھۇلمەھپۇزدا (خاتىرىلەكلىكتۇر)، پەرۋەردىگارىم ھەم خاتالاشمايدۇ ھەم ئۇنتۇمايدۇ
الَّذِي جَعَلَ لَكُمُ الْأَرْضَ مَهْدًا وَسَلَكَ لَكُمْ فِيهَا سُبُلًا وَأَنزَلَ مِنَ السَّمَاءِ مَاءً فَأَخْرَجْنَا بِهِ أَزْوَاجًا مِّن نَّبَاتٍ شَتَّىٰ ﴿٥٣﴾
ئۇ زېمىننى سىلەرگە بىسات قىلىپ بەردى، سىلەر ئۈچۈن نۇرغۇن يوللارنى ئېچىپ بەردى، بۇلۇتتىن يامغۇر ياغدۇرۇپ بەردى، ئۇنىڭ بىلەن نۇرغۇن ئۆسۈملۈكلەرنى ئۆستۈرۈپ بەردى»
كُلُوا وَارْعَوْا أَنْعَامَكُمْ ۗ إِنَّ فِي ذَٰلِكَ لَآيَاتٍ لِّأُولِي النُّهَىٰ ﴿٥٤﴾
(ئۇلاردىن) يەڭلار ۋە مال چارۋىلىرىڭلارنى بېقىڭلار، ئۇنىڭدا ئەقىل ئىگىلىرى ئۈچۈن (اﷲ نىڭ بارلىقى ۋە بىرلىكىنى كۆرسىتىدىغان) نۇرغۇن دەلىللەر بار
مِنْهَا خَلَقْنَاكُمْ وَفِيهَا نُعِيدُكُمْ وَمِنْهَا نُخْرِجُكُمْ تَارَةً أُخْرَىٰ ﴿٥٥﴾
سىلەرنى زېمىندىن خەلق ئەتتۇق، (ئۆلگۈنۈڭلاردىن كېيىن) سىلەرنى يەنە زېمىنغا قايتۇرىمىز (توپىغا ئايلىنىسىلەر)، سىلەرنى (ھېساب ئېلىش ئۈچۈن) يەنە بىر قېتىم زېمىندىن چىقىرىمىز
وَلَقَدْ أَرَيْنَاهُ آيَاتِنَا كُلَّهَا فَكَذَّبَ وَأَبَىٰ ﴿٥٦﴾
شەك - شۈبھىسىزكى، ئۇنىڭغا بىزنىڭ ھەممە مۆجىزىلىرىمىزنى كۆرسەتتۇق. ئۇ ئىنكار قىلدى ۋە (ئىمان ئېيتىشتىن) باش تارتتى
قَالَ أَجِئْتَنَا لِتُخْرِجَنَا مِنْ أَرْضِنَا بِسِحْرِكَ يَا مُوسَىٰ ﴿٥٧﴾
پىرئەۋن ئېيتتى: «ئى مۇسا! سېھرىڭ بىلەن بىزنى زېمىنىمىز (مىسىر) دىن چىقىرىۋېتىش ئۈچۈن كەلدىڭمۇ؟
فَلَنَأْتِيَنَّكَ بِسِحْرٍ مِّثْلِهِ فَاجْعَلْ بَيْنَنَا وَبَيْنَكَ مَوْعِدًا لَّا نُخْلِفُهُ نَحْنُ وَلَا أَنتَ مَكَانًا سُوًى ﴿٥٨﴾
ساڭا بىز شۇنىڭغا ئوخشاش سېھىر بىلەن تاقابىل تۇرىمىز، سەن بىزگە ئۆز ئارا يىغىلىدىغان ۋاقىتنى مۇئەييەن جاينى بەلگىلەپ بەرگىن، سەنمۇ، بىزمۇ ئۇنىڭغا خىلاپلىق قىلمايلى»
قَالَ مَوْعِدُكُمْ يَوْمُ الزِّينَةِ وَأَن يُحْشَرَ النَّاسُ ضُحًى ﴿٥٩﴾
مۇسا ئېيتتى: «سىلەرگە ۋەدە قىلىنغان چاغ بايرام كۈنى بولۇپ، كىشىلەر چاشكا ۋاقتىدا يىغىلسۇن»
فَتَوَلَّىٰ فِرْعَوْنُ فَجَمَعَ كَيْدَهُ ثُمَّ أَتَىٰ ﴿٦٠﴾
پىرئەۋن قايتىپ كېتىپ ھىيلە - مىكرنى (يەنى سېھىرگەرلەرنى) توپلىدى. ئاندىن ئۇ (سېھىرگەرلەرنى ۋە سېھرىدە ئىشلىنىدىغان نەرسىلەرنى ئېلىپ ۋەدىلەشكەن ۋاقىتتا) كەلدى
قَالَ لَهُم مُّوسَىٰ وَيْلَكُمْ لَا تَفْتَرُوا عَلَى اللَّهِ كَذِبًا فَيُسْحِتَكُم بِعَذَابٍ ۖ وَقَدْ خَابَ مَنِ افْتَرَىٰ ﴿٦١﴾
مۇسا ئۇلار غا ئېيتتى: «سىلەرگە ۋاي! اﷲ قا يالغاننى چاپلىماڭلار (يەنى سېھرىڭلار ئارقىلىق اﷲ نىڭ مۆجىزىلىرىگە تاقابىل تۇرۇشنى دەۋا قىلماڭلار)، اﷲ سىلەرنى قاتتىق ئازاب بىلەن ھالاك قىلىدۇ، اﷲ قا بوھتان چاپلىغان ئادەم مەغلۇب بولىدۇ»
فَتَنَازَعُوا أَمْرَهُم بَيْنَهُمْ وَأَسَرُّوا النَّجْوَىٰ ﴿٦٢﴾
ئۇلار ئۆز ئىشى (يەنى مۇسا ۋە ھارۇننىڭ ئىشى) ئۈستىدە دەتالاش قىلىشتى، قىلغان دەتالاشنى يوشۇرۇشتى
قَالُوا إِنْ هَٰذَانِ لَسَاحِرَانِ يُرِيدَانِ أَن يُخْرِجَاكُم مِّنْ أَرْضِكُم بِسِحْرِهِمَا وَيَذْهَبَا بِطَرِيقَتِكُمُ الْمُثْلَىٰ ﴿٦٣﴾
ئۇلار ئېيتتى: «بۇ ئىككىسى راستلا سېھرىگەر بولۇپ، سېھىر بىلەن سىلەرنى زېمىنىڭلاردىن چىقىرىۋەتمەكچى ۋە توغرا دىنىڭلارنى يوقاتماقچى بولىدۇ
فَأَجْمِعُوا كَيْدَكُمْ ثُمَّ ائْتُوا صَفًّا ۚ وَقَدْ أَفْلَحَ الْيَوْمَ مَنِ اسْتَعْلَىٰ ﴿٦٤﴾
ھىيلەڭلارنىڭ ھەممىسىنى جەملەڭلار (يەنى پۈتۈن چارە - تەدبىرلىرىڭلارنى بىر نۇقتىغا يىغىپ بىر ياقىدىن باش چىقىرىڭلار)، ئاندىن (قارىغۇچىلارغا ھەيۋەتلىك كۆرۈنۈش ئۈچۈن مەيدانغا) سەپ تارتىپ كېلىڭلار، بۈگۈن ئۈستۈنلۈك قازانغان ئادەم راستلا مەقسىتىگە ئېرىشىدۇ»
قَالُوا يَا مُوسَىٰ إِمَّا أَن تُلْقِيَ وَإِمَّا أَن نَّكُونَ أَوَّلَ مَنْ أَلْقَىٰ ﴿٦٥﴾
ئۇلار ئېيتتى: «ئى مۇسا! (ھاساڭنى) سەن ئاۋۋال تاشلامسەن؟ ياكى (ئارغامچا ھاسىلىرىمىزنى) بىز ئاۋۋال تاشلامدۇق؟»
قَالَ بَلْ أَلْقُوا ۖ فَإِذَا حِبَالُهُمْ وَعِصِيُّهُمْ يُخَيَّلُ إِلَيْهِ مِن سِحْرِهِمْ أَنَّهَا تَسْعَىٰ ﴿٦٦﴾
مۇسا: «بەلكى سىلەر ئاۋۋال تاشلاڭلار!» دېدى. (ئۇلار تاشلىغان ئىدى) ناگاھان ئۇلارنىڭ ئاغامچىلىرى، ھاسىلىرى ئۇلارنىڭ سېھىرىدىن (يەنى سېھىرنىڭ تەسىرىدىن) ئۇنىڭغا ھەرىكەتلىنىپ مېڭىۋاتقاندەك تۇيۇلدى
فَأَوْجَسَ فِي نَفْسِهِ خِيفَةً مُّوسَىٰ ﴿٦٧﴾
(بۇنىڭدىن) مۇسا ئۆزىدە قورقۇنچ ھېس قىلدى
قُلْنَا لَا تَخَفْ إِنَّكَ أَنتَ الْأَعْلَىٰ ﴿٦٨﴾
بىز (ئۇنىڭغا) ئېيتتۇقكى، «قورقمىغىن سەن چوقۇم ئۈستۈنلۈك قازىنىسەن،
وَأَلْقِ مَا فِي يَمِينِكَ تَلْقَفْ مَا صَنَعُوا ۖ إِنَّمَا صَنَعُوا كَيْدُ سَاحِرٍ ۖ وَلَا يُفْلِحُ السَّاحِرُ حَيْثُ أَتَىٰ ﴿٦٩﴾
قولۇڭدىكىنى تاشلىغىن، ئۇ ئۇلارنىڭ كۆرسەتكەن سېھىرلىرىنى دەم تارتىپ يۇتۇۋېتىدۇ، ئۇلارنىڭ كۆرسەتكەنلىكىنى سېھىرگەرلەرنىڭ ھىيلىسىدۇر، سېھىرگەرلەر قەيەرگە بارسا مۇۋەپپەقىيەت قازىنالمايدۇ»
فَأُلْقِيَ السَّحَرَةُ سُجَّدًا قَالُوا آمَنَّا بِرَبِّ هَارُونَ وَمُوسَىٰ ﴿٧٠﴾
(مۇسا ھاسىسىنى تاشلىۋىدى، ئۇلارنىڭ كۆرسەتكەن سېھىرلىرىنى دەم تارتىپ يۇتۇۋەتتى) شۇنىڭ بىلەن سېھىرگەرلەر (سەجدە قىلغان ھالدا يىقىلىپ): «ھارۇن بىلەن مۇسانىڭ پەرۋەردىگارىغا ئىمان ئېيتتۇق» دېيىشتى
قَالَ آمَنتُمْ لَهُ قَبْلَ أَنْ آذَنَ لَكُمْ ۖ إِنَّهُ لَكَبِيرُكُمُ الَّذِي عَلَّمَكُمُ السِّحْرَ ۖ فَلَأُقَطِّعَنَّ أَيْدِيَكُمْ وَأَرْجُلَكُم مِّنْ خِلَافٍ وَلَأُصَلِّبَنَّكُمْ فِي جُذُوعِ النَّخْلِ وَلَتَعْلَمُنَّ أَيُّنَا أَشَدُّ عَذَابًا وَأَبْقَىٰ ﴿٧١﴾
پىرئەۋن ئېيتتى: «سىلەر مەن رۇخسەت بېرىشتىن بۇرۇن ئۇنىڭغا ئىمان ئېيتتىڭلار، شۈبھىسىزكى، ئۇ چوقۇم سىلەرگە سېھىرنى ئۆگەتكەن چوڭۇڭلار ئىكەن، مەن چوقۇم سىلەرنىڭ قوللىرىڭلارنى، پۇتلىرىڭلارنى ئوڭ - چەپ قىلىپ (يەنى ئوڭ قول بىلەن سول پۇتنى، ئوڭ پۇت بىلەن سول قولنى) كېسىپ تاشلايمەن، سىلەرنى چوقۇم خورما دەرەخلىرىگە ئاسىمەن، سىلەر قايسىمىزنىڭ (يەنى مېنىڭ ياكى مۇسانىڭ پەرۋەردىگارىنىڭ) ئازابىنىڭ ئەڭ قاتتىق ۋە ئەڭ ئۇزۇن ئىكەنلىكىنى چوقۇم بىلىسىلەر»
قَالُوا لَن نُّؤْثِرَكَ عَلَىٰ مَا جَاءَنَا مِنَ الْبَيِّنَاتِ وَالَّذِي فَطَرَنَا ۖ فَاقْضِ مَا أَنتَ قَاضٍ ۖ إِنَّمَا تَقْضِي هَٰذِهِ الْحَيَاةَ الدُّنْيَا ﴿٧٢﴾
ئۇلار ئېيتتى: «سېنى (يەنى ساڭا ئەگىشىشنى) ئىختىيار قىلىپ، بىزگە كەلگەن (مۇسانىڭ راستلىقىنى كۆرسىتىدىغان) روشەن دەلىللەرنى ۋە بىزنى خەلق ئەتكەن اﷲ نى ھەرگىز تاشلىمايمىز. (بىزنىڭ ھەققىمىزدە) نېمىنى ھۆكۈم قىلساڭ، شۇنى ھۆكۈم قىلغىن، سېنىڭ ھۆكمىڭ پەقەت بۇ دۇنيادىلا ئۆتىدۇ
إِنَّا آمَنَّا بِرَبِّنَا لِيَغْفِرَ لَنَا خَطَايَانَا وَمَا أَكْرَهْتَنَا عَلَيْهِ مِنَ السِّحْرِ ۗ وَاللَّهُ خَيْرٌ وَأَبْقَىٰ ﴿٧٣﴾
بىزنىڭ خاتالىقىمىزنى، سېنىڭ مەجبۇرلىشىڭ ئارقىسىدا بىزنىڭ سېھىر كۆرسەتكەنلىكىمىزدىن ئىبارەت گۇناھىمىزنى مەغپىرەت قىلسۇن دەپ بىز ئىشەنچلىك ھالدا پەرۋەردىگارىمىزغا ئىمان ئېيتتۇق، اﷲ نىڭ (ساۋابىنى سېنىڭ مۇكاپاتىڭدىن) ئارتۇقتۇر؛ اﷲ نىڭ (ئازابى سېنىڭ جازايىڭدىن) دائىمىيدۇر»
إِنَّهُ مَن يَأْتِ رَبَّهُ مُجْرِمًا فَإِنَّ لَهُ جَهَنَّمَ لَا يَمُوتُ فِيهَا وَلَا يَحْيَىٰ ﴿٧٤﴾
كىمكى گۇناھكار ھالدا اﷲ قا مۇلاقات بولىدىكەن، ئۇ دوزاخقا كىرىدۇ، دوزاختا ئۇ ئۆلمەيدۇ ۋە (ئوبدان) ياشىمايدۇ
وَمَن يَأْتِهِ مُؤْمِنًا قَدْ عَمِلَ الصَّالِحَاتِ فَأُولَٰئِكَ لَهُمُ الدَّرَجَاتُ الْعُلَىٰ ﴿٧٥﴾
كىمكى مۆمىن بولۇپ، ياخشى ئەمەللەرنى قىلغان ھالدا اﷲ قا مۇلاقات بولىدىكەن، ئۇ (اﷲ نىڭ دەرگاھىدا) يۇقىرى دەرىجىگە ئېرىشىدۇ
جَنَّاتُ عَدْنٍ تَجْرِي مِن تَحْتِهَا الْأَنْهَارُ خَالِدِينَ فِيهَا ۚ وَذَٰلِكَ جَزَاءُ مَن تَزَكَّىٰ ﴿٧٦﴾
ئۇ (دەرىجىلەر) دائىمىي تۇرالغۇ بولغان مەڭگۈلۈك باغلارلەر بولۇپ، ئۇلارنىڭ ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدۇ، ئۇ يەرلەردە ئۇ مەڭگۈ قالىدۇ، ئەنە شۇ (كۇفرىدىن ۋە گۇناھلاردىن) پاك بولغان ئادەمگە بېرىلىدىغان مۇكاپاتتۇر
وَلَقَدْ أَوْحَيْنَا إِلَىٰ مُوسَىٰ أَنْ أَسْرِ بِعِبَادِي فَاضْرِبْ لَهُمْ طَرِيقًا فِي الْبَحْرِ يَبَسًا لَّا تَخَافُ دَرَكًا وَلَا تَخْشَىٰ ﴿٧٧﴾
بىز شەك - شۈبھىسىز مۇساغا ۋەھيى قىلدۇقكى، «سەن مېنىڭ بەندىلىرىمنى كېچىدە (مىسىردىن) ئېلىپ چىقىپ كەت، سەن (ھاساڭنى) دېڭىزغا ئۇرۇپ ئۇلارغا بىر قۇرۇق يول ئېچىپ بەرگىن، (پىرئەۋننىڭ ئارقاڭدىن قوغلاپ) يېتىۋېلىشىدىن قورقمىغىن، (دېڭىزدا غەرق بولۇپ كېتىشتىنمۇ) قورقمىغىن»
فَأَتْبَعَهُمْ فِرْعَوْنُ بِجُنُودِهِ فَغَشِيَهُم مِّنَ الْيَمِّ مَا غَشِيَهُمْ ﴿٧٨﴾
پىرئەۋن لەشكەرلىرى بىلەن قوغلىدى، ئۇلارنى (يەنى پىرئەۋن بىلەن ئۇنىڭ لەشكەرلىرىنى) دېڭىزدا دەھشەتلىك ھالەت ئورىۋالدى (يەنى ئۇلار دېڭىزدا غەرق بولدى)
وَأَضَلَّ فِرْعَوْنُ قَوْمَهُ وَمَا هَدَىٰ ﴿٧٩﴾
پىر ئەۋن ئۆز قەۋمىنى ئازدۇردى، (ئۇلارنى) توغرا يولغا باشلىمىدى
يَا بَنِي إِسْرَائِيلَ قَدْ أَنجَيْنَاكُم مِّنْ عَدُوِّكُمْ وَوَاعَدْنَاكُمْ جَانِبَ الطُّورِ الْأَيْمَنَ وَنَزَّلْنَا عَلَيْكُمُ الْمَنَّ وَالسَّلْوَىٰ ﴿٨٠﴾
ئى ئىسرائىل ئەۋلادى! سىلەرنى ئىشەنچلىك ھالدا دۈشمىنىڭلاردىن قۇتقۇزدۇق، (مۇناجات ئۈچۈن ۋە مۇساغا تەۋراتنى نازىل قىلىش ئۈچۈن) سىلەرگە (تۇرسىنا) نىڭ ئوڭ تەرىپىنى ۋەدە قىلدۇق، سىلەرگە تەرەنجىبىن بىلەن بۆدۈنىنى چۈشۈرۈپ بەردۇق
كُلُوا مِن طَيِّبَاتِ مَا رَزَقْنَاكُمْ وَلَا تَطْغَوْا فِيهِ فَيَحِلَّ عَلَيْكُمْ غَضَبِي ۖ وَمَن يَحْلِلْ عَلَيْهِ غَضَبِي فَقَدْ هَوَىٰ ﴿٨١﴾
(ئېيتتۇقكى) «بىز رىزىق قىلىپ بەرگەن پاك - ھالال نەرسىلەردىن يەڭلار، بۇ ھەقتە چەكتىن ئاشماڭلار، (ئۇنداق قىلساڭلار) سىلەرگە غەزىپىم نازىل بولىدۇ، كىمگە مېنىڭ غەزىپىم نازىل بولىدىكەن، ئۇ چوقۇم ھالاك بولىدۇ
وَإِنِّي لَغَفَّارٌ لِّمَن تَابَ وَآمَنَ وَعَمِلَ صَالِحًا ثُمَّ اهْتَدَىٰ ﴿٨٢﴾
تەۋبە قىلغان، ئىمان ئېيتقان، ياخشى ئەمەللەرنى قىلغان، ئاندىن توغرا يولدا ماڭغان ئادەمنى مەن ئەلۋەتتە ناھايىتى مەغپىرەت قىلغۇچىمەن»
وَمَا أَعْجَلَكَ عَن قَوْمِكَ يَا مُوسَىٰ ﴿٨٣﴾
«ئى مۇسا! ئالدىراپ (تۇرسىناغا) قەۋمىڭدىن ئىلگىرى كېلىشىڭنىڭ سەۋەبى نېمە؟» دېدۇق
قَالَ هُمْ أُولَاءِ عَلَىٰ أَثَرِي وَعَجِلْتُ إِلَيْكَ رَبِّ لِتَرْضَىٰ ﴿٨٤﴾
مۇسا ئېيتتى: «ئۇلار ئىزىمنى قوغلىشىپ كېلىۋاتىدۇ، پەرۋەردىگارىم! سېنى رازى بولسۇن دەپ بۇ يەرگە ئالدىراپ كەلدىم»
قَالَ فَإِنَّا قَدْ فَتَنَّا قَوْمَكَ مِن بَعْدِكَ وَأَضَلَّهُمُ السَّامِرِيُّ ﴿٨٥﴾
اﷲ ئېيتتى: «سەندىن (يەنى سەن كەتكەندىن) كېيىن قەۋمىڭنى ئىشەنچلىك ھالدا سىنىدۇق، ئۇلارنى سامىرى ئازدۇردى»
فَرَجَعَ مُوسَىٰ إِلَىٰ قَوْمِهِ غَضْبَانَ أَسِفًا ۚ قَالَ يَا قَوْمِ أَلَمْ يَعِدْكُمْ رَبُّكُمْ وَعْدًا حَسَنًا ۚ أَفَطَالَ عَلَيْكُمُ الْعَهْدُ أَمْ أَرَدتُّمْ أَن يَحِلَّ عَلَيْكُمْ غَضَبٌ مِّن رَّبِّكُمْ فَأَخْلَفْتُم مَّوْعِدِي ﴿٨٦﴾
مۇسا قەۋمىگە غەزەپلەنگەن ھالدا خاپا قايتتى، ئۇ ئېيتتى: «سىلەرگە پەرۋەردىگارىڭلار چىرايلىق ۋەدە قىلمىغانمىدى؟ (سىلەر ئەھدىنى ئۇنتۇغۇدەك) ۋاقىت شۇنچە ئۇزۇن بولۇپ كەتتىمۇ؟ ياكى سىلەر ئۆزۈڭلار پەرۋەردىگارىڭلارنىڭ غەزىپى چۈشۈشىنى ئىرادە قىلدىڭلارمۇ؟ شۇنىڭ ئۈچۈن سىلەرنىڭ ماڭا بەرگەن ۋەدەڭلارغا خىلاپلىق قىلدىڭلارمۇ؟»
قَالُوا مَا أَخْلَفْنَا مَوْعِدَكَ بِمَلْكِنَا وَلَٰكِنَّا حُمِّلْنَا أَوْزَارًا مِّن زِينَةِ الْقَوْمِ فَقَذَفْنَاهَا فَكَذَٰلِكَ أَلْقَى السَّامِرِيُّ ﴿٨٧﴾
ئۇلار ئېيتتى: «ساڭا بەرگەن ۋەدىگە ئۆزلۈكىمىزدىن خىلاپلىق قىلغىنىمىز يوق، لېكىن بىز (پىرئەۋن) قەۋمىنىڭ زىننەتلىرىدىن نۇرغۇن ئېلىپ چىققان ئىدۇق، (سامىرىنىڭ بۇيرۇقى بويىچە) ئۇلارنى (ئوتقا) تاشلىدۇق، سامىرىمۇ (ئۆزى ئېلىپ چىققان زىبۇزىننەتلىرىنى) تاشلىدى»
فَأَخْرَجَ لَهُمْ عِجْلًا جَسَدًا لَّهُ خُوَارٌ فَقَالُوا هَٰذَا إِلَٰهُكُمْ وَإِلَٰهُ مُوسَىٰ فَنَسِيَ ﴿٨٨﴾
سامىرى (ئېرىتىلگەن زىبۇزىننەتلەردىن) ئۇلار ئۈچۈن بىر موزاي ھەيكەل ياسىدى، ئۇنىڭدىن موزاينىڭ ئاۋازىغا ئوخشاش ئاۋاز چىقاتتى، ئۇلار (يەنى سامىرى ۋە ئۇنىڭ تەۋەلىرى) ئېيتتى: «بۇ سىلەرنىڭ ئىلاھىڭلاردۇر ۋە مۇسانىڭ ئىلاھىدۇر، مۇسا (ئىلاھنى بۇ يەردە) ئۇنتۇپ قېلىپ (ئۇنى ئىزدەپ تۇرسىناغا كەتتى)»
أَفَلَا يَرَوْنَ أَلَّا يَرْجِعُ إِلَيْهِمْ قَوْلًا وَلَا يَمْلِكُ لَهُمْ ضَرًّا وَلَا نَفْعًا ﴿٨٩﴾
ئۇ موزاينىڭ ئۇلارنىڭ سۆزىگە جاۋاب بېرەلمەيدىغانلىقىنى، ئۇلارغا پايدا - زىيان يەتكۈزەلمەيدىغانلىقىنى ئۇلار ئۇقمامدۇ؟
وَلَقَدْ قَالَ لَهُمْ هَارُونُ مِن قَبْلُ يَا قَوْمِ إِنَّمَا فُتِنتُم بِهِ ۖ وَإِنَّ رَبَّكُمُ الرَّحْمَٰنُ فَاتَّبِعُونِي وَأَطِيعُوا أَمْرِي ﴿٩٠﴾
شەك - شۈبھىسىزكى، ھارۇن ئۇلارغا (مۇسا قايتىپ كېلىشتىن) بۇرۇن: «ئى قەۋمىم! سىلەر موزاي بىلەن قايمۇقتۇرۇلدۇڭلار، سىلەرنىڭ پەرۋەردىگارىڭلار مەرھەمەتلىك اﷲ دۇر، ماڭا بوي سۇنۇڭلار، (موزايغا چوقۇنۇشنى تەرك ئېتىش بىلەن) بۇيرىقىمغا ئىتائەت قىلىڭلار!» دېدى
قَالُوا لَن نَّبْرَحَ عَلَيْهِ عَاكِفِينَ حَتَّىٰ يَرْجِعَ إِلَيْنَا مُوسَىٰ ﴿٩١﴾
ئۇلار: «مۇسا قايتىپ كەلگەنگە قەدەر ئۇنىڭغا چوقۇنىۋېرىمىز» دېيىشتى
قَالَ يَا هَارُونُ مَا مَنَعَكَ إِذْ رَأَيْتَهُمْ ضَلُّوا ﴿٩٢﴾
مۇسا ئېيتتى: «ئى ھارۇن! ئۇلارنىڭ ئازغانلىقىنى كۆرگەن چېغىڭدا (خۇدالىق ئۈچۈن غەزەپلىنىشتە) ماڭا ئەگىشىشتىن نېمە توسالغۇ بولدى؟ مېنىڭ بۇيرىقىمغا خىلاپلىق قىلدىڭمۇ؟»
أَلَّا تَتَّبِعَنِ ۖ أَفَعَصَيْتَ أَمْرِي ﴿٩٣﴾
مۇسا ئېيتتى: «ئى ھارۇن! ئۇلارنىڭ ئازغانلىقىنى كۆرگەن چېغىڭدا (خۇدالىق ئۈچۈن غەزەپلىنىشتە) ماڭا ئەگىشىشتىن نېمە توسالغۇ بولدى؟ مېنىڭ بۇيرىقىمغا خىلاپلىق قىلدىڭمۇ؟»
قَالَ يَا ابْنَ أُمَّ لَا تَأْخُذْ بِلِحْيَتِي وَلَا بِرَأْسِي ۖ إِنِّي خَشِيتُ أَن تَقُولَ فَرَّقْتَ بَيْنَ بَنِي إِسْرَائِيلَ وَلَمْ تَرْقُبْ قَوْلِي ﴿٩٤﴾
ھارۇن ئېيتتى: «ئى قېرىندىشىم! مېنىڭ ساقىلىمنى ۋە چېچىمنى تارتمىغىن، مەن ئىشەنچلىك ھالدا سېنىڭ، ئىسرائىل ئەۋلادىنى بۆلۈۋېتىپسەن، سۆزۈمگە دىققەت قىلماپسەن، دېيىشىڭدىن قورقتۇم»
قَالَ فَمَا خَطْبُكَ يَا سَامِرِيُّ ﴿٩٥﴾
مۇسا ئېيتتى: «ئى سامىرى! سەن نېمە بولدۇڭ؟ (يەنى مۇنداق يامان ئىشنى قىلىشىڭدىن غەرىزىڭ نېمە؟)»
قَالَ بَصُرْتُ بِمَا لَمْ يَبْصُرُوا بِهِ فَقَبَضْتُ قَبْضَةً مِّنْ أَثَرِ الرَّسُولِ فَنَبَذْتُهَا وَكَذَٰلِكَ سَوَّلَتْ لِي نَفْسِي ﴿٩٦﴾
سامىرى ئېيتتى: «مەن ئۇلار كۆرمىگەننى كۆردۈم، ئەلچى (يەنى جىبرىئىل) نىڭ ئىزىدىن بىر چاڭگال توپىنى ئالدىم - دە، ئۇنى (موزايغا) چاچتىم، شۇنىڭ بىلەن ئۇنىڭدىن موزاينىڭ ئاۋازىغا ئوخشاش ئاۋاز چىقتى)، شۇنىڭدەك (بۇنى) ماڭا نەپسىم چىرايلىق كۆرسەتتى»
قَالَ فَاذْهَبْ فَإِنَّ لَكَ فِي الْحَيَاةِ أَن تَقُولَ لَا مِسَاسَ ۖ وَإِنَّ لَكَ مَوْعِدًا لَّن تُخْلَفَهُ ۖ وَانظُرْ إِلَىٰ إِلَٰهِكَ الَّذِي ظَلْتَ عَلَيْهِ عَاكِفًا ۖ لَّنُحَرِّقَنَّهُ ثُمَّ لَنَنسِفَنَّهُ فِي الْيَمِّ نَسْفًا ﴿٩٧﴾
مۇسا (سامىرىغا) ئېيتتى: «سەن بارغىن! سەن ئۆمۈرۋايەت (كۆرگەنلا كىشىگە) ماڭا يېقىنلاشماڭلار! دېگەيسەن (يەنى سېنىڭ كىشىگە يېقىنلاشماسلىقىڭ، كىشىنىڭ ساڭا يېقىنلاشماسلىقى بۇ دۇنيادا ساڭا بېرىلگەن جازادۇر)، شۈبھىسىزكى (ئاخىرەتتە جازالىنىدىغان) مۇئەييەن بىر ۋاقتىڭ بار، اﷲ ساڭا قىلغان بۇ ۋەدىسىگە خىلاپلىق قىلمايدۇ، سەن ھامان چوقۇنۇپ كېلىۋاتقان ئىلاھىڭغا قارىغىن، بىز ئۇنى چوقۇم (ئوتتا) كۆيدۈرىمىز، ئاندىن ئۇنىڭ (كۈلىنى) دېڭىزغا چېچىۋېتىمىز»
إِنَّمَا إِلَٰهُكُمُ اللَّهُ الَّذِي لَا إِلَٰهَ إِلَّا هُوَ ۚ وَسِعَ كُلَّ شَيْءٍ عِلْمًا ﴿٩٨﴾
سىلەرنىڭ ئىلاھىڭلار بىر اﷲ دۇر، ئۇنىڭدىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، ئۇ ھەر نەرسىنى تولۇق بىلگۈچىدۇر
كَذَٰلِكَ نَقُصُّ عَلَيْكَ مِنْ أَنبَاءِ مَا قَدْ سَبَقَ ۚ وَقَدْ آتَيْنَاكَ مِن لَّدُنَّا ذِكْرًا ﴿٩٩﴾
(ئى مۇھەممەد!) شۇنىڭدەك، ساڭا ئۆتكەنكىلەرنىڭ بەزى خەۋەرلىرىنى بايان قىلىمىز، ساڭا ئۆز تەرىپىمىزدىن قۇرئاننى ئاتا قىلدۇق
مَّنْ أَعْرَضَ عَنْهُ فَإِنَّهُ يَحْمِلُ يَوْمَ الْقِيَامَةِ وِزْرًا ﴿١٠٠﴾
كىمكى قۇرئاندىن يۈز ئۆرۈيدىكەن، ئۇنىڭغا قىيامەت كۈنى چوڭ گۇناھتىن ئېغىر يۈك يۈكلىنىدۇ
خَالِدِينَ فِيهِ ۖ وَسَاءَ لَهُمْ يَوْمَ الْقِيَامَةِ حِمْلًا ﴿١٠١﴾
ئۇ گۇناھى ئۈچۈن بېرىلگەن ئازابنى مەڭگۈ تارتىدۇ) قىيامەت كۈنى ئۇلارغا يۈكلەنگەن بۇ يۈك نېمىدېگەن يامان!
يَوْمَ يُنفَخُ فِي الصُّورِ ۚ وَنَحْشُرُ الْمُجْرِمِينَ يَوْمَئِذٍ زُرْقًا ﴿١٠٢﴾
بۇ سۇر چىلىنىدىغان كۈندۇر، بۇ كۈندە (شەكلى ئۆزگەرگەن) كۆك كۆزلۈك (قارا يۈزلۈك) گۇناھكارلارنى يىغىمىز
يَتَخَافَتُونَ بَيْنَهُمْ إِن لَّبِثْتُمْ إِلَّا عَشْرًا ﴿١٠٣﴾
(قىيامەتنىڭ دەھشىتىنى كۆرۈپ دۇنيادا تۇرغان ۋاقتى قىسقا بىلىنگەنلىكتىن) (ئۇلار) «(دۇنيادا) پەقەت ئون كۈنلا تۇردۇڭلار» دېيىشىپ پىچىرلىشىدۇ
نَّحْنُ أَعْلَمُ بِمَا يَقُولُونَ إِذْ يَقُولُ أَمْثَلُهُمْ طَرِيقَةً إِن لَّبِثْتُمْ إِلَّا يَوْمًا ﴿١٠٤﴾
ئۇلارنىڭ دەيدىغانلىرىنى بىز ئوبدان بىلىمىز، ئەينى زاماندا ئۇلارنىڭ ئەڭ توغرا سۆزلەيدىغىنى: «سىلەر پەقەت بىر كۈن تۇردۇڭلار» دەيدۇ
وَيَسْأَلُونَكَ عَنِ الْجِبَالِ فَقُلْ يَنسِفُهَا رَبِّي نَسْفًا ﴿١٠٥﴾
ئۇلار سەندىن تاغلار توغرۇلۇق (يەنى تاغنىڭ ئاقىۋىتىنىڭ قانداق بولىدىغانلىقى توغرۇلۇق) سورايدۇ، ئېيتقىنكى، «پەرۋەردىگارىم ئۇلارنى كۈكۈم تالقان قىلىپ (ئاندىن شامالنى ئەۋەتىپ ئۇنى سورۇيدۇ)
فَيَذَرُهَا قَاعًا صَفْصَفًا ﴿١٠٦﴾
ئۇلارنى تۈزلەڭلىككە ئايلاندۇرىدۇ
لَّا تَرَىٰ فِيهَا عِوَجًا وَلَا أَمْتًا ﴿١٠٧﴾
ئۇ يەرلەردە نە چۇڭقۇرلۇقنى، نە ئېگىزلىكنى كۆرمەيسەن»
يَوْمَئِذٍ يَتَّبِعُونَ الدَّاعِيَ لَا عِوَجَ لَهُ ۖ وَخَشَعَتِ الْأَصْوَاتُ لِلرَّحْمَٰنِ فَلَا تَسْمَعُ إِلَّا هَمْسًا ﴿١٠٨﴾
بۇ كۈندە (كىشىلەر اﷲ نىڭ مەھشەرگاھقا) دەۋەت قىلغۇچىسىغا ئەگىشىدۇ، قىڭغىر كەتمەيدۇ، مەرھەمەتلىك اﷲ نىڭ ئۇلۇغلۇقى ۋە ھەيۋىتىدىن قورقۇپ، بارلىق ئاۋازلار بېسىقىدۇ، پىچىرلاشقاندىن باشقىنى ئاڭلىمايسەن
يَوْمَئِذٍ لَّا تَنفَعُ الشَّفَاعَةُ إِلَّا مَنْ أَذِنَ لَهُ الرَّحْمَٰنُ وَرَضِيَ لَهُ قَوْلًا ﴿١٠٩﴾
بۇ كۈندە مەرھەمەتلىك اﷲ ئىزىن بەرگەن ۋە سۆزىدىن اﷲ رازى بولغان ئادەمنىڭ شاپائىتىدىن باشقا ھېچقانداق شاپائەت پايدا بەرمەيدۇ
يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلَا يُحِيطُونَ بِهِ عِلْمًا ﴿١١٠﴾
اﷲ ئۇلارنىڭ ئالدىدىكىنى ۋە كەينىدىكىنى بىلىدۇ، ئۇلار بولسا ئۇنى بىلمەيدۇ
وَعَنَتِ الْوُجُوهُ لِلْحَيِّ الْقَيُّومِ ۖ وَقَدْ خَابَ مَنْ حَمَلَ ظُلْمًا ﴿١١١﴾
گۇناھكارلار مەخلۇقاتنى ئىدارە قىلىپ تۇرغۇچى (مەڭگۈ) ھايات اﷲ قا ئېگىلىدۇ، اﷲ قا شېرىك كەلتۈرگەنلەر شەكسىز زىيان تارتىدۇ
وَمَن يَعْمَلْ مِنَ الصَّالِحَاتِ وَهُوَ مُؤْمِنٌ فَلَا يَخَافُ ظُلْمًا وَلَا هَضْمًا ﴿١١٢﴾
مۆمىن تۇرۇپ ياخشى ئەمەللەرنى قىلغانلار يامانلىقلىرىنىڭ ئېشىپ كېتىشىدىن ۋە ياخشىلىقلىرىنىڭ كېمىيىپ كېتىشىدىن قورقمايدۇ
وَكَذَٰلِكَ أَنزَلْنَاهُ قُرْآنًا عَرَبِيًّا وَصَرَّفْنَا فِيهِ مِنَ الْوَعِيدِ لَعَلَّهُمْ يَتَّقُونَ أَوْ يُحْدِثُ لَهُمْ ذِكْرًا ﴿١١٣﴾
ئۇلارنىڭ تەقۋادار بولۇشلىرى ئۈچۈن ياكى ئۇلارنى ۋەز - نەسىھەتكە ئىگە قىلىش ئۈچۈن، ساڭا ئەرەبچە قۇرئاننى نازىل قىلدۇق ۋە ئۇنىڭدا تۈرلۈك ئاگاھلاندۇرۇشلارنى بايان قىلدۇق
فَتَعَالَى اللَّهُ الْمَلِكُ الْحَقُّ ۗ وَلَا تَعْجَلْ بِالْقُرْآنِ مِن قَبْلِ أَن يُقْضَىٰ إِلَيْكَ وَحْيُهُ ۖ وَقُل رَّبِّ زِدْنِي عِلْمًا ﴿١١٤﴾
اﷲ (زاتى، سۈپىتى جەھەتتە پەتەن مەخلۇقاتتىن) ئۈستۈندۇر، ھەق پادىشاھتۇر، (ئى مۇھەممەد!) ساڭا قۇرئاننىڭ ۋەھيىسى تاماملىنىشتىن بۇرۇن ئوقۇشقا ئالدىراپ كەتمىگىن، (يەنى جىبرىئىل ساڭا قۇرئاننى ئوقۇپ بېرىۋاتقاندا، بىللە ئوقۇشقا ئالدىراپ كەتمەي، ئۇ تىلاۋىتىدىن فارىغ بولغۇچە تىڭشاپ تۇرغىن). «ئى پەرۋەردىگارىم! ئىلمىمنى زىيادە قىلغىن!» دېگىن
وَلَقَدْ عَهِدْنَا إِلَىٰ آدَمَ مِن قَبْلُ فَنَسِيَ وَلَمْ نَجِدْ لَهُ عَزْمًا ﴿١١٥﴾
بىز ئىلگىرى ئادەمگە (دەرەخنىڭ مېۋىسىنى يېمەسلىكنى) تەۋسىيە قىلدۇق، ئۇ (بۇ تەۋسىيەنى) يادىدىن چىقاردى، ئۇنىڭدا ھېچقانداق چىڭ ئىرادە كۆرمىدۇق
وَإِذْ قُلْنَا لِلْمَلَائِكَةِ اسْجُدُوا لِآدَمَ فَسَجَدُوا إِلَّا إِبْلِيسَ أَبَىٰ ﴿١١٦﴾
ئۆز ۋاقتىدا بىز خۇدانىڭ ئەلچى پەرىشتىلىرىگە: «ئادەمگە (يەنى ئادەم ئەلەيھىسسالامغا) سەجدە قىلىڭلار» دېدۇق، ئىبلىستىن باشقىسى سەجدە قىلدى، ئۇ (سەجدە قىلىشتىن) باش تارتتى
فَقُلْنَا يَا آدَمُ إِنَّ هَٰذَا عَدُوٌّ لَّكَ وَلِزَوْجِكَ فَلَا يُخْرِجَنَّكُمَا مِنَ الْجَنَّةِ فَتَشْقَىٰ ﴿١١٧﴾
بىز دېدۇق: «ئى ئادەم! بۇ ئىشەنچلىك ھالدا سېنىڭ دۈشمىنىڭدۇر، خوتۇنۇڭنىڭ دۈشمىنىدۇر ئۇ سىلەر ئىككىڭلارنىڭ مەڭگۈلۈك باغلارتىن چىقىپ كېتىشىڭلارغا سەۋەبچى بولۇپ قالمىسۇن، (ئۇ چاغدا) سىلەر جاپا مۇشەققەتتە قالىسىلەر
إِنَّ لَكَ أَلَّا تَجُوعَ فِيهَا وَلَا تَعْرَىٰ ﴿١١٨﴾
مەڭگۈلۈك باغلارتە سەن ئىشەنچلىك ھالدا ئاچ قالمايسەن ۋە يالىڭاچ قالمايسەن
وَأَنَّكَ لَا تَظْمَأُ فِيهَا وَلَا تَضْحَىٰ ﴿١١٩﴾
ھەمدە ئۇنىڭدا ئىشەنچلىك ھالدا ئۇسساپ كەتمەيسەن، ئىسسىپ كەتمەيسەن»
فَوَسْوَسَ إِلَيْهِ الشَّيْطَانُ قَالَ يَا آدَمُ هَلْ أَدُلُّكَ عَلَىٰ شَجَرَةِ الْخُلْدِ وَمُلْكٍ لَّا يَبْلَىٰ ﴿١٢٠﴾
شەيتان (ئازدۇرغۇچى) ئۇنىڭغا ۋەسۋەسە قىلىپ: «ئى ئادەم! ساڭا (مەن يېگەن ئادەم) ئۆلمەيدىغان دەرەخنى ۋە زاۋال تاپماس پادىشاھلىقنى كۆرسىتىپ قويايمۇ؟» دېدى
فَأَكَلَا مِنْهَا فَبَدَتْ لَهُمَا سَوْآتُهُمَا وَطَفِقَا يَخْصِفَانِ عَلَيْهِمَا مِن وَرَقِ الْجَنَّةِ ۚ وَعَصَىٰ آدَمُ رَبَّهُ فَغَوَىٰ ﴿١٢١﴾
ئۇلار ئۇ دەرەخنىڭ مېۋىسىنى يېدى، شۇنىڭ بىلەن ئۇلارنىڭ ئەۋرەتلىرى ئېچىلىپ قالدى، ئۇلار مەڭگۈلۈك باغلار (دەرەخلىرىنىڭ) يوپۇرماقلىرى بىلەن سەترى ئەۋرەت قىلىشقا كىرىشتى، ئادەم پەرۋەردىگارىنىڭ (ئەمرىگە) خىلاپلىق قىلدى، شۇنىڭ بىلەن ئۇ ئازدى
ثُمَّ اجْتَبَاهُ رَبُّهُ فَتَابَ عَلَيْهِ وَهَدَىٰ ﴿١٢٢﴾
ئاندىن پەرۋەردىگارى ئۇنى تاللىدى، ئۇنىڭ تەۋبىسىنى قوبۇل قىلدى ۋە ئۇنى ھىدايەت قىلدى
قَالَ اهْبِطَا مِنْهَا جَمِيعًا ۖ بَعْضُكُمْ لِبَعْضٍ عَدُوٌّ ۖ فَإِمَّا يَأْتِيَنَّكُم مِّنِّي هُدًى فَمَنِ اتَّبَعَ هُدَايَ فَلَا يَضِلُّ وَلَا يَشْقَىٰ ﴿١٢٣﴾
اﷲ (ئادەم بىلەن ھەۋۋاغا) ئېيتتى: «سىلەر ھەممىڭلار مەڭگۈلۈك باغلارتىن بەزىڭلار بەزىڭلارغا دۈشمەن بولغان ھالدا چۈشۈڭلار، ئەگەر سىلەرگە مېنىڭ تەرىپىمدىن ھىدايەت كەلسە، كىمكى مېنىڭ ھىدايىتىمگە ئەگەشسە، ئۇ ئازمايدۇ ۋە ئاخىرەتتە شەقىي بولمايدۇ
وَمَنْ أَعْرَضَ عَن ذِكْرِي فَإِنَّ لَهُ مَعِيشَةً ضَنكًا وَنَحْشُرُهُ يَوْمَ الْقِيَامَةِ أَعْمَىٰ ﴿١٢٤﴾
كىمكى مېنىڭ زىكرىمدىن يۈز ئۆرۈيدىكەن، ئۇنىڭ ھاياتى تار (يەنى خاتىرجەمسىز) بولىدۇ، قىيامەت كۈنى ئۇنى بىز كور قوپۇرىمىز»
قَالَ رَبِّ لِمَ حَشَرْتَنِي أَعْمَىٰ وَقَدْ كُنتُ بَصِيرًا ﴿١٢٥﴾
ئۇ: «ئى پەرۋەردىگارىم! نېمىشقا مېنى كور قوپۇردۇڭ، ۋاھالەنكى، (دۇنيادىكى چاغدا) مېنىڭ كۆزۈم كۆرەتتىغۇ؟» دەيدۇ
قَالَ كَذَٰلِكَ أَتَتْكَ آيَاتُنَا فَنَسِيتَهَا ۖ وَكَذَٰلِكَ الْيَوْمَ تُنسَىٰ ﴿١٢٦﴾
اﷲ ئېيتىدۇ: «شۇنىڭدەك ساڭا (روشەن) ئايەتلىرىمىز كەلدى، سەن ئۇلارنى تەرك ئەتتىڭ، سەن بۈگۈن شۇنىڭغا ئوخشاش تەرك ئېتىلىسەن»
وَكَذَٰلِكَ نَجْزِي مَنْ أَسْرَفَ وَلَمْ يُؤْمِن بِآيَاتِ رَبِّهِ ۚ وَلَعَذَابُ الْآخِرَةِ أَشَدُّ وَأَبْقَىٰ ﴿١٢٧﴾
ھەددىدىن ئاشقان ۋە پەرۋەردىگارىنى ئايەتلىرىگە ئىمان ئېيتمىغانلارغا شۇنىڭغا ئوخشاش جازا بېرىمىز، ئاخىرەتنىڭ ئازابى تېخىمۇ قاتتىقتۇر، تېخىمۇ باقىيدۇر
أَفَلَمْ يَهْدِ لَهُمْ كَمْ أَهْلَكْنَا قَبْلَهُم مِّنَ الْقُرُونِ يَمْشُونَ فِي مَسَاكِنِهِمْ ۗ إِنَّ فِي ذَٰلِكَ لَآيَاتٍ لِّأُولِي النُّهَىٰ ﴿١٢٨﴾
ئۇلاردىن ئىلگىرى ئۆتكەن (خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلغان) نۇرغۇن ئۈممەتلەرنى ھالاك قىلغانلىقىمىزنى ئۇلار ئۇقمامدۇ؟ ئۇلار شۇ ئۈممەتلەرنىڭ ماكانلىرىدا مېڭىپ يۈرىدۇ، ئەقىل ئىگىلىرى ئۈچۈن ئۇنىڭدا ئىشەنچلىك ھالدا نۇرغۇن ئىبرەتلەر بار
وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ لَكَانَ لِزَامًا وَأَجَلٌ مُّسَمًّى ﴿١٢٩﴾
ئەگەر پەرۋەردىگارىڭنىڭ (ئازابنى ئۇلاردىن كېچىكتۈرۈشكە) قىلىۋەتكەن ھۆكمى ۋە ئالدىنئالا بەلگىلەپ قويغان ۋاقتى بولمىسا ئىدى، ئۇ چاغدا (ئۇلارنىڭ ھالاك بولۇشى) چوقۇم ئىدى
فَاصْبِرْ عَلَىٰ مَا يَقُولُونَ وَسَبِّحْ بِحَمْدِ رَبِّكَ قَبْلَ طُلُوعِ الشَّمْسِ وَقَبْلَ غُرُوبِهَا ۖ وَمِنْ آنَاءِ اللَّيْلِ فَسَبِّحْ وَأَطْرَافَ النَّهَارِ لَعَلَّكَ تَرْضَىٰ ﴿١٣٠﴾
ئۇلارنىڭ سۆزلىرىگە سەۋرى قىلغىن، مەمنۇن بولۇشۇڭ ئۈچۈن، كۈن چىقىشتىن ئىلگىرى، كۈن پېتىشتىن بۇرۇن، كېچە ۋاقىتلىرىدا ۋە كۈندۈزنىڭ دەسلەپكى ۋە ئاخىرقى ۋاقىتلىرىدا پەرۋەردىگارىڭغا تەسبىھ ئېيتقىن (يەنى ناماز ئوقۇغىن) ۋە ھەمدە ئېيتقىن
وَلَا تَمُدَّنَّ عَيْنَيْكَ إِلَىٰ مَا مَتَّعْنَا بِهِ أَزْوَاجًا مِّنْهُمْ زَهْرَةَ الْحَيَاةِ الدُّنْيَا لِنَفْتِنَهُمْ فِيهِ ۚ وَرِزْقُ رَبِّكَ خَيْرٌ وَأَبْقَىٰ ﴿١٣١﴾
بىز (كۇففارلاردىن) تۈرلۈك جامائەنى بەھرىمەن قىلغان دۇنيانىڭ نېمەتلىرى ۋە زىبۇزىننەتلىرىگە كۆز سالمىغىن، بۇنىڭ بىلەن ئۇلارنى سىنايمىز، پەرۋەردىگارىڭنىڭ رىزقى (يەنى ساۋابى بۇ پانىي نېمەتتىن) ياخشىدۇر ۋە ئەڭ باقىيدۇر
وَأْمُرْ أَهْلَكَ بِالصَّلَاةِ وَاصْطَبِرْ عَلَيْهَا ۖ لَا نَسْأَلُكَ رِزْقًا ۖ نَّحْنُ نَرْزُقُكَ ۗ وَالْعَاقِبَةُ لِلتَّقْوَىٰ ﴿١٣٢﴾
(ئى مۇھەممەد!) ئائىلەڭدىكىلەرنى (ۋە ئۈممىتىڭنى) نامازغا بۇيرۇغىن، ئۆزۈڭمۇ ئۇنى ئادا قىلىشقا چىداملىق بولغىن، سەندىن بىز رىزىق تەلەپ قىلمايمىز. ساڭا بىز رىزىق بېرىمىز، ياخشى ئاقىۋەت پەقەت تەقۋادارلارغا خاستۇر
وَقَالُوا لَوْلَا يَأْتِينَا بِآيَةٍ مِّن رَّبِّهِ ۚ أَوَلَمْ تَأْتِهِم بَيِّنَةُ مَا فِي الصُّحُفِ الْأُولَىٰ ﴿١٣٣﴾
ئۇلار (يەنى مۇشرىكلار): «(مۇھەممەد) پەرۋەردىگارى تەرىپىدىن بىزگە بىرەر مۆجىزە كەلتۈرسىچۇ!» دەيدۇ، ئۇلارغا بۇرۇنقى ئۈممەتلەرنىڭ خەۋەرلىرىنى ئۆز ئىچىگە ئالغان قۇرئان كەلمىدىمۇ؟
وَلَوْ أَنَّا أَهْلَكْنَاهُم بِعَذَابٍ مِّن قَبْلِهِ لَقَالُوا رَبَّنَا لَوْلَا أَرْسَلْتَ إِلَيْنَا رَسُولًا فَنَتَّبِعَ آيَاتِكَ مِن قَبْلِ أَن نَّذِلَّ وَنَخْزَىٰ ﴿١٣٤﴾
ئەگەر بىز ئۇلارنى ئۇنىڭدىن (يەنى مۇھەممەد ئەلەيھىسسالامنى ئەۋەتىشتىن) ئىلگىرى ھالاك قىلغان بولساق، ئۇلار چوقۇم: «نېمىشقا بىزگە بىرەر خۇدانىڭ ئەلچىسى ئەۋەتمىدىڭ؟ بىز ۋە رەسۋا بولۇشتىن ئىلگىرى سېنىڭ ئايەتلىرىڭگە ئەگەشتۇق» دەيتتى
قُلْ كُلٌّ مُّتَرَبِّصٌ فَتَرَبَّصُوا ۖ فَسَتَعْلَمُونَ مَنْ أَصْحَابُ الصِّرَاطِ السَّوِيِّ وَمَنِ اهْتَدَىٰ ﴿١٣٥﴾
ئېيتقىنكى، ھەممە (سىلەرنىڭ ۋە بىزنىڭ ئىشلىرىمىزنى ئاقىۋىتىنى) كۈتمەكتە. سىلەرمۇ كۈتۈڭلار، كىملەرنىڭ توغرا يول ئىگىلىرى ئىكەنلىكىنى ۋە كىملەرنىڭ ھىدايەت تاپقانلىقىنى (ئۇزاققا قالماي) بىلىسىلەر»
