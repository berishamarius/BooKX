بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
حم ﴿١﴾
حامىم،
عسق ﴿٢﴾
عين، سىين، قاف
كَذَٰلِكَ يُوحِي إِلَيْكَ وَإِلَى الَّذِينَ مِن قَبْلِكَ اللَّهُ الْعَزِيزُ الْحَكِيمُ ﴿٣﴾
غالىب، ھېكمەت بىلەن ئىش قىلغۇچى اﷲ ساڭا ۋە سەندىن ئىلگىرىكى (خۇدانىڭ ئەلچىسى) لەرگە مۇشۇنداق ۋەھىي قىلىدۇ
لَهُ مَا فِي السَّمَاوَاتِ وَمَا فِي الْأَرْضِ ۖ وَهُوَ الْعَلِيُّ الْعَظِيمُ ﴿٤﴾
ئاسمانلاردىكى شەيئىلەر ۋە زېمىندىكى شەيئىلەر اﷲ نىڭدۇر (يەنى اﷲ نىڭ مۈلكىدۇر، اﷲ نىڭ تەسەررۇپى ئاستىدىدۇر)، اﷲ ئۈستۈندۇر، كاتتىدۇر
تَكَادُ السَّمَاوَاتُ يَتَفَطَّرْنَ مِن فَوْقِهِنَّ ۚ وَالْمَلَائِكَةُ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ وَيَسْتَغْفِرُونَ لِمَن فِي الْأَرْضِ ۗ أَلَا إِنَّ اللَّهَ هُوَ الْغَفُورُ الرَّحِيمُ ﴿٥﴾
(اﷲ نىڭ كاتتىلىقىدىن) ئاسمانلار يۇقىرى تەرىپىدىن يېرىلىپ كېتىشكە تاس قالىدۇ، خۇدانىڭ ئەلچى پەرىشتىلىرى پەرۋەردىگارىغا تەسبىھ ئېيتىدۇ، ھەمدى ئېيتىدۇ، يەر يۈزىدىكىلەرگە مەغپىرەت تىلەيدۇ. بىلىڭلاركى، اﷲ ئىشەنچلىك ھالدا ناھايىتى مەغپىرەت قىلغۇچىدۇر، ناھايىتى مېھرىباندۇر
وَالَّذِينَ اتَّخَذُوا مِن دُونِهِ أَوْلِيَاءَ اللَّهُ حَفِيظٌ عَلَيْهِمْ وَمَا أَنتَ عَلَيْهِم بِوَكِيلٍ ﴿٦﴾
اﷲ نى قويۇپ (بۇتلارنى) ئىگە قىلىۋالغانلارنىڭ (ئەمەللىرىنى) اﷲ كۈزىتىپ تۇرغۇچىدۇر، سەن ئۇلارغا ھامىي ئەمەسسەن
وَكَذَٰلِكَ أَوْحَيْنَا إِلَيْكَ قُرْآنًا عَرَبِيًّا لِّتُنذِرَ أُمَّ الْقُرَىٰ وَمَنْ حَوْلَهَا وَتُنذِرَ يَوْمَ الْجَمْعِ لَا رَيْبَ فِيهِ ۚ فَرِيقٌ فِي الْجَنَّةِ وَفَرِيقٌ فِي السَّعِيرِ ﴿٧﴾
مەككە ئاھالىسىنى ۋە ئۇنىڭ ئەتراپىدىكىلەرنى ئاگاھلاندۇرۇشۇڭ ئۈچۈن، (كىشىلەرنى) قىيامەت كۈنىدىن ئاگاھلاندۇرۇشۇڭ ئۈچۈن، ساڭا بىز مۇشۇنداق ئەرەبچە قۇرئاننى نازىل قىلدۇق، ئۇ كۈندە ھېچ شەك يوقتۇر، (ئۇ كۈندە) بىر پىرقە (كىشىلەر) مەڭگۈلۈك باغلارتە ۋە بىر پىرقە كىشىلەر مەڭگۈلۈك ئوتدە بولىدۇ
وَلَوْ شَاءَ اللَّهُ لَجَعَلَهُمْ أُمَّةً وَاحِدَةً وَلَٰكِن يُدْخِلُ مَن يَشَاءُ فِي رَحْمَتِهِ ۚ وَالظَّالِمُونَ مَا لَهُم مِّن وَلِيٍّ وَلَا نَصِيرٍ ﴿٨﴾
ئەگەر اﷲ خالىسا، ئۇلارنى (يەنى پۈتۈن ئىنسانلارنى) ئەلۋەتتە بىر ئۈممەت (يەنى بىر دىندا) قىلاتتى، لېكىن اﷲ خالىغان ئادەمنى رەھمىتىگە داخىل قىلىدۇ، زالىملارغا ھېچقانداق ئىگە، ھېچقانداق مەدەتكار بولمايدۇ
أَمِ اتَّخَذُوا مِن دُونِهِ أَوْلِيَاءَ ۖ فَاللَّهُ هُوَ الْوَلِيُّ وَهُوَ يُحْيِي الْمَوْتَىٰ وَهُوَ عَلَىٰ كُلِّ شَيْءٍ قَدِيرٌ ﴿٩﴾
بەلكى ئۇلار اﷲ تىن باشقىلارنى ئىگە قىلىۋالدى، پەقەت اﷲ ئىگىدۇر، اﷲ ئۆلۈكلەرنى تىرىلدۈرىدۇ، اﷲ ھەر نەرسىگە قادىر
وَمَا اخْتَلَفْتُمْ فِيهِ مِن شَيْءٍ فَحُكْمُهُ إِلَى اللَّهِ ۚ ذَٰلِكُمُ اللَّهُ رَبِّي عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ أُنِيبُ ﴿١٠﴾
سىلەر (دىن ئۈستىدە ۋە ئۇنىڭدىن باشقا) ھەر قانداق نەرسە ئۈستىدە ئىختىلاپ قىلىشماڭلار، ئۇنىڭغا اﷲ ھۆكۈم چىقىرىدۇ. (ئى مۇھەممەد! ئۇلارغا ئېيتقىنكى) ئەنە شۇ اﷲ مېنىڭ پەرۋەردىگارىمدۇر، ئۇنىڭغا تەۋەككۈل قىلدىم، (ھەممە ئىشتا) اﷲ قا مۇراجىئەت قىلىمەن
فَاطِرُ السَّمَاوَاتِ وَالْأَرْضِ ۚ جَعَلَ لَكُم مِّنْ أَنفُسِكُمْ أَزْوَاجًا وَمِنَ الْأَنْعَامِ أَزْوَاجًا ۖ يَذْرَؤُكُمْ فِيهِ ۚ لَيْسَ كَمِثْلِهِ شَيْءٌ ۖ وَهُوَ السَّمِيعُ الْبَصِيرُ ﴿١١﴾
اﷲ ئاسمانلارنى ۋە زېمىننى ياراتقۇچىدۇر، اﷲ سىلەرگە ئۆز جىنسىڭلاردىن جۈپتىلەرنى (يەنى ئاياللارنى) ياراتتى. ھايۋانلارغىمۇ ئۆز تىپىدىن جۈپتىلەرنى ياراتتى. بۇ ئارقىلىق اﷲ سىلەرنى كۆپەيتىدۇ، ھېچ شەيئى اﷲ قا ئوخشاش ئەمەستۇر، اﷲ ھەممىنى ئاڭلاپ تۇرغۇچىدۇر، ھەممىنى كۆرۈپ تۇرغۇچىدۇر
لَهُ مَقَالِيدُ السَّمَاوَاتِ وَالْأَرْضِ ۖ يَبْسُطُ الرِّزْقَ لِمَن يَشَاءُ وَيَقْدِرُ ۚ إِنَّهُ بِكُلِّ شَيْءٍ عَلِيمٌ ﴿١٢﴾
ئاسمانلارنىڭ ۋە زېمىننىڭ (خەزىنىلىرىنىڭ) ئاچقۇچلىرى اﷲ نىڭ ئىلكىدىدۇر، اﷲ خالىغان كىشىنىڭ رىزقىنى (سىناش يۈزىسىدىن) كەڭ قىلىدۇ، خالىغان كىشىنىڭ رىزقىنى (بالاغا مۇپتىلا قىلىپ سىناش يۈزىسىدىن) تار قىلىدۇ، اﷲ ئىشەنچلىك ھالدا ھەممە نەرسىنى بىلىپ تۇرغۇچىدۇر
شَرَعَ لَكُم مِّنَ الدِّينِ مَا وَصَّىٰ بِهِ نُوحًا وَالَّذِي أَوْحَيْنَا إِلَيْكَ وَمَا وَصَّيْنَا بِهِ إِبْرَاهِيمَ وَمُوسَىٰ وَعِيسَىٰ ۖ أَنْ أَقِيمُوا الدِّينَ وَلَا تَتَفَرَّقُوا فِيهِ ۚ كَبُرَ عَلَى الْمُشْرِكِينَ مَا تَدْعُوهُمْ إِلَيْهِ ۚ اللَّهُ يَجْتَبِي إِلَيْهِ مَن يَشَاءُ وَيَهْدِي إِلَيْهِ مَن يُنِيبُ ﴿١٣﴾
اﷲ سىلەرگە دىندىن نۇھقا تەۋسىيە قىلغان نەرسىنى، ساڭا (يەنى مۇھەممەد ئەلەيھىسسالامغا) بىز ۋەھىي قىلغان نەرسىنى، ئىبراھىمغا، مۇساغا ۋە ئىساغا بىز تەۋسىيە قىلغان نەرسىنى بايان قىلدى. سىلەر دىننى بەرپا قىلىڭلار، دىندا تەپرىقىچىلىك قىلماڭلار، مۇشرىكلارغا سەن ئۇلارنى دەۋەت قىلغان نەرسە (يەنى تەۋھىد) ئېغىر كەلدى، اﷲ تەۋھىدكە خالىغان ئادەمنى تاللايدۇ، اﷲ نىڭ (تائىتىگە) قايتىدىغان ئادەمنى تەۋھىدكە باشلايدۇ
وَمَا تَفَرَّقُوا إِلَّا مِن بَعْدِ مَا جَاءَهُمُ الْعِلْمُ بَغْيًا بَيْنَهُمْ ۚ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ إِلَىٰ أَجَلٍ مُّسَمًّى لَّقُضِيَ بَيْنَهُمْ ۚ وَإِنَّ الَّذِينَ أُورِثُوا الْكِتَابَ مِن بَعْدِهِمْ لَفِي شَكٍّ مِّنْهُ مُرِيبٍ ﴿١٤﴾
(يەھۇدىي، ناسارا قاتارلىقلار) ئۇلارغا ئىلىم كەلگەندىن كېيىن (يەنى ئۇلار رەسۇلۇللاھنىڭ ھەق خۇدانىڭ ئەلچىسىلىكىنى بىلگەندىن كېيىن)، ئاندىن ئۆزئارا ھەسەت قىلىشىپ پىرقىلەرگە بۆلۈندى. ئەگەر پەرۋەردىگارىڭنىڭ (ئۇلارنى جازالاشنى) قىيامەتكىچە تەخىر قىلىشتىن ئىبارەت سۆزى بولمىسا ئىدى، ئۇلارنىڭ ئارىسىدا ئەلۋەتتە ھۆكۈم چىقىرىلىپ بولغان بولاتتى (يەنى ئۇلار تېز ھالاك قىلىنغان بولاتتى). ئۇلاردىن كېيىن مۇقەددەس كىتاب-ۋەھىيقا ۋارىسلىق قىلغانلار (يەنى رەسۇلۇللاھ بىلەن زامانداش بولغان ئەھلى مۇقەددەس كىتاب-ۋەھىي) (مۇقەددەس كىتاب-ۋەھىي ھەققىدە) قاتتىق دەرگۇماندىدۇر
فَلِذَٰلِكَ فَادْعُ ۖ وَاسْتَقِمْ كَمَا أُمِرْتَ ۖ وَلَا تَتَّبِعْ أَهْوَاءَهُمْ ۖ وَقُلْ آمَنتُ بِمَا أَنزَلَ اللَّهُ مِن كِتَابٍ ۖ وَأُمِرْتُ لِأَعْدِلَ بَيْنَكُمُ ۖ اللَّهُ رَبُّنَا وَرَبُّكُمْ ۖ لَنَا أَعْمَالُنَا وَلَكُمْ أَعْمَالُكُمْ ۖ لَا حُجَّةَ بَيْنَنَا وَبَيْنَكُمُ ۖ اللَّهُ يَجْمَعُ بَيْنَنَا ۖ وَإِلَيْهِ الْمَصِيرُ ﴿١٥﴾
شۇنىڭ ئۈچۈن (يەنى دىندا ئىختىلاپ قىلىشقانلىقى ئۈچۈن) (ئى مۇھەممەد! بارلىق ئىنسانلارنى توغرا دىن ئاساسىدا ئىتتىپاق بولۇشقا) چاقىرغىن، بۇيرۇلغىنىڭ بويىچە توغرا يولدا چىڭ تۇرغىن، ئۇلار (يەنى مۇشرىكلار) نىڭ نەپسى خاھىشلىرىغا ئەگەشمىگىن ۋە ئېيتقىن: «اﷲ نازىل قىلغان (بارلىق) مۇقەددەس كىتاب-ۋەھىيلارغا ئىمان ئېيتتىم، ئاراڭلاردا ئادىل بولۇشقا بۇيرۇلدۇم، اﷲ بىزنىڭ پەرۋەردىگارىمىزدۇر، سىلەرنىڭمۇ پەرۋەردىگارىڭلاردۇر، بىزنىڭ ئەمەللىرىمىز ئۆزىمىز ئۈچۈندۇر، سىلەرنىڭ ئەمەللىرىڭلارمۇ ئۆزلىرىڭلار ئۈچۈندۇر، سىلەر بىلەن بىزنىڭ ئارىمىزدا جېدەللىشىشكە (ئورۇن) يوق (چۈنكى ھەقىقەت ئاشكارىدۇر)، اﷲ (قىيامەت كۈنى) بىزنى (ھۆكۈم چىقىرىش ئۈچۈن) يىغىدۇ، ئاخىر قايتىدىغان جاي اﷲ نىڭ دەرگاھىدۇر»
وَالَّذِينَ يُحَاجُّونَ فِي اللَّهِ مِن بَعْدِ مَا اسْتُجِيبَ لَهُ حُجَّتُهُمْ دَاحِضَةٌ عِندَ رَبِّهِمْ وَعَلَيْهِمْ غَضَبٌ وَلَهُمْ عَذَابٌ شَدِيدٌ ﴿١٦﴾
كىشىلەر اﷲ نىڭ دىنىنى قوبۇل قىلغاندىن كېيىن، (كىشىلەرنى ئىماندىن توسۇش ئۈچۈن) اﷲ نىڭ دىنىدا خۇسۇمەت قىلىشقانلارنىڭ دەلىللىرى اﷲ نىڭ دەرگاھىدا باتىلدۇر، ئۇلار (دۇنيادا) غەزەپكە ئۇچرايدۇ، (ئاخىرەتتە) قاتتىق ئازابقا دۇچار بولىدۇ
اللَّهُ الَّذِي أَنزَلَ الْكِتَابَ بِالْحَقِّ وَالْمِيزَانَ ۗ وَمَا يُدْرِيكَ لَعَلَّ السَّاعَةَ قَرِيبٌ ﴿١٧﴾
اﷲ مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) نى ھەق بىلەن نازىل قىلدى ۋە ئادالەتنى نازىل قىلدى. سەن نېمە بىلىسەن؟ قىيامەت يېقىن بولۇشى مۇمكىن
يَسْتَعْجِلُ بِهَا الَّذِينَ لَا يُؤْمِنُونَ بِهَا ۖ وَالَّذِينَ آمَنُوا مُشْفِقُونَ مِنْهَا وَيَعْلَمُونَ أَنَّهَا الْحَقُّ ۗ أَلَا إِنَّ الَّذِينَ يُمَارُونَ فِي السَّاعَةِ لَفِي ضَلَالٍ بَعِيدٍ ﴿١٨﴾
قىيامەتكە ئىشەنمەيدىغانلار ئۇنىڭ بولۇشىنى ئالدىرىتىدۇ، ئۇنىڭغا ئىشىنىدىغانلار بولسا قىيامەتتىن قورقىدۇ، قىيامەتنى ھەق دەپ بىلىدۇ، بىلىڭلاركى، قىيامەت توغرۇلۇق مۇجادىلە قىلىدىغانلار ئەلۋەتتە چوڭقۇر گۇمراھلىقتىدۇر
اللَّهُ لَطِيفٌ بِعِبَادِهِ يَرْزُقُ مَن يَشَاءُ ۖ وَهُوَ الْقَوِيُّ الْعَزِيزُ ﴿١٩﴾
اﷲ بەندىلىرىگە كۆيۈنگۈچىدۇر، ئۇ خالىغان ئادەمگە (كەڭ) رىزىق بېرىدۇ، اﷲ كۈچلۈكتۇر، غالىبتۇر
مَن كَانَ يُرِيدُ حَرْثَ الْآخِرَةِ نَزِدْ لَهُ فِي حَرْثِهِ ۖ وَمَن كَانَ يُرِيدُ حَرْثَ الدُّنْيَا نُؤْتِهِ مِنْهَا وَمَا لَهُ فِي الْآخِرَةِ مِن نَّصِيبٍ ﴿٢٠﴾
كىمكى (ئۆزىنىڭ ئەمەلى بىلەن) ئاخىرەتنىڭ ساۋابىنى كۆزلىسە، ئۇنىڭ ساۋابىنى زىيادە بېرىمىز، كىمكى (ئەمەلى بىلەن) دۇنيانىڭ مەنپەئىتىنى كۆزلىسە، (ئۇنىڭ تىلىگەن) مەنپەئىتىنىڭ بەزىسىنى بېرىمىز، ئۇنىڭغا ئاخىرەتتە (ساۋابتىن) ھېچ نېسىۋە يوق
أَمْ لَهُمْ شُرَكَاءُ شَرَعُوا لَهُم مِّنَ الدِّينِ مَا لَمْ يَأْذَن بِهِ اللَّهُ ۚ وَلَوْلَا كَلِمَةُ الْفَصْلِ لَقُضِيَ بَيْنَهُمْ ۗ وَإِنَّ الظَّالِمِينَ لَهُمْ عَذَابٌ أَلِيمٌ ﴿٢١﴾
ئۇلارنىڭ اﷲ رۇخسەت قىلمىغان نەرسىلەرنى دىن قىلىپ بېكەتكەن مەبۇدلىرى بارمۇ؟ (اﷲ نىڭ ئۇلارغا بېرىلىدىغان ئازابنى قىيامەتكىچە تەخىر قىلىش توغرىسىدىكى) ھۆكمى بولمىسا ئىدى، ئۇلارنىڭ ئارىسىدا ئەلۋەتتە (دۇنيادىلا) ھۆكۈم چىقىرىلغان بولاتتى، (ئۆزلىرىگە) زۇلۇم قىلغۇچىلار (يەنى ھەقىقەتنى رەت قىلغۇچىلار) قاتتىق ئازابقا دۇچار بولىدۇ
تَرَى الظَّالِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا وَهُوَ وَاقِعٌ بِهِمْ ۗ وَالَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ فِي رَوْضَاتِ الْجَنَّاتِ ۖ لَهُم مَّا يَشَاءُونَ عِندَ رَبِّهِمْ ۚ ذَٰلِكَ هُوَ الْفَضْلُ الْكَبِيرُ ﴿٢٢﴾
ھەقىقەتنى رەت قىلغۇچىلارنىڭ (قىيامەت كۈنى دۇنيادا) قىلغان گۇناھلىرىنىڭ جازاسىدىن قورقۇپ تۇرغانلىقىنى كۆرىسەن، (ئۇلار مەيلى قورقسۇن، قورقمىسۇن) جازا ئۇلارغا (چوقۇم) نازىل بولغۇسىدۇر، ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلار مەڭگۈلۈك باغلارلەرنىڭ باغچىلىرىدا بولىدۇ، ئۇلار پەرۋەردىگارىنىڭ ھۇزۇرىدا خالىغان نەرسىلەردىن بەھرىمەن بولىدۇ، بۇ كاتتا ئىنئامدۇر
ذَٰلِكَ الَّذِي يُبَشِّرُ اللَّهُ عِبَادَهُ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ ۗ قُل لَّا أَسْأَلُكُمْ عَلَيْهِ أَجْرًا إِلَّا الْمَوَدَّةَ فِي الْقُرْبَىٰ ۗ وَمَن يَقْتَرِفْ حَسَنَةً نَّزِدْ لَهُ فِيهَا حُسْنًا ۚ إِنَّ اللَّهَ غَفُورٌ شَكُورٌ ﴿٢٣﴾
اﷲ ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغان بەندىلىرىگە ئەنە شۇ ئىنئام بىلەن خۇش - خەۋەر بېرىدۇ. «(خۇدانىڭ ئەلچىسىلىكنى) تەبلىغ قىلغانلىقىمغا سىلەردىن ھەق تەلەپ قىلمايمەن، پەقەت تۇغقانچىلىق ھەققى ئۈچۈن مېنى دوست تۇتۇشۇڭلارنى (يەنى پەرۋەردىگارىمنىڭ دەۋىتىنى يەتكۈزۈشۈمگە چېقىلماسلىقىڭلارنى) تىلەيمەن» دېگىن، كىمكى بىرەر ياخشى ئىش قىلسا، ئۇنىڭغا بىز ساۋابنى زىيادە بېرىمىز، اﷲ ئىشەنچلىك ھالدا ناھايىتى مەغپىرەت قىلغۇچىدۇر، ئاز ياخشىلىققا كۆپ ساۋاب بەرگۈچىدۇر
أَمْ يَقُولُونَ افْتَرَىٰ عَلَى اللَّهِ كَذِبًا ۖ فَإِن يَشَإِ اللَّهُ يَخْتِمْ عَلَىٰ قَلْبِكَ ۗ وَيَمْحُ اللَّهُ الْبَاطِلَ وَيُحِقُّ الْحَقَّ بِكَلِمَاتِهِ ۚ إِنَّهُ عَلِيمٌ بِذَاتِ الصُّدُورِ ﴿٢٤﴾
ئۇلار (يەنى قۇرەيش ھەقىقەتنى رەت قىلغۇچىلىرى): «ئۇ (يەنى مۇھەممەد) اﷲ قا يالغاننى توقۇدى» دەمدۇ؟ ئەگەر اﷲ خالىسا (يەنى ئۇلار گۇمان قىلغاندەك، اﷲ قا يالغاننى توقۇغان بولساڭ) سېنىڭ قەلبىڭنى پېچەتلىۋەتكەن بولاتتى، اﷲ باتىلنى يوق قىلىدۇ، (پەيغەمبىرىگە نازىل قىلغان) سۆزلىرى ئارقىلىق ھەقنى ھەق قىلىدۇ، اﷲ ئىشەنچلىك ھالدا دىللاردىكى سىرلارنى بىلىپ تۇرغۇچىدۇر
وَهُوَ الَّذِي يَقْبَلُ التَّوْبَةَ عَنْ عِبَادِهِ وَيَعْفُو عَنِ السَّيِّئَاتِ وَيَعْلَمُ مَا تَفْعَلُونَ ﴿٢٥﴾
اﷲ بەندىلىرىدىن تەۋبىنى قوبۇل قىلىدۇ، يامان ئىشلارنى ئەپۇ قىلىدۇ، قىلىۋاتقان ئىشىڭلارنى بىلىپ تۇرىدۇ
وَيَسْتَجِيبُ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ وَيَزِيدُهُم مِّن فَضْلِهِ ۚ وَالْكَافِرُونَ لَهُمْ عَذَابٌ شَدِيدٌ ﴿٢٦﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنىڭ دۇئاسىنى اﷲ ئىجابەت قىلىدۇ، (كامالى) پەزلىدىن ئۇلارغا ساۋابنى زىيادە بېرىدۇ، ھەقىقەتنى رەت قىلغۇچىلار قاتتىق ئازابقا دۇچار بولىدۇ
وَلَوْ بَسَطَ اللَّهُ الرِّزْقَ لِعِبَادِهِ لَبَغَوْا فِي الْأَرْضِ وَلَٰكِن يُنَزِّلُ بِقَدَرٍ مَّا يَشَاءُ ۚ إِنَّهُ بِعِبَادِهِ خَبِيرٌ بَصِيرٌ ﴿٢٧﴾
ئەگەر اﷲ بەندىلىرىنىڭ رىزقىنى كەڭ قىلسا، ئۇلار ئەلۋەتتە زېمىندا پىتنە - پاسات تېرىيتتى. لېكىن اﷲ خالىغان نەرسىنى ئۆلچەم بىلەن چۈشۈرىدۇ، اﷲ ئىشەنچلىك ھالدا بەندىلىرىنىڭ (ئەھۋالىدىن) تولۇق خەۋەرداردۇر، (ئۇنى) كۆرۈپ تۇرغۇچىدۇر
وَهُوَ الَّذِي يُنَزِّلُ الْغَيْثَ مِن بَعْدِ مَا قَنَطُوا وَيَنشُرُ رَحْمَتَهُ ۚ وَهُوَ الْوَلِيُّ الْحَمِيدُ ﴿٢٨﴾
ئۇلار (يامغۇرنىڭ يېغىشىدىن ئۈمىدسىزلەنگەندىن كېيىن) اﷲ يامغۇر ياغدۇرۇپ بېرىدۇ، رەھمىتىنى يايىدۇ، اﷲ (ھەممىنىڭ) ئىگىسىدۇر، مەدھىيىگە لايىقتۇر
وَمِنْ آيَاتِهِ خَلْقُ السَّمَاوَاتِ وَالْأَرْضِ وَمَا بَثَّ فِيهِمَا مِن دَابَّةٍ ۚ وَهُوَ عَلَىٰ جَمْعِهِمْ إِذَا يَشَاءُ قَدِيرٌ ﴿٢٩﴾
ئاسمانلارنىڭ، زېمىننىڭ يارىتىلىشى ۋە ئۇ ئىككىسىدە جاندارلارنىڭ يېيىلىشى اﷲ نىڭ (قۇدرىتىنىڭ) دەلىللىرىدىندۇر، اﷲ ئەگەر خالىسا، ئۇلارنى جەمئى قىلىشقا قادىردۇر
وَمَا أَصَابَكُم مِّن مُّصِيبَةٍ فَبِمَا كَسَبَتْ أَيْدِيكُمْ وَيَعْفُو عَن كَثِيرٍ ﴿٣٠﴾
سىلەرگە (يەنى جېنىڭلار ياكى مېلىڭلارغا) ھەرقانداق بىر مۇسىبەت يەتسە، ئۇ سىلەرنىڭ قىلغان گۇناھىڭلار تۈپەيلىدىن كەلگەن بولىدۇ، اﷲ نۇرغۇن گۇناھىڭلارنى ئەپۇ قىلىدۇ (ئەپۇ قىلمايدىغان بولسا، بالا ۋە مۇسىبەتتىن ھېچ قۇتۇلماس ئىدىڭلار)
وَمَا أَنتُم بِمُعْجِزِينَ فِي الْأَرْضِ ۖ وَمَا لَكُم مِّن دُونِ اللَّهِ مِن وَلِيٍّ وَلَا نَصِيرٍ ﴿٣١﴾
(ئى مۇشرىكلار!) سىلەر زېمىندا اﷲ نىڭ ئازابىدىن قېچىپ قۇتۇلالمايسىلەر، سىلەرگە اﷲ تىن باشقا ھېچ ئىگە ۋە مەدەتكار يوق
وَمِنْ آيَاتِهِ الْجَوَارِ فِي الْبَحْرِ كَالْأَعْلَامِ ﴿٣٢﴾
دېڭىزدا قاتنىغۇچى تاغدەك كېمىلەر اﷲ نىڭ قۇدرىتىنىڭ دەلىللىرىدىندۇر
إِن يَشَأْ يُسْكِنِ الرِّيحَ فَيَظْلَلْنَ رَوَاكِدَ عَلَىٰ ظَهْرِهِ ۚ إِنَّ فِي ذَٰلِكَ لَآيَاتٍ لِّكُلِّ صَبَّارٍ شَكُورٍ ﴿٣٣﴾
ئەگەر اﷲ خالىسا، شامالنى توختىتىۋېتىدۇ - دە، كېمىلەر دېڭىزنىڭ ئۈستىدە توختاپ قالىدۇ، بۇنىڭدا (قاتتىقلىقتا) سەۋر قىلغۇچى، (كەڭچىلىكتە) شۈكۈر قىلغۇچى ھەر مۆمىن ئۈچۈن نۇرغۇن ئىبرەتلەر بار
أَوْ يُوبِقْهُنَّ بِمَا كَسَبُوا وَيَعْفُ عَن كَثِيرٍ ﴿٣٤﴾
ياكى اﷲ خالىسا، ئىچىدىكى كىشىلەرنىڭ قىلغان گۇناھلىرى تۈپەيلىدىن كېمىلەرنى (بوران بىلەن) ھالاك قىلىدۇ، اﷲ نۇرغۇن كىشلەرنىڭ گۇناھلىرىنى ئەپۇ قىلىدۇ
وَيَعْلَمَ الَّذِينَ يُجَادِلُونَ فِي آيَاتِنَا مَا لَهُم مِّن مَّحِيصٍ ﴿٣٥﴾
بىزنىڭ ئايەتلىرىمىز ئۈستىدە مۇجادىلە قىلىدىغانلار ئۆزلىرىگە اﷲ نىڭ ئازابىدىن قېچىپ قۇتۇلىدىغان جاي يوق ئىكەنلىكىنى بىلىدۇ
فَمَا أُوتِيتُم مِّن شَيْءٍ فَمَتَاعُ الْحَيَاةِ الدُّنْيَا ۖ وَمَا عِندَ اللَّهِ خَيْرٌ وَأَبْقَىٰ لِلَّذِينَ آمَنُوا وَعَلَىٰ رَبِّهِمْ يَتَوَكَّلُونَ ﴿٣٦﴾
(دۇنيانىڭ نېمەتلىرىدىن) سىلەرگە بېرىلگەن ھەر قانداق نەرسە دۇنيا تىرىكچىلىكىدە پايدىلىنىدىغان نەرسىلەردۇر، ئىمان ئېيتقان، پەرۋەردىگارىغا تەۋەككۈل قىلىدىغان كىشىلەرگە، گۇناھى كەبرىلەردىن، قەبىھ ئىشلاردىن ساقلانغۇچىلارغا، دەرغەزەپ بولغانلىرىدا كەچۈرەلەيدىغانلارغا، پەرۋەردىگارىنىڭ دەۋىتىگە ئاۋاز قوشالايدىغانلارغا، نامازنى (تەئدىل ئەركان بىلەن) ئۆتەيدىغانلارغا، ئىشلىرىنى مەسلىھەت بىلەن قارار قىلىدىغانلارغا، بىز رىزىق قىلىپ بەرگەن نەرسىلەردىن سەدىقە قىلىدىغانلارغا، ئۇچرىغان زۇلۇمغا قارشى تۇرالايدىغانلارغا اﷲ نىڭ ھۇزۇرىدىكى ساۋاب تېخىمۇ ياخشىدۇر، تېخىمۇ باقىدۇر
وَالَّذِينَ يَجْتَنِبُونَ كَبَائِرَ الْإِثْمِ وَالْفَوَاحِشَ وَإِذَا مَا غَضِبُوا هُمْ يَغْفِرُونَ ﴿٣٧﴾
(دۇنيانىڭ نېمەتلىرىدىن) سىلەرگە بېرىلگەن ھەر قانداق نەرسە دۇنيا تىرىكچىلىكىدە پايدىلىنىدىغان نەرسىلەردۇر، ئىمان ئېيتقان، پەرۋەردىگارىغا تەۋەككۈل قىلىدىغان كىشىلەرگە، گۇناھى كەبرىلەردىن، قەبىھ ئىشلاردىن ساقلانغۇچىلارغا، دەرغەزەپ بولغانلىرىدا كەچۈرەلەيدىغانلارغا، پەرۋەردىگارىنىڭ دەۋىتىگە ئاۋاز قوشالايدىغانلارغا، نامازنى (تەئدىل ئەركان بىلەن) ئۆتەيدىغانلارغا، ئىشلىرىنى مەسلىھەت بىلەن قارار قىلىدىغانلارغا، بىز رىزىق قىلىپ بەرگەن نەرسىلەردىن سەدىقە قىلىدىغانلارغا، ئۇچرىغان زۇلۇمغا قارشى تۇرالايدىغانلارغا اﷲ نىڭ ھۇزۇرىدىكى ساۋاب تېخىمۇ ياخشىدۇر، تېخىمۇ باقىدۇر
وَالَّذِينَ اسْتَجَابُوا لِرَبِّهِمْ وَأَقَامُوا الصَّلَاةَ وَأَمْرُهُمْ شُورَىٰ بَيْنَهُمْ وَمِمَّا رَزَقْنَاهُمْ يُنفِقُونَ ﴿٣٨﴾
(دۇنيانىڭ نېمەتلىرىدىن) سىلەرگە بېرىلگەن ھەر قانداق نەرسە دۇنيا تىرىكچىلىكىدە پايدىلىنىدىغان نەرسىلەردۇر، ئىمان ئېيتقان، پەرۋەردىگارىغا تەۋەككۈل قىلىدىغان كىشىلەرگە، گۇناھى كەبرىلەردىن، قەبىھ ئىشلاردىن ساقلانغۇچىلارغا، دەرغەزەپ بولغانلىرىدا كەچۈرەلەيدىغانلارغا، پەرۋەردىگارىنىڭ دەۋىتىگە ئاۋاز قوشالايدىغانلارغا، نامازنى (تەئدىل ئەركان بىلەن) ئۆتەيدىغانلارغا، ئىشلىرىنى مەسلىھەت بىلەن قارار قىلىدىغانلارغا، بىز رىزىق قىلىپ بەرگەن نەرسىلەردىن سەدىقە قىلىدىغانلارغا، ئۇچرىغان زۇلۇمغا قارشى تۇرالايدىغانلارغا اﷲ نىڭ ھۇزۇرىدىكى ساۋاب تېخىمۇ ياخشىدۇر، تېخىمۇ باقىدۇر
وَالَّذِينَ إِذَا أَصَابَهُمُ الْبَغْيُ هُمْ يَنتَصِرُونَ ﴿٣٩﴾
(دۇنيانىڭ نېمەتلىرىدىن) سىلەرگە بېرىلگەن ھەر قانداق نەرسە دۇنيا تىرىكچىلىكىدە پايدىلىنىدىغان نەرسىلەردۇر، ئىمان ئېيتقان، پەرۋەردىگارىغا تەۋەككۈل قىلىدىغان كىشىلەرگە، گۇناھى كەبرىلەردىن، قەبىھ ئىشلاردىن ساقلانغۇچىلارغا، دەرغەزەپ بولغانلىرىدا كەچۈرەلەيدىغانلارغا، پەرۋەردىگارىنىڭ دەۋىتىگە ئاۋاز قوشالايدىغانلارغا، نامازنى (تەئدىل ئەركان بىلەن) ئۆتەيدىغانلارغا، ئىشلىرىنى مەسلىھەت بىلەن قارار قىلىدىغانلارغا، بىز رىزىق قىلىپ بەرگەن نەرسىلەردىن سەدىقە قىلىدىغانلارغا، ئۇچرىغان زۇلۇمغا قارشى تۇرالايدىغانلارغا اﷲ نىڭ ھۇزۇرىدىكى ساۋاب تېخىمۇ ياخشىدۇر، تېخىمۇ باقىدۇر
وَجَزَاءُ سَيِّئَةٍ سَيِّئَةٌ مِّثْلُهَا ۖ فَمَنْ عَفَا وَأَصْلَحَ فَأَجْرُهُ عَلَى اللَّهِ ۚ إِنَّهُ لَا يُحِبُّ الظَّالِمِينَ ﴿٤٠﴾
بىر يامانلىقنىڭ جازاسى شۇنىڭغا ئوخشاش بىر يامانلىقتۇر (يەنى ساڭا بىر كىشى قانچىلىك چېقىلسا، شۇنچىلىك چېقىلماي ئاشۇرۇۋەتسەڭ بولمايدۇ)، كىمكى (ئىنتىقام ئېلىشقا قادىر تۇرۇقلۇق) ئەپۇ قىلسا ۋە (ئۆزى بىلەن يامانلىق قىلغۇچىنىڭ ئارىسىنى) تۈزىسە، ئۇنىڭ ئەجرىنى اﷲ بېرىدۇ، اﷲ ئىشەنچلىك ھالدا زۇلۇم قىلغۇچىلارنى دوست تۇتمايدۇ
وَلَمَنِ انتَصَرَ بَعْدَ ظُلْمِهِ فَأُولَٰئِكَ مَا عَلَيْهِم مِّن سَبِيلٍ ﴿٤١﴾
زۇلۇمغا ئۇچرىغۇچى ئادەم ئىنتىقامنى ئالسا ئۇنى ئەيىبلەشكە بولمايدۇ
إِنَّمَا السَّبِيلُ عَلَى الَّذِينَ يَظْلِمُونَ النَّاسَ وَيَبْغُونَ فِي الْأَرْضِ بِغَيْرِ الْحَقِّ ۚ أُولَٰئِكَ لَهُمْ عَذَابٌ أَلِيمٌ ﴿٤٢﴾
ئەيىبلىنىدىغانلار پەقەت كىشىلەرگە زۇلۇم قىلىدىغان، زېمىندا ناھەق رەۋىشتە پىتنە - پاسات تېرىيدىغان ئادەملەردۇر، ئەنە شۇلار قاتتىق ئازابقا دۇچار بولىدۇ
وَلَمَن صَبَرَ وَغَفَرَ إِنَّ ذَٰلِكَ لَمِنْ عَزْمِ الْأُمُورِ ﴿٤٣﴾
كىمكى (ئەزىيەتكە) سەۋر قىلسا، (اﷲ نىڭ رازىلىقى ئۈچۈن) ئىنتىقام ئالمىسا، بۇ ئەلۋەتتە مەرغۇپ ئىشلاردىندۇر
وَمَن يُضْلِلِ اللَّهُ فَمَا لَهُ مِن وَلِيٍّ مِّن بَعْدِهِ ۗ وَتَرَى الظَّالِمِينَ لَمَّا رَأَوُا الْعَذَابَ يَقُولُونَ هَلْ إِلَىٰ مَرَدٍّ مِّن سَبِيلٍ ﴿٤٤﴾
اﷲ گۇمراھ قىلغان ئادەمگە شۇنىڭدىن كېيىن ھېچ ئىگە يوقتۇر، زالىملار ئازابنى كۆرگەن چاغدا، ئۇلارنىڭ «(دۇنياغا) قايتىشقا يول بارمىدۇ؟» دېگەنلىكىنى كۆرىسەن
وَتَرَاهُمْ يُعْرَضُونَ عَلَيْهَا خَاشِعِينَ مِنَ الذُّلِّ يَنظُرُونَ مِن طَرْفٍ خ��فِيٍّ ۗ وَقَالَ الَّذِينَ آمَنُوا إِنَّ الْخَاسِرِينَ الَّذِينَ خَسِرُوا أَنفُسَهُمْ وَأَهْلِيهِمْ يَوْمَ الْقِيَامَةِ ۗ أَلَا إِنَّ الظَّالِمِينَ فِي عَذَابٍ مُّقِيمٍ ﴿٤٥﴾
ئۇلارنىڭ قورققان، خارلانغان ھالدا، (دوزاخقا) كۆزىنىڭ قۇيرۇقى بىلەن قارىغان ھالدا دوزاخقا توغرىلىنىدىغانلىقىنى كۆرىسەن، مۆمىنلەر: «شۈبھىسىزكى، زىيان تارتقۇچىلار قىيامەت كۈنى ئۆزلىرىگە ۋە ئائىلىسىدىكىلەرگە زىيان سالغۇچىلاردۇر» دەيدۇ، بىلىڭلاركى، زالىملار ئىشەنچلىك ھالدا دائىملىق ئازابتىدۇر
وَمَا كَانَ لَهُم مِّنْ أَوْلِيَاءَ يَنصُرُونَهُم مِّن دُونِ اللَّهِ ۗ وَمَن يُضْلِلِ اللَّهُ فَمَا لَهُ مِن سَبِيلٍ ﴿٤٦﴾
ئۇلارغا اﷲ تىن باشقا ياردەم بېرىدىغان دوستلار بولمايدۇ، اﷲ كىمنى گۇمراھ قىلسا، ئۇنىڭغا (ئۇنى دۇنيادا ھەقكە، ئاخىرەتتە مەڭگۈلۈك باغلاركە ئېرىشتۈرگۈچى) ھېچ يول بولمايدۇ
اسْتَجِيبُوا لِرَبِّكُم مِّن قَبْلِ أَن يَأْتِيَ يَوْمٌ لَّا مَرَدَّ لَهُ مِنَ اللَّهِ ۚ مَا لَكُم مِّن مَّلْجَإٍ يَوْمَئِذٍ وَمَا لَكُم مِّن نَّكِيرٍ ﴿٤٧﴾
اﷲ تەرىپىدىن كەلگەن، قارشى تۇرغىلى بولمايدىغان بىر كۈن كېلىشتىن بۇرۇن پەرۋەردىگارىڭلارنىڭ دەۋىتىنى قوبۇل قىلىڭلار، بۇ كۈندە سىلەرگە قاچىدىغان جاي تېپىلماي قالىدۇ. (قىلمىشىڭلار نامە - ئەمالىڭلارغا يېزىقلىق بولغانلىقى، ئەزايىڭلار ئۇنىڭغا گۇۋاھ بولغانلىقى ئۈچۈن) سىلەر (ئۇنى) ئىنكار قىلالمايسىلەر
فَإِنْ أَعْرَضُوا فَمَا أَرْسَلْنَاكَ عَلَيْهِمْ حَفِيظًا ۖ إِنْ عَلَيْكَ إِلَّا الْبَلَاغُ ۗ وَإِنَّا إِذَا أَذَقْنَا الْإِنسَانَ مِنَّا رَحْمَةً فَرِحَ بِهَا ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌ بِمَا قَدَّمَتْ أَيْدِيهِمْ فَإِنَّ الْإِنسَانَ كَفُورٌ ﴿٤٨﴾
ئەگەر ئۇلار (يەنى مۇشرىكلار) (ئىماندىن) يۈز ئۆرۈسە، بىز سېنى ئۇلارغا كۆزەتچى قىلىپ ئەۋەتكىنىمىز يوق، سېنىڭ ۋەزىپەڭ پەقەت تەبلىغ قىلىشتۇر، ئەگەر ئىنسانغا بىزنىڭ رەھمىتىمىزنى تېتىتساق خۇش بولۇپ كېتىدۇ، ئۇلارنىڭ قىلغان گۇناھلىرى تۈپەيلىدىن (بېشىغا) بىرەر ھادىسە كەلسە، ئىنسان تولىمۇ كۇفرانە نېمەت قىلغۇچىدۇر
لِّلَّهِ مُلْكُ السَّمَاوَاتِ وَالْأَرْضِ ۚ يَخْلُقُ مَا يَشَاءُ ۚ يَهَبُ لِمَن يَشَاءُ إِنَاثًا وَيَهَبُ لِمَن يَشَاءُ الذُّكُورَ ﴿٤٩﴾
ئاسمانلارنىڭ ۋە زېمىننىڭ پادىشاھلىقى اﷲ قا خاستۇر، اﷲ نېمىنى خالىسا شۇنى يارىتىدۇ، خالىغان ئادەمگە قىز پەرزەنت ئاتا قىلىدۇ، خالىغان ئادەمگە ئوغۇل پەرزەنت ئاتا قىلىدۇ
أَوْ يُزَوِّجُهُمْ ذُكْرَانًا وَإِنَاثًا ۖ وَيَجْعَلُ مَن يَشَاءُ عَقِيمًا ۚ إِنَّهُ عَلِيمٌ قَدِيرٌ ﴿٥٠﴾
ياكى ئۇلارغا ئوغۇل، قىزنى ئارىلاش بېرىدۇ، خالىغان ئادەمنى تۇغماس قىلىدۇ، اﷲ ئىشەنچلىك ھالدا ھەممىنى بىلگۈچىدۇر، ھەممىگە قادىردۇر
وَمَا كَانَ لِبَشَرٍ أَن يُكَلِّمَهُ اللَّهُ إِلَّا وَحْيًا أَوْ مِن وَرَاءِ حِجَابٍ أَوْ يُرْسِلَ رَسُولًا فَيُوحِيَ بِإِذْنِهِ مَا يَشَاءُ ۚ إِنَّهُ عَلِيٌّ حَكِيمٌ ﴿٥١﴾
اﷲ ھەر قانداق ئادەمگە پەقەت ۋەھيى ئارقىلىق ياكى پەردە ئارقىسىدىنلا ياكى بىر خۇدانىڭ ئەلچىسىنى ئەۋەتىپ ئۆز ئىزنى بىلەن ئۇنىڭغا خالىغان ۋەھيىنى قىلىش ئارقىلىقلا سۆز قىلغان. اﷲ ئىشەنچلىك ھالدا (مەخلۇقلارنىڭ سۈپەتلىرىدىن) ئۈستۈندۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
وَكَذَٰلِكَ أَوْحَيْنَا إِلَيْكَ رُوحًا مِّنْ أَمْرِنَا ۚ مَا كُنتَ تَدْرِي مَا الْكِتَابُ وَلَا الْإِيمَانُ وَلَٰكِن جَعَلْنَاهُ نُورًا نَّهْدِي بِهِ مَن نَّشَاءُ مِنْ عِبَادِنَا ۚ وَإِنَّكَ لَتَهْدِي إِلَىٰ صِرَاطٍ مُّسْتَقِيمٍ ﴿٥٢﴾
شۇنىڭدەك (يەنى باشقا خۇدانىڭ ئەلچى-رەسۇللىرىگە ۋەھيى قىلغاندەك) ئەمرىمىز بويىچە ساڭا قۇرئاننى ۋەھيى قىلدۇق، سەن (ۋەھيىدىن ئىلگىرى) قۇرئاننىڭ ۋە ئىماننىڭ نېمە ئىكەنلىكىنى ئۇقمايتتىڭ، لېكىن بىز قۇرئاننى بىر نۇر قىلدۇقكى، ئۇنىڭ بىلەن بەندىلىرىمىزدىن خالىغان كىشىلەرنى ھىدايەت قىلىمىز. شەك - شۈبھىسىزكى سەن توغرا يولغا باشلايسەن
صِرَاطِ اللَّهِ الَّذِي لَهُ مَا فِي السَّمَاوَاتِ وَمَا فِي الْأَرْضِ ۗ أَلَا إِلَى اللَّهِ تَصِيرُ الْأُمُورُ ﴿٥٣﴾
