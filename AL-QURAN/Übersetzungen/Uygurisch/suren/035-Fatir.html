بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
✦ ✦
الْحَمْدُ لِلَّهِ فَاطِرِ السَّمَاوَاتِ وَالْأَرْضِ جَاعِلِ الْمَلَائِكَةِ رُسُلًا أُولِي أَجْنِحَةٍ مَّثْنَىٰ وَثُلَاثَ وَرُبَاعَ ۚ يَزِيدُ فِي الْخَلْقِ مَا يَشَاءُ ۚ إِنَّ اللَّهَ عَلَىٰ كُلِّ شَيْءٍ قَدِيرٌ ﴿١﴾
جىمى ھەمدۇسانا ئاسمانلارنى ۋە زېمىننى ئۆرنەكسىز ياراتقۇچى، خۇدانىڭ ئەلچى پەرىشتىلىرىنى ئىككى قاناتلىق، ئۈچ قاناتلىق، تۆت قاناتلىق ئەلچىلەر قىلغۇچى اﷲ قا خاستۇر! اﷲ يارىتىشتا خالىغىنىنى زىيادە قىلىدۇ، اﷲ ئىشەنچلىك ھالدا ھەر نەرسىگە قادىردۇر
مَّا يَفْتَحِ اللَّهُ لِلنَّاسِ مِن رَّحْمَةٍ فَلَا مُمْسِكَ لَهَا ۖ وَمَا يُمْسِكْ فَلَا مُرْسِلَ لَهُ مِن بَعْدِهِ ۚ وَهُوَ الْعَزِيزُ الْحَكِيمُ ﴿٢﴾
اﷲ رەھمىتىدىن كىشىلەرگە ئاچقان نەرسىنى ھەرگىزمۇ توسۇپ قالالىغۇچى بولمايدۇ، اﷲ رەھمىتىدىن توسقان نەرسىنى، اﷲ توسقاندىن كېيىن، ئۇنى ھېچ قويۇپ بەرگۈچى بولمايدۇ، اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر
يَا أَيُّهَا النَّاسُ اذْكُرُوا نِعْمَتَ اللَّهِ عَلَيْكُمْ ۚ هَلْ مِنْ خَالِقٍ غَيْرُ اللَّهِ يَرْزُقُكُم مِّنَ السَّمَاءِ وَالْأَرْضِ ۚ لَا إِلَٰهَ إِلَّا هُوَ ۖ فَأَنَّىٰ تُؤْفَكُونَ ﴿٣﴾
ئى ئىنسانلار! اﷲ نىڭ سىلەرگە بەرگەن نېمىتىنى ياد ئېتىڭلار، اﷲ تىن باشقا سىلەرگە ئاسماندىن ۋە زېمىندىن رىزىق بېرىپ تۇرىدىغان ياراتقۇچى بارمۇ؟ اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، (اﷲ نىڭ ئىبادىتىدىن) قانداقمۇ باش تارتىسىلەر؟
وَإِن يُكَذِّبُوكَ فَقَدْ كُذِّبَتْ رُسُلٌ مِّن قَبْلِكَ ۚ وَإِلَى اللَّهِ تُرْجَعُ الْأُمُورُ ﴿٤﴾
(ئى مۇھەممەد!) ئەگەر ئۇلار سېنى ئىنكار قىلسا (قايغۇرۇپ كەتمە)، ئىشەنچلىك ھالدا سەندىن ئىلگىرىكى خۇدانىڭ ئەلچى-رەسۇللىرىمۇ ئىنكار قىلىنغان، ھەممە ئىش اﷲ نىڭ دەرگاھىغا قايتۇرۇلىدۇ
يَا أَيُّهَا النَّاسُ إِنَّ وَعْدَ اللَّهِ حَقٌّ ۖ فَلَا تَغُرَّنَّكُمُ الْحَيَاةُ الدُّنْيَا ۖ وَلَا يَغُرَّنَّكُم بِاللَّهِ الْغَرُورُ ﴿٥﴾
ئى ئىنسانلار! اﷲ نىڭ ۋەدىسى ئىشەنچلىك ھالدا ھەقتۇر، سىلەرنى ھەرگىز دۇنيا تىرىكچىلىكى مەغرۇر قىلمىسۇن، شەيتان (ئازدۇرغۇچى)نىڭ سىلەرنى اﷲ نىڭ ئەپۇسىنىڭ كەڭلىكى بىلەن مەغرۇر قىلىشىغا يول قويماڭلار
إِنَّ الشَّيْطَانَ لَكُمْ عَدُوٌّ فَاتَّخِذُوهُ عَدُوًّا ۚ إِنَّمَا يَدْعُو حِزْبَهُ لِيَكُونُوا مِنْ أَصْحَابِ السَّعِيرِ ﴿٦﴾
شەيتان (ئازدۇرغۇچى) ئىشەنچلىك ھالدا سىلەرگە دۈشمەندۇر، ئۇنى دۈشمەن تۇتۇڭلار، شەيتان (ئازدۇرغۇچى) ئۆزىنىڭ تەۋەلىرىنى ئەھلى دوزاختىن بولۇشقا چاقىرىدۇ
الَّذِينَ كَفَرُوا لَهُمْ عَذَابٌ شَدِيدٌ ۖ وَالَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ لَهُم مَّغْفِرَةٌ وَأَجْرٌ كَبِيرٌ ﴿٧﴾
ھەقىقەتنى رەت قىلغۇچىلار قاتتىق ئازابقا دۇچار بولىدۇ، ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلار مەغپىرەتكە ۋە كاتتا ساۋابقا ئېرىشىدۇ
أَفَمَن زُيِّنَ لَهُ سُوءُ عَمَلِهِ فَرَآهُ حَسَنًا ۖ فَإِنَّ اللَّهَ يُضِلُّ مَن يَشَاءُ وَيَهْدِي مَن يَشَاءُ ۖ فَلَا تَذْهَبْ نَفْسُكَ عَلَيْهِمْ حَسَرَاتٍ ۚ إِنَّ اللَّهَ عَلِيمٌ بِمَا يَصْنَعُونَ ﴿٨﴾
ئۆزىنىڭ يامان ئەمەلى چىرايلىق كۆرسىتىلگەن، شۇنىڭ بىلەن ئۇنى چىرايلىق ھېسابلىغان كىشى (ھىدايەت تاپقان كىشى بىلەن ئوخشاشمۇ؟) اﷲ ئىشەنچلىك ھالدا خالىغان كىشىنى گۇمراھ قىلىدۇ (خالىغان كىشىنى ھىدايەت قىلىدۇ). شۇڭا سەن ئۇلار ئۈچۈن (يەنى ئىمان ئېيتمىغانلىقلىرى ئۈچۈن) قايغۇرۇپ ئۆزۈڭنى ھالاك قىلىۋالمىغىن، اﷲ ئىشەنچلىك ھالدا ئۇلارنىڭ قىلمىشلىرىنى بىلىپ تۇرغۇچىدۇر
وَاللَّهُ الَّذِي أَرْسَلَ الرِّيَاحَ فَتُثِيرُ سَحَابًا فَسُقْنَاهُ إِلَىٰ بَلَدٍ مَّيِّتٍ فَأَحْيَيْنَا بِهِ الْأَرْضَ بَعْدَ مَوْتِهَا ۚ كَذَٰلِكَ النُّشُورُ ﴿٩﴾
اﷲ شاماللارنى ئەۋەتىپ (ئۇنىڭ بىلەن) بۇلۇتنى قوزغايدۇ، بىز بۇلۇتلارنى ئۆلۈك (يەنى قاغجىراق) بىر يەرگە ھەيدەيمىز (بۇلۇتتىن يامغۇر ياغدۇرىمىز)، شۇ يامغۇر بىلەن ئۆلگەن زېمىننى تىرىلدۈرىمىز. ئۆلۈكلەرنىڭ تىرىلدۈرۈلۈشى ئەنە شۇنداقتۇر
مَن كَانَ يُرِيدُ الْعِزَّةَ فَلِلَّهِ الْعِزَّةُ جَمِيعًا ۚ إِلَيْهِ يَصْعَدُ الْكَلِمُ الطَّيِّبُ وَالْعَمَلُ الصَّالِحُ يَرْفَعُهُ ۚ وَالَّذِينَ يَمْكُرُونَ السَّيِّئَاتِ لَهُمْ عَذَابٌ شَدِيدٌ ۖ وَمَكْرُ أُولَٰئِكَ هُوَ يَبُورُ ﴿١٠﴾
كىمكى (دۇنيادا) شەرەپ ئىزدەيدىكەن، شەرەپنىڭ ھەممىسى اﷲ قا مەنسۇپ (ئۇنى اﷲ تىن تىلىسۇن). ياخشى سۆز اﷲ نىڭ دەرگاھىغا ئۆرلەيدۇ، ياخشى ئەمەل ئۇنى كۆتۈرىدۇ، (خۇدانىڭ ئەلچىسىگە قارشى) ھىيلە - مىكىرلەرنى تۈزىدىغانلار قاتتىق ئازابقا دۇچار بولىدۇ، ئۇلارنىڭ ھىيلە - مىكرى ئىشقا ئاشمايدۇ
وَاللَّهُ خَلَقَكُم مِّن تُرَابٍ ثُمَّ مِن نُّطْفَةٍ ثُمَّ جَعَلَكُمْ أَزْوَاجًا ۚ وَمَا تَحْمِلُ مِنْ أُنثَىٰ وَلَا تَضَعُ إِلَّا بِعِلْمِهِ ۚ وَمَا يُعَمَّرُ مِن مُّعَمَّرٍ وَلَا يُنقَصُ مِنْ عُمُرِهِ إِلَّا فِي كِتَابٍ ۚ إِنَّ ذَٰلِكَ عَلَى اللَّهِ يَسِيرٌ ﴿١١﴾
اﷲ سىلەرنى (يەنى ئەسلىڭلار ئادەم ئەلەيھىسسالامنى) تۇپراقتىن ياراتتى. ئاندىن ئابىمەنىيدىن پەيدا قىلدى. ئاندىن سىلەرنى (ئەر - ئايالدىن ئىبارەت) بىر جۈپ قىلدى، ھەر قانداق ئايالنىڭ ھامىلدار بولۇشى ۋە تۇغۇشى اﷲ نىڭ ئىلمىنىڭ سىرتىدا ئەمەس، ھەرقانداق ئادەمنىڭ ئۆمرىنىڭ ئۇزۇن ياكى قىسقا بولۇشى لەۋھۇلمەھپۇزدا يېزىلغان، بۇ ئىشەنچلىك ھالدا اﷲ قا ئاسان
وَمَا يَسْتَوِي الْبَحْرَانِ هَٰذَا عَذْبٌ فُرَاتٌ سَائِغٌ شَرَابُهُ وَهَٰذَا مِلْحٌ أُجَاجٌ ۖ وَمِن كُلٍّ تَأْكُلُونَ لَحْمًا طَرِيًّا وَتَسْتَخْرِجُونَ حِلْيَةً تَلْبَسُونَهَا ۖ وَتَرَى الْفُلْكَ فِيهِ مَوَاخِرَ لِتَبْتَغُوا مِن فَضْلِهِ وَلَعَلَّكُمْ تَشْكُرُونَ ﴿١٢﴾
ئىككى دېڭىز (يەنى دېڭىز بىلەن دەريا) ئوخشاش ئەمەس. بۇنىڭ (يەنى دەريانىڭ) سۈيى تاتلىق، تەملىك بولۇپ (گالدىن) سىلىق ئۆتىدۇ، بۇ (يەنى دېڭىز) ناھايىتى ئاچچىق، تۇزلۇق، ئۇلارنىڭ ھەر ئىككىسىدىن يېڭى گۆشلەرنى (يەنى بېلىقلارنى) يەيسىلەر، (دېڭىزدىن) سىلەر تاقايدىغان زىننەت بۇيۇملىرى (يەنى ئۈنچە - مارجانلار) نى چىقىرىسىلەر، اﷲ نىڭ نېمىتىنى تەلەپ قىلىشڭلار ئۈچۈن، شۈكۈر قىلىشىڭلار ئۈچۈن (اﷲ نىڭ ئەمرى بىلەن يۈك - تاق، يېمەك - ئىچمەكلەر قاچىلانغان) كېمىلەرنىڭ دېڭىزدا دولقۇن يېرىپ كېتىۋاتقانلىقىنى كۆرىسەن
يُولِجُ اللَّيْلَ فِي النَّهَارِ وَيُولِجُ النَّهَارَ فِي اللَّيْلِ وَسَخَّرَ الشَّمْسَ وَالْقَمَرَ كُلٌّ يَجْرِي لِأَجَلٍ مُّسَمًّى ۚ ذَٰلِكُمُ اللَّهُ رَبُّكُمْ لَهُ الْمُلْكُ ۚ وَالَّذِينَ تَدْعُونَ مِن دُونِهِ مَا يَمْلِكُونَ مِن قِطْمِيرٍ ﴿١٣﴾
اﷲ كېچىنى كۈندۈزگە كىرگۈزىدۇ، كۈندۈزنى كېچىگە كىرگۈزىدۇ (شۇنىڭ بىلەن كېچە - كۈندۈزنىڭ ئۇزۇن - قىسقا بولۇشى جايلار، پەسىللەر بويىچە نۆۋەتلىشىپ تۇرىدۇ)، اﷲ كۈن بىلەن ئاينى (بەندىلىرىنىڭ مەنپەئىتىگە) بويسۇندۇرۇپ بەردى. ھەر ئىككىلىسى مۇئەييەن مۇددەتكىچە (يەنى قىيامەتكىچە ئۆز ئوقىدا) سەير قىلىدۇ، ئەنە شۇ اﷲ سىلەرنىڭ پەرۋەردىگارىڭلاردۇر، پادىشاھلىق ئۇنىڭغا خاستۇر، اﷲ نى قويۇپ چوقۇنۇۋاتقان بۇتلىرىڭلار قىلچىلىك نەرسىگە ئىگە ئەمەس
إِن تَدْعُوهُمْ لَا يَسْمَعُوا دُعَاءَكُمْ وَلَوْ سَمِعُوا مَا اسْتَجَابُوا لَكُمْ ۖ وَيَوْمَ الْقِيَامَةِ يَكْفُرُونَ بِشِرْكِكُمْ ۚ وَلَا يُنَبِّئُكَ مِثْلُ خَبِيرٍ ﴿١٤﴾
ئەگەر ئۇلارنى چاقىرساڭلار، سىلەرنىڭ چاقىرغىنىڭلارنى ئاڭلىمايدۇ، ئاڭلىغان تەقدىردىمۇ سىلەرگە جاۋاپ قايتۇرالمايدۇ، قىيامەت كۈنى ئۇلار (اﷲ قا ئۇلارنى) شېرىك قىلغانلىقىڭلارنى ئىنكار قىلىدۇ، (مۇشرىكلارنىڭ ۋە ئۇلارنىڭ بۇتلىرىنىڭ قىيامەت كۈنىدىكى ئەھۋالىنى ھېچ ئەھەدى) ساڭا ھەممىدىن خەۋەردار زاتتەك (يەنى اﷲ دەك) ئېيتىپ بېرەلمەيدۇ
يَا أَيُّهَا النَّاسُ أَنتُمُ الْفُقَرَاءُ إِلَى اللَّهِ ۖ وَاللَّهُ هُوَ الْغَنِيُّ الْحَمِيدُ ﴿١٥﴾
ئى ئىنسانلار! سىلەر اﷲ قا موھتاجسىلەر، اﷲ (ھەممىدىن) بىھاجەتتۇر، مەدھىيىگە لايىقتۇر
إِن يَشَأْ يُذْهِبْكُمْ وَيَأْتِ بِخَلْقٍ جَدِيدٍ ﴿١٦﴾
ئەگەر اﷲ خالىسا، سىلەرنى ھالاك قىلىپ (ئورنۇڭلارغا) يېڭى بىر خەلقنى پەيدا قىلىدۇ
وَمَا ذَٰلِكَ عَلَى اللَّهِ بِعَزِيزٍ ﴿١٧﴾
بۇ اﷲ قا قىيىن ئەمەس
وَلَا تَزِرُ وَازِرَةٌ وِزْرَ أُخْرَىٰ ۚ وَإِن تَدْعُ مُثْقَلَةٌ إِلَىٰ حِمْلِهَا لَا يُحْمَلْ مِنْهُ شَيْءٌ وَلَوْ كَانَ ذَا قُرْبَىٰ ۗ إِنَّمَا تُنذِرُ الَّذِينَ يَخْشَوْنَ رَبَّهُم بِالْغَيْبِ وَأَقَامُوا الصَّلَاةَ ۚ وَمَن تَزَكَّىٰ فَإِنَّمَا يَتَزَكَّىٰ لِنَفْسِهِ ۚ وَإِلَى اللَّهِ الْمَصِيرُ ﴿١٨﴾
ھېچبىر گۇناھكار ئادەم بىراۋنىڭ گۇناھىنى ئۈستىگە ئالالمايدۇ، گۇناھى ئېغىر بىر ئادەم باشقا بىراۋنى ئۆز گۇناھىنى ئۈستىگە ئېلىۋېلىشقا چاقىرسا، ئۇ يېقىن تۇغقىنى بولغان تەقدىردىمۇ، ئۇنىڭ گۇناھىدىن ئازراق نەرسىنى ئۈستىگە ئېلىۋالالمايدۇ، سەن پەقەت پەرۋەردىگارىنى كۆرمەي تۇرۇپ ئۇنىڭدىن قورقىدىغانلارنى، نامازنى ئادا قىلىدىغانلارنى ئاگاھلاندۇرىسەن، كىمكى (گۇناھلاردىن) پاكلىنىدىكەن، ئۇ ئۆزى ئۈچۈن پاكلانغان بولىدۇ. ئاخىر قايتىدىغان جاي اﷲ نىڭ دەرگاھىدۇر
وَمَا يَسْتَوِي الْأَعْمَىٰ وَالْبَصِيرُ ﴿١٩﴾
كور ئادەم بىلەن كۆزى ساق ئادەم (يەنى ھەقىقەتنى رەت قىلغۇچى بىلەن مۆمىن) باراۋەر بولمايدۇ
وَلَا الظُّلُمَاتُ وَلَا النُّورُ ﴿٢٠﴾
زۇلمەت بىلەن نۇر (يەنى باتىل بىلەن ھەق) باراۋەر بولمايدۇ
وَلَا الظِّلُّ وَلَا الْحَرُورُ ﴿٢١﴾
سايە بىلەن ئىسسىق (يەنى مەڭگۈلۈك باغلار بىلەن دوزاخ) باراۋەر بولمايدۇ
وَمَا يَسْتَوِي الْأَحْيَاءُ وَلَا الْأَمْوَاتُ ۚ إِنَّ اللَّهَ يُسْمِعُ مَن يَشَاءُ ۖ وَمَا أَنتَ بِمُسْمِعٍ مَّن فِي الْقُبُورِ ﴿٢٢﴾
تىرىكلەر بىلەن ئۆلۈكلەر (يەنى مۆمىنلەر بىلەن ھەقىقەتنى رەت قىلغۇچىلار) باراۋەر بولمايدۇ، شۈبھىسىزكى، اﷲ (ھەق دەۋەتنى) خالىغان كىشىلەرگە ئاڭلىتىدۇ. سەن قەبرىلەردىكىلەرگە (يەنى ھەقىقەتنى رەت قىلغۇچىلارغا) ئاڭلىتالمايسەن
إِنْ أَنتَ إِلَّا نَذِيرٌ ﴿٢٣﴾
سەن پەقەت بىر ئاگاھلاندۇرغۇچىسەن
إِنَّا أَرْسَلْنَاكَ بِالْحَقِّ بَشِيرًا وَنَذِيرًا ۚ وَإِن مِّنْ أُمَّةٍ إِلَّا خَلَا فِيهَا نَذِيرٌ ﴿٢٤﴾
ئىشەنچلىك ھالدا بىز سېنى ھەق (دىن) بىلەن (مۆمىنلەرگە) خۇش خەۋەر بەرگۈچى، (ھەقىقەتنى رەت قىلغۇچىلارغا) ئاگاھلاندۇرغۇچى قىلىپ ئەۋەتتۇق، قانداقلا بىر ئۈممەت بولمىسۇن، ئۇنىڭغا خۇدانىڭ ئەلچىسى كەلگەن
وَإِن يُكَذِّبُوكَ فَقَدْ كَذَّبَ الَّذِينَ مِن قَبْلِهِمْ جَاءَتْهُمْ رُسُلُهُم بِالْبَيِّنَاتِ وَبِالزُّبُرِ وَبِالْكِتَابِ الْمُنِيرِ ﴿٢٥﴾
ئۇلار سېنى ئىنكار قىلسا، ئۇلاردىن بۇرۇن ئۆتكەن ئۈممەتلەرمۇ (ئۆزلىرىنىڭ خۇدانىڭ ئەلچىسىلىرىنى) ئىنكار قىلغان، خۇدانىڭ ئەلچىسىلىرى ئۇلارغا روشەن مۆجىزىلەرنى، (ئىبراھىم ئەلەيھىسسالامغا نازىل قىلىنغان سەھىپىلەرگە ئوخشاش) سەھىپىلەرنى، (تەۋرات، ئىنجىلدەك) نۇرلۇق مۇقەددەس كىتاب-ۋەھىيلارنى ئېلىپ كەلدى
ثُمَّ أَخَذْتُ الَّذِينَ كَفَرُوا ۖ فَكَيْفَ كَانَ نَكِيرِ ﴿٢٦﴾
ئاندىن ھەقىقەتنى رەت قىلغۇچىلارنى جازالىدىم، (ئۇلارغا بەرگەن) ئازابىم قانداق ئىكەن؟ (يەنى ئۇلارغا قانداق قاتتىق ئازاب قىلدىم؟)
أَلَمْ تَرَ أَنَّ اللَّهَ أَنزَلَ مِنَ السَّمَاءِ مَاءً فَأَخْرَجْنَا بِهِ ثَمَرَاتٍ مُّخْتَلِفًا أَلْوَانُهَا ۚ وَمِنَ الْجِبَالِ جُدَدٌ بِيضٌ وَحُمْرٌ مُّخْتَلِفٌ أَلْوَانُهَا وَغَرَابِيبُ سُودٌ ﴿٢٧﴾
كۆرمەمسەنكى، اﷲ بۇلۇتتىن يامغۇر ياغدۇردى. ئۇنىڭ بىلەن تۈرلۈك، رەڭگارەڭ مېۋىلەرنى چىقاردى، (شۇنىڭدەك اﷲ تاغلارنى ياراتتى) تاغلارنىڭ ئاق، قىزىللىرى، ھەر خىل رەڭدىكى يوللۇقلىرى ۋە قاپقارىلىرىمۇ بار
وَمِنَ النَّاسِ وَالدَّوَابِّ وَالْأَنْعَامِ مُخْتَلِفٌ أَلْوَانُهُ كَذَٰلِكَ ۗ إِنَّمَا يَخْشَى اللَّهَ مِنْ عِبَادِهِ الْعُلَمَاءُ ۗ إِنَّ اللَّهَ عَزِيزٌ غَفُورٌ ﴿٢٨﴾
شۇنىڭدەك ئىنسانلارنى، ھايۋانلارنى، چاھارپايلارنىمۇ خىلمۇخىل رەڭلىك قىلىپ ياراتتى، اﷲ نىڭ بەندىلىرى ئىچىدە اﷲ تىن پەقەت ئالىملارلا قورقىدۇ، اﷲ ئىشەنچلىك ھالدا غالىبتۇر، (بەندىلىرى ئىچىدە تەۋبە قىلغانلارنى) مەغپىرەت قىلغۇچىدۇر
إِنَّ الَّذِينَ يَتْلُونَ كِتَابَ اللَّهِ وَأَقَامُوا الصَّلَاةَ وَأَنفَقُوا مِمَّا رَزَقْنَاهُمْ سِرًّا وَعَلَانِيَةً يَرْجُونَ تِجَارَةً لَّن تَبُورَ ﴿٢٩﴾
شۈبھىسىزكى، اﷲ نىڭ مۇقەددەس كىتاب-ۋەھىيىنى تىلاۋەت قىلىپ تۇرىدىغانلار، نامازنى ئادا قىلىدىغانلار كاسات بولمايدىغان تىجارەتنى ئۈمىد قىلىدۇ، بىز رىزىق قىلىپ بەرگەن نەرسىلەرنى (اﷲ نىڭ رازىلىقى ئۈچۈن) يوشۇرۇن ۋە ئاشكارا سەرپ قىلىدىغان كىشىلەرگە
لِيُوَفِّيَهُمْ أُجُورَهُمْ وَيَزِيدَهُم مِّن فَضْلِهِ ۚ إِنَّهُ غَفُورٌ شَكُورٌ ﴿٣٠﴾
اﷲ ئەجىرلىرىنى تولۇق بېرىدۇ ۋە مەرھەمىتىدىن ئۇلارغا ئاشۇرۇپ بېرىدۇ، اﷲ ئىشەنچلىك ھالدا ناھايىتى مەغپىرەت قىلغۇچىدۇر، ئاز ياخشىلىققا كۆپ ساۋاب بەرگۈچىدۇر
وَالَّذِي أَوْحَيْنَا إِلَيْكَ مِنَ الْكِتَابِ هُوَ الْحَقُّ مُصَدِّقًا لِّمَا بَيْنَ يَدَيْهِ ۗ إِنَّ اللَّهَ بِعِبَادِهِ لَخَبِيرٌ بَصِيرٌ ﴿٣١﴾
ساڭا بىز ۋەھيى قىلغان مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) ھەقتۇر. ئىلگىرىكى مۇقەددەس كىتاب-ۋەھىيلارنىڭ (راستلىقىنى) ئىسپاتلىغۇچىدۇر، اﷲ ئىشەنچلىك ھالدا بەندىلىرىنى تولۇق بىلگۈچىدۇر، كۆرۈپ تۇرغۇچىدۇر
ثُمَّ أَوْرَثْنَا الْكِتَابَ الَّذِينَ اصْطَفَيْنَا مِنْ عِبَادِنَا ۖ فَمِنْهُمْ ظَالِمٌ لِّنَفْسِهِ وَمِنْهُم مُّقْتَصِدٌ وَمِنْهُمْ سَابِقٌ بِالْخَيْرَاتِ بِإِذْنِ اللَّهِ ۚ ذَٰلِكَ هُوَ الْفَضْلُ الْكَبِيرُ ﴿٣٢﴾
ئاندىن بىز مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) نى بەندىلىرىمىزدىن بىز تاللىغان كىشىلەرگە مىراس قىلىپ بەردۇق. ئۇلارنىڭ بەزىسى ئۆزىگە زۇلۇم قىلغۇچىدۇر (يەنى قۇرئاننى تىلاۋەت قىلغان بىلەن ئەمەل قىلمايدۇ)، بەزىسى ئوتتۇراھالدۇر (يەنى كۆپ چاغلاردا قۇرئانغا ئەمەل قىلىپ، قىسمەن چاغلاردا ئەمەل قىلمايدۇ) ۋە بەزىسى اﷲ نىڭ ئىزنى بويىچە ياخشى ئىشلارنى قىلىشقا ئالدىرىغۇچىدۇر. ئەنە شۇ كاتتا مەرھەمەتتۇر
جَنَّاتُ عَدْنٍ يَدْخُلُونَهَا يُحَلَّوْنَ فِيهَا مِنْ أَسَاوِرَ مِن ذَهَبٍ وَلُؤْلُؤًا ۖ وَلِبَاسُهُمْ فِيهَا حَرِيرٌ ﴿٣٣﴾
ئۇلار ھەمىشە تۇرىدىغان مەڭگۈلۈك باغلارلەرگە كىرىدۇ. ئۇلار مەڭگۈلۈك باغلارلەردە ئالتۇن بىلەيزۈكلەرنى، مەرۋايىتلارنى تاقايدۇ، ئۇلارنىڭ كىيىمى يىپەكتىن بولىدۇ
وَقَالُوا الْحَمْدُ لِلَّهِ الَّذِي أَذْهَبَ عَنَّا الْحَزَنَ ۖ إِنَّ رَبَّنَا لَغَفُورٌ شَكُورٌ ﴿٣٤﴾
ئۇلار ئېيتتى: «جىمى ھەمدۇسانا بىزدىن غەم - قايغۇنى كەتكۈزۈۋەتكەن اﷲ قا مەنسۇپتۇر! بىزنىڭ پەرۋەردىگارىمىز ئىشەنچلىك ھالدا ناھايىتى مەغپىرەت قىلغۇچىدۇر، ئاز ياخشىلىققا كۆپ ساۋاب بەرگۈچىدۇر
الَّذِي أَحَلَّنَا دَارَ الْمُقَامَةِ مِن فَضْلِهِ لَا يَمَسُّنَا فِيهَا نَصَبٌ وَلَا يَمَسُّنَا فِيهَا لُغُوبٌ ﴿٣٥﴾
اﷲ مەرھەمىتىدىن بىزنى مەڭگۈلۈك باغلارتە تۇرغۇزدى. مەڭگۈلۈك باغلارتە بىز ھەرگىز جاپا تارتمايمىز، ھەرگىز چارچىمايمىز»
وَالَّذِينَ كَفَرُوا لَهُمْ نَارُ جَهَنَّمَ لَا يُقْضَىٰ عَلَيْهِمْ فَيَمُوتُوا وَلَا يُخَفَّفُ عَنْهُم مِّنْ عَذَابِهَا ۚ كَذَٰلِكَ نَجْزِي كُلَّ كَفُورٍ ﴿٣٦﴾
ھەقىقەتنى رەت قىلغۇچىلار دوزاخ ئازابىغا دۇچار بولىدۇ، ئۇلارنىڭ ئۆلۈپ (ئارام تېپىپ قالماسلىقى ئۈچۈن) جانلىرى ئېلىنمايدۇ، ئۇلاردىن ئازابمۇ يېنىكلىتىلمەيدۇ، كۇفرىدا ھەددىدىن ئاشقان ھەر ئىنساننى ئەنە شۇنداق جازالايمىز
وَهُمْ يَصْطَرِخُونَ فِيهَا رَبَّنَا أَخْرِجْنَا نَعْمَلْ صَالِحًا غَيْرَ الَّذِي كُنَّا نَعْمَلُ ۚ أَوَلَمْ نُعَمِّرْكُم مَّا يَتَذَكَّرُ فِيهِ مَن تَذَكَّرَ وَجَاءَكُمُ النَّذِيرُ ۖ فَذُوقُوا فَمَا لِلظَّالِمِينَ مِن نَّصِيرٍ ﴿٣٧﴾
ئۇلار دوزاختا: «پەرۋەردىگارىمىز! بىزنى چىقىرىۋەتسەڭ (دۇنيادىكى ۋاقىتتا) قىلغان ئەمەللىرىمىزدىن باشقا ئەمەللەرنى قىلساق» دەپ يالۋۇرۇپ توۋلايدۇ. (اﷲ ئېيتىدۇ) «سىلەرگە ئۆمرۈڭلارنى ئۇزۇن قىلىپ، ۋەز - نەسىھەت ئالىدىغان ئادەم ۋەز - نەسىھەت ئالالىغۇدەك ۋاقىت بەرمىدىممۇ؟ سىلەرگە ئاگاھلاندۇرغۇچى (يەنى خۇدانىڭ ئەلچىسى) كەلدىغۇ، ئەمدى ئازابنى تېتىڭلار، زالىملارغا ھېچقانداق ياردەم بەرگۈچى يوق»
إِنَّ اللَّهَ عَالِمُ غَيْبِ السَّمَاوَاتِ وَالْأَرْضِ ۚ إِنَّهُ عَلِيمٌ بِذَاتِ الصُّدُورِ ﴿٣٨﴾
اﷲ ئىشەنچلىك ھالدا ئاسمانلاردىكى ۋە زېمىندىكى غەيبنى بىلگۈچىدۇر. اﷲ ئىشەنچلىك ھالدا دىللاردىكى سىرلارنى بىلگۈچىدۇر
هُوَ الَّذِي جَعَلَكُمْ خَلَائِفَ فِي الْأَرْضِ ۚ فَمَن كَفَرَ فَعَلَيْهِ كُفْرُهُ ۖ وَلَا يَزِيدُ الْكَافِرِينَ كُفْرُهُمْ عِندَ رَبِّهِمْ إِلَّا مَقْتًا ۖ وَلَا يَزِيدُ الْكَافِرِينَ كُفْرُهُمْ إِلَّا خَسَارًا ﴿٣٩﴾
اﷲ سىلەرنى زېمىندا ئورۇنباسارلار قىلدى، كىمكى ھەقىقەتنى رەت قىلغۇچى بولىدىكەن، ئۇنىڭ كۇفرىنىڭ زىيىنى ئۆزىگە بولىدۇ، ھەقىقەتنى رەت قىلغۇچىلارنىڭ كۇفرى پەرۋەردىگارىنىڭ دەرگاھىدا پەقەت غەزەپنىلا زىيادە قىلىدۇ (يەنى ئۇلارنىڭ كۇفرى ئۇلارنى اﷲ نىڭ رەھمىتىدىن تېخىمۇ يىراقلاشتۇرىدۇ)، ھەقىقەتنى رەت قىلغۇچىلارنىڭ كۇفرى ئۇلارغا پەقەت ھالاكەتنىلا زىيادە قىلىدۇ
قُلْ أَرَأَيْتُمْ شُرَكَاءَكُمُ الَّذِينَ تَدْعُونَ مِن دُونِ اللَّهِ أَرُونِي مَاذَا خَلَقُوا مِنَ الْأَرْضِ أَمْ لَهُمْ شِرْكٌ فِي السَّمَاوَاتِ أَمْ آتَيْنَاهُمْ كِتَابًا فَهُمْ عَلَىٰ بَيِّنَتٍ مِّنْهُ ۚ بَلْ إِن يَعِدُ الظَّالِمُونَ بَعْضُهُم بَعْضًا إِلَّا غُرُورًا ﴿٤٠﴾
ئېيتقىنكى، «ئېيتىپ بېقىڭلارچۇ؟ سىلەر اﷲ نى قويۇپ چوقۇنۇۋاتقان بۇتلىرىڭلار (نېمە بىلەن چوقۇنۇشقا لايىق بولدى؟) ماڭا ئېيتىپ بېرىڭلاركى، ئۇلار زېمىندىن نېمىلەرنى ياراتتى؟ يا ئۇلار ئاسمانلارنى اﷲ بىلەن بىرلىكتە ياراتتىمۇ؟ يا ئۇلارغا بىز بىر مۇقەددەس كىتاب-ۋەھىي بەرگەن بولۇپ، ئۇلار شۇ مۇقەددەس كىتاب-ۋەھىيتىكى روشەن دەلىللەرگە ئاساسلىنامدۇ؟» ھەرگىز ئۇنداق ئەمەس، زالىملار بىر - بىرىگە پەقەت يالغاننىلا ۋەدە قىلىشىدۇ
إِنَّ اللَّهَ يُمْسِكُ السَّمَاوَاتِ وَالْأَرْضَ أَن تَزُولَا ۚ وَلَئِن زَالَتَا إِنْ أَمْسَكَهُمَا مِنْ أَحَدٍ مِّن بَعْدِهِ ۚ إِنَّهُ كَانَ حَلِيمًا غَفُورًا ﴿٤١﴾
اﷲ ئىشەنچلىك ھالدا ئاسمانلارنى ۋە زېمىننى (قۇدرىتى ۋە گۈزەل ھېكمىتى بىلەن) چۈشۈپ كېتىشتىن توختىتىپ تۇرىدۇ. ئەگەر ئۇلار چۈشۈپ كەتسە، ئۇلارنى ھېچ ئەھەدى توختىتىپ تۇرالمايدۇ. اﷲ ئىشەنچلىك ھالدا ھەلىمدۇر (يەنى كۇففارلار ئازابقا تېگىشلىك بولسىمۇ، ئۇلارنى جازالاشقا ئالدىراپ كەتمەيدۇ)، ناھايىتى مەغپىرەت قىلغۇچىدۇر
وَأَقْسَمُوا بِاللَّهِ جَهْدَ أَيْمَانِهِمْ لَئِن جَاءَهُمْ نَذِيرٌ لَّيَكُونُنَّ أَهْدَىٰ مِنْ إِحْدَى الْأُمَمِ ۖ فَلَمَّا جَاءَهُمْ نَذِيرٌ مَّا زَادَهُمْ إِلَّا نُفُورًا ﴿٤٢﴾
ئۇلار ئۆزلىرىگە ئاگاھلاندۇرغۇچى (يەنى خۇدانىڭ ئەلچىسى) كەلسە، ھەر قانداق ئۈممەتكە قارىغاندا ئەڭ ھىدايەت تاپقۇچى بولىدىغانلىقلىرى بىلەن قاتتىق قەسەم ئىچىشتى. ئۇلارغا ئاگاھلاندۇرغۇچى كەلگەن چاغدا (ھىدايەتتىن) تېخىمۇ يىراق بولۇشتى
اسْتِكْبَارًا فِي الْأَرْضِ وَمَكْرَ السَّيِّئِ ۚ وَلَا يَحِيقُ الْمَكْرُ السَّيِّئُ إِلَّا بِأَهْلِهِ ۚ فَهَلْ يَنظُرُونَ إِلَّا سُنَّتَ الْأَوَّلِينَ ۚ فَلَن تَجِدَ لِسُنَّتِ اللَّهِ تَبْدِيلًا ۖ وَلَن تَجِدَ لِسُنَّتِ اللَّهِ تَحْوِيلًا ﴿٤٣﴾
(بۇنداق بولۇشى) ئۇلارنىڭ زېمىندا تەكەببۇرلۇق قىلغانلىقلىرى ۋە ھىيلە - مىكىر ئىشلەتكەنلىكلىرىدىندۇر. ھىيلە - مىكىرنىڭ ۋابالى پەقەت ھىيلە - مىكىر ئىشلەتكەن ئادەمنىڭ ئۆزىگە بولىدۇ، ئۇلار پەقەت ئىلگىرىكى ئۈممەتلەرگە قوللىنىلغان (اﷲ نىڭ ئۇلارنى ھالاك قىلىش ۋە جازالاشتىن ئىبارەت) يولىنىلا كۈتىدۇ، اﷲ نىڭ (مەخلۇقاتلار ئۈستىدە قوللانغان) يولىدا ھەرگىزمۇ ھېچقانداق ئۆزگىرىشنى كۆرمەيسەن، اﷲ يولىدا ھەرگىز يۆتكىلىشنىمۇ (يەنى ئازابنىڭ ئازابقا تېگىشلىك بولغانلار ئۈستىدىن باشقىلارنىڭ ئۈستىگە يۆتكىلىشىنى) كۆرمەيسەن
أَوَلَمْ يَسِيرُوا فِي الْأَرْضِ فَيَنظُرُوا كَيْفَ كَانَ عَاقِبَةُ الَّذِينَ مِن قَبْلِهِمْ وَكَانُوا أَشَدَّ مِنْهُمْ قُوَّةً ۚ وَمَا كَانَ اللَّهُ لِيُعْجِزَهُ مِن شَيْءٍ فِي السَّمَاوَاتِ وَلَا فِي الْأَرْضِ ۚ إِنَّهُ كَانَ عَلِيمًا قَدِيرًا ﴿٤٤﴾
ئۇلار زېمىندا سەير قىلىپ، ئۇلاردىن ئىلگىرىكىلەر (يەنى ھالاك بولغان ئۈممەتلەر) نىڭ ئاقىۋىتىنىڭ قانداق بولغانلىقىنى كۆرمىدىمۇ؟ ئۇلار (يەنى ئۆتكەنكى ئۈممەتلەر) بۇلاردىن تېخىمۇ كۈچلۈك ئىدى. ئاسمانلاردىكى ۋە زېمىندىكى ھېچ نەرسە اﷲ نى ئىلاجسىز قالدۇرالمايدۇ. اﷲ ئىشەنچلىك ھالدا ھەممىنى بىلگۈچىدۇر، ھەممىگە قادىردۇر
وَلَوْ يُؤَاخِذُ اللَّهُ النَّاسَ بِمَا كَسَبُوا مَا تَرَكَ عَلَىٰ ظَهْرِهَا مِن دَابَّةٍ وَلَٰكِن يُؤَخِّرُهُمْ إِلَىٰ أَجَلٍ مُّسَمًّى ۖ فَإِذَا جَاءَ أَجَلُهُمْ فَإِنَّ اللَّهَ كَانَ بِعِبَادِهِ بَصِيرًا ﴿٤٥﴾
ئەگەر اﷲ ئىنسانلارنى ئۇلارنىڭ قىلمىشلىرى تۈپەيلىدىن جازالايدىغان بولسا، يەر يۈزىدە ھېچبىر جان ئىگىسىنى قويمىغان بولاتتى ۋە لېكىن اﷲ ئۇلار (دىن ھېساب ئېلىش) نى مۇئەييەن ۋاقىتقىچە تەخىر قىلىدۇ، ئۇلار (دىن ھېساب ئېلىش) نىڭ ۋاقتى يېتىپ كەلگەندە (ئۇلارنىڭ ئەمەللىرىگە قاراپ جازا بېرىدۇ)، اﷲ ئىشەنچلىك ھالدا بەندىلىرىنى كۆرۈپ تۇرغۇچىدۇر
