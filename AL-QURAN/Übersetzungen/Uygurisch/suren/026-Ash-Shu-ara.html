بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
✦ ✦
طسم ﴿١﴾
طا، سىم، مىم
تِلْكَ آيَاتُ الْكِتَابِ الْمُبِينِ ﴿٢﴾
بۇ، روشەن مۇقەددەس كىتاب-ۋەھىي (يەنى قۇرئان) ئايەتلىرىدۇر
لَعَلَّكَ بَاخِعٌ نَّفْسَكَ أَلَّا يَكُونُوا مُؤْمِنِينَ ﴿٣﴾
(ئى مۇھەممەد!) ئۇلارنىڭ ئىمان ئېيتمىغانلىقىدىن ئۆزۈڭنى ھالاك قىلىۋېتىشىڭ مۇمكىن
إِن نَّشَأْ نُنَزِّلْ عَلَيْهِم مِّنَ السَّمَاءِ آيَةً فَظَلَّتْ أَعْنَاقُهُمْ لَهَا خَاضِعِينَ ﴿٤﴾
ئەگەر بىز ئۇلارنىڭ (ئىمان ئېيتىشىنى) خالايدىغان بولساق، ئۇلارغا ئاسماندىن (ئۇلارنى ئىمانغا مەجبۇرلايدىغان) بىر ئايەتنى (ئالامەتنى) نازىل قىلاتتۇقتە، ئۇنىڭغا ئۇلار باش ئەگكەن بولاتتى
وَمَا يَأْتِيهِم مِّن ذِكْرٍ مِّنَ الرَّحْمَٰنِ مُحْدَثٍ إِلَّا كَانُوا عَنْهُ مُعْرِضِينَ ﴿٥﴾
ئۇلارغا مەرھەمەتلىك اﷲ تەرىپىدىن قۇرئاندىن يېڭىدىن بىرنەرسە نازىل بولسىلا، ئۇلار ئۇنىڭدىن يۈز ئۆرۈيدۇ
فَقَدْ كَذَّبُوا فَسَيَأْتِيهِمْ أَنبَاءُ مَا كَانُوا بِهِ يَسْتَهْزِئُونَ ﴿٦﴾
ئۇلار راستلا (قۇرئاننى) ئىنكار قىلدى، مەسخىرە قىلغان نەرسىنىڭ خەۋەرلىرى (يەنى ئاقىۋىتى) ئۇلارغا كېلىدۇ
أَوَلَمْ يَرَوْا إِلَى الْأَرْضِ كَمْ أَنبَتْنَا فِيهَا مِن كُلِّ زَوْجٍ كَرِيمٍ ﴿٧﴾
ئۇلار زېمىننى (يەنى زېمىننىڭ ئاجايىباتلىرىنى) كۈزەتمىدىمۇ؟ زېمىندا تۈرلۈك پايدىلىق ئۆسۈملۈكلەرنى ئۆستۈردۇق
إِنَّ فِي ذَٰلِكَ لَآيَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿٨﴾
بۇنىڭدا ئەلۋەتتە (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) ئالامەت بار، ئۇلارنىڭ تولىسى ئىمان ئېيتقۇچى بولمىدى
وَإِنَّ رَبَّكَ لَهُوَ الْعَزِيزُ الرَّحِيمُ ﴿٩﴾
سېنىڭ پەرۋەردىگارىڭ ئىشەنچلىك ھالدا غالىبتۇر، ناھايىتى مېھرىباندۇر
وَإِذْ نَادَىٰ رَبُّكَ مُوسَىٰ أَنِ ائْتِ الْقَوْمَ الظَّالِمِينَ ﴿١٠﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭ مۇساغا نىدا قىلدى: «سەن زالىم قەۋمگە بارغىن
قَوْمَ فِرْعَوْنَ ۚ أَلَا يَتَّقُونَ ﴿١١﴾
(ئۇلار) پىرئەۋننىڭ قەۋمىدۇر، ئۇلار (اﷲ نىڭ جازالىشىدىن) قورقمامدۇ؟»
قَالَ رَبِّ إِنِّي أَخَافُ أَن يُكَذِّبُونِ ﴿١٢﴾
مۇسا ئېيتتى: «پەرۋەردىگارىم! مەن ئىشەنچلىك ھالدا ئۇلارنىڭ مېنى ئىنكار قىلىشىدىن قورقىمەن
وَيَضِيقُ صَدْرِي وَلَا يَنطَلِقُ لِسَانِي فَأَرْسِلْ إِلَىٰ هَارُونَ ﴿١٣﴾
(ئۇلارنىڭ مېنى ئىنكار قىلىشىدىن) يۈرىكىم سىقىلىدۇ، تىلىم كېكەچ، شۇڭا (ماڭا ياردەمدە بولۇش ئۈچۈن) ھارۇننى (خۇدانىڭ ئەلچىسى) قىلىپ ئەۋەتكىن
وَلَهُمْ عَلَيَّ ذَنبٌ فَأَخَافُ أَن يَقْتُلُونِ ﴿١٤﴾
(ئۇلارنىڭ دەۋاسىچە) ئۇلارنىڭ ئالدىدا مېنىڭ گۇناھىم بار، ئۇلارنىڭ مېنى ئۆلتۈرۈشىدىن قورقىمەن»
قَالَ كَلَّا ۖ فَاذْهَبَا بِآيَاتِنَا ۖ إِنَّا مَعَكُم مُّسْتَمِعُونَ ﴿١٥﴾
اﷲ ئېيتتى: «ھەرگىز ئۇنداق ئەمەس (يەنى سېنى ئۇلار ھەرگىز ئۆلتۈرەلمەيدۇ)، سىلەر مېنىڭ مۆجىزىلىرىمنى ئېلىپ بېرىڭلار، بىز ئىشەنچلىك ھالدا سىلەر بىلەن بىللە (مۇنازىرەڭلارنى) ئاڭلاپ تۇرىمىز
فَأْتِيَا فِرْعَوْنَ فَقُولَا إِنَّا رَسُولُ رَبِّ الْعَالَمِينَ ﴿١٦﴾
سىلەر پىرئەۋنگە بېرىپ: بىز ئىشەنچلىك ھالدا ئالەملەرنىڭ پەرۋەردىگارىنىڭ ئەلچىسىمىز
أَنْ أَرْسِلْ مَعَنَا بَنِي إِسْرَائِيلَ ﴿١٧﴾
ئىسرائىل ئەۋلادىنى (قۇللۇقۇڭدىن) بوشاتقىن، ئۇلار بىز بىلەن (شامغا) كەتسۇن دەڭلار»
قَالَ أَلَمْ نُرَبِّكَ فِينَا وَلِيدً�� وَلَبِثْتَ فِينَا مِنْ عُمُرِكَ سِنِينَ ﴿١٨﴾
(ئۇلار پىرئەۋنگە كېلىپ ئەلچىلىكنى يەتكۈزدى) پىرئەۋن (مۇساغا) ئېيتتى: «سېنى بىز ئۆيىمىزدە كىچىكىڭدىن تەربىيلەپ چوڭ قىلمىدۇقمۇ؟ ئارىمىزدا كۆپ يىللار تۇرمىدىڭمۇ؟
وَفَعَلْتَ فَعْلَتَكَ الَّتِي فَعَلْتَ وَأَنتَ مِنَ الْكَافِرِينَ ﴿١٩﴾
سەن ھېلىقى قىلغان ئىشىڭنى قىلمىدىڭمۇ؟ (يەنى قىبتىنى ئۆلتۈرمىدىڭمۇ؟) سەن تۇزكورلاردىنسەن»
قَالَ فَعَلْتُهَا إِذًا وَأَنَا مِنَ الضَّالِّينَ ﴿٢٠﴾
مۇسا ئېيتتى: «مەن ئۇ ئىشنى قىلغان چېغىمدا نادانلاردىن ئىدىم
فَفَرَرْتُ مِنكُمْ لَمَّا خِفْتُكُمْ فَوَهَبَ لِي رَبِّي حُكْمًا وَجَعَلَنِي مِنَ الْمُرْسَلِينَ ﴿٢١﴾
سىلەردىن قورقۇپ، سىلەردىن قاچتىم، پەرۋەردىگارىم ماڭا ھېكمەت ئاتا قىلدى، مېنى خۇدانىڭ ئەلچىسى قىلدى
وَتِلْكَ نِعْمَةٌ تَمُنُّهَا عَلَيَّ أَنْ عَبَّدتَّ بَنِي إِسْرَائِيلَ ﴿٢٢﴾
سەن ئىسرائىل ئەۋلادىنى قۇل قىلدىڭ، سېنىڭ ماڭا مىننەت قىلغان نېمىتىڭ ئەنە شۇدۇر»
قَالَ فِرْعَوْنُ وَمَا رَبُّ الْعَالَمِينَ ﴿٢٣﴾
پىرئەۋن ئېيتتى: «ئالەملەرنىڭ پەرۋەردىگارى دېگەن نېمە؟»
قَالَ رَبُّ السَّمَاوَاتِ وَالْأَرْضِ وَمَا بَيْنَهُمَا ۖ إِن كُنتُم مُّوقِنِينَ ﴿٢٤﴾
مۇسا ئېيتتى: «ئۇ ئاسمانلارنىڭ، زېمىننىڭ ۋە ئۇلارنىڭ ئارىسىدىكى پۈتۈن مەخلۇقاتلارنىڭ پەرۋەردىگارىدۇر. ئەگەر سىلەر ھەقىقىي ئىشىنىدىغان بولساڭلار»
قَالَ لِمَنْ حَوْلَهُ أَلَا تَسْتَمِعُونَ ﴿٢٥﴾
پىرئەۋن چۆرىسىدىكىلەرگە: «(ئۇنىڭ جاۋابىنى) ئاڭلاۋاتامسىلەر؟» دېدى
قَالَ رَبُّكُمْ وَرَبُّ آبَائِكُمُ الْأَوَّلِينَ ﴿٢٦﴾
مۇسا ئيتتى: «(ئۇ) سىلەرنىڭ پەرۋەردىگارىڭلاردۇر ۋە سىلەرنىڭ ئاتا - بوۋاڭلارنىڭ پەرۋەردىگارىدۇر»
قَالَ إِنَّ رَسُولَكُمُ الَّذِي أُرْسِلَ إِلَيْكُمْ لَمَجْنُونٌ ﴿٢٧﴾
پىرئەۋن ئېيتتى: «سىلەرگە ئەۋەتىلگەن (بۇ) ئەلچى ئەلۋەتتە مەجنۇندۇر»
قَالَ رَبُّ الْمَشْرِقِ وَالْمَغْرِبِ وَمَا بَيْنَهُمَا ۖ إِن كُنتُمْ تَعْقِلُونَ ﴿٢٨﴾
مۇسا ئېيتتى: «(ئۇ) مەشرىقنىڭ، مەغرىبنىڭ ۋە ئۇلارنىڭ ئارىسىدىكى مەخلۇقاتنىڭ پەرۋەردىگارىدۇر. ئەگەر سىلەر چۈشىنىدىغان بولساڭلار»
قَالَ لَئِنِ اتَّخَذْتَ إِلَٰهًا غَيْرِي لَأَجْعَلَنَّكَ مِنَ الْمَسْجُونِينَ ﴿٢٩﴾
پىرئەۋن ئېيتتى: «ئەگەر مەندىن غەيرىنى ئىلاھ قىلىۋالىدىغان بولساڭ، سېنى چوقۇم زىندانغا تاشلايمەن»
قَالَ أَوَلَوْ جِئْتُكَ بِشَيْءٍ مُّبِينٍ ﴿٣٠﴾
مۇسا ئېيتتى: «(مېنىڭ راست خۇدانىڭ ئەلچىسى ئىكەنلىكىمنى ئىسپاتلايدىغان) روشەن دەلىل كەلتۈرسەممۇ (زىندانغا تاشلامسەن؟)»
قَالَ فَأْتِ بِهِ إِن كُنتَ مِنَ الصَّادِقِينَ ﴿٣١﴾
پىرئەۋن ئېيتتى: «ئەگەر سۆزۈڭ راست بولسا روشەن دەلىلىڭنى كەلتۈرگىن»
فَأَلْقَىٰ عَصَاهُ فَإِذَا هِيَ ثُعْبَانٌ مُّبِينٌ ﴿٣٢﴾
مۇسا ھاسىسىنى تاشلىۋىدى، ناگاھان ئوپئوچۇق ئەجدىھاغا ئايلاندى
وَنَزَعَ يَدَهُ فَإِذَا هِيَ بَيْضَاءُ لِلنَّاظِرِينَ ﴿٣٣﴾
مۇسا (قوينىدىن) قولىنى چىقىرىۋىدى، ناگاھان ئۇ قارىغۇچىلارغا (نۇر چاقناپ تۇرىدىغان) ئاپئاق بولۇپ كۆرۈندى
قَالَ لِلْمَلَإِ حَوْلَهُ إِنَّ هَٰذَا لَسَاحِرٌ عَلِيمٌ ﴿٣٤﴾
پىرئەۋن چۆرىسىدىكى (قەۋمنىڭ) چوڭلىرىغا ئېيتتى: «بۇ ئىشەنچلىك ھالدا ئۇستا سېھىرگەر ئىكەن، ئۇ سېھىر ئارقىلىق سىلەرنى زېمىنىڭلاردىن ھەيدەپ چىقارماقچى، (بۇ ھەقتە) نېمە مەسلىھەت بېرىسىلەر؟»
يُرِيدُ أَن يُخْرِجَكُم مِّنْ أَرْضِكُم بِسِحْرِهِ فَمَاذَا تَأْمُرُونَ ﴿٣٥﴾
پىرئەۋن چۆرىسىدىكى (قەۋمنىڭ) چوڭلىرىغا ئېيتتى: «بۇ ئىشەنچلىك ھالدا ئۇستا سېھىرگەر ئىكەن، ئۇ سېھىر ئارقىلىق سىلەرنى زېمىنىڭلاردىن ھەيدەپ چىقارماقچى، (بۇ ھەقتە) نېمە مەسلىھەت بېرىسىلەر؟»
قَالُوا أَرْجِهْ وَأَخَاهُ وَابْعَثْ فِي الْمَدَائِنِ حَاشِرِينَ ﴿٣٦﴾
ئۇلار ئېيتتى: «(ئۇلارنىڭ ئىشىنى) تەخىر قىلغىن، شەھەرلەرگە (سېھىرگەرلەرنى) يىققۇچى كىشىلەرنى ئەۋەتكىن
يَأْتُوكَ بِكُلِّ سَحَّارٍ عَلِيمٍ ﴿٣٧﴾
ئۇلار سېنىڭ ھوزۇرۇڭغا ناھايىتى ئۇستا سېھىرگەرلەرنىڭ ھەممىسىنى ئېلىپ كەلسۇن»
فَجُمِعَ السَّحَرَةُ لِمِيقَاتِ يَوْمٍ مَّعْلُومٍ ﴿٣٨﴾
مۇئەييەن كۈندىكى بەلگىلەنگەن ۋاقىتتا سېھىرگەرلەر توپلاندى
وَقِيلَ لِلنَّاسِ هَلْ أَنتُم مُّجْتَمِعُونَ ﴿٣٩﴾
كىشىلەرگە: «سىلەر يىغىلىپ بولدۇڭلارمۇ؟ ئەگەر سېھىرگەرلەر غالىب چىقسا ئۇلارغا بويسۇنۇشىمىز مۇمكىن» دېيىلدى
لَعَلَّنَا نَتَّبِعُ السَّحَرَةَ إِن كَانُوا هُمُ الْغَالِبِينَ ﴿٤٠﴾
كىشىلەرگە: «سىلەر يىغىلىپ بولدۇڭلارمۇ؟ ئەگەر سېھىرگەرلەر غالىب چىقسا ئۇلارغا بويسۇنۇشىمىز مۇمكىن» دېيىلدى
فَلَمَّا جَاءَ السَّحَرَةُ قَالُوا لِفِرْعَوْنَ أَئِنَّ لَنَا لَأَجْرًا إِن كُنَّا نَحْنُ الْغَالِبِينَ ﴿٤١﴾
سېھىرگەرلەر پىرئەۋننىڭ قېشىغا يېتىپ كەلگەندە، ئۇلار پىرئەۋنگە: «ئەگەر غەلىبە قىلساق بىزگە چوقۇم مۇكاپات بېرىلەمدۇ؟» دېدى
قَالَ نَعَمْ وَإِنَّكُمْ إِذًا لَّمِنَ الْمُقَرَّبِينَ ﴿٤٢﴾
پىرئەۋن: «ھەئە، ئۇ چاغدا (سىلەرگە مۇكاپات بېرىلىدۇ، ئۇنىڭ ئۈستىگە) سىلەر چوقۇم مېنىڭ يېقىن ئادەملىرىمدىن بولۇپ قالىسىلەر» دېدى
قَالَ لَهُم مُّوسَىٰ أَلْقُوا مَا أَنتُم مُّلْقُونَ ﴿٤٣﴾
مۇسا ئۇلارغا: «تاشلايدىغان نەرسەڭلارنى تاشلاڭلار!» دېدى
فَأَلْقَوْا حِبَالَهُمْ وَعِصِيَّهُمْ وَقَالُوا بِعِزَّةِ فِرْعَوْنَ إِنَّا لَنَحْنُ الْغَالِبُونَ ﴿٤٤﴾
ئۇلار ئارغامچىلىرىنى، ھاسسىلىرىنى تاشلىدى. ئۇلار: «پىرئەۋننىڭ كاتتىلىقى بىلەن قەسەمكى، بىز شەك شۈبھىسىز غەلىبە قىلىمىز» دېدى
فَأَلْقَىٰ مُوسَىٰ عَصَاهُ فَإِذَا هِيَ تَلْقَفُ مَا يَأْفِكُونَ ﴿٤٥﴾
مۇسا ھاسىسىنى تاشلىدى، (ئۇ ئەجدىھاغا ئايلىىنىپ) ئۇلارنىڭ ئويدۇرما نەرسىلىرىنى يۈتۈۋەتتى
فَأُلْقِيَ السَّحَرَةُ سَاجِدِينَ ﴿٤٦﴾

إِذْ نُسَوِّيكُم بِرَبِّ الْعَالَمِينَ ﴿٩٨﴾
ئۆز ۋاقتىدا بىز سىلەرنى (ئىبادەتتە) ئالەملەرنىڭ پەرۋەردىگارى بىلەن ئوخشاش ئورۇندا قوياتتۇق
وَمَا أَضَلَّنَا إِلَّا الْمُجْرِمُونَ ﴿٩٩﴾
بىزنى پەقەت گۇناھكارلارلا ئازدۇردى
فَمَا لَنَا مِن شَافِعِينَ ﴿١٠٠﴾
شۇنىڭ ئۈچۈن بىزگە شاپائەت قىلغۇچىلار يوق
وَلَا صَدِيقٍ حَمِيمٍ ﴿١٠١﴾
ۋە يېقىن دوستىمۇ يوق
فَلَوْ أَنَّ لَنَا كَرَّةً فَنَكُونَ مِنَ الْمُؤْمِنِينَ ﴿١٠٢﴾
ئەگەر بىزگە (دۇنياغا) قايتىشقا بولسا ئىدى، بىز مۆمىنلەردىن بولاتتۇق»
إِنَّ فِي ذَٰلِكَ لَآيَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿١٠٣﴾
شۈبھىسىزكى، بۇنىڭدا (يەنى ئىبراھىم ۋە ئۇنىڭ قەۋمىنىڭ قىسسىسىدە) (ئەقىل ئىگىلىرى ئۈچۈن) ئەلۋەتتە (چوڭ) ئىبرەت بار، ئۇلارنىڭ تولىسى ئىمان ئېيتقۇچى بولمىدى
وَإِنَّ رَبَّكَ لَهُوَ الْعَزِيزُ الرَّحِيمُ ﴿١٠٤﴾
سېنىڭ پەرۋەردىگارىڭ ئىشەنچلىك ھالدا ناھايىتى غالىبتۇر، ناھايىتى مېھرىباندۇر
كَذَّبَتْ قَوْمُ نُوحٍ الْمُرْسَلِينَ ﴿١٠٥﴾
نۇھنىڭ قەۋمى خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلدى
إِذْ قَالَ لَهُمْ أَخُوهُمْ نُوحٌ أَلَا تَتَّقُونَ ﴿١٠٦﴾
ئۆز ۋاقتىدا ئۇلارغا قېرىندىشى نۇھ ئېيتتى: «سىلەر (اﷲ تىن) قورقمامسىلەر؟
إِنِّي لَكُمْ رَسُولٌ أَمِينٌ ﴿١٠٧﴾
مەن ئىشەنچلىك ھالدا سىلەرگە سادىق بىر خۇدانىڭ ئەلچىسىمەن
فَاتَّقُوا اللَّهَ وَأَطِيعُونِ ﴿١٠٨﴾
سىلەر اﷲ تىن قورقۇڭلار، ماڭا ئىتائەت قىلىڭلار
وَمَا أَسْأَلُكُمْ عَلَيْهِ مِنْ أَجْرٍ ۖ إِنْ أَجْرِيَ إِلَّا عَلَىٰ رَبِّ الْعَالَمِينَ ﴿١٠٩﴾
تەبلىغ قىلغانلىقىمغا سىلەردىن ھېچقانداق ھەق تەلەپ قىلمايمەن، ئۇنى پەقەت ئالەملەرنىڭ پەرۋەردىگارىدىن تىلەيمەن
فَاتَّقُوا اللَّهَ وَأَطِيعُونِ ﴿١١٠﴾
اﷲ تىن قورقۇڭلار، ماڭا ئىتائەت قىلىڭلار»
قَالُوا أَنُؤْمِنُ لَكَ وَاتَّبَعَكَ الْأَرْذَلُونَ ﴿١١١﴾
ئۇلار: «(ئى نۇھ!) ساڭا تۆۋەن تەبىقىدىكى ئادەملەر ئەگەشكەن تۇرسا، ساڭا بىز ئىمان ئېيتامدۇق؟» دېدى
قَالَ وَمَا عِلْمِي بِمَا كَانُوا يَعْمَلُونَ ﴿١١٢﴾
نۇھ ئېيتتى: «مەن ئۇلارنىڭ نېمە قىلغانلىقىنى ئۇقمايمەن
إِنْ حِسَابُهُمْ إِلَّا عَلَىٰ رَبِّي ۖ لَوْ تَشْعُرُونَ ﴿١١٣﴾
ئەگەر بىلسەڭلار ئۇلاردىن ھېساب ئېلىشنى پەقەت پەرۋەردىگارىم ئۈستىگە ئالغان
وَمَا أَنَا بِطَارِدِ الْمُؤْمِنِينَ ﴿١١٤﴾
مەن مۆمىنلەرنى قوغلىۋەتمەيمەن
إِنْ أَنَا إِلَّا نَذِيرٌ مُّبِينٌ ﴿١١٥﴾
مەن پەقەت ئوچۇق - ئاشكارا ئاگاھلاندۇرغۇچىمەن»
قَالُوا لَئِن لَّمْ تَنتَهِ يَا نُوحُ لَتَكُونَنَّ مِنَ الْمَرْجُومِينَ ﴿١١٦﴾
ئۇلار ئېيتتى: «ئى نۇھ! ئەگەر سەن (خۇدانىڭ ئەلچىسىلىك دەۋەتىڭدىن) يانمىساڭ، سەن چوقۇم تاش - كېسەك قىلىنىپ ئۆلتۈرۈلگۈچىلەردىن بولىسەن»
قَالَ رَبِّ إِنَّ قَوْمِي كَذَّبُونِ ﴿١١٧﴾
نۇھ ئېيتتى: «پەرۋەردىگارىم! قەۋمىم مېنى ئىشەنچلىك ھالدا ئىنكار قىلدى
فَافْتَحْ بَيْنِي وَبَيْنَهُمْ فَتْحًا وَنَجِّنِي وَمَن مَّعِيَ مِنَ الْمُؤْمِنِينَ ﴿١١٨﴾
مەن بىلەن ئۇلارنىڭ ئارىسىدا ھۆكۈم چىقارغىن، مېنى ۋە مەن بىلەن بىللە بولغان مۆمىنلەرنى قۇتقۇزغىن»
فَأَنجَيْنَاهُ وَمَن مَّعَهُ فِي الْفُلْكِ الْمَشْحُونِ ﴿١١٩﴾
بىز ئۇنى ۋە (ئادەملەر، ھايۋانلار بىلەن) لىق تولغان كېمىدە ئۇنىڭ بىلەن بىللە بولغانلارنى قۇتقۇزدۇق
ثُمَّ أَغْرَقْنَا بَعْدُ الْبَاقِينَ ﴿١٢٠﴾
(ئۇلارنى قۇتقۇزغاندىن) كېيىن ئاندىن قالغانلارنى غەرق قىلدۇق
إِنَّ فِي ذَٰلِكَ لَآيَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿١٢١﴾
بۇنىڭدا (تەپەككۇر قىلغۇچىلار ئۈچۈن) ئەلۋەتتە (چوڭ) ئىبرەت بار، ئۇلار (يەنى كىشىلەر) نىڭ تولىسى ئىمان ئېيتقۇچى بولمىدى
وَإِنَّ رَبَّكَ لَهُوَ الْعَزِيزُ الرَّحِيمُ ﴿١٢٢﴾
سېنىڭ پەرۋەردىگارىڭ ئىشەنچلىك ھالدا ناھايىتى غالىبتۇر، ناھايىتى مېھرىباندۇر
كَذَّبَتْ عَادٌ الْمُرْسَلِينَ ﴿١٢٣﴾
ئاد (خەلقى) خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلدى
إِذْ قَالَ لَهُمْ أَخُوهُمْ هُودٌ أَلَا تَتَّقُونَ ﴿١٢٤﴾
ئۆز ۋاقتىدا ئۇلارغا قېرىندىشى ھۇد ئېيتتى: «سىلەر (اﷲ تىن) قورقمامسىلەر؟
إِنِّي لَكُمْ رَسُولٌ أَمِينٌ ﴿١٢٥﴾
مەن ئىشەنچلىك ھالدا سىلەرگە سادىق خۇدانىڭ ئەلچىسىمەن
فَاتَّقُوا اللَّهَ وَأَطِيعُونِ ﴿١٢٦﴾
سىلەر اﷲ تىن قورقۇڭلار، ماڭا ئىتائەت قىلىڭلار
وَمَا أَسْأَلُكُمْ عَلَيْهِ مِنْ أَجْرٍ ۖ إِنْ أَجْرِيَ إِلَّا عَلَىٰ رَبِّ الْعَالَمِينَ ﴿١٢٧﴾
تەبلىغ قىلغانلىقىمغا سىلەردىن ھېچقانداق ھەق تەلەپ قىلمايمەن، ئۇنى پەقەت ئالەملەرنىڭ پەرۋەردىگارىدىن تىلەيمەن
أَتَبْنُونَ بِكُلِّ رِيعٍ آيَةً تَعْبَثُونَ ﴿١٢٨﴾
سىلەر ھەر بىر ئېگىز جايغا ئويۇن - كۈلكە ئۈچۈن بىر ئالامەت (يەنى ئېگىز بىنا) سالامسىلەر
وَتَتَّخِذُونَ مَصَانِعَ لَعَلَّكُمْ تَخْلُدُونَ ﴿١٢٩﴾
(گويا سىلەر ئۆلمەيدىغاندەك) دۇنيادا مەڭگۈ قېلىشنى ئۈمىد قىلىپ پۇختا سارايلارنى سالامسىلەر
وَإِذَا بَطَشْتُم بَطَشْتُمْ جَبَّارِينَ ﴿١٣٠﴾
ئەگەر (بىراۋنى) جازالىساڭلار، زالىملاردەك جازالايسىلەر
فَاتَّقُوا اللَّهَ وَأَطِيعُونِ ﴿١٣١﴾
سىلەر اﷲ تىن قورقۇڭلار، ماڭا ئىتائەت قىلىڭلار
وَاتَّقُوا الَّذِي أَمَدَّكُم بِمَا تَعْلَمُونَ ﴿١٣٢﴾
سىلەرگە سىلەر بىلىدىغان نېمەتلەرنى ئاتا قىلغان اﷲ تىن قورقۇڭلار
أَمَدَّكُم بِأَنْعَامٍ وَبَنِينَ ﴿١٣٣﴾
ئۇ سىلەرگە چارۋىلارنى، ئوغۇللارنى، باغلارنى، بۇلاقلارنى ئاتا قىلدى
وَجَنَّاتٍ وَعُيُونٍ ﴿١٣٤﴾
ئۇ سىلەرگە چارۋىلارنى، ئوغۇللارنى، باغلارنى، بۇلاقلارنى ئاتا قىلدى
إِنِّي أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ عَظِيمٍ ﴿١٣٥﴾
مەن ئىشەنچلىك ھالدا سىلەرنىڭ بۈيۈك كۈننىڭ ئازابىغا قېلىشىڭلاردىن قورقىمەن»
قَالُوا سَوَاءٌ عَلَيْنَا أَوَعَظْتَ أَمْ لَمْ تَكُن مِّنَ الْوَاعِظِينَ ﴿١٣٦﴾
ئۇلار ئېيتتى: «نەسىھەت قىلامسەن، قىلمامسەن، بەرىبىر بىزگە ئوخشاش
إِنْ هَٰذَا إِلَّا خُلُقُ الْأَوَّلِينَ ﴿١٣٧﴾
بۇ (يەنى بۇتلارغا چوقۇنۇش) پەقەت بۇرۇنقىلاردىن قالغان ئادەت
وَمَا نَحْنُ بِمُعَذَّبِينَ ﴿١٣٨﴾
بىز ھەرگىز جازالانمايمىز»
فَكَذَّبُوهُ فَأَهْلَكْنَاهُمْ ۗ إِنَّ فِي ذَٰلِكَ لَآيَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿١٣٩﴾
ئۇلار ئۇنى (يەنى ھۇدنى) ئىنكار قىلدى، ئۇلارنى بىز ھالاك قىلدۇق، بۇنىڭدا ئەلۋەتتە (چوڭ) ئىبرەت بار، ئۇلارنىڭ تولىسى ئىمان ئېيتقۇچى بولمىدى
وَإِنَّ رَبَّكَ لَهُوَ الْعَزِيزُ الرَّحِيمُ ﴿١٤٠﴾
سېنىڭ پەرۋەردىگارىڭ ئىشەنچلىك ھالدا ناھايىتى غالىبتۇر، ناھايىتى مېھرىباندۇر
كَذَّبَتْ ثَمُودُ الْمُرْسَلِينَ ﴿١٤١﴾
سەمۇد (خەلقى) خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلدى
إِذْ قَالَ لَهُمْ أَخُوهُمْ صَالِحٌ أَلَا تَتَّقُونَ ﴿١٤٢﴾
ئۆز ۋاقتىدا ئۇلارغا قېرىندىشى سالىھ ئېيتتى: «(اﷲ تىن) قورقمامسىلەر؟
إِنِّي لَكُمْ رَسُولٌ أَمِينٌ ﴿١٤٣﴾
مەن ئىشەنچلىك ھالدا سىلەرگە سادىق خۇدانىڭ ئەلچىسىمەن
فَاتَّقُوا اللَّهَ وَأَطِيعُونِ ﴿١٤٤﴾
سىلەر اﷲ تىن قورقۇڭلار، ماڭا ئىتائەت قىلىڭلار
وَمَا أَسْأَلُكُمْ عَلَيْهِ مِنْ أَجْرٍ ۖ إِنْ أَجْرِيَ إِلَّا عَلَىٰ رَبِّ الْعَالَمِينَ ﴿١٤٥﴾
تەبلىغ قىلغانلىقىمغا سىلەردىن ھېچقانداق ھەق تەلەپ قىلمايمەن، ئۇنى پەقەت ئالەملەرنىڭ پەرۋەردىگارىدىن تىلەيمەن
أَتُتْرَكُونَ فِي مَا هَاهُنَا آمِنِينَ ﴿١٤٦﴾
سىلەر بۇ يەرلەردە، باغچىلاردىن، بۇلاقلاردىن، زىرائەتلەردىن، يۇمشاق پىشقان خورمىلاردىن بەھرىمەن بولۇپ خاتىرجەم ھالدا داۋاملىق قالىمىز (دەپ ئويلامسىلەر)
فِي جَنَّاتٍ وَعُيُونٍ ﴿١٤٧﴾
سىلەر بۇ يەرلەردە، باغچىلاردىن، بۇلاقلاردىن، زىرائەتلەردىن، يۇمشاق پىشقان خورمىلاردىن بەھرىمەن بولۇپ خاتىرجەم ھالدا داۋاملىق قالىمىز (دەپ ئويلامسىلەر)
وَزُرُوعٍ وَنَخْلٍ طَلْعُهَا هَضِيمٌ ﴿١٤٨﴾
سىلەر بۇ يەرلەردە، باغچىلاردىن، بۇلاقلاردىن، زىرائەتلەردىن، يۇمشاق پىشقان خورمىلاردىن بەھرىمەن بولۇپ خاتىرجەم ھالدا داۋاملىق قالىمىز (دەپ ئويلامسىلەر)
وَتَنْحِتُونَ مِنَ الْجِبَالِ بُيُوتًا فَارِهِينَ ﴿١٤٩﴾
خۇشال ھالدا تاغلارنى تېشىپ ئۆي ياسامسىلەر
فَاتَّقُوا اللَّهَ وَأَطِيعُونِ ﴿١٥٠﴾
اﷲ تىن قورقۇڭلار، ماڭا ئىتائەت قىلىڭلار
وَلَا تُطِيعُوا أَمْرَ الْمُسْرِفِينَ ﴿١٥١﴾
زېمىندا بۇزغۇنچىلىق قىلىدىغان، ئىسلاھ قىلمايدىغان ھەددىدىن ئاشقۇچىلارنىڭ ئەمرىگە ئىتائەت قىلماڭلار»
الَّذِينَ يُفْسِدُونَ فِي الْأَرْضِ وَلَا يُصْلِحُونَ ﴿١٥٢﴾
زېمىندا بۇزغۇنچىلىق قىلىدىغان، ئىسلاھ قىلمايدىغان ھەددىدىن ئاشقۇچىلارنىڭ ئەمرىگە ئىتائەت قىلماڭلار»
قَالُوا إِنَّمَا أَنتَ مِنَ الْمُسَحَّرِينَ ﴿١٥٣﴾
ئۇلار ئېيتتى: «سەن ئىشەنچلىك ھالدا سېھىر قىلىنغانلاردىنسەن
مَا أَنتَ إِلَّا بَشَرٌ مِّثْلُنَا فَأْتِ بِآيَةٍ إِن كُنتَ مِنَ الصَّادِقِينَ ﴿١٥٤﴾
سەن پەقەت بىزگە ئوخشاش بىر ئىنسانسەن، ئەگەر سەن راستچىللاردىن بولساڭ، بىرەر مۆجىزە كەلتۈرۈپ باققىن»

مَا أَغْنَىٰ عَنْهُم مَّا كَانُوا يُمَتَّعُونَ ﴿٢٠٧﴾
ئېيتىپ باققىنا! ئەگەر ئۇلارنى (نۇرغۇن) يىللار (نېمەتلىرىمىزدىن) بەھرىمەن قىلساق، ئاندىن ئۇلارغا ئاگاھلاندۇرۇلغان ئازاب كەلسە، ئۇلارنىڭ بەھرىمەن بولغان نېمەتلىرى (ئۇلاردىن ئازابنى دەپئى قىلىشتا) ھېچ نەرسىگە ئەسقاتمايدۇ
وَمَا أَهْلَكْنَا مِن قَرْيَةٍ إِلَّا لَهَا مُنذِرُونَ ﴿٢٠٨﴾
قانداقلىكى شەھەر بولمىسۇن، ۋەز - نەسىھەت قىلىش ئۈچۈن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتمىگىچە، ئۇنى ھالاك قىلمىدۇق، (ئۇلارنى جازالاشتا) بىز زالىم بولمىدۇق
ذِكْرَىٰ وَمَا كُنَّا ظَالِمِينَ ﴿٢٠٩﴾
قانداقلىكى شەھەر بولمىسۇن، ۋەز - نەسىھەت قىلىش ئۈچۈن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئەۋەتمىگىچە، ئۇنى ھالاك قىلمىدۇق، (ئۇلارنى جازالاشتا) بىز زالىم بولمىدۇق
وَمَا تَنَزَّلَتْ بِهِ الشَّيَاطِينُ ﴿٢١٠﴾
قۇرئاننى شەيتان (ئازدۇرغۇچى)لار ئېلىپ چۈشكىنى يوق
وَمَا يَنبَغِي لَهُمْ وَمَا يَسْتَطِيعُونَ ﴿٢١١﴾
بۇ ئۇلارغا لايىق ئەمەس. ئۇلارمۇ (ئۇنىڭغا) قادىر بولالمايدۇ
إِنَّهُمْ عَنِ السَّمْعِ لَمَعْزُولُونَ ﴿٢١٢﴾
شۈبھىسىزكى، ئۇلار (يەنى شەيتان (ئازدۇرغۇچى)لار) (ئوغرىلىقچە) تىڭشاشتىن مەنئى قىلىنغاندۇر
فَلَا تَدْعُ مَعَ اللَّهِ إِلَٰهًا آخَرَ فَتَكُونَ مِنَ الْمُعَذَّبِينَ ﴿٢١٣﴾
اﷲ تىن باشقا ھېچ ئىلاھقا ئىبادەت قىلمىغىن، ئۇنداق قىلساڭ، ئازاب قىلىنغۇچىلاردىن بولۇپ قالىسەن
وَأَنذِرْ عَشِيرَتَكَ الْأَقْرَبِينَ ﴿٢١٤﴾
يېقىن خىش – ئەقرىبالىرىڭنى ئاگاھلاندۇغىن
وَاخْفِضْ جَنَاحَكَ لِمَنِ اتَّبَعَكَ مِنَ الْمُؤْمِنِينَ ﴿٢١٥﴾
ساڭا ئەگەشكەن مۆمىنلەرگە مۇلايىم بولغىن
فَإِنْ عَصَوْكَ فَقُلْ إِنِّي بَرِيءٌ مِّمَّا تَعْمَلُونَ ﴿٢١٦﴾
ئەگەر خىش - ئەقرىبالىرىڭ ساڭا ئاسىيلىق قىلسا: «مەن سىلەرنىڭ قىلمىشلىرىڭلاردىن ئىشەنچلىك ھالدا ئادا - جۇدا مەن» دېگىن
وَتَوَكَّلْ عَلَى الْعَزِيزِ الرَّحِيمِ ﴿٢١٧﴾
ناھايىتى غالىب، مېھرىبان اﷲ قا يۆلەنگىن
الَّذِي يَرَاكَ حِينَ تَقُومُ ﴿٢١٨﴾
اﷲ سېنى (نامازغا) قوپقىنىڭدا كۆرۈپ تۇرىدۇ
وَتَقَلُّبَكَ فِي السَّاجِدِينَ ﴿٢١٩﴾
ناماز ئوقۇغۇچىلار ئارىسىدىكى (سەجدىگە بارغانلىق، رۇكۇغا تۇرغانلىق ۋە قىيامدا تۇرغانلىق) ھەرىكىتىڭنى كۆرۈپ تۇرىدۇ
إِنَّهُ هُوَ السَّمِيعُ الْعَلِيمُ ﴿٢٢٠﴾
شۈبھىسىزكى، اﷲ ھەممىنى ئاڭلاپ تۇرغۇچىدۇر، ھەممىنى بىلىپ تۇرغۇچىدۇر
هَلْ أُنَبِّئُكُمْ عَلَىٰ مَن تَنَزَّلُ الشَّيَاطِينُ ﴿٢٢١﴾
(ئى مۇھەممەد! مەككە كۇففارلىرىغا ئېيتقىنكى) سىلەرگە مەن شەيتان (ئازدۇرغۇچى)لارنىڭ كىمگە چۈشىدىغانلىقىنى ئېيتىپ بېرەيمۇ؟
تَنَزَّلُ عَلَىٰ كُلِّ أَفَّاكٍ أَثِيمٍ ﴿٢٢٢﴾
ئۇلار ھەربىر ئىغۋا توقۇغۇچى، گۇناھكارغا چۈشىدۇ
يُلْقُونَ السَّمْعَ وَأَكْثَرُهُمْ كَاذِبُونَ ﴿٢٢٣﴾
ئۇلار (خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ سۆزلىرىنى ئوغرىلىقچە) ئاڭلايدۇ، ئۇلار (يەنى شەيتان (ئازدۇرغۇچى)لار) نىڭ تولىسى يالغانچىلاردۇر
وَالشُّعَرَاءُ يَتَّبِعُهُمُ الْغَاوُونَ ﴿٢٢٤﴾
شائىرلارغا گۇمراھلار ئەگىشىدۇ
أَلَمْ تَرَ أَنَّهُمْ فِي كُلِّ وَادٍ يَهِيمُونَ ﴿٢٢٥﴾
ئۇلارنىڭ (سۆز) ۋادىلىرىدا تېڭىرقاپ يۈرگەنلىكىنى كۆرمەمسەن؟
وَأَنَّهُمْ يَقُولُونَ مَا لَا يَفْعَلُونَ ﴿٢٢٦﴾
ئۇلار قىلمايدىغان نەرسىلىرىنى قىلدۇق دەپ سۆزلەيدۇ
إِلَّا الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ وَذَكَرُوا اللَّهَ كَثِيرًا وَانتَصَرُوا مِن بَعْدِ مَا ظُلِمُوا ۗ وَسَيَعْلَمُ الَّذِينَ ظَلَمُوا أَيَّ مُنقَلَبٍ يَنقَلِبُونَ ﴿٢٢٧﴾
پەقەت ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغان، اﷲ نى كۆپ زىكرى قىلغان، زۇلۇمغا ئۇچرىغاندىن كېيىن ئۆزىنى قوغدىغان شائىرلار بۇنىڭدىن مۇستەسنا، زۇلۇم قىلغۇچىلار ئۇزاققا قالماي قايسى جايغا قايتىدىغانلىقىنى بىلىدۇ
