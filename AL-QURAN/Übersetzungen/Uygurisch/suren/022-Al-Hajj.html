Al-Hajj · The Pilgrimage
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
يَا أَيُّهَا النَّاسُ اتَّقُوا رَبَّكُمْ ۚ إِنَّ زَلْزَلَةَ السَّاعَةِ شَيْءٌ عَظِيمٌ ﴿١﴾
ئى ئىنسانلار! پەرۋەردىگارىڭلاردىن (يەنى ئازابىدىن) قورقۇڭلار، قىيامەتنىڭ زىلزىلىسى ئىشەنچلىك ھالدا چوڭ ئىشتۇر
يَوْمَ تَرَوْنَهَا تَذْهَلُ كُلُّ مُرْضِعَةٍ عَمَّا أَرْضَعَتْ وَتَضَعُ كُلُّ ذَاتِ حَمْلٍ حَمْلَهَا وَتَرَى النَّاسَ سُكَارَىٰ وَمَا هُم بِسُكَارَىٰ وَلَٰكِنَّ عَذَابَ اللَّهِ شَدِيدٌ ﴿٢﴾
ئۇ كۈندە ھەر بىر سۈت ئېمىتىۋاتقان ئايال (قاتتىق قورقۇنچتىن تېڭىرقاپ قېلىش بىلەن) ئېمىۋاتقان بوۋىقىنى ئۇنتۇيدۇ، ھەر بىر ھامىلىدار ئايالنىڭ بويىدىن ئاجراپ كېتىدۇ، ئىنسانلارنى مەست كۆرۈسەن، ھالبۇكى، ئۇلار (ھەقىقەتتە) مەست ئەمەس (دېمەك، قىيامەتنىڭ دەششىتى ئۇلارنى ئەس - ھوشىدىن كەتكۈزۈۋەتكەن)، لېكىن اﷲ نىڭ ئازابى قاتتىقتۇر
وَمِنَ النَّاسِ مَن يُجَادِلُ فِي اللَّهِ بِغَيْرِ عِلْمٍ وَيَتَّبِعُ كُلَّ شَيْطَانٍ مَّرِيدٍ ﴿٣﴾
بەزى كىشىلەر اﷲ (نىڭ قۇدرىتى ۋە سۈپەتلىرى) ئۈستىدە ھېچقانداق دەلىلسىز مۇنازىرىلىشىدۇ، ھەر بىر سەركەش شەيتان (ئازدۇرغۇچى)غا (يەنى ھەقتى توسقۇچى كۇففارلارنىڭ كاتتىباشلىرىغا) ئەگىشىدۇ
كُتِبَ عَلَيْهِ أَنَّهُ مَن تَوَلَّاهُ فَأَنَّهُ يُضِلُّهُ وَيَهْدِيهِ إِلَىٰ عَذَابِ السَّعِيرِ ﴿٤﴾
اﷲ ھۆكۈم قىلىدۇكى، كىمكى شەيتان (ئازدۇرغۇچى)غا ئەگىشىدىكەن شەيتان (ئازدۇرغۇچى) ئۇنى ئازدۇرىدۇ، ۋە مەڭگۈلۈك ئوتنىڭ ئازابىغا سۆرەيدۇ
يَا أَيُّهَا النَّاسُ إِن كُنتُمْ فِي رَيْبٍ مِّنَ الْبَعْثِ فَإِنَّا خَلَقْنَاكُم مِّن تُرَابٍ ثُمَّ مِن نُّطْفَةٍ ثُمَّ مِنْ عَلَقَةٍ ثُمَّ مِن مُّضْغَةٍ مُّخَلَّقَةٍ وَغَيْرِ مُخَلَّقَةٍ لِّنُبَيِّنَ لَكُمْ ۚ وَنُقِرُّ فِي الْأَرْحَامِ مَا نَشَاءُ إِلَىٰ أَجَلٍ مُّسَمًّى ثُمَّ نُخْرِجُكُمْ طِفْلًا ثُمَّ لِتَبْلُغُوا أَشُدَّكُمْ ۖ وَمِنكُم مَّن يُتَوَفَّىٰ وَمِنكُم مَّن يُرَدُّ إِلَىٰ أَرْذَلِ الْعُمُرِ لِكَيْلَا يَعْلَمَ مِن بَعْدِ عِلْمٍ شَيْئًا ۚ وَتَرَى الْأَرْضَ هَامِدَةً فَإِذَا أَنزَلْنَا عَلَيْهَا الْمَاءَ اهْتَزَّتْ وَرَبَتْ وَأَنبَتَتْ مِن كُلِّ زَوْجٍ بَهِيجٍ ﴿٥﴾
ئى ئىنسانلار! ئەگەر سىلەر قايتا تىرىلىشتىن شەكلەنسەڭلار (گۇمانىڭلارنىڭ تۈگىشى ئۈچۈن ئەسلىدىكى يارىتىلىشىڭلارغا قاراڭلار)، بىز سىلەرگە (اﷲ نىڭ قۇدرىتىنى بايان قىلىش ئۈچۈن ئاتاڭلار ئادەم ئەلەيھىسسالامنى) تۇپراقتىن، ئۇنىڭ ئەۋلادىنى (ئالدى بىلەن) ئابىمەنىيدىن، ئاندىن لەختە قاندىن، ئاندىن شەكىلگە كىرگۈزۈلگەن ۋە كىرگۈزۈلمىگەن پارچە گۆشتىن ياراتتۇق. بىز خالىغان ئادەمنى بەچچىداندان مۇئەييەن مۇددەتكىچە قالدۇرىمىز، ئاندىن سىلەرنى (ئاناڭلارنىڭ قارنىدىن) بوۋاقلىق ھالىتىڭلاردا چىقىرىمىز (ئاندىن سىلەرنى ئاستا - ئاستا كۈچ - قۇۋۋەتكە تولدۇرىمىز)، سىلەر (كامالەت يېشى) قىران ۋاقتىڭلارغا يېتىسىلەر، بەزىڭلار ياشلىقىڭلاردا ۋاپات بولۇپ كېتىسىلەر، بەزىڭلار، (ئىلگىرى) بىلگەندىن كېيىن ھېچ نەرسىنى بىلمەس ھالىتىڭلارغا قايتۇرۇلۇشىڭلار ئۈچۈن، ناھايىتى قېرىپ كەتكۈچە (ھايات) قالدۇرۇلىسىلەر، (اﷲ تائالانىڭ ئۆلگەنلەرنىڭ قايتا تىرىلدۈرۈشكە قادىر ئىكەنلىكىنى كۆرسىتىدىغان يەنە بىر دەلىل شۇكى) سەن زېمىننى قاقاس كۆرۈسەن، ئۇنىڭغا بىز يامغۇر ياغدۇرساق ئۇ جانلىنىدۇ ۋە كۆپۈشىدۇ، تۈرلۈك چىرايلىق ئۆسۈملۈكلەرنى ئۈندۈرۈپ بېرىدۇ
ذَٰلِكَ بِأَنَّ اللَّهَ هُوَ الْحَقُّ وَأَنَّهُ يُحْيِي الْمَوْتَىٰ وَأَنَّهُ عَلَىٰ كُلِّ شَيْءٍ قَدِيرٌ ﴿٦﴾
بۇ (يەنى اﷲ تائالانىڭ ئىنساننى ۋە ئۆسۈملۈكلەرنى يۇقىرىقىدەك يارىتىشى)، اﷲ نىڭ ھەق ئىكەنلىكىنى، ئۆلۈكلەرنى تىرىلدۈرەلەيدىغانلىقىنى ۋە ھەر نەرسىگە قادىر ئىكەنلىكىنى سىلەرگە بىلدۈرۈش ئۈچۈندۇر
وَأَنَّ السَّاعَةَ آتِيَةٌ لَّا رَيْبَ فِيهَا وَأَنَّ اللَّهَ يَبْعَثُ مَن فِي الْقُبُورِ ﴿٧﴾
شۇنداقلا قىيامەتنىڭ شەكسىز بولىدىغانلىقىنى، قەبرىلەردىكى ئۆلۈكلەرنى (ئۆلۈك زېمىننى ئۆسۈملۈكلەرنى ئۈندۈرۈش بىلەن تىرىلدۈرگەندەك) اﷲ نىڭ تىرىلدۈرىدىغانلىقىنى بىلدۈرۈش ئۈچۈندۇر
وَمِنَ النَّاسِ مَن يُجَادِلُ فِي اللَّهِ بِغَيْرِ عِلْمٍ وَلَا هُدًى وَلَا كِتَابٍ مُّنِيرٍ ﴿٨﴾
بەزىلەر (توغرا تونۇشقا ئېلىپ بارىدىغان) ھېچ قانداق ھەقىقى ئىلىمسىز، ھىدايەتسىز ۋە نۇرلۇق مۇقەددەس كىتاب-ۋەھىيسىز ھالدا اﷲ توغرىسىدا مۇنازىرىلىشىدۇ، (ئۇنىڭ مۇنازىرىلىشىشى نوقۇل نەپسى خاھىشى ۋە تەرسالىقىدىندۇر)
ثَانِيَ عِطْفِهِ لِيُضِلَّ عَن سَبِيلِ اللَّهِ ۖ لَهُ فِي الدُّنْيَا خِزْيٌ ۖ وَنُذِيقُهُ يَوْمَ الْقِيَامَةِ عَذَابَ الْحَرِيقِ ﴿٩﴾
ئۇ (كىشىلەرنى) اﷲ نىڭ يولىدىن ئازدۇرۇش ئۈچۈن مەغرۇرانە يۈرىدۇ، ئۇ دۇنيادا خارلىققا دۇچار بولىدۇ، ئاخىرەتتە ئۇنىڭغا كۆيدۈرگۈچى ئوتنىڭ ئازابىنى تېتىتىمىز
ذَٰلِكَ بِمَا قَدَّمَتْ يَدَاكَ وَأَنَّ اللَّهَ لَيْسَ بِظَلَّامٍ لِّلْعَبِيدِ ﴿١٠﴾
(ئۇنىڭغا): «(خارلىق بىلەن ئازاب) سېنىڭ قىلمىشلىرىڭ يەنى كۇفرىڭ بىلەن گۇمراھلىقىڭ) تۈپەيلىدىندۇر، اﷲ بەندىلىرىگە ھەرگىز زۇلۇم قىلغۇچى ئەمەستۇر» (دېيىلىدۇ)
وَمِنَ النَّاسِ مَن يَعْبُدُ اللَّهَ عَلَىٰ حَرْفٍ ۖ فَإِنْ أَصَابَهُ خَيْرٌ اطْمَأَنَّ بِهِ ۖ وَإِنْ أَصَابَتْهُ فِتْنَةٌ انقَلَبَ عَلَىٰ وَجْهِهِ خَسِرَ الدُّنْيَا وَالْآخِرَةَ ۚ ذَٰلِكَ هُوَ الْخُسْرَانُ الْمُبِينُ ﴿١١﴾
بەزىلەر اﷲ قا شەك - شۈبھە بىلەن (يەنى تۇراقسىز ھالدا) ئىبادەت قىلىدۇ، ئەگەر ئۇنىڭغا ياخشىلىق يەتسە دىنىدا تۇرىدۇ (بۇنداق ئادەم مۇناپىق بولۇپ، اﷲ قا دىلى بىلەن ئەمەس پەقەت تىلى بىلەن ئىبادەت قىلىدۇ)، ئەگەر ئۇنىڭغا بەختسىزلىك يەتسە دىنىدىن يېنىۋالىدۇ، ئۇ دۇنيا ۋە ئاخىرەتتە زىيان تارتىدۇ، بۇ روشەن زىياندۇر
يَدْعُو مِن دُونِ اللَّهِ مَا لَا يَضُرُّهُ وَمَا لَا يَنفَعُهُ ۚ ذَٰلِكَ هُوَ الضَّلَالُ الْبَعِيدُ ﴿١٢﴾
ئۇ اﷲ نى قويۇپ پايدا - زىيان يەتكۈزەلمەيدىغان بۇتقا چوقۇنىدۇ، بۇ چوڭقۇر گۇمراھلىقتۇر
يَدْعُو لَمَن ضَرُّهُ أَقْرَبُ مِن نَّفْعِهِ ۚ لَبِئْسَ الْمَوْلَىٰ وَلَبِئْسَ الْعَشِيرُ ﴿١٣﴾
ئۇ (مۇشرىكلارنىڭ قارىشى بويىچە بۇت قىيامەت كۈنى شاپائەت قىلىدۇ دېگەندىمۇ) ئەلۋەتتە زىيىنى پايدىسىدىن يېقىن بولغان نەرسىگە چوقۇنىدۇ، (ئۇ بۇت) راستتىنلا يامان مەدەتكاردۇر، راستتىنلا يامان ھەمراھتۇر
إِنَّ اللَّهَ يُدْخِلُ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ جَنَّاتٍ تَجْرِي مِن تَحْتِهَا الْأَنْهَارُ ۚ إِنَّ اللَّهَ يَفْعَلُ مَا يُرِيدُ ﴿١٤﴾
شۈبھىسىزكى، اﷲ ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنى ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان مەڭگۈلۈك باغلارلەرگە كىرگۈزىدۇ، اﷲ ئىشەنچلىك ھالدا خالىغىنىنى قىلىدۇ (يەنى خالىغان ئادەمگە ساۋاب بېرىدۇ، خالىغان ئادەمگە ئازاب قىلىدۇ)
مَن كَانَ يَظُنُّ أَن لَّن يَنصُرَهُ اللَّهُ فِي الدُّنْيَا وَالْآخِرَةِ فَلْيَمْدُدْ بِسَبَبٍ إِلَى السَّمَاءِ ثُمَّ لْيَقْطَعْ فَلْيَنظُرْ هَلْ يُذْهِبَنَّ كَيْدُهُ مَا يَغِيظُ ﴿١٥﴾
كىمكى اﷲ خۇدانىڭ ئەلچىسىگە دۇنيا ۋە ئاخىرەتتە ھەرگىز ياردەم بەرمەيدۇ دەپ گۇمان قىلىدىكەن، ئۇ تورۇسقا ئارغامچا سېلىپ بوغۇلۇپ ئالسۇن، قارىسۇنكى، ئۇنىڭ (بۇ) تەدبىرى غەزىپىنى باسالامدۇ؟ (يەنى اﷲ پەيغەمبىرى مۇھەممەد ئەلەيھىسسالامغا ياردەم بەرمەيدۇ دەپ ئويلايدىكەن ئۇ ئېسىلىپ ئۆلۈۋالسۇن، ئەگەر بۇ ئۇنىڭ خاپىلىقىنى باسالايدىغان بولسا، چۈنكى اﷲ خۇدانىڭ ئەلچىسىگە چوقۇم ياردەم بېرىدۇ)
وَكَذَٰلِكَ أَنزَلْنَاهُ آيَاتٍ بَيِّنَاتٍ وَأَنَّ اللَّهَ يَهْدِي مَن يُرِيدُ ﴿١٦﴾
شۇنىڭغا ئوخشاش (يەنى ئىلگىرىكى مۇقەددەس كىتاب-ۋەھىيلارنى نازىل قىلغىنىمىزغا ئوخشاش). قۇرئاننى (مۇھەممەد ئەلەيھىسسالامغا) روشەن ئايەتلەر قىلىپ نازىل قىلدۇق، اﷲ خالىغان ئادەمنى ھىدايەت قىلىدۇ (يەنى اﷲ تىن باشقا ھېچ ھىدايەت قىلغۇچى يوق، اﷲ خالىغان ئادەمنى توغرا يولغا باشلايدۇ
إِنَّ الَّذِينَ آمَنُوا وَالَّذِينَ هَادُوا وَالصَّابِئِينَ وَالنَّصَارَىٰ وَالْمَجُوسَ وَالَّذِينَ أَشْرَكُوا إِنَّ اللَّهَ يَفْصِلُ بَيْنَهُمْ يَوْمَ الْقِيَامَةِ ۚ إِنَّ اللَّهَ عَلَىٰ كُلِّ شَيْءٍ شَهِيدٌ ﴿١٧﴾
مۆمىنلەر، يەھۇدىيلار، يۇلتۇزلارغا چوقۇنغۇچىلار، ناسارالار، مەجۇسىيلەر (يەنى ئاتەشپەرەسلەر) ۋە مۇشرىكلار ئۈچۈن قىيامەت كۈنى اﷲ ئىشەنچلىك ھالدا ھۆكۈم چىقىرىدۇ، (يەنى اﷲ تائالا مۆمىنلەر بىلەن مەزكۇر بەش پىرقە ئارىسىدا توغرا ھۆكۈم چىقىرىپ، مۆمىنلەرنى مەڭگۈلۈك باغلاركە، كۇففارلارنى دوزاخقا كىرگۈزىدۇ)، اﷲ ئىشەنچلىك ھالدا ھەممە نەرسىنى كۆرۈپ تۇرغۇچىدۇر
أَلَمْ تَرَ أَنَّ اللَّهَ يَسْجُدُ لَهُ مَن فِي السَّمَاوَاتِ وَمَن فِي الْأَرْضِ وَالشَّمْسُ وَالْقَمَرُ وَالنُّجُومُ وَالْجِبَالُ وَالشَّجَرُ وَالدَّوَابُّ وَكَثِيرٌ مِّنَ النَّاسِ ۖ وَكَثِيرٌ حَقَّ عَلَيْهِ الْعَذَابُ ۗ وَمَن يُهِنِ اللَّهُ فَمَا لَهُ مِن مُّكْرِمٍ ۚ إِنَّ اللَّهَ يَفْعَلُ مَا يَشَاءُ ﴿١٨﴾
بىلمەمسەنكى، ئاسماندىكىلەر (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى)، زېمىندىكىلەر (يەنى ئىنسانلار، جىنلار ۋە زېمىندا ياشايدىغان بارلىق مەخلۇقاتلار)، كۈن، ئاي، يۇلتۇزلار، تاغلار، دەرەخلەر، ھايۋاناتلار اﷲ قا بويسۇنۇپ، اﷲ نىڭ تەسەررۇپ قىلىشى بويىچە ھەرىكەت قىلىدۇ) ۋە نۇرغۇن كىشىلەر اﷲ قا سەجدە قىلىدۇ، نۇرغۇن كىشىلەر (كۇفرى سەۋەبلىك) ئازابقا دۇچار بولدى، اﷲ نىڭ خارلىشىغا ئۇچرىغان ئادەمگە ھۆرمەتلىگۈچى چىقمايدۇ، اﷲ ئىشەنچلىك ھالدا خالىغىنىنى قىلىدۇ (يەنى ئازاب قىلىش، رەھىم قىلىش، ئەزىز قىلىش، خار قىلىش، باي قىلىش، كەمبەغەل قىلىش اﷲ نىڭ خاھىشىدىكى ئىش بولۇپ، اﷲ قا ھېچ كىشى تەئەررۇز قىلالمايدۇ)
هَٰذَانِ خَصْمَانِ اخْتَصَمُوا فِي رَبِّهِمْ ۖ فَالَّذِينَ كَفَرُوا قُطِّعَتْ لَهُمْ ثِيَابٌ مِّن نَّارٍ يُصَبُّ مِن فَوْقِ رُءُوسِهِمُ الْحَمِيمُ ﴿١٩﴾
(مۆمىنلەر ۋە كۇففارلاردىن ئىبارەت) بۇ ئىككى (پىرقە) پەرۋەردىگارى توغرىسىدا مۇنازىرىلەشتى (يەنى مۆمىنلەر بىلەن كۇففارلار اﷲ نىڭ دىنى ئۈستىدە بەس - مۇنازىرە قىلىشىپ، مۆمىنلەر اﷲ نىڭ دىنىغا ياردەم بەرمەكچى بولدى، كۇففارلار اﷲ نىڭ نۇرىنى ئۆچۈرمەكچى بولدى)، ھەقىقەتنى رەت قىلغۇچىلارغا ئوتتىن كىيىملەر پىچىلىدۇ، ئۇلارنىڭ باشلىرى ئۈستىدىن يۇقىرى ھارارەتلىك قايناقسۇ قويۇلىدۇ
يُصْهَرُ بِهِ مَا فِي بُطُونِهِمْ وَالْجُلُودُ ﴿٢٠﴾
ئۇنىڭ بىلەن ئۇلارنىڭ ئىچ –باغرى ۋە تېرىلىرى ئېرىتىلىدۇ
وَلَهُم مَّقَامِعُ مِنْ حَدِيدٍ ﴿٢١﴾
ئۇلار تۆمۈر - توقماقلار بىلەن ئۇرۇلىدۇ
كُلَّمَا أَرَادُوا أَن يَخْرُجُوا مِنْهَا مِنْ غَمٍّ أُعِيدُوا فِيهَا وَذُوقُوا عَذَابَ الْحَرِيقِ ﴿٢٢﴾
ھەر قاچان ئۇلار (يېتىۋاتقان) غەم - قايغۇنىڭ قاتتىقلىقىدىن دوزاختىن چىقماقچى بولسا، ئۇلار دوزاخقا قايتۇرۇلىدۇ، (ئۇلارغا) كۆيدۈرگۈچى (دوزاخ) ئازابىنى تېتىڭلار (دېيىلىدۇ)
إِنَّ اللَّهَ يُدْخِلُ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ جَنَّاتٍ تَجْرِي مِن تَحْتِهَا الْأَنْهَارُ يُحَلَّوْنَ فِيهَا مِنْ أَسَاوِرَ مِن ذَهَبٍ وَلُؤْلُؤًا ۖ وَلِبَاسُهُمْ فِيهَا حَرِيرٌ ﴿٢٣﴾
اﷲ ئىشەنچلىك ھالدا مۆمىنلەرنى ۋە ياخشى ئەمەللەرنى قىلغانلارنى ئاستىدىن ئۆستەڭلار ئېقىپ تۇرىدىغان مەڭگۈلۈك باغلارلەرگە كىرگۈزىدۇ، ئۇلار مەڭگۈلۈك باغلارلەردە ئالتۇن بىلەيزۈكلەرنى ۋە مەرۋايىتلارنى زىننەت بۇيۇملىرى قىلىدۇ، ئۇلارنىڭ كىيىمى يىپەكتىن بولىدۇ
وَهُدُوا إِلَى الطَّيِّبِ مِنَ الْقَوْلِ وَهُدُوا إِلَىٰ صِرَاطِ الْحَمِيدِ ﴿٢٤﴾
ئۇلار ياخشى سۆزگە (يەنى كەلىمە تەۋھىدكە) ھىدايەت قىلىندى. اﷲ نىڭ يولىغا ھىدايەت قىلىندى
إِنَّ الَّذِينَ كَفَرُوا وَيَصُدُّونَ عَن سَبِيلِ اللَّهِ وَالْمَسْجِدِ الْحَرَامِ الَّذِي جَعَلْنَاهُ لِلنَّاسِ سَوَاءً الْعَاكِفُ فِيهِ وَالْبَادِ ۚ وَمَن يُرِدْ فِيهِ بِإِلْحَادٍ بِظُلْمٍ نُّذِقْهُ مِنْ عَذَابٍ أَلِيمٍ ﴿٢٥﴾
ھەقىقەتنى رەت قىلغۇچىلارغا، (كىشىلەرنى) اﷲ نىڭ يولىدىن توسقۇچىلارغا، - ئۇ ئولتۇرۇشلۇق بولسۇن ياكى سىرتتىن كەلگەن بولسۇن - بىز كىشىلەرگە ئىبادەتگاھ قىلغان مەسجىدى ھەرەمدىن توسقۇچىلارغا (يەنى ھەج پەرزنى ئادا قىلىش ئۈچۈن مەسجىدى ھەرەمغا كەلگەن مۆمىنلەرنى توسقۇچىلارغا) (قاتتىق ئازابنى تېتىتىمىز)، كىمكى مەسجىدى ھەرەمدا زۇلۇم بىلەن گۇناھ قىلماقچى بولىدىكەن، ئۇنىڭغا قاتتىق ئازابنى تېتىتىمىز
وَإِذْ بَوَّأْنَا لِإِبْرَاهِيمَ مَكَانَ الْبَيْتِ أَن لَّا تُشْرِكْ بِي شَيْئًا وَطَهِّرْ بَيْتِيَ لِلطَّائِفِينَ وَالْقَائِمِينَ وَالرُّكَّعِ السُّجُودِ ﴿٢٦﴾
ئۆز ۋاقتىدا ئىبراھىمغا بەيتۇللاھنىڭ ئورنىنى تەيىنلەپ بەردۇق، (ئۇنىڭغا ئېيتتۇقكى) «ماڭا ھېچ نەرسىنى شېرىك كەلتۈرمىگىن، تاۋاپ قىلغۇچىلارغا، قىيام قىلغۇچىلارغا، رۇكۇ قىلغۇچىلارغا، سەجدە قىلغۇچىلارغا مىنىڭ ئۆيۈمنى پاك قىلغىن
وَأَذِّن فِي النَّاسِ بِالْحَجِّ يَأْتُوكَ رِجَالًا وَعَلَىٰ كُلِّ ضَامِرٍ يَأْتِينَ مِن كُلِّ فَجٍّ عَمِيقٍ ﴿٢٧﴾
كىشىلەرنىڭ ئارىسىدا (ئۇلارنى) ھەجىگە چاقىرىپ نىدا قىلغىن، ئۇلار پىيادە ۋە ئۇرۇق تۆگىلەرگە مىنىپ كېلىدۇ، ئۇرۇق تۆگىلەر يىراق يوللارنى بېسىپ كېلىدۇ
لِّيَشْهَدُوا مَنَافِعَ لَهُمْ وَيَذْكُرُوا اسْمَ اللَّهِ فِي أَيَّامٍ مَّعْلُومَاتٍ عَلَىٰ مَا رَزَقَهُم مِّن بَهِيمَةِ الْأَنْعَامِ ۖ فَكُلُوا مِنْهَا وَأَطْعِمُوا الْبَائِسَ الْفَقِيرَ ﴿٢٨﴾
كىشىلەر ئۆزلىرىگە تېگىشلىك بولغان (دىنى ۋە دۇنياۋى) مەنپەئەتلەرنى كۆرسۇن، بەلگىلەنگەن كۈنلەردە (يەنى قۇربانلىق كۈنلىرىدە) اﷲ ئۇلارغا رىزىق قىلىپ بەرگەن چارۋا ماللارنى (يەنى تۆگە، قوي، ئۆچكىلەرنى اﷲ نىڭ نېمەتلىرىگە شۈكۈر قىلىش يۈزىسىدىن) اﷲ نىڭ ئىسمىنى ئېيتىپ قۇربانلىق قىلسۇن». سىلەر قۇربانلىقلارنى گۆشىدىن يەڭلار، موھتاجقا، پېقىرغا بېرىڭلار
ثُمَّ لْيَقْضُوا تَفَثَهُمْ وَلْيُوفُوا نُذُورَهُمْ وَلْيَطَّوَّفُوا بِالْبَيْتِ الْعَتِيقِ ﴿٢٩﴾
ئاندىن ئۇلار كىرلىرىنى تازىلىسۇن (يەنى ئېھرامدىن چىققاندىن كېيىن چاچلىرىنى، تىرناقلىرىنى ئالسۇن)، ئۆز ئۈستىگە ئالغان ئىبادەتلىرىنى ئادا قىلسۇن، قەدىمىي بەيت (يەنى بەيتۇللاھنى) تاۋاپ قىلسۇن
ذَٰلِكَ وَمَن يُعَظِّمْ حُرُمَاتِ اللَّهِ فَهُوَ خَيْرٌ لَّهُ عِندَ رَبِّهِ ۗ وَأُحِلَّتْ لَكُمُ الْأَنْعَامُ إِلَّا مَا يُتْلَىٰ عَلَيْكُمْ ۖ فَاجْتَنِبُوا الرِّجْسَ مِنَ الْأَوْثَانِ وَاجْتَنِبُوا قَوْلَ الزُّورِ ﴿٣٠﴾
(ھەج ئىبادىتى) ئەنە شۇدۇر، كىمكى اﷲ نىڭ دىنىنىڭ ئەھكاملىرىنى ئۇلۇغلىسا (يەنى ھەج جەريانىدا اﷲ نىڭ بۇيرۇغانلىرىنى بەجا كەلتۈرۈپ، مەنئى قىلغان ئىشلىرىدىن چەكلەنسە)، بۇ، پەرۋەردىگارنىڭ دەرگاھىدا ئۇنىڭ (دۇنيا ۋە ئاخىرەتلىكى) ئۈچۈن ياخشىدۇر، سىلەرگە چاھارپايلاردىن قۇرئاندا (ھاراملىقى) بايان قىلىنغانلىرىدىن باشقىلىرى ھالال قىلىندى. سىلەر بۇتلاردىن ئىبارەت نىجىستىن ساقلىنىڭلار، يالغان گۇۋاھلىق بېرىشتىن ساقلىنىڭلار
حُنَفَاءَ لِلَّهِ غَيْرَ مُشْرِكِينَ بِهِ ۚ وَمَن يُشْرِكْ بِاللَّهِ فَكَأَنَّمَا خَرَّ مِنَ السَّمَاءِ فَتَخْطَفُهُ الطَّيْرُ أَوْ تَهْوِي بِهِ الرِّيحُ فِي مَكَانٍ سَحِيقٍ ﴿٣١﴾
اﷲ قا خالىس ئىبادەت قىلىڭلار، ئۇنىڭغا شېرىك كەلتۈرمەڭلار، كىمكى اﷲ قا شېرىك كەلتۈرىدىكەن، ئۇ گويا ئاسماندىن تاشلىنىپ، قۇشلار ئۇنى ئېلىپ قاچقاندەك، ياكى بوران ئۇنى ئېلىپ بېرىپ يىراق جايغا تاشلىۋەتكەندەك بولۇپ قالىدۇ
ذَٰلِكَ وَمَن يُعَظِّمْ شَعَائِرَ اللَّهِ فَإِنَّهَا مِن تَقْوَى الْقُلُوبِ ﴿٣٢﴾
ئىش مانا شۇنداق، كىمكى دىنىي ئىشلارنى (جۈملىدىن ھەجنىڭ ئەمەللىرىنى قۇربانلىقلارنى) ئۇلۇغلايدىكەن، بۇ، دىللارنىڭ تەقۋادارلىقىدىندۇر
لَكُمْ فِيهَا مَنَافِعُ إِلَىٰ أَجَلٍ مُّسَمًّى ثُمَّ مَحِلُّهَا إِلَى الْبَيْتِ الْعَتِيقِ ﴿٣٣﴾
مۇئەييەن مۇددەتكىچە (يەنى بوغۇزلايدىغان ۋاقىتقىچە قۇربانلىق ماللىرىدىن (سېغىپ، نەسىللەندۈرۈپ، مىنىپ) پايدىلىنىسىلەر، ئاندىن ئۇنى بوغۇزلاشقا تېگىشلىك جاي بەيتۇللاھنىڭ يېنىدۇر (يەنى (يەنى ھەرەمدۇر)
وَلِكُلِّ أُمَّةٍ جَعَلْنَا مَنسَكًا لِّيَذْكُرُوا اسْمَ اللَّهِ عَلَىٰ مَا رَزَقَهُم مِّن بَهِيمَةِ الْأَنْعَامِ ۗ فَإِلَٰهُكُمْ إِلَٰهٌ وَاحِدٌ فَلَهُ أَسْلِمُوا ۗ وَبَشِّرِ الْمُخْبِتِينَ ﴿٣٤﴾
ھەر ئۈممەت. اﷲ ئۇلارغا رىزىق قىلىپ بەرگەن چارۋىلارنى (شۈكۈر قىلىش يۈزىسىدىن قۇربانلىق قىلغانلىرىدا) اﷲ نىڭ ئىسمىنى ئېيتسۇن دەپ، ئۇلارغا قۇربانلىقنى بەلگىلىدۇق، سىلەرنىڭ ئىلاھىڭلار بىر ئىلاھتۇر، ئۇنىڭغا بويسۇنۇڭلار، ئىتائەتمەنلەرگە خۇش خەۋەر بەرگىن
الَّذِينَ إِذَا ذُكِرَ اللَّهُ وَجِلَتْ قُلُوبُهُمْ وَالصَّابِرِينَ عَلَىٰ مَا أَصَابَهُمْ وَالْمُقِيمِي الصَّلَاةِ وَمِمَّا رَزَقْنَاهُمْ يُنفِقُونَ ﴿٣٥﴾
ئۇلار شۇنداق كىشىلەركى اﷲ ياد ئېتىلسە، ئۇلارنىڭ دىللىرى قورقۇپ (تىترەيدۇ)، ئۆزلىرىگە يەتكەن ئەزىيەتلەرگە سەۋر قىلىدۇ، ناماز ئۆتەيدۇ، ئۇلارغا بىز رىزىق قىلىپ بەرگەن نەرسىلەردىن (ياخشىلىق يوللىرىغا) بېرىدۇ
وَالْبُدْنَ جَعَلْنَاهَا لَكُم مِّن شَعَائِرِ اللَّهِ لَكُمْ فِيهَا خَيْرٌ ۖ فَاذْكُرُوا اسْمَ اللَّهِ عَلَيْهَا صَوَافَّ ۖ فَإِذَا وَجَبَتْ جُنُوبُهَا فَكُلُوا مِنْهَا وَأَطْعِمُوا الْقَانِعَ وَالْمُعْتَرَّ ۚ كَذَٰلِكَ سَخَّرْنَاهَا لَكُمْ لَعَلَّكُمْ تَشْكُرُونَ ﴿٣٦﴾

وَلِيَعْلَمَ الَّذِينَ أُوتُوا الْعِلْمَ أَنَّهُ الْحَقُّ مِن رَّبِّكَ فَيُؤْمِنُوا بِهِ فَتُخْبِتَ لَهُ قُلُوبُهُمْ ۗ وَإِنَّ اللَّهَ لَهَادِ الَّذِينَ آمَنُوا إِلَىٰ صِرَاطٍ مُّسْتَقِيمٍ ﴿٥٤﴾
(اﷲ تائالانىڭ مۇنداق قىلىشى) ئىلىم ئەھلىلىرىنىڭ قۇرئاننىڭ پەرۋەردىگارىڭ تەرىپىدىن كەلگەن ھەقىقەت ئىكەنلىكىنى بىلىشلىرى، شۇنىڭ ئۈچۈندۇر. اﷲ مۆمىنلەرنى ئىشەنچلىك ھالدا توغرا يولغا باشلىغۇچىدۇر
وَلَا يَزَالُ الَّذِينَ كَفَرُوا فِي مِرْيَةٍ مِّنْهُ حَتَّىٰ تَأْتِيَهُمُ السَّاعَةُ بَغْتَةً أَوْ يَأْتِيَهُمْ عَذَابُ يَوْمٍ عَقِيمٍ ﴿٥٥﴾
ھەقىقەتنى رەت قىلغۇچىلار تاكى ئۇلارغا قىيامەت ئۇشتۇمتۇت كەلگۈچە ياكى ئۇلارغا قىيامەت كۈنىنىڭ ئازابى يەتكۈچە قۇرئاندىن ھامان شەكلىنىدۇ
الْمُلْكُ يَوْمَئِذٍ لِّلَّهِ يَحْكُمُ بَيْنَهُمْ ۚ فَالَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ فِي جَنَّاتِ النَّعِيمِ ﴿٥٦﴾
بۇ كۈندە پادىشاھلىق (يالغۇز) اﷲ قا خاستۇر، اﷲ بەندىلەر ئارىسىدا (ئادىل) ھۆكۈم چىقىرىدۇ، ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلار (تۈرلۈك) نازۇنېمەتلەر بار مەڭگۈلۈك باغلارلەردە بولىدۇ
وَالَّذِينَ كَفَرُوا وَكَذَّبُوا بِآيَاتِنَا فَأُولَٰئِكَ لَهُمْ عَذَابٌ مُّهِينٌ ﴿٥٧﴾
ھەقىقەتنى رەت قىلغۇچىلار ۋە بىزنىڭ ئايەتلىرىمىزنى ئىنكار قىلغانلار (دوزاختا) خار قىلغۇچى ئازابقا دۇچار بولىدۇ
وَالَّذِينَ هَاجَرُوا فِي سَبِيلِ اللَّهِ ثُمَّ قُتِلُوا أَوْ مَاتُوا لَيَرْزُقَنَّهُمُ اللَّهُ رِزْقًا حَسَنًا ۚ وَإِنَّ اللَّهَ لَهُوَ خَيْرُ الرَّازِقِينَ ﴿٥٨﴾
اﷲ نىڭ يولىدا ھىجرەت قىلغانلارغا، ئاندىن (جىھاد قىلىپ) ئۆلتۈرۈلگەنلەرگە ياكى (كېسەل بىلەن) ئۆلگەنلەرگە اﷲ چوقۇم ياخشى رىزىق (يەنى مەڭگۈلۈك باغلارنىڭ مەڭگۈلۈك نېمەتلىرىنى) ئاتا قىلىدۇ. رىزىق بەرگۈچىلەرنىڭ ئەڭ ياخشىسى ئىشەنچلىك ھالدا يالغۇز اﷲ دۇر
لَيُدْخِلَنَّهُم مُّدْخَلًا يَرْضَوْنَهُ ۗ وَإِنَّ اللَّهَ لَعَلِيمٌ حَلِيمٌ ﴿٥٩﴾
اﷲ ئۇلارنى ئەلۋەتتە ئۇلار مەمنۇن بولىدىغان جايغا (يەنى مەڭگۈلۈك باغلاركە) كىرگۈزىدۇ، اﷲ ئىشەنچلىك ھالدا (ئۇلارنى مەمنۇن قىلىدىغان نەرسىلەرنى) بىلگۈچىدۇر، ھەلىمدۇر (يەنى ئۇلارنىڭ دۈشمەنلىرىنى جازالاشقا ئالدىراپ كەتمەيدۇ)
ذَٰلِكَ وَمَنْ عَاقَبَ بِمِثْلِ مَا عُوقِبَ بِهِ ثُمَّ بُغِيَ عَلَيْهِ لَيَنصُرَنَّهُ اللَّهُ ۗ إِنَّ اللَّهَ لَعَفُوٌّ غَفُورٌ ﴿٦٠﴾
ئىش مانا شۇنداق، كىمكى قانچىلىك زۇلۇمغا ئۇچراپ شۇنچىلىك ئىنتىقامىنى ئالغان بولسا، ئاندىن يەنە زۇلۇمغا ئۇچرىسا، اﷲ ئۇنىڭغا چوقۇم ياردەم بېرىدۇ، اﷲ ئىشەنچلىك ھالدا ناھايىتى ئەپۇ قىلغۇچىدۇر، ناھايىتى مەغپىرەت قىلغۇچىدۇر
ذَٰلِكَ بِأَنَّ اللَّهَ يُولِجُ اللَّيْلَ فِي النَّهَارِ وَيُولِجُ النَّهَارَ فِي اللَّيْلِ وَأَنَّ اللَّهَ سَمِيعٌ بَصِيرٌ ﴿٦١﴾
(شۇ نەرسە اﷲ نىڭ قۇدرىتىنىڭ ئالامەتلىرىدىندۇركى) اﷲ كېچىنى كۈندۈزگە، كۈندۈزنى كېچىگە كىرگۈزىدۇ (شۇنىڭ بىلەن كېچە - كۈندۈزنىڭ ئۇزۇن - قىسقا بولۇشى پەسىللەر بويىچە نۆۋەتلىشىپ تۇرىدۇ)، اﷲ (بەندىلىرىنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (ئەھۋالىنى) كۆرۈپ تۇرغۇچىدۇر
ذَٰلِكَ بِأَنَّ اللَّهَ هُوَ الْحَقُّ وَأَنَّ مَا يَدْعُونَ مِن دُونِهِ هُوَ الْبَاطِلُ وَأَنَّ اللَّهَ هُوَ الْعَلِيُّ الْكَبِيرُ ﴿٦٢﴾
بۇ (بۇ) (قۇدرەت) شۇنىڭ بۇنىڭ ئۈچۈندۇركى، اﷲ ھەقتۇر، (مۇشرىكلارنىڭ) اﷲ نى قويۇپ ئىبادەت قىلىدىغان نەرسىلىرى (يەنى بۇتلىرى) باتىلدۇر، اﷲ (ھەممە نەرسىدىن) ئۈستۈندۇر، بۈيۈكتۇر
أَلَمْ تَرَ أَنَّ اللَّهَ أَنزَلَ مِنَ السَّمَاءِ مَاءً فَتُصْبِحُ الْأَرْضُ مُخْضَرَّةً ۗ إِنَّ اللَّهَ لَطِيفٌ خَبِيرٌ ﴿٦٣﴾
اﷲ نىڭ بۇلۇتتىن يامغۇر ياغدۇرۇپ بېرىدىغانلىقىنى، شۇنىڭ بىلەن زېمىننى يېشىلزارلىققا ئايلاندۇرىدىغانلىقىنى كۆرمىدىڭمۇ؟ اﷲ ئىشەنچلىك ھالدا ناھايىتى مېھرىباندۇر، ھەممىدىن خەۋەرداردۇر
لَّهُ مَا فِي السَّمَاوَاتِ وَمَا فِي الْأَرْضِ ۗ وَإِنَّ اللَّهَ لَهُوَ الْغَنِيُّ الْحَمِيدُ ﴿٦٤﴾
ئاسمانلاردىكى ۋە زېمىندىكى شەيئىلەرنىڭ ھەممىسى اﷲ نىڭدۇر (يەنى اﷲ نىڭ مۈلكىدۇر، اﷲ نىڭ مەخلۇقاتىدۇر، اﷲ نىڭ تەسەررۇپى ئاستىدىدۇر)، اﷲ ئىشەنچلىك ھالدا ھەممىدىن بىھاجەتتۇر، ھەمدۇساناغا لايىقتۇر
أَلَمْ تَرَ أَنَّ اللَّهَ سَخَّرَ لَكُم مَّا فِي الْأَرْضِ وَالْفُلْكَ تَجْرِي فِي الْبَحْرِ بِأَمْرِهِ وَيُمْسِكُ السَّمَاءَ أَن تَقَعَ عَلَى الْأَرْضِ إِلَّا بِإِذْنِهِ ۗ إِنَّ اللَّهَ بِالنَّاسِ لَرَءُوفٌ رَّحِيمٌ ﴿٦٥﴾
اﷲ نىڭ يەر يۈزىدىكى نەرسىلەرنى، ئۆزىنىڭ ئەمرى بويىچە دېڭىزدا يۈرۈۋاتقان كېمىلەرنى سىلەرگە تويۇندۇرۇپ بەرگەنلىكىنى كۆرمىدىڭمۇ؟ (ئاسماننىڭ زېمىنغا چۈشۈپ كەتمەسلىكى ئۈچۈن، اﷲ ئاسماننى توختىتىپ تۇرىدۇ، پەقەت اﷲ نىڭ ئىزنى بىلەن (ئاسماننىڭ قىيامەت بولغان چاغدا) چۈشۈپ كېتىشى بۇنىڭدىن مۇستەسنا. اﷲ ئىنسانلارغا ئىشەنچلىك ھالدا ناھايىتى شەپقەتلىكتۇر، ناھايىتى مېھرىباندۇر
وَهُوَ الَّذِي أَحْيَاكُمْ ثُمَّ يُمِيتُكُمْ ثُمَّ يُحْيِيكُمْ ۗ إِنَّ الْإِنسَانَ لَكَفُورٌ ﴿٦٦﴾
اﷲ سىلەرگە ھاياتلىق بېرىدۇ، ئاندىن سىلەرنىڭ جېنىڭلارنى ئالىدۇ، ئاندىن سىلەرنى تىرىلدۈرىدۇ، ئىنسان ئىشەنچلىك ھالدا (اﷲ نىڭ نېمەتلىرىگە) ناشۈكۈرلۈك قىلغۇچىدۇر
لِّكُلِّ أُمَّةٍ جَعَلْنَا مَنسَكًا هُمْ نَاسِكُوهُ ۖ فَلَا يُنَازِعُنَّكَ فِي الْأَمْرِ ۚ وَادْعُ إِلَىٰ رَبِّكَ ۖ إِنَّكَ لَعَلَىٰ هُدًى مُّسْتَقِيمٍ ﴿٦٧﴾
ھەر ئۈممەت ئۈچۈن بىر شەرىئەت بېكىتتۇق، ئۇلار شۇ شەرىئەتكە ئەمەل قىلىدۇ، (مۇشرىكلار) شەرىئەت ئىشىدا سەن بىلەن جېدەللەشمىسۇن، كىشىلەرنى پەرۋەردىگارىڭنىڭ (ئىبادىتىگە) دەۋەت قىلغىن، سەن ئىشەنچلىك ھالدا شەكسىز توغرا يولدىدۇرسەن
وَإِن جَادَلُوكَ فَقُلِ اللَّهُ أَعْلَمُ بِمَا تَعْمَلُونَ ﴿٦٨﴾

اللَّهُ يَحْكُمُ بَيْنَكُمْ يَوْمَ الْقِيَامَةِ فِيمَا كُنتُمْ فِيهِ تَخْتَلِفُونَ ﴿٦٩﴾

بىلمەمسەنكى، اﷲ ئاسمان، زېمىندىكى شەيئىلەرنى بىلىپ تۇرىدۇ، بۇ ئىشەنچلىك ھالدا لەۋھۇلمەھپۇزدا (يېزىلغان)، بۇ (يەنى بىلىش) ئىشەنچلىك ھالدا اﷲ قا ئاسان

ئۇلار (يەنى قۇرەيش كۇففارلىرى) اﷲ نى قويۇپ، (ئىبادەت قىلىشنىڭ دۇرۇسلۇقىغا) اﷲ ھېچقانداق دەلىل چۈشۈرمىگەن نەرسىلەرگە ۋە ئۆزلىرى بىلمەيدىغان نەرسىلەرگە (يەنى بوتلارغا) چوقۇنىدۇ، زالىملارغا ھېچقانداق مەدەتكار يوق

ئۇلار (ئەرزىمەس بۇتلارنى اﷲ قا شېرىك قىلىۋېلىپ) اﷲ نى تېگىشلىك رەۋىشتە ئۇلۇغلاشمىدى، شۈبھىسىزكى، اﷲ ئىشەنچلىك ھالدا كۈچلۈكتۇر، غالىبتۇر
اللَّهُ يَصْطَفِي مِنَ الْمَلَائِكَةِ رُسُلًا وَمِنَ النَّاسِ ۚ إِنَّ اللَّهَ سَمِيعٌ بَصِيرٌ ﴿٧٥﴾
اﷲ خۇدانىڭ ئەلچى پەرىشتىلىرىدىنمۇ (خۇدانىڭ ئەلچىسىلىرىگە ۋەھيى يەتكۈزۈش ئۈچۈن)، ئەلچىلەرنى تاللايدۇ، اﷲ ئىشەنچلىك ھالدا (بەندىلىرىنىڭ سۆزلىرىنى) ئاڭلاپ تۇرغۇچىدۇر، (قىلمىشلىرىنى) كۆرۈپ تۇرغۇچىدۇر
يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ ۗ وَإِلَى اللَّهِ تُرْجَعُ الْأُمُورُ ﴿٧٦﴾
اﷲ ئۇلارنىڭ ئالدىدىكى نەرسىلەرنى (يەنى ئۇلارنىڭ ھازىرقى ئەھۋالىنى ۋە كەلگۈسىدىكى ئىشلىرىنى) بىلىدۇ. (بەندىلەرنىڭ) ئىشلىرىنىڭ ھەممىسى اﷲ قا قايتۇرۇلىدۇ (اﷲ ئۇلارغا يا مۇكاپات، يا جازا بېرىدۇ)
يَا أَيُّهَا الَّذِينَ آمَنُوا ارْكَعُوا وَاسْجُدُوا وَاعْبُدُوا رَبَّكُمْ وَافْعَلُوا الْخَيْرَ لَعَلَّكُمْ تُفْلِحُونَ ﴿٧٧﴾
ئى مۆمىنلەر! بەختكە ئېرىشىشىڭلار ئۈچۈن رۇكۇ قىلىڭلار، سەجدە قىلىڭلار (يەنى تەزەررۇ بىلەن ناماز ئوقۇڭلار)، (يالغۇز) پەرۋەردىگارىڭلارغا ئىبادەت قىلىڭلار، ياخشى ئىشلارنى (يەنى خىش - ئەقرىبالارغا سىلەر - رەھىم قىلىش، يېتىم - يېسىرلەرنىڭ بېشىنى سىلاش، كېچىسى تەھەججۇد نامىزى ئۆتەش قاتارلىق ئىشلارنى) قىلىڭلار
وَجَاهِدُوا فِي اللَّهِ حَقَّ جِهَادِهِ ۚ هُوَ اجْتَبَاكُمْ وَمَا جَعَلَ عَلَيْكُمْ فِي الدِّينِ مِنْ حَرَجٍ ۚ مِّلَّةَ أَبِيكُمْ إِبْرَاهِيمَ ۚ هُوَ سَمَّاكُمُ الْمُسْلِمِينَ مِن قَبْلُ وَفِي هَٰذَا لِيَكُونَ الرَّسُولُ شَهِيدًا عَلَيْكُمْ وَتَكُونُوا شُهَدَاءَ عَلَى النَّاسِ ۚ فَأَقِيمُوا الصَّلَاةَ وَآتُوا الزَّكَاةَ وَاعْتَصِمُوا بِاللَّهِ هُوَ مَوْلَاكُمْ ۖ فَنِعْمَ الْمَوْلَىٰ وَنِعْمَ النَّصِيرُ ﴿٧٨﴾
اﷲ نىڭ يولىدا (پۇل - مېلىڭلار بىلەن، جېنىڭلار بىلەن) تاقىتىڭلارنىڭ يېتىشىچە جىھاد قىلىڭلار، اﷲ (دىنغا خۇدانىڭ ئەلچىسىنى سىلەرگە خاس قىلدى). سىلەرگە دىندا ھېچقانداق مۈشكۈللۈكنى قىلمىدى (سىلەرنى سىلەر تاقەت قىلالايدىغان ئىشلارنى قىلىشقا تەكلىپ قىلمىدى). (بۇ) ئاتاڭلار ئىبراھىمنىڭ دىنىدۇر (بۇ توغرا دىن بولغانلىقى ئۈچۈن، ئۇنى مەھكەم ئۇچلاڭلار)، اﷲ سىلەرنى ئىلگىرى (يەنى قۇرئاندىن ئىلگىرىكى مۇقەددەس كىتاب-ۋەھىيلاردا) مۇسۇلمان دەپ ئاتىدۇ. قۇرئاندىمۇ شۇنداق ئاتىدۇ. (اﷲ) خۇدانىڭ ئەلچىسىنى سىلەرگە گۇۋاھ بولۇشقا ۋە سىلەرنى كىشىلەرگە گۇۋاھ بولۇشقا (تاللىدى)، ناماز ئوقۇڭلار، زاكات (پاكلاش ئۇلۇشى) بېرىڭلار، اﷲ قا يېپىشىڭلار، اﷲ سىلەرنىڭ ئىگەڭلاردۇر، اﷲ نېمىدېگەن ياخشى ئىگە! نېمىدېگەن ياخشى مەدەتكار!
