Sad · The Letter "Saad"
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
ص ۚ وَالْقُرْآنِ ذِي الذِّكْرِ ﴿١﴾
صاد. شانلىق قۇرئان بىلەن قەسەمكى، (بۇ قۇرئان ئىشەنچلىك ھالدا مۆجىزىدۇر)
بَلِ الَّذِينَ كَفَرُوا فِي عِزَّةٍ وَشِقَاقٍ ﴿٢﴾
بەلكى ھەقىقەتنى رەت قىلغۇچىلار ئىماندىن باش تارتماقتا ۋە (رەسۇلۇللاھقا) مۇخالىپەتچىلىك قىلماقتا
كَمْ أَهْلَكْنَا مِن قَبْلِهِم مِّن قَرْنٍ فَنَادَوا وَّلَاتَ حِينَ مَنَاصٍ ﴿٣﴾
ئۇلاردىن ئىلگىرى (كۇفرى تۈپەيلىدىن) نۇرغۇن ئۈممەتلەرنى ھالاك قىلدۇق، ئۇلار (ئازابنى كۆرگەندە) پەرياد قىلدى، ھالبۇكى، (بۇ) قاچىدىغان ۋاقىت ئەمەس
وَعَجِبُوا أَن جَاءَهُم مُّنذِرٌ مِّنْهُمْ ۖ وَقَالَ الْكَافِرُونَ هَٰذَا سَاحِرٌ كَذَّابٌ ﴿٤﴾
ئۇلارغا ئۆزلىرىنىڭ ئىچىدىن بىر ئاگاھلاندۇرغۇچى (يەنى خۇدانىڭ ئەلچىسى) كەلگەنلىكىگە ئۇلار ھەيران قېلىشتى، ھەقىقەتنى رەت قىلغۇچىلار ئېيتتى: «بۇ بىر يالغانچى سېھىرگەردۇر
أَجَعَلَ الْآلِهَةَ إِلَٰهًا وَاحِدًا ۖ إِنَّ هَٰذَا لَشَيْءٌ عُجَابٌ ﴿٥﴾
ئۇ نۇرغۇن ئىلاھلارنى بىر ئىلاھ قىلماقچىمۇ؟ بۇ ئىشەنچلىك ھالدا غەلىتە ئىش»
وَانطَلَقَ الْمَلَأُ مِنْهُمْ أَنِ امْشُوا وَاصْبِرُوا عَلَىٰ آلِهَتِكُمْ ۖ إِنَّ هَٰذَا لَشَيْءٌ يُرَادُ ﴿٦﴾
ئۇلارنىڭ (يەنى قۇرەيشنىڭ) كاتتىلىرى (سورۇنلىرىدىن) قوزغىلىپ چىقىپ ئېيتتى: «سىلەر مېڭىڭلار، مەبۇدلىرىڭلارغا ئىبادەت قىلىشتا چىڭ تۇرۇڭلار، بۇ ئىشەنچلىك ھالدا (بىزدىن) ئىرادە قىلىنىدىغان بىر ئىش (يەنى مۇھەممەد سىلەرنى دىنىڭلاردىن تاندۇرۇپ، سىلەرگە ئۆزىنىڭ ئۈستۈنلۈك قازىنىشىنى ئىرادە قىلىدۇ)
مَا سَمِعْنَا بِهَٰذَا فِي الْمِلَّةِ الْآخِرَةِ إِنْ هَٰذَا إِلَّا اخْتِلَاقٌ ﴿٧﴾
بۇنى بىز باشقا بىر دىندىن (يەنى ناسارا دىنىدىكىلەردىن) ئاڭلىمىدۇق، بۇ پەقەت ئويدۇرمىدۇر
أَأُنزِلَ عَلَيْهِ الذِّكْرُ مِن بَيْنِنَا ۚ بَلْ هُمْ فِي شَكٍّ مِّن ذِكْرِي ۖ بَل لَّمَّا يَذُوقُوا عَذَابِ ﴿٨﴾
قۇرئان بىزگە نازىل بولماي ئۇنىڭغا نازىل بولدىمۇ؟» بەلكى ئۇلار مېنىڭ زىكرىم (يەنى قۇرئان) دىن شەكتىدۇر، بەلكى ئۇلار تېخى مېنىڭ ئازابىمنى تېتىغىنى يوق
أَمْ عِندَهُمْ خَزَائِنُ رَحْمَةِ رَبِّكَ الْعَزِيزِ الْوَهَّابِ ﴿٩﴾
ياكى ئۇلاردا غالىب، ناھايىتى كەرەملىك پەرۋەردىگارىڭنىڭ رەھمىتىنىڭ خەزىنىلىرى بارمۇ؟
أَمْ لَهُم مُّلْكُ السَّمَاوَاتِ وَالْأَرْضِ وَمَا بَيْنَهُمَا ۖ فَلْيَرْتَقُوا فِي الْأَسْبَابِ ﴿١٠﴾
ياكى ئۇلار ئاسمانلارنىڭ، زېمىننىڭ ۋە ئۇلارنىڭ ئارىسىدىكى نەرسىلەرنىڭ پادىشاھلىقىغا ئىگىمۇ؟ (ئەگەر ئۇلار ئىگە بولسا، ئۇلار ئۆزلىرىنى ئاسمانغا ئېلىپ چىقىدىغان) شوتىلارغا چىقسۇن
جُندٌ مَّا هُنَالِكَ مَهْزُومٌ مِّنَ الْأَحْزَابِ ﴿١١﴾
ئۇلار (رەسۇلۇللاھقا قارشى) جامائەلەردىن ئۇيۇشقان مەغلۇب قىلىنىدىغان قوشۇندۇر
كَذَّبَتْ قَبْلَهُمْ قَوْمُ نُوحٍ وَعَادٌ وَفِرْعَوْنُ ذُو الْأَوْتَادِ ﴿١٢﴾
ئۇلاردىن (يەنى قۇرەيش مۇشرىكلىرىدىن) ئىلگىرى نۇھ قەۋمى، ئاد (قەۋمى) ۋە قوزۇقلار ئىگىسى پىرئەۋن (قەۋمى) (خۇدانىڭ ئەلچىسىلىرىنى) ئىنكار قىلدى
وَثَمُودُ وَقَوْمُ لُوطٍ وَأَصْحَابُ الْأَيْكَةِ ۚ أُولَٰئِكَ الْأَحْزَابُ ﴿١٣﴾
سەمۇد قەۋمى، لۇت قەۋمى ۋە ئەيكىلىكلەر (يەنى شۇئەيب ئەلەيھىسسالام قەۋمى) مۇ (خۇدانىڭ ئەلچىسىلىرىنى ئىنكار قىلدى)، ئەنە شۇلار (خۇدانىڭ ئەلچى-رەسۇللىرىگە قارشى ئۇيۇشقان) جامائەلەر ئىدى
إِن كُلٌّ إِلَّا كَذَّبَ الرُّسُلَ فَحَقَّ عِقَابِ ﴿١٤﴾
ئۇلار (يەنى جامائەلەر) دىن خۇدانىڭ ئەلچى-رەسۇللىرىنى ئىنكار قىلمىغان بىرىمۇ يوق، شۇنىڭ ئۈچۈن ئۇلارغا مېنىڭ ئازابىم تېگىشلىك بولدى
وَمَا يَنظُرُ هَٰؤُلَاءِ إِلَّا صَيْحَةً وَاحِدَةً مَّا لَهَا مِن فَوَاقٍ ﴿١٥﴾
بۇلار (يەنى مەككە مۇشرىكلىرى) پەقەت بىر ئاۋاز (يەنى بىرىنچى قېتىملىق سۇر) نىلا كۈتىدۇ، ئۇ قايتىلانمايدۇ
وَقَالُوا رَبَّنَا عَجِّل لَّنَا قِطَّنَا قَبْلَ يَوْمِ الْحِسَابِ ﴿١٦﴾
ئۇلار (مەسخىرە قىلىش يولى بىلەن): «پەرۋەردىگارىمىز! (سەن بىزگە ئازابتىن ۋەدە قىلغان) نېسىۋىمىزنى بىزگە بالدۇرراق – قىيامەت كۈنىدىن ئىلگىرى بەرگىن» دېيىشتى
اصْبِرْ عَلَىٰ مَا يَقُولُونَ وَاذْكُرْ عَبْدَنَا دَاوُودَ ذَا الْأَيْدِ ۖ إِنَّهُ أَوَّابٌ ﴿١٧﴾
ئۇلارنىڭ سۆزلىرىگە سەۋر قىلغىن، (دىندا) كۈچلۈك بەندىمىز داۋۇدنى ئەسلىگىن، ئۇ ئىشەنچلىك ھالدا اﷲ قا ئىتائەت قىلغۇچى ئىدى
إِنَّا سَخَّرْنَا الْجِبَالَ مَعَهُ يُسَبِّحْنَ بِالْعَشِيِّ وَالْإِشْرَاقِ ﴿١٨﴾
بىز ئىشەنچلىك ھالدا تاغلارنى (داۋۇدقا) بويسۇندۇرۇپ بەردۇق، ئۇلار داۋۇد بىلەن ئەتىگىنى - ئاخشىمى تەسبىھ ئېيتاتتى
وَالطَّيْرَ مَحْشُورَةً ۖ كُلٌّ لَّهُ أَوَّابٌ ﴿١٩﴾
قۇشلارنىمۇ ئۇنىڭغا بويسۇندۇرۇپ بەردۇق، ئۇلارمۇ (تەرەپ - تەرەپتىن) توپلىنىپ، داۋۇد بىلەن بىللە تەسبىھ ئېيتاتتى، (تاغلارنىڭ، قۇشلارنىڭ) ھەممىسى اﷲ قا ئىتائەت قىلغۇچىدۇر
وَشَدَدْنَا مُلْكَهُ وَآتَيْنَاهُ الْحِكْمَةَ وَفَصْلَ الْخِطَابِ ﴿٢٠﴾
داۋۇدنىڭ سەلتەنىتىنى كۈچەيتتۇق، ئۇنىڭغا ھېكمەت ۋە (ھەممە ئادەم چۈشىنىدىغان) روشەن سۆز - ئىبارىلەرنى ئاتا قىلدۇق
وَهَلْ أَتَاكَ نَبَأُ الْخَصْمِ إِذْ تَسَوَّرُوا الْمِحْرَابَ ﴿٢١﴾
(ئى مۇھەممەد!) دەۋا قىلغۇچىلارنىڭ قىسسىسى ساڭا يەتتىمۇ؟ ئۆز ۋاقتىدا ئۇلار (داۋۇد ئىبادەت قىلىۋاتقان) مەسجىدنىڭ تېمىدىن ئارتىلىپ چۈشۈشتى
إِذْ دَخَلُوا عَلَىٰ دَاوُودَ فَفَزِعَ مِنْهُمْ ۖ قَالُوا لَا تَخَفْ ۖ خَصْمَانِ بَغَىٰ بَعْضُنَا عَلَىٰ بَعْضٍ فَاحْكُم بَيْنَنَا بِالْحَقِّ وَلَا تُشْطِطْ وَاهْدِنَا إِلَىٰ سَوَاءِ الصِّرَاطِ ﴿٢٢﴾
ئەينى ۋاقىتتا ئۇلار داۋۇدنىڭ يېنىغا كىردى. داۋۇد ئۇلاردىن قورقتى، ئۇلار (داۋۇدقا): «قورقمىغىن، بىز بىرىمىز - بىرىمىزگە چېقىلغان ئىككى دەۋاگەرمىز، بىزنىڭ ئارىمىزدا ئادىللىق بىلەن ھۆكۈم قىلغىن، ناھەق ھۆكۈم چىقارمىغىن، بىزنى توغرا يولغا باشلىغىن» دېدى
إِنَّ هَٰذَا أَخِي لَهُ تِسْعٌ وَتِسْعُونَ نَعْجَةً وَلِيَ نَعْجَةٌ وَاحِدَةٌ فَقَالَ أَكْفِلْنِيهَا وَعَزَّنِي فِي الْخِطَابِ ﴿٢٣﴾
(ئۇلارنىڭ بىرى) «شۈبھىسىزكى، مېنىڭ بۇ قېرىندىشىمنىڭ 99 ساغلىقى بار، مېنىڭ بىر ساغلىقىم بار، قېرىندىشىم ئۇ بىر ساغلىقنىمۇ ماڭا بەرگىن دەيدۇ، (ئارىمىزدا بۇ توغرۇلۇق مۇنازىرە بولۇپ) ئۇ مېنى سۆزدە يېڭىپ قويدى» دېدى
قَالَ لَقَدْ ظَلَمَكَ بِسُؤَالِ نَعْجَتِكَ إِلَىٰ نِعَاجِهِ ۖ وَإِنَّ كَثِيرًا مِّنَ الْخُلَطَاءِ لَيَبْغِي بَعْضُهُمْ عَلَىٰ بَعْضٍ إِلَّا الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ وَقَلِيلٌ مَّا هُمْ ۗ وَظَنَّ دَاوُودُ أَنَّمَا فَتَنَّاهُ فَاسْتَغْفَرَ رَبَّهُ وَخَرَّ رَاكِعًا وَأَنَابَ ﴿٢٤﴾
داۋۇد ئېيتتى: «ئۇ سېنىڭ ساغلىقىڭنى ئۆزىنىڭ ساغلىقلىرىغا قوشۇۋېلىشنى تەلەپ قىلىپ راستلا سېنى بوزەك قىلىپتۇ، نۇرغۇن شېرىكلەر (يەنى دوستلار)، شۈبھىسىزكى، بىر - بىرىگە چېقىلىدۇ، پەقەت ئىمان ئېيتقان ۋە ياخشى ئەمەل قىلغانلارلا (بىر - بىرىگە چېقىلمايدۇ) بۇلار ئازدۇر». داۋۇد بىزنىڭ ئۇنى سىنىغانلىقىمىزنى بىلدى. پەرۋەردىگارىدىن مەغپىرەت تەلەپ قىلدى، سەجدىگە باردى. (اﷲ قا) تەۋبە قىلدى
فَغَفَرْنَا لَهُ ذَٰلِكَ ۖ وَإِنَّ لَهُ عِندَنَا لَزُلْفَىٰ وَحُسْنَ مَآبٍ ﴿٢٥﴾
بىز ئۇنىڭ خاتالىقىنى ئەپۇ قىلدۇق، ئۇ بىزنىڭ دەرگاھىمىزدا ئەلۋەتتە يېقىنلىققا ۋە ياخشى ئاقىۋەتكە ئىگە بولىدۇ
يَا دَاوُودُ إِنَّا جَعَلْنَاكَ خَلِيفَةً فِي الْأَرْضِ فَاحْكُم بَيْنَ النَّاسِ بِالْحَقِّ وَلَا تَتَّبِعِ الْهَوَىٰ فَيُضِلَّكَ عَن سَبِيلِ اللَّهِ ۚ إِنَّ الَّذِينَ يَضِلُّونَ عَن سَبِيلِ اللَّهِ لَهُمْ عَذَابٌ شَدِيدٌ بِمَا نَسُوا يَوْمَ الْحِسَابِ ﴿٢٦﴾
ئى داۋۇد! سېنى بىز ئىشەنچلىك ھالدا يەر يۈزىدە خەلىپە قىلدۇق، كىشىلەرنىڭ ئارىسىدا ئادىللىق بىلەن ھۆكۈم چىقارغىن، نەپسى خاھىشقا ئەگەشمىگىنكى، ئۇ سېنى اﷲ نىڭ يولىدىن ئازدۇرىدۇ، اﷲ نىڭ يولىدىن ئازغانلار ھېساب كۈنى (يەنى قىيامەت كۈنى) نى ئۇنتۇغانلىقلىرى ئۈچۈن ئىشەنچلىك ھالدا قاتتىق ئازابقا دۇچار بولىدۇ
وَمَا خَلَقْنَا السَّمَاءَ وَالْأَرْضَ وَمَا بَيْنَهُمَا بَاطِلًا ۚ ذَٰلِكَ ظَنُّ الَّذِينَ كَفَرُوا ۚ فَوَيْلٌ لِّلَّذِينَ كَفَرُوا مِنَ النَّارِ ﴿٢٧﴾
ئاسماننى، زېمىننى ۋە ئۇلارنىڭ ئارىسىدىكى شەيئىلەرنى بىكار ياراتمىدۇق، ئۇ (يەنى ئۇلارنى بىكار يارىتىلغان دەپ قاراش) ھەقىقەتنى رەت قىلغۇچىلارنىڭ گۇمانىدۇر، ھەقىقەتنى رەت قىلغۇچىلارغا دوزاختىن ۋاي!
أَمْ نَجْعَلُ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ كَالْمُفْسِدِينَ فِي الْأَرْضِ أَمْ نَجْعَلُ الْمُتَّقِينَ كَالْفُجَّارِ ﴿٢٨﴾
ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنى يەر يۈزىدە بۇزغۇنچىلىق قىلغانلار بىلەن ئوخشاش قىلامدۇق؟ ياكى تەقۋادارلارنى فاجىرلارغا ئوخشاش قىلامدۇق؟
كِتَابٌ أَنزَلْنَاهُ إِلَيْكَ مُبَارَكٌ لِّيَدَّبَّرُوا آيَاتِهِ وَلِيَتَذَكَّرَ أُولُو الْأَلْبَابِ ﴿٢٩﴾
(بۇ) ئۇلارنىڭ ئايەتلەرنى تەپەككۈر قىلىشلىرى ئۈچۈن، ئەقىل ئىگىلىرىنىڭ ۋەز - نەسىھەت ئېلىشلىرى ئۈچۈن بىز ساڭا نازىل قىلغان مۇبارەك مۇقەددەس كىتاب-ۋەھىيتۇر
وَوَهَبْنَا لِدَاوُودَ سُلَيْمَانَ ۚ نِعْمَ الْعَبْدُ ۖ إِنَّهُ أَوَّابٌ ﴿٣٠﴾
بىز داۋۇدقا سۇلەيماننى ئاتا قىلدۇق، سۇلەيمان نېمىدېگەن ياخشى بەندە! ئۇ (اﷲ قا) ئىشەنچلىك ھالدا ئىتائەت قىلغۇچى ئىدى
إِذْ عُرِضَ عَلَيْهِ بِالْعَشِيِّ الصَّافِنَاتُ الْجِيَادُ ﴿٣١﴾
ئۆز ۋاقتىدا كەچقۇرۇنلىقى ئۇنىڭغا ياخشى يۈگۈرۈك ئاتلار توغرىلاندى

إِذْ قَالَ رَبُّكَ لِلْمَلَائِكَةِ إِنِّي خَالِقٌ بَشَرًا مِّن طِينٍ ﴿٧١﴾
ئۆز ۋاقتىدا پەرۋەردىگارىڭ خۇدانىڭ ئەلچى پەرىشتىلىرىگە ئېيتتى: «مەن ئىشەنچلىك ھالدا لايدىن بىر ئادەم يارىتىمەن
فَإِذَا سَوَّيْتُهُ وَنَفَخْتُ فِيهِ مِن رُّوحِي فَقَعُوا لَهُ سَاجِدِينَ ﴿٧٢﴾
ئۇنى مەن راۋۇرۇس ياراتقان (يەنى ئۇنى يارىتىپ سۈرەتكە كىرگۈزۈپ، ئەزالىرىنى تولۇق، مۇكەممەل ئىنسان ھالىتىگە كەلتۈرگەن) ۋە ئۇنىڭغا جان كىرگۈزگەن ۋاقتىمدا ئۇنىڭغا سەجدە قىلىڭلار»
فَسَجَدَ الْمَلَائِكَةُ كُلُّهُمْ أَجْمَعُونَ ﴿٧٣﴾
خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ بىرىمۇ قالماي، ھەممىسى (ئادەم ئەلەيھىسسالامغا) سەجدە قىلدى
إِلَّا إِبْلِيسَ اسْتَكْبَرَ وَكَانَ مِنَ الْكَافِرِينَ ﴿٧٤﴾
پەقەت ئىبلىسلا بويۇنتاۋلىق قىلدى، ئۇ ھەقىقەتنى رەت قىلغۇچىلاردىن بولۇپ كەتتى
قَالَ يَا إِبْلِيسُ مَا مَنَعَكَ أَن تَسْجُدَ لِمَا خَلَقْتُ بِيَدَيَّ ۖ أَسْتَكْبَرْتَ أَمْ كُنتَ مِنَ الْعَالِينَ ﴿٧٥﴾
اﷲ ئېيتتى: «ئى ئىبلىس! مەن ئۆز قولۇم بىلەن ياراتقان ئىنسانغا سەجدە قىلىشتىن ساڭا نېمە توسقۇنلۇق قىلدى (ئۇنىڭغا سەجدە قىلماسلىقىڭ تەكەببۇرلۇق قىلغانلىقىڭدىنمۇ؟ ياكى يۇقىرى مەرتىۋىلىكلەردىن بولغانلىقىڭدىنمۇ؟»
قَالَ أَنَا خَيْرٌ مِّنْهُ ۖ خَلَقْتَنِي مِن نَّارٍ وَخَلَقْتَهُ مِن طِينٍ ﴿٧٦﴾
ئىبلىس: «مەن ئۇنىڭدىن ئارتۇق، مېنى ئوتتىن ياراتتىڭ، ئۇنى لايدىن ياراتتىڭ» دېدى
قَالَ فَاخْرُجْ مِنْهَا فَإِنَّكَ رَجِيمٌ ﴿٧٧﴾
اﷲ: «سەن مەڭگۈلۈك باغلارتىن يوقال! سەن ئىشەنچلىك ھالدا قوغلاندى بولدۇڭ، ساڭا ئىشەنچلىك ھالدا قىيامەتكىچە مېنىڭ لەنىتىم بولسۇن» دېدى
وَإِنَّ عَلَيْكَ لَعْنَتِي إِلَىٰ يَوْمِ الدِّينِ ﴿٧٨﴾
اﷲ: «سەن مەڭگۈلۈك باغلارتىن يوقال! سەن ئىشەنچلىك ھالدا قوغلاندى بولدۇڭ، ساڭا ئىشەنچلىك ھالدا قىيامەتكىچە مېنىڭ لەنىتىم بولسۇن» دېدى
قَالَ رَبِّ فَأَنظِرْنِي إِلَىٰ يَوْمِ يُبْعَثُونَ ﴿٧٩﴾
ئىبلىس: «پەرۋەردىگارىم! ماڭا (خالايىق) تىرىلدۈرۈلىدىغان كۈن (يەنى قىيامەت) گىچە مۆھلەت بەرسەڭ (يەنى مېنى ھايات قالدۇرساڭ)» دېدى
قَالَ فَإِنَّكَ مِنَ الْمُنظَرِينَ ﴿٨٠﴾
اﷲ: «جەزمەن ساڭا مەلۇم ۋاقىتقىچە مۆھلەت بېرىلىدۇ» دېدى
إِلَىٰ يَوْمِ الْوَقْتِ الْمَعْلُومِ ﴿٨١﴾
اﷲ: «جەزمەن ساڭا مەلۇم ۋاقىتقىچە مۆھلەت بېرىلىدۇ» دېدى
قَالَ فَبِعِزَّتِكَ لَأُغْوِيَنَّهُمْ أَجْمَعِينَ ﴿٨٢﴾
ئىبلىس ئېيتتى: «ئىززىتىڭ بىلەن قەسەمكى، ئۇلارنىڭ ھەممىسىنى چوقۇم ئازدۇرىمەن
إِلَّا عِبَادَكَ مِنْهُمُ الْمُخْلَصِينَ ﴿٨٣﴾
ئۇلاردىن پەقەت تاللانغان بەندىلىرىڭلا بۇنىڭدىن مۇستەسنا (يەنى ئۇلارنى ئازدۇرالمايمەن)»
قَالَ فَالْحَقُّ وَالْحَقَّ أَقُولُ ﴿٨٤﴾
اﷲ ئېيتتى: «مېنىڭ سۆزۈم ھەقتۇر، مەن ھەقنى سۆزلەيمەن
لَأَمْلَأَنَّ جَهَنَّمَ مِنكَ وَمِمَّن تَبِعَكَ مِنْهُمْ أَجْمَعِينَ ﴿٨٥﴾
سەن بىلەن ۋە ئۇلاردىن ساڭا ئەگەشكەنلەرنىڭ ھەممىسى بىلەن چوقۇم مەڭگۈلۈك ئوتنى توشقۇزىمەن»
قُلْ مَا أَسْأَلُكُمْ عَلَيْهِ مِنْ أَجْرٍ وَمَا أَنَا مِنَ الْمُتَكَلِّفِينَ ﴿٨٦﴾
ئېيتقىنكى، «(اﷲ تەرىپىدىن كەلگەن ۋەھيىنى) تەبلىغ قىلغانلىقىمغا سىلەردىن ھېچ ھەق تەلەپ قىلمايمەن، مەن (قۇرئاننى) ئىپتىرا قىلغۇچىلاردىن (يەنى يالغاندىن توقۇغۇچىلاردىن) ئەمەسمەن
إِنْ هُوَ إِلَّا ذِكْرٌ لِّلْعَالَمِينَ ﴿٨٧﴾
قۇرئان پەقەت جاھان ئەھلى ئۈچۈن ۋەز - نەسىھەتتۇر
وَلَتَعْلَمُنَّ نَبَأَهُ بَعْدَ حِينٍ ﴿٨٨﴾
بىر مۇددەتتىن كېيىن ئۇنىڭ راستلىقىنى چوقۇم بىلىسىلەر»
