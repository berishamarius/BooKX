Saba · Sheba
بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ
ٱلْحَمْدُ لِلَّهِ ٱلَّذِى لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ وَلَهُ ٱلْحَمْدُ فِى ٱلْـَٔاخِرَةِ ۚ وَهُوَ ٱلْحَكِيمُ ٱلْخَبِيرُ ﴿١﴾
جىمى ھەمدۇسانا اﷲ قا خاستۇر، ئاسمانلاردىكى ۋە زېمىندىكى ھەممە نەرسە اﷲ نىڭ مۈلكىدۇر، ئاخىرەتتىكى ھەمدۇسانامۇ اﷲ قا خاستۇر، اﷲ ھېكمەت بىلەن ئىش قىلغۇچىدۇر، ھەممىدىن خەۋەرداردۇر
يَعْلَمُ مَا يَلِجُ فِى ٱلْأَرْضِ وَمَا يَخْرُجُ مِنْهَا وَمَا يَنزِلُ مِنَ ٱلسَّمَآءِ وَمَا يَعْرُجُ فِيهَا ۚ وَهُوَ ٱلرَّحِيمُ ٱلْغَفُورُ ﴿٢﴾
اﷲ زېمىننىڭ قوينىغا كىرىپ كېتىدىغان (يەنى يامغۇر، كان ۋە ئۆلۈكلەرگە ئوخشاش) نەرسىلەرنى، زېمىندىن چىقىدىغان (ئۆسۈملۈك، بۇلاق ۋە قۇدۇقلارنىڭ سۇلىرى قاتارلىق) نەرسىلەرنى، ئاسماندىن چۈشىدىغان (پەرىشتە، يامغۇر، ئەجەل، رىزىق قاتارلىق) نەرسىلەرنى، ئاسمانغا ئۆرلەيدىغان (ياخشى ئەمەللەر، مەقبۇل دۇئالار قاتارلىق) نەرسىلەرنى بىلىپ تۇرىدۇ، اﷲ بەندىلىرىگە ناھايىتى مېھرىباندۇر، (تەۋبە قىلغۇچىلارنى) ناھايىتى مەغپىرەت قىلغۇچىدۇر
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لَا تَأْتِينَا ٱلسَّاعَةُ ۖ قُلْ بَلَىٰ وَرَبِّى لَتَأْتِيَنَّكُمْ عَـٰلِمِ ٱلْغَيْبِ ۖ لَا يَعْزُبُ عَنْهُ مِثْقَالُ ذَرَّةٍ فِى ٱلسَّمَـٰوَٰتِ وَلَا فِى ٱلْأَرْضِ وَلَآ أَصْغَرُ مِن ذَٰلِكَ وَلَآ أَكْبَرُ إِلَّا فِى كِتَـٰبٍ مُّبِينٍ ﴿٣﴾
ھەقىقەتنى رەت قىلغۇچىلار: «بىزگە قىيامەت كەلمەيدۇ» دەيدۇ، ئېيتقىنكى، «ئۇنداق ئەمەس، پەرۋەردىگارىم بىلەن قەسەمكى، قىيامەت سىلەرگە چوقۇم كېلىدۇ، اﷲ غەيىبنى بىلگۈچىدۇر، ئاسمانلاردىكى ۋە زېمىندىكى زەررە چاغلىق نەرسە اﷲ تىن يىراق ئەمەستۇر، ئۇنىڭ (يەنى زەررە) دىن كىچىكرەك نەرسە بولسۇن، چوڭراق نەرسە بولسۇن، لەۋھۇلمەھپۇزدا خاتىرىلەنمىگىنى يوق»
لِّيَجْزِىَ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ ۚ أُو۟لَـٰٓئِكَ لَهُم مَّغْفِرَةٌ وَرِزْقٌ كَرِيمٌ ﴿٤﴾
(ئۇلارنىڭ لەۋھۇلمەھپۇزدا خاتىرىلىنىشى) ئىمان ئېيتقان ۋە ياخشى ئەمەللەرنى قىلغانلارنى مۇكاپاتلاش ئۈچۈندۇر، ئەنە شۇلار مەغپىرەتكە ۋە ئېسىل رىزىققا ئېرىشىدۇ
وَٱلَّذِينَ سَعَوْ فِىٓ ءَايَـٰتِنَا مُعَـٰجِزِينَ أُو۟لَـٰٓئِكَ لَهُمْ عَذَابٌ مِّن رِّجْزٍ أَلِيمٌ ﴿٥﴾
بىزنىڭ ئايەتلىرىمىزگە كۈچىنىڭ بېرىچە قارشى تۇرغانلار قاتتىق ئازابقا دۇچار بولىدۇ
وَيَرَى ٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ ٱلَّذِىٓ أُنزِلَ إِلَيْكَ مِن رَّبِّكَ هُوَ ٱلْحَقَّ وَيَهْدِىٓ إِلَىٰ صِرَٰطِ ٱلْعَزِيزِ ٱلْحَمِيدِ ﴿٦﴾
ئىلىم بېرىلگەنلەر (يەنى ساھابىلەر ۋە كېيىنكى ئەمەل قىلغۇچى ئۆلىمالار) ساڭا پەرۋەردىگارىڭ تەرىپىدىن نازىل قىلىنغان قۇرئاننىڭ شۈبھىسىز ھەق ئىكەنلىكىنى، (ئىنسانلارنى) غالىب، ھەمدۇساناغا لايىق اﷲ نىڭ يولىغا باشلايدىغانلىقىنى بىلىدۇ
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ هَلْ نَدُلُّكُمْ عَلَىٰ رَجُلٍ يُنَبِّئُكُمْ إِذَا مُزِّقْتُمْ كُلَّ مُمَزَّقٍ إِنَّكُمْ لَفِى خَلْقٍ جَدِيدٍ ﴿٧﴾
ھەقىقەتنى رەت قىلغۇچىلار (بىر - بىرىگە) ئېيتىدۇ: «سىلەرگە سىلەر (قەبرىلەردە چىرىپ) تىتما - تالاڭ بولۇپ كەتكەندىن كېيىن چوقۇم يېڭىدىن يارىتىلىسىلەر دەپ خەۋەر بېرىدىغان بىر ئادەمنى كۆرسىتىپ قويايلىمۇ؟
أَفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا أَم بِهِۦ جِنَّةٌۢ ۗ بَلِ ٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْـَٔاخِرَةِ فِى ٱلْعَذَابِ وَٱلضَّلَـٰلِ ٱلْبَعِيدِ ﴿٨﴾
ئۇ اﷲ قا يالغاننى توقۇدىمۇ؟ يا ئۇ ئېلىشىپ قالدىمۇ؟» ھەرگىز ئۇنداق ئەمەس، ئاخىرەتكە ئىمان كەلتۈرمەيدىغانلار ئازابتا ۋە چوڭقۇر گۇمراھلىقتىدۇر
أَفَلَمْ يَرَوْا۟ إِلَىٰ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُم مِّنَ ٱلسَّمَآءِ وَٱلْأَرْضِ ۚ إِن نَّشَأْ نَخْسِفْ بِهِمُ ٱلْأَرْضَ أَوْ نُسْقِطْ عَلَيْهِمْ كِسَفًا مِّنَ ٱلسَّمَآءِ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّكُلِّ عَبْدٍ مُّنِيبٍ ﴿٩﴾
ئۇلار ئالدىدىكى ۋە ئارقىسىدىكى (ئۇلارنى ئوراپ تۇرغان) ئاسمان بىلەن زېمىنغا قارىمامدۇ؟ ئەگەر خالىساق ئۇلارنى يەرگە يۇتقۇزىمىز، ياكى ئاسماننى ئۇلارنىڭ ئۈستىگە پارچە - پارچە قىلىپ چۈشۈرىمىز. بۇنىڭدا اﷲ قا قايتقۇچى ھەر بەندە ئۈچۈن ئەلۋەتتە (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان) دەلىل بار
۞ وَلَقَدْ ءَاتَيْنَا دَاوُۥدَ مِنَّا فَضْلًا ۖ يَـٰجِبَالُ أَوِّبِى مَعَهُۥ وَٱلطَّيْرَ ۖ وَأَلَنَّا لَهُ ٱلْحَدِيدَ ﴿١٠﴾
بىز داۋۇدقا دەرگاھىمىزدىن ئىشەنچلىك ھالدا پەزل (يەنى خۇدانىڭ ئەلچىسىلىك، زەبۇر، تاغلارنىڭ ۋە قۇشلارنىڭ بويسۇنۇشى، تۆمۈرنىڭ يۇمشاق بولۇپ بېرىشى، ساۋۇت ياساشنى بىلدۈرۈشلەر) ئاتا قىلدۇق. «ئى تاغلار! ئى قۇشلار! داۋۇت بىلەن بىرلىكتە تەسبىھ ئېيتىڭلار» (دېدۇق). تۆمۈرنى داۋۇدقا يۇمشاق قىلىپ بەردۇق
أَنِ ٱعْمَلْ سَـٰبِغَـٰتٍ وَقَدِّرْ فِى ٱلسَّرْدِ ۖ وَٱعْمَلُوا۟ صَـٰلِحًا ۖ إِنِّى بِمَا تَعْمَلُونَ بَصِيرٌ ﴿١١﴾
(بىز ئۇنىڭغا ئېيتتۇق) «مۇكەممەل ساۋۇتلارنى ياسىغىن، ساۋۇتلارنى ياساشتا (ھالقىلىرىنى بىر - بىرىگە) تەكشى قىلغىن، (ئى داۋۇد ئائىلىسىدىكىلەر!) ياخشى ئىش قىلىڭلار، مەن ئىشەنچلىك ھالدا سىلەرنىڭ ئەمەللىرىڭلارنى كۆرۈپ تۇرغۇچىمەن»
وَلِسُلَيْمَـٰنَ ٱلرِّيحَ غُدُوُّهَا شَهْرٌ وَرَوَاحُهَا شَهْرٌ ۖ وَأَسَلْنَا لَهُۥ عَيْنَ ٱلْقِطْرِ ۖ وَمِنَ ٱلْجِنِّ مَن يَعْمَلُ بَيْنَ يَدَيْهِ بِإِذْنِ رَبِّهِۦ ۖ وَمَن يَزِغْ مِنْهُمْ عَنْ أَمْرِنَا نُذِقْهُ مِنْ عَذَابِ ٱلسَّعِيرِ ﴿١٢﴾
سۇلەيمانغا شامالنى (بويسۇندۇرۇپ بەردۇق)، شامال چۈشتىن ئىلگىرى بىر ئايلىق مۇساپىنى، چۈشتىن كېيىن بىر ئايلىق مۇساپىنى باساتتى، ئۇنىڭغا مىس (چىقىدىغان) بۇلاقنى ئاققۇزۇپ بەردۇق، بەزى جىنلار پەرۋەردىگارىنىڭ ئەمرى بويىچە سۇلەيماننىڭ ئالدىدا ئىشلەيتتى، ئۇلاردىن كىمكى بىزنىڭ ئەمرىمىزگە خىلاپلىق قىلىدىكەن، ئۇنىڭغا بىز دوزاخ ئازابىنى تېتىتىمىز
يَعْمَلُونَ لَهُۥ مَا يَشَآءُ مِن مَّحَـٰرِيبَ وَتَمَـٰثِيلَ وَجِفَانٍ كَٱلْجَوَابِ وَقُدُورٍ رَّاسِيَـٰتٍ ۚ ٱعْمَلُوٓا۟ ءَالَ دَاوُۥدَ شُكْرًا ۚ وَقَلِيلٌ مِّنْ عِبَادِىَ ٱلشَّكُورُ ﴿١٣﴾
ئۇلار سۇلەيمانغا ئۇ خالىغان كاتتا سارايلارنى، (مىستىن - ئەينەكتىن ئاجايىپ) ھەيكەللەرنى، كۆلدەك چوڭ لېگەنلەرنى، (چوڭلۇقتىن) مىدىرلىمايدىغان قازانلارنى ياسايتتى، (ئېيتتۇقكى) «ئى داۋۇد ئائىلىسىدىكىلەر! (اﷲ نىڭ بۇ چوڭ نېمەتلىرىگە) شۈكۈر قىلىڭلار»، مېنىڭ بەندىلىرىمدىن (اﷲ نىڭ نېمەتلىرىگە) شۈكۈر قىلغۇچى بەك ئاز
فَلَمَّا قَضَيْنَا عَلَيْهِ ٱلْمَوْتَ مَا دَلَّهُمْ عَلَىٰ مَوْتِهِۦٓ إِلَّا دَآبَّةُ ٱلْأَرْضِ تَأْكُلُ مِنسَأَتَهُۥ ۖ فَلَمَّا خَرَّ تَبَيَّنَتِ ٱلْجِنُّ أَن لَّوْ كَانُوا۟ يَعْلَمُونَ ٱلْغَيْبَ مَا لَبِثُوا۟ فِى ٱلْعَذَابِ ٱلْمُهِينِ ﴿١٤﴾
سۇلەيماننىڭ ۋاپاتىنى ھۆكۈم قىلغىنىمىزدا (يەنى ئۇنى ۋاپات قىلدۇرغىنىمىزدا) سۇلەيماننىڭ ئۆلگەنلىكىدىن جىنلارنى پەقەت ئۇنىڭ ھاسىسىنى يېگەن قۇرۇتلار خەۋەردار قىلدى. سۇلەيمان (ھاسىسىنى قۇرۇت يەپ) يىقىلغان چاغدا جىنلارغا ئېنىق بولدىكى، ئەگەر ئۇلار غەيبنى بىلىدىغان بولسا ئىدى، خار قىلغۇچى ئازابتا (يەنى شۇنچە ئۇزاق ۋاقىت ئېغىر ئەمگەكتە) قالمىغان بولاتتى
لَقَدْ كَانَ لِسَبَإٍ فِى مَسْكَنِهِمْ ءَايَةٌ ۖ جَنَّتَانِ عَن يَمِينٍ وَشِمَالٍ ۖ كُلُوا۟ مِن رِّزْقِ رَبِّكُمْ وَٱشْكُرُوا۟ لَهُۥ ۚ بَلْدَةٌ طَيِّبَةٌ وَرَبٌّ غَفُورٌ ﴿١٥﴾
شۈبھىسىزكى، سەبەئ قەۋمىگە ئۇلارنىڭ تۇرىدىغان جايىدا (اﷲ نىڭ قۇدرىتىنى كۆرسىتىدىغان بۈيۈك) دەلىل بار ئىدى، ئۇنىڭ ئوڭ تەرىپىدە بىر تۈركۈم باغلار، سول تەرىپىدە بىر تۈركۈم باغلار بار ئىدى. (ئۇلارغا ئېيتتۇقكى) «پەرۋەردىگارىڭلار بەرگەن رىزىقتىن يەڭلار ۋە ئۇنىڭغا شۈكۈر قىلىڭلار، (سىلەرنىڭ جايىڭلار) بەلەن جايدۇر، (سىلەرگە رىزىق بەرگەن پەرۋەردىگارىڭلار شۈكۈر قىلغۇچىنى) مەغپىرەت قىلغۇچى پەرۋەردىگاردۇر
فَأَعْرَضُوا۟ فَأَرْسَلْنَا عَلَيْهِمْ سَيْلَ ٱلْعَرِمِ وَبَدَّلْنَـٰهُم بِجَنَّتَيْهِمْ جَنَّتَيْنِ ذَوَاتَىْ أُكُلٍ خَمْطٍ وَأَثْلٍ وَشَىْءٍ مِّن سِدْرٍ قَلِيلٍ ﴿١٦﴾
ئۇلار (شۈكۈردىن) باش تارتتى، شۇنىڭ بىلەن ئۇلارغا قاتتىق سەلنى ئەۋەتتۇق. ئۇلارنىڭ باغلىرىنى ئاچچىق مېۋىلىك، يۇلغۇنلۇق، ئاز - تولا سىدرىلىق (يەنى ياۋا چىلانلىق) باغلارغا ئايلاندۇردۇق
ذَٰلِكَ جَزَيْنَـٰهُم بِمَا كَفَرُوا۟ ۖ وَهَلْ نُجَـٰزِىٓ إِلَّا ٱلْكَفُورَ ﴿١٧﴾
ئۇلار كۇفرىلىق قىلغانلىقلىرى ئۈچۈن مۇشۇنداق جازالىدۇق، بىز پەقەت كۇفرىلىق قىلغۇچىلارنىلا جازالايمىز
وَجَعَلْنَا بَيْنَهُمْ وَبَيْنَ ٱلْقُرَى ٱلَّتِى بَـٰرَكْنَا فِيهَا قُرًى ظَـٰهِرَةً وَقَدَّرْنَا فِيهَا ٱلسَّيْرَ ۖ سِيرُوا۟ فِيهَا لَيَالِىَ وَأَيَّامًا ءَامِنِينَ ﴿١٨﴾
ئۇلار بىلەن بىز بەرىكەتلىك قىلغان قىشلاقلار ئارىسىدا، كۆرۈنۈپ تۇرىدىغان (تۇتىشىپ كەتكەن) نۇرغۇن قىشلاقلارنى بەرپا قىلدۇق، ئۇلار ئارسىدىكى مېڭىشنى ئۆلچەملىك قىلدۇق (يەنى سەپەردە يېمەك - ئىچمەك ئېلىۋالمىسىمۇ بولىدىغان، ئەتىگەن چىققان يولۇچى چۈشتە بىر ئۆتەڭگە يېتىپ بارسا، كەچتە بىر ئۆتەڭگە يېتىپ بارىدىغان قىلدۇق)، (ئۇلارغا ئېيتتۇقكى) «ئۇلارنىڭ ئارىسىدا كېچىلەردە ۋە كۈندۈزلەردە تىنچ - ئامان مېڭىڭلار!»
فَقَالُوا۟ رَبَّنَا بَـٰعِدْ بَيْنَ أَسْفَارِنَا وَظَلَمُوٓا۟ أَنفُسَهُمْ فَجَعَلْنَـٰهُمْ أَحَادِيثَ وَمَزَّقْنَـٰهُمْ كُلَّ مُمَزَّقٍ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّكُلِّ صَبَّارٍ شَكُورٍ ﴿١٩﴾
ئۇلار: «پەرۋەردىگارىمىز! سەپەرلىرىمىز ئارىسىنى (يەنى ئۆتەڭلار ئارىسىدىكى مۇساپىنى) يىراق قىلغىن» دېدى. ئۇلار ئۆزلىرىگە زۇلۇم قىلدى. ئۇلارنى بىز (كېيىنكى كىشىلەرگە) چۆچەك قىلىپ قالدۇردۇق. ئۇلارنى تىرىپىسىن قىلىۋەتتۇق، بۇنىڭغا ھەر بىر (بالاغا) سەۋر قىلغۇچى، (نېمەتكە) شۈكۈر قىلغۇچى ئۈچۈن نۇرغۇن ئىبرەتلەر بار
وَلَقَدْ صَدَّقَ عَلَيْهِمْ إِبْلِيسُ ظَنَّهُۥ فَٱتَّبَعُوهُ إِلَّا فَرِيقًا مِّنَ ٱلْمُؤْمِنِينَ ﴿٢٠﴾
ئىبلىس ئۇلارنى (ئازدۇرۇشتىن ئىبارەت) گۇمانىنى ئىشەنچلىك ھالدا ئىشقا ئاشۇردى. چۈنكى بىر تۈركۈم مۆمىنلەردىن باشقا ئۇلارنىڭ ھەممىسى ئىبلىسقا ئەگەشتى
وَمَا كَانَ لَهُۥ عَلَيْهِم مِّن سُلْطَـٰنٍ إِلَّا لِنَعْلَمَ مَن يُؤْمِنُ بِٱلْـَٔاخِرَةِ مِمَّنْ هُوَ مِنْهَا فِى شَكٍّ ۗ وَرَبُّكَ عَلَىٰ كُلِّ شَىْءٍ حَفِيظٌ ﴿٢١﴾
ئىبلىسنىڭ ئۇلار (يەنى ئەگەشكۈچىلەر) ئۈستىدىن (ۋەسۋەسە قىلىش بىلەن) ھۆكۈمرانلىق قىلىش پەقەت بىزنىڭ ئاخىرەتكە ئىمان ئېيتقۇچىلار بىلەن ئۇنىڭغا شەك كەلتۈرگۈچىلەرنى بىلىۋېلىشىمىز ئۈچۈندۇر، پەرۋەردىگارىڭ ھەممە نەرسىنى كۆزەتكۈچىدۇر
قُلِ ٱدْعُوا۟ ٱلَّذِينَ زَعَمْتُم مِّن دُونِ ٱللَّهِ ۖ لَا يَمْلِكُونَ مِثْقَالَ ذَرَّةٍ فِى ٱلسَّمَـٰوَٰتِ وَلَا فِى ٱلْأَرْضِ وَمَا لَهُمْ فِيهِمَا مِن شِرْكٍ وَمَا لَهُۥ مِنْهُم مِّن ظَهِيرٍ ﴿٢٢﴾
ئېيتقىنكى، «سىلەر اﷲ نى قويۇپ مەبۇد دەپ قارىغىنىڭلارنى چاقىرىڭلار، ئۇلار ئاسمانلاردا ۋە زېمىندا زەررە چاغلىق نەرسىگە ئىگە بولالمايدۇ، ئۇلارنىڭ ئاسمان - زېمىننىڭ (يارىتىلىشى ۋە باشقۇرۇلۇشىدا اﷲ قا) ھېچ شېرىكچىلىگى يوق، ئۇلارنىڭ ئىچىدە اﷲ قا (ئاسمان - زېمىننى ئىدارە قىلىشتا ياردەملىشىدىغان) ھېچ ياردەمچىمۇ يوق»
وَلَا تَنفَعُ ٱلشَّفَـٰعَةُ عِندَهُۥٓ إِلَّا لِمَنْ أَذِنَ لَهُۥ ۚ حَتَّىٰٓ إِذَا فُزِّعَ عَن قُلُوبِهِمْ قَالُوا۟ مَاذَا قَالَ رَبُّكُمْ ۖ قَالُوا۟ ٱلْحَقَّ ۖ وَهُوَ ٱلْعَلِىُّ ٱلْكَبِيرُ ﴿٢٣﴾
اﷲ ئىزنى بەرگەن ئادەمدىن باشقىغا اﷲ نىڭ دەرگاھىدا شاپائەت پايدا بەرمەيدۇ. شاپائەت قىلغۇچىلارنىڭ دىللىرىدىن قورقۇنچ كۆتۈرۈلگەندە، ئۇلار ئاندىن (بەزىسى بەزىسىگە): «پەرۋەردىگارىڭلار (شاپائەت توغرولۇق) نېمە دېدى؟» دەيدۇ. ئۇلار (جاۋابەن): «ھەقنى (دېدى) (يەنى مۆمىنلەرگە شاپائەت قىلىشقا رۇخسەت قىلدى)» دەيدۇ. اﷲ يۈكسەكتۇر، كاتتىدۇر
۞ قُلْ مَن يَرْزُقُكُم مِّنَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۖ قُلِ ٱللَّهُ ۖ وَإِنَّآ أَوْ إِيَّاكُمْ لَعَلَىٰ هُدًى أَوْ فِى ضَلَـٰلٍ مُّبِينٍ ﴿٢٤﴾
ئېيتقىنكى، «سىلەرگە ئاسمانلاردىن ۋە زېمىندىن كىم رىزىق بېرىپ تۇرىدۇ؟» ئېيتقىنكى، «اﷲ (رىزىق بېرىپ تۇرىدۇ)، شۈبھىسىزكى، بىز ياكى سىلەر چوقۇم ھىدايەتتىدۇرمىز، يا روشەن گۇمراھلىقتىدۇرمىز»
قُل لَّا تُسْـَٔلُونَ عَمَّآ أَجْرَمْنَا وَلَا نُسْـَٔلُ عَمَّا تَعْمَلُونَ ﴿٢٥﴾
ئېيتقىنكى، «بىزنىڭ قىلغان گۇناھىمىز ئۈچۈن سىلەر سوراققا تارتىلمايسىلەر، سىلەرنىڭ قىلمىشىڭلار ئۈچۈن بىز سوراققا تارتىلمايمىز»
قُلْ يَجْمَعُ بَيْنَنَا رَبُّنَا ثُمَّ يَفْتَحُ بَيْنَنَا بِٱلْحَقِّ وَهُوَ ٱلْفَتَّاحُ ٱلْعَلِيمُ ﴿٢٦﴾
ئېيتقىنكى، «پەرۋەردىگارىمىز بىزنى (قىيامەت كۈنى) يىغىدۇ. ئاندىن بىزنىڭ ئارىمىزدا ھەقلىق بىلەن ھۆكۈم چىقىرىدۇ، اﷲ ئادىل ھۆكۈم قىلغۇچىدۇر، (خالايىقنىڭ ئەھۋالىنى) بىلگۈچىدۇر»
قُلْ أَرُونِىَ ٱلَّذِينَ أَلْحَقْتُم بِهِۦ شُرَكَآءَ ۖ كَلَّا ۚ بَلْ هُوَ ٱللَّهُ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٧﴾
ئېيتقىنكى، «ماڭا ئېيتىپ بېقىڭلارچۇ، سىلەرنىڭ اﷲ نىڭ شېرىكى دەپ ئاتىۋالغىنىڭلار ھەرگىزمۇ ئۇنداق ئەمەس، بەلكى ئۇ - اﷲ غالىبتۇر، ھېكمەت بىلەن ئىش قىلغۇچىدۇر»
وَمَآ أَرْسَلْنَـٰكَ إِلَّا كَآفَّةً لِّلنَّاسِ بَشِيرًا وَنَذِيرًا وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَعْلَمُونَ ﴿٢٨﴾
سېنى بىز پەقەت پۈتۈن ئىنسانلار ئۈچۈن (مۆمىنلەرگە مەڭگۈلۈك باغلار بىلەن) خۇش خەۋەر بەرگۈچى ۋە (ھەقىقەتنى رەت قىلغۇچىلارنى ئازاب بىلەن) ئاگاھلاندۇرغۇچى قىلىپ ئەۋەتتۇق. ۋە لېكىن ئىنسانلارنىڭ تولىسى (بۇنى) بىلمەيدۇ
وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٢٩﴾
ئۇلار: «ئەگەر سىلەر راستچىل بولساڭلار، بۇ ۋەدە (يەنى سىلەر بىزنى قورقۇتۇۋاتقان ئازاب) قاچان ئىشقا ئاشىدۇ؟» دەيدۇ
قُل لَّكُم مِّيعَادُ يَوْمٍ لَّا تَسْتَـْٔخِرُونَ عَنْهُ سَاعَةً وَلَا تَسْتَقْدِمُونَ ﴿٣٠﴾
ئېيتقىنكى، «سىلەرگە (ئازاب ئۈچۈن) ۋەدە قىلىنغان بىر كۈن باركى، ئۇنىڭدىن بىردەممۇ ئىلگىرى - كېيىن بولمايسىلەر»
وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لَن نُّؤْمِنَ بِهَـٰذَا ٱلْقُرْءَانِ وَلَا بِٱلَّذِى بَيْنَ يَدَيْهِ ۗ وَلَوْ تَرَىٰٓ إِذِ ٱلظَّـٰلِمُونَ مَوْقُوفُونَ عِندَ رَبِّهِمْ يَرْجِعُ بَعْضُهُمْ إِلَىٰ بَعْضٍ ٱلْقَوْلَ يَقُولُ ٱلَّذِينَ ٱسْتُضْعِفُوا۟ لِلَّذِينَ ٱسْتَكْبَرُوا۟ لَوْلَآ أَنتُمْ لَكُنَّا مُؤْمِنِينَ ﴿٣١﴾
ھەقىقەتنى رەت قىلغۇچىلار: «بۇ قۇرئانغا ۋە ئۇنىڭدىن ئىلگىرىكى مۇقەددەس كىتاب-ۋەھىيلارغا ھەرگىز ئىشەنمەيمىز» دەيدۇ. ئەگەر زالىملارنى پەرۋەردىگارىڭنىڭ دەرگاھىدا توختىتىپ قويۇلغان چاغدا بەزىسى بەزىسى بىلەن جېدەللىشىۋاتقاندا كۆرسەڭ ئىدىڭ (ئەلۋەتتە قورقۇنچلۇق ھالنى كۆرەتتىڭ)، بوزەك قىلىنغانلار چوڭچىلىق قىلغانلارغا: «ئەگەر سىلەر بولمىساڭلار بىز چوقۇم ئىمان ئېيتاتتۇق» دەيدۇ
قَالَ ٱلَّذِينَ ٱسْتَكْبَرُوا۟ لِلَّذِينَ ٱسْتُضْعِفُوٓا۟ أَنَحْنُ صَدَدْنَـٰكُمْ عَنِ ٱلْهُدَىٰ بَعْدَ إِذْ جَآءَكُم ۖ بَلْ كُنتُم مُّجْرِمِينَ ﴿٣٢﴾
چوڭچىلىق قىلغانلار بوزەك قىلىنغانلارغا: «سىلەرگە ھىدايەت كەلگەندىن كېيىن بىز سىلەرنى ئۇنىڭدىن توستۇقمۇ؟ ھەرگىز توسمىدۇق، سىلەر ئۆزۈڭلار گۇناھ قىلغۇچى بولدۇڭلار» دەيدۇ
وَقَالَ ٱلَّذِينَ ٱسْتُضْعِفُوا۟ لِلَّذِينَ ٱسْتَكْبَرُوا۟ بَلْ مَكْرُ ٱلَّيْلِ وَٱلنَّهَارِ إِذْ تَأْمُرُونَنَآ أَن نَّكْفُرَ بِٱللَّهِ وَنَجْعَلَ لَهُۥٓ أَندَادًا ۚ وَأَسَرُّوا۟ ٱلنَّدَامَةَ لَمَّا رَأَوُا۟ ٱلْعَذَابَ وَجَعَلْنَا ٱلْأَغْلَـٰلَ فِىٓ أَعْنَاقِ ٱلَّذِينَ كَفَرُوا۟ ۚ هَلْ يُجْزَوْنَ إِلَّا مَا كَانُوا۟ يَعْمَلُونَ ﴿٣٣﴾
بوزەك قىلىنغانلار چوڭچىلىق قىلغانلارغا: «ھەرگىز ئۇنداق ئەمەس، (سىلەرنىڭ بىزگە) كېچە - كۈندۈز قىلغان ھىيلە - مىكرىڭلار (بىزنى ئىماندىن توستى)، ئۆز ۋاقتىدا سىلەر بىزنى اﷲ نى ئىنكار قىلىشقا ۋە ئۇنىڭغا شېرىك كەلتۈرۈشكە بۇيرۇيتتۇڭلار» دەيدۇ. ئازابنى كۆرگەن چاغدا، ئۇلار (ھەر ئىككى گۇرۇھ ئىمان ئېيتمىغانلىقلىرىغا) ئىچىدە پۇشايمان قىلىدۇ، ھەقىقەتنى رەت قىلغۇچىلارنىڭ بويۇنلىرىغا تاقاقلارنى سالىمىز، ئۇلارغا پەقەت قىلمىشلىرىنىڭ جازاسى بېرىلىدۇ
وَمَآ أَرْسَلْنَا فِى قَرْيَةٍ مِّن نَّذِيرٍ إِلَّا قَالَ مُتْرَفُوهَآ إِنَّا بِمَآ أُرْسِلْتُم بِهِۦ كَـٰفِرُونَ ﴿٣٤﴾
قايسى شەھەرگە بولمىسۇن، بىرەر ئاگاھلاندۇرغۇچى ئەۋەتسەكلا ئۇنىڭ دۆلەتمەن ئادەملىرى: «شۈبھىسىزكى، بىز سىلەرنىڭ خۇدانىڭ ئەلچىسىلىكىڭلارنى ئىنكار قىلىمىز» دېدى
وَقَالُوا۟ نَحْنُ أَكْثَرُ أَمْوَٰلًا وَأَوْلَـٰدًا وَمَا نَحْنُ بِمُعَذَّبِينَ ﴿٣٥﴾
ئۇلار يەنە: «بىزنىڭ ماللىرىمىز ۋە بالىلىرىمىز ئەڭ كۆپ، بىز ئازابلانمايمىز» دېدى
قُلْ إِنَّ رَبِّى يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ وَيَقْدِرُ وَلَـٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يَعْلَمُونَ ﴿٣٦﴾
ئېيتقىنكى، «ئىشەنچلىك ھالدا، پەرۋەردىگارىم كىمنىڭ رىزقىنى كەڭ قىلىشنى خالىسا ئۇنى كەڭ قىلىدۇ، (كىمنىڭ رىزقىنى تار قىلىشنى خالىسا ئۇنى) تار قىلىدۇ، لېكىن كىشىلەرنىڭ تولىسى (بۇنى) ئۇقمايدۇ»
وَمَآ أَمْوَٰلُكُمْ وَلَآ أَوْلَـٰدُكُم بِٱلَّتِى تُقَرِّبُكُمْ عِندَنَا زُلْفَىٰٓ إِلَّا مَنْ ءَامَنَ وَعَمِلَ صَـٰلِحًا فَأُو۟لَـٰٓئِكَ لَهُمْ جَزَآءُ ٱلضِّعْفِ بِمَا عَمِلُوا۟ وَهُمْ فِى ٱلْغُرُفَـٰتِ ءَامِنُونَ ﴿٣٧﴾
ماللىرىڭلار ۋە بالىلىرىڭلار سىلەرنى بىزگە يېقىنلاشتۇرالمايدۇ، پەقەت ئىمان ئېيتقان ۋە ياخشى ئەمەل قىلغانلارغا قىلغان ئەمەللىرى ئۈچۈن ھەسسىلەپ ساۋاب بېرىلىدۇ، ئۇلار (مەڭگۈلۈك باغلارنىڭ) ئېسىل ئۆيلىرىدە (ھەر قانداق كۆڭۈلسىزلىكتىن) ئەمىن بولغان ھالدا تۇرىدۇ
وَٱلَّذِينَ يَسْعَوْنَ فِىٓ ءَايَـٰتِنَا مُعَـٰجِزِينَ أُو۟لَـٰٓئِكَ فِى ٱلْعَذَابِ مُحْضَرُونَ ﴿٣٨﴾
بىزنىڭ ئايەتلىرىمىزگە كۈچىنىڭ بېرىچە قارشى تۇرغانلار ئازابقا دۇچار قىلىنغۇچىلاردۇر
قُلْ إِنَّ رَبِّى يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ مِنْ عِبَادِهِۦ وَيَقْدِرُ لَهُۥ ۚ وَمَآ أَنفَقْتُم مِّن شَىْءٍ فَهُوَ يُخْلِفُهُۥ ۖ وَهُوَ خَيْرُ ٱلرَّٰزِقِينَ ﴿٣٩﴾
ئېيتقىنكى، «ئىشەنچلىك ھالدا، پەرۋەردىگارىم قايسى بەندىسىنىڭ رىزقىنى كەڭ قىلىشنى خالىسا ئۇنى كەڭ قىلىدۇ، (قايسى بەندىسىنىڭ رىزقىنى تار قىلىشنى خالىسا ئۇنى) تار قىلىدۇ، (اﷲ نىڭ يولىدا) بەرگەن نەرسەڭلارنىڭ ئورنىنى اﷲ تولدۇرۇپ بېرىدۇ، ئۇ رىزىق بەرگۈچىلەرنىڭ ياخشىسىدۇر»
وَيَوْمَ يَحْشُرُهُمْ جَمِيعًا ثُمَّ يَقُولُ لِلْمَلَـٰٓئِكَةِ أَهَـٰٓؤُلَآءِ إِيَّاكُمْ كَانُوا۟ يَعْبُدُونَ ﴿٤٠﴾
ئۇ كۈندە اﷲ ئۇلار (يەنى مۇشرىكلار) نىڭ ھەممىسىنى (ھېساب ئېلىش ئۈچۈن) يىغىدۇ، ئاندىن خۇدانىڭ ئەلچى پەرىشتىلىرىگە: «بۇ كىشىلەر سىلەرگە چوقۇنغانمىدى؟» دەيدۇ
قَالُوا۟ سُبْحَـٰنَكَ أَنتَ وَلِيُّنَا مِن دُونِهِم ۖ بَلْ كَانُوا۟ يَعْبُدُونَ ٱلْجِنَّ ۖ أَكْثَرُهُم بِهِم مُّؤْمِنُونَ ﴿٤١﴾
ئۇلار ئېيتىدۇ: «سەن پاكتۇرسەن، بىزنىڭ دوستىمىز ئۇلار ئەمەس، سەن. (ئۇلار بىزگە چوقۇنمايتتى) بەلكى جىنغا چوقۇناتتى. ئۇلارنىڭ تولىسى جىنغا ئىمان كەلتۈرەتتى»
فَٱلْيَوْمَ لَا يَمْلِكُ بَعْضُكُمْ لِبَعْضٍ نَّفْعًا وَلَا ضَرًّا وَنَقُولُ لِلَّذِينَ ظَلَمُوا۟ ذُوقُوا۟ عَذَابَ ٱلنَّارِ ٱلَّتِى كُنتُم بِهَا تُكَذِّبُونَ ﴿٤٢﴾
بۈگۈن سىلەر بىر - بىرىڭلارغا پايدا - زىيان يەتكۈزەلمەيسىلەر، بىز (اﷲ تىن غەيرىيگە چوقۇنغۇچى) زالىملارغا: «سىلەر چىنپۈتمىگەن دوزاخ ئازابىنى تېتىڭلار» دەيمىز
وَإِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُنَا بَيِّنَـٰتٍ قَالُوا۟ مَا هَـٰذَآ إِلَّا رَجُلٌ يُرِيدُ أَن يَصُدَّكُمْ عَمَّا كَانَ يَعْبُدُ ءَابَآؤُكُمْ وَقَالُوا۟ مَا هَـٰذَآ إِلَّآ إِفْكٌ مُّفْتَرًى ۚ وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لِلْحَقِّ لَمَّا جَآءَهُمْ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ﴿٤٣﴾
ئۇلارغا بىزنىڭ روشەن ئايەتلىرىمىز ئوقۇپ بېرىلگەندە، ئۇلار: «بۇ پەقەت سىلەرنى ئاتا - بوۋاڭلار چوقۇنغان بۇتلارغا چوقۇنۇشتىن توسماقچى بولغان ئادەمدۇر» دەيدۇ ھەمدە ئۇلار: «بۇ پەقەت توقۇلغان يالغان سۆزدۇر» دەيدۇ. ھەقىقەتنى رەت قىلغۇچىلارغا ھەقىقەت يېتىپ كەلگەندە، ئۇلار: «بۇ پەقەت ئوپئوچۇق سېھىردۇر» دېدى
وَمَآ ءَاتَيْنَـٰهُم مِّن كُتُبٍ يَدْرُسُونَهَا ۖ وَمَآ أَرْسَلْنَآ إِلَيْهِمْ قَبْلَكَ مِن نَّذِيرٍ ﴿٤٤﴾
ئۇلارغا ئوقۇيدىغان ھېچقانداق مۇقەددەس كىتاب-ۋەھىيلارنى ئاتا قىلغىنىمىز يوق. سەندىن ئىلگىرى ئۇلارغا ھېچقانداق خۇدانىڭ ئەلچىسىمۇ ئەۋەتكىنىمىز يوق
وَكَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ وَمَا بَلَغُوا۟ مِعْشَارَ مَآ ءَاتَيْنَـٰهُمْ فَكَذَّبُوا۟ رُسُلِى ۖ فَكَيْفَ كَانَ نَكِيرِ ﴿٤٥﴾
ئۇلاردىن ئىلگىرى ئۆتكەنلەر (ھەقنى) ئىنكار قىلغان، ئۇلار (يەنى مەككە كۇففارلىرى) ئۇلار (يەنى ئىلگىرىكىلەر) ئېرىشكەن كۈچ - قۇۋۋەتنىڭ ئوندىن بىرىگىمۇ ئېرىشمىگەن، ئۇلار مېنىڭ خۇدانىڭ ئەلچىسىلىرىمنى ئىنكار قىلدى. ئۇلارغا مېنىڭ جازايىم قانداق بولدى؟
۞ قُلْ إِنَّمَآ أَعِظُكُم بِوَٰحِدَةٍ ۖ أَن تَقُومُوا۟ لِلَّهِ مَثْنَىٰ وَفُرَٰدَىٰ ثُمَّ تَتَفَكَّرُوا۟ ۚ مَا بِصَاحِبِكُم مِّن جِنَّةٍ ۚ إِنْ هُوَ إِلَّا نَذِيرٌ لَّكُم بَيْنَ يَدَىْ عَذَابٍ شَدِيدٍ ﴿٤٦﴾
ئېيتقىنكى، «مەن سىلەرگە بىر ئىشنى تەۋسىيە قىلىمەن: سىلەر خۇدالىق ئۈچۈن ئىككىدىن، يا بىردىن تۇرۇڭلار (يەنى اﷲ نىڭ رازىلىقى ئۈچۈن يىغىلىپ ياكى يەككە ھالدا ھەقنى تىلەڭلار)، ئاندىن پىكىر يۈرگۈزۈڭلار». ھەمراھىڭلاردا (يەنى مۇھەممەدتە) ھېچ مەجنۇنلۇق يوق، پەقەت ئۇ قاتتىق ئازاب نازىل بولۇشتىن ئىلگىرى سىلەرگە ئەۋەتىلگەن بىر ئاگاھلاندۇرغۇچىدۇر
قُلْ مَا سَأَلْتُكُم مِّنْ أَجْرٍ فَهُوَ لَكُمْ ۖ إِنْ أَجْرِىَ إِلَّا عَلَى ٱللَّهِ ۖ وَهُوَ عَلَىٰ كُلِّ شَىْءٍ شَهِيدٌ ﴿٤٧﴾
ئېيتقىنكى، «سىلەردىن ھېچقانداق ھەق تەلەپ قىلغىنىم يوق، ئۇ سىلەرگە بولسۇن؛ ئەجرىمنى پەقەت اﷲ تىن تىلەيمەن. اﷲ ھەممە نەرسىنى كۆزىتىپ تۇرغۇچىدۇر»
قُلْ إِنَّ رَبِّى يَقْذِفُ بِٱلْحَقِّ عَلَّـٰمُ ٱلْغُيُوبِ ﴿٤٨﴾
ئېيتقىنكى، «پەرۋەردىگارىم، ئىشەنچلىك ھالدا ھەق ئارقىلىق (باتىلنى) ئاتىدۇ (يەنى پاكىتنى روشەن بايان قىلىدۇ)، اﷲ غەيبنى تامامەن بىلگۈچىدۇر»
قُلْ جَآءَ ٱلْحَقُّ وَمَا يُبْدِئُ ٱلْبَـٰطِلُ وَمَا يُعِيدُ ﴿٤٩﴾
ئېيتقىنكى، «ھەق (يەنى ئىسلام) كەلدى (باتىل يوقالدى)، باتىل (بىرەر مەخلۇقنى) پەيدا قىلالمايدۇ، (يوقالغان بىرەر مەخلۇقنى) ئەسلىگە كەلتۈرەلمەيدۇ»
قُلْ إِن ضَلَلْتُ فَإِنَّمَآ أَضِلُّ عَلَىٰ نَفْسِى ۖ وَإِنِ ٱهْتَدَيْتُ فَبِمَا يُوحِىٓ إِلَىَّ رَبِّىٓ ۚ إِنَّهُۥ سَمِيعٌ قَرِيبٌ ﴿٥٠﴾
ئېيتقىنكى، «ئەگەر ئازسام ئازغانلىقىمنىڭ زىيىنى ئۆزۈمگىدۇر، ئەگەر ھىدايەت تاپسام (بۇ) پەرۋەردىگارىمنىڭ ماڭا ۋەھيى قىلغان قۇرئانىنىڭ سەۋەبىدىندۇر، اﷲ ئىشەنچلىك ھالدا ئاڭلاپ تۇرغۇچىدۇر، يېقىندۇر»
وَلَوْ تَرَىٰٓ إِذْ فَزِعُوا۟ فَلَا فَوْتَ وَأُخِذُوا۟ مِن مَّكَانٍ قَرِيبٍ ﴿٥١﴾
ئەگەر سەن ئەينى ۋاقىتتا ئۇلارنىڭ قورقۇپ قاچىدىغان يەر تاپالمايۋاتقانلىقىنى، دوزاخقا (ئېلىپ بېرىلىش ئۈچۈن مەھشەرگاھقا) يېقىن جايدا تۇتۇلۇۋاتقانلىقىنى كۆرسەڭ (ئەلۋەتتە قورقۇنچلۇق ھالىنى كۆرەتتىڭ)
وَقَالُوٓا۟ ءَامَنَّا بِهِۦ وَأَنَّىٰ لَهُمُ ٱلتَّنَاوُشُ مِن مَّكَانٍۭ بَعِيدٍ ﴿٥٢﴾
ئۇلار (ئازابنى كۆرگەندە): «ئىمان ئېيتتۇق» دەيدۇ، ئۇلار قانداقمۇ يىراق جايدىن ئىماننى قولغا كەلتۈرەلەيدۇ
وَقَدْ كَفَرُوا۟ بِهِۦ مِن قَبْلُ ۖ وَيَقْذِفُونَ بِٱلْغَيْبِ مِن مَّكَانٍۭ بَعِيدٍ ﴿٥٣﴾
ئۇلار ئىلگىرى ئۇنى (يەنى خۇدانىڭ ئەلچىسىنى) ئىنكار قىلغان ئىدى. ئۇلار غەيب (ئىشلىرى) ئۈستىدە يىراق جايدىن قارىسىغا سۆزلەيدۇ
وَحِيلَ بَيْنَهُمْ وَبَيْنَ مَا يَشْتَهُونَ كَمَا فُعِلَ بِأَشْيَاعِهِم مِّن قَبْلُ ۚ إِنَّهُمْ كَانُوا۟ فِى شَكٍّ مُّرِيبٍۭ ﴿٥٤﴾
ئۇلار بىلەن ئۇلارنىڭ ئارزۇ قىلغان نەرسىلىرى (يەنى ئىمان ئېيتىش ۋە مەڭگۈلۈك باغلاركە كىرىش) ئارىسىدا توسالغۇ پەيدا قىلىندى. ئۇلارغا ئىلگىرى ئۇلارغا ئوخشاشلارغا قىلىنغاندەك قىلىندى. چۈنكى ئۇلار ھازىر ئىمان كەلتۈرگەن نەرسىلىرى ئۈستىدە (دۇنيادىكى چاغلىرىدا) چوڭقۇر شەكتە ئىدى
