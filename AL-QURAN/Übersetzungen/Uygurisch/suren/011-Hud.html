الر ۚ كِتَابٌ أُحْكِمَتْ آيَاتُهُ ثُمَّ فُصِّلَتْ مِن لَّدُنْ حَكِيمٍ خَبِيرٍ ﴿١﴾
ئەلىف، لام، را. بۇ، ھېكمەت بىلەن ئىش قىلغۇچى، ھەممىدىن خەۋەردار اﷲ تەرىپىدىن نازىل بولغان ئايەتلەرنىڭ تۈزۈلۈشى پۇختا، (ئەقىدە، ئەھكام ۋەز ۋە قىسسىلەر) تەپسىلىي بايان قىلىنغان مۇقەددەس كىتاب-ۋەھىيتۇر
أَلَّا تَعْبُدُوا إِلَّا اللَّهَ ۚ إِنَّنِي لَكُم مِّنْهُ نَذِيرٌ وَبَشِيرٌ ﴿٢﴾
(ئېيتقىنكى) «سىلەر پەقەت اﷲ قا ئىبادەت قىلىڭلار، شۈبھىسىزكى، مەن سىلەرگە اﷲ تەرىپىدىن ئەۋەتىلگەن، (ھەقىقەتنى رەت قىلغۇچى بولساڭلار سىلەرنى اﷲ نىڭ ئازابىدىن) ئاگاھلاندۇرغۇچىمەن، (ئىمان ئېيتساڭلار اﷲ نىڭ ساۋابى بىلەن سىلەرگە) خۇش خەۋەر بەرگۈچىمەن
وَأَنِ اسْتَغْفِرُوا رَبَّكُمْ ثُمَّ تُوبُوا إِلَيْهِ يُمَتِّعْكُم مَّتَاعًا حَسَنًا إِلَىٰ أَجَلٍ مُّسَمًّى وَيُؤْتِ كُلَّ ذِي فَضْلٍ فَضْلَهُ ۖ وَإِن تَوَلَّوْا فَإِنِّي أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ كَبِيرٍ ﴿٣﴾
سىلەر پەرۋەردىگارىڭلاردىن مەغپىرەت تەلەپ قىلىڭلار، ئاندىن ئۇنىڭغا تەۋبە قىلىڭلار، اﷲ سىلەرنى مۇئەييەن مۇددەتكىچە (يەنى ئۆمرۈڭلار ئاخىرلىشىپ، ئەجىلىڭلار يەتكۈچە ھاياتىي دۇنيادىن) ئوبدان بەھرىمەن قىلىدۇ، ياخشى ئىش قىلغۇچىغا قىلغان ياخشىلىقنىڭ ساۋابىنى بېرىدۇ، ئەگەر (ئىماندىن) يۈز ئۆرۈسەڭلار، (ئازابقا دۇچار بولىسىلەر) سىلەرنىڭ قىيامەت كۈنىنىڭ چوڭ ئازابىغا قېلىشىڭلاردىن ئەنسىرەيمەن
إِلَى اللَّهِ مَرْجِعُكُمْ ۖ وَهُوَ عَلَىٰ كُلِّ شَيْءٍ قَدِيرٌ ﴿٤﴾
(ئۆلگەندىن كېيىن) اﷲ نىڭ دەرگاھىغا قايتىسىلەر، اﷲ ھەر نەرسىگە قادىردۇر»
أَلَا إِنَّهُمْ يَثْنُونَ صُدُورَهُمْ لِيَسْتَخْفُوا مِنْهُ ۚ أَلَا حِينَ يَسْتَغْشُونَ ثِيَابَهُمْ يَعْلَمُ مَا يُسِرُّونَ وَمَا يُعْلِنُونَ ۚ إِنَّهُ عَلِيمٌ بِذَاتِ الصُّدُورِ ﴿٥﴾
راستتىنلا ئۇلار (يەنى ھەقىقەتنى رەت قىلغۇچىلار) (رەسۇلۇللاھ، قۇرئان ئوقۇۋاتقاندا ئۇنىڭ يېنىدىن ئۆتكەنلىرىدە) ئۇنىڭدىن يوشۇرۇش ئۈچۈن يۈز ئۆرۈيدۇ، راستلا ئۇلار كىيىملىرى بىلەن چۈمكىنىۋالغانلىرىدا، اﷲ ئۇلارنىڭ يوشۇرۇن ۋە ئاشكارا ئىشلىرىنى بىلىپ تۇرىدۇ، اﷲ ئىشەنچلىك ھالدا دىللاردىكىنى بىلگۈچىدۇر
وَمَا مِن دَابَّةٍ فِي الْأَرْضِ إِلَّا عَلَى اللَّهِ رِزْقُهَا وَيَعْلَمُ مُسْتَقَرَّهَا وَمُسْتَوْدَعَهَا ۚ كُلٌّ فِي كِتَابٍ مُّبِينٍ ﴿٦﴾
يەر يۈزىدىكى ھايۋانلارنىڭ ھەممىسىگە رىزىق بېرىشنى اﷲ (مەرھەمەت يۈزىسىدىن) ئۈستىگە ئالغان، اﷲ ئۇلارنىڭ تۇرار جايىنى ۋە (ئۆلگەندىن كېيىن) كۆمۈلىدىغان جاينى بىلىدۇ، ئۇلارنىڭ ھەممىسى روشەن مۇقەددەس كىتاب-ۋەھىيقا (يەنى لەۋھۇلمەھپۇزغا يېزىلغاندۇر
وَهُوَ الَّذِي خَلَقَ السَّمَاوَاتِ وَالْأَرْضَ فِي سِتَّةِ أَيَّامٍ وَكَانَ عَرْشُهُ عَلَى الْمَاءِ لِيَبْلُوَكُمْ أَيُّكُمْ أَحْسَنُ عَمَلًا ۗ وَلَئِن قُلْتَ إِنَّكُم مَّبْعُوثُونَ مِن بَعْدِ الْمَوْتِ لَيَقُولَنَّ الَّذِينَ كَفَرُوا إِنْ هَٰذَا إِلَّا سِحْرٌ مُّبِينٌ ﴿٧﴾
اﷲ ئاسمانلارنى ۋە زېمىننى ئالتە كۈندە ياراتتى. (ئاسمان - زېمىننى يارتىشتىن ئىلگىرى) اﷲ نىڭ ئەرشى سۇ ئۈستىدە ئىدى. سىلەرنىڭ قايسىڭلارنىڭ ئەمەلى ياخشى ئىكەنلىكىنى سىناش ئۈچۈن اﷲ (ئۇلارنى ياراتتى). (ئى مۇھەممەد!) ئەگەر سەن (مەككە كۇففارلىرىغا): «سىلەر ئۆلگەندىن كېيىن چوقۇم تىرىلىسىلەر!» دېسەڭ، ھەقىقەتنى رەت قىلغۇچىلار چوقۇم ئېيتىدۇ: «بۇ (سۆزنى ئىچىگە ئالغان قۇرئان) پەقەت روشەن سېھىردۇر»
وَلَئِنْ أَخَّرْنَا عَنْهُمُ الْعَذَابَ إِلَىٰ أُمَّةٍ مَّعْدُودَةٍ لَّيَقُولُنَّ مَا يَحْبِسُهُ ۗ أَلَا يَوْمَ يَأْتِيهِمْ لَيْسَ مَصْرُوفًا عَنْهُمْ وَحَاقَ بِهِم مَّا كَانُوا بِهِ يَسْتَهْزِئُونَ ﴿٨﴾
ئەگەر بىز ئۇلارغا نازىل بولىدىغان ئازابنى چەكلىك ۋاقىت كېچىكتۈرسەك، ئۇلار چوقۇم (مەسخىرە قىلىش يۈزىسىدىن): «ئازاب نېمىشقا چۈشمەيدۇ؟» دەيدۇ، راستلا ئۇلارغا اﷲ نىڭ ئازابى چۈشكەن كۈندە، ئۇلار قاچىدىغان يەر تاپالمايدۇ ئۇلار مەسخىرە قىلغان ئازاب ئۇلارنى قورشىۋالىدۇ
وَلَئِنْ أَذَقْنَا الْإِنسَانَ مِنَّا رَحْمَةً ثُمَّ نَزَعْنَاهَا مِنْهُ إِنَّهُ لَيَئُوسٌ كَفُورٌ ﴿٩﴾
ئەگەر بىز بىرەر ئىنسانغا رەھمىتىمىزنى تېتىتساق (يەنى نېمىتىمىزنى بەرسەك)، ئاندىن ئۇنى ئۇنىڭدىن تارتىۋالساق، ئۇ چوقۇم (اﷲ نىڭ رەھمىتىدىن) ئۈمىدسىزلىنىپ كېتىدۇ، كۇفرانى نېمەت قىلىدۇ
وَلَئِنْ أَذَقْنَاهُ نَعْمَاءَ بَعْدَ ضَرَّاءَ مَسَّتْهُ لَيَقُولَنَّ ذَهَبَ السَّيِّئَاتُ عَنِّي ۚ إِنَّهُ لَفَرِحٌ فَخُورٌ ﴿١٠﴾
ئەگەر بىز (پېقىرلىق، كېسەللىك، قاتتىقچىلىق قاتارلىق) بالالارغا مۇپتىلا قىلغاندىن كېيىن، ئۇنىڭغا نېمىتىمىزنى تېتىتساق، ئۇ چوقۇم: «مەندىن بالالار كۆتۈرۈلۈپ كەتتى (ئەمدى كەلمەيدۇ)» دەپ شەك شۈبھىسىز خۇشاللىنىپ، مەغرۇرلىنىپ كېتىدۇ
إِلَّا الَّذِينَ صَبَرُوا وَعَمِلُوا الصَّالِحَاتِ أُولَٰئِكَ لَهُم مَّغْفِرَةٌ وَأَجْرٌ كَبِيرٌ ﴿١١﴾
پەقەت (بالاغا ئۇچرىغاندا) سەۋرى قىلغان، (نېمەتكە يولۇققاندا) ياخشى ئىشلارنى قىلغان كىشىلەر (ياخشىلاردۇر). ئۇلارنىڭ گۇناھى مەغپىرەت قىلىنىدۇ، ئۇلار (ئاخىرەتتە) چوڭ مۇكاپاتقا ئېرىشىدۇ
فَلَعَلَّكَ تَارِكٌ بَعْضَ مَا يُوحَىٰ إِلَيْكَ وَضَائِقٌ بِهِ صَدْرُكَ أَن يَقُولُوا لَوْلَا أُنزِلَ عَلَيْهِ كَنزٌ أَوْ جَاءَ مَعَهُ مَلَكٌ ۚ إِنَّمَا أَنتَ نَذِيرٌ ۚ وَاللَّهُ عَلَىٰ كُلِّ شَيْءٍ وَكِيلٌ ﴿١٢﴾
(ئى مۇھەممەد!) ئۇلارنىڭ (يەنى مۇشرىكلارنىڭ) «نېمىشقا ئۇنىڭغا بىر خەزىنە چۈشمىدى؟ ياكى ئۇنىڭ بىلەن بىللە (ئۇنى تەستىقلايدىغان) بىرەر پەرىشتە كەلمىدى؟» دەيدىغانلىقى ئۈچۈن، ساڭا نازىل قىلىنغان ۋەھيىنىڭ بىر قىسىمىنى يەتكۈزگۈڭ كەلمەيدىغاندۇ، يەتكۈزۈشتىن يۈرىكىڭ سىقىلىدىغاندۇ، سەن پەقەت (ئۇلارنى اﷲ نىڭ ئازابىدىن) ئاگاھلاندۇرغۇچىسىمەن، اﷲ ھەممە نەرسىنى تەسەررۇپ قىلغۇچىدۇر
أَمْ يَقُولُونَ افْتَرَاهُ ۖ قُلْ فَأْتُوا بِعَشْرِ سُوَرٍ مِّثْلِهِ مُفْتَرَيَاتٍ وَادْعُوا مَنِ اسْتَطَعْتُم مِّن دُونِ اللَّهِ إِن كُنتُمْ صَادِقِينَ ﴿١٣﴾
ئۇلار قۇرئاننى (مۇھەممەد) ئۆزى ئىجاد قىلغان دېيىشەمدۇ؟ ئېيتقىنكى، «(بۇ سۆزۈڭلاردا) راستچىل بولساڭلار، اﷲ تىن غەيرىي چاقىرالايدىغانلىكى كىشىلىرىڭلارنى (ياردەمگە) چاقىرىپ، (پاساھەت ۋە بالاغەت جەھەتتىن) قۇرئاندىكىگە ئوخشايدىغان ئون سۆز ئىجاد قىلىپ بېقىڭلار»
فَإِلَّمْ يَسْتَجِيبُوا لَكُمْ فَاعْلَمُوا أَنَّمَا أُنزِلَ بِعِلْمِ اللَّهِ وَأَن لَّا إِلَٰهَ إِلَّا هُوَ ۖ فَهَلْ أَنتُم مُّسْلِمُونَ ﴿١٤﴾
ئەگەر ئۇلار (بۇ چاقىرىقىڭلارغا) ئاۋاز قوشمىسا، بىلىڭلاركى، قۇرئان اﷲ تىڭ ۋەھيى قىلىشى بىلەن نازىل قىلىنغان. اﷲ تىن باشقا ھېچ ئىلاھ يوقتۇر، سىلەر (اﷲ قا) بويسۇنامسىلەر؟
مَن كَانَ يُرِيدُ الْحَيَاةَ الدُّنْيَا وَزِينَتَهَا نُوَفِّ إِلَيْهِمْ أَعْمَالَهُمْ فِيهَا وَهُمْ فِيهَا لَا يُبْخَسُونَ ﴿١٥﴾
(قىلغان ياخشى ئەمەللىرى بىلەن پەقەت) ھاياتىي دۇنيانى ۋە ئۇنىڭ زىبۇزىننىتىنى (يەنى دۇنيانىڭ نېمەتلىرىنى) كۆزلەيدىغان كىشىلەرگە ئۇلارنىڭ ئەمەللىرىنىڭ ئەجرىنى بۇ دۇنيادىلا تولۇق بېرىمىز، دۇنيادا ئۇلارنىڭ ئەجرىدىن ھېچ نەرسە كېمەيتىلمەيدۇ
أُولَٰئِكَ الَّذِينَ لَيْسَ لَهُمْ فِي الْآخِرَةِ إِلَّا النَّارُ ۖ وَحَبِطَ مَا صَنَعُوا فِيهَا وَبَاطِلٌ مَّا كَانُوا يَعْمَلُونَ ﴿١٦﴾
بۇنداق كىشىلەرگە ئاخىرەتتە دوزاختىن باشقا نەرسە يوقتۇر، ئۇلارنىڭ قىلغان ئىشلىرى يوققا چىقىرىلىدۇ، قىلغان (ياخشى) ئەمەللىرى ھېسابقا ئېلىنمايدۇ
أَفَمَن كَانَ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّهِ وَيَتْلُوهُ شَاهِدٌ مِّنْهُ وَمِن قَبْلِهِ كِتَابُ مُوسَىٰ إِمَامًا وَرَحْمَةً ۚ أُولَٰئِكَ يُؤْمِنُونَ بِهِ ۚ وَمَن يَكْفُرْ بِهِ مِنَ الْأَحْزَابِ فَالنَّارُ مَوْعِدُهُ ۚ فَلَا تَكُ فِي مِرْيَةٍ مِّنْهُ ۚ إِنَّهُ الْحَقُّ مِن رَّبِّكَ وَلَٰكِنَّ أَكْثَرَ النَّاسِ لَا يُؤْمِنُونَ ﴿١٧﴾
پەرۋەردىگارى تەرىپىدىن كەلگەن روشەن دەلىل (يەنى قۇرئان) غا ئاساسلانغان ئادەم (يەنى مۇھەممەد ئەلەيھىسسالام) (ھاياتىي دۇنيانى كۆزلەيدىغان كىشىلەر بىلەن ئوخشاشمۇ؟) ئۇنىڭغا اﷲ تەرىپىدىن كەلگەن گۇۋاھلىق بەرگۈچى (يەنى جىبرىئىل ئەلەيھىسسالام) قۇرئاننى تىلاۋەت قىلىدۇ، قۇرئاندىن ئىلگىرى مۇساغا نازىل قىلىنغان مۇقەددەس كىتاب-ۋەھىي (يەنى تەۋرات) (ئىسرائىل ئەۋلادىغا) يېتەكچى ۋە رەھمەت قىلىپ (نازىل قىلىندى)، ئەنە شۇنداق كىشىلەر قۇرئانغا (ھەقىقىي ئىشىنىدۇ، (كۇففار) جامائەلىرىدىن كىمكى قۇرئاننى ئىنكار قىلىدىكەن، ئۇنىڭغا ۋەدە قىلىنغان جاي دوزاختۇر، قۇرئانغا شەك كەلتۈرمىگىن، ئۇ ئىشەنچلىك ھالدا پەرۋەردىگارىڭ تەرىپىدىن نازىل قىلىنغان ھەقىقەتتۇر، لېكىن ئىنسانلارنىڭ تولىسى (قۇرئاننىڭ ئالەملەرنىڭ پەرۋەردىگارى تەرىپىدىن نازىل قىلىنغانلىقىغا) ئىشەنمەيدۇ
وَمَنْ أَظْلَمُ مِمَّنِ افْتَرَىٰ عَلَى اللَّهِ كَذِبًا ۚ أُولَٰئِكَ يُعْرَضُونَ عَلَىٰ رَبِّهِمْ وَيَقُولُ الْأَشْهَادُ هَٰؤُلَاءِ الَّذِينَ كَذَبُوا عَلَىٰ رَبِّهِمْ ۚ أَلَا لَعْنَةُ اللَّهِ عَلَى الظَّالِمِينَ ﴿١٨﴾
(اﷲ نىڭ شېرىكى بار، بالىسى بار دەپ) اﷲ قا يالغاننى چاپلىغان كىشىدىنمۇ زالىم كىشى بارمۇ؟ ئەنە شۇلار (قىيامەت كۈنى) پەرۋەردىگارىغا توغرىلىنىدۇ، گۇۋاھچىلار: «بۇلار پەرۋەردىگارى ھەققىدە يالغان سۆزلىگەنلەردۇر» دەيدۇ. راستلا اﷲ نىڭ لەنىتى زالىملارغا بولىدۇ (يەنى اﷲ زالىملارنى رەھمىتىدىن يىراق قىلىدۇ)
الَّذِينَ يَصُدُّونَ عَن سَبِيلِ اللَّهِ وَيَبْغُونَهَا عِوَجًا وَهُم بِالْآخِرَةِ هُمْ كَافِرُونَ ﴿١٩﴾
ئۇلار (كىشىلەرنى) اﷲ نىڭ يولىدىن (يەنى ئىسلام يولىدىن) توسىدۇ، اﷲ نىڭ يولىنىڭ ئەگرى بولىشىنى تىلەيدۇ، ئۇلار ئاخىرەتنى ئىنكار قىلغۇچىلاردۇر
أُولَٰئِكَ لَمْ يَكُونُوا مُعْجِزِينَ فِي الْأَرْضِ وَمَا كَانَ لَهُم مِّن دُونِ اللَّهِ مِنْ أَوْلِيَاءَ ۘ يُضَاعَفُ لَهُمُ الْعَذَابُ ۚ مَا كَانُوا يَسْتَطِيعُونَ السَّمْعَ وَمَا كَانُوا يُبْصِرُونَ ﴿٢٠﴾
ئەنە شۇلار زېمىندا اﷲ نىڭ ئازابىدىن قېچىپ قۇتۇلالمايدۇ، ئۇلارغا اﷲ تىن باشقا (ئىگە بولىدىغان) ياردەمچىلەرمۇ بولمايدۇ، ئۇلار قاتمۇ قات ئازابقا دۇچار بولىدۇ، ئۇلار (ھەقنى) ئاڭلاشقا ۋە (ھەقنى) كۆرۈشكە قادىر بولالمايدۇ
أُولَٰئِكَ الَّذِينَ خَسِرُوا أَنفُسَهُمْ وَضَلَّ عَنْهُم مَّا كَانُوا يَفْتَرُونَ ﴿٢١﴾
ئەنە شۇلار ئۆزلىرىگە زىيان سالغۇچىلاردۇر. ئۇلارنىڭ ئويدۇرۇپ چىقارغان بۇتلىرى ئۇلاردىن ئۆزلىرىنى چەتكە ئالدى
لَا جَرَمَ أَنَّهُمْ فِي الْآخِرَةِ هُمُ الْأَخْسَرُونَ ﴿٢٢﴾
شەك يوقكى، ئۇلار ئاخىرەتتە ئەڭ زىيان تارتقۇچىلاردۇر
إِنَّ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ وَأَخْبَتُوا إِلَىٰ رَبِّهِمْ أُولَٰئِكَ أَصْحَابُ الْجَنَّةِ ۖ هُمْ فِيهَا خَالِدُونَ ﴿٢٣﴾
شۈبھىسىزكى، ئىمان ئېيتقانلار ۋە ياخشى ئەمەللەرنى قىلغانلار، پەرۋەردىگارىنىڭ ھۇزۇرىدا ئاجىزلىق بىلدۈرگەنلەر - ئەنە شۇلار ئەھلى مەڭگۈلۈك باغلارتۇر، ئۇلار مەڭگۈلۈك باغلارتە مەڭگۈ قالغۇچىلاردۇر
مَثَلُ الْفَرِيقَيْنِ كَالْأَعْمَىٰ وَالْأَصَمِّ وَالْبَصِيرِ وَالسَّمِيعِ ۚ هَلْ يَسْتَوِيَانِ مَثَلًا ۚ أَفَلَا تَذَكَّرُونَ ﴿٢٤﴾
(ھەقىقەتنى رەت قىلغۇچىلار ۋە مۆمىنلەردىن ئىبارەت) ئىككى پىرقە) ئەماغا ۋە گاسقا ئوخشايدۇ، (يەنە بىر پىرقە بولسا) كۆزى كۆرۈدىغان، قۇلىقى كۆرۈدىغان، قۇلىقى ئاڭلايدىغان ئادەمگە ئوخشايدۇ، ئۇلارنىڭ ئەھۋالى بىر - بىرىگە ئوخشامدۇ؟ ئىبرەت ئامامسىلەر؟
وَلَقَدْ أَرْسَلْنَا نُوحًا إِلَىٰ قَوْمِهِ إِنِّي لَكُمْ نَذِيرٌ مُّبِينٌ ﴿٢٥﴾
شۈبھىسىزكى، بىز نۇھنى قەۋمىگە خۇدانىڭ ئەلچىسى قىلىپ ئەۋەتتۇق، ئۇ: «مەن ئىشەنچلىك ھالدا سىلەرگە ئوچۇق ئاگاھلاندۇرغۇچىمەن
أَن لَّا تَعْبُدُوا إِلَّا اللَّهَ ۖ إِنِّي أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ أَلِيمٍ ﴿٢٦﴾
بىر اﷲ تىن باشقىغا ئىبادەت قىلماڭلار، سىلەرنىڭ قاتتىق كۈننىڭ ئازابىغا ئۇچرىشىڭلاردىن قورقىمەن» دېدى
فَقَالَ الْمَلَأُ الَّذِينَ كَفَرُوا مِن قَوْمِهِ مَا نَرَاكَ إِلَّا بَشَرًا مِّثْلَنَا وَمَا نَرَاكَ اتَّبَعَكَ إِلَّا الَّذِينَ هُمْ أَرَاذِلُنَا بَادِيَ الرَّأْيِ وَمَا نَرَىٰ لَكُمْ عَلَيْنَا مِن فَضْلٍ بَلْ نَظُنُّكُمْ كَاذِبِينَ ﴿٢٧﴾
ئۇنىڭ قەۋمىنىڭ ھەقىقەتنى رەت قىلغۇچى كاتتىلىرى: «بىزنىڭچە، سەن پەقەت بىزگە ئوخشاش بىر ئادەمسەن، بىزنىڭچە، ئارىمىزدىكى پەس كىشىلەر يەڭگىللىك بىلەن ساڭا ئەگەشكەن، بىزنىڭچە، سىلەرنىڭ ھېچقانداق ئارتۇقچىلىقىڭلار يوق، بەلكى بىز سىلەرنى يالغانچى دەپ ئويلايمىز» دېدى
قَالَ يَا قَوْمِ أَرَأَيْتُمْ إِن كُنتُ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّي وَآتَانِي رَحْمَةً مِّنْ عِندِهِ فَعُمِّيَتْ عَلَيْكُمْ أَنُلْزِمُكُمُوهَا وَأَنتُمْ لَهَا كَارِهُونَ ﴿٢٨﴾
نۇھ ئېيتتى: «ئى قەۋمىم، ئېيتىڭلارچۇ، مەن پەرۋەردىگارىم تەرىپىدىن نازىل بولغان (دەۋىتىمنىڭ توغرىلىقىنى ئىسپاتلايدىغان) ئېنىق دەلىلگە ئاساسلانسام ۋە ئۇنىڭ رەھمىتىگە (يەنى خۇدانىڭ ئەلچىسىلىككە) سازاۋەر بولغان بولسام، ئۇ دەلىل سىلەر ئۈچۈن مەخپىي بولسا، ئۇنى يامان كۆرسەڭلار، ئۇنى قوبۇل قىلىشقا سىلەرنى مەجبۇرلامدۇق؟
وَيَا قَوْمِ لَا أَسْأَلُكُمْ عَلَيْهِ مَالًا ۖ إِنْ أَجْرِيَ إِلَّا عَلَى اللَّهِ ۚ وَمَا أَنَا بِطَارِدِ الَّذِينَ آمَنُوا ۚ إِنَّهُم مُّلَاقُو رَبِّهِمْ وَلَٰكِنِّي أَرَاكُمْ قَوْمًا تَجْهَلُونَ ﴿٢٩﴾
ئى قەۋمىم! سىلەرگە (دىننى) تەبلىغ قىلغانلىقىم ئۈچۈن سىلەردىن پۇل - مال تەلەپ قىلمايمەن، ماڭا ئەجىر بېرىشنى پەقەت اﷲ ئۆز ئۈستىگە ئالغان، مەن ئىمان ئېيتقان كىشىلەرنى (يېنىمدىن) قوغلىۋەتمەيمەن، (چۈنكى) ئۇلار، شۈبھىسىزكى، پەرۋەردىگارىغا مۇلاقات بولىدۇ، لېكىن مەن سىلەرنى نادان قەۋم كۆرۈمەن
وَيَا قَوْمِ مَن يَنصُرُنِي مِنَ اللَّهِ إِن طَرَدتُّهُمْ ۚ أَفَلَا تَذَكَّرُونَ ﴿٣٠﴾
ئى قەۋمىم! ئەگەر مەن ئۇلارنى قوغلىۋەتسەم، اﷲ نىڭ ئازابىدىن مېنى كىم قۇتقۇزالايدۇ؟ ئويلىنىپ باقمامسىلەر؟
وَلَا أَقُولُ لَكُمْ عِندِي خَزَائِنُ اللَّهِ وَلَا أَعْلَمُ الْغَيْبَ وَلَا أَقُولُ إِنِّي مَلَكٌ وَلَا أَقُولُ لِلَّذِينَ تَزْدَرِي أَعْيُنُكُمْ لَن يُؤْتِيَهُمُ اللَّهُ خَيْرًا ۖ اللَّهُ أَعْلَمُ بِمَا فِي أَنفُسِهِمْ ۖ إِنِّي إِذًا لَّمِنَ الظَّالِمِينَ ﴿٣١﴾
مەن سىلەرگە، مەندە اﷲ نىڭ خەزىنىلىرى بار، دېمەيمەن، غەيبنى بىلىمەن دېمەيمەن، مەن ئەلۋەتتە (ئۆزۈمنى) پەرىشتە دەپمۇ ئېيتمايمەن ھەمدە سىلەر كۆزگە ئىلمايدىغان كىشىلەرنى اﷲ ئۇلارغا ياخشىلىق بەرمەيدۇ دېمەيمەن، اﷲ ئۇلارنىڭ دىللىرىدىكىنى ئوبدان بىلىدۇ، بولمىسا مەن چوقۇم زالىملاردىن بولۇپ قالىمەن»
قَالُوا يَا نُوحُ قَدْ جَادَلْتَنَا فَأَكْثَرْتَ جِدَالَنَا فَأْتِنَا بِمَا تَعِدُنَا إِن كُنتَ مِنَ الصَّادِقِينَ ﴿٣٢﴾
ئۇلار (يەنى نۇھنىڭ قەۋمى): «ئى نۇھ! بىز بىلەن مۇنازىرلەشتىڭ، ناھايىتى كۆپ مۇنازىرلەشتىڭ. ئەگەر راست سۆزلىگۈچىلەردىن بولساڭ، بىزنى قورقۇتقان ئازابنى چۈشۈرۈپ باققىن» دېدى
قَالَ إِنَّمَا يَأْتِيكُم بِهِ اللَّهُ إِن شَاءَ وَمَا أَنتُم بِمُعْجِزِينَ ﴿٣٣﴾
نۇھ ئېيتتى: «ئۇنى (يەنى ئازابنى) ئەگەر خالىسا اﷲ لا چۈشۈرەلەيدۇ، سىلەر ھەرگىز قېچىپ قۇتۇلالمايسىلەر
وَلَا يَنفَعُكُمْ نُصْحِي إِنْ أَرَدتُّ أَنْ أَنصَحَ لَكُمْ إِن كَانَ اللَّهُ يُرِيدُ أَن يُغْوِيَكُمْ ۚ هُوَ رَبُّكُمْ وَإِلَيْهِ تُرْجَعُونَ ﴿٣٤﴾
مەن سىلەرگە قىلغان نەسىھىتىمنىڭ سىلەرگە پايدىسى بولمايدۇ، اﷲ سىلەرنىڭ پەرۋەردىگارىڭلاردۇر، سىلەر اﷲ نىڭ دەرگاھىغا قايتىسىلەر»
أَمْ يَقُولُونَ افْتَرَاهُ ۖ قُلْ إِنِ افْتَرَيْتُهُ فَعَلَيَّ إِجْرَامِي وَأَنَا بَرِيءٌ مِّمَّا تُجْرِمُونَ ﴿٣٥﴾
ياكى ئۇلار (يەنى قۇرەيشنىڭ ھەقىقەتنى رەت قىلغۇچىلىرى) قۇرئاننى ئۇ (يەنى مۇھەممەد ئەلەيھىسسالام) ئۆزى ئىجاد قىلدى دېيىشەمدۇ؟ «ئەگەر ئۇنى مەن ئىجاد قىلغان بولسام، ئۇ گۇناھىمغا ئۆزۈم مەسئۇل، مەن سىلەرنىڭ گۇناھىڭلاردىن ئادا - جۇدا مەن» دېگىن
وَأُوحِيَ إِلَىٰ نُوحٍ أَنَّهُ لَن يُؤْمِنَ مِن قَوْمِكَ إِلَّا مَن قَدْ آمَنَ فَلَا تَبْتَئِسْ بِمَا كَانُوا يَفْعَلُونَ ﴿٣٦﴾
نۇھقا (مۇنداق) ۋەھيى قىلىندى: «قەۋمىڭدىن ئىلگىرى ساڭا ئىمان ئېيتقانلاردىن باشقا يەنە ئىمان ئېيتقۇچىلار بولمايدۇ، ئۇلارنىڭ قىلمىشىدىن قايغۇرمىغىن
وَاصْنَعِ الْفُلْكَ بِأَعْيُنِنَا وَوَحْيِنَا وَلَا تُخَاطِبْنِي فِي الَّذِينَ ظَلَمُوا ۚ إِنَّهُم مُّغْرَقُونَ ﴿٣٧﴾
بىزنىڭ كۆز ئوڭىمىزدا ۋەھيىمىز بويىچە كېمە ياسىغىن، (ئۆزلىرىگە) زۇلۇم قىلغانلار توغرىسىدا ماڭا سۆز ئاچمىغىن (يەنى ئۇلارغا شاپائەت قىلمىغىن)، ئۇلار چوقۇم (توپان بالاسىدا) غەرق قىلىنىدۇ»
وَيَصْنَعُ الْفُلْكَ وَكُلَّمَا مَرَّ عَلَيْهِ مَلَأٌ مِّن قَوْمِهِ سَخِرُوا مِنْهُ ۚ قَالَ إِن تَسْخَرُوا مِنَّا فَإِنَّا نَسْخَرُ مِنكُمْ كَمَا تَسْخَرُونَ ﴿٣٨﴾
نۇھ كېمە ياسىدى، قەۋمنىڭ چوڭلىرى ئۇنىڭ ئالدىدىن ئۆتكەندە ئۇنى مەسخىرە قىلىشتى، نۇھ ئېيتتى: «ئەگەر سىلەر (بۈگۈن) بىزنى مەسخىرە قىلساڭلار (كېلەچەكتە سىلەر غەرق قىلىنغاندا)، سىلەر بىزنى مەسخىرە قىلغاندەك بىزمۇ سىلەرنى چوقۇم مەسخىرە قىلىمىز
فَسَوْفَ تَعْلَمُونَ مَن يَأْتِيهِ عَذَابٌ يُخْزِيهِ وَيَحِلُّ عَلَيْهِ عَذَابٌ مُّقِيمٌ ﴿٣٩﴾
خار قىلغۇچى ئازاب (يەنى توپان بالاسى بىلەن غەرق بولۇش) نىڭ كىمنىڭ ئۈستىگە كېلىدىغانلىقىنى، دائىمىي ئازاب (يەنى مەڭگۈلۈك ئوت ئازابى) نىڭ كىمگە چۈشىدىغانلىقىنى كېلەچەكتە بىلىسىلەر»
حَتَّىٰ إِذَا جَاءَ أَمْرُنَا وَفَارَ التَّنُّورُ قُلْنَا احْمِلْ فِيهَا مِن كُلٍّ زَوْجَيْنِ اثْنَيْنِ وَأَهْلَكَ إِلَّا مَن سَبَقَ عَلَيْهِ الْقَوْلُ وَمَنْ آمَنَ ۚ وَمَا آمَنَ مَعَهُ إِلَّا قَلِيلٌ ﴿٤٠﴾
(توپان بالاسى توغرىسىدىكى) بۇيرۇقىمىز يېتىپ كەلگەن ۋە يەر يۈزىدىن سۇ ئېتىلىپ چىققان چاغدا: «كېمىگە ھايۋاناتلاردىن (ئەركەك چىشى بولۇپ) بىر جۇپتىن ئېلىۋالغىن، ئائىلەڭدىكىلەر بىلەن ھالاك بولۇشى ھۆكۈم قىلىنغانلار (يەنى نۇھ ئەلەيھىسسالامنىڭ ئايالى ۋائىلە بىلەن ئوغلى كەنئان) بۇنىڭ سىرتىدا - ۋە ئىمان ئېيتقانلار بىلەن كېمىگە چىققىن» دېدۇق، ئۇنىڭ بىلەن پەقەت ئازغىنا كىشىلەر ئىمان ئېيتتى
وَقَالَ ارْكَبُوا فِيهَا بِسْمِ اللَّهِ مَجْرَاهَا وَمُرْسَاهَا ۚ إِنَّ رَبِّي لَغَفُورٌ رَّحِيمٌ ﴿٤١﴾
نۇھ (ئىمان ئېيتقان تەۋەلىرىگە) ئېيتتى: «كېمىگە چىقىڭلار، كېمىنىڭ مېڭىشى ۋە توختىشى اﷲ نىڭ ئىسمى بىلەندۇر، شۈبھىسىزكى، مېنىڭ پەرۋەردىگارىم (تەۋبە قىلغۇچىلارنىڭ گۇناھىنى) ئىشەنچلىك ھالدا مەغپىرەت قىلغۇچىدۇر، (مۆمىنلەرگە) ناھايىتى مېھرىباندۇر»
وَهِيَ تَجْرِي بِهِمْ فِي مَوْجٍ كَالْجِبَالِ وَنَادَىٰ نُوحٌ ابْنَهُ وَكَانَ فِي مَعْزِلٍ يَا بُنَيَّ ارْكَب مَّعَنَا وَلَا تَكُن مَّعَ الْكَافِرِينَ ﴿٤٢﴾
كېمە ئۇلارنى ئېلىپ تاغدەك دولقۇنلار ئىچىدە ئۈزۈپ باراتتى، نۇھ كېمە مېڭىشتىن بۇرۇن ئۆزىدىن يىراقتا تۇرغان ئوغلىنى: «ئى ئوغلۇم، بىز بىلەن بىللە كېمىگە چىققىن، ھەقىقەتنى رەت قىلغۇچىلار بىلەن بىللە بولمىغىن» دەپ توۋلىدى
قَالَ سَآوِي إِلَىٰ جَبَلٍ يَعْصِمُنِي مِنَ الْمَاءِ ۚ قَالَ لَا عَاصِمَ الْيَوْمَ مِنْ أَمْرِ اللَّهِ إِلَّا مَن رَّحِمَ ۚ وَحَالَ بَيْنَهُمَا الْمَوْجُ فَكَانَ مِنَ الْمُغْرَقِينَ ﴿٤٣﴾
ئۇ: «مېنى سۇدا غەرق بولۇشتىن ساقلاپ قالىدىغان بىر تاغنىڭ ئۈستىگە چىقىۋالىمەن» دېدى. نۇھ: «بۈگۈن اﷲ رەھىم قىلغان ئادەمدىن باشقا ھېچ ئادەم ئۇنىڭ جازالىشىدىن قۇتۇلۇپ قالالمايدۇ» دېدى. ئۇلارنىڭ ئارىسىنى دولقۇن ئايرىۋەتتى - دە، ئۇ غەرق بولۇپ كەتتى
وَقِيلَ يَا أَرْضُ ابْلَعِي مَاءَكِ وَيَا سَمَاءُ أَقْلِعِي وَغِيضَ الْمَاءُ وَقُضِيَ الْأَمْرُ وَاسْتَوَتْ عَلَى الْجُودِيِّ ۖ وَقِيلَ بُعْدًا لِّلْقَوْمِ الظَّالِمِينَ ﴿٤٤﴾
«ئى زېمىن سۈيۈڭنى يۇتقىن، ئى بۇلۇت تارقىغىن» دېيىلدى، سۇ سوغۇلدى، اﷲ نىڭ (ھەقىقەتنى رەت قىلغۇچىلارنى غەرق قىلىشتىن ئىبارەت) ھۆكمى ئىجرا قىلىندى، كېمە جۇدى تېغىنىڭ ئۈستىدە توختىدى، «زالىم قەۋم ھالاك بولسۇن» دېيىلدى
وَنَادَىٰ نُوحٌ رَّبَّهُ فَقَالَ رَبِّ إِنَّ ابْنِي مِنْ أَهْلِي وَإِنَّ وَعْدَكَ الْحَقُّ وَأَنتَ أَحْكَمُ الْحَاكِمِينَ ﴿٤٥﴾
نۇھ پەرۋەردىگارىغا دۇئا قىلىپ: «پەرۋەردىگارىم! ئوغلۇم مېنىڭ ئائىلەمدىكىلەردىن ئىدى (سەن ماڭا ئۇلارنىڭ نىجات تېپىشىنى ۋەدە قىلغان ئىدىڭ)، سېنىڭ ۋەدەڭ ئەلۋەتتە ھەقتۇر، سەن ئىشەنچلىك ھالدا ئەڭ توغرا ھۆكۈم قىلغۇچىسەن» دېدى
قَالَ يَا نُوحُ إِنَّهُ لَيْسَ مِنْ أَهْلِكَ ۖ إِنَّهُ عَمَلٌ غَيْرُ صَالِحٍ ۖ فَلَا تَسْأَلْنِ مَا لَيْسَ لَكَ بِهِ عِلْمٌ ۖ إِنِّي أَعِظُكَ أَن تَكُونَ مِنَ الْجَاهِلِينَ ﴿٤٦﴾
اﷲ ئېيتتى: «ئى نۇھ! ئۇ (مەن نىجات تېپىشىنى ۋەدە قىلغان) ئائىلەڭدىكىلەردىن ئەمەس، ئۇنىڭ ئەمەلى ياماندۇر، سەن ئېنىق بىلمىگەن نەرسەڭنى مەندىن سورىمىغىن، مەن سېنىڭ جاھىللاردىن بولماسلىقىڭنى نەسىھەت قىلىمەن»
قَالَ رَبِّ إِنِّي أَعُوذُ بِكَ أَنْ أَسْأَلَكَ مَا لَيْسَ لِي بِهِ عِلْمٌ ۖ وَإِلَّا تَغْفِرْ لِي وَتَرْحَمْنِي أَكُن مِّنَ الْخَاسِرِينَ ﴿٤٧﴾
نۇھ ئېيتتى: «پەرۋەردىگارىم! مەن بىلمىگەن نەرسىنى سوراشتىن سېنىڭ پاناھىڭغا سېغىنىمەن، ئەگەر ماڭا مەغپىرەت قىلمىساڭ، زىيان تارتقۇچىلاردىن بولىمەن»
قِيلَ يَا نُوحُ اهْبِطْ بِسَلَامٍ مِّنَّا وَبَرَكَاتٍ عَلَيْكَ وَعَلَىٰ أُمَمٍ مِّمَّن مَّعَكَ ۚ وَأُمَمٌ سَنُمَتِّعُهُمْ ثُمَّ يَمَسُّهُم مِّنَّا عَذَابٌ أَلِيمٌ ﴿٤٨﴾
ئېيتىلدى (يەنى اﷲ ئېيتتى): «ئى نۇھ! ساڭا، سەن بىلەن (كېمىدە) بىللە بولغانلارنىڭ بىر قىسىم ئەۋلادىغا بىز تەرەپتىن نازىل بولغان ئامانلىق ۋە بەرىكەتلەر بىلەن (كېمىدىن) چۈشكىن، ئۇلارنىڭ يەنە بىر قىسىم ئەۋلادىنى (ھاياتىي دۇنيادىن) بەھرىمەن قىلىمىز، ئاندىن ئۇلار بىزنىڭ قاتتىق ئازابىمىزغا دۇچار بولىدۇ»
تِلْكَ مِنْ أَنبَاءِ الْغَيْبِ نُوحِيهَا إِلَيْكَ ۖ مَا كُنتَ تَعْلَمُهَا أَنتَ وَلَا قَوْمُكَ مِن قَبْلِ هَٰذَا ۖ فَاصْبِرْ ۖ إِنَّ الْعَاقِبَةَ لِلْمُتَّقِينَ ﴿٤٩﴾
(ئى مۇھەممەد!) ئەنە شۇ (قىسسە) غەيبكە ئائىت خەۋەرلەردىندۇر، ساڭا ئۇنى ۋەھيى قىلدۇق، سەن ۋە سېنىڭ قەۋمىڭ بۇنىڭدىن ئىلگىرى ئۇنى بىلمەيتتىڭلار، سەۋر قىلغىن، شۈبھىسىزكى، ياخشى ئاقىۋەت تەقۋادارلارغا مەنسۇپتۇر
وَإِلَىٰ عَادٍ أَخَاهُمْ هُودًا ۚ قَالَ يَا قَوْمِ اعْبُدُوا اللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُ ۖ إِنْ أَنتُمْ إِلَّا مُفْتَرُونَ ﴿٥٠﴾
ئاد قەۋمىگە ئۇلارنىڭ قېرىندىشى ھۇدنى ئەۋەتتۇق، ھۇد ئېيتتى: «ئى قەۋمىم! اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە اﷲ تىن باشقا مەبۇد (بەرھەق) يوقتۇر، سىلەر (اﷲ تىن غەيرىيگە چوقۇنۇش بىلەن) پەقەت (اﷲ قا) يالغاننى توقۇغۇچىلارسىلەر»
يَا قَوْمِ لَا أَسْأَلُكُمْ عَلَيْهِ أَجْرًا ۖ إِنْ أَجْرِيَ إِلَّا عَلَى الَّذِي فَطَرَنِي ۚ أَفَلَا تَعْقِلُونَ ﴿٥١﴾
«ئى قەۋمىم! مەن تەبلىغ ئۈچۈن سىلەردىن ئەجىر سورىمايمەن، ماڭا ئەجىر بېرىشنى پەقەت مېنى ياراتقان اﷲ ئۈستىگە ئالغان، سىلەر چۈشەنمەمسىلەر؟»
وَيَا قَوْمِ اسْتَغْفِرُوا رَبَّكُمْ ثُمَّ تُوبُوا إِلَيْهِ يُرْسِلِ السَّمَاءَ عَلَيْكُم مِّدْرَارًا وَيَزِدْكُمْ قُوَّةً إِلَىٰ قُوَّتِكُمْ وَلَا تَتَوَلَّوْا مُجْرِمِينَ ﴿٥٢﴾
«ئى قەۋمىم! پەرۋەردىگارىڭلاردىن مەغپىرەت تىلەڭلار، ئاندىن ئۇنىڭغا تەۋبە قىلىڭلار، سىلەرگە ئۇ كۆپ يېغىن ياغدۇرۇپ بېرىدۇ، سىلەرگە تېخىمۇ كۈچ - قۇۋۋەت بېرىدۇ، سىلەر گۇناھقا چۆمگەن ھالدا (مېنىڭ دەۋىتىمدىن) يۈز ئۆرۈمەڭلار»
قَالُوا يَا هُودُ مَا جِئْتَنَا بِبَيِّنَةٍ وَمَا نَحْنُ بِتَارِكِي آلِهَتِنَا عَن قَوْلِكَ وَمَا نَحْنُ لَكَ بِمُؤْمِنِينَ ﴿٥٣﴾
ئۇلار ئېيتتى: «ئى ھۇد! سەن بىزگە (ئۆزۈڭنىڭ راستلىقىڭنى كۆرسىتىدىغان) ھېچقانداق روشەن دەلىل كەلتۈرمىدىڭ، سېنىڭ سۆزۈڭ بىلەن بىز ئىلاھلىرىمىزنى تاشلىۋەتمەيمىز ۋە سېنىڭ (خۇدانىڭ ئەلچىسىلىكىڭگە) ئىشەنمەيمىز
إِن نَّقُولُ إِلَّا اعْتَرَاكَ بَعْضُ آلِهَتِنَا بِسُوءٍ ۗ قَالَ إِنِّي أُشْهِدُ اللَّهَ وَاشْهَدُوا أَنِّي بَرِيءٌ مِّمَّا تُشْرِكُونَ ﴿٥٤﴾
بىز پەقەت سېنى بەزى ئىلاھلىرىمىز ساراڭ قىلىپ قويۇپتۇ دەيمىز» ھۇد ئېيتتى: «مەن ئىشەنچلىك ھالدا اﷲ نى گۇۋاھ قىلىمەن، سىلەرمۇ گۇۋاھ بولۇڭلاركى، مەن سىلەرنىڭ اﷲ نى قويۇپ شېرىك كەلتۈرگەن بۇتلىرىڭلاردىن ئادا - جۇدامەن، ھەممىڭلار ماڭا سۇيىقەست قىلىڭلار، ماڭا مۆھلەت بەرمەڭلار
مِن دُونِهِ ۖ فَكِيدُونِي جَمِيعًا ثُمَّ لَا تُنظِرُونِ ﴿٥٥﴾
بىز پەقەت سېنى بەزى ئىلاھلىرىمىز ساراڭ قىلىپ قويۇپتۇ دەيمىز» ھۇد ئېيتتى: «مەن ئىشەنچلىك ھالدا اﷲ نى گۇۋاھ قىلىمەن، سىلەرمۇ گۇۋاھ بولۇڭلاركى، مەن سىلەرنىڭ اﷲ نى قويۇپ شېرىك كەلتۈرگەن بۇتلىرىڭلاردىن ئادا - جۇدامەن، ھەممىڭلار ماڭا سۇيىقەست قىلىڭلار، ماڭا مۆھلەت بەرمەڭلار
إِنِّي تَوَكَّلْتُ عَلَى اللَّهِ رَبِّي وَرَبِّكُم ۚ مَّا مِن دَابَّةٍ إِلَّا هُوَ آخِذٌ بِنَاصِيَتِهَا ۚ إِنَّ رَبِّي عَلَىٰ صِرَاطٍ مُّسْتَقِيمٍ ﴿٥٦﴾
شۈبھىسىزكى، مېنىڭ پەرۋەردىگارىم ۋە سىلەرنىڭ پەرۋەردىگارىڭلار بولغان اﷲ قا مەن تەۋەككۈل قىلدىم، اﷲ نىڭ باشقۇرۇشىدا بولمىغان بىرمۇ مەخلۇق يوقتۇر؛ پەرۋەردىگارىم ئىشەنچلىك ھالدا توغرا يولدىدۇر (يەنى ئادىل بولۇپ، ھېچ كىشىگە زۇلۇم قىلمايدۇ)
فَإِن تَوَلَّوْا فَقَدْ أَبْلَغْتُكُم مَّا أُرْسِلْتُ بِهِ إِلَيْكُمْ ۚ وَيَسْتَخْلِفُ رَبِّي قَوْمًا غَيْرَكُمْ وَلَا تَضُرُّونَهُ شَيْئًا ۚ إِنَّ رَبِّي عَلَىٰ كُلِّ شَيْءٍ حَفِيظٌ ﴿٥٧﴾
ئەگەر سىلەر (مېنىڭ دەۋىتىمنى قوبۇل قىلىشتىن) يۈز ئۆرۈسەڭلار (يۈز ئۆرۈگىنىڭلارنىڭ ماڭا زىيىنى يوق)، سىلەرگە مەن پەرۋەردىگارىمنىڭ ئەلچىلىكىنى يەتكۈزدۈم (خۇدانىڭ ئەلچىسىنىڭ ۋەزىپىسى پەقەت تەبلىغ قىلىشتۇر)، پەرۋەردىگارىم (سىلەرنى ھالاك قىلىپ) ئورنۇڭلارغا باشقا بىر قەۋمنى كەلتۈرىدۇ، (شېرىك كەلتۈرۈش بىلەن) اﷲ قا قىلچە زىيان يەتكۈزەلمەيسىلەر، پەرۋەردىگارىم ئىشەنچلىك ھالدا ھەربىر نەرسىنى كۈزىتىپ تۇرغۇچىدۇر»
وَلَمَّا جَاءَ أَمْرُنَا نَجَّيْنَا هُودًا وَالَّذِينَ آمَنُوا مَعَهُ بِرَحْمَةٍ مِّنَّا وَنَجَّيْنَاهُم مِّنْ عَذَابٍ غَلِيظٍ ﴿٥٨﴾
بىزنىڭ (ئازابلاش) بۇيرۇقىمىز كەلگەندە، ھۇدنى ۋە ئۇنىڭ بىلەن بىللە بولغان مۆمىنلەرنى مەرھەمەت قىلىپ قۇتقۇزدۇق، ئۇلارنى قاتتىق ئازابتىن خالاس قىلدۇق
وَتِلْكَ عَادٌ ۖ جَحَدُوا بِآيَاتِ رَبِّهِمْ وَعَصَوْا رُسُلَهُ وَاتَّبَعُوا أَمْرَ كُلِّ جَبَّارٍ عَنِيدٍ ﴿٥٩﴾
ئەنە شۇ ئاد قەۋمى پەرۋەردىگارىنىڭ مۆجىزىلىرىنى ئىنكار قىلدى، ئۇنىڭ خۇدانىڭ ئەلچىسىلىكىگە ئاسىيلىق قىلدى، ھەربىر ئۇچىغا چىققان تەرسا، شەپقەتسىز (يەنى ھەققە قارشىلىق قىلىپ جېدەللەشكۈچى) نىڭ بۇيرۇقىغا ئەگەشتى
وَأُتْبِعُوا فِي هَٰذِهِ الدُّنْيَا لَعْنَةً وَيَوْمَ الْقِيَامَةِ ۗ أَلَا إِنَّ عَادًا كَفَرُوا رَبَّهُمْ ۗ أَلَا بُعْدًا لِّعَادٍ قَوْمِ هُودٍ ﴿٦٠﴾
ئۇلارغا بۇ دۇنيادا لەنەت قىلىندى، قىيامەت كۈنىمۇ ئۇلارغا لەنەت قىلىنىدۇ، بىلىڭلاركى، ئاد قەۋمى ئىشەنچلىك ھالدا پەرۋەردىگارىنى ئىنكار قىلدى. ئاگاھ بولۇڭلاركى، ھۇد قەۋمى بولغان ئاد ھالاك بولسۇن
وَإِلَىٰ ثَمُودَ أَخَاهُمْ صَالِحًا ۚ قَالَ يَا قَوْمِ اعْبُدُوا اللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُ ۖ هُوَ أَنشَأَكُم مِّنَ الْأَرْضِ وَاسْتَعْمَرَكُمْ فِيهَا فَاسْتَغْفِرُوهُ ثُمَّ تُوبُوا إِلَيْهِ ۚ إِنَّ رَبِّي قَرِيبٌ مُّجِيبٌ ﴿٦١﴾
سەمۇدقا ئۇلارنىڭ قېرىندىشى سالىھنى (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، ئۇ ئېيتتى: «ئى قەۋمىم! اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، ئۇ سىلەرنى زېمىندىن (يەنى تۇپراقتىن) ياراتتى، سىلەرنى زېمىندا تۇرغۇزدى، اﷲ تىن مەغپىرەت تىلەڭلار، ئاندىن ئۇنىڭغا تەۋبە قىلىڭلار، پەرۋەردىگارىم ئىشەنچلىك ھالدا يېقىندۇر، (دۇئانى) ئىجابەت قىلغۇچىدۇر»
قَالُوا يَا صَالِحُ قَدْ كُنتَ فِينَا مَرْجُوًّا قَبْلَ هَٰذَا ۖ أَتَنْهَانَا أَن نَّعْبُدَ مَا يَعْبُدُ آبَاؤُنَا وَإِنَّنَا لَفِي شَكٍّ مِّمَّا تَدْعُونَا إِلَيْهِ مُرِيبٍ ﴿٦٢﴾
ئۇلار: «ئى سالىھ! بۇنىڭدىن بۇرۇن سەن ئارىمىزدا ئۈمىد قىلىنغان كىشى ئىدىڭ (يەنى بۇ سۆزنى ئېيتىشىڭدىن بۇرۇن ئارىمىزدا چوڭ ئادەم بولۇپ قېلىشىڭنى ئۈمىد قىلاتتۇق)، بىزنى ئاتا - بوۋىلىرىمىز ئىبادەت قىلىپ كەلگەن بۇتلارغا ئىبادەت قىلىشتىن توسۇمسەن؟ شەك شۈبھىسىزكى، ئىشەنچلىك ھالدا بىز سەن دەۋەت قىلغان ئىشقا زور گۇماندىمىز» دېيىشتى
قَالَ يَا قَوْمِ أَرَأَيْتُمْ إِن كُنتُ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّي وَآتَانِي مِنْهُ رَحْمَةً فَمَن يَنصُرُنِي مِنَ اللَّهِ إِنْ عَصَيْتُهُ ۖ فَمَا تَزِيدُونَنِي غَيْرَ تَخْسِيرٍ ﴿٦٣﴾
سالىھ ئېيتتى: «ئى قەۋمىم! ئېيتىڭلارچۇ، ئەگەر مەن پەرۋەردىگارىم تەرىپىدىن نازىل بولغان ئېنىق دەلىلگە ئاساسلانسام ۋە ئۇنىڭ رەھمىتى (يەنى خۇدانىڭ ئەلچىسىلىك) گە سازاۋەر بولسام، شۇنداق تۇرۇقلۇق اﷲ قا قارشىلىق قىلسام، اﷲ نىڭ ئازابىدىن مېنى كىم قۇتقۇزىدۇ؟ (مېنى ئۆزۈڭلارغا ئەگىشىشكە دەۋەت قىلىش بىلەن) ماڭا پەقەت زىياندىن باشقىسىنى زىيادە قىلالمايسىلەر
وَيَا قَوْمِ هَٰذِهِ نَاقَةُ اللَّهِ لَكُمْ آيَةً فَذَرُوهَا تَأْكُلْ فِي أَرْضِ اللَّهِ وَلَا تَمَسُّوهَا بِسُوءٍ فَيَأْخُذَكُمْ عَذَابٌ قَرِيبٌ ﴿٦٤﴾
ئى قەۋمىم! بۇ اﷲ (بىۋاسىتە) ياراتقان چىشى تۆگە سىلەرگە مېنىڭ مۆجىزەمدۇر، ئۇنى اﷲ نىڭ زېمىنىغا قويۇۋېتىڭلار، ئىختىيارى ئوتلىسۇن، ئۇنىڭغا زىيان زەخمەت يەتكۈزمەڭلار، بولمىسا سىلەرنى پات كېلىدىغان ئازاب ھالاك قىلىدۇ»
فَعَقَرُوهَا فَقَالَ تَمَتَّعُوا فِي دَارِكُمْ ثَلَاثَةَ أَيَّامٍ ۖ ذَٰلِكَ وَعْدٌ غَيْرُ مَكْذُوبٍ ﴿٦٥﴾
ئۇلار چىشى تۆگىنى بوغۇزلىدى، سالىھ ئېيتتى: «ئۆيۈڭلاردا (ھاياتلىقتىن) ئۈچ كۈن بەھرىمەن بولۇۋېلىڭلار. بۇ ۋەدە يالغان ئەمەس»
فَلَمَّا جَاءَ أَمْرُنَا نَجَّيْنَا صَالِحًا وَالَّذِينَ آمَنُوا مَعَهُ بِرَحْمَةٍ مِّنَّا وَمِنْ خِزْيِ يَوْمِئِذٍ ۗ إِنَّ رَبَّكَ هُوَ الْقَوِيُّ الْعَزِيزُ ﴿٦٦﴾
بىزنىڭ (ئازاب قىلىش) ئەمرىمىز كەلگەندە، سالىھنى ۋە ئۇنىڭ بىلەن بىللە ئىمان ئېيتقانلارنى رەھمىتىمىز بىلەن شۇ كۈننىڭ شەرمەندىچىلىكىدىن قۇتقۇزدۇق، پەرۋەردىگارىڭ ئىشەنچلىك ھالدا كۈچلۈكتۇر، غالىبتۇر
وَأَخَذَ الَّذِينَ ظَلَمُوا الصَّيْحَةُ فَأَصْبَحُوا فِي دِيَارِهِمْ جَاثِمِينَ ﴿٦٧﴾
زالىملارغا قاتتىق ئاۋاز يۈزلەندى - دە، ئۇلار ئۆيلىرىدە ئولتۇرغان پېتى قېتىپ قالدى
كَأَن لَّمْ يَغْنَوْا فِيهَا ۗ أَلَا إِنَّ ثَمُودَ كَفَرُوا رَبَّهُمْ ۗ أَلَا بُعْدًا لِّثَمُودَ ﴿٦٨﴾
گويا ئۇلار ئۆيلىرىدە تۇرمىغاندەك (يەنى ئۇلارنىڭ تۇرغان جايلىرى ئادەمزات تەرىپىدىن ئاۋات قىلىنمىغاندەك) بولۇپ قالدى، بىلىڭلاركى، سەمۇد پەرۋەردىگارىنى ئىنكار قىلدى، سەمۇد ھالاك بولسۇن
وَلَقَدْ جَاءَتْ رُسُلُنَا إِبْرَاهِيمَ بِالْبُشْرَىٰ قَالُوا سَلَامًا ۖ قَالَ سَلَامٌ ۖ فَمَا لَبِثَ أَن جَاءَ بِعِجْلٍ حَنِيذٍ ﴿٦٩﴾
شۈبھىسىزكى، بىزنىڭ ئەلچىلىرىمىز (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى) ئىبراھىمغا خۇش خەۋەر ئېلىپ كەلدى. ئۇلار: «ساڭا ئامانلىق تىلەيمىز» دېدى. ئىبراھىم: «سىلەرگە ئامانلىق تىلەيمەن» دېدى. ئۇ ئۇزاققا قالماي بىر موزاينى كاۋاپ قىلىپ ئېلىپ كەلدى
فَلَمَّا رَأَىٰ أَيْدِيَهُمْ لَا تَصِلُ إِلَيْهِ نَكِرَهُمْ وَأَوْجَسَ مِنْهُمْ خِيفَةً ۚ قَالُوا لَا تَخَفْ إِنَّا أُرْسِلْنَا إِلَىٰ قَوْمِ لُوطٍ ﴿٧٠﴾
ئۇلارنىڭ (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرىنىڭ) ئۇنىڭغا قول ئۇزاتمايۋاتقانلىقىنى كۆرۈپ، ئۇلارنى خوش كۆرمىدى، ئۇلاردىن قورقۇنچ ھېس قىلدى، ئۇلار: «قورقمىغىن (بىز رەببىڭنىڭ پەرىشتىلىرى، تاماق يېمەيمىز)، بىز ئىشەنچلىك ھالدا لۇتنىڭ قەۋمىنى (ھالاك قىلىش ئۈچۈن) ئەۋەتىلدۇق» دېدى
وَامْرَأَتُهُ قَائِمَةٌ فَضَحِكَتْ فَبَشَّرْنَاهَا بِإِسْحَاقَ وَمِن وَرَاءِ إِسْحَاقَ يَعْقُوبَ ﴿٧١﴾
ئۇنىڭ ئايالى (يەنى سارە) بىر تەرەپتە تۇرۇپ (لۇت قەۋمىنىڭ ھالاك بولىدىغانلىقىدىن خۇشاللىنىپ) كۈلدى، بىز ئۇنىڭغا (يەنى سارەگە) ئىسھاق (ئاتلىق بالىسى بولىدىغانلىقى) بىلەن ۋە ئىسھاقتىن كېيىن كېلىدىغان يەئقۇب (ئاتلىق نەۋرىسى بولىدىغانلىقى) بىلەن خۇش خەۋەر بەردۇق
قَالَتْ يَا وَيْلَتَىٰ أَأَلِدُ وَأَنَا عَجُوزٌ وَهَٰذَا بَعْلِي شَيْخًا ۖ إِنَّ هَٰذَا لَشَيْءٌ عَجِيبٌ ﴿٧٢﴾
ئۇ (يەنى سارە): «ۋىيەي! مەن بىر موماي تۇرسام، ئېرىم بىر بوۋاي تۇرسا، تۇغامدىمەن؟ بۇ ئىشەنچلىك ھالدا قىزىق ئىشقۇ!» دېدى
قَالُوا أَتَعْجَبِينَ مِنْ أَمْرِ اللَّهِ ۖ رَحْمَتُ اللَّهِ وَبَرَكَاتُهُ عَلَيْكُمْ أَهْلَ الْبَيْتِ ۚ إِنَّهُ حَمِيدٌ مَّجِيدٌ ﴿٧٣﴾
ئۇلار: «اﷲ نىڭ ئەمرىدىن (يەنى قۇدرىتىدىن) ئەجەبلىنەمسە؟ اﷲ نىڭ رەھمىتى ۋە بەرىكەتلىرى سىلەرنىڭ ئائىلەڭلارغا (يەنى ئىبراھىمنىڭ ئائىلىسىدىكىلەرگە) نازىل بولسۇن. اﷲ ئىشەنچلىك ھالدا مەدھىيىلەشكە لايىقتۇر، ئۇلۇغلاشقا لايىقتۇر» دېدى
فَلَمَّا ذَهَبَ عَنْ إِبْرَاهِيمَ الرَّوْعُ وَجَاءَتْهُ الْبُشْرَىٰ يُجَادِلُنَا فِي قَوْمِ لُوطٍ ﴿٧٤﴾
ئىبراھىمدىن قورقۇنچ يوقالغان ھەمدە خۇش خەۋەر ئۇنىڭغا يەتكەن چاغدا، ئۇ بىز بىلەن لۇتنىڭ قەۋمى توغرىسىدا مۇنازىرىلىشىشىگە كىرىشتى
إِنَّ إِبْرَاهِيمَ لَحَلِيمٌ أَوَّاهٌ مُّنِيبٌ ﴿٧٥﴾
ئىبراھىم بولسا ئىشەنچلىك ھالدا كۆڭلى كۆكسى كەڭ، كۆپ ئاھ چەككۈچى (يەنى دىلى يۇمشاق)، (ھەممە ئىشتا) اﷲ قا مۇراجىئەت قىلغۇچى زاتتۇر
يَا إِبْرَاهِيمُ أَعْرِضْ عَنْ هَٰذَا ۖ إِنَّهُ قَدْ جَاءَ أَمْرُ رَبِّكَ ۖ وَإِنَّهُمْ آتِيهِمْ عَذَابٌ غَيْرُ مَرْدُودٍ ﴿٧٦﴾
(خۇدانىڭ ئەلچى پەرىشتىلىرى ئېيتتى) «ئى ئىبراھىم! بۇنى (يەنى لۇت قەۋمى توغرىسىدا مۇنازىرىلىشىشنى) قويغىن، شۈبھىسىزكى، پەرۋەردىگارىڭنىڭ (ئۇلارنى ھالاك قىلىش توغرىسىدىكى) ئەمرى ئاللىقاچان چۈشتى، ئۇلارغا چوقۇم (مۇنازىرىلىشىش ياكى شاپائەت قىلىش بىلەن) قايتۇرغىلى بولمايدىغان ئازاب چۈشىدۇ»
وَلَمَّا جَاءَتْ رُسُلُنَا لُوطًا سِيءَ بِهِمْ وَضَاقَ بِهِمْ ذَرْعًا وَقَالَ هَٰذَا يَوْمٌ عَصِيبٌ ﴿٧٧﴾
بىزنىڭ ئەلچىلىرىمىز (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى) لۇتنىڭ يېنىغا كەلگەن چاغدا (لۇت ئۆز قەۋمىنىڭ ئۇلارغا چېقىلىپ قويۇشىدىن قورقۇپ) قىيىن ئەھۋالدا قالدى، ئۇلار توغرىسىدا يۈرىكى سىقىلدى ۋە: «بۇ بىر دىشۋار كۈندۇر» دېدى
وَجَاءَهُ قَوْمُهُ يُهْرَعُونَ إِلَيْهِ وَمِن قَبْلُ كَانُوا يَعْمَلُونَ السَّيِّئَاتِ ۚ قَالَ يَا قَوْمِ هَٰؤُلَاءِ بَنَاتِي هُنَّ أَطْهَرُ لَكُمْ ۖ فَاتَّقُوا اللَّهَ وَلَا تُخْزُونِ فِي ضَيْفِي ۖ أَلَيْسَ مِنكُمْ رَجُلٌ رَّشِيدٌ ﴿٧٨﴾
لۇتنىڭ قەۋمى ئۇنىڭ قېشىغا ئالدىراپ كېلىشتى، ئۇلار بۇرۇنلا يامان ئىشلارنى قىلاتتى (يەنى لىۋاتەت قىلاتتى). لۇت ئېيتتى: «ئى قەۋمىم! مېنىڭ بۇ قىزلىرىم سىلەرگە ئەڭ پاكتۇر، اﷲ تىن قورقۇڭلار، مېھمانلىرىم ئۈستىدە مېنى رەسۋا قىلماڭلار، ئاراڭلاردا قەبىھ ئىشتىن توسىدىغان بىرەر كاللىسى جايىدا ئادەم يوقمۇ؟»
قَالُوا لَقَدْ عَلِمْتَ مَا لَنَا فِي بَنَاتِكَ مِنْ حَقٍّ وَإِنَّكَ لَتَعْلَمُ مَا نُرِيدُ ﴿٧٩﴾
ئۇلار (لۇتقا) ئېيتتى: «قىزلىرىڭغا ھاجىتىمىز يوق ئىكەنلىكىنى سەن ئوبدان بىلىسەن، شۇنىڭدەك بىزنىڭ مەقسىتىمىزنىمۇ ئوبدان بىلىسەن»
قَالَ لَوْ أَنَّ لِي بِكُمْ قُوَّةً أَوْ آوِي إِلَىٰ رُكْنٍ شَدِيدٍ ﴿٨٠﴾
لۇت ئېيتتى: «كاشكى مېنىڭ (سىلەرگە قارشى تۇرىدىغان) كۈچ قۇۋۋىتىم بولسا ئىدى، ياكى ئىلتىجا قىلىدىغان كۈچلۈك جەمەتىم بولسا ئىدى (چوقۇم سىلەرگە زەربە بەرگەن بولاتتىم)»
قَالُوا يَا لُوطُ إِنَّا رُسُلُ رَبِّكَ لَن يَصِلُوا إِلَيْكَ ۖ فَأَسْرِ بِأَهْلِكَ بِقِطْعٍ مِّنَ اللَّيْلِ وَلَا يَلْتَفِتْ مِنكُمْ أَحَدٌ إِلَّا امْرَأَتَكَ ۖ إِنَّهُ مُصِيبُهَا مَا أَصَابَهُمْ ۚ إِ��َّ مَوْعِدَهُمُ الصُّبْحُ ۚ أَلَيْسَ الصُّبْحُ بِقَرِيبٍ ﴿٨١﴾
ئۇلار (يەنى خۇدانىڭ ئەلچى پەرىشتىلىرى): «ئى لۇت! شۈبھىسىزكى، بىز پەرۋەردىگارىڭنىڭ ئەلچىلىرىدۇرمىز، ئۇلار ھەرگىزمۇ ساڭا زىيان يەتكۈزەلمەيدۇ، ئائىلەڭدىكىلەرنى ئېلىپ كېچىنىڭ ئاخىرىدا (ئۇلارنىڭ ئىچىدىن) چىقىپ كەتكىن، ئايالىڭدىن باشقا، سىلەردىن ھېچ كىشى ئارقىسىغا قارىمىسۇن، ئۇ (يەنى ئايالىڭ) ئۇلار بىلەن بىرگە ھالاك بولغۇچىدۇر، شۈبھىسىزكى، ئۇلارغا ئازاب چۈشىدىغان چاغ تاڭ ۋاقتىدۇر، تاڭ يېقىن ئەمەسمۇ؟» دېدى
فَلَمَّا جَاءَ أَمْرُنَا جَعَلْنَا عَالِيَهَا سَافِلَهَا وَأَمْطَرْنَا عَلَيْهَا حِجَارَةً مِّن سِجِّيلٍ مَّنضُودٍ ﴿٨٢﴾
بىزنىڭ (جازالاش) ئەمرىمىز چۈشكەندە، ئۇلارنىڭ يۇرتلىرىنى ئاستىن - ئۈستۈن قىلىۋەتتۇق (يەنى كۆمتۈرۈۋەتتۇق)، ئۇ يۇرتلارغا بىز ئۈستى - ئۈستىلەپ ساپال تاشلارنى ياغدۇردۇق
مُّسَوَّمَةً عِندَ رَبِّكَ ۖ وَمَا هِيَ مِنَ الظَّالِمِينَ بِبَعِيدٍ ﴿٨٣﴾
ئۇ ساپال تاشلارغا پەرۋەردىگارىڭنىڭ دەرگاھىدا بەلگە سېلىنغان ئىدى، ئۇ يۇرتلار زالىملار (يەنى قۇرەيش ھەقىقەتنى رەت قىلغۇچىلىرى) دىن يىراق ئەمەس
وَإِلَىٰ مَدْيَنَ أَخَاهُمْ شُعَيْبًا ۚ قَالَ يَا قَوْمِ اعْبُدُوا اللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُ ۖ وَلَا تَنقُصُوا الْمِكْيَالَ وَالْمِيزَانَ ۚ إِنِّي أَرَاكُم بِخَيْرٍ وَإِنِّي أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ مُّحِيطٍ ﴿٨٤﴾
مەديەن (ئەھلى) ئۇلارنىڭ قېرىندىشى شۇئەيبنى (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، ئۇ ئېيتتى: «ئى قەۋمىم! اﷲ قا ئىبادەت قىلىڭلار، سىلەرگە اﷲ تىن باشقا ھېچ مەبۇد (بەرھەق) يوقتۇر، ئۆلچەمدە ۋە تارازىدا كەم بەرمەڭلار، مەن سىلەرنى ئىشەنچلىك ھالدا باي ھېسابلايمەن، مەن ئىشەنچلىك ھالدا سىلەرنىڭ ھەممىنى قورشاپ تۇرغۇچى كۈن (يەنى قىيامەت كۈنى) نىڭ ئازابىغا دۇچار بولۇشۇڭلاردىن قورقىمەن
وَيَا قَوْمِ أَوْفُوا الْمِكْيَالَ وَالْمِيزَانَ بِالْقِسْطِ ۖ وَلَا تَبْخَسُوا النَّاسَ أَشْيَاءَهُمْ وَلَا تَعْثَوْا فِي الْأَرْضِ مُفْسِدِينَ ﴿٨٥﴾
ئى قەۋمىم! سىلەر ئۆلچەمنى توغرا ئۆلچەڭلار، تارازىنى توغرا تارتىڭلار، كىشىلەرگە نەرسىلەرنى كەم بەرمەڭلار، يەر يۈزىدە بۇزغۇنچىلىق قىلىپ پىتنە - پاسات تېرىماڭلار
بَقِيَّتُ اللَّهِ خَيْرٌ لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ ۚ وَمَا أَنَا عَلَيْكُم بِحَفِيظٍ ﴿٨٦﴾
اﷲ سىلەرگە قالدۇرغان ھالال رىزىق (سىلەر يىغقان ھارامدىن) ياخشىدۇر، ئەگەر سىلەر مۆمىن بولساڭلار. مەن سىلەرگە مۇھاپىزەتچى ئەمەسمەن»
قَالُوا يَا شُعَيْبُ أَصَلَاتُكَ تَأْمُرُكَ أَن نَّتْرُكَ مَا يَعْبُدُ آبَاؤُنَا أَوْ أَن نَّفْعَلَ فِي أَمْوَالِنَا مَا نَشَاءُ ۖ إِنَّكَ لَأَنتَ الْحَلِيمُ الرَّشِيدُ ﴿٨٧﴾
ئۇلار: «ئى شۇئەيب! سېنىڭ نامىزىڭ (يەنى دىنىڭ) سېنى ئاتا - بوۋىلىرىمىز ئىبادەت قىلىپ كېلىۋاتقان نەرسىلەرنى (يەنى بۇتلارنى) بىزنى تاشلاتقۇزۇشقا ياكى ماللىرىمىزنى خالىغىنىڭچە تەسەررۇپ قىلىشىمىزنى تەرك ئەتكۈزۈشكە بۇيرۇمدۇ؟ سەن تازىمۇ كۆڭلى - كۆكسى كەڭ، تازىمۇ توغرا يول تاپقان ئادەمسەن - دە» دېدى
قَالَ يَا قَوْمِ أَرَأَيْتُمْ إِن كُنتُ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّي وَرَزَقَنِي مِنْهُ رِزْقًا حَسَنًا ۚ وَمَا أُرِيدُ أَنْ أُخَالِفَكُمْ إِلَىٰ مَا أَنْهَاكُمْ عَنْهُ ۚ إِنْ أُرِيدُ إِلَّا الْإِصْلَاحَ مَا اسْتَطَعْتُ ۚ وَمَا تَوْفِيقِي إِلَّا بِاللَّهِ ۚ عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ أُنِيبُ ﴿٨٨﴾
شۇئەيب ئېيتتى: «ئى قەۋمىم! ئېيتىپ بېقىڭلارچۇ، پەرۋەردىگارىم ماڭا روشەن پاكىت (يەنى خۇدانىڭ ئەلچىسىلىك) ۋە كەڭتاشا ھالال رىزىق ئاتا قىلغان تۇرسا، (مەن شۇنداق نېمەتلەرنى ئاتا قىلغان پەرۋەردىگارىمنىڭ ئەمرىگە خىلاپلىق قىلامدىم؟) سىلەرگە قىلىش مەنئى قىلىنغان نەرسىنى ئۆزۈممۇ قىلىشنى خالىمايمەن، مەن پەقەت (سىلەرنى) قولۇمدىن كېلىشىچە تۈزەشنى خالايمەن، مەن پەقەت اﷲ نىڭ ياردىمى بولغاندىلا مۇۋەپپەقىيەت قازىنالايمەن، (ھەممە ئىشتا) اﷲ قا تايىنىمەن ۋە ئۇنىڭغا يۈزلىنىمەن
وَيَا قَوْمِ لَا يَجْرِمَنَّكُمْ شِقَاقِي أَن يُصِيبَكُم مِّثْلُ مَا أَصَابَ قَوْمَ نُوحٍ أَوْ قَوْمَ هُودٍ أَوْ قَوْمَ صَالِحٍ ۚ وَمَا قَوْمُ لُوطٍ مِّنكُم بِبَعِيدٍ ﴿٨٩﴾
ئى قەۋمىم! سىلەرنىڭ ماڭا قارشى تۇرغانلىقىڭلار سىلەرنى نۇھنىڭ قەۋمى ياكى ھۇدنىڭ قەۋمى ۋە ياكى سالىھنىڭ قەۋمى دۇچار بولغان ئازابقا دۇچار قىلمىسۇن. لۇت قەۋمى (نىڭ دىيارى) سىلەر (نىڭ جايىڭلار) دىن يىراق ئەمەس
وَاسْتَغْفِرُوا رَبَّكُمْ ثُمَّ تُوبُوا إِلَيْهِ ۚ إِنَّ رَبِّي رَحِيمٌ وَدُودٌ ﴿٩٠﴾
پەرۋەردىگارىڭلاردىن مەغپىرەت تەلەپ قىلىڭلار، ئاندىن ئۇنىڭغا تەۋبە قىلىڭلار، مېنىڭ پەرۋەردىگارىم (تەۋبە قىلغۇچىلارغا) ئىشەنچلىك ھالدا مەرھەمەتلىكتۇر، (ئۇلارنى) دوست تۇتقۇچىدۇر»
قَالُوا يَا شُعَيْبُ مَا نَفْقَهُ كَثِيرًا مِّمَّا تَقُولُ وَإِنَّا لَنَرَاكَ فِينَا ضَعِيفًا ۖ وَلَوْلَا رَهْطُكَ لَرَجَمْنَاكَ ۖ وَمَا أَنتَ عَلَيْنَا بِعَزِيزٍ ﴿٩١﴾
ئۇلار: «ئى شۇئەيب! بىز سېنىڭ ئېيتقانلىرىڭنىڭ نۇرغۇنىنى چۈشەنمەيمىز. بىز ئەلۋەتتە سېنى ئارىمىزدا كۈچسىز ئادەم دەپ قارايمىز، سېنىڭ قەۋم - قېرىنداشلىرىڭ بولمىغاندا ئىدى، بىز سېنى چوقۇم تاش - كېسەك قىلىپ ئۆلتۈرەتتۇق، سەن بىزگە ئەتىۋارلىق ئەمەسسەن» دېدى
قَالَ يَا قَوْمِ أَرَهْطِي أَعَزُّ عَلَيْكُم مِّنَ اللَّهِ وَاتَّخَذْتُمُوهُ وَرَاءَكُمْ ظِهْرِيًّا ۖ إِنَّ رَبِّي بِمَا تَعْمَلُونَ مُحِيطٌ ﴿٩٢﴾
شۇئەيب ئېيتتى: «ئى قەۋمىم! مېنىڭ قەۋم قېرىنداشلىرىم سىلەرگە اﷲ تىنمۇ ئەتىۋارلىقمۇ؟ سىلەر اﷲ نى ئارقاڭلارغا چۆرىۋەتتىڭلار (يەنى اﷲ قا ئىتائەت قىلمىدىڭلار ۋە اﷲ نى ئۇلۇغلىمىدىڭلار)، پەرۋەردىگارىم سىلەرنىڭ قىلمىشلىرىڭلاردىن ئەلۋەتتە خەۋەرداردۇر
وَيَا قَوْمِ اعْمَلُوا عَلَىٰ مَكَانَتِكُمْ إِنِّي عَامِلٌ ۖ سَوْفَ تَعْلَمُونَ مَن يَأْتِيهِ عَذَابٌ يُخْزِيهِ وَمَنْ هُوَ كَاذِبٌ ۖ وَارْتَقِبُوا إِنِّي مَعَكُمْ رَقِيبٌ ﴿٩٣﴾
ئى قەۋمىم! سىلەر ئۆز يولۇڭلار بويىچە ھەرىكەت قىلىڭلار، مەنمۇ ئەلۋەتتە ئۆز يولۇم بويىچە ھەرىكەت قىلاي! ئۇزاققا قالماي كىمنىڭ رەسۋا قىلغۇچى ئازابقا دۇچار بولىدىغانلىقى ۋە كىمنىڭ يالغانچى ئىكەنلىكىنى بىلىسىلەر، (ئىشىڭلارنىڭ ئاقىۋىتىنى) كۈتۈڭلار! شۈبھىسىزكى. مەنمۇ سىلەر بىلەن بىللە كۈتىمەن»
وَلَمَّا جَاءَ أَمْرُنَا نَجَّيْنَا شُعَيْبًا وَالَّذِينَ آمَنُوا مَعَهُ بِرَحْمَةٍ مِّنَّا وَأَخَذَتِ الَّذِينَ ظَلَمُوا الصَّيْحَةُ فَأَصْبَحُوا فِي دِيَارِهِمْ جَاثِمِينَ ﴿٩٤﴾
بىزنىڭ (ئۇلارنى ھالاك قىلىش) پەرمانىمىز چۈشكەن چاغدا، شۇئەيبنى ۋە ئۇنىڭ بىلەن بىللە ئىمان ئېيتقانلارنى مەرھەمەت قىلىپ قۇتۇلدۇردۇق، (ئۆزلىرىگە) زۇلۇم قىلغانلارغا قاتتىق ئاۋاز (دىن پەيدا بولغان زىلزىلە) يۈزلەندى، ئۇلار ئۆز ئۆيلىرىدە ئولتۇرغان پېتى قېتىپ قالدى
كَأَن لَّمْ يَغْنَوْا فِيهَا ۗ أَلَا بُعْدًا لِّمَدْيَنَ كَمَا بَعِدَتْ ثَمُودُ ﴿٩٥﴾
گويا ئۇلار ئۆيلىرىدە تۇرمىغاندەك (يەنى ياشىمىغاندەك) بولۇپ قالدى. مەديەن ئاھالىسى خۇددى سەمۇد قەۋمى ھالاك بولغاندەك ھالاك بولسۇن!
وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِآيَاتِنَا وَسُلْطَانٍ مُّبِينٍ ﴿٩٦﴾
بىز مۇسانى ئىشەنچلىك ھالدا ئايەتلىرىمىز (يەنى اﷲ تىن نازىل بولغان دىنىي ئەھكاملار) ۋە روشەن دەلىللەر (يەنى مۆجىزىلەر) بىلەن پىرئەۋنگە ۋە ئۇنىڭ قەۋمىنىڭ چوڭلىرىغا (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، ئۇلار پىرئەۋننىڭ بۇيرۇقىغا بويسۇندى، پىرئەۋننىڭ بۇيرۇقى توغرا ئەمەس ئىدى
إِلَىٰ فِرْعَوْنَ وَمَلَئِهِ فَاتَّبَعُوا أَمْرَ فِرْعَوْنَ ۖ وَمَا أَمْرُ فِرْعَوْنَ بِرَشِيدٍ ﴿٩٧﴾
بىز مۇسانى ئىشەنچلىك ھالدا ئايەتلىرىمىز (يەنى اﷲ تىن نازىل بولغان دىنىي ئەھكاملار) ۋە روشەن دەلىللەر (يەنى مۆجىزىلەر) بىلەن پىرئەۋنگە ۋە ئۇنىڭ قەۋمىنىڭ چوڭلىرىغا (خۇدانىڭ ئەلچىسى قىلىپ) ئەۋەتتۇق، ئۇلار پىرئەۋننىڭ بۇيرۇقىغا بويسۇندى، پىرئەۋننىڭ بۇيرۇقى توغرا ئەمەس ئىدى
يَقْدُمُ قَوْمَهُ يَوْمَ الْقِيَامَةِ فَأَوْرَدَهُمُ النَّارَ ۖ وَبِئْسَ الْوِرْدُ الْمَوْرُودُ ﴿٩٨﴾
پىرئەۋن قىيامەت كۈنى باشلامچىلىق بىلەن ئۆز قەۋمىنى دوزاخقا باشلاپ كىرىدۇ، ئۇ جاي نېمىدېگەن يامان!
وَأُتْبِعُوا فِي هَٰذِهِ لَعْنَةً وَيَوْمَ الْقِيَامَةِ ۚ بِئْسَ الرِّفْدُ الْمَرْفُودُ ﴿٩٩﴾
ئۇلار بۇ دۇنيادىمۇ، قىيامەت كۈنىدىمۇ لەنەتكە ئۇچرايدۇ، ئۇلارغا بېرىلگەن (ئىككىلا دۇنيادا لەنەتكە ئۇچراشتىن ئىبارەت) بۇ سوغا نېمىدېگەن يامان!
ذَٰلِكَ مِنْ أَنبَاءِ الْقُرَىٰ نَقُصُّهُ عَلَيْكَ ۖ مِنْهَا قَائِمٌ وَحَصِيدٌ ﴿١٠٠﴾
مانا بۇلار (ئاھالىسىنى بىز ھالاك قىلغان) يۇرتلارنىڭ خەۋەرلىرىدۇر، ئۇنى ساڭا (ۋەھىيى ئارقىلىق) بايان قىلىمىز. ئۇ يۇرتلاردىن ئىزى قالغانلىرىمۇ بار (يەنى ئاھالىسى ھالاك قىلىنىپ، يۇرتلىرىنىڭ ئورنى قالدۇرۇلغانلىرىمۇ بار)، ئىزى قالمىغانلىرىمۇ بار (يەنى ئاھالىسى بىلەن قوشۇپ ئۆرۈۋېتىلگەن زىرائەتتەك ۋەيران قىلىنغانلىرىمۇ بار)

سائادەتمەنلەرگە كەلسەك، ئۇلار مەڭگۈلۈك باغلاركە كىرىدۇ، (مەڭگۈلۈك باغلارنىڭ) ئاسمان - زېمىنى يوقالمايلا تۇرسا، ئۇلار مەڭگۈلۈك باغلارتە پەرۋەردىگارىڭ خالىغان زامانغىچە داۋاملىق تۇرىۋېرىدۇ (بۇ ئۇلارغا قىلىنغان) ئۈزۈلۈپ قالمايدىغان ئىنئامدۇر

ئۇلار (يەنى مۇشرىكلار) نىڭ بۇتلارغا چوقۇنۇىشىنىڭ (گۇمراھلىق) ئىكەنلىكىدىن شەكلەنمە، ئۇلار پەقەت ئۆزلىرىنىڭ بىز ساڭا قىسسىسىنى بايان قىلغان ئاتا - بوۋىلىرىغا ئوخشاشلا چوقۇنىدۇ، بىز ئۇلارنىڭ (ئازابتىن) تېگىشلىك نېسىۋىسىنى چوقۇم كېمەيتمەي بېرىمىز

وَإِنَّ كُلًّا لَّمَّا لَيُوَفِّيَنَّهُمْ رَبُّكَ أَعْمَالَهُمْ ۚ إِنَّهُ بِمَا يَعْمَلُونَ خَبِيرٌ ﴿١١١﴾
پەرۋەردىگارىڭ ھەر بىر ئادەمگە ئۇنىڭ قىلغان ئەمەللىرىڭنىڭ مۇكاپاتىنى تولۇق بېرىدۇ، اﷲ، ئەلۋەتتە، ئۇلارنىڭ ئەمەللىرىدىن تولۇق خەۋەرداردۇر
فَاسْتَقِمْ كَمَا أُمِرْتَ وَمَن تَابَ مَعَكَ وَلَا تَطْغَوْا ۚ إِنَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿١١٢﴾
